def _make_sinc_shape(n, cutoff_frac):
    L = _next_pow2(2*n-1)
    K = int(cutoff_frac * L // 2)  # number of positive frequencies (including zero)
    # create spectrum of length L//2+1 (real rfft length)
    spec = np.zeros(L//2+1, dtype=float)

    # IFFT via irfft
    p = np.fft.irfft(spec, n=L)[:n]  # take first n values
    p = np.maximum(p, 0.0)
    if p.sum() > 0:
        p /= p.sum()
    else:
        p = np.full(n, 1.0 / n)
    return p

"""
Hybrid optimizer for the step‑function attack.

1. Build a small pool of diverse probability vectors.
   In addition to the shapes used before we add a *spectral* shape:
   low‑frequency rectangular spectrum → sinc‑like time shape.

2. Run a continuation / smooth‑max Adam optimizer.
   The true objective 2·n·max(conv(p,p)) is approximated by
   2·n·log‑sum‑exp(α·conv(p,p))/α with a slowly increasing α.
   The gradient of the smoothed max is obtained analytically:
       g = 2·n·2·(p * w)[i]   where w = softmax(α·conv)   and * denotes
   correlation.  Correlation is performed by a single FFT‑based convolution.
   The variables are the logits of a soft‑max parametrisation, i.e. p = softmax(z),
   which guarantees non‑negativity and Σp = 1.

3. A short simulated‑annealing polish (mass moves) is applied on the best
   candidate found by the Adam run.

The whole routine respects the given time budget (≤ 1000 s) and returns a
plain list of non‑negative floats (a probability vector).  Scaling of the
vector does not affect the evaluation score because the objective is
homogeneous; we keep the sum = 1 to stay in the simplex.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Exact full linear convolution using FFT (real input)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = b.shape[0]
    L = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    c = np.fft.irfft(A * B, n=L)[: n + m - 1]
    return c


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
#   value = 2·n·max(conv(p,p))
# ----------------------------------------------------------------------
def _objective(p: np.ndarray) -> float:
    n = p.shape[0]
    conv = _conv_full(p, p)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and the associated weight vector w
#   smooth_max = logsumexp(α·conv) / α
#   w_i       = exp(α·conv_i) / Σ exp(α·conv)
# ----------------------------------------------------------------------
def _smooth_max_and_weights(p: np.ndarray, alpha: float):
    n = p.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]

    # stable log‑sum‑exp
    c = alpha * conv
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    sumexp = exp_c.sum()
    smooth_max = (c_max + np.log(sumexp)) / alpha
    w = exp_c / sumexp                     # weights (size 2·n‑1)
    return smooth_max, w


# ----------------------------------------------------------------------
# Gradient of the smoothed max w.r.t. p (vector of length n)
#   g = 2·n·2·Σ_j p_j w_{i+j}
# Using correlation implemented as a convolution with the reversed w.
# ----------------------------------------------------------------------
def _gradient_wrt_p(p: np.ndarray, w: np.ndarray):
    n = p.shape[0]
    w_rev = w[::-1]

    L = _next_pow2(3 * n - 2)                     # length for p (n) + w_rev (2n‑1)
    Ap = np.fft.rfft(p, n=L)
    Aw = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(Ap * Aw, n=L)  # length L, we only need a slice

    # slice conv_rev[n‑1 : 2·n‑1] (size n) and reverse it
    slice_start = n - 1
    slice_end = 2 * n - 1
    g = 2.0 * conv_rev[slice_start:slice_end][::-1]   # factor 2 from ∂conv/∂p
    return g


# ----------------------------------------------------------------------
# Simple simulated‑annealing mass‑move polish (uses the exact max‑objective)
# ----------------------------------------------------------------------
def _simulated_annealing(p0: np.ndarray, deadline: float,
                         start_time: float, rng: np.random.Generator,
                         max_move_frac: float = 0.03):
    p = p0.copy()
    cur_val = _objective(p)
    best_p = p.copy()
    best_val = cur_val
    n = p.shape[0]

    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / (deadline - start_time))
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=p)
        if p[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = p[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        p_prop = p.copy()
        p_prop[src] -= delta
        p_prop[dst] += delta

        new_val = _objective(p_prop)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / T):
            p = p_prop
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_p = p.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} best={best_val:.6f}")

    return best_p, best_val


# ----------------------------------------------------------------------
# Spectral (sinc‑like) shape: keep low frequencies of a rectangular mask,
# inverse FFT and clip to the simplex.
# ----------------------------------------------------------------------
def _make_sinc_shape(n: int, cutoff_frac: float) -> np.ndarray:
    # n → length of the resulting vector, cutoff_frac ∈ (0,1)
    L = _next_pow2(2 * n - 1)                     # FFT length used later
    # number of retained non‑negative frequencies (including DC)
    keep = max(1, int(round(cutoff_frac * (L // 2 + 1))))
    spec = np.zeros(L // 2 + 1, dtype=float)
    spec[:keep] = 1.0
    v = np.fft.irfft(spec, n=L)[:n]              # keep first n samples
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        v /= s
    else:
        v = np.full(n, 1.0 / n)
    return v


# ----------------------------------------------------------------------
# Keep the K best candidates (sorted by score)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed objective with continuation on α
# ----------------------------------------------------------------------
def _adam_optimize(p0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   rng: np.random.Generator,
                   lr: float = 0.5,
                   alpha_start: float = 0.5,
                   alpha_end: float = 2000.0,
                   ramp_iters: int = 50000) -> tuple[np.ndarray, float]:
    n = p0.shape[0]
    eps = 1e-12
    # logits of a soft‑max parametrisation
    z = np.log(p0 + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_p = None
    best_score = np.inf
    it = 0

    while time.time() < deadline:
        it += 1
        # continuation schedule for α
        alpha = alpha_start + (alpha_end - alpha_start) * min(it / ramp_iters, 1.0)

        # soft‑max to obtain a probability vector
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        p = exp_z / exp_z.sum()

        # smooth max and its weight vector
        smooth_max, w = _smooth_max_and_weights(p, alpha)
        cur_score = 2.0 * n * smooth_max
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} α={alpha:.1f} best={best_score:.6f}")

        # gradient wrt p
        g = _gradient_wrt_p(p, w)          # size n, already contains the factor 2
        grad_p = 2.0 * n * g                # chain rule for the outer factor 2·n

        # gradient wrt logits (soft‑max Jacobian)
        dot = np.dot(p, grad_p)
        grad_z = p * (grad_p - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # safety: keep logits in a reasonable range
        if np.any(np.isnan(z)) or np.any(np.isinf(z)):
            z = np.log(p + eps)

    return best_p, best_score

                                                    start_time, rng,
                                                    max_move_frac=0.04)
        print(f"[{time.time() - start_time:.2f}s] After SA best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Normalise and return as plain Python list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0:
        best_vec /= s
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    # expose the best sequence for the next round (optional)
    globals()['height_sequence_1'] = best_vec.tolist()

    return [float(v) for v in best_vec]

def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Power‑ramp with optional flat plateau, symmetrised."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape
