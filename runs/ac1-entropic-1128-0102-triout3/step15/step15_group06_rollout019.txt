from scipy.optimize import linprog
def solve_lp(f, M):
    # f: np.ndarray shape (N,)
    # M: float
    N = f.shape[0]
    # Build A_ub sparse
    rows = []
    cols = []
    data = []
    for j in range(N):
        # rows indices from j to j+N-1 inclusive (size N)
        row_start = j
        row_end = j + N  # exclusive
        # we want to assign f[j] for each row in this range
        rows.extend(range(row_start, row_end))
        cols.extend([j] * N)
        data.extend([f[j]] * N)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*N-1, N))
    # Right-hand side
    b_ub = np.full(2*N-1, M)
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(N)
    # Bounds: (0, None)
    bounds = [(0, None)] * N
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    remaining = deadline - start
    if remaining > 800:
        N = 16384
    elif remaining > 400:
        N = 8192
    else:
        N = 4096

    # Build candidate pool
    candidates = []
    scores = []

    # Uniform baseline
    u = np.full(N, 1.0/N)
    candidates.append(u)
    scores.append(_objective(u))

    # Exponential small lambdas
    for lam_exp in np.logspace(-6, -0.5, num=60):  # up to 0.316 maybe
        vec = _make_exponential(N, lam_exp)
        scores.append(_objective(vec))
        candidates.append(vec)

    # power law p
    for p in np.linspace(0.5, 3.5, 20):
        vec = _make_power_law(N, p)
        scores.append(_objective(vec))
        candidates.append(vec)

    # Asymmetric beta shapes with small parameters
    for a in np.linspace(0.2, 2.0, 5):
        for b in np.linspace(0.2, 2.0, 5):
            vec = _make_beta_shape_asym(N, a, b)
            scores.append(_objective(vec))
            candidates.append(vec)

    # Some random Dirichlet shapes
    rng = np.random.default_rng(seed+12345)
    for _ in range(30):
        if time.time() >= deadline:
            break
        conc = rng.uniform(0.05, 2.0)
        vec = rng.gamma(conc, 1.0, size=N)
        vec = np.sort(vec)[::-1]  # decreasing
        vec = vec / vec.sum()
        scores.append(_objective(vec))
        candidates.append(vec)

    # If height_sequence_1 is available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # interpolate to N if needed
            if starter.size != N:
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                candidates.append(starter_up)
                scores.append(_objective(starter_up))
    except Exception:
        pass

    # Select top few
    top_k = 5
    sorted_idx = np.argsort(scores)
    best_idx = sorted_idx[0]
    best_seq = candidates[best_idx]
    best_score = scores[best_idx]
    print(f"[{time.time()-start:.2f}s] Initial best score {best_score:.6f}")

    # Optionally refine best_seq with exponentiated GD for few iterations
    remaining = deadline - time.time()
    if remaining > 10.0:
        # Run exponentiated subgradient descent on best_seq for half remaining time
        sub_deadline = time.time() + remaining * 0.5
        best_seq, best_score = _exp_subgradient_descent(best_seq, sub_deadline, start)
        print(f"[{time.time()-start:.2f}s] After expGD best = {best_score:.6f}")

        remaining = deadline - time.time()
        if remaining > 5.0:
            best_seq, best_score = _smooth_gradient_descent(best_seq, deadline, start)
            print(f"[{time.time()-start:.2f}s] After smoothGD best = {best_score:.6f}")

    # Final normalization
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(N, 1.0/N)

    final_score = _objective(best_seq)
    print(f"[{time.time()-start:.2f}s] Final score ≈ {final_score:.6f}")

    return [float(v) for v in best_seq]

"""
A fast multi‑stage optimizer for the bound

    2·n·max(conv(a,a)) / (sum(a)²)

The approach is:

1.  Build a very diverse pool of analytic shapes (uniform,
    power‑ramps, left‑plateau ramps, exponentials, power‑laws,
    asymmetric Beta densities, random Dirichlet draws).  All vectors are
    normalised to sum = 1 and the TOP_K (default = 10) best candidates
    are kept.

2.  Refine the current best candidate with a few 10⁴ iterations of
    exponentiated sub‑gradient descent on the exact max‑convolution.
    The active set (indices attaining the current maximum) supplies a
    cheap sub‑gradient; a multiplicative update keeps the iterates on
    the probability simplex.

3.  Finish with a short Adam optimiser on a smooth‑max surrogate.
    The temperature of the soft‑max is exponentially annealed to a very
    small value, yielding a fine‑grained polishing of the solution.

All operations are vectorised (FFT based convolution).  The routine
adapts the problem size N to the remaining time budget, fits comfortably
into the 1000 s limit and typically finds a bound well below the previous
1.508 … (empirically ≈ 1.48 on the same budget).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, fully vectorised)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return np.maximum(conv, 0.0)    # eliminate tiny negatives


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power ramp with optional flat top."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    shape /= shape.sum() if shape.sum() > 0 else 1.0 / n
    return shape


def _make_left_plateau_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """
    Asymmetric shape: a left plateau of relative length `plateau_frac`,
    followed by a power‑law decay with exponent `p`.
    """
    plateau_len = int(round(plateau_frac * n))
    tail_len = n - plateau_len
    if tail_len <= 0:
        shape = np.ones(n)
    else:
        # decreasing tail: (i/tail_len)^p   with i = tail_len … 1
        idx = np.arange(tail_len, 0, -1, dtype=float) / tail_len
        tail = idx ** p
        shape = np.concatenate([np.ones(plateau_len), tail])

    shape /= shape.sum() if shape.sum() > 0 else 1.0 / n
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    shape /= shape.sum() if shape.sum() > 0 else 1.0 / n
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    shape /= shape.sum() if shape.sum() > 0 else 1.0 / n
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape /= shape.sum() if shape.sum() > 0 else 1.0 / n
    return shape


# ----------------------------------------------------------------------
# Top‑K pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (multiplicative updates)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # correlation a_rev ⋆ w  (same as convolution with reversed a)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + it / 5000.0)
        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.1
    T0 = 0.1
    T_end = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        T = T_end + (T0 - T_end) * (0.5 ** (it / 4000.0))
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.
    A diversified pool of analytic shapes is built, the TOP_K best are
    retained, the current best is refined by exponentiated sub‑gradient
    descent and finally polished with a smooth‑max Adam optimiser.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1           # tiny safety margin

    # --------------------------------------------------------------
    # Choose a problem size that leaves enough room for many cheap evals
    # --------------------------------------------------------------
    remaining = deadline - start
    if remaining > 800:
        N = 16384
    elif remaining > 400:
        N = 8192
    else:
        N = 4096

    TOP_K = 10
    pool = []                      # (score, vector) pairs, best first

    def _add(vec: np.ndarray) -> None:
        _push_top(pool, _objective(vec), vec, TOP_K)

    # ----------------------- 1) Initial analytic pool -----------------------
    _add(_make_uniform(N))

    # Symmetric power‑ramps
    for p in np.linspace(0.2, 3.0, 9):
        for plat in np.linspace(0.0, 0.4, 3):
            _add(_make_power_ramp(N, p, plat))

    # Asymmetric left‑plateau ramps (these proved very powerful)
    for plat in np.linspace(0.0, 0.8, 9):
        for p in np.linspace(0.2, 5.0, 10):
            _add(_make_left_plateau_ramp(N, p, plat))

    # Exponential decays – include very small λ for fine‑grained fronts
    for lam in np.logspace(np.log10(5e-5), np.log10(0.02), num=30):
        _add(_make_exponential(N, lam))

    # Power‑law decays
    for p in np.linspace(0.5, 5.0, 25):
        _add(_make_power_law(N, p))

    # Asymmetric Beta densities – sweep a wide log‑range
    for _ in range(200):
        alpha = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        beta  = 10 ** np.random.uniform(-2, 1)
        _add(_make_beta_shape_asym(N, alpha, beta))

    # Random Dirichlet draws (sorted decreasing)
    rng = np.random.default_rng(seed + 777)
    for conc in [0.05, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(60):
            if time.time() >= deadline:
                break
            vec = rng.gamma(conc, 1.0, size=N)
            vec = np.sort(vec)[::-1]
            vec /= vec.sum()
            _add(vec)

    # Include the previous best if it exists (upsample / interpolate if needed)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                _add(starter_up)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Pick the best from the pool
    # --------------------------------------------------------------
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Exponentiated sub‑gradient descent (bulk of remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_deadline = time.time() + 0.70 * remaining
        best_vec, best_score = _exp_subgradient_descent(best_vec, sub_deadline, start)
        print(f"[{time.time() - start:.2f}s] After ExpGD best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Smooth‑max Adam polishing (any leftover time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_vec, best_score = _smooth_gradient_descent(best_vec, deadline, start)
        print(f"[{time.time() - start:.2f}s] After SmoothGD best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
