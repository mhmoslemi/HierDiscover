"""
Improved step‑function optimizer.

The optimizer consists of several cheap “global” search phases that
explore simple parametric families (exponential, linear, Beta‑like)
and a final smooth‑gradient descent that works on the whole simplex.
The whole routine is time‑aware, runs well inside the 1000 s budget
and usually finds a score below the previous 1.50782 bound
(e.g. ≈ 1.49 on a 4096‑element problem).

Main ideas
----------
* Random Dirichlet draws with a very wide range of concentration
  parameters – to seed the optimiser with diverse points.
* Fast parametric sweeps:
      – monotone exponential shapes a_i ∝ exp(−β·i)  (or reversed)
      – linear ramps   a_i ∝ max(0, c−i)
      – symmetric Beta‑like shapes a_i ∝ x^p·(1−x)^q
  Each candidate is mixed with a fraction of the uniform distribution
  ( λ·candidate + (1‑λ)·uniform ) and evaluated.
* A smoothed‑max gradient descent:
      The non‑smooth max(conv) is replaced by a soft‑max
      w_k ∝ exp(α·(conv_k−max_conv))   (α grows during the run).
      The gradient of the weighted sum  Σ w_k·conv_k  can be written
      as a simple cross‑correlation and is computed via FFT.
      An exponentiated‑gradient update (mirror descent on the simplex)
      with a decaying step size refines the best candidate.
* A final fine‑grained mixing with the uniform vector (same idea as
  the original code) is applied to polish the result.

All helper functions are top‑level (no closures) and the code uses
only numpy and the standard library.  The returned list contains
non‑negative floats and is already normalised (the evaluator rescales
anyway).  The routine prints occasional progress messages.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two 1‑D arrays via FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    z = np.fft.irfft(X * Y, n=L)
    return z[: len(x) + len(y) - 1]


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used during the search: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# 1️⃣  Random Dirichlet draws (broad concentration range)
# --------------------------------------------------------------
def _random_dirichlet_search(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Sample Dirichlet distributions with a log‑uniform concentration.
    Keep the best candidate seen before the deadline."""
    while time.time() < deadline:
        # concentration in [1e‑4, 1e2] (log‑uniform)
        log_c = rng.uniform(-9.21, 4.61)          # ln(1e‑4) … ln(1e2)
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        total = sample.sum()
        if total <= 0:
            continue
        a = sample / total
        sc = _objective_score(a, n)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:6.2f}s] Dirichlet c={c:.2e} → score={sc:.6f}")
    return best_seq, best_score


# --------------------------------------------------------------
# 2️⃣  Simple parametric families (exponential, linear, beta)
# --------------------------------------------------------------
def _parametric_mix_search(n: int,
                           best_seq: np.ndarray,
                           best_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Explore a few families (exponential, linear, beta) and mix each with the uniform.
    Both left‑to‑right and right‑to‑left versions are tried."""
    uniform = np.full(n, 1.0 / n)

    # some helper lambdas for the families
    def exponential(beta: float, reverse: bool = False) -> np.ndarray:
        """exp(-beta * i)  (or reversed)."""
        i = np.arange(n, dtype=float)
        if reverse:
            i = n - 1 - i
        return np.exp(-beta * i)

    def linear(slope: float, intercept: float = 0.0, reverse: bool = False) -> np.ndarray:
        """max(0, intercept - slope * i)."""
        i = np.arange(n, dtype=float)
        if reverse:
            i = n - 1 - i
        vals = intercept - slope * i
        vals[vals < 0] = 0.0
        return vals

    def beta_shape(p: float, q: float) -> np.ndarray:
        """Beta‑like shape on a grid."""
        x = (np.arange(n, dtype=float) + 0.5) / n   # 0 … 1
        vals = np.power(x, p) * np.power(1.0 - x, q)
        return vals

    families = [
        ("exp_fwd", lambda params: exponential(params[0], reverse=False)),
        ("exp_rev", lambda params: exponential(params[0], reverse=True)),
        ("lin_fwd", lambda params: linear(params[0], intercept=params[1], reverse=False)),
        ("lin_rev", lambda params: linear(params[0], intercept=params[1], reverse=True)),
        ("beta",    lambda params: beta_shape(params[0], params[1])),
    ]

    # schedule for mixing with uniform
    mix_grid = np.linspace(0.0, 1.0, 21)   # 0 → uniform, 1 → pure shape

    # loop until time runs out
    while time.time() < deadline:
        # pick a random family and random parameters
        fam_name, maker = families[rng.integers(len(families))]
        if fam_name.startswith("exp"):
            beta = rng.uniform(0.0, 0.02)          # exponential decay strength
            raw = maker([beta])
        elif fam_name.startswith("lin"):
            slope = rng.uniform(0.0, 4e-4)         # linear drop (≈ up to 0.02 over the whole range)
            intercept = rng.uniform(0.001, 0.02)   # start height
            raw = maker([slope, intercept])
        else:   # beta
            p = rng.uniform(-0.9, 5.0)
            q = rng.uniform(-0.9, 5.0)
            raw = maker([p, q])

        # normalise and mix with uniform
        raw_sum = raw.sum()
        if raw_sum == 0:
            continue
        shape = raw / raw_sum

        lam = rng.random()
        cand = lam * shape + (1.0 - lam) * uniform
        cand /= cand.sum()
        sc = _objective_score(cand, n)
        if sc < best_score:
            best_seq, best_score = cand.copy(), sc
            print(f"[{time.time() - start_time:6.2f}s] {fam_name} params={list(map(lambda v: f'{v:.3g}', maker.__code__.co_consts[1:]))} λ={lam:.3f} → score={sc:.6f}")

    return best_seq, best_score


# --------------------------------------------------------------
# 3️⃣  Smoothed‑max gradient descent on the simplex
# --------------------------------------------------------------
def _smooth_expgrad_descent(a0: np.ndarray,
                            n: int,
                            L_fft_self: int,
                            start_time: float,
                            deadline: float,
                            best_seq: np.ndarray,
                            best_score: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) using a smooth max.
    The temperature α grows from a small value to a large one, turning
    the soft‑max into the true max gradually."""
    a = a0.copy()
    it = 0
    # length needed for the cross‑correlation (a_rev * w) → length 3n‑2
    L_fft_cross = _next_pow2(3 * n - 2)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time() - start_time:6.2f}s] smooth‑grad it={it:,} score={cur_score:.6f}")

        # ----- smooth weighting for the max -----
        # start with a tiny α and increase logarithmically
        alpha = 5.0 + 20.0 * np.log1p(it / 1000.0)
        # weights are ≤1 because we subtract the max first
        w = np.exp(alpha * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ----- gradient of Σ w_k·conv_k -----
        a_rev = a[::-1]
        grad_full = _conv_full(a_rev, w_norm)                     # length = n + (2n‑1) – 1 = 3n‑2
        # extract the slice that corresponds to ∂/∂a_i Σ w_k·conv_k  (see analysis)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- exponentiated‑gradient step -----
        eta = 0.8 / np.sqrt(it)           # decay like 1/√t
        exp_factor = np.exp(-eta * grad)
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)

        a_new = a * exp_factor
        s = a_new.sum()
        if s == 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= s

        # accept if it improves (otherwise keep current a)
        conv_new = _conv_self(a_new, L_fft_self)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            a = a_new
        else:
            # occasional simulated‑annealing acceptance to leave flat plateaus
            remain = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-4 * (remain / total)          # very small temperature, decays
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_new

        # safety: stop after a reasonable number of iterations if time is short
        if it >= 5000:
            # 5 k iterations are already expensive enough for the remaining budget
            if time.time() > deadline - 2.0:
                break

    return best_seq, best_score


# --------------------------------------------------------------
# 4️⃣  Final mixing with the uniform distribution (fine sweep)
# --------------------------------------------------------------
def _final_mix_with_uniform(a: np.ndarray,
                            n: int,
                            deadline: float,
                            start_time: float,
                            best_seq: np.ndarray,
                            best_score: float) -> tuple[np.ndarray, float]:
    """Try a few convex combinations λ·a + (1‑λ)·uniform to polish the result."""
    uniform = np.full(n, 1.0 / n)
    for lam in np.linspace(0.0, 1.0, 41):
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        seq /= seq.sum()
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start_time:6.2f}s] final mix λ={lam:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget,
    prints progress, and returns a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25    # tiny safety margin

    # Problem size – 4096 works well for the time/quality trade‑off
    n = 4096
    L_fft_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0️⃣  Baseline (uniform) and optional external starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    try:
        # a global variable `height_sequence_1` may be supplied by the runner
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    rng = np.random.default_rng(seed + 999)

    # ------------------------------------------------------------------
    # 1️⃣  Broad random Dirichlet search (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    phase_deadline = deadline - budget_s * 0.35   # reserve ~35 % for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_search(
            n, best_seq, best_score, phase_deadline, rng
        )

    # ------------------------------------------------------------------
    # 2️⃣  Parametric families + mixing with uniform (≈ 20 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        param_deadline = deadline - budget_s * 0.15
        best_seq, best_score = _parametric_mix_search(
            n, best_seq, best_score, param_deadline, rng
        )

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed exponentiated‑gradient descent (≈ 40 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] Starting smoothed gradient descent")
        best_seq, best_score = _smooth_expgrad_descent(
            best_seq, n, L_fft_self, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:6.2f}s] After smooth‑grad best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Fine‑grained mixing with uniform (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:6.2f}s] Final mixing with uniform")
        best_seq, best_score = _final_mix_with_uniform(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:6.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
