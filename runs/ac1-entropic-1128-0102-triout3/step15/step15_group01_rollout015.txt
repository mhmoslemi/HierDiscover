"""
Sidon‑set step‑function optimizer.

The evaluation function
    score = 2·n·max(conv(a,a)) / (sum(a)²)

has the following useful properties:

* It is invariant under scaling of the whole vector.
* For a vector whose non‑zero entries lie on a *Sidon set* (all pairwise
  differences are distinct) the convolution peak equals the square of a
  single entry, because each shift contains at most one product term.
* With m equally‑weighted non‑zero entries (sum = 1) the score becomes
      2·n / m².
  Hence to minimise the score we need a Sidon set as large as possible.
  Theoretical limit:  m ≤ (√(8·n+1)–1)/2 ≈ √(2n) (≈90 for n=4096).

The program repeatedly builds Sidon sets by a fast greedy algorithm
using a random order of the indices.  Several attempts are made until the
time budget (≈1000 s) is exhausted, keeping the largest set found.
The final step‑function puts weight 1/m on the selected indices and 0
elsewhere, normalises it (the sum is already 1) and returns it as a
plain Python list.

Output: a list of 4096 non‑negative floats.  The score is guaranteed to
be ≤ 2·n / m_best²; with m_best ≥ 75 we already beat the target
1.503 (for n=4096 this gives ≤1.46).  The routine prints progress and
the final score.

No external files, network access or hidden state are used.
"""

# Helper: greedy construction of a Sidon set given an order of indices
# ----------------------------------------------------------------------
def _greedy_sidon(order, n):
    """
    Build a Sidon set by scanning `order` (an array of candidate indices).
    Returns the list of selected indices.
    """
    selected = []                     # chosen indices
    # used[d] == True  ⇔  difference d already appears (d > 0)
    used = np.zeros(n, dtype=np.bool_)
    # we never need the difference 0, so it stays False
    for i in order:
        if not selected:
            selected.append(i)
            continue
        # compute absolute differences to already selected indices
        diffs = np.abs(np.subtract(i, np.array(selected, dtype=int)))
        # if any of those differences is already used we cannot add i
        if used[diffs].any():
            continue
        # otherwise add i and mark the new differences as used
        selected.append(i)
        used[diffs] = True
    # second pass: try to add any remaining index that now fits
    # (this can increase the size a little after the first greedy pass)
    present = np.zeros(n, dtype=np.bool_)
    present[selected] = True
    for i in range(n):
        if present[i]:
            continue
        diffs = np.abs(np.subtract(i, np.array(selected, dtype=int)))
        if not used[diffs].any():
            selected.append(i)
            used[diffs] = True
            present[i] = True
    return selected


# ----------------------------------------------------------------------
# Search for the largest Sidon set within the time budget
# ----------------------------------------------------------------------
def _search_best_sidon(n, time_limit, rng):
    """
    Perform many random greedy attempts until `time_limit` seconds have
    elapsed.  Returns the largest Sidon set found.
    """
    best_set = []
    # theoretical upper bound for the size of a Sidon set in [0,n-1]
    max_possible = int((np.sqrt(8 * n + 1) - 1) // 2)

    attempt = 0
    while time.time() < time_limit:
        attempt += 1
        # random permutation of the indices 0..n-1
        order = rng.permutation(n)
        cand = _greedy_sidon(order, n)
        if len(cand) > len(best_set):
            best_set = cand
            print(
                f"[{time.time() - start_time:.2f}s] "
                f"Attempt {attempt:,} → sidon size {len(best_set)} (theoretical {max_possible})"
            )
            # if we have reached the theoretical bound we can stop early
            if len(best_set) >= max_possible:
                break
    return best_set


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Build a step‑function whose support is a large Sidon set.
    The returned list has length 4096, non‑negative entries, and sum≈1.
    """
    global start_time
    start_time = time.time()
    # leave a tiny safety margin for the runner
    deadline = start_time + budget_s - 0.25

    rng = np.random.default_rng(seed)

    # problem size – the evaluator uses the length of the list as n
    n = 4096

    # ------------------------------------------------------------------
    # 0️⃣  Prepare a trivial uniform baseline (for comparison)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    baseline_score = evaluate_sequence(uniform.tolist())
    print(f"[0.00s] Uniform baseline score = {baseline_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Search for the largest Sidon set possible within the time budget
    # ------------------------------------------------------------------
    # allocate most of the time to the Sidon search; the remaining
    # seconds will be used only for final packaging.
    search_deadline = deadline - 5.0   # keep a few seconds for final steps
    best_sidon = _search_best_sidon(n, search_deadline, rng)

    m = len(best_sidon)
    if m == 0:
        # fallback – return the uniform vector if something went wrong
        best_seq = uniform
        final_score = baseline_score
    else:
        # construct the step function: equal mass on each chosen index
        best_seq = np.zeros(n, dtype=float)
        best_seq[best_sidon] = 1.0 / m   # sum is exactly 1

        # sanity check – compute the true score using the provided evaluator
        final_score = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] Sidon size = {m} → score = {final_score:.6f}"
        )
        # theoretical score (ignoring rounding) for reference
        theory = 2.0 * n / (m * m)
        print(f"    (theoretical score = {theory:.6f})")

    # ------------------------------------------------------------------
    # 2️⃣  Final normalisation (the vector already sums to 1, but we
    #     apply the same safeguards as the evaluator does)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = uniform

    # ------------------------------------------------------------------
    # Done – return as plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    return [float(v) for v in best_seq]
