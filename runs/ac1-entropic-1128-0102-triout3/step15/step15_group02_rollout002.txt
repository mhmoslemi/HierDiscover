"""
Optimisation of a non‑negative step function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm works for any length `n`.  It performs

1.   A quick parametric sweep (power, beta, stretched‑exp) on the selected
     length.
2.   Massive random sampling: Dirichlet draws + mixtures of two beta
     distributions (the latter dramatically diversifies the shapes).
3.   Sub‑gradient descent on the exact `max`‑convolution (projected onto the
     probability simplex) – a classic mirror‑descent for a non‑smooth convex
     objective.
4.   An aggressive hill‑climb / simulated‑annealing phase that moves mass from
     indices with large sub‑gradient to indices with small sub‑gradient.
     The temperature slowly drops, allowing occasional uphill steps to escape
     plateaux.

The whole pipeline respects a 1000 s wall‑clock budget (a 0.3 s safety margin
is kept).  The best sequence found across several candidate lengths
(1024, 2048, 4096) is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Linear convolution a*a (full length) using a pre‑computed FFT size."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(seq: np.ndarray) -> float:
    """Score used during the search: 2·n·max(conv(seq,seq)) (seq sums to 1)."""
    n = seq.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    conv = _conv_self(seq, L_fft)
    return float(2.0 * n * conv.max())


def _beta_pdf(alpha: float, beta: float, n: int) -> np.ndarray:
    """Beta pdf evaluated at the mid‑points of a length‑`n` grid (unnormalised)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    x = idx / n
    # use log to stay stable for extreme alphas/betas
    log_val = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # clip to avoid -inf when x≈0/1 and exponent <0
    log_val = np.where(np.isfinite(log_val), log_val, -700.0)
    return np.exp(log_val)


# ----------------------------------------------------------------------
# Search primitives
# ----------------------------------------------------------------------
def _sweep_families(n: int, best_seq: np.ndarray, best_score: float,
                    deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """A light sweep over a few classical families (power, beta, stretched‑exp)."""
    idx = np.arange(n, dtype=np.float64)

    # uniform baseline (already examined but kept for safety)
    uniform = np.full(n, 1.0 / n)
    sc = _objective_score(uniform)
    if sc < best_score:
        best_score, best_seq = sc, uniform.copy()
        print(f"[{time.time() - start_time:.2f}s] Uniform score={sc:.6f}")

    # ---- power ---------------------------------------------------------
    p_grid = np.linspace(-3.0, 3.0, 121)                 # extended range
    for p in p_grid:
        if time.time() >= deadline:
            break
        pow_seq = np.power(idx + 1.0, p)
        s = pow_seq.sum()
        if s == 0:
            continue
        pow_seq /= s
        sc = _objective_score(pow_seq)
        if sc < best_score:
            best_score, best_seq = sc, pow_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")

    # ---- beta -----------------------------------------------------------
    alphas = np.geomspace(0.01, 8.0, 30)               # log‑spaced, includes very sharp U‑shapes
    betas = np.geomspace(0.01, 8.0, 30)
    for a in alphas:
        log_a = (a - 1.0) * np.log(idx + 1.0)
        for b in betas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * np.log(np.maximum(n - idx, 1.0))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta α={a:.3f} β={b:.3f} score={sc:.6f}")

    # ---- stretched exponential -----------------------------------------
    scales = np.linspace(0.2, 1.0, 9)
    p_vals = np.linspace(0.2, 5.0, 35)
    for scale in scales:
        for p in p_vals:
            if time.time() >= deadline:
                break
            factor = (idx / (scale * n)) ** p
            seq = np.exp(-np.clip(factor, 0, 700))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Stretched p={p:.2f} scale={scale:.2f} score={sc:.6f}")

    return best_seq, best_score


def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Draw many Dirichlet vectors with a wide range of concentration parameters."""
    trials = 6000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.005), np.log(50.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


def _mixture_beta_trials(n: int, best_seq: np.ndarray,
                         best_score: float, deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Two‑component beta mixtures give a very expressive family."""
    trials = 5000
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # sample parameters from a log‑uniform distribution
        a1 = np.exp(rng.uniform(np.log(0.01), np.log(10.0)))
        b1 = np.exp(rng.uniform(np.log(0.01), np.log(10.0)))
        a2 = np.exp(rng.uniform(np.log(0.01), np.log(10.0)))
        b2 = np.exp(rng.uniform(np.log(0.01), np.log(10.0)))
        w = rng.random()                         # weight of the first component
        pdf1 = _beta_pdf(a1, b1, n)
        pdf2 = _beta_pdf(a2, b2, n)
        seq = w * pdf1 + (1.0 - w) * pdf2
        s = seq.sum()
        if s == 0 or not np.isfinite(s):
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] MixBeta w={w:.3f} a1={a1:.2f} b1={b1:.2f} a2={a2:.2f} b2={b2:.2f} score={sc:.6f}")
    return best_seq, best_score


def _subgradient_descent(start_seq: np.ndarray,
                         best_seq: np.ndarray, best_score: float,
                         deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact L∞‑norm of the self‑convolution.
    The step size decays as η / sqrt(t) (standard for sub‑gradient methods).
    """
    a = start_seq.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    eta0 = 0.8        # initial step size (quite aggressive)
    t = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        t += 1
        eta = eta0 / math.sqrt(t)

        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        a_next = a - eta * grad
        a_next = _project_simplex(a_next)

        new_score = _objective_score(a_next)

        if new_score < best_score:
            best_score, best_seq = new_score, a_next.copy()
            print(f"[{time.time() - start_time:.2f}s] SubGrad it={t:,} score={new_score:.6f}")

        if new_score < cur_score - 1e-14:
            a = a_next
            cur_score = new_score
        else:
            # occasional random perturbation to shake out of plateaus
            if rng.random() < 0.001:
                a = a_next
                cur_score = new_score

    return best_seq, best_score


def _hill_climb(seq: np.ndarray,
                best_seq: np.ndarray, best_score: float,
                deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climb.
    Moves a small amount of mass from a high‑gradient index to a low‑gradient one.
    Temperature T decays exponentially; uphill moves are occasionally accepted.
    """
    a = seq.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    T0 = 0.02
    # decay factor to reach ~1e‑7 at the end of the allotted time
    total_time = max(deadline - time.time(), 1e-9)
    decay = (1e-7 / T0) ** (1.0 / total_time)

    cur_score = _objective_score(a)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # indices with non‑zero mass are candidates for source
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            # pick random pair if the deterministic choice collapses
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # transfer amount – adaptively sized but never larger than 10 % of the source bucket
        eps = min(a[src] * rng.uniform(0.001, 0.02), a[src] * 0.1)

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # re‑normalisation to avoid drift
        a_prop = np.maximum(a_prop, 0)
        a_prop_sum = a_prop.sum()
        if a_prop_sum == 0:
            continue
        a_prop /= a_prop_sum

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))

        accept = False
        if Δ < -1e-14:
            accept = True
        else:
            prob = math.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine runs a multi‑stage exploration (parametric sweep, random sampling,
    sub‑gradient descent, hill‑climbing) and returns the best sequence found.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Prepare a list of candidate lengths – smaller lengths give a cheaper
    # objective factor 2·n but also coarser resolution.
    # ------------------------------------------------------------------
    candidate_lengths = [4096, 2048, 1024]

    best_global_seq = None
    best_global_score = np.inf

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # start from a uniform distribution and optionally from a global starter
        # ------------------------------------------------------------------
        uniform = np.full(n, 1.0 / n)
        best_seq = uniform.copy()
        best_score = _objective_score(best_seq)

        # try to incorporate the external starter if it exists
        starter = globals().get("height_sequence_1", None)
        if isinstance(starter, (list, np.ndarray)):
            try:
                s = np.asarray(starter, dtype=float)
                if s.ndim == 1 and s.size > 0:
                    s = np.maximum(s, 0.0)
                    total = s.sum()
                    if total > 1e-12:
                        s = s / total
                        sc = _objective_score(s)
                        if sc < best_score:
                            best_seq, best_score = s.copy(), sc
                            print(f"[{time.time() - start_time:.2f}s] External starter score={sc:.6f}")
            except Exception:
                pass

        # -------------------------- 1️⃣ Parametric sweep ---------------------------
        phase_deadline = deadline - 250.0
        if time.time() < phase_deadline:
            best_seq, best_score = _sweep_families(
                n, best_seq, best_score, phase_deadline, rng
            )

        # -------------------------- 2️⃣ Dirichlet draws ---------------------------
        phase_deadline = deadline - 200.0
        if time.time() < phase_deadline:
            best_seq, best_score = _random_dirichlet_trials(
                n, best_seq, best_score, phase_deadline, rng
            )

        # -------------------------- 3️⃣ Beta‑mixture draws -----------------------
        phase_deadline = deadline - 150.0
        if time.time() < phase_deadline:
            best_seq, best_score = _mixture_beta_trials(
                n, best_seq, best_score, phase_deadline, rng
            )

        # -------------------------- 4️⃣ Sub‑gradient descent --------------------
        phase_deadline = deadline - 80.0
        if time.time() < phase_deadline:
            best_seq, best_score = _subgradient_descent(
                best_seq, best_seq, best_score, phase_deadline, rng
            )

        # -------------------------- 5️⃣ Hill‑climb (simulated annealing) ---------
        if time.time() < deadline:
            best_seq, best_score = _hill_climb(
                best_seq, best_seq, best_score, deadline, rng
            )

        # ------------------------------------------------------------------
        # Record global optimum
        # ------------------------------------------------------------------
        if best_score < best_global_score:
            best_global_score = best_score
            best_global_seq = best_seq.copy()
            print(f"[{time.time() - start_time:.2f}s] New global best (n={n}) score={best_global_score:.6f}")

    # ----------------------------------------------------------------------
    # Normalise the final sequence (just in case) and convert to plain list
    # ----------------------------------------------------------------------
    if best_global_seq is None:
        # fallback – uniform of length 1024
        best_global_seq = np.full(1024, 1.0 / 1024)

    best_global_seq = np.maximum(best_global_seq, 0.0)
    total = best_global_seq.sum()
    if total > 0:
        best_global_seq /= total

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_global_score:.6f}")
    return [float(x) for x in best_global_seq]
