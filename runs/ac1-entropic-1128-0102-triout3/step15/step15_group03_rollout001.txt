"""
Propose a non‑negative step‑function that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1.  Initialise the candidate from a known good global (`height_sequence_1`)
    or from a uniform distribution.
2.  Perform a long “soft‑max’’ gradient descent on the smoothed
    max‑convolution objective.  The loss is
        L_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    whose gradient can be written as a simple convolution.
    We use the Adam optimiser and project the iterate onto the probability
    simplex after every step.  The smoothing parameter β is increased
    gradually (continuation) so that the optimisation first sees a smooth
    landscape and finally concentrates on the true maximum.
3.  Refine the result with a fast hill‑climb that moves tiny amounts of
    mass from indices with a large sub‑gradient of the max‑convolution to
    those with a small sub‑gradient (the same idea as the original code).

The whole search respects a 1000 s wall‑clock budget and prints intermediate
scores for monitoring.  The returned list is ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Directed hill‑climb that moves small mass from an over‑represented
    index to an under‑represented one (using the sub‑gradient of max‑conv)."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)   # up to 0.4 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # random jitter to escape plateaus
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Soft‑max gradient descent on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _softmax_descent(a0: np.ndarray,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Adam‑based gradient descent on L_β(a) = (1/β)·log Σ exp(β·conv)"""

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.25

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    n = a.shape[0]

    it = 0
    # schedule for β (smooth → sharp)
    beta_start = 5.0
    beta_end = 200.0
    smooth_start = time.time()
    smooth_end = deadline - 30.0                     # keep ~30 s for hill‑climb

    while time.time() < smooth_end:
        it += 1

        # current objective (exact max‑conv)
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Softmax it={it} score={cur_score:.6f}")

        # current smoothing parameter β
        if smooth_end > smooth_start:
            frac = (time.time() - smooth_start) / (smooth_end - smooth_start)
            beta = beta_start + (beta_end - beta_start) * min(max(frac, 0.0), 1.0)
        else:
            beta = beta_end

        # soft‑max probabilities over the convolution entries
        max_conv = conv.max()
        # subtract max to keep exponentials ≤ 1
        exp_vals = np.exp(beta * (conv - max_conv))
        p = exp_vals / exp_vals.sum()                     # shape (2n‑1,)

        # gradient of the smoothed loss:
        #   ∂L/∂a_i = 2 * Σ_k p_k * a_{k-i}
        a_rev = a[::-1]
        grad_full = _conv_fft(p, a_rev)                # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]        # slice back to length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # keep a on the simplex (non‑negative, sum=1)
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that minimises the evaluation metric.

    The routine follows the three‑phase strategy described in the module
    docstring.  It returns a plain Python list of floats ready for
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate n keeps each iteration cheap.
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    # start from a known good global if available, else uniform
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)

    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    print(f"[{time.time() - start_time:.2f}s] Baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent (most of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 35.0:   # leave a few seconds for hill‑climb
        best_seq, best_score = _softmax_descent(
            best_seq, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After softmax score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final hill‑climb (tiny mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
