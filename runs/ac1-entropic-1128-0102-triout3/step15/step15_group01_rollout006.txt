"""
Smooth‑gradient exponentiated‑descent optimizer.

We minimise 
    2·n·max(conv(a,a)) / (sum a)²

for a non‑negative vector a (the sum is 1 by normalisation).
The optimisation uses a smooth approximation of the max
  g_β(a)= (1/β)·log∑_k exp(β·conv_k(a))
and the exact gradient of g_β, which is a weighted convolution.
A multiplicative (exponentiated‑gradient) update on the simplex
is applied with a 1/√t step‑size schedule.  The temperature β is
gradually increased so that the smoothed objective converges to the
original max‑convolution.  The routine starts from a few Dirichlet
samples and the optional global starter, runs several restarts,
injects occasional random noise to escape plateaux and finishes
with a fine‑grained mixing with the uniform distribution.
All work respects the 1000 s time budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max of β·conv (returns probabilities that sum to 1)."""
    # max(conv) is used for numerical stability
    max_c = conv.max()
    scaled = beta * (conv - max_c)
    ex = np.exp(scaled)
    return ex / ex.sum()


def _grad_from_weights(a: np.ndarray, w: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution.
    grad_i = 2·∑_k w_k·a_{k-i} (zero outside the support).
    Implemented as a cross‑correlation via FFT.
    """
    n = a.shape[0]
    # reversed a for the correlation
    a_rev = a[::-1]

    # convolution w * a_rev (full length = len(w)+len(a)-1 = 3·n‑2)
    Wf = np.fft.rfft(w, n=L_grad)
    Af = np.fft.rfft(a_rev, n=L_grad)
    conv_full = np.fft.irfft(Wf * Af, n=L_grad)
    # keep only the needed part of the linear convolution
    needed_len = len(w) + n - 1  # = 3·n‑2
    conv_full = conv_full[:needed_len]

    # extract the slice that corresponds to the gradient
    grad = 2.0 * conv_full[n - 1 : 2 * n - 1]
    return grad


def _objective_score(a: np.ndarray, n: int, L_self: int) -> float:
    """Exact objective used for reporting (uses the true max)."""
    conv = _conv_self_fft(a, L_self)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a tiny safety margin
    deadline = start + budget_s - 0.25

    # problem size – 4096 worked well in earlier experiments
    n = 4096
    L_self = _next_pow2(2 * n - 1)          # FFT length for a*a
    L_grad = _next_pow2(3 * n)              # FFT length for w*a_rev convolution

    # ------------------------------------------------------------
    # Helper to compute the exact score (used for reporting and final mix)
    # ------------------------------------------------------------
    def _score(a):
        return _objective_score(a, n, L_self)

    # ------------------------------------------------------------
    # 0️⃣  Initialise (uniform + optional global starter)
    # ------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _score(best_seq)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # optional global starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _score(starter)
                if sc < best_score:
                    best_seq, best_score = starter, sc
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    rng = np.random.default_rng()

    # ------------------------------------------------------------
    # 1️⃣  A few Dirichlet initialisations (cheap, give a head‑start)
    # ------------------------------------------------------------
    for _ in range(30):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        total = sample.sum()
        if total == 0:
            continue
        a = sample / total
        sc = _score(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------
    # 2️⃣  Smoothed‑gradient exponentiated descent (single long run)
    # ------------------------------------------------------------
    # Parameters for the stochastic descent
    eta0 = 0.7                # base learning‑rate
    beta = 1.0                # start of the soft‑max temperature
    max_beta = 200.0
    beta_inc = 0.005          # increase per iteration
    it = 0
    a = best_seq.copy()      # start from the best found so far
    # simple patience counter – used to inject random noise occasionally
    patience = 0
    patience_limit = 20000   # if no improvement for this many iters, we perturb

    while time.time() < deadline:
        it += 1

        # ---- current objective -------------------------------------------------
        conv = _conv_self_fft(a, L_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_seq, best_score = a.copy(), cur_score
            patience = 0
            print(f"[{time.time() - start:.2f}s] it={it:,} new best {cur_score:.6f}")
        else:
            patience += 1

        # ---- smooth max via soft‑max weights ------------------------------------
        w = _softmax_weights(conv, beta)

        # ---- gradient of the smoothed objective ---------------------------------
        grad = _grad_from_weights(a, w, L_grad)

        # ---- multiplicative (exponentiated) update on the simplex -----------------
        eta = eta0 / np.sqrt(it)                # 1/√t schedule
        exp_factor = np.exp(-eta * grad)        # grad ≥ 0 ⇒ exponent ≤ 0
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a_new = a * exp_factor
        s = a_new.sum()
        if s == 0:
            a_new = uniform.copy()
        else:
            a_new = a_new / s

        # ---- accept / reject (tiny simulated‑annealing flavour) -----------------
        conv_new = _conv_self_fft(a_new, L_self)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            a = a_new
        else:
            # decreasing temperature as we near the deadline
            T0 = 1e-3
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start)
            T = T0 * (remaining / total)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a_new

        # ---- anneal the soft‑max temperature ------------------------------------
        beta = min(beta + beta_inc, max_beta)

        # ---- occasional random perturbation to escape plateaux ------------------
        if patience > patience_limit:
            # add a small Gaussian jitter, re‑normalise, and reset the patience
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            s = a.sum()
            if s == 0:
                a = uniform.copy()
            else:
                a = a / s
            patience = 0
            print(f"[{time.time() - start:.2f}s] Injected noise at it={it:,}")

    # ------------------------------------------------------------
    # 3️⃣  Final fine‑grained mixing with the uniform distribution
    # ------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    for lam in np.linspace(0.0, 1.0, 41):
        if time.time() >= deadline:
            break
        seq = lam * best_seq + (1.0 - lam) * uniform
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq = seq / seq_sum
        sc = _score(seq)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------
    # 4️⃣  Normalise one final time and return as a plain Python list
    # ------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    tot = best_seq.sum()
    if tot > 0:
        best_seq = best_seq / tot
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
