"""
Simple local search for a low‑peak self‑convolution step function.

The evaluation function is
    score = 2·n·max(conv(a,a)) / (sum(a)^2)
and is homogeneous, so we may normalise the vector to sum 1.
The algorithm starts from the supplied global best (``height_sequence_1``) or
uniform weights, then repeatedly moves a small amount of mass from the index
with the largest sub‑gradient (i.e. the index that contributes most to the
current peak) to the index with the smallest sub‑gradient.  The convolution
is updated incrementally (O(n) per move) and the move is accepted if it does
not raise the peak; otherwise a tiny simulated‑annealing acceptance is used.
Every few hundred successful moves the full convolution is recomputed via
FFT to avoid drift.  After the time budget a short uniform‑mix sweep is
performed.  The best sequence found is returned.

The routine respects the 1000 s budget (parameter ``budget_s``) and prints
progress information.
"""

import time
import numpy as np

# --------------------------------------------------------------
# low‑level helpers
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    # numerical noise may produce tiny negative values
    return np.maximum(conv, 0.0)


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat values within 1e-12 of the max as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # contribution a_{k‑i}
    if active.size:
        grad = 2.0 * grad / active.size    # factor 2 because each term appears twice
    return grad


# --------------------------------------------------------------
# main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.25

    # ------------------------------------------------------------------
    # initialise – try to use the globally supplied best sequence
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            a = np.maximum(starter, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = None
        else:
            a = None
    except Exception:
        a = None

    # fallback to uniform if no usable starter
    if a is None:
        n = 4096
        a = np.full(n, 1.0 / n)
    else:
        n = a.shape[0]

    # pre‑compute FFT size for fast convolution
    L_fft = _next_pow2(2 * n - 1)

    # initial full convolution and score
    conv = _conv_self(a, L_fft)
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv
    best_seq = a.copy()
    print(f"[{time.time() - start_time:.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # local search – move mass from high‑gradient to low‑gradient indices
    # ------------------------------------------------------------------
    it = 0                # total iterations
    accepted = 0          # successful moves
    recompute_every = 500   # recompute convolution fully after this many accepts

    while time.time() < deadline:
        it += 1

        # --- sub‑gradient of the current peak ---------------------------------
        grad = _subgradient_maxconv(a, conv)

        # choose source (largest gradient) and destination (smallest)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # avoid degenerate moves
        if src == dst or a[src] <= 1e-14:
            # pick a random source with non‑zero mass
            nonzero = np.where(a > 1e-14)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # step size decays like 1/√t, capped to keep it reasonable
        step_factor = 0.5 / np.sqrt(it)
        eps = min(step_factor * a[src], a[src])
        if eps < 1e-15:
            continue

        # --- incremental update of the convolution ---------------------------
        # conv_new = conv - 2·eps·shift(src) + 2·eps·shift(dst)
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * eps * a
        conv_new[dst:dst + n] += 2.0 * eps * a

        new_max = conv_new.max()

        # --- accept if non‑increasing; otherwise occasional SA acceptance -----
        if new_max <= max_conv * (1.0 + 1e-12):
            accept = True
        else:
            # linear cooling simulated‑annealing
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 1e-3 * (remaining / total)
            prob = np.exp(-(new_max - max_conv) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            # apply the move
            a[src] -= eps
            a[dst] += eps
            conv = conv_new
            max_conv = new_max
            accepted += 1

            cur_score = 2.0 * n * max_conv
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.2f}s] it={it:,} acc={accepted:,} "
                      f"score={cur_score:.6f}")

            # occasional full recomputation to avoid accumulated numerical error
            if accepted % recompute_every == 0:
                conv = _conv_self(a, L_fft)
                max_conv = conv.max()
                cur_score = 2.0 * n * max_conv
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:.2f}s] recompute it={it:,} "
                          f"score={cur_score:.6f}")

        # safety break in very long loops
        if it >= 2_000_000:
            break

    # ------------------------------------------------------------------
    # final fine‑grained mixing with the uniform distribution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        uniform = np.full(n, 1.0 / n)
        lam_grid = np.linspace(0.0, 1.0, 41)   # coarse sweep
        for lam in lam_grid:
            if time.time() >= deadline:
                break
            seq = lam * best_seq + (1.0 - lam) * uniform
            seq_sum = seq.sum()
            if seq_sum == 0:
                continue
            seq = seq / seq_sum
            conv_test = _conv_self(seq, L_fft)
            score_test = 2.0 * n * conv_test.max()
            if score_test < best_score:
                best_score = score_test
                best_seq = seq.copy()
                print(f"[{time.time() - start_time:.2f}s] mix λ={lam:.3f} → score={score_test:.6f}")

    # ------------------------------------------------------------------
    # final normalisation and return as plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
