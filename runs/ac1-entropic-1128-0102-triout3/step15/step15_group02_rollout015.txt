assistantfinal**Solution Overview**

The objective is  

\[
E(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the denominator scales quadratically with a global factor, we may restrict ourselves to the
probability simplex \(\sum_i a_i=1\).  
For a probability vector the score reduces to  

\[
E(a)=2\,n\;\max_k (a*a)_k .
\]

The problem is highly non‑convex.  The code below combines several complementary
search ideas:

1. **Broad random start** – many Dirichlet samples of different concentration.
2. **Smooth‑max Adam optimisation** – a soft‑max surrogate of the max‑convolution,
   projected onto the simplex.
3. **Exact sub‑gradient mirror descent** (exponentiated‑gradient) on the true
   \(\ell_\infty\) loss.
4. **Linear‑programming refinement** – the classic “LP–step” described in the
   statement.  For a fixed \(a\) the constraints  
   \((a*b)_k\le M\) are linear in \(b\); maximising \(\sum b_i\) yields a new
   direction that is mixed back into the current iterate.
5. **Final simulated‑annealing hill‑climb** – occasional random mass transfers
   accepted with a temperature schedule, able to escape small plateaus.

All phases respect the global time budget (default 1000 s).  
The implementation works with a modest problem size (`n = 256`); the objective
depends mainly on the *shape* of the distribution, not on the absolute length,
so a small \(n\) gives the same constant factor but runs much faster,
leaving plenty of time for extensive refinement.

```python
"""
A multi‑stage optimiser for the step‑function bound

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works on the probability simplex (sum = 1) and consists of
  – many random Dirichlet starts,
  – Adam on a soft‑max surrogate,
  – exact mirror‑descent on the L∞ loss,
  – a linear‑programming “LP‑refinement” step (the classic f←(1‑t)f+t·g update),
  – a short simulated‑annealing hill‑climb.

All phases share the same time budget (default 1000 s).  The final
candidate is returned as a plain Python list of floats.
"""

import time
import numpy as np

# optional linear‑programming import
try:
    import cvxpy as cp
except Exception:                     # pragma: no cover
    cp = None


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(seq: np.ndarray) -> float:
    """Score used during the search (2·n·max(conv(seq,seq))) – sum(seq) is 1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max‑convolution at a.
    The gradient is the average of the gradients of every
    shift that attains the maximum.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Random Dirichlet initialisers (intensive)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 4000                     # more draws than the original version
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # Log‑uniform concentration in [0.005, 30]
        log_c = rng.uniform(np.log(0.005), np.log(30.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → {sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, start_time: float,
                   deadline: float, best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.06
    tau = 30.0
    tau_factor = 1.5
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} score={cur_score:.6f}")

        # smooth max
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of smooth‑max (same derivation as the original script)
        a_rev = a[::-1]
        conv_w = np.fft.irfft(
            np.fft.rfft(w, n=_next_pow2(3 * n - 2)) *
            np.fft.rfft(a_rev, n=_next_pow2(3 * n - 2)),
            n=_next_pow2(3 * n - 2)
        )[:n]
        grad = 2.0 * conv_w

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new
        if it % 2000 == 0:
            tau = min(tau * tau_factor, 1e9)
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Exact mirror‑descent on the L∞ loss
# ----------------------------------------------------------------------
def _eg_maxconv_descent(a0: np.ndarray,
                        start_time: float,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    eta = 0.07      # initial step size (multiplicative)
    it = 0
    cur_score = _objective_score(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        # decaying step size
        eta_t = eta / (1.0 + it * 1e-4)

        # exponentiated‑gradient step
        exp_factor = np.exp(np.clip(-eta_t * grad, -700, 700))
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-15)
        a_new = _project_simplex(a_new)

        new_score = _objective_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] EG it={it:,} score={cur_score:.6f}")
        else:
            # gentle decay when we cannot improve
            eta *= 0.99
            if np.random.rand() < 0.001:
                a = a_new
                cur_score = new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  LP‑refinement step (the classic “f←(1‑t)f+t·g” iteration)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray,
               start_time: float,
               deadline: float,
               best_seq: np.ndarray,
               best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the linear programme

        max   Σ b_i
        s.t.  (a * b)_k ≤ M   ∀k
              b_i ≥ 0

    where M = max(conv(a,a)).  The optimum b is mixed back
    into a (mixing factor t=0.15).  The procedure is a faithful
    implementation of the inspiration described in the statement.
    """
    if cp is None:                         # pragma: no cover
        print("cvxpy not available – skipping LP refinement.")
        return best_seq, best_score

    a = a0.copy()
    n = a.shape[0]
    rows = 2 * n - 1
    t_mix = 0.15

    while time.time() < deadline:
        M = _conv_self(a).max()

        # Build the (rows × n) convolution matrix of a
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            # i runs, j = k‑i
            for i in range(i_min, i_max + 1):
                j = k - i
                A[k, j] = a[i]

        # Linear programme
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1500, verbose=False)
        except Exception:                    # pragma: no cover
            break

        if b.value is None:
            break
        b_vec = np.maximum(b.value, 0.0)
        b_vec = _project_simplex(b_vec)

        # convex mixing
        a = (1.0 - t_mix) * a + t_mix * b_vec
        a = _project_simplex(a)

        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] LP refine → {sc:.6f}")

        # stop early if we are out of time
        if time.time() > deadline - 0.5:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# 5️⃣  Final simulated‑annealing hill‑climb
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    # temperature schedule (starts relatively warm, ends near zero)
    T0 = 0.04
    total_time = max(0.0, deadline - time.time())
    if total_time <= 0.0:
        return best_seq, best_score
    decay = (1e-5 / T0) ** (1.0 / total_time)

    while time.time() < deadline:
        # propose a move: transfer a random fraction of mass from i→j
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] < 1e-12:
            continue
        frac = np.random.rand() * 0.01 + 1e-4
        eps = a[i] * frac
        a_prop = a.copy()
        a_prop[i] -= eps
        a_prop[j] += eps
        a_prop = _project_simplex(a_prop)

        new_score = _objective_score(a_prop)
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))

        if Δ < -1e-12 or np.random.rand() < np.exp(-Δ / max(T, 1e-12)):
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine follows a five‑stage pipeline
    (random starts → Adam → mirror‑descent → LP refinement → SA)
    and returns the best sequence found.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate n keeps the optimisation cheap
    # --------------------------------------------------------------
    n = 256                    # enough resolution, cheap enough for many iterations
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialisation (uniform or optional global starter)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter `height_sequence_1`
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # truncate / pad to length n
            if starter.size > n:
                starter = starter[:n]
            elif starter.size < n:
                starter = np.pad(starter, (0, n - starter.size))
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Random Dirichlet sampling – allocate ~200 s
    # --------------------------------------------------------------
    phase_deadline = deadline - 600.0   # leave ~600 s for the refinements
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Adam optimisation of the smooth‑max surrogate – ~250 s
    # --------------------------------------------------------------
    phase_deadline = deadline - 340.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam optimisation")
        best_seq, best_score = _adam_optimize(
            best_seq, start_time, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After Adam best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Exact mirror‑descent (exponentiated‑gradient) – ~150 s
    # --------------------------------------------------------------
    phase_deadline = deadline - 180.0
    if time.time() < phase_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting EG descent")
        best_seq, best_score = _eg_maxconv_descent(
            best_seq, start_time, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After EG best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Linear‑programming refinement – ~80 s
    # --------------------------------------------------------------
    phase_deadline = deadline - 80.0
    if time.time() < phase_deadline and cp is not None:
        print(f"[{time.time() - start_time:.2f}s] Starting LP refinement")
        best_seq, best_score = _lp_refine(
            best_seq, start_time, phase_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After LP best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing hill‑climb – remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final SA hill‑climb")
        best_seq, best_score = _simulated_annealing(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
