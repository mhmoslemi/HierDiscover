"""
Proposes a non‑negative step‑function that (approximately) minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four stages tuned for the 1000 s budget:

1️⃣  Fast parametric sweep over geometric‑power shapes mixed with the uniform
    distribution (coarse grid).  This yields a good warm‑start.
2️⃣  Local random refinement of the three shape parameters (r, p, λ).
3️⃣  Adam‑style gradient descent on a smoothed max (soft‑max of the
    self‑convolution) with a continuation scheme for the temperature β.
4️⃣  Exact projected sub‑gradient descent on the true L∞‑objective,
    followed by a tiny random hill‑climb.

All vectors are projected onto the probability simplex after each move,
so the sum is always 1 (the objective is scale‑invariant).  The routine
respects the 1000 s time limit and returns a plain Python list of floats.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    # use log‑space to keep numbers stable
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)            # prevent overflow
    shape = np.exp(log_shape)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Mixture shape = λ·uniform + (1‑λ)·shape."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed_sum = mixed.sum()
    if mixed_sum == 0.0:
        mixed = uniform.copy()
    else:
        mixed /= mixed_sum
    return mixed


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights p_k = exp(β·conv_k) / Σ exp(·)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max objective:
        g_i = 2 * Σ_j a_j * p_{i+j}
    Computed by a cross‑correlation via FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]                     # reverse for cross‑correlation
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    conv_full = np.fft.irfft(Af * Pf, n=L_grad)

    # keep the central n entries (indices n-1 … 2n-2)
    start = n - 1
    g = conv_full[start:start + n][::-1]
    return 2.0 * g


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact L∞‑objective.
    Uses the same sub‑gradient formula as in the reference solution.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]

        step = step0 / (it ** 0.6)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            # progress report
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            prob = np.exp(-(cur_score - best_score) * 12.0 /
                          (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    # ---------------------------------------------------------------

    L_grad = _next_pow2(3 * n - 2)

    try:

            s = starter.sum()
            if s == 0.0:

            else:
                starter = starter / s
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_score = cur
                best_seq = starter.copy()
                print(f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}")
    except Exception:
        pass

    # 1️⃣  Fast parametric sweep (≈150 s)
    # ---------------------------------------------------------------
    sweep_deadline = start_time + 150.0

    p_grid = np.linspace(-3.0, 2.0, 31)
    lam_grid = np.linspace(0.0, 1.0, 11)

    best_r = best_p = best_lam = None

    for r in r_grid:
        if time.time() >= sweep_deadline:
            break
        for p in p_grid:
            shape = _make_shape(n, r, p)
            for lam in lam_grid:
                cand = _mix_with_uniform(shape, lam)
                sc = _eval_fast(cand, L_fft)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    best_r, best_p, best_lam = r, p, lam
                    print(f"[{time.time() - start_time:.2f}s] sweep r={r:.5f} "
                          f"p={p:.3f} λ={lam:.2f} score={sc:.6f}")

    # ---------------------------------------------------------------
    # 2️⃣  Local refinement of (r, p, λ) – random hill‑climb (≈30 s)

    sigma_p = 0.3
    sigma_l = 0.05

    while time.time() < local_deadline:
        # propose new parameters
        r_prop = np.clip(best_r * np.exp(sigma_r * rng.standard_normal()), 1e-6, 0.999999)
        p_prop = best_p + sigma_p * rng.standard_normal()
        lam_prop = np.clip(best_lam + sigma_l * rng.standard_normal(), 0.0, 1.0)

        shape = _make_shape(n, r_prop, p_prop)
        cand = _mix_with_uniform(shape, lam_prop)
        sc = _eval_fast(cand, L_fft)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_r, best_p, best_lam = r_prop, p_prop, lam_prop
            print(f"[{time.time() - start_time:.2f}s] local refine r={r_prop:.5f} "
                  f"p={p_prop:.3f} λ={lam_prop:.2f} score={sc:.6f}")

        # slowly decay exploration radii
        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_l *= 0.9995

    # ---------------------------------------------------------------
    # 3️⃣  Adam‑style descent on smoothed max (≈remaining time – ≈200 s)
    # ---------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.18
    t = 0

    # continuation schedule for β

        if time.time() >= deadline - 200.0:

    while time.time() < deadline:

    best_seq = np.maximum(best_seq, 0.0)

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
