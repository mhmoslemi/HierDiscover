Hybrid random‑search / simulated‑annealing optimiser for the step‑function bound

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three stages:

1. **Diverse analytic seed pool** – a large variety of monotone,
   symmetric, and strongly skewed shapes (uniform, exponential,
   power‑law, Beta, Dirichlet samples, plus their reversed versions and
   random permutations).  The TOP_K best seeds are kept.

2. **Simulated annealing** – starting from each seed we perform a
   temperature‑controlled random walk on the probability simplex.
   An elementary move transfers a small amount of mass from one index
   to another while preserving the sum.  Moves that improve the
   objective are always accepted; worsening moves are accepted with a
   probability exp(−Δ / T).  The temperature and step size decay
   exponentially, allowing a broad exploration early on and fine‑grained
   adjustments later.

3. **Adam polishing** – the best vector from the annealing phase is
   refined with a few hundred Adam steps on a smooth‑max surrogate of
   the max‑convolution.  This yields a high‑precision “final” solution.

All heavy work is vectorised (FFT based convolution) and the whole

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return np.maximum(conv, 0.0)   # numerical safety


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:

    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape

    vec = rng.gamma(conc, 1.0, size=n)

    return pool

# Simulated annealing ----------------------------------------------------
# ----------------------------------------------------------------------


def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex with an exponentially decaying temperature.
    Moves transfer a small amount of probability mass from one index to
    another while preserving the total sum.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective(a)
    best = a.copy()
    best_score = cur_score

    rng = np.random.default_rng()

    # temperature schedule (log‑linear)
    T0 = max(cur_score * 0.2, 1e-3)          # initial temperature
    T_end = max(cur_score * 1e-6, 1e-8)       # final temperature
    total_time = deadline - start_time
    # anticipating ~200 000 iterations for the whole annealing phase
    max_iters = int(2e5 * (total_time / 800.0) + 1)

    log_T0, log_Tend = math.log(T0), math.log(T_end)
    for it in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break
        # exponential cooling
        frac = it / max_iters
        T = math.exp(log_T0 + frac * (log_Tend - log_T0))

        # step size tied to temperature (larger early on)
        # we move at most ‘step_frac’ of the donor’s current mass.
        step_frac = max(0.02, T / T0) * 0.5   # shrink to ~1 % later

        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # we always move from the larger entry to the smaller one;
        # this tends to flatten peaks and respects non‑negativity.
        if a[i] < a[j]:
            i, j = j, i
        max_transfer = a[i]                     # we cannot take more than what i holds
        eps = step_frac * max_transfer * rng.random()
        a[i] -= eps
        a[j] += eps

        # evaluate
        new_score = _objective(a)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability exp(-(Δ)/T)
            if rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{now - start_time:.2f}s] Anneal it={it:,} new best={best_score:.6f}")

        else:
            # revert move
            a[i] += eps
            a[j] -= eps

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max Adam polishing ---------------------------------------------
# ----------------------------------------------------------------------


def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.12            # slightly larger base step
    T0 = 0.2                # initial temperature for soft‑max
    T_end = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={it:,} best={cur_score:.6f}")

        # anneal temperature
        T = T_end + (T0 - T_end) * (0.5 ** (it / 4000.0))

        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # correlation gradient (see the original code)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it + 1)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_simplex(a)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a small evaluation score.

    The routine:
      1. builds a rich pool of analytical and random shapes,
      2. runs simulated annealing on the best seeds,
      3. finishes with a short Adam polishing phase,
      4. returns the normalised vector as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1      # safety margin

    # --------------------------------------------------------------
    # Choose a problem size that balances quality and speed.
    # --------------------------------------------------------------
    remaining = deadline - start_time
    if remaining > 800:
        N = 16384
    elif remaining > 400:
        N = 8192
    else:
        N = 4096

    rng = np.random.default_rng(seed + 12345)

    # --------------------------------------------------------------
    # 1) Build the initial diverse pool
    # --------------------------------------------------------------
    TOP_K = 15
    pool = _build_initial_pool(N, rng, top_k=TOP_K)

    # keep the best seed for the annealing phase
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Simulated annealing – spend the bulk of the time here
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        anneal_end = time.time() + 0.70 * remaining
        # run annealing from each of the top pool members in a round‑robin way
        idx = 0
        while time.time() < anneal_end:
            seed_vec = pool[idx % len(pool)][1]
            cand, sc = _simulated_annealing(seed_vec, anneal_end, start_time)
            if sc < best_score:
                best_score, best_vec = sc, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal improvement → {sc:.6f}")
            idx += 1

    # --------------------------------------------------------------
    # 3) Adam polishing for the remaining fraction of the budget
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        cand, sc = _smooth_gradient_descent(best_vec, deadline, start_time)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam final improvement → {sc:.6f}")

    # --------------------------------------------------------------
    # 4) Final normalisation and conversion to list
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0.0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start_time:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
