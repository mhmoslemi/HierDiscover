"""
Search for a non‑negative step‑function that minimises

    2·n·max(conv(a, a)) / (sum(a)^2)

The algorithm combines several stages:

1. Broad parametric sweep (geometric, power, beta families, two‑exponential).
2. Random Dirichlet draws and random mixtures of simple shapes.
3. Projected gradient descent on a *smooth* approximation of the
   max‑convolution (soft‑max).  The smoothness parameter τ is slowly
   increased, turning the problem into the original non‑smooth one.
4. Final hill‑climb that moves tiny amounts of mass from the most
   “over‑represented’’ index to the most “under‑represented’’ one.

All stages respect a 1000 s wall‑clock budget.  The best sequence found
is returned as a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalized to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (all return a copy of the best found)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 250)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.05, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SymBeta a={a:.2f} b={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.1, 5.0, 21)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta a={a:.2f} b={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_exponential(n, best_seq, best_score, deadline):
    """
    U‑shaped candidates: mixture of a decreasing geometric left‑hand
    tail, a mirrored decreasing right‑hand tail and uniform mass.
    """
    idx = np.arange(n, dtype=np.float64)
    r_grid = np.linspace(0.90, 0.995, 30)          # geometric ratio
    for r in r_grid:
        if time.time() >= deadline:
            break
        left = np.power(r, idx)                    # mass concentrated at start
        right = np.power(r, n - 1 - idx)           # mirrored version
        left /= left.sum()
        right /= right.sum()
        uniform = np.full(n, 1.0 / n)

        for w_left in np.linspace(0.0, 0.6, 7):
            if time.time() >= deadline:
                break
            for w_right in np.linspace(0.0, 0.6, 7):
                w_unif = 1.0 - w_left - w_right
                if w_unif < 0.0:
                    continue
                seq = w_left * left + w_right * right + w_unif * uniform
                seq /= seq.sum()
                sc = _objective_score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] TwoExp r={r:.3f} wL={w_left:.2f} wR={w_right:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=250):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)      # concentration between ~0.01 and 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


def _random_mixture_trials(n, best_seq, best_score, deadline, trials=200):
    """Mixtures of left‑geometric, right‑geometric and uniform components."""
    rng = np.random.default_rng()
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    for _ in range(trials):
        if time.time() >= deadline:
            break
        r = rng.uniform(0.85, 0.995)
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        w = rng.random(3)
        w /= w.sum()
        seq = w[0] * left + w[1] * right + w[2] * uniform
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] RandMix r={r:.3f} w={w} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smooth approximation of the max‑convolution + projected GD
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray, deadline, best_seq, best_score):
    """
    Minimise the smooth objective
        f_τ(a) = (1/τ)·log Σ exp( τ·conv_k(a) )
    by projected gradient descent.  τ is slowly increased, turning
    f_τ into the original max‑convolution as the optimisation proceeds.
    """
    a = a0.copy()
    n = a.shape[0]

    # start with a modest smoothness and increase it gradually
    tau = 5.0
    tau_max = 3000.0
    step_factor = 0.2          # base step (will be normalised by grad norm)
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # convolution of the current iterate
        conv = _conv_self(a)
        max_conv = conv.max()

        # soft‑max weights over the convolution entries
        diff = conv - max_conv                     # ≤ 0, max at 0
        w = np.exp(tau * diff)                    # w_i ∝ e^{τ·(conv_i-max)}
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the weighted sum (2·conv(w, a_rev) on the slice)
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # projective step
        grad_norm = np.linalg.norm(grad) + 1e-12
        step = step_factor / grad_norm
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # evaluate the *real* objective (max‑convolution)
        conv_new = _conv_self(a_new)
        score_new = float(2.0 * n * conv_new.max())

        # keep the best ever seen
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth τ={tau:.1f} it={iteration} score={score_new:.6f}")

        # accept the step only if it does not increase the true objective
        if score_new <= 2.0 * n * max_conv:
            a = a_new
            step_factor *= 1.02                     # a little acceleration
        else:
            step_factor *= 0.5                      # be more cautious

        # increase smoothness every few iterations
        if iteration % 30 == 0:
            tau = min(tau * 1.25, tau_max)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # allow a slightly larger transfer than the original 0.4 %
        eps = a[src] * (rng.random() * 0.010 + 1e-6)   # up to 1 % of src
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # random jitter to escape plateaux
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the evaluation function.
    The algorithm follows several stages (parametric sweep, random draws,
    smooth‑max projected gradient descent, final hill‑climb) and respects a
    1000 s wall‑clock budget.  The returned list is ready for
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25      # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – keep n moderate to finish comfortably.
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep (geometric, power, beta families, 2‑exp)
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_exponential(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet + random mixtures
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _random_mixture_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max projected gradient descent (new stage)
    # --------------------------------------------------------------
    smooth_deadline = deadline - 60.0          # keep ~60 s for the final hill‑climb
    if time.time() < smooth_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max gradient descent")
        best_seq, best_score = _smooth_max_descent(best_seq, smooth_deadline,
                                                   best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After smooth descent score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
