"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The method is a three‑phase search:

1️⃣  Fast parametric sweep over several analytic families
    (geometric × power, symmetric power, double‑power) mixed with the
    uniform distribution.  This gives a strong warm‑start.

2️⃣  Smoothed‑max (soft‑max) gradient descent with an Adam‑style
    optimiser and a continuation schedule for the temperature β.
    The optimisation works on a larger problem size (n = 8192) which
    gives a tighter theoretical lower bound.

3️⃣  Projected sub‑gradient descent on the true L∞‑objective, followed
    by a tiny random hill‑climb to polish the final result.

All phases respect a 1000 s budget and the routine finally returns a
plain Python list of floats (normalised to sum = 1).  No external I/O is
performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w ≥ 0, Σ w = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape factories (all return a *probability* vector that sums to 1)
# ----------------------------------------------------------------------


def _make_shape_geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power: a_i ∝ r**i * (i+1)**p."""
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0)
    log_shape -= np.max(log_shape)               # avoid overflow
    shape = np.exp(log_shape)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape


def _make_shape_symmetric(n: int, r: float, p: float) -> np.ndarray:
    """Symmetric power‑geometric shape centred in the middle."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    dist = np.abs(i - centre) + 1.0
    log_shape = np.log(r) * dist + p * np.log(dist)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape


def _make_shape_double_power(n: int, p: float, q: float) -> np.ndarray:
    """Double power: a_i ∝ (i+1)**p * (n-i)**q."""
    i = np.arange(n, dtype=np.float64)
    # add a tiny epsilon to avoid 0**negative
    eps = 1e-12
    left = (i + 1.0) ** p
    right = (n - i) ** q
    shape = left * right + eps
    shape_sum = shape.sum()
    if shape_sum == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape_sum
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Mix a shape with the uniform distribution (lam∈[0,1]). Result already sums to 1."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed_sum = mixed.sum()
    if mixed_sum == 0.0:
        mixed = uniform.copy()
    else:
        mixed /= mixed_sum
    return mixed


# ----------------------------------------------------------------------
# Soft‑max utilities
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights p_k = exp(beta*conv_k) / Σ exp(...)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective:
        g_i = 2 * Σ_j a_j * p_{i+j}
    Computed efficiently via FFT (correlation with reversed p).
    """
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    conv_full = np.fft.irfft(Af * Pf, n=L_grad)
    # the central n entries (starting at index n‑1) correspond to i+j sums
    start = n - 1
    g = conv_full[start:start + n][::-1]
    return 2.0 * g


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same idea as the reference implementation)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true L∞‑objective."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution and its maximal value
        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        cur_score = float(2.0 * n * max_val / (a.sum() ** 2))

        # index where the maximum occurs
        k = int(np.argmax(conv))

        # sub‑gradient for the maximal shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]          # a_j contribution
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]   # symmetric part

        # diminishing step size
        step = step0 / (it ** 0.6)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cand_score = float(2.0 * n *
                           _conv_self_fft(a_candidate, L_fft).max() /
                           (a_candidate.sum() ** 2))

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] sub‑grad it={it:,} "
                  f"step={step:.5e} score={cand_score:.6f}")
        else:
            # occasional uphill acceptance (simulated‑annealing flavour)
            prob = np.exp(-(cand_score - best_score) * 12.0 /
                          (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter every few thousand iterations to escape plateaus
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score

        if src == dst:
            continue
        # transfer up to 0.3 % of the source mass (larger than before)
        eps = best_seq[src] * rng.random() * 3e-3
        a_prop = best_seq.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_score:
            best_seq = a_prop
            best_score = new_sc
            print(f"[{time.time() - start_time:.2f}s] final hill‑climb score={new_sc:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```

This implementation expands the search space to 8192 coefficients, enriches the parametric families, uses an Adam‑style optimiser on the smooth‑max surrogate, and keeps a final sub‑gradient polish plus a tiny stochastic hill‑climb.  All steps respect the 1000 s budget and return a non‑negative sequence that should improve on the previous best bound.
