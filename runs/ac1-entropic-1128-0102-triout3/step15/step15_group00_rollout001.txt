"""
Optimisation via smoothed max‑convolution and Adam on the simplex.

The objective to minimise is

        J(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Because the denominator is homogeneous we may enforce sum(a)=1.
We therefore minimise  max(conv(a,a)).

We approximate the non‑smooth maximum by a soft‑max
    w_k ∝ exp(β·c_k)   (c = a∗a)                (β → ∞ ⇒ true max)

For a given weight vector w the gradient of  Σ_k w_k·c_k  w.r.t. a is obtained
by Correlation(a,w) and a simple diagonal correction.
The variables a are kept on the probability simplex by the soft‑max
parameterisation a = softmax(z) and an Adam optimiser on the free
parameters z.

The algorithm starts from the best known construction (global variable
``height_sequence_1`` if available) and performs many thousands of cheap
Adam steps, slowly increasing β so that the optimisation first moves
the mass in a smooth landscape and finally focuses on the true peak.
The best sequence found during the run is returned.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _correlate_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Correlation of a (length n) with w (length m=2·n‑1).
    Returns the n entries with non‑negative lag:
        corr[i] = Σ_j a[j]·w[i+j]    (0 ≤ i < n)
    """
    n = a.shape[0]
    m = w.shape[0]          # must be 2·n‑1
    L = _next_pow2(n + m - 1)
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(A * np.conj(W), n=L)[: n + m - 1]
    start = m - 1           # lag = 0 sits at this index
    return corr_full[start: start + n]


# ----------------------------------------------------------------------
# Objective helpers (sum is held equal to 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a∈Δ (Δ = simplex)."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine uses a smoothed
    max‑convolution loss together with Adam on the simplex.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1      # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – keep the same size that was used previously
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # Initialise from the global construction if it exists,
    # otherwise start from the uniform distribution.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size:
            starter = np.maximum(starter, 0.0)
            if starter.shape[0] != n:
                # truncate or pad with zeros to length n
                if starter.shape[0] > n:
                    starter = starter[:n]
                else:
                    starter = np.concatenate([starter,
                                              np.zeros(n - starter.shape[0])])
            s = starter.sum()
            if s > 0:
                a_cur = starter / s
            else:
                a_cur = np.full(n, 1.0 / n)
        else:
            a_cur = np.full(n, 1.0 / n)
    except Exception:
        a_cur = np.full(n, 1.0 / n)

    # Parameterise the simplex by logits z (soft‑max)
    eps = 1e-12
    z = np.log(a_cur + eps)          # shape (n,)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.07                       # step‑size (tuned empirically)

    # Smoothing factor for the soft‑max over the convolution entries
    beta_smooth = 0.5               # starts very smooth
    max_beta = 400.0
    beta_mul = 1.25                 # multiply every few thousand steps

    # Best solution seen so far
    best_a = a_cur.copy()
    best_score = _objective_score(best_a)
    print(f"[{time.time() - start_time:.2f}s] start score = {best_score:.6f}")

    it = 0
    no_improve = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- 1. Retrieve the current point on the simplex -----------------
        a = np.exp(z - np.max(z))            # soft‑max, numerically stable
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # ---- 2. Self‑convolution of a ------------------------------------
        conv = _conv_self_fft(a)               # length 2·n‑1

        # ---- 3. Soft‑max weights over the convolution entries -------------
        # w_k ∝ exp(beta·conv_k)  (stable by subtracting the maximum)
        shift = conv.max()
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- 4. Gradient of Σ_k w_k·conv_k w.r.t. a -----------------------
        # correlation part
        corr = _correlate_fft(a, w)               # length n
        diag = w[2 * np.arange(n)] * a            # diagonal correction
        grad_a = 2.0 * corr - diag                # shape (n,)

        # ---- 5. Chain rule to obtain gradient w.r.t. logits z ------------
        # ∂a/∂z = diag(a) - a a^T   ⇒   grad_z = a * (grad_a - <grad_a,a>)
        inner = np.dot(grad_a, a)
        grad_z = a * (grad_a - inner)

        # ---- 6. Adam update on z -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- 7. Track the true (non‑smooth) objective --------------------
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] iter {it:,} "
                  f"β={beta_smooth:.2f} score={cur_score:.6f}")
        else:
            no_improve += 1

        # ---- 8. Anneal the smoothing parameter ----------------------------
        if it % 3000 == 0 and beta_smooth < max_beta:
            beta_smooth = min(max_beta, beta_smooth * beta_mul)

        # ---- 9. Random shake‑up if we are stuck for a long time -----------
        if no_improve > 20000:
            # inject a moderate Gaussian perturbation in log‑space
            z += 0.15 * np.random.randn(n)
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] shake‑up at iter {it:,}")

        # ---- 10. Safety break if we are very close to the deadline ----------
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Final polishing – a tiny deterministic hill‑climb
    # --------------------------------------------------------------
    a = best_a.copy()
    n = a.shape[0]
    # Simple transfer‑mass hill‑climb (few hundred moves)
    for _ in range(2000):
        conv = _conv_self_fft(a)
        # sub‑gradient of the true max (as in the original code)
        max_val = conv.max()
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size
        # choose source/destination
        src = np.argmax(grad)
        dst = np.argmin(grad)
        if src == dst or a[src] < 1e-12:
            break
        # transfer a small fraction
        delta = min(0.02 * a[src], a[src] / 2.0)
        a[src] -= delta
        a[dst] += delta
        # re‑project onto the simplex (Euclidean)
        # using the O(n) algorithm of Duchi et al.
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1.0)
            a = np.maximum(a - theta, 0.0)
        # keep the best seen
        cur_score = float(2.0 * n * _conv_self_fft(a).max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if (_ + 1) % 500 == 0:
                print(f"[{time.time() - start_time:.2f}s] hill‑climb step {_+1} "
                      f"score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Normalise once more (just in case) and return the list
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0:
        best_a = best_a / total
    else:
        best_a = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_a]
