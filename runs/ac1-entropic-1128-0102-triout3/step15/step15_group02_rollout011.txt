def _simulated_annealing(a0, start_time, deadline, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    L_fft = _next_pow2(2*n - 1)
    cur_score = _objective_score(a)
    rng = np.random.default_rng()
    # temperature schedule: start high (maybe 0.02) and decay to near 0.
    T0 = 0.05
    # iterate
    it = 0
    while time.time() < deadline:
        it += 1
        # compute gradient
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        # alternative randomization: with small prob choose random pair
        if rng.random() < 0.1:
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue
        # fraction decreases over time
        frac = max(0.001, 0.2 * (1 - (time.time() - start_time) / (deadline - start_time)))
        delta = a[src] * frac
        if delta <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        # ensure non-negative
        if a_prop[src] < -1e-12:
            continue
        # re-normalize sum=1 (should be unchanged)
        # compute new score
        new_score = _objective_score(a_prop)
        # acceptance
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear cooling temperature
        T = T0 * max(0.01, 1 - elapsed / total)
        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
        else:
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                a = a_prop
                cur_score = new_score
        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SA it={it:,} best score={best_score:.6f}")
        # occasional random perturbation (reset)
        if rng.random() < 0.001:
            # random jitter: add noise and re-normalize
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            if a.sum() == 0:
                a = np.full(n, 1.0/n)
            else:
                a = a / a.sum()
            cur_score = _objective_score(a)
    return best_seq, best_score
```

"""
Improved step‑function search.

The routine searches a sequence a = (a_0,…,a_{n‑1}) (non‑negative,
∑a_i = 1) that minimises

        2·n·max(conv(a,a))

where conv is the linear (non‑reversed) convolution.
The lower bound of the original problem is 1, the uniform distribution
gives 2, and the best known attack is ≈1.508 for n=4096.
We increase the dimension to n=8192 (more degrees of freedom) and use
a multi‑stage heuristic:

  * a rich parametric sweep over several analytic families (geometric,
    power, inverse‑power, stretched‑exponential, logistic, sine‑power,
    linear, etc.);
  * many random Dirichlet draws with a wide concentration range;
  * a simulated‑annealing / greedy local search that moves mass from
    indices with large sub‑gradient to low‑gradient indices;
  * a short final hill‑climb (the same as the reference implementation)
    to polish the solution.

All stages respect a global time budget (default 1000 s) and keep the
best sequence found.  The final sequence is normalised to sum 1 and
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(af * af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """
    Fast surrogate for evaluate_sequence.
    a → non‑negative, not necessarily normalised.
    Returns 2·n·max(conv(a,a)) / (sum(a)²).
    """
    s = a.sum()
    if s <= 0:
        return np.inf
    conv = _conv_self(a, L_fft)
    return float(2.0 * n * conv.max() / (s * s))

    seq = (n - idx).astype(float)
    seq /= seq.sum()
    sc = _objective_score(seq, n, L_fft)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Linear decreasing score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet trials
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)
    while time.time() < deadline:
        # log‑uniform concentration in [1e‑2, 20]
        log_c = rng.uniform(np.log(1e-2), np.log(20.0))
        c = np.exp(log_c)
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a, n, L_fft)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Simulated Annealing / greedy mass‑transfer
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, start_time: float,
                         deadline: float, best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Greedy SA that repeatedly moves a fraction of mass from the index with
    largest sub‑gradient (most responsible for the current max‑convolution)
    to the index with smallest sub‑gradient.
    Temperature and transfer fraction are linearly cooled.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    cur_score = _objective_score(a, n, L_fft)

    rng = np.random.default_rng()
    T0 = 0.05                     # initial temperature
    max_frac_start = 0.15        # start moving up to 15 % of source mass
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # occasional random source/destination to escape flat zones
        if rng.random() < 0.07:
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            if src == dst:
                continue

        # linear cooling of both temperature and step size
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = max(0.001, max_frac_start * (1.0 - elapsed / total))
        T = max(1e-6, T0 * (1.0 - elapsed / total))

        delta = a[src] * frac
        if delta <= 0:
            continue
        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        if a_prop[src] < -1e-12:
            continue
        # (the sum stays equal to 1)

        new_score = _objective_score(a_prop, n, L_fft)

        if new_score < cur_score:
            a = a_prop
            cur_score = new_score
        else:
            # uphill moves with probability exp(-Δ/T)
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                a = a_prop
                cur_score = new_score

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SA it={it:,} best score={best_score:.6f}")

        # tiny random perturbation once in a while
        if rng.random() < 0.001:
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            if a.sum() == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a.sum()
            cur_score = _objective_score(a, n, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb (as in the reference implementation)
# ----------------------------------------------------------------------
def _final_hill_climb(a0: np.ndarray,
                      start_time: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    rng = np.random.default_rng()
    cur_score = float(2.0 * n * _conv_self(a, L_fft).max())
    if cur_score < best_score:
        best_score, best_seq = cur_score, a.copy()

    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    T0 = 0.02
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)

    while time.time() < deadline:
        conv = _conv_self(a, L_fft)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        frac = rng.random() * 0.004 + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue

        new_score = float(2.0 * n * _conv_self(a_prop, L_fft).max())
        Δ = new_score - cur_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = Δ < -1e-12 or (np.exp(-Δ / max(T, 1e-12)) > rng.random())
        if accept:
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb best score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # --------------------------------------------------------------
    # Problem size (larger n gives more freedom)
    # --------------------------------------------------------------
    n = 8192
    L_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialisation (uniform or external starter if available)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n, L_fft)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # optional external starter (height_sequence_1)
    try:
        starter = globals().get('height_sequence_1', None)
        if starter is not None:
            starter = np.asarray(starter, dtype=float).ravel()
            if starter.size > 0:
                starter = np.maximum(starter, 0.0)
                if starter.size < n:
                    # pad with zeros on the right
                    padded = np.zeros(n, dtype=float)
                    padded[: starter.size] = starter
                    starter = padded
                elif starter.size > n:
                    starter = starter[:n]
                s = starter.sum()
                if s > 0:
                    starter = starter / s
                    sc = _objective_score(starter, n, L_fft)
                    if sc < best_score:
                        best_score, best_seq = sc, starter.copy()
                        print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 260.0   # reserve time for later stages
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_families(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 210.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline
        )

    # --------------------------------------------------------------
    # 3️⃣  Simulated annealing / greedy mass‑transfer
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After SA best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final hill‑climb for polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _final_hill_climb(
            best_seq, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] Final best score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
