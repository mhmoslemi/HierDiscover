"""
Propose a step‑function (non‑negative vector) that minimizes

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The objective is scale‑invariant, so we keep the vector on the probability
simplex (∑a_i = 1).  The algorithm combines three phases:

1. **Warm start** – a few Dirichlet draws are used to obtain a decent
   initial point (if the global variable ``height_sequence_1`` is present it
   is also tried).

2. **Smooth‑max mirror descent** – we minimise a smooth approximation of
   ``max(conv)``:
        f(a) = (1/α)·log(∑_k exp(α·conv_k)).
   The gradient of ``f`` is obtained analytically:
        ∇f = 2·corr(w , a)   with   w_k = softmax(α·conv_k).
   An exponentiated‑gradient (mirror descent on the simplex) step is taken:
        a_i ← a_i·exp(−η·∇_i) ,   then renormalise.
   The temperature ``α`` is slowly increased (annealed) so that the smooth
   objective approaches the true maximum, while the stepsize η decays like
   1/√t.

3. **Fine‐grained mixing & hill‑climbing** – after the main descent we
   scan a convex combination with the uniform distribution and perform a
   light random‑pair hill‑climb to squeeze out the last bits of improvement.

The whole routine is time‑aware (default budget ≈1000 s) and prints
intermediate scores.  The returned list contains non‑negative floats that
sum to 1 (the evaluator will re‑normalise them automatically).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)−1) via FFT."""
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * a.shape[0] - 1]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_k = exp(α·conv_k) / Σ exp(α·conv)
    in a numerically stable way.
    """
    # shift by max for stability
    shifted = alpha * conv
    m = shifted.max()
    exp_shifted = np.exp(shifted - m)
    w = exp_shifted / exp_shifted.sum()
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, L_corr: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective w.r.t. a:
        ∇ = 2 * corr(w, a)   (valid correlation, length = len(a))

    The correlation is performed via FFT for speed.
    """
    # correlation(w, a) = conv(w_rev, a) (valid part)
    w_rev = w[::-1]
    # FFT of both sequences, zero‑padded to L_corr
    Wr = np.fft.rfft(w_rev, n=L_corr)
    Ar = np.fft.rfft(a, n=L_corr)
    full = np.fft.irfft(Wr * Ar, n=L_corr)
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


def _objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Exact score used by the evaluator: 2·n·max(conv(a,a))."""
    conv = _conv_self_fft(a, L_fft)
    return float(2.0 * n * conv.max()), conv


# ----------------------------------------------------------------------
# 1️⃣  Warm‑start with Dirichlet draws (optional)
# ----------------------------------------------------------------------
def _warm_start(n: int,
                best_a: np.ndarray,
                best_score: float,
                deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # a few thousand draws are cheap; we continue while time permits
    while time.time() < deadline:
        # concentration drawn log‑uniformly in [0.02, 20]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        a = sample / sample.sum()
        sc, _ = _objective_score(a, n, _next_pow2(2 * n - 1))
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# 2️⃣  Smooth‑max mirror‑descent on the simplex
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        n: int,
                        L_fft: int,
                        L_corr: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror descent using exponentiated updates and a smooth approximation
    of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score, _ = _objective_score(a, n, L_fft)

    it = 0
    # annealing schedule for the soft‑max temperature
    alpha = 0.5          # start very smooth
    alpha_target = 250.0  # final value (approaches true max)

    while time.time() < deadline:
        it += 1
        # ---- exact score (for bookkeeping) ----
        cur_score, conv = _objective_score(a, n, L_fft)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] "
                  f"Iter={it:,}  score={cur_score:.6f}")

        # ---- smooth max computation ----
        w = _softmax_weights(conv, alpha)          # shape (2n‑1,)
        grad = _gradient_from_weights(a, w, L_corr)   # shape (n,)

        # ---- stepsize (η) – classic 1/√t schedule ----
        eta = 0.6 / np.sqrt(it + 1)

        # ---- multiplicative mirror‑descent update ----
        exp_factor = np.exp(-eta * grad)
        # clip to avoid overflow/underflow
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a = a * exp_factor
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ---- anneal α (slowly increase) ----
        # geometric increase until reaching the target
        if alpha < alpha_target:
            alpha = min(alpha * 1.001, alpha_target)

        # occasional random jitter to escape shallow plateaus
        if np.random.rand() < 0.005:
            eps = np.random.normal(scale=0.002, size=n)
            a = np.maximum(a + eps, 0.0)
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# 3️⃣  Final mixing with uniform + light hill‑climb
# ----------------------------------------------------------------------
def _final_refinement(a: np.ndarray,
                      n: int,
                      L_fft: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan convex mixtures with the uniform distribution and then perform a
    cheap random‑pair hill‑climb for the remaining time.
    """
    best_a = a.copy()
    best_score, _ = _objective_score(best_a, n, L_fft)

    uniform = np.full(n, 1.0 / n)

    # coarse λ‑grid
    for lam in np.linspace(0.0, 1.0, 31):
        if time.time() >= deadline:
            break
        cand = lam * a + (1.0 - lam) * uniform
        cand = cand / cand.sum()
        sc, _ = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # fine‑grained hill‑climb (random pair transfers)
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # move a small amount δ from i to j
        delta = rng.uniform(0.0, min(best_a[i], 0.005))
        if delta == 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc, _ = _objective_score(cand, n, L_fft)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb i={i} j={j} δ={delta:.5f} → score={sc:.6f}")

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and prints
    progress.  The returned list is a plain Python list with non‑negative
    entries and (approximately) unit sum.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 0.15

    # ------------------------------------------------------------------
    # Problem size – 4096 works well, keep it fixed for reproducibility
    # ------------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)          # for a*a
    L_corr = _next_pow2(3 * n - 2)         # for correlation w⋆a

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – uniform or provided starter
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score, _ = _objective_score(best_a, n, L_fft)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # Try to use a globally supplied starter (height_sequence_1) if present
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 1:
            # truncate or pad to length n
            if starter.size != n:
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size), constant_values=0.0)
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc, _ = _objective_score(starter, n, L_fft)
                if sc < best_score:
                    best_score, best_a = sc, starter.copy()
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start via Dirichlet draws (10 % of the budget)
    # ------------------------------------------------------------------
    warm_deadline = min(deadline, start_time + 0.10 * (deadline - start_time))
    if time.time() < warm_deadline:
        best_a, best_score = _warm_start(n, best_a, best_score, warm_deadline)

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max mirror descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_a, best_score = _smooth_max_descent(
            best_a, n, L_fft, L_corr, deadline
        )
        print(f"[{time.time() - start_time:.2f}s] After descent best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Mixing with uniform + hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Final refinement (mix + hill‑climb)")
        best_a, best_score = _final_refinement(best_a, n, L_fft, deadline)

    # ------------------------------------------------------------------
    # Normalisation & return
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # return plain Python list
    return [float(v) for v in best_a]
