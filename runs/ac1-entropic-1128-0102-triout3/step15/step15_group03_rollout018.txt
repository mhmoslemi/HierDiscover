def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    import time, numpy as np
    start_time = time.time()
    deadline = start_time + budget_s - 0.1 # safety
    np.random.seed(seed)
    n = 4096

    # arcsine distribution bins
    xs = np.arange(n+1, dtype=np.float64) / n   # boundaries 0..1 inclusive
    cdf = (2.0/np.pi) * np.arcsin(np.sqrt(xs))
    a = cdf[1:] - cdf[:-1]   # lengths n
    # Ensure positivity
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= a_sum

    best_seq = a.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time()-start_time:.2f}s] Arcsine init score={best_score:.6f}")

    # Subgradient descent on max(conv)
    iteration = 0
    step0 = 0.2
    a_cur = best_seq.copy()
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a_cur)
        grad = _subgradient_maxconv(a_cur, conv)
        gnorm = np.linalg.norm(grad) + 1e-12
        step = step0 / np.sqrt(iteration) / gnorm
        a_next = a_cur - step * grad
        # project onto simplex
        a_next = np.maximum(a_next, 0.0)
        a_next = _project_simplex(a_next)
        cur_score = _objective_score(a_next)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_next.copy()
            print(f"[{time.time()-start_time:.2f}s] iter {iteration} improved score={best_score:.6f}")
        a_cur = a_next
    # final projection and normalization
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(n, 1.0/n)
    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    import numpy as np, time
    start = time.time()
    deadline = start + budget_s - 0.1
    np.random.seed(seed)

    n = 4096 # could adjust based on remaining time
    # try to get global starter
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim ==1 and starter.size>0:
            # clip negatives -> 0
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s>0:
                starter = starter / s
                # If not length n, resample: if length differs, we can interpolate or pad/truncate
                if starter.shape[0] != n:
                    # if more, truncate; if less, repeat or upsample
                    # For simplicity, use interpolation
                    xi = np.linspace(0,1, starter.shape[0])
                    x_new = np.linspace(0,1,n)
                    starter = np.interp(x_new, xi, starter)
                # ensure sum=1
                starter = starter / starter.sum()
                best_seq = starter.copy()
                best_score = _objective_score(best_seq)
                print(f"[{time.time()-start:.2f}s] Using global starter score={best_score:.6f}")
            else:
                raise ValueError
        else:
            raise ValueError
    except Exception:
        # fallback arcsine
        xs = np.linspace(0,1,n+1)
        cdf = (2.0/np.pi) * np.arcsin(np.sqrt(xs))
        a = cdf[1:] - cdf[:-1]
        a = np.maximum(a,0.0)
        a = a / a.sum()
        best_seq = a.copy()
        best_score = _objective_score(best_seq)
        print(f"[{time.time()-start:.2f}s] Arcsine init score={best_score:.6f}")

    # Subgradient descent
    a = best_seq.copy()
    iteration = 0
    step0 = 0.5  # start larger to move quickly
    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)
        # compute step
        gnorm = np.linalg.norm(grad) + 1e-12
        step = step0 / np.sqrt(iteration) / gnorm
        a_next = a - step * grad
        a_next = np.maximum(a_next, 0.0)
        a_next = _project_simplex(a_next)
        cur_score = _objective_score(a_next)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_next.copy()
            print(f"[{time.time()-start:.2f}s] iter {iteration} improved score={best_score:.6f}")
        a = a_next

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0/n)

    print(f"[{time.time()-start:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(v) for v in best_seq]

"""
Propose a step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three fast phases (≈ 1000 s total).

1. **Starter** – reuse the best sequence found so far (`height_sequence_1`);
   if it is unavailable we fall back to the analytically known
   arcsine‑distribution (the continuous law whose self‑convolution is flat).

2. **Projected sub‑gradient descent** – the objective is `max(conv(a,a))`
   (the denominator is constant after normalising to sum = 1).  The
   sub‑gradient is obtained analytically (see `_subgradient_maxconv`).  At
   each iteration we move a step opposite to the sub‑gradient and project the
   result onto the probability simplex.  The step size decays as
   `α / sqrt(iter) / ||grad||` (α≈0.5) which is a classic schedule for
   sub‑gradient methods.

3. **Escaping plateaux** – after a long streak without improvement a small
   Gaussian jitter is added and the iterate is re‑projected; this helps the
   optimiser leave shallow local minima.

All heavy work (convolution and gradient) is done with FFTs (`_conv_self`);
the algorithm is fully deterministic apart from the occasional jitter and
uses only one CPU core.  The final sequence is normalised to sum = 1 and
returned as a plain Python list ready for `evaluate_sequence`.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine first tries to reuse the globally supplied
    ``height_sequence_1`` (the best result from a previous run).  If that is
    unavailable it falls back to the arcsine distribution.  A long projected
    sub‑gradient descent with occasional random jitter is then performed.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate n gives a good trade‑off.
    # --------------------------------------------------------------
    n = 4096

    # --------------------------------------------------------------
    # 0️⃣  Obtain a good starter (global if available, else arcsine).
    # --------------------------------------------------------------
    best_seq = None
    try:
        # ``height_sequence_1`` may be defined in the surrounding namespace.
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                # If the length differs, re‑interpolate to the desired n.
                if starter.shape[0] != n:
                    xi = np.linspace(0.0, 1.0, starter.shape[0])
                    x_new = np.linspace(0.0, 1.0, n)
                    starter = np.interp(x_new, xi, starter)
                starter = starter / starter.sum()
                best_seq = starter.copy()
                best_score = _objective_score(best_seq)
                print(f"[{time.time() - start_time:.2f}s] Global starter score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        # fallback: arcsine distribution (Cumulative of Beta(½,½))
        xs = np.linspace(0.0, 1.0, n + 1, dtype=np.float64)
        cdf = (2.0 / np.pi) * np.arcsin(np.sqrt(xs))
        a = cdf[1:] - cdf[:-1]               # probability mass of each bin
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        best_seq = a.copy()
        best_score = _objective_score(best_seq)
        print(f"[{time.time() - start_time:.2f}s] Arcsine start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Projected sub‑gradient descent on max(conv)
    # --------------------------------------------------------------
    a = best_seq.copy()
    iteration = 0
    no_improve = 0
    rng = np.random.default_rng(seed)
    step0 = 0.5               # initial step magnitude (will be scaled)

    while time.time() < deadline:
        iteration += 1
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        gnorm = np.linalg.norm(grad) + 1e-12
        step = step0 / np.sqrt(iteration) / gnorm

        a_next = a - step * grad
        a_next = np.maximum(a_next, 0.0)
        a_next = _project_simplex(a_next)

        cur_score = _objective_score(a_next)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_next.copy()
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] iter {iteration} improved score={best_score:.6f}")
        else:
            no_improve += 1

        # occasional jitter to escape plateaus
        if no_improve > 30000:
            noise = rng.normal(scale=0.01, size=n)   # ~1 % of mass per entry
            a_next = a_next + noise
            a_next = np.maximum(a_next, 0.0)
            a_next = _project_simplex(a_next)
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] jitter applied (iter {iteration})")

        a = a_next

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
