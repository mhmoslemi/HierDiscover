    n = a.shape[0]
    iteration = 0
    step_scale = 0.5  # starting scale
    while time.time() < subgrad_deadline:
        iteration += 1
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(...)
        # compute subgrad
        grad = _subgradient_maxconv(a, conv)
        # compute step size normalized
        grad_norm = np.linalg.norm(grad) + 1e-12
        step = step_scale / grad_norm
        # try reduce step if not improvement
        improved = False
        for _ in range(5):
            a_new = a - step * grad
            # enforce nonneg
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            conv_new = _conv_self(a_new)
            new_score = 2.0 * n * conv_new.max()
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                improved = True
                break
            else:
                step *= 0.5
        if not improved:
            # random jitter to escape
            rng = np.random.default_rng()

            a[i2] += eps
            a = _project_simplex(np.maximum(a,0))
        # gradually decay step_scale
        step_scale *= 0.9995
    return best_seq, best_score

"""
Propose a non‑negative step function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a lightweight local optimiser:

1. Initialise from the global `height_sequence_1` if it exists,
   otherwise start from the uniform distribution.
2. Run a long sub‑gradient descent on the *max* of the self‑convolution.
   The sub‑gradient is computed analytically and the iterate is projected
   onto the probability simplex after each step.  A simple back‑tracking line
   search adapts the step size; occasional random jitter helps to escape
   plateaus.
3. Finish with a fast hill‑climb that moves tiny amounts of mass from the
   “over‑represented’’ index to the “under‑represented’’ one.
4. Return the best sequence found (normalised to sum = 1).

The routine respects a 1000 s wall‑clock budget, printing intermediate
scores for monitoring.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Local optimisation stages
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max(conv).  A simple back‑tracking line
    search adapts the step size; random jitter is added when no improvement
    can be found.
    """
    a = a0.copy()
    n = a.shape[0]

    # step scale (L2 norm of the step) – start modestly and decay slowly.
    step_scale = 5e-3
    min_step = 1e-10
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it} score={cur_score:.6f}")

        # compute sub‑gradient
        grad = _subgradient_maxconv(a, conv)
        grad_norm = np.linalg.norm(grad) + 1e-16

        # try a few back‑tracking step sizes
        step = step_scale
        improved = False
        for _ in range(6):
            direction = grad / grad_norm
            a_candidate = a - step * direction
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

            new_score = _objective_score(a_candidate)
            if new_score < cur_score:
                a = a_candidate
                cur_score = new_score
                # on success slightly increase step scale (but keep it bounded)
                step_scale = min(step_scale * 1.02, 5e-2)
                improved = True
                break
            else:
                step *= 0.5

        if not improved:
            # no descent found – shrink step scale and inject a tiny jitter
            step_scale = max(step_scale * 0.5, min_step)
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * (rng.random() * 1e-5 + 1e-9)
            if eps > 0:
                a[i1] -= eps
                a[i2] += eps
                a = _project_simplex(np.maximum(a, 0.0))

    return best_seq, best_score


def _hill_climb(a0: np.ndarray,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climb that repeatedly moves a tiny fraction of mass from the
    index with the largest sub‑gradient (most “over‑represented’’) to the one
    with the smallest sub‑gradient.  The transferred amount is up to 1 % of
    the source mass, which is larger than the original 0.4 % and helps to
    break shallow plateaus.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # source – highest gradient among entries that still have mass
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        # destination – lowest gradient (could be same as src)
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # move up to 1 % of the source mass
        eps = a[src] * (rng.random() * 0.01 + 1e-8)
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # random jitter to escape a plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 1e-5 + 1e-9)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising the evaluation function.
    The routine uses a projected sub‑gradient descent followed by a short
    hill‑climb, respecting the 1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin

    # --------------------------------------------------------------
    # Problem size – moderate to keep each iteration cheap.
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Initialise from a global starter if available, else uniform.
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                best_seq = starter.copy()
            else:
                best_seq = uniform.copy()
        else:
            best_seq = uniform.copy()
    except Exception:
        best_seq = uniform.copy()

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = _project_simplex(best_seq)   # ensure exact normalisation
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – sub‑gradient descent (most of the budget)
    # --------------------------------------------------------------
    subgrad_deadline = deadline - 120.0      # reserve ~2 min for hill‑climb
    if time.time() < subgrad_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_seq, best_score = _subgradient_descent(
            best_seq, subgrad_deadline, best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Stage 2 – final hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, deadline, best_seq, best_score, rng)

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()
    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
