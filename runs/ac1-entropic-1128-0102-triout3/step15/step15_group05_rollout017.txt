"""
Mirror‑descent optimizer on a smooth approximation of the
max‑convolution objective.

The algorithm:
* Initialise from the global starter (if available) or from a
  uniform probability vector.
* Perform a brief random‑parametric sweep and a small hill‑climb
  (as in the reference implementation) to obtain a decent starting
  point.
* Run a mirror‑descent / exponentiated‑gradient loop that minimises
  a smoothed version of the max‑convolution (soft‑max with a
  temperature that is increased during the run).  The gradient is
  obtained efficiently with two FFT‑convolutions per iteration.
* Finish with a cheap random “mass‑transfer’’ local search.
All updates keep the vector on the probability simplex, respect the
non‑negativity requirement and stay well inside the 0 … 1000 s time budget.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (same as the reference code)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to r**i * (i+1)**p * (n-i)**q   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return lam·uniform + (1‑lam)·shape  (both already sum to 1).
    The result is renormalised to guard against numerical drift.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

# Random parametric sweep (as in the reference code)
# ----------------------------------------------------------------------


def _random_param_search(n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray,
                                                     float,
                                                     tuple[float, float, float, float]]:
    """
    Uniform‑in‑log random sampling of (r,p,q,λ) and keep the best vector.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
        p = rng.uniform(-3.0, 3.0)
        q = rng.uniform(-3.0, 3.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(f"[{time.time() - start_time:.2f}s] random search r={r:.4f} "
                  f"p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}")

    return best_seq, best_score, best_params


def _hillclimb_params(initial_params: tuple[float, float, float, float],
                      start_seq: np.ndarray,
                      start_score: float,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb on (r,p,q,λ) with exponentially shrinking
    perturbations.
    """
    cur_r, cur_p, cur_q, cur_lam = initial_params
    cur_seq = start_seq.copy()
    cur_score = start_score

    sigma_r = 0.015
    sigma_p = 0.30
    sigma_q = 0.30
    sigma_lam = 0.12

    rng = np.random.default_rng()
    n = cur_seq.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_seq = cur_seq.copy()
    best_score = cur_score

    while time.time() < deadline:
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        r_new = float(np.clip(r_new, 0.85, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape_general(n, r_new, p_new, q_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        if sc_new < cur_score:
            cur_r, cur_p, cur_q, cur_lam = r_new, p_new, q_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(f"[{time.time() - start_time:.2f}s] param hill r={r_new:.6f} "
                      f"p={p_new:.4f} q={q_new:.4f} λ={lam_new:.4f} score={sc_new:.6f}")

        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_q *= 0.9995
        sigma_lam *= 0.9995

        if rng.random() < 0.005:
            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient (new)
# ----------------------------------------------------------------------


def _smooth_max_gradient(a: np.ndarray,
                         alpha: float,
                         L_self: int,
                         L_grad: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (conv, grad) where
    - conv = a * a (linear convolution, length 2n‑1)
    - grad  = ∂/∂a [ Σ_k w_k conv_k ] with
      w_k ∝ exp(alpha * conv_k) (soft‑max weights).
    The gradient is computed via a second FFT‑convolution.
    """
    # 1) self‑convolution
    conv = _conv_self_fft(a, L_self)          # length 2n‑1

    # 2) soft‑max weights
    scaled = conv * alpha
    max_scaled = np.max(scaled)
    expw = np.exp(scaled - max_scaled)        # stability trick
    w = expw / np.sum(expw)                   # shape (2n‑1,)

    # 3) gradient: g_i = 2 * Σ_j a_j w_{i+j}
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L_grad)
    Wf = np.fft.rfft(w, n=L_grad)
    conv_rev = np.fft.irfft(Af * Wf, n=L_grad)
    # take the central segment that corresponds to indices 0…n‑1
    offset = (len(w) - 1) // 2               # equals n‑1
    g = 2.0 * conv_rev[offset: offset + a.shape[0]]
    return conv, g

# Mirror‑descent optimiser on the smoothed objective
# ----------------------------------------------------------------------


def _mirror_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on the simplex) that
    minimises a smoothed version of the max‑convolution.
    The temperature α (soft‑max sharpness) is increased slowly,
    allowing a smooth start and a focus on the true max at the end.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    n = a.shape[0]
    L_self = _next_pow2(2 * n - 1)           # for self‑convolution
    L_grad = _next_pow2(3 * n - 2)           # for a_rev * w convolution
    rng = np.random.default_rng()
    eta0 = 0.5                               # initial step size

"""
Mirror‑descent optimizer on a smooth approximation of the max‑convolution
objective.

The algorithm proceeds as follows:

* Initialise from the global starter (if present) or from a uniform vector.
* Run a brief random‑parameter sweep and a short hill‑climb (as in the
  reference code) to obtain a decent starting point.
* Perform a mirror‑descent / exponentiated‑gradient loop that minimises a
  softened version of the max‑convolution (soft‑max with a temperature
  that is slowly increased).  The gradient is obtained efficiently with two
  FFT‑convolutions per iteration while staying on the probability simplex.
* End with a cheap random “mass‑transfer’’ local search.
All updates respect non‑negativity and the 0 … 1000 s time budget.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to r**i * (i+1)**p * (n-i)**q   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return lam·uniform + (1‑lam)·shape  (both already sum to 1).
    The result is renormalised to guard against numerical drift.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed

def _random_param_search(n: int,

                         start_time: float) -> tuple[np.ndarray,

    Uniform‑in‑log random sampling of (r,p,q,λ) and keep the best vector.

    best_seq = None
    best_score = float("inf")
    best_params = None

        p = rng.uniform(-3.0, 3.0)
        q = rng.uniform(-3.0, 3.0)

def _smooth_max_gradient(a: np.ndarray,
                         alpha: float,
                         L_self: int,
                         L_grad: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (conv, grad) where
    - conv = a * a (linear convolution, length 2n‑1)
    - grad  = ∂/∂a [ Σ_k w_k conv_k ] with
      w_k ∝ exp(alpha * conv_k) (soft‑max weights).
    The gradient is computed via a second FFT‑convolution.
    """
    # 1) self‑convolution
    conv = _conv_self_fft(a, L_self)          # length 2n‑1

    # 2) soft‑max weights
    scaled = conv * alpha
    max_scaled = np.max(scaled)
    expw = np.exp(scaled - max_scaled)        # stability trick
    w = expw / np.sum(expw)                   # shape (2n‑1,)

    # 3) gradient: g_i = 2 * Σ_j a_j w_{i+j}
    a_rev = a[::-1]
    Af = np.fft.rfft(a_rev, n=L_grad)
    Wf = np.fft.rfft(w, n=L_grad)
    conv_rev = np.fft.irfft(Af * Wf, n=L_grad)
    offset = (len(w) - 1) // 2               # equals n‑1
    g = 2.0 * conv_rev[offset: offset + a.shape[0]]
    return conv, g

        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Log official evaluation (uses the provided security‑checked function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
