def _smooth_obj_and_grad(w, alpha, n):
    a = np.exp(w)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= s
    # conv_self
    conv = _conv_self(a)  # length 2n-1
    # smooth max
    max_c = conv.max()
    exp_term = np.exp(alpha * (conv - max_c))
    sum_exp = exp_term.sum()
    weights = exp_term / sum_exp
    smooth_max = (np.log(sum_exp) + alpha * max_c) / alpha
    obj = 2.0 * n * smooth_max
    # gradient w.r.t a
    rev_weights = weights[::-1]
    full = _conv_fft(rev_weights, a)  # length 3n-2
    start_idx = len(rev_weights) - 1
    grad_a = 2.0 * full[start_idx - np.arange(n)]
    # Chain to w
    scalar = np.dot(grad_a, a)
    grad_w = a * (grad_a - scalar)
    return obj, grad_w, a, conv.max()  # also return actual max conv for tracking

"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

using a smooth‑max surrogate and an Adam optimiser on the simplex.
The vector is represented by log‑variables w (a_i = exp(w_i) / sum(exp(w))).
A schedule of increasing smoothness (α) guides the optimiser towards the true
maximum.  Symmetry (a_i = a_{n‑1‑i}) is enforced each iteration, which matches
the best known constructions.  Several random restarts are performed within a
1000 s budget; the best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities (same as in the reference code)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full convolution of two real vectors using an FFT of length L."""
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT of length L."""
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (sum(a) = 1 ⇒ denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) under the constraint sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, alpha: float, n: int, L: int):
    """
    Returns (smooth_objective, gradient_wrt_w, a, max_conv_exact).

    a = softmax(w) (non‑negative, sum = 1)
    smooth_objective = 2·n·smooth_max(conv) with smooth_max being a
    log‑sum‑exp approximation of max(conv).
    """
    # softmax
    a = np.exp(w)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # self‑convolution a*a (length 2n‑1)
    conv = _conv_self(a, L)               # shape (2n‑1,)

    # smooth max via log‑sum‑exp
    max_c = conv.max()
    exp_shift = np.exp(alpha * (conv - max_c))          # shift to avoid overflow
    sum_exp = exp_shift.sum()
    weights = exp_shift / sum_exp                       # softmax over shift‑indices
    smooth_max = (np.log(sum_exp) + alpha * max_c) / alpha
    smooth_obj = 2.0 * n * smooth_max

    # gradient w.r.t. a :  ∂ ( Σ_k w_k·conv_k )/∂a = 2· Σ_j w_{i+j}·a_j
    # where w_k = weights[k]
    rev_w = weights[::-1]                               # length 2n‑1
    full = _conv_fft(rev_w, a, L)                       # length 3n‑2
    start_idx = len(rev_w) - 1                          # = 2n‑2
    idx = start_idx - np.arange(n)                      # indices for i=0..n‑1
    grad_a = 2.0 * full[idx]                            # shape (n,)

    # chain rule through the softmax: w_i = log a_i  (up to normalisation)
    scalar = np.dot(grad_a, a)
    grad_w = a * (grad_a - scalar)

    return smooth_obj, grad_w, a, conv.max()

                best_obj = exact_obj
                best_a = a_curr.copy()
                print(f"[{time.time() - start_time:.2f}s] Restart {restart} it {it:,} new best obj={best_obj:.6f}")

            # Adam update on w
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_w
            v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            w -= step

            # enforce symmetry on w (and consequently on m, v)
            w = (w + w[::-1]) * 0.5
            m = (m + m[::-1]) * 0.5
            v = (v + v[::-1]) * 0.5

            # occasional cheap check (every few thousand iterations)
            if it % 5000 == 0:
                # recompute exact objective without the smooth surrogate
                exact_obj_check = _objective_score(a_curr)
                if exact_obj_check < best_obj:
                    best_obj = exact_obj_check
                    best_a = a_curr.copy()
                    print(f"[{time.time() - start_time:.2f}s] Restart {restart} it {it:,} exact check best={best_obj:.6f}")

            # early exit if we already beat the target
            if best_obj <= 1.503:
                break

        restart += 1
        if best_obj <= 1.503:
            break   # stop all restarts early

    # ------------------------------------------------------------------
    # Final tidy‑up: enforce non‑negativity & normalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total > 0.0:
        best_a = best_a / total
    else:
        best_a = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best objective ≈ {best_obj:.6f}")
    return [float(x) for x in best_a]
