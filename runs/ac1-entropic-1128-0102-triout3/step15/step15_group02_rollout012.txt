import time, numpy as np

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    def conv_self(a):
        n = a.shape[0]
        L = 1 << ((2*n - 2).bit_length())
        Af = np.fft.rfft(a, n=L)
        c = np.fft.irfft(Af*Af, n=L)[:2*n-1]
        return c

    def score_norm(a):
        # assumes sum(a)=1
        n = a.shape[0]
        c = conv_self(a)
        return 2.0 * n * c.max()
    
    def proj_simplex(v):
        # Euclidean simplex projection
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1)
            w = np.maximum(v - theta, 0)
        else:
            w = np.full_like(v, 1.0 / len(v))
        return w

    # We'll consider a few lengths
    lengths = [1024, 2048, 4096]
    best_seq = None
    best_score = float('inf')

    for n in lengths:
        if time.time() >= deadline:
            break
        # baseline uniform
        uniform = np.full(n, 1.0 / n)
        cur_best = uniform.copy()
        cur_score = score_norm(cur_best)
        if cur_score < best_score:
            best_score, best_seq = cur_score, cur_best.copy()
        print(f"[{time.time() - start:.2f}s] n={n} uniform score {cur_score:.6f}")

        # parametric families
        # 1) Beta shape
        alphas = np.logspace(-2, 2, 25)  # 0.01 to 100
        betas = alphas
        idx = np.arange(n, dtype=np.float64)
        for a in alphas:
            if time.time() >= deadline: break
            log_a = (a-1) * np.log(idx+1)
            for b in betas:
                if time.time() >= deadline: break
                log_seq = log_a + (b-1) * np.log(np.maximum(n-idx, 1))
                seq = np.exp(log_seq - np.max(log_seq))
                seq_sum = seq.sum()
                if seq_sum == 0 or not np.isfinite(seq_sum):
                    continue
                seq /= seq_sum
                s = score_norm(seq)
                if s < cur_score:
                    cur_score = s
                    cur_best = seq.copy()
                    if s < best_score:
                        best_score, best_seq = s, seq.copy()
                    print(f"[{time.time() - start:.2f}s] n={n} Beta a={a:.3f} b={b:.3f} score={s:.6f}")

        # 2) Power shape
        pvals = np.linspace(-3,3,31)
        for p in pvals:
            if time.time() >= deadline: break
            seq = np.power(idx+1, p)
            seq_sum = seq.sum()
            if seq_sum == 0: continue
            seq /= seq_sum
            s = score_norm(seq)
            if s < cur_score:
                cur_score = s
                cur_best = seq.copy()
                if s < best_score:
                    best_score, best_seq = s, seq.copy()
                print(f"[{time.time() - start:.2f}s] n={n} power p={p:.2f} score={s:.6f}")

        # 3) Random Dirichlet sampling
        rng = np.random.default_rng()
        for _ in range(5000):
            if time.time() >= deadline: break
            log_c = rng.uniform(np.log(0.01), np.log(50.0))
            c = np.exp(log_c)
            a = rng.gamma(shape=c, scale=1.0, size=n)
            a_sum = a.sum()
            if a_sum ==0: continue
            a /= a_sum
            s = score_norm(a)
            if s < cur_score:
                cur_score = s
                cur_best = a.copy()
                if s < best_score:
                    best_score, best_seq = s, a.copy()
                # maybe print
                print(f"[{time.time() - start:.2f}s] n={n} Dirichlet c={c:.3f} score={s:.6f}")

        # 4) Exponentiated gradient descent from cur_best
        a = cur_best.copy()
        eta0 = 0.1
        it = 0
        while time.time() < deadline and it < 20000:
            it += 1
            conv = conv_self(a)
            max_val = conv.max()
            # compute subgradient
            max_val = conv.max()
            tol = 1e-12
            active = np.where(conv >= max_val - tol)[0]
            grad = np.zeros_like(a)
            T = len(a)
            for k in active:
                lo = max(0, k - (T-1))
                hi = min(T-1, k)
                idx_range = np.arange(lo, hi+1)
                grad[idx_range] += a[k - idx_range]
            if active.size:
                grad = 2.0 * grad / active.size
            # adapt step
            eta = eta0 / (1.0 + it * 1e-5)
            exp_factor = np.exp(-eta * grad)
            a_new = a * exp_factor
            a_new = proj_simplex(a_new)
            new_score = score_norm(a_new)
            if new_score + 1e-12 < cur_score:
                a = a_new
                cur_score = new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, a_new.copy()
                    print(f"[{time.time() - start:.2f}s] n={n} EG iteration {it} score {new_score:.6f}")
            else:
                # maybe occasionally accept worse move to escape
                if np.random.rand() < 0.001:
                    a = a_new
            # Stop after some iterations
        # after EG, store
        # 5) Simulated annealing random swaps
        a = cur_best.copy()
        cur_score = score_norm(a)
        T0 = 0.02
        # linear decay exponent
        total_time = deadline - time.time()
        end_time = time.time() + total_time
        while time.time() < deadline:
            # pick random src,dst
            src = rng.integers(n)
            if a[src] < 1e-12:
                continue
            dst = rng.integers(n)
            if src == dst:
                continue
            eps = a[src] * rng.random() * 0.1  # up to 10%
            a_prop = a.copy()
            a_prop[src] -= eps
            a_prop[dst] += eps
            # ensure nonnegative
            if a_prop[src] < -1e-12:
                continue
            new_score = score_norm(a_prop)
            delta = new_score - cur_score
            # temperature schedule
            elapsed = time.time() - start
            T = T0 * (0.99 ** (elapsed / (budget_s/10)))  # decays
            if delta < 0 or np.exp(-delta / max(T, 1e-12)) > rng.random():
                a = a_prop
                cur_score = new_score
                if new_score < best_score:
                    best_score, best_seq = new_score, a.copy()
                    print(f"[{time.time() - start:.2f}s] n={n} SA moved score {new_score:.6f}")

        print(f"[{time.time() - start:.2f}s] Finished length {n} best_score {cur_score:.6f}")

    # final rescale/clip
    if best_seq is None:
        best_seq = uniform
    # enforce positivity/cap
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full(best_seq.shape[0], 1.0 / best_seq.shape[0])

    print(f"[{time.time() - start:.2f}s] Overall best score ≈ {best_score:.6f} length {len(best_seq)}")
    return [float(x) for x in best_seq]

   The best sample is retained.

3. **Local refinement** – an exponentiated‑gradient descent on the exact
   max‑convolution (sub‑gradient of the L∞ norm) is run, followed by a
   simple Gaussian‑perturbation hill‑climb (simulated‑annealing style).
   The best vector encountered over all lengths is returned.

All operations respect a global time budget (default 1000 s).  The routine
always returns a normalised (sum = 1) non‑negative list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a * a (full length) using FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _objective_norm_score(a: np.ndarray) -> float:
    """
    Objective when the vector is already normalised (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative numbers that minimises
    `evaluate_sequence`.  The routine respects the time budget
    and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin so that we always finish before the external timeout
    deadline = start_time + budget_s - 0.3

    # Helper to check remaining time
    def _time_left() -> float:
        return deadline - time.time()

    # ------------------------------------------------------------------
    # Global state (best sequence over all lengths)
    # ------------------------------------------------------------------
    best_seq_global = None
    best_score_global = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – small enough for many iterations, large enough
    # for a good asymptotic constant.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if _time_left() <= 0:
            break

        print(f"[{time.time() - start_time:.2f}s] ==== Exploring length n={n} ====")

        # ---- baseline uniform -----------------------------------------
        uniform = np.full(n, 1.0 / n)
        cur_best = uniform.copy()
        cur_score = _objective_norm_score(cur_best)

        # Incorporate a possible external starter (height_sequence_1)
        try:
            starter = np.asarray(height_sequence_1, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                # Resample / truncate / pad to current n
                if starter.size != n:
                    # simple linear interpolation to required size
                    x_old = np.linspace(0, 1, starter.size)
                    x_new = np.linspace(0, 1, n)
                    starter = np.interp(x_new, x_old, starter)
                starter = np.maximum(starter, 0.0)
                s = starter.sum()
                if s > 0:
                    starter /= s
                    sc = _objective_norm_score(starter)
                    if sc < cur_score:
                        cur_score, cur_best = sc, starter.copy()
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 1️⃣  Parametric families
        # ------------------------------------------------------------------
        idx = np.arange(n, dtype=np.float64)

        # ---- Beta family (wide range) ---------------------------------
        alphas = np.logspace(-2, 2, 25)   # 0.01 … 100
        betas = alphas
        log_idx = np.log(idx + 1.0)
        log_rev = np.log(np.maximum(n - idx, 1.0))

        for a in alphas:
            if _time_left() <= 0:
                break
            log_a = (a - 1.0) * log_idx
            for b in betas:
                if _time_left() <= 0:
                    break
                log_seq = log_a + (b - 1.0) * log_rev
                seq = np.exp(log_seq - np.max(log_seq))
                s = seq.sum()
                if s == 0 or not np.isfinite(s):
                    continue
                seq /= s
                sc = _objective_norm_score(seq)
                if sc < cur_score:
                    cur_score, cur_best = sc, seq.copy()
                    print(f"[{time.time() - start_time:.2f}s] n={n} Beta a={a:.3f} b={b:.3f} → {sc:.6f}")

        # ---- Power shape ------------------------------------------------
        p_vals = np.linspace(-3.0, 3.0, 31)
        for p in p_vals:
            if _time_left() <= 0:
                break
            seq = np.power(idx + 1.0, p)
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_norm_score(seq)
            if sc < cur_score:
                cur_score, cur_best = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] n={n} power p={p:.2f} → {sc:.6f}")

        # ---- Sine‑power -------------------------------------------------
        p_vals = np.linspace(0.1, 5.0, 31)
        for p in p_vals:
            if _time_left() <= 0:
                break
            seq = np.sin(np.pi * (idx + 0.5) / n) ** p
            seq = np.clip(seq, 0, None)
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_norm_score(seq)
            if sc < cur_score:
                cur_score, cur_best = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] n={n} sine‑power p={p:.2f} → {sc:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Random Dirichlet sampling (intensive)
        # ------------------------------------------------------------------
        rng = np.random.default_rng()
        dirichlet_trials = 8000 if n <= 2048 else 5000
        for _ in range(dirichlet_trials):
            if _time_left() <= 0:
                break
            log_c = rng.uniform(np.log(0.01), np.log(50.0))
            c = np.exp(log_c)
            a = rng.gamma(shape=c, scale=1.0, size=n)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            sc = _objective_norm_score(a)
            if sc < cur_score:
                cur_score, cur_best = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] n={n} Dirichlet c={c:.3f} → {sc:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣  Exponentiated‑gradient descent on exact max‑convolution
        # ------------------------------------------------------------------
        a = cur_best.copy()
        eta0 = 0.1
        it = 0
        max_it = 20000
        while _time_left() > 0 and it < max_it:
            it += 1
            conv = _conv_self(a)
            grad = _subgradient_maxconv(a, conv)

            # decrease step size very slowly – this keeps the method
            # exploratory for a long time.
            eta = eta0 / (1.0 + it * 1e-5)
            exp_factor = np.exp(-eta * grad)
            a_new = a * exp_factor
            a_new = _project_simplex(a_new)

            new_score = _objective_norm_score(a_new)
            if new_score + 1e-12 < cur_score:
                a = a_new
                cur_score = new_score
                if new_score < best_score_global:
                    best_score_global, best_seq_global = new_score, a_new.copy()
                    print(f"[{time.time() - start_time:.2f}s] n={n} EG it={it:,} score={new_score:.6f}")

        # ------------------------------------------------------------------
        # 4️⃣  Simple Gaussian‑perturbation hill‑climb (simulated annealing)
        # ------------------------------------------------------------------
        a = cur_best.copy()
        cur_score = _objective_norm_score(a)

        # exponential cooling – start relatively warm and finish cold
        T0 = 0.02
        total_sa_time = max(0.0, _time_left())
        sa_end = time.time() + total_sa_time
        while time.time() < sa_end:
            if _time_left() <= 0:
                break
            # random Gaussian perturbation
            scale = 0.05 * np.sqrt(a)  # adapt to magnitude of entries
            delta = rng.normal(scale=scale, size=n)
            a_prop = a + delta
            a_prop = np.maximum(a_prop, 0.0)
            a_prop = _project_simplex(a_prop)

            new_score = _objective_norm_score(a_prop)
            delta_obj = new_score - cur_score

            # current temperature (linear decay)
            elapsed = time.time() - start_time
            T = T0 * max(0.0, 1.0 - elapsed / budget_s)

            accept = False
            if delta_obj < -1e-12:
                accept = True
            else:
                prob = np.exp(-delta_obj / max(T, 1e-12))
                accept = rng.random() < prob

            if accept:
                a = a_prop
                cur_score = new_score
                if new_score < best_score_global:
                    best_score_global, best_seq_global = new_score, a.copy()
                    print(f"[{time.time() - start_time:.2f}s] n={n} SA move → {new_score:.6f}")

        # ------------------------------------------------------------------
        # End of this length – maybe update global best with cur_best
        # ------------------------------------------------------------------
        if cur_score < best_score_global:
            best_score_global, best_seq_global = cur_score, cur_best.copy()
        print(f"[{time.time() - start_time:.2f}s] Finished n={n} best_score={cur_score:.6f}")

    # ----------------------------------------------------------------------
    # Normalise final answer (the evaluation function is scale‑invariant,
    # but we keep a sane normalisation anyway).
    # ----------------------------------------------------------------------
    if best_seq_global is None:
        # fallback – uniform of length 1024
        best_seq_global = np.full(1024, 1.0 / 1024)

    best_seq_global = np.maximum(best_seq_global, 0.0)
    total = best_seq_global.sum()
    if total > 0:
        best_seq_global = best_seq_global / total
    else:
        best_seq_global = np.full_like(best_seq_global, 1.0 / best_seq_global.shape[0])

    print(f"[{time.time() - start_time:.2f}s] Overall best score ≈ {best_score_global:.6f} "
          f"(length {len(best_seq_global)})")
    return [float(x) for x in best_seq_global]
