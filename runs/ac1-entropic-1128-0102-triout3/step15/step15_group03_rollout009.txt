def _smooth_max_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    beta = 1.0
    beta_max = 300.0  # high enough
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute conv self
        conv = _conv_self(a)   # using FFT for speed
        max_val = conv.max()
        # compute weights
        # stable exponentials:
        # subtract max to avoid overflow; multiply by beta
        exp_arg = beta * (conv - max_val)
        # To avoid underflow for very negative arguments: use np.exp with safe handling.
        # May need to clip to -700 for float64.
        exp_arg = np.clip(exp_arg, -700, 0)   # because max arg is 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        # if w_sum is zero (shouldn't happen), revert to uniform
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0
        w /= w_sum
        # gradient via convolution
        w_rev = w[::-1]
        conv_full = _conv_fft(a, w_rev)   # result length 3n-2
        idx_base = 2*n - 2
        grad = 2.0 * conv_full[idx_base - np.arange(n)]
        # step size
        gnorm = np.linalg.norm(grad) + 1e-12
        # step scaling schedule decreasing as beta increases
        step = 0.1 / (beta + 1.0)  # base step
        # Optionally compute step based on gnorm
        step = step / gnorm
        # update
        a_new = a - step * grad
        # project onto simplex
        a_new = _project_simplex(np.maximum(a_new, 0.0))
        a = a_new

        # evaluate actual score using objective score
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth descent iter={iteration} beta={beta:.2f} score={cur_score:.6f}")

        # increase beta gradually
        if beta < beta_max:
            beta = min(beta_max, beta * 1.01)

    return best_seq, best_score

assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where `a*a` is the linear self‑convolution of the vector `a`.  
If we normalise the vector to a probability distribution (`∑a_i = 1`) the
objective reduces to  

\[
F(a)=2\,n\;\max (a*a).
\]

The problem is a non‑convex quadratic programme.  A very effective way to
drive the maximum of the convolution down is to minimise a *smooth*
approximation of the `max` operator, namely the **log‑sum‑exp** (soft‑max)

\[
\phi_\beta(a)=\frac1\beta\log\!\Bigl(\sum_{k}e^{\beta\,(a*a)_k}\Bigr),
\qquad \beta>0 .
\]

When \(\beta\) is small \(\phi_\beta\) is a gentle average; as \(\beta\)
grows it converges to the true maximum.
The gradient of \(\phi_\beta\) is available in closed form and can be
computed with two FFT‑based convolutions, which makes each iteration
\(O(n\log n)\).

The algorithm proceeds in three phases

1. **Initialisation** – a handful of simple families (uniform, geometric,
   two‑exponential “U‑shape”, a symmetric Beta distribution) and a few
   random Dirichlet samples are evaluated; the best of them is kept.
2. **Smooth‑max descent** – starting with \(\beta=1\) the vector is updated
   by projected gradient descent on \(\phi_\beta\).  After every iteration
   the step size is reduced if no improvement is seen, and \(\beta\) is
   multiplied by a small factor (≈ 1.02) so that the optimisation gradually
   focuses on the true supremum.
3. **Final projection** – the best vector found is projected onto the
   probability simplex and returned as a plain Python list.

All operations respect the required non‑negativity and the
\(0\le a_i\le 1000\) limits; the whole search stops well before the 1000 s
budget, but the time‑check is kept for safety.

The code below implements the whole pipeline.  It works for any
reasonable size; the default is `n = 8192`, which gives the algorithm more
degrees of freedom than the previous 4096‑point search while still being
fast enough on a single CPU.

```python
"""
Propose a non‑negative step‑function (list of floats) that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

The search consists of three stages

1.  Quick evaluation of a few analytically‑motivated families
    (uniform, geometric, two‑exponential U‑shape, symmetric Beta) and a
    handful of random Dirichlet draws.  The best of these seeds is kept.

2.  Projected gradient descent on a smooth approximation of the
    supremum of the self‑convolution.
    The smooth max is  φβ(a)= (1/β)·log Σk exp(β·(a*a)k) .
    Its gradient is  ∇φβ = 2·conv(a, w_rev)  with
    w = soft‑maxβ(conv)  (length 2·n‑1),  w_rev = w[::-1].
    Two FFT‑based convolutions give the gradient in O(n log n) time.
    β starts at 1 and is slowly increased (≈1 % per iteration) so that
    the optimisation first smoothes the profile and then concentrates on
    the true maximum.  After each step the iterate is projected onto the
    probability simplex.

3.  The best vector found is normalised, projected onto the feasible set
    and returned as a plain list for ``evaluate_sequence``.

The whole routine respects the 1000 s wall‑clock budget; intermediate
best scores are printed for monitoring.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using the FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    # length 2*n-1
    return _conv_fft(a, a)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (components ≥0 and sum = 1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helper (assumes sum(a) == 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth max
    φβ(a) = (1/β)·log Σ exp(β·(a*a)k).

    Parameters
    ----------
    a0 : np.ndarray
        Starting probability vector (sum = 1, non‑negative).
    deadline : float
        Absolute time at which the routine must stop.
    best_seq, best_score : current best solution.

    Returns
    -------
    best_seq, best_score
        Updated best vector and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    beta = 1.0            # smoothing parameter
    beta_max = 300.0      # after this we essentially optimise the true max
    it = 0

    # step candidates for a simple back‑tracking line search
    step_candidates = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]

    while time.time() < deadline:
        it += 1

        # ----- current convolution -------------------------------------------------
        conv = _conv_self(a)                 # length 2n‑1
        max_c = conv.max()

        # ----- soft‑max weights ----------------------------------------------------
        # w_i = exp(β·(conv_i−max)) / Σ exp(...)
        exp_arg = beta * (conv - max_c)       # max argument is 0 → numerically safe
        # clip to avoid underflow (exp(-700)≈5e-305)
        np.clip(exp_arg, -700, 0, out=exp_arg)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:                     # fallback (should not happen)
            w = np.full_like(w, 1.0 / w.size)
            w_sum = 1.0
        w /= w_sum

        # ----- gradient of φβ ------------------------------------------------------
        # ∇φβ = 2·conv(a, w_rev) evaluated at indices 2n‑2−j
        w_rev = w[::-1]                       # length 2n‑1
        conv_full = _conv_fft(a, w_rev)       # length 3n‑2
        idx = 2 * n - 2
        grad = 2.0 * conv_full[idx - np.arange(n)]

        # ----- line‑search ---------------------------------------------------------
        cur_score = _objective_score(a)
        improved = False

        for step in step_candidates:
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            a_try = _project_simplex(a_try)

            cand_score = _objective_score(a_try)
            if cand_score < cur_score - 1e-12:
                a = a_try
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a.copy()
                    print(
                        f"[{time.time() - start_time:.2f}s] "
                        f"smooth‑descent it={it} β={beta:.2f} step={step:.3f} "
                        f"score={cand_score:.6f}"
                    )
                break

        # if none of the candidates improved, just keep the current a
        # (the next iteration will use a larger β and a smaller step)

        # ----- update smoothing parameter -------------------------------------------
        if beta < beta_max:
            beta = min(beta_max, beta * 1.02)   # about +2 % per iteration

    return best_seq, best_score


# ----------------------------------------------------------------------
# Helper functions to generate a few analytical starting points
# ----------------------------------------------------------------------
def _geometric(n: int, r: float) -> np.ndarray:
    """Left‑geometric distribution (sum = 1)."""
    idx = np.arange(n, dtype=np.float64)
    g = np.power(r, idx)
    g /= g.sum()
    return g


def _two_exponential_mixture(n: int, r: float, w_left: float, w_right: float) -> np.ndarray:
    """U‑shaped mixture of a left and a right geometric tail plus uniform."""
    idx = np.arange(n, dtype=np.float64)
    left = np.power(r, idx)
    right = np.power(r, n - 1 - idx)
    left /= left.sum()
    right /= right.sum()
    uniform = np.full(n, 1.0 / n)
    mix = w_left * left + w_right * right + (1.0 - w_left - w_right) * uniform
    mix /= mix.sum()
    return mix


def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta density on the discrete grid."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    # work in log‑space to avoid overflow
    log_seq = (a - 1.0) * np.log(idx) + (b - 1.0) * np.log(n - idx + 1.0)
    seq = np.exp(log_seq - np.max(log_seq))
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation engine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the evaluation function.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    n = 8192
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Quick analytical seeds
    # ------------------------------------------------------------------
    # geometric left‑tail
    geo = _geometric(n, r=0.998)
    sc = _objective_score(geo)
    if sc < best_score:
        best_score, best_seq = sc, geo.copy()
        print(f"[{time.time() - start_time:.2f}s] Geometric seed score={sc:.6f}")

    # U‑shaped two‑exponential mixture
    ue = _two_exponential_mixture(n, r=0.995, w_left=0.3, w_right=0.3)
    sc = _objective_score(ue)
    if sc < best_score:
        best_score, best_seq = sc, ue.copy()
        print(f"[{time.time() - start_time:.2f}s] Two‑Exp seed score={sc:.6f}")

    # Symmetric Beta(0.5,0.5) – strongly U‑shaped
    beta01 = _beta_distribution(n, a=0.5, b=0.5)
    sc = _objective_score(beta01)
    if sc < best_score:
        best_score, best_seq = sc, beta01.copy()
        print(f"[{time.time() - start_time:.2f}s] Beta(0.5,0.5) seed score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (very cheap, only a few)
    # ------------------------------------------------------------------
    rng = np.random.default_rng()
    for _ in range(80):
        if time.time() >= deadline:
            break
        conc = 10 ** rng.uniform(-2.0, 0.7)          # concentration between ~0.01 and 5
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet seed (c={conc:.3f}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max projected gradient descent (main work)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max descent")
        best_seq, best_score = _smooth_max_descent(
            best_seq, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] End of descent, best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final sanitisation
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – final best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
```
