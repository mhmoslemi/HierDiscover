def smooth_gradient(a, conv, tau):
    maxc = conv.max()
    # for numeric stability subtract maxc
    exp_weights = np.exp(tau * (conv - maxc))
    weights = exp_weights / exp_weights.sum()
    # compute gradient = 2 * sum_k weights_k * a_{k-i}
    # We'll compute convolution of a with weights and then extract slice.
    # Use FFT.

    # w = weights
    # Compute conv_aw = convolution a * w (full) via FFT of size L
    L = _next_pow2(len(a) + len(weights) - 1)
    Af = np.fft.rfft(a, n=L)
    Wf = np.fft.rfft(weights, n=L)
    conv_aw = np.fft.irfft(Af * Wf, n=L)
    # conv_aw length = len(a) + len(weights) -1 = n + (2n-1) -1 = 3n-2.
    # Gradient_i = 2 * conv_aw[i + n - 1]
    grad = 2 * conv_aw[n-1:n-1+len(a)]
    return grad
```

def _mix_refine(a, n, deadline, start_time, best_seq, best_score):

    phi = (1 + np.sqrt(5)) / 2

    lam_a, lam_b = 0.0, 1.0

    seq_a = uniform
    seq_b = a
    # compute at lam_a (0) and lam_b (1) already known best

    # proceed:
    # golden section search:
    for _ in range(30):
        if time.time() >= deadline:
            break
        lam1 = lam_b - (lam_b - lam_a) / phi
        lam2 = lam_a + (lam_b - lam_a) / phi
        seq1 = lam1 * a + (1-lam1) * uniform
        seq1 = seq1 / seq1.sum()
        score1 = _objective_score(seq1, n)

        score2 = _objective_score(seq2, n)
        if score1 < score2:
            lam_b = lam2
            if score1 < best_score:
                best_score = score1; best_seq = seq1.copy()
        else:
            lam_a = lam1
            if score2 < best_score:
                best_score = score2; best_seq = seq2.copy()

def _beta_shape(alpha, beta, n):

"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm proceeds in four stages:

1️⃣  Broad random search – Dirichlet draws with a log‑uniform concentration
    together with a few deterministic Beta‑shaped seeds (edge‑heavy
    distributions).  The best candidate is kept.

2️⃣  Mirror descent on the simplex (exponentiated‑gradient).  The update
    uses the sub‑gradient of the max‑convolution, a 1/√t step‑size and
    occasional random perturbations to escape stagnation.

3️⃣  Targeted hill‑climbing – repeatedly moves a small amount of mass
    from the largest entry to the smallest (or random) entry, accepting
    improvements or occasional uphill moves (simulated‑annealing style).

4️⃣  Fine mixing with the uniform distribution – a golden‑section search
    over λ∈[0,1] for the convex combination λ·a+(1−λ)·uniform, keeping the
    best mixture.

All phases respect a global time‑budget (default 1000 s).  The final
sequence is normalised to sum 1, contains only non‑negative numbers and
is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat everything within a tiny tolerance as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # contribution a_{k‑i}
    if active.size:
        grad = 2.0 * grad / active.size  # factor 2 because each term appears twice
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta(α,β)‑shaped discrete density on n points."""
    if alpha <= 0 or beta <= 0:
        return None
    x = (np.arange(n, dtype=float) + 0.5) / n          # mid‑point of each bin
    # pdf ∝ x^{α‑1}(1‑x)^{β‑1}
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return None
    return a / s


# ----------------------------------------------------------------------
# 0️⃣  Global starter (if provided by the runner)
# ----------------------------------------------------------------------
def _maybe_use_global_starter(n: int) -> tuple[np.ndarray, float]:
    """If a global variable `height_sequence_1` exists, try to use it."""
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter, n)
                return starter, sc
    except Exception:
        pass
    return None, np.inf


# ----------------------------------------------------------------------
# 1️⃣  Broad random search (Dirichlet + Beta seeds)
# ----------------------------------------------------------------------
def _broad_random_search(n: int,
                         best_seq: np.ndarray,
                         best_score: float,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    # deterministic Beta seeds (evaluated once)
    beta_params = [(0.2, 0.2), (0.3, 0.3), (0.5, 0.5),
                   (0.2, 0.8), (0.8, 0.2), (0.3, 0.7), (0.7, 0.3)]
    beta_seeds = []
    for a, b in beta_params:
        v = _beta_shape(a, b, n)
        if v is not None:
            beta_seeds.append(v)

    while time.time() < deadline:
        # ---- Dirichlet draw ----
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s > 0:
            a = sample / s
            sc = _objective_score(a, n)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")

        # ---- Beta seeds (cheaper, evaluated only occasionally) ----
        # evaluate one random beta seed every few thousand Dirichlet draws
        if rng.random() < 0.005 and beta_seeds:
            v = beta_seeds[rng.integers(len(beta_seeds))]
            sc = _objective_score(v, n)
            if sc < best_score:
                best_score, best_seq = sc, v.copy()
                print(f"[{time.time() - start_time:.2f}s] Beta seed → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Exponentiated‑gradient descent with occasional perturbation
# ----------------------------------------------------------------------
def _expgrad_descent(a0: np.ndarray,
                     n: int,
                     L_fft: int,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the sub‑gradient of max(conv).
    η decays like 1/√t.  After a long stall we add a small random perturbation.
    """
    a = a0.copy()
    rng = np.random.default_rng()
    it = 0
    last_improve_it = 0
    restart_interval = 300_000  # restart if no improvement for this many iters

    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_improve_it = it
            print(f"[{time.time() - start_time:.2f}s] Exp‑grad it={it:,} score={cur_score:.6f}")

        # sub‑gradient of the current max‑convolution
        grad = _subgradient_maxconv(a, conv)

        # step size η – classic 1/√t schedule (a little more aggressive)
        eta = 0.8 / math.sqrt(it)

        # multiplicative (exponential) update, then project onto the simplex
        exp_factor = np.exp(-eta * grad)
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a_new = a * exp_factor
        s = a_new.sum()
        if s <= 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s

        # evaluate the trial point
        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()

        # accept improvements; otherwise accept uphill with a tiny temperature
        if new_score < cur_score:
            a = a_new
        else:
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 1e-4 * (remaining / total)          # decaying temperature
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a_new

        # stall detection → random perturbation (helps to leave flat plateaux)
        if it - last_improve_it > restart_interval:
            perturb = rng.normal(scale=0.01, size=n) * a
            a = np.maximum(a + perturb, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s
            last_improve_it = it
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Targeted hill‑climbing with simulated‑annealing style acceptance
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                L_fft: int,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> tuple[np.ndarray, float]:
    """
    Small deterministic mass‑transfer moves (largest → smallest entry)
    plus occasional random moves.  Accept if improvement, otherwise
    with a temperature‑scaled probability.
    """
    a = a0.copy()
    cur_score = 2.0 * n * _conv_self(a, L_fft).max()
    rng = np.random.default_rng()

    while time.time() < deadline:
        # decide donor and receiver
        if rng.random() < 0.7:
            i = np.argmax(a)                     # biggest mass
        else:
            i = rng.integers(n)                  # random donor
        if rng.random() < 0.7:
            j = np.argmin(a)                     # smallest mass
        else:
            j = rng.integers(n)                  # random receiver

        if i == j or a[i] <= 1e-14:
            continue

        # move a tiny fraction of the donor’s mass
        eps = min(0.02 * a[i], 1e-4)              # at most 2 % of donor or an absolute cap
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0:
            continue  # should never happen but keep safety

        new_score = 2.0 * n * _conv_self(a_new, L_fft).max()
        if new_score < cur_score:
            a, cur_score = a_new, new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time() - start_time:.2f}s] Hill‑climb improved to {new_score:.6f}")
        else:
            # annealing acceptance
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 5e-5 * (remaining / total)
            if np.random.rand() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a, cur_score = a_new, new_score
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fine mixing with the uniform distribution (golden‑section search)
# ----------------------------------------------------------------------
def _mix_with_uniform_fine(a: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """
    Optimise λ∈[0,1] for seq = λ·a + (1‑λ)·uniform using a few golden‑section
    evaluations.  The best mixture is kept.
    """
    uniform = np.full(n, 1.0 / n)

    def _score_lambda(lam: float) -> float:
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        return _objective_score(seq, n)

    # golden‑section constants
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    lo, hi = 0.0, 1.0
    # already have the ends (λ=0 → uniform, λ=1 → a)
    # they are implicitly considered via best_seq / best_score

    while time.time() < deadline:
        lam1 = hi - (hi - lo) / phi
        lam2 = lo + (hi - lo) / phi
        sc1 = _score_lambda(lam1)
        sc2 = _score_lambda(lam2)

        if sc1 < sc2:
            hi = lam2
            if sc1 < best_score:
                best_score = sc1
                best_seq = lam1 * a + (1.0 - lam1) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam1:.4f} → score={sc1:.6f}")
        else:
            lo = lam1
            if sc2 < best_score:
                best_score = sc2
                best_seq = lam2 * a + (1.0 - lam2) * uniform
                best_seq = best_seq / best_seq.sum()
                print(f"[{time.time() - start_time:.2f}s] Mix λ={lam2:.4f} → score={sc2:.6f}")

        if hi - lo < 1e-5:          # sufficient precision
            break
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores, finally returning the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin (the runner will kill the process after budget_s seconds)
    deadline = start_time + budget_s - 0.25

    # problem size – 4096 works well and matches the previous best
    n = 4096
    L_fft = _next_pow2(2 * n - 1)

    # ---- 0️⃣ Initialise with uniform & optional global starter ----
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    starter_seq, starter_sc = _maybe_use_global_starter(n)
    if starter_seq is not None and starter_sc < best_score:
        best_seq, best_score = starter_seq, starter_sc
        print(f"[0.00s] Global starter improved score={starter_sc:.6f}")

    # ---- 1️⃣ Broad random search (≈20 % of the budget) ----
    phase1_deadline = start_time + 0.20 * (deadline - start_time)
    if time.time() < phase1_deadline:
        best_seq, best_score = _broad_random_search(
            n, best_seq, best_score, phase1_deadline, start_time
        )

    # ---- 2️⃣ Exponentiated‑gradient descent (≈55 % of the budget) ----
    phase2_deadline = start_time + 0.75 * (deadline - start_time)
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting exponentiated‑gradient descent")
        best_seq, best_score = _expgrad_descent(
            best_seq, n, L_fft, start_time, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After exp‑grad best score≈{best_score:.6f}")

    # ---- 3️⃣ Hill‑climbing local refinements (≈20 % of the budget) ----
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting hill‑climb")
        best_seq, best_score = _hill_climb(
            best_seq, n, L_fft, start_time, deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After hill‑climb best score≈{best_score:.6f}")

    # ---- 4️⃣ Fine mixing with uniform (remaining time) ----
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Fine mixing with uniform")
        best_seq, best_score = _mix_with_uniform_fine(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ---- Final normalisation and return ----
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    # Convert to plain Python list of floats
    return [float(v) for v in best_seq]
