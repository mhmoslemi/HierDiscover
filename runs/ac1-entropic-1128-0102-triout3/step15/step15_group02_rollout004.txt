def _hill_climb_seq(a0, start_time, deadline, current_best_score):
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective_score(a)
    # temperature schedule: start T0 = 0.02; but we can set a bit higher
    T0 = 0.05
    total_hc_time = max(0.0, deadline - time.time())
    if total_hc_time <= 0:
        return best_seq, best_score
    decay = (1e-6 / T0) ** (1.0 / total_hc_time)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        # compute subgradient to get gradient direction
        grad = _subgradient_maxconv(a, conv)
        # choose src as index with largest positive gradient among a>0
        positive_indices = np.where(a > 1e-12)[0]
        if positive_indices.size == 0:
            break
        src = positive_indices[np.argmax(grad[positive_indices])]
        dst = np.argmin(grad)
        # Ensure src != dst, else random.
        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if a[src] < 1e-12:
                continue
        # choose epsilon fraction up to 0.2 of source mass
        max_frac = 0.2
        frac = rng.random() * max_frac + 1e-6
        eps = a[src] * frac
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-12:
            continue
        # Renormalize to preserve sum? It stays sum unchanged.
        # Compute new score
        new_score = _objective_score(a_prop)
        delta = new_score - best_score
        # Temperature
        T = T0 * (decay ** (time.time() - start_time))
        if delta < -1e-12:
            a = a_prop
            best_score = new_score
            best_seq = a.copy()
        else:
            prob = np.exp(-delta / max(T, 1e-12))
            if rng.random() < prob:
                a = a_prop
                best_score = new_score
        # occasionally random shake
        if rng.random() < 0.001:
            # random perturbation of all entries
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum
    return best_seq, best_score

def _upsample_seq(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_pos = np.linspace(0, 1, old_n)
    new_pos = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_pos, old_pos, seq)
    # Ensure nonneg
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 0:
        new_seq = new_seq / s
    else:
        new_seq = np.full(new_n, 1.0 / new_n)
    return new_seq

"""
Hierarchical optimisation for the max‑convolution problem.

The routine builds candidate step functions on increasingly fine
grids (256 → 512 → 1024 → 2048 → 4096).  For the small grids a massive
random Dirichlet sweep is performed, keeping the best normalised
candidate.  The best candidate is up‑sampled to the next grid and a few
perturbations are optimised by a smooth‑max Adam optimiser, an
exponentiated‑gradient (EG) descent on the exact max‑convolution and a
simulated‑annealing hill‑climb with relatively large mass moves.
The whole pipeline respects a 1000 s wall‑clock budget and returns the
best sequence it found (as a plain list of floats).  The objective that
is minimised (after normalising the sum to 1) is

    f(a) = 2·n·max(conv(a,a))

which coincides with ``evaluate_sequence`` for non‑negative,
normalised inputs.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Low‑level utilities
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv_full


def _objective_score(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a normalised (sum=1) sequence."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# -------------------------------------------------------------
# Random Dirichlet generator
# -------------------------------------------------------------
_rng = np.random.default_rng()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex using a log‑uniform concentration."""
    log_c = _rng.uniform(np.log(0.01), np.log(20.0))
    c = np.exp(log_c)
    a = _rng.gamma(shape=c, scale=1.0, size=n)
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


# -------------------------------------------------------------
# Upsampling / interpolation
# -------------------------------------------------------------
def _upsample_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, then re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_pos = np.linspace(0, 1, old_n)
    new_pos = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_pos, old_pos, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0:
        return np.full(new_n, 1.0 / new_n)
    return new_seq / s


# -------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# -------------------------------------------------------------
def _adam_optimize_seq(a0: np.ndarray,
                       max_iters: int,
                       lr: float,
                       tau_start: float,
                       tau_factor: float,
                       tau_inc_interval: int,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth‑max approximation of the objective.
    Returns the best sequence encountered and its (exact) score.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    tau = tau_start

    best_seq = a.copy()
    best_score = _objective_score(a)

    L_conv = _next_pow2(2 * n - 1)          # for the convolution a*a
    L_grad = _next_pow2(3 * n - 2)          # for w * a_rev

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and current exact score
        conv = _conv_self(a, L_conv)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()

        # smooth‑max weights (softmax over the convolution values)
        max_c = conv.max()
        w_raw = np.exp(tau * (conv - max_c))
        w = w_raw / w_raw.sum()

        # gradient of the smooth‑max (2 * (w * a_rev) convolution)
        w_hat = np.fft.rfft(w, n=L_grad)
        a_rev_hat = np.fft.rfft(a[::-1], n=L_grad)
        conv_w_full = np.fft.irfft(w_hat * a_rev_hat, n=L_grad)
        grad = 2.0 * conv_w_full[:n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        a = a_new

        if it % tau_inc_interval == 0:
            tau = min(tau * tau_factor, 1e9)

    # final exact evaluation (may be slightly better than best_score)
    final_score = _objective_score(a)
    if final_score < best_score:
        best_score, best_seq = final_score, a.copy()
    return best_seq, best_score


# -------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# -------------------------------------------------------------
def _eg_optimize_seq(a0: np.ndarray,
                     max_iters: int,
                     eta0: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the simplex using the exact sub‑gradient."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        eta_t = eta0 / (1.0 + it * 1e-4)
        exp_factor = np.exp(np.clip(-eta_t * grad, -700, 700))
        a_new = a * exp_factor
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        new_score = _objective_score(a_new)
        if new_score < best_score:
            best_score, best_seq = new_score, a_new.copy()
            a = a_new
        else:
            # modest decay of the step size if no progress
            if it % 100 == 0:
                eta0 *= 0.98
    return best_seq, best_score


# -------------------------------------------------------------
# Simulated‑annealing hill climb with larger moves
# -------------------------------------------------------------
def _hill_climb_seq(a0: np.ndarray,
                    start_time: float,
                    deadline: float,
                    current_best_score: float) -> tuple[np.ndarray, float]:
    """Greedy hill climbing with a temperature schedule and
    relatively large mass transfers.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _objective_score(a)

    # Temperature parameters (a bit hotter than the original)
    T0 = 0.04
    total_time = max(0.0, deadline - time.time())
    if total_time <= 0:
        return best_seq, best_score
    decay = (1e-6 / T0) ** (1.0 / total_time)

    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        # source: index with large *positive* gradient among non‑zero entries
        pos_idx = np.where(a > 1e-12)[0]
        if pos_idx.size == 0:
            break
        src = pos_idx[np.argmax(grad[pos_idx])]
        dst = np.argmin(grad)

        if src == dst:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or a[src] < 1e-12:
                continue

        # allow a fairly large fraction of the source mass to move
        max_frac = 0.2                     # up to 20 % of src can be transferred
        frac = rng.random() * max_frac + 1e-6
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # ensure feasibility
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        Δ = new_score - best_score
        T = T0 * (decay ** (time.time() - start_time))
        accept = False
        if Δ < -1e-12:
            accept = True
        else:
            prob = np.exp(-Δ / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_prop
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()

        # occasional random “shake” to escape plateaus
        if rng.random() < 0.001:
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s

    return best_seq, best_score


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    evaluate_sequence(seq).  The algorithm proceeds hierarchically,
    performing massive random Dirichlet sampling on small grids,
    up‑sampling the best candidate and refining it with Adam,
    exponentiated‑gradient descent and a simulated‑annealing hill‑climb.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.3       # small safety margin

    # -----------------------------------------------------------------
    # Optional global starter (if a previous construction is available)
    # -----------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                best_seq, best_score = starter, sc
                print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # -----------------------------------------------------------------
    # Hierarchical grid sizes
    # -----------------------------------------------------------------
    grid_sequence = [256, 512, 1024, 2048, 4096]
    # For the smallest grids we only evaluate random draws (no heavy optimisation)
    random_draw_limits = {256: 8000, 512: 5000, 1024: 3000}
    # For the larger grids we will run a handful of local optimiser restarts
    opt_restart_counts = {2048: 30, 4096: 25}
    # number of Adam iterations for the large‑grid restarts
    adam_iters = {2048: 2500, 4096: 5000}
    # EG and hill‑climb will be applied to the *current* best after the last
    # Adam phase on each grid

    for n in grid_sequence:
        if time.time() >= deadline:
            break

        # -------------------------------------------------------------
        # 1) Massive random search on the small grids
        # -------------------------------------------------------------
        if n in random_draw_limits:
            draws = random_draw_limits[n]
            for _ in range(draws):
                if time.time() >= deadline:
                    break
                cand = _random_dirichlet(n)
                score = _objective_score(cand)
                if score < best_score:
                    best_score, best_seq = score, cand.copy()
                    print(f"[{time.time() - start_time:.2f}s] Grid {n} random best={score:.6f}")
            # after the random sweep, up‑sample to the next grid if any
            continue

        # -------------------------------------------------------------
        # 2) Upsample the current best to the current grid size
        # -------------------------------------------------------------
        if best_seq is None:
            best_seq = np.full(n, 1.0 / n)
        else:
            best_seq = _upsample_seq(best_seq, n)

        # -------------------------------------------------------------
        # 3) Local optimiser restarts on the up‑sampled candidate
        # -------------------------------------------------------------
        restarts = opt_restart_counts.get(n, 5)
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # perturb the up‑sampled candidate a little (except for the first restart)
            if r == 0:
                cand = best_seq.copy()
            else:
                noise = _random_dirichlet(n)
                lam = 0.15                     # mixing weight for the noise
                cand = (1 - lam) * best_seq + lam * noise
                cand = cand / cand.sum()

            # Adam optimisation on the smooth‑max surrogate
            sub_deadline = deadline - 0.1
            cand_opt, score_opt = _adam_optimize_seq(
                cand,
                max_iters=adam_iters.get(n, 3000),
                lr=0.04,
                tau_start=25.0,
                tau_factor=1.3,
                tau_inc_interval=500,
                deadline=sub_deadline,
            )
            # record improvement
            if score_opt < best_score:
                best_score, best_seq = score_opt, cand_opt.copy()
                print(f"[{time.time() - start_time:.2f}s] Grid {n} Adam restart {r} new best={score_opt:.6f}")

        # -------------------------------------------------------------
        # 4) Exponentiated‑gradient descent (exact max‑convolution)
        # -------------------------------------------------------------
        if time.time() < deadline:
            eg_deadline = deadline - 0.05
            best_seq, eg_score = _eg_optimize_seq(
                best_seq,
                max_iters=4000,
                eta0=0.08,
                deadline=eg_deadline,
            )
            if eg_score < best_score:
                best_score, best_seq = eg_score, best_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Grid {n} EG improved to {eg_score:.6f}")

        # -------------------------------------------------------------
        # 5) Final hill‑climb / simulated‑annealing refinement
        # -------------------------------------------------------------
        if time.time() < deadline:
            hc_deadline = deadline - 0.05
            best_seq, hc_score = _hill_climb_seq(
                best_seq,
                start_time,
                hc_deadline,
                best_score,
            )
            if hc_score < best_score:
                best_score, best_seq = hc_score, best_seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Grid {n} hill‑climb improved to {hc_score:.6f}")

    # -----------------------------------------------------------------
    # Final normalisation & return
    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence on a modest size
        n = 1024
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    final_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")
    return [float(x) for x in best_seq]
