def _adam_descent(a0, deadline, start_time, best_score_initial, L_self, L_grad):
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0.4
    t = 0
    beta = 5.0
    beta_multiplier = 1.02
    max_beta = 200.0
    while time.time() < deadline:
        t += 1
        # compute gradient via smoothing
        conv, maxc, w, g = _smooth_gradient(a, beta, L_self, L_grad)
        # evaluate current score (unsmoothed)
        cur_score = 2.0 * len(a) * maxc / (a.sum()**2)  # sum=1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam it={t} beta={beta:.2f} score={cur_score:.6f}")
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_simplex(a)
        # update beta
        beta = min(beta * beta_multiplier, max_beta)
        # optionally decay step size
        step *= 0.9999
        # occasional jitter
        if t % 5000 == 0:
            eps_j = 1e-4 * np.random.default_rng().random(a.shape) * a
            a = np.maximum(a + eps_j, 0.0)
            a = _project_simplex(a)
            m = np.zeros_like(a); v = np.zeros_like(a); t=0  # reset moment
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The optimizer works on a relatively large vector (n = 8192) – the finer
discretisation gives a better asymptotic bound.  The search consists of:

1️⃣  Random sampling of a 4‑parameter geometric‑power‑tail family, mixed
    with a uniform background.
2️⃣  Stochastic hill‑climb on the four parameters.
3️⃣  A projected Adam‑style gradient descent on a *smoothed* version of
    the objective (log‑sum‑exp of the convolution entries).  The smoothing
    parameter β is increased during the run so that the algorithm first
    follows a smooth landscape and finally focuses on the true maximum.
4️⃣  A cheap final random “mass‑transfer” hill‑climb for fine‑tuning.

All steps respect the overall 1000 s budget; intermediate best vectors
are printed and finally returned as a plain list of floats.  The routine
uses only NumPy and the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT; returns 2·len(a)‑1 entries."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) / (sum(a)**2) (sum assumed ≈1)."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to r**i * (i+1)**p * (n-i)**q   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    # log‑space expression
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)                     # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·uniform + (1‑lam)·shape, both already sum to 1."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Random parametric sweep
# ----------------------------------------------------------------------


def _random_param_search(n: int, deadline: float, start_time: float) -> tuple[
        np.ndarray | None, float, tuple[float, float, float, float] | None]:
    """Uniform‑in‑log random sampling of (r,p,q,λ) and keep the best vector."""
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
        p = rng.uniform(-4.0, 4.0)
        q = rng.uniform(-4.0, 4.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(f"[{time.time() - start_time:.2f}s] random search r={r:.5f} "
                  f"p={p:.3f} q={q:.3f} λ={lam:.3f} score={sc:.6f}")

    return best_seq, best_score, best_params


# ----------------------------------------------------------------------
# Hill‑climb on the four shape parameters
# ----------------------------------------------------------------------


def _hillclimb_params(initial_params: tuple[float, float, float, float],
                      start_seq: np.ndarray,
                      start_score: float,
                      deadline: float,
                      start_time: float) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb on (r,p,q,λ) with exponentially shrinking perturbations."""
    cur_r, cur_p, cur_q, cur_lam = initial_params
    cur_seq = start_seq.copy()
    cur_score = start_score

    sigma_r = 0.015
    sigma_p = 0.30
    sigma_q = 0.30
    sigma_lam = 0.12

    rng = np.random.default_rng()
    n = cur_seq.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_seq = cur_seq.copy()
    best_score = cur_score

    while time.time() < deadline:
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        r_new = float(np.clip(r_new, 0.85, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape_general(n, r_new, p_new, q_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        if sc_new < cur_score:
            cur_r, cur_p, cur_q, cur_lam = r_new, p_new, q_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(f"[{time.time() - start_time:.2f}s] param hill r={r_new:.6f} "
                      f"p={p_new:.4f} q={q_new:.4f} λ={lam_new:.4f} score={sc_new:.6f}")

        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_q *= 0.9995
        sigma_lam *= 0.9995

        # occasional jitter
        if rng.random() < 0.005:
            cur_seq = np.maximum(cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq,
                                0.0)
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed objective + Adam‑style descent
# ----------------------------------------------------------------------


def _smooth_info(a: np.ndarray, beta: float,
                 L_self: int, L_grad: int) -> tuple[
        np.ndarray, float, np.ndarray, np.ndarray]:
    """
    Given a (sum≈1) compute:
        conv    = a * a (linear convolution)
        maxc    = max(conv)
        w       = soft‑max weights (log‑sum‑exp) with smoothing β
        grad    = ∇ max_β(conv) = 2 * (a ⋆ w) where ⋆ is cross‑correlation.
    Returns (conv, maxc, w, grad).
    """
    n = a.shape[0]
    conv = _conv_self_fft(a, L_self)          # length 2n‑1
    maxc = conv.max()

    # soft‑max (log‑sum‑exp) weights
    exp_weights = np.exp(beta * (conv - maxc))
    w = exp_weights / exp_weights.sum()

    # gradient = 2 * (a cross‑correlated with w)
    w_rev = w[::-1]                           # reverse for convolution
    Fa = np.fft.rfft(a, n=L_grad)
    Fw = np.fft.rfft(w_rev, n=L_grad)
    conv_aw_full = np.fft.irfft(Fa * Fw, n=L_grad)

    idx_start = 2 * n - 2                     # position where i = 0 aligns
    grad = 2.0 * conv_aw_full[idx_start - np.arange(n)]
    return conv, maxc, w, grad


def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  best_score_initial: float,
                  L_self: int,
                  L_grad: int) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent on the true objective using a smoothed
    max (log‑sum‑exp) as surrogate.  β is increased slowly so that the
    algorithm moves from a smooth landscape to the genuine max.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    m = np.zeros_like(a)          # 1st‑moment
    v = np.zeros_like(a)          # 2nd‑moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0.35                   # initial learning rate
    t = 0

    beta = 5.0                    # initial smoothing
    beta_mult = 1.02
    max_beta = 200.0

    while time.time() < deadline:
        t += 1

        conv, maxc, w, grad = _smooth_info(a, beta, L_self, L_grad)

        # true (unsmoothed) objective for bookkeeping
        cur_score = 2.0 * n * maxc / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] adam it={t:,} "
                  f"β={beta:.1f} score={cur_score:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

        # schedule updates
        beta = min(beta * beta_mult, max_beta)
        step *= 0.9999

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            eps_j = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps_j, 0.0)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny random mass‑transfer hill‑climb
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """Randomly move a tiny fraction of mass between two coordinates."""
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper to resize a possibly‑different‑length starter
# ----------------------------------------------------------------------


def _resize_to(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `vec` to length `new_n`."""
    old_n = vec.shape[0]
    if old_n == new_n:
        return vec.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    return np.interp(x_new, x_old, vec)

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes the target
    evaluation function.  Respects the 1000 s time budget and returns
    the best sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    # Problem size – 8192 offers a better asymptotic bound than 4096
    n = 8192
    L_self = _next_pow2(2 * n - 1)
    L_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0️⃣  Uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_self)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣+  Use global starter if available, resizing if necessary
    # ------------------------------------------------------------------
    try:
        starter_raw = np.asarray(height_sequence_1, dtype=float)
        if starter_raw.ndim == 1 and starter_raw.size > 0:
            starter = _resize_to(np.maximum(starter_raw, 0.0), n)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            sc = _eval_fast(starter, L_self)
            if sc < best_score:
                best_seq = starter.copy()
                best_score = sc
                print(f"[{time.time() - start_time:.2f}s] global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep (geometric × power × tail)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 150.0
    rp_seq, rp_score, rp_params = _random_param_search(n, sweep_deadline, start_time)
    if rp_seq is not None and rp_score < best_score:
        best_seq = rp_seq.copy()
        best_score = rp_score
        best_params = rp_params
        print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")
    else:
        # fallback parameters if the sweep failed to improve
        best_params = (0.97, 0.0, 0.0, 0.3)

    # ------------------------------------------------------------------
    # 2️⃣  Fine hill‑climb on the four parameters
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    hill_deadline = time.time() + min(150.0, remaining * 0.25)
    best_seq, best_score = _hillclimb_params(best_params,
                                             best_seq,
                                             best_score,
                                             hill_deadline,
                                             start_time)

    # ------------------------------------------------------------------
    # 3️⃣  Adam descent with smoothed max (log‑sum‑exp)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_deadline = time.time() + min(500.0, remaining * 0.55)
    if time.time() < adam_deadline:
        a_adam, sc_adam = _adam_descent(best_seq,
                                        adam_deadline,
                                        start_time,
                                        best_score,
                                        L_self,
                                        L_grad)
        if sc_adam < best_score:
            best_seq = a_adam.copy()
            best_score = sc_adam
            print(f"[{time.time() - start_time:.2f}s] adam final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny hill‑climb (random mass transfers)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_final, sc_final = _final_local_search(best_seq,
                                                deadline,
                                                start_time,
                                                best_score)
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Log the official evaluation (uses the provided security‑checked function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
