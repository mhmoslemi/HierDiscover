"""
propose_candidate
-----------------

A 1000 s meta‑heuristic for the step‑function problem.
The algorithm works in four stages (all respecting the time budget)

1️⃣  Broad parametric families (geometric, power, beta, exponential,
    plus two new families: sinusoidal and inverse‑square‑root).
2️⃣  Random constructions – many Dirichlet draws and random sparse
    supports (≈ √(2 n) non‑zero indices, where n=4096).
3️⃣  A long simulated‑annealing phase that repeatedly moves a random
    fraction of mass from one index to another.  The acceptance rule
    is the classic Metropolis criterion with a linearly decreasing
    temperature; the convolution (and the objective) is recomputed
    with a fast FFT at every step.
4️⃣  Final projection onto the probability simplex and return.

All operations stay in the simplex (∑aᵢ=1, aᵢ≥0); the objective that is
minimised is

        2·n·max(conv(a,a))          (the denominator sum² cancels for ∑aᵢ=1)

which is exactly what ``evaluate_sequence`` computes.
The routine is deliberately simple, uses only NumPy and the standard
library and fits well within the 1 000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)); assumes ∑seq = 1."""
    n = seq.shape[0]
    return float(2.0 * n * _conv_self(seq).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 300)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.05, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline:

        log_seq = (a - 1.0) * (np.log(idx + 1.0) + np.log(np.maximum(n - idx, 1)))
        seq = np.exp(log_seq - np.max(log_seq))
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymBeta α={a:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):

def _sweep_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 100):
        if time.time() >= deadline:

        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:

            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    for L_frac in np.linspace(0.01, 0.4, 40):
        L = int(L_frac * n)
        if L < 1:

                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score

def _simulated_annealing(a0, deadline):
    """Classic SA on the simplex – moves mass between two coordinates."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    cur_score = best_score
    rng = np.random.default_rng()
    start = time.time()
    total_time = deadline - start
    # initial temperature – a fraction of the current score
    T0 = max(1e-3, best_score * 0.1)

    max_frac = 0.30   # up to 30 % of the source mass can be moved

    it = 0
    while time.time() < deadline:
        it += 1
        # linearly cooled temperature
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / total_time)

        # propose a move
        src = rng.integers(n)
        if a[src] < 1e-12:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        frac = rng.random() * max_frac
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        # (the sum stays 1 by construction, non‑negativity is guaranteed)

        prop_score = _objective_score(a_prop)

        # Metropolis acceptance
        if prop_score < cur_score or np.exp((cur_score - prop_score) / max(T, 1e-12)) > rng.random():
            a = a_prop
            cur_score = prop_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} score={best_score:.6f}")

    return best_a, best_score

        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_sinusoidal(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_inverse_sqrt(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random constructions (Dirichlet & sparse)
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _random_sparse_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated annealing")
        best_seq, best_score = _simulated_annealing(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Annealing finished – best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
