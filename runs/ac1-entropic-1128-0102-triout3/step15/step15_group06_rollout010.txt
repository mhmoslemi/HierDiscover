"""
Hybrid random‑search + soft‑max Adam optimiser.

* Build a modest but diverse pool of candidate shapes (uniform, expo,
  power‑law, power‑ramp, asymmetric Beta, random Dirichlet, simple
  convex mixtures).  Keep the TOP_K best according to the exact bound.

* For each top candidate run a long “soft‑max Adam” phase: the hard max
  in the objective is approximated by a temperature‑controlled soft‑max.
  The gradient of the smooth surrogate is computed by a single FFT‑based
  convolution; Adam‐style moments give an adaptive step‑size.  The
  temperature is exponentially annealed from 0.1 down to 1e‑8, which lets
  the optimiser first move in a smooth landscape and then focus on the
  true maximum.

* After the main phase a quick hill‑climbing stage performs tiny random
  mass swaps and accepts only improvements – a cheap way to escape the
  final flat region.

The whole routine respects the 1000 s budget, works on a single CPU
and returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negatives caused by numerical noise.
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (normalised to sum‑1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    """Flat profile."""
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with optional flat top."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2

    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])

        # Pad / trim to exact length n
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]

    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-lam·i)."""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Core optimisation phases
# ----------------------------------------------------------------------
def _softmax_adam_phase(a0: np.ndarray,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Soft‑max Adam descent on the exact max‑convolution objective.
    Temperature is exponentially annealed from 0.1 to 1e‑8.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    T0 = 0.1
    T_end = 1e-8
    # learning‑rate schedule
    base_lr = 0.5

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ADAM iter={it:,} best={cur_score:.6f}")

        # exponential cooling of the soft‑max temperature
        frac = (time.time() - start_time) / (deadline - start_time)
        T = T_end + (T0 - T_end) * (0.5 ** (frac * 12.0))   # ~12 halvings

        # soft‑max weights (stable formulation)
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient w.r.t a (correlation with reversed a)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / math.sqrt(it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps sum = 1, non‑negative)
        a = _project_simplex(a)

        # occasional tiny random jitter to avoid stagnation
        if it % 5000 == 0:
            noise = np.random.default_rng().normal(scale=1e-4, size=n)
            a += noise
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best, best_score


def _hill_climbing_refine(a0: np.ndarray,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: move a tiny amount of mass between
    two random positions and keep the move if it improves the bound.
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    rng = np.random.default_rng()

    attempts = 0
    while time.time() < deadline and attempts < 2000:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 1 % of the mass from i to j (but not more than a[i])
        delta = min(a[i], 0.01) * rng.random()
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        # evaluate
        sc = _objective(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb improvement → {sc:.6f}")
        else:
            # revert
            a[i] += delta
            a[j] -= delta

    return best, best_score

    # --------------------------------------------------------------

        while time.time() < phase_end:
            vec = pool_vecs[idx % len(pool_vecs)]
            refined, sc = _softmax_adam_phase(vec, phase_end, start)
            if sc < best_score:
                best_score, best_vec = sc, refined.copy()
                print(f"[{time.time() - start:.2f}s] ADAM improvement → {sc:.6f}")
            idx += 1

    # --------------------------------------------------------------
    # 3) Final cheap hill‑climbing refinement (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        refined, sc = _hill_climbing_refine(best_vec, deadline, start)
        if sc < best_score:
            best_score, best_vec = sc, refined.copy()
            print(f"[{time.time() - start:.2f}s] Hill‑climb final improvement → {sc:.6f}")

    # --------------------------------------------------------------
    # 4) Normalise and return
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    # Convert to plain Python list as required by the evaluator.
    return [float(v) for v in best_vec]
