def _softmax_weights(conv, alpha):
    # compute stable softmax of alpha * conv
    z = alpha * conv
    max_z = np.max(z)
    expz = np.exp(z - max_z)
    return expz / np.sum(expz)

def _build_lp_constraints(f, M):
   # f: numpy array length n
   n = f.shape[0]
   m = 2*n - 1
   rows = []
   cols = []
   data = []
   for k in range(m):
      lo = max(0, k - (n - 1))
      hi = min(n - 1, k)
      # j runs from lo to hi (inclusive)
      # for each j, coefficient is f[k - j]
      for j in range(lo, hi+1):
          rows.append(k)
          cols.append(j)
          data.append(f[k - j])
   A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))
   b = np.full(m, M)
   return A, b

def _softmax_gradient_descent(a, n, L_fft, start_time, deadline, best_seq, best_score):
   iteration = 0
   while time.time() < deadline:
       iteration +=1
       conv = _conv_self(a, L_fft)
       cur_score = 2.0 * n * conv.max()
       if cur_score < best_score:
           best_score, best_seq = cur_score, a.copy()
           print(f"[{time.time() - start_time:.2f}s] Softgrad it={iteration} score={cur_score:.6f}")

       # schedule for alpha: increase from low to high
       alpha = 1e3 * min(1.0, iteration / 200000)  # start small then go up to 1000? maybe.

       # compute softmax weights p
       z = alpha * conv
       max_z = np.max(z)
       exp_z = np.exp(z - max_z)
       p = exp_z / np.sum(exp_z)

       # compute gradient
       grad = _gradient_softmax(a, p)  # we will implement _gradient_softmax as described

       # step size
       eta = 0.6 / np.sqrt(iteration)

       # multiplicative update
       factor = np.exp(-eta * grad)
       factor = np.clip(factor, 1e-12, 1e12)

"""
Step‑function search – hybrid of random Dirichlet starts,
soft‑max (entropy‑regularised) mirror descent,
and a tiny active‑set linear‑program to enlarge the feasible mass.
The algorithm respects a 1000 s budget, keeps the best sequence
found and finishes with a short uniform‑mix sweep.
"""

import time
import math
import numpy as np

# optional: a cheap LP fallback (used only if SciPy is available)
try:
    import scipy.optimize
    import scipy.sparse
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)).  a is already normalised."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _gradient_softmax(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of 2·n·⟨p, a∗a⟩ where p is a probability vector on the self‑convolution.
    It equals 2 * (p ⋆ a)[::-1] restricted to the original n indices.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(p) + n - 1)
    P_fft = np.fft.rfft(p, n=L)
    A_fft = np.fft.rfft(a_rev, n=L)
    conv_full = np.fft.irfft(P_fft * A_fft, n=L)
    offset = len(p) - 1                     # index that aligns with a[0]
    grad = 2.0 * conv_full[offset:offset + n]   # shape (n,)
    return grad


# ----------------------------------------------------------------------
# 0️⃣  Random Dirichlet draws (wide concentration range)
# ----------------------------------------------------------------------
def _random_dirichlet_search(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap random search – draws Dirichlet samples with many concentration values."""
    while time.time() < deadline:
        # pick concentration log‑uniform in [log 0.02, log 20]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = math.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s <= 0.0:
            continue
        a = sample / s
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3g} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 1️⃣  Soft‑max (entropy‑regularised) mirror‑descent
# ----------------------------------------------------------------------
def _softmax_descent(a0: np.ndarray,
                     n: int,
                     L_fft: int,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the gradient of a
    soft‑max approximation of ``max(conv)``.
    The temperature (inverse softness) α grows linearly so that the
    method initially sees a smooth landscape and later focuses on the true
    max‑peak.  An Ada‑step η≈1/√t is used together with a multiplicative
    update and a tiny Simulated‑Annealing acceptance rule.
    """
    a = a0.copy()
    it = 0
    # parameters for the schedule
    alpha_start = 1e2          # relatively smooth at the beginning
    alpha_end = 1e6           # very sharp at the end (≈ true max)
    total_iters_est = 1e6     # rough scale for α schedule (does not need to be exact)

    while time.time() < deadline:
        it += 1
        # current convolution and score
        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Soft‑grad it={it:,} score={cur_score:.6f}")

        # ---- schedule for the soft‑max temperature α --------------------
        # grow α logarithmically from start to end
        progress = min(1.0, it / total_iters_est)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        # ---- soft‑max weights over the convolution entries ----------------
        # stable implementation: shift by the maximum
        z = alpha * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)               # all ≤ 1, avoids overflow
        p = exp_z / np.sum(exp_z)               # p sums to 1

        # ---- gradient of the smoothed objective -------------------------
        grad = _gradient_softmax(a, p)           # shape (n,)

        # ---- step size (AdaGrad‑like) -----------------------------------
        eta = 0.7 / math.sqrt(it)

        # multiplicative update, kept in a safe numeric range
        factor = np.exp(-eta * grad)
        factor = np.clip(factor, 1e-12, 1e12)

        a_new = a * factor
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s

        # acceptance test – always accept an improvement, otherwise
        # occasional SA‑style acceptance to escape flat plateaus
        conv_new = _conv_self(a_new, L_fft)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            a = a_new
        else:
            # temperature decays linearly with remaining time
            T0 = 5e-4
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = T0 * (remaining / total)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                a = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Tiny active‑set LP to enlarge the L₁‑mass while keeping the peak
# ----------------------------------------------------------------------
def _lp_enlarge(a: np.ndarray,
                n: int,
                L_fft: int,
                start_time: float,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves a small linear programme:
        max   Σ b_j
        s.t.  b_j ≥ 0,
              (a * b)_k ≤ M   for all *active* shifts k,
    where M = max(a * a).  The solution b typically has a larger
    L₁‑norm; after normalising it we convex‑combine it with the current a.
    Only the constraints that are (almost) tight are kept – this keeps
    the LP tiny (≈ tens of rows) and fast.
    """
    if not _SCIPY_AVAILABLE:
        # No LP capability – return unchanged
        return best_seq, best_score

    a_curr = a.copy()
    while time.time() < deadline:
        conv = _conv_self(a_curr, L_fft)
        M = conv.max()

        # active set – shifts whose value is within a tiny tolerance of the peak
        active_tol = max(1e-12, 1e-9 * M)   # relative tolerance
        active = np.where(conv >= M - active_tol)[0]
        if active.size == 0:
            break

        # Build a sparse matrix A_ub where each row corresponds to an active shift.
        rows, cols, data = [], [], []
        for row, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # b_j appears with coefficient a_{k-j}
            # j runs from lo … hi
            # vectorised version of the loop (still fast enough for small active sets)
            idx = np.arange(lo, hi + 1, dtype=int)
            rows.extend([row] * idx.size)
            cols.extend(idx.tolist())
            data.extend(a_curr[k - idx].tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        b_ub = np.full(len(active), M, dtype=float)

        # maximise Σ b_j  → minimise -Σ b_j
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None)] * n

        # solve the LP (highs is fast and can respect a time limit)
        time_left = deadline - time.time()
        if time_left <= 0.0:
            break
        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                         bounds=bounds,
                                         method='highs',
                                         options={'time_limit': time_left})
        except Exception:  # pragma: no cover
            break

        if not res.success:
            # Something went wrong – abort the LP phase
            print(f"[{time.time() - start_time:.2f}s] LP failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # No L₁‑gain → stop looping
            break

        # Normalise the candidate so that it lives on the same simplex as a_curr
        g = b / sum_b

        # Try a few mixing coefficients λ∈[0,1] and keep the best improvement
        improved = False
        best_local_seq = a_curr
        best_local_score = best_score
        for lam in np.linspace(0.0, 1.0, 11):          # 0,0.1,…,1
            seq = (1.0 - lam) * a_curr + lam * g
            seq = np.maximum(seq, 0.0)
            seq /= seq.sum()
            sc = _objective_score(seq, n)
            if sc < best_local_score - 1e-9:
                best_local_score = sc
                best_local_seq = seq.copy()
                improved = True
                print(f"[{time.time() - start_time:.2f}s] LP mix λ={lam:.2f} → score={sc:.6f}")

        if not improved:
            # mixing did not help – stop the LP loop
            break

        # accept the improved mixture
        a_curr = best_local_seq
        best_seq, best_score = best_local_seq, best_local_score

    return best_seq, best_score

# 3️⃣  Final fine‑grained mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Scan λ∈[0,1] for seq = λ·a + (1‑λ)·uniform and keep the best."""
    uniform = np.full(n, 1.0 / n)
    lam_grid = np.linspace(0.0, 1.0, 41)   # coarse sweep
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        seq = np.maximum(seq, 0.0)
        seq_sum = seq.sum()
        if seq_sum == 0.0:
            continue
        seq /= seq_sum
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Uniform mix λ={lam:.3f} → score={sc:.6f}")
    return best_seq, best_score

        print(f"[{time.time() - start_time:.2f}s] LP enlargement phase")
        best_seq, best_score = _lp_enlarge(
            best_seq, n, L_fft, start_time, deadline, best_seq, best_score, rng
        )
        print(f"[{time.time() - start_time:.2f}s] After LP best score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final fine‑grained mixing with uniform
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Mixing with uniform")
        best_seq, best_score = _mix_with_uniform(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Normalise one final time and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
