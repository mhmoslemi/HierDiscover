"""
Optimisation of a non‑negative step function to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three main phases:

1.  A quick, massive random Dirichlet initialisation.  Each draw is
    refined by a few iterations of a *soft‑max* gradient descent that
    smoothly turns the average‑conv‑flattening problem into a true
    max‑conv minimisation.

2.  A long soft‑max gradient descent starting from the best candidate.
    The temperature is increased from a low value (almost averaging) to
    a high value (focussing on the current maximum).  The gradient is
    obtained analytically via FFTs and the iterate is projected onto the
    probability simplex after every step.

3.  A final hill‑climb that moves tiny amounts of mass from indices with
    large sub‑gradient to those with small sub‑gradient, followed by a
    symmetrisation test.

All phases respect a total wall‑clock budget of 1000 s.  The best
sequence found is returned as a plain Python list ready for the
``evaluate_sequence`` checker.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolutions, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Phase 1 – quick random Dirichlet draws + a few soft‑max steps
# ----------------------------------------------------------------------
def _quick_softmax_descent(a0: np.ndarray,
                           deadline: float,
                           max_iters: int = 200) -> tuple[np.ndarray, float]:
    """Run a very short soft‑max descent (used for cheap refinement)."""
    a = a0.copy()
    n = a.shape[0]
    step_scale = 0.5
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        max_val = conv.max()
        # temperature goes from 0.5 → 3.0 linearly
        T = 0.5 + 2.5 * it / max_iters
        w = np.exp(T * (conv - max_val))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws
        grad = 2.0 * _conv_fft(w, a[::-1])[n - 1:n - 1 + n]
        gnorm = np.linalg.norm(grad) + 1e-12
        step = step_scale / gnorm
        a = _project_simplex(np.maximum(a - step * grad, 0.0))
        step_scale *= 0.9995
    return a, _objective_score(a)


# ----------------------------------------------------------------------
# Phase 2 – long soft‑max gradient descent
# ----------------------------------------------------------------------
def _softmax_gradient_descent(a0: np.ndarray,
                              deadline: float,
                              best_seq: np.ndarray,
                              best_score: float) -> tuple[np.ndarray, float]:
    """
    Runs the main soft‑max descent until the deadline.
    Temperature T is increased from ≈0.5 to ≈10, turning the
    objective from a smooth average into the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    step_scale = 0.5
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_val = conv.max()
        # Linear schedule based on elapsed fraction of the whole run
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = 0.5 + (10.0 - 0.5) * min(elapsed / total, 1.0)
        w = np.exp(T * (conv - max_val))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws
        grad = 2.0 * _conv_fft(w, a[::-1])[n - 1:n - 1 + n]
        gnorm = np.linalg.norm(grad) + 1e-12
        step = step_scale / gnorm
        a = _project_simplex(np.maximum(a - step * grad, 0.0))

        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Soft‑max it={it} T={T:.2f} "
                  f"score={cur_score:.6f}")

        # gentle decay of the step scale
        step_scale *= 0.9999
        if step_scale < 1e-6:
            step_scale = 1e-6

        # early stop if we already beat the target
        if best_score < 1.503:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase 3 – final hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Directed hill‑climb that moves a tiny fraction of mass each step."""
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)   # ≤0.4 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                # jitter to escape a plateau
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the evaluation function.
    The strategy combines massive random Dirichlet sampling, a soft‑max
    gradient descent that gradually focuses on the worst convolution entry,
    and a final hill‑climb.  The whole search respects the 1000 s budget.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # safety margin

    # --------------------------------------------------------------
    # Problem size – medium enough for fast FFTs but still flexible.
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1, if supplied by the runner).
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – massive random Dirichlet draws + cheap refinement
    # --------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.20
    rng = np.random.default_rng()
    while time.time() < phase1_deadline:
        conc = rng.uniform(0.05, 5.0)                 # concentration parameter
        a = rng.gamma(shape=conc, scale=1.0, size=n)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        # a few quick soft‑max steps (no more than 150 ms each)
        a_ref, sc = _quick_softmax_descent(a, phase1_deadline, max_iters=150)
        if sc < best_score:
            best_score, best_seq = sc, a_ref.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={conc:.3f} refined score={sc:.6f}")
        if best_score < 1.503:       # achieved target ⇒ abort early
            break

    # --------------------------------------------------------------
    # Phase 2 – long soft‑max gradient descent from the current best
    # --------------------------------------------------------------
    phase2_deadline = start_time + budget_s * 0.75
    if time.time() < phase2_deadline and best_score >= 1.503:
        print(f"[{time.time() - start_time:.2f}s] Starting long soft‑max descent")
        best_seq, best_score = _softmax_gradient_descent(
            best_seq, phase2_deadline, best_seq, best_score)

    # --------------------------------------------------------------
    # Phase 3 – final hill‑climb and optional symmetrisation
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)

        # try forcing symmetry (a_i = a_{n-1-i}) – sometimes helps
        sym_a = (best_seq + best_seq[::-1]) * 0.5
        sym_a = _project_simplex(sym_a)    # keep it a valid probability vector
        sym_score = _objective_score(sym_a)
        if sym_score < best_score:
            best_seq, best_score = sym_a, sym_score
            print(f"[{time.time() - start_time:.2f}s] Symmetrised improved score={sym_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
