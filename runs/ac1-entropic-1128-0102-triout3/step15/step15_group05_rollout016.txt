        A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')

        c = -np.ones(n, dtype=np.float64)

            break

        if sum_g <= 1e-12:
            print("g sum zero; breaking")
            break
        # scale g to target sum sqrt(2n)
        scaling = total_sum_target / sum_g
        g_scaled = g * scaling
        # mix
        f_new = (1 - t_mix) * f + t_mix * g_scaled
        M_new = compute_M(f_new)
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
            print(f"[{time.time() - start:.2f}s] LP improvement M={M_new:.6f}")
        # accept partly if improved else keep old f
        if M_new < M:
            f = f_new
            M = M_new
        else:
            # maybe shrink mix factor
            t_mix *= 0.8
            if t_mix < 0.01:
                t_mix = 0.01
        # if improvement minimal, break
        if abs(M_new - M) < 1e-6:
            break
    # Now enlarge to larger n for refinement
    # Determine target size based on remaining time
    remaining = deadline - time.time()
    # choose final n = 4096 if time sufficient else keep n_small
    target_n = 4096 if remaining > 200 else n_small
    if target_n > n_small:
        # upsample best_f to target_n via linear interpolation
        x_small = np.linspace(0, 1, n_small)
        x_big = np.linspace(0, 1, target_n)
        a_big = np.interp(x_big, x_small, best_f)
        # renormalize sum to sqrt(2*target_n)
        total_target = np.sqrt(2 * target_n)
        a_big = a_big / a_big.sum() * total_target
    else:
        a_big = best_f.copy()
    # finalize with subgradient descent on a_big
    L_fft_big = _next_pow2(2 * target_n - 1)
    # run subgradient descent for remaining time
    sub_deadline = time.time() + remaining * 0.8
    best_vec, best_M2 = _subgradient_descent_multi(a_big, sub_deadline, start, best_M)
    # final local mass transfer for a few seconds
    final_deadline = deadline
    best_vec2, best_M3 = _final_local_search(best_vec, final_deadline, start, best_M2)
    # Normalize to sum 1 for final output
    final_seq = np.maximum(best_vec2, 0.0)
    final_seq = final_seq / final_seq.sum()
    # Evaluate
    try:
        final_score = evaluate_sequence(final_seq.tolist())
        print(f"[{time.time() - start:.2f}s] final evaluated score: {final_score:.6f}")
    except Exception as e:
        print("Evaluation error:", e)
    return [float(v) for v in final_seq]

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a, L=None):
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2*n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[:2*n-1]
    return conv

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _eval_fast(a, L_fft):
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

def _compute_subgrad_multi(a, conv, tol=1e-12):
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    return g

def _subgradient_descent_multi(a0, deadline, start_time, best_score_initial, active_tol=1e-12):
    n = a0.shape[0]
    L_fft = _next_pow2(2*n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    beta = 0.9
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=active_tol)
        v = beta * v + (1.0 - beta) * g
        step = step0 / (it ** 0.6)
        a_candidate = a - step * v
        a_candidate = _project_simplex(a_candidate)
        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            v = np.zeros_like(a)
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} step={step:.5e} score={cur_score:.6f}")
        else:
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            v = np.zeros_like(a)
    return best_a, best_score

def _final_local_search(a, deadline, start_time, best_score_initial):
    n = a.shape[0]
    L_fft = _next_pow2(2*n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            if rng.random() < 0.001:
                a = a_prop
    return best_a, best_sc

def _lp_improve(f, M, t_mix, deadline, start_time):
    """Run a single LP improvement iteration and return new f, M, maybe improved flag."""
    n = f.shape[0]
    # Build convolution matrix using scipy.sparse.diags
    offsets = -np.arange(n)  # 0, -1, -2, ...
    # Build list of diagonal data
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2*n - 1, n), format='csr')
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    b = np.full(2*n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    # limit time for linprog if available
    time_left = max(0.5, deadline - time.time())
    options = {'presolve': True}

"""
A fast, multi‑stage optimiser for the step‑function objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works as follows:

1️⃣  Initialise from the global best sequence (if the variable
    ``height_sequence_1`` exists) or from a uniform vector.  The
    vector is normalised to have sum = sqrt(2·n); in this scale the
    objective equals the maximal self‑convolution value M.

2️⃣  Perform a few linear‑programming “boost” steps.
    For a fixed shape f we solve

        maximise   Σ g_j
        subject to  g ≥ 0,  f * g ≤ M                (elementwise)

    The LP is built from the convolution matrix (a Toeplitz
    structure).  The optimiser returns a vector g with a larger total
    mass while respecting the same bound M.  After scaling g to the
    required sum we mix it back into f (f ← (1‑t)f + t·g).  Each
    successful LP iteration strictly reduces M.

3️⃣  Refine the result with a projected sub‑gradient descent that
    works directly on the true objective.  The sub‑gradient is the
    sum of gradients of all convolution entries that are within a tiny
    tolerance of the current maximum, providing a stable descent
    direction.

4️⃣  Finish with a cheap random “mass‑transfer’’ hill‑climb that moves
    a tiny amount of probability mass between two coordinates.

5️⃣  Normalise to sum 1 (the objective is scale‑invariant) and return
    the best sequence found before the 1000 s budget expires.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a, aggregated over
    all shifts whose value is within `tol` of the maximum.
    """
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals

        g[i_vals] += a[j_vals]

        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (with momentum & jitter)
# ----------------------------------------------------------------------


def _subgradient_descent_multi(a0: np.ndarray,
                               deadline: float,
                               start_time: float,
                               best_score_initial: float,
                               active_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the true objective,
    using the full active set of near‑maximal convolution entries.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    beta = 0.9
    v = np.zeros_like(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=active_tol)
        v = beta * v + (1.0 - beta) * g
        step = step0 / (it ** 0.6)
        a_candidate = a - step * v
        a_candidate = _project_simplex(a_candidate)
        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            v = np.zeros_like(a)  # reset momentum after improvement
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"step={step:.5e} score={cur_score:.6f}")
        else:
            prob = np.exp(-(cur_score - best_score) * 12.0 / (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)
            v = np.zeros_like(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final cheap random “mass‑transfer” hill‑climb
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two
    coordinates, keeping the vector on the simplex.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            if rng.random() < 0.001:
                a = a_prop
    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming “boost” step
# ----------------------------------------------------------------------


def _lp_improve(f: np.ndarray,
                M: float,
                t_mix: float,
                deadline: float,
                start_time: float) -> tuple[np.ndarray, float, bool]:
    """
    One LP‑boost iteration:
        maximise Σ g_j   s.t.   g≥0 ,  f*g ≤ M .
    The solution g is scaled to the same L1‑norm as f and mixed back:
        f ← (1‑t)·f + t·g .
    Returns the new f, the new maximal convolution M, and a flag
    indicating whether an improvement was obtained.
    """
    n = f.shape[0]

    # Build the convolution matrix A (2n‑1) × n using constant diagonals.
    offsets = -np.arange(n)                     # 0, -1, -2, …
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    c = -np.ones(n, dtype=np.float64)           # maximise Σ g_j  → minimise -Σ g_j
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # Solve the LP (HiGHS interior‑point method, robust for moderate size)
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                         method="highs", options={"presolve": True})
    except Exception as exc:                    # fallback on unexpected failures
        print(f"Linprog exception: {exc}")
        return f, M, False

    if res.status != 0:
        print(f"LP status {res.status}: {res.message}")
        return f, M, False

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return f, M, False

    target_sum = np.sqrt(2.0 * n)
    scaling = target_sum / sum_g
    g_scaled = g * scaling

    f_new = (1.0 - t_mix) * f + t_mix * g_scaled
    L = _next_pow2(2 * n - 1)
    M_new = _conv_self_fft(f_new, L).max()
    improved = M_new < M - 1e-9
    return f_new, M_new, improved


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
