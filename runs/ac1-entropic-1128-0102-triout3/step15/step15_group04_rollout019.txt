"""
Enhanced search for the step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm uses three complementary ideas:

1️⃣  **Coarse LP‑based shape (small n)** – Starting from a uniform vector,
    we solve a tiny linear programme (maximise total mass under the
    constraint f∗b ≤ M, where M is the current L∞‑norm of f∗f).  Repeating a
    few times yields a highly non‑uniform shape that already improves the
    objective dramatically for a modest size (n≈256).

2️⃣  **Upsampling + Adam optimisation** – The small‑scale shape is
    interpolated to a large dimension (n≈8192).  Then we run an Adam‑type
    optimiser on a smooth approximation of the L∞‑objective
    (soft‑max of the self‑convolution) with a continuation schedule of the
    temperature β.  This fine‑tunes the whole vector while respecting the
    simplex constraint.

3️⃣  **Projected sub‑gradient polish & tiny hill‑climb** – The final
    minutes are spent on exact sub‑gradient descent (the same routine as
    the reference) and a light random mass‑transfer hill‑climb to catch
    any remaining improvements.

All phases respect the 1000 s budget, use only NumPy (and SciPy for the
small LP), and finally return a plain Python list of floats (sum≈1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional import – SciPy provides the LP solver for the small‑scale phase.
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """Full linear self‑convolution a*a via FFT."""
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:  # numerically unlikely but safe
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact objective (sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum≈1."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric × power) and mixture with uniform
# ----------------------------------------------------------------------


def _make_shape(n: int, r: float, p: float) -> np.ndarray:
    """Geometric × power shape, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    idx = i + 1.0
    # log‑space for stability
    log_shape = i * np.log(r) + p * np.log(idx)
    log_shape -= np.max(log_shape)          # avoid overflow
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Lam∈[0,1] mixes uniform and shape – result already sums to 1."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Soft‑max (smooth max) utilities for the gradient step
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights p_k = exp(beta*conv_k)/Σexp(.) (stable)."""
    shift = conv.max()
    exps = np.exp(beta * (conv - shift))
    return exps / exps.sum()


def _grad_softmax(a: np.ndarray, p: np.ndarray, L_grad: int) -> np.ndarray:
    """
    Gradient of the smooth‑max objective:
        g_i = 2 * Σ_j a_j * p_{i+j}
    Computed via FFT as a cross‑correlation.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    Af = np.fft.rfft(a, n=L_grad)
    Pf = np.fft.rfft(p_rev, n=L_grad)
    conv_full = np.fft.irfft(Af * Pf, n=L_grad)
    # keep the first (2n‑1 + n‑1) = 3n‑2 entries
    conv_full = conv_full[: (3 * n - 2)]
    start = n - 1
    g = conv_full[start:start + n][::-1]
    return 2.0 * g


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (same as reference)
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_score_initial: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on   2·n·max(conv(a,a))."""
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial
    rng = np.random.default_rng()
    step0 = 0.16
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_self_fft(a, L_fft)
        max_val = conv.max()
        k = int(np.argmax(conv))

        # sub‑gradient for shift k (exact)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]          # contribution a_j
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]   # symmetric part

        # diminishing step
        step = step0 / (it ** 0.6)
        a_candidate = a - step * g
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            best_a = _project_simplex(best_a)   # safety
            a = a_candidate
            print(f"[{time.time() - start_time:.2f}s] subgrad it={it:,} "
                  f"score={cur_score:.6f}")

        else:
            # occasional uphill moves (simulated‑annealing flavour)
            prob = np.exp(-(cur_score - best_score) * 12.0 /
                          (np.sqrt(it) + 1.0))
            if rng.random() < prob:
                a = a_candidate

        # jitter every few thousand iterations
        if it % 2500 == 0:
            eps = 1e-4 * rng.random(n) * a
            a = np.maximum(a + eps, 0.0)
            a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Small‑scale linear‑programming step (n_small ≤ 512)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return dense matrix A such that A @ b = f * b (full linear conv)."""
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        A[k, j] = f[k - j]
    return A


def _lp_improve(f: np.ndarray, rng: np.random.Generator,
                max_iter: int = 15) -> np.ndarray:
    """
    Perform a few LP iterations on a small vector f.
    Each iteration solves:
        maximise Σ b_j
        s.t.    f * b ≤ M   (M = max(f∗f))
                b ≥ 0
    The optimiser (HiGHS) returns a basic feasible point that is used
    to move f a little towards the LP solution.
    """
    if not _SCIPY_AVAILABLE:
        return f  # fall back to unchanged shape

    n = f.shape[0]
    for it in range(max_iter):
        conv = np.convolve(f, f)               # full linear conv
        M = conv.max()
        A = _build_conv_matrix(f)              # (2n‑1) × n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)      # maximise Σb → minimise –Σb
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break
        b = res.x
        s = b.sum()
        if s <= 0.0:
            break
        g = b / s                              # normalised LP solution
        # convex combination: move a little towards g
        t = 0.2 / (it + 1)                     # decreasing step size
        f = (1.0 - t) * f + t * g
        f = np.maximum(f, 0.0)
        f = f / f.sum()                        # keep exactly on simplex
    return f

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine mixes a tiny LP on a small problem, up‑samples,
    refines with Adam on a smooth max, then finishes with exact
    sub‑gradient descent and a light hill‑climb.  The whole process respects
    the 1000 s time budget and returns the best sequence as a Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # small safety margin

    # Phase 0 – small‑scale LP (n_small = 256)
    # ------------------------------------------------------------------
    n_small = 256
    a_small = np.full(n_small, 1.0 / n_small, dtype=np.float64)   # uniform start
    a_small = _lp_improve(a_small, rng, max_iter=20)

    # Phase 1 – optional parametric warm‑start on the small scale
    # (quick random sweep – 20 s)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 20.0

    p_grid = np.linspace(-3.0, 2.0, 21)
    lam_grid = np.linspace(0.0, 1.0, 7)

    best_score_small = _eval_fast(best_small,
                                  _next_pow2(2 * n_small - 1))

    while time.time() < sweep_deadline:
        r = rng.choice(r_grid)
        p = rng.choice(p_grid)
        lam = rng.choice(lam_grid)
        shape = _make_shape(n_small, r, p)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, _next_pow2(2 * n_small - 1))
        if sc < best_score_small:
            best_score_small = sc
            best_small = cand.copy()
            print(f"[{time.time() - start_time:.2f}s] small sweep r={r:.5f} "

    # ------------------------------------------------------------------

    best_score = _eval_fast(best_seq, L_fft)

    # ------------------------------------------------------------------

    # reserve ~5 min for the final sub‑gradient phase
    subgrad_reserve = 300.0
    cur_score = best_score
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Normalise finally (sum → 1) and return a plain Python list.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n_large, 1.0 / n_large)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
