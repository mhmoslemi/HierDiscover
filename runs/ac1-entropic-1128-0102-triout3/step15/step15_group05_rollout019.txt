def golden_section_search(f, a, b, tol=1e-8):
    gr = (np.sqrt(5) + 1) / 2
    c = b - (b - a) / gr
    d = a + (b - a) / gr
    while abs(b - a) > tol:
        if f(c) < f(d):
            b = d
            d = c
            c = b - (b - a) / gr
        else:
            a = c
            c = d
            d = a + (b - a) / gr
    return (a + b) / 2
```

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2

    n = 4096  # can maybe adjust based on time.

    L_fft = _next_pow2(2*n - 1)

    # baseline uniform
    uniform = np.full(n, 1.0/n)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(f"Uniform baseline score={best_score}")

    # step 1: exponential shape search
    # define objective in log domain
    def obj_logr(logr):
        r = np.exp(logr)
        shape = _make_shape_general(n, r, 0.0, 0.0)
        return _eval_fast(shape, L_fft)

    # golden-section
    a_log = np.log(0.85)
    b_log = np.log(0.999999)
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    c_log = b_log - phi*(b_log - a_log)
    d_log = a_log + phi*(b_log - a_log)
    fc = obj_logr(c_log)
    fd = obj_logr(d_log)
    it = 0
    while time.time() < deadline and (b_log - a_log) > 1e-10 and it < 60:
        it += 1
        if fc < fd:
            b_log = d_log
            d_log = c_log
            fd = fc
            c_log = b_log - phi*(b_log - a_log)
            fc = obj_logr(c_log)
        else:
            a_log = c_log
            c_log = d_log
            fc = fd
            d_log = a_log + phi*(b_log - a_log)
            fd = obj_logr(d_log)
    best_r_log = (a_log + b_log)/2
    best_r = np.exp(best_r_log)
    best_shape = _make_shape_general(n, best_r, 0.0, 0.0)
    best_shape_score = _eval_fast(best_shape, L_fft)
    if best_shape_score < best_score:
        best_seq = best_shape
        best_score = best_shape_score
    print(f"Exponential search r={best_r:.6f} score={best_shape_score:.6f}")

    # step 2: simple hillclimb on p,q around 0
    cur_p = 0.0
    cur_q = 0.0
    cur_r = best_r
    cur_seq = best_seq
    cur_score = best_score
    sigma_p = 0.5
    sigma_q = 0.5
    hill_deadline = time.time() + min(180.0, max(0.0, deadline - time.time()))
    while time.time() < hill_deadline:
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        # keep r unchanged
        shape_new = _make_shape_general(n, cur_r, p_new, q_new)
        sc_new = _eval_fast(shape_new, L_fft)
        if sc_new < cur_score:
            cur_p, cur_q = p_new, q_new
            cur_seq, cur_score = shape_new, sc_new
            if sc_new < best_score:
                best_seq, best_score = shape_new.copy(), sc_new
                print(f"p,q hill: p={p_new:.4f} q={q_new:.4f} score={sc_new:.6f}")
        sigma_p *= 0.999
        sigma_q *= 0.999
        # occasional jitter:
        if rng.random() < 0.005:
            # small noise
            shape_new = cur_seq * (1 + 1e-4 * rng.standard_normal(n))
            shape_new = np.maximum(shape_new, 0.0)
            shape_new = shape_new / shape_new.sum()
            sc_new = _eval_fast(shape_new, L_fft)
            if sc_new < cur_score:
                cur_seq, cur_score = shape_new, sc_new
                if sc_new < best_score:
                    best_seq, best_score = shape_new.copy(), sc_new
                    print(f"jitter improved score={sc_new:.6f}")

    # step 3: optional subgradient descent if time left
    if time.time() < deadline - 10 and best_score > 1.503:
        # run Adam subgradient descent for remaining time
        a_sub, sc_sub = _adam_subgradient_descent(best_seq, deadline, start_time, best_score)
        if sc_sub < best_score:
            best_seq, best_score = a_sub.copy(), sc_sub
            print(f"Adam subgrad final score={best_score:.6f}")

    # final small random mass-transfer hill climb
    # similar to earlier final local search but simpler
    if time.time() < deadline - 5:
        # allocate some time
        final_deadline = deadline
        n = best_seq.shape[0]
        L_fft = _next_pow2(2*n - 1)
        while time.time() < final_deadline:
            src = rng.integers(n)
            dst = rng.integers(n)
            if src == dst or best_seq[src] < 1e-15:
                continue
            eps = best_seq[src] * rng.random() * 5e-4
            proposal = best_seq.copy()
            proposal[src] -= eps
            proposal[dst] += eps
            sc = _eval_fast(proposal, L_fft)
            if sc < best_score:
                best_seq, best_score = proposal.copy(), sc
                print(f"mass transfer improved score={sc:.6f}")

    # final normalization
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform

    # final evaluation
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"evaluate_sequence => {final_val:.6f}")
    except Exception as e:
        print(f"Evaluation error: {e}")

    print(f"Finished – best score ≈ {best_score:.6f}")

    return [float(v) for v in best_seq]

def _adam_subgradient_descent(a0, deadline, start_time, best_score_initial):
    n = a0.shape[0]
    L_fft = _next_pow2(2*n - 1)
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    step_size = 0.05  # adjust
    it = 0
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=1e-12)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = step_size / (np.sqrt(v_hat) + epsilon)
        a_candidate = a - step * m_hat
        a_candidate = _project_simplex(a_candidate)
        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            # optionally reset m,v
            m = np.zeros_like(a)
            v = np.zeros_like(a)
        else:
            # occasional acceptance
            if rng.random() < 0.001:
                a = a_candidate
        # jitter every few thousand
        if it % 2000 == 0:
            eps = 1e-4 * rng.standard_normal(n) * a
            a += eps
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
    return best_a, best_score
```

"""
Propose a sequence of non‑negative heights that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The method is a lightweight multi‑stage optimiser:

1️⃣  A very wide random search over the 4‑parameter family
    a_i ∝ r**i·(i+1)**p·(n‑i)**q  mixed with a uniform background λ.
    The ranges for the parameters are deliberately generous
    (r∈[0.5,0.999999], p,q∈[‑6,6], λ∈[0,1]) so that the search
    can discover shapes that the previous optimiser missed.

2️⃣  Stochastic hill‑climb on (r,p,q,λ) starting from the best
    random candidate.  The perturbation scales decay slowly,
    enabling a fine‑grained local optimisation.

3️⃣  Projected Adam‑style sub‑gradient descent directly on the true
    objective (max‑convolution).  The gradient is the sum of the
    sub‑gradients of all shifts whose value is within a tiny tolerance
    of the current maximum, which yields a stable direction.  Adam’s
    adaptive step‑size and momentum help to escape shallow plateaux.

4️⃣  A final cheap “mass‑transfer” hill‑climb that moves a tiny amount
    of probability mass between two coordinates, keeping the vector on
    the probability simplex.

All stages respect the overall 1000 s budget, and the best vector
found is returned as a plain list of floats.  The routine also prints
progress information and verifies the final result with the supplied
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:  # should never happen – guard anyway
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation
# ----------------------------------------------------------------------


def _make_shape_general(
    n: int, r: float, p: float, q: float
) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to r**i * (i+1)**p * (n-i)**q   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    # log‑space expression
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    # shift to avoid overflow/underflow
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """Return lam·uniform + (1‑lam)·shape  (both already sum to 1)."""
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Gradient for the active set of maximal convolution entries
# ----------------------------------------------------------------------


def _compute_subgrad_multi(
    a: np.ndarray, conv: np.ndarray, tol: float = 1e-12
) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) with respect to a, aggregating over
    all shifts whose value is within `tol` of the maximum.
    """
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    return g


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def _adam_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on 2·n·max(conv(a,a)).
    The iterate is always kept on the probability simplex.
    """
    n = a0.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    a = a0.copy()
    best_a = a.copy()
    best_score = best_score_initial

    rng = np.random.default_rng()
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha0 = 0.16      # base learning rate
    t = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    while time.time() < deadline:
        t += 1
        conv = _conv_self_fft(a, L_fft)
        g = _compute_subgrad_multi(a, conv, tol=1e-12)

        # gradient of the objective J = 2·n·max(conv)
        grad = 2.0 * n * g

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha0 / (np.sqrt(v_hat) + eps)
        candidate = a - step * m_hat
        candidate = _project_simplex(candidate)

        cur_score = _eval_fast(candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = candidate.copy()
            a = candidate
            # reset moments after improvement – helps escaping shallow minima
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{time.time()-start_time:.2f}s] Adam iter improved score={cur_score:.6f}")
        else:
            # occasional uphill move
            if rng.random() < 0.001:
                a = candidate

        # jitter occasionally to shake out of plateaux
        if t % 2500 == 0:
            eps_jit = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + eps_jit, 0.0)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)

    return best_a, best_score


# ----------------------------------------------------------------------
# Final tiny hill‑climb (random mass transfers)
# ----------------------------------------------------------------------


def _final_local_search(
    a: np.ndarray,
    deadline: float,
    start_time: float,
    best_score_initial: float,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two
    coordinates, keeping the vector on the simplex.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # move up to 0.2 % of the source mass
        eps = a[src] * rng.random() * 2e-3
        prop = a.copy()
        prop[src] -= eps
        prop[dst] += eps
        new_sc = _eval_fast(prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = prop.copy()
            a = prop
            print(f"[{time.time()-start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            if rng.random() < 0.001:
                a = prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Random parametric search (wide ranges)
# ----------------------------------------------------------------------


def _random_param_search(
    n: int,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float, tuple[float, float, float, float]]:
    """
    Very wide random sampling of (r,p,q,λ) and keep the best vector.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        logr = rng.uniform(np.log(0.5), np.log(0.999999))
        r = np.exp(logr)
        p = rng.uniform(-6.0, 6.0)
        q = rng.uniform(-6.0, 6.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(
                f"[{time.time()-start_time:.2f}s] random ext r={r:.6f} p={p:.3f} "
                f"q={q:.3f} λ={lam:.3f} score={sc:.6f}"
            )

    return best_seq, best_score, best_params


# ----------------------------------------------------------------------
# Local hill‑climb on the four shape parameters
# ----------------------------------------------------------------------


def _hillclimb_params(
    initial_params: tuple[float, float, float, float],
    start_seq: np.ndarray,
    start_score: float,
    deadline: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb on (r,p,q,λ) with slowly decaying perturbations.
    """
    cur_r, cur_p, cur_q, cur_lam = initial_params
    cur_seq = start_seq.copy()
    cur_score = start_score

    sigma_r = 0.02
    sigma_p = 0.6
    sigma_q = 0.6
    sigma_lam = 0.2

    rng = np.random.default_rng()
    n = cur_seq.shape[0]
    L_fft = _next_pow2(2 * n - 1)

    best_seq = cur_seq.copy()
    best_score = cur_score

    while time.time() < deadline:
        # propose a new point
        r_new = cur_r * np.exp(sigma_r * rng.standard_normal())
        p_new = cur_p + sigma_p * rng.standard_normal()
        q_new = cur_q + sigma_q * rng.standard_normal()
        lam_new = cur_lam + sigma_lam * rng.standard_normal()

        # clamp
        r_new = float(np.clip(r_new, 0.5, 0.999999))
        lam_new = float(np.clip(lam_new, 0.0, 1.0))

        shape_new = _make_shape_general(n, r_new, p_new, q_new)
        cand_new = _mix_with_uniform(shape_new, lam_new)
        sc_new = _eval_fast(cand_new, L_fft)

        if sc_new < cur_score:
            cur_r, cur_p, cur_q, cur_lam = r_new, p_new, q_new, lam_new
            cur_seq = cand_new
            cur_score = sc_new
            if sc_new < best_score:
                best_score = sc_new
                best_seq = cand_new.copy()
                print(
                    f"[{time.time()-start_time:.2f}s] hillclimb r={r_new:.6f} "
                    f"p={p_new:.3f} q={q_new:.3f} λ={lam_new:.3f} score={sc_new:.6f}"
                )

        # decay perturbations very gently
        sigma_r *= 0.9995
        sigma_p *= 0.9995
        sigma_q *= 0.9995
        sigma_lam *= 0.9995

        # occasional jitter
        if rng.random() < 0.005:
            cur_seq = np.maximum(
                cur_seq + 1e-4 * rng.standard_normal(n) * cur_seq, 0.0
            )
            cur_seq = _project_simplex(cur_seq)
            cur_score = _eval_fast(cur_seq, L_fft)

    return best_seq, best_score
