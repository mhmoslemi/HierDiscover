"""
Hybrid optimiser for the step‑function bound

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine works inside the 1000 s budget and tries to beat the previously
published score ≈1.508.  Three phases are used:

1. **Diverse initial pool** – analytic families (uniform, power‑ramps,
   exponentials, power‑laws, asymmetric Beta, Dirichlet draws) are sampled.
   The best `TOP_K` candidates are kept.

2. **Exponentiated sub‑gradient descent** – a multiplicative update
   driven by the active set of convolution entries that attain the current
   maximum.  A slower decay of the learning‑rate and a larger number of
   passes (the bulk of the time budget) give a finer exploration.

3. **Soft‑max Adam polishing + random local search** – the hard max is
   smoothed with an annealed temperature and Adam updates keep the iterate
   on the simplex.  After this deterministic phase a short stochastic
   “hill‑climbing” phase attempts tiny random moves that still improve the
   objective.

All operations are vectorised (FFT based convolution).  The final vector
is normalised to sum = 1 and returned as a plain Python list.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # eliminate numerical noise
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    shape /= shape.sum()
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    shape /= shape.sum()
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.gamma(conc, 1.0, size=n)
    vec /= vec.sum()
    return vec


# ----------------------------------------------------------------------
# Top‑k pool handling
# ----------------------------------------------------------------------
def _push_top(pool: List[Tuple[float, np.ndarray]], score: float,
              vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (active‑set ℓ∞ optimisation)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> Tuple[np.ndarray, float]:
    """Multiplicative update driven by the active set of the current max."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 0.7               # slightly larger base step
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()

        # active set (indices attaining the max, up to tolerance)
        tol = max(1e-14, max_conv * 1e-14)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient of the max‑convolution w.r.t. a
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1: n - 1 + n]

        eta = eta0 / (1.0 + it / 4000.0)          # slower decay
        factor = np.exp(np.clip(-eta * grad, -40.0, 40.0))
        a *= factor
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} "
                  f"best={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Soft‑max Adam polishing
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     deadline: float,
                     start_time: float) -> Tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.12
    T0, T_end = 0.2, 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} "
                  f"best={cur_score:.6f}")

        # anneal temperature exponentially
        T = T_end + (T0 - T_end) * (0.5 ** (it / 3500.0))

        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1: n - 1 + n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best, best_score

# Tiny stochastic hill‑climbing refinement
# ----------------------------------------------------------------------
def _random_local_search(a0: np.ndarray,
                         deadline: float,
                         start_time: float) -> Tuple[np.ndarray, float]:
    """Small random perturbations that are kept when they improve the score."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    rng = np.random.default_rng(int(time.time()))
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two positions and move a tiny amount δ from i to j
        i, j = rng.choice(n, size=2, replace=False)
        delta = rng.uniform(0.0, 1e-3) * a[i]   # relative move
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # re‑project onto simplex (the move already keeps sum constant, but
        # clipping may violate non‑negativity)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        cur_score = _objective(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            a = a_new
            print(f"[{time.time() - start_time:.2f}s] LocalSearch step={it:,} "
                  f"best={cur_score:.6f}")
        # occasional restart to avoid being stuck
        if it % 2000 == 0:
            # tiny random jitter around the current best
            jitter = rng.exponential(scale=1e-4, size=n)
            a = np.maximum(best + jitter, 0.0)
            a /= a.sum()

    return best, best_score
