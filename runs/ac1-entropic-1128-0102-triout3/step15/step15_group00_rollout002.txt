def _smoothed_descent(a0, deadline, best_seq, best_score, target_beta=200, init_beta=5, beta_inc=0.5, step0=1.0, step_decay=0.995, step_min=1e-4):
    a = a0.copy()
    beta = init_beta
    step = step0
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and current objective
        conv = _conv_self_fft(a)  # using FFT (full conv)
        cur_score = 2.0 * len(a) * conv.max()
        if cur_score < best_score:
            best_score = cur_score; best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothedDescent it={it}, beta={beta:.2f}, score={cur_score:.6f}")
        # compute smoothed weights
        # To improve stability, shift conv by max (to avoid overflow)
        shift = conv.max()
        exp_beta = np.exp(beta * (conv - shift))
        w = exp_beta / exp_beta.sum()
        # compute gradient
        g = _gradient_softmax_fft(a, w)  # shape n
        # center gradient (subtract mean) to preserve sum after exponentiation
        g_center = g - g.mean()
        # compute factor
        factor = np.exp(-step * g_center)
        # clip factor for stability
        factor = np.clip(factor, 1e-12, 1e12)
        a = a * factor
        # normalize to sum 1
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a = a / a_sum

        # update step size (maybe schedule)
        step = max(step * step_decay, step_min)

        # increase beta gradually
        if beta < target_beta:
            beta = min(target_beta, beta + beta_inc)

    return best_seq, best_score

    to possibly discover a better region of the search space.
3.  **Soft‑max gradient descent** – the true objective  max(conv) is
    approximated by  Sβ = (1/β)·log∑exp(β·conv).  Its gradient is cheap to
    compute via an FFT‑based convolution.  An exponentiated‑gradient update
    on the probability simplex (mirror descent with KL‑divergence) is then
    performed, while β is slowly increased (annealing) so that the surrogate
    becomes tighter.
4.  A final hill‑climb that repeatedly transfers a small fraction of mass
    from a “high‑gradient” index to a “low‑gradient” one, with occasional
    random shake‑ups to escape shallow plateaux.

The best sequence found (normalised to sum = 1) is returned as a plain Python
list.  All operations respect the security checks built into
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities (re‑used many times)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a cached FFT length."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def _gradient_softmax_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max‐surrogate Sβ w.r.t. a:
        g_i = 2 Σ_j a_j·w_{j+i},
    where w are the soft‑max weights (Σ w = 1).
    Computed via one FFT‑based convolution.
    """
    n = a.shape[0]
    L = _next_pow2(n + w.size - 1)          # = next_pow2(3·n‑2)
    A_rev = np.fft.rfft(a[::-1], n=L)       # reversed a
    W = np.fft.rfft(w, n=L)
    conv = np.fft.irfft(A_rev * W, n=L)[: n + w.size - 1]
    start = n - 1
    return 2.0 * conv[start:start + n]      # slice of length n


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (sum is kept = 1, so denominator = 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Very light parametric families (geometric & power)
# ----------------------------------------------------------------------
def _quick_parametric_sweep(n: int, best_seq: np.ndarray,
                            best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """Search a few geometric / power families to obtain a non‑uniform seed."""
    idx = np.arange(n, dtype=np.float64)

    # geometric decay – rough grid
    for eps in np.logspace(-5, -1, 30):
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        sc = _objective_score(geo)
        if sc < best_score:
            best_seq, best_score = geo.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] geo eps={eps:.1e} score={sc:.6f}")

    # power law – coarse grid
    for p in np.linspace(-2.0, 2.0, 41):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] power p={p:.3f} score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws (few but diverse)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n: int, best_seq: np.ndarray,
                             best_score: float, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    trials = 200
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)  # concentration ≈ 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a.copy(), sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smoothed‑max (log‑sum‑exp) gradient descent on the simplex
# ----------------------------------------------------------------------
def _softmax_descent(a0: np.ndarray, deadline: float,
                     best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using the soft‑max surrogate
    of max(conv).  β (inverse temperature) is annealed from a small to a
    larger value, tightening the approximation.
    """
    a = a0.copy()
    n = a.shape[0]

    # Hyper‑parameters (tuned experimentally)
    beta = 5.0            # start temperature (low β → smooth surrogate)
    beta_target = 250.0   # final temperature (high β → close to true max)
    beta_inc = 0.5        # per‑iteration increase
    eta = 0.5             # base step size for the exponentiated update
    eta_decay = 0.9995    # exponential decay of η
    min_eta = 5e-3

    it = 0
    while time.time() < deadline:
        it += 1

        # 1) current objective (exact max) – used for tracking only
        conv = _conv_self_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] SoftmaxDescent it={it:,} β={beta:.1f} score={cur_score:.6f}")

        # 2) soft‑max weights over the convolution entries
        shift = conv.max()
        # (conv - shift) ≤ 0 ⇒ exp never overflows
        w_exp = np.exp(beta * (conv - shift))
        w = w_exp / w_exp.sum()                     # Σ w = 1

        # 3) gradient of the surrogate (size n)
        g = _gradient_softmax_fft(a, w)               # g_i = ∂Sβ/∂a_i

        # 4) centred gradient (preserves the simplex after exponentiation)
        g_center = g - g.mean()

        # 5) multiplicative (exponentiated) update, then renormalise
        factor = np.exp(-eta * g_center)
        factor = np.clip(factor, 1e-12, 1e12)
        a = a * factor
        a_sum = a.sum()
        a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)

        # 6) schedule updates
        eta = max(eta * eta_decay, min_eta)
        if beta < beta_target:
            beta = min(beta_target, beta + beta_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb (larger, directed mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that repeatedly moves a small fraction of mass
    from the index with the largest sub‑gradient to the index with the
    smallest sub‑gradient.  Occasional random reshuffles prevent stalling.
    """
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self_fft(a)
        max_val = conv.max()
        # sub‑gradient of max(conv) (averaged over all active positions)
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += a[k - idx]
        if active.size:
            grad = 2.0 * grad / active.size

        src_cand = np.where(a > 1e-14)[0]
        if src_cand.size == 0:
            break
        src = src_cand[np.argmax(grad[src_cand])]
        dst = np.argmin(grad)

        # transfer up to a few percent of the source mass
        max_frac = 0.06
        frac = rng.random() * max_frac + 5e-5
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # no negative entries allowed (tiny tolerance for floating‑point)
        if a_prop[src] < -1e-12:
            continue

        new_score = _objective_score(a_prop)
        if new_score < best_score:
            a = a_prop
            best_a, best_score = a.copy(), new_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] HillClimb score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                # Random shake‑up to escape a flat region
                i1, i2 = rng.choice(n, size=2, replace=False)
                shift = a[i1] * (rng.random() * max_frac + 1e-4)
                a[i1] -= shift
                a[i2] += shift
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights (sum = 1) that minimises
    ``evaluate_sequence``.  The routine uses a fast soft‑max surrogate
    gradient descent followed by a simple hill‑climb.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between flexibility and speed
    # ------------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1, if present)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Very quick parametric sweep (geometric / power)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _quick_parametric_sweep(
            n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers (diverse starts)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(
            n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 3️⃣  Soft‑max gradient descent with temperature annealing
    # ------------------------------------------------------------------
    descent_deadline = deadline - 150.0
    if time.time() < descent_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting soft‑max descent")
        best_seq, best_score = _softmax_descent(
            best_seq, descent_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb (sharpen the solution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list (the evaluator will clamp)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
