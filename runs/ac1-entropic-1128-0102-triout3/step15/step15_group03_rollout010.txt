def _subgradient_maxconv(a, conv):
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

"""
Improved hill‑climb for the self‑convolution max problem.

The algorithm starts from the best known sequence (global variable
``height_sequence_1`` if it exists, otherwise a uniform distribution),
normalises it to sum = 1 and computes the full linear self‑convolution.
A fast O(1) update of the convolution is derived for any tiny transfer
of mass ε from a source index s to a destination index d:

    conv'[2s]   = conv[2s]   - 2·a_s·ε + ε²
    conv'[s+d]  = conv[s+d]  + 2·(a_s-a_d)·ε - 2·ε²
    conv'[2d]   = conv[2d]   + 2·a_d·ε + ε²

Only these three entries change, therefore the new maximum can be
evaluated instantly.  At each iteration we:
  * compute a sub‑gradient of the “max‑convolution’’ functional,
  * pick a source index with largest positive gradient (large contribution
    to the current peak) and a destination with the smallest gradient,
  * test a handful of candidate ε (fractions of the source mass) and
    apply the one that yields the biggest drop of the peak,
  * keep the best sequence found so far.

If no improving move exists a tiny random jitter is performed to escape
plateaus.  The loop runs until the time budget (≈ 1000 s) is exhausted.
The final sequence is returned as a plain Python list; the evaluation
function will normalise it automatically.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution a*a (real vectors, zero‑padded).
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (as used in the original code).
    For each shift attaining the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # a[k-idx] is a reversed slice of a
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# Precise hill‑climb with analytic 3‑point convolution update
# ----------------------------------------------------------------------
def _hill_climb_precise(
    a: np.ndarray,
    conv: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_obj: float,
) -> tuple[np.ndarray, float]:
    """
    Performs the hill‑climb described in the module docstring.
    Returns the best sequence (as a NumPy array) and its objective
    2·n·max(conv).
    """
    n = a.shape[0]
    best_score = best_obj
    best_a = best_seq.copy()

    # fractions of the source mass to try in each move
    eps_fractions = np.array([0.001, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0])
    no_improve = 0
    rng = np.random.default_rng()

    # occasional full recomputation to avoid drift
    recompute_interval = 20000
    it = 0

    while time.time() < deadline:
        it += 1
        M = conv.max()
        # sub‑gradient of the maximal-convolution functional
        grad = _subgradient_maxconv(a, conv)

        # source: largest positive gradient among indices that still have mass
        mask_positive = (a > 1e-15)
        if not np.any(mask_positive):
            break
        src = int(np.argmax(grad * mask_positive))

        # destination: smallest gradient (could be negative)
        dst = int(np.argmin(grad))
        if src == dst:
            # fallback: pick a random different pair
            src_candidates = np.where(mask_positive)[0]
            src = int(rng.choice(src_candidates))
            dst = int(rng.choice(np.delete(np.arange(n), src)))
        eps_max = a[src]
        if eps_max <= 0:
            # nothing to move – random jitter and continue
            src = int(rng.choice(np.where(a > 1e-15)[0]))
            eps_max = a[src]

        # compute the maximal value among indices that are NOT affected
        affected = np.unique([2 * src, src + dst, 2 * dst])
        # safe‑guard: ensure indices are within the convolution length
        affected = affected[(affected >= 0) & (affected < 2 * n - 1)]
        # mask out affected entries
        temp = conv.copy()
        temp[affected] = -np.inf
        M2 = temp.max()

        # try several epsilon candidates
        best_eps = 0.0
        best_new_max = M
        a_src0 = a[src]
        a_dst0 = a[dst]

        for f in eps_fractions:
            eps = eps_max * f
            # analytically updated convolution entries
            c2s = conv[2 * src] - 2.0 * a_src0 * eps + eps * eps
            csd = conv[src + dst] + 2.0 * (a_src0 - a_dst0) * eps - 2.0 * eps * eps
            c2d = conv[2 * dst] + 2.0 * a_dst0 * eps + eps * eps

            # new overall maximum after this move
            new_max = c2s
            if csd > new_max:
                new_max = csd
            if c2d > new_max:
                new_max = c2d
            if M2 > new_max:
                new_max = M2

            if new_max < best_new_max - 1e-14:
                best_new_max = new_max
                best_eps = eps

        if best_eps > 0:
            # apply the best move
            eps = best_eps
            a[src] -= eps
            a[dst] += eps
            # update the three convolution entries in‑place
            conv[2 * src] = conv[2 * src] - 2.0 * a_src0 * eps + eps * eps
            conv[src + dst] = conv[src + dst] + 2.0 * (a_src0 - a_dst0) * eps - 2.0 * eps * eps
            conv[2 * dst] = conv[2 * dst] + 2.0 * a_dst0 * eps + eps * eps

            # keep track of the best objective found so far
            cur_score = 2.0 * n * best_new_max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] New best 2·n·max = {best_score:.6f}")
            no_improve = 0
        else:
            # no improving move – perform a tiny random jitter
            no_improve += 1
            if no_improve > 300:
                src = int(rng.choice(np.where(a > 1e-15)[0]))
                dst = int(rng.integers(0, n))
                if src == dst:
                    continue
                eps = a[src] * (rng.random() * 0.004 + 1e-6)
                # apply jitter (even if it does not improve)
                a_src0 = a[src]
                a_dst0 = a[dst]
                a[src] -= eps
                a[dst] += eps
                conv[2 * src] = conv[2 * src] - 2.0 * a_src0 * eps + eps * eps
                conv[src + dst] = conv[src + dst] + 2.0 * (a_src0 - a_dst0) * eps - 2.0 * eps * eps
                conv[2 * dst] = conv[2 * dst] + 2.0 * a_dst0 * eps + eps * eps
                no_improve = 0

        # periodically recompute the full convolution to suppress numerical drift
        if it % recompute_interval == 0:
            conv = _conv_self(a)

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising
        2·n·max(conv(a,a)) / (sum(a)**2)

    The implementation starts from the globally best known sequence,
    then performs a precise hill‑climb that moves mass between two
    indices using an exact three‑point update of the self‑convolution.
    The loop runs until the supplied time budget (≈ 1000 s) is exhausted.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose problem size and initialise a
    # ------------------------------------------------------------------
    # prefer the length of the global starter if it exists
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            n = starter.size
            a = np.maximum(starter, 0.0)
        else:
            raise ValueError
    except Exception:
        n = 4096
        a = np.full(n, 1.0 / n)

    # Normalise to a probability vector (sum = 1)
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    # ------------------------------------------------------------------
    # Compute initial self‑convolution and objective
    # ------------------------------------------------------------------
    conv = _conv_self(a)
    cur_max = conv.max()
    best_score = 2.0 * n * cur_max
    best_seq = a.copy()
    print(f"[{time.time() - start_time:.2f}s] Initial objective 2·n·max = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional cheap random Dirichlet exploration (first 30 s)
    # ------------------------------------------------------------------
    dirichlet_deadline = min(deadline - 600, start_time + 30)  # at most 30 s, leave most time for hill‑climb
    if time.time() < dirichlet_deadline:
        rng = np.random.default_rng()
        trials = 2000
        for _ in range(trials):
            if time.time() >= dirichlet_deadline:
                break
            # concentration between 0.01 and 5 (log‑uniform)
            logc = rng.uniform(-2.0, 0.7)
            c = 10 ** logc
            x = rng.gamma(shape=c, scale=1.0, size=n)
            if x.sum() == 0:
                continue
            x /= x.sum()
            cur = _conv_self(x)
            cur_obj = 2.0 * n * cur.max()
            if cur_obj < best_score:
                best_score = cur_obj
                best_seq = x.copy()
                a = x.copy()
                conv = cur
                print(f"[{time.time() - start_time:.2f}s] Dirichlet improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Precise hill‑climb on the current best candidate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting precise hill‑climb")
        best_seq, best_score = _hill_climb_precise(
            a,
            conv,
            deadline,
            best_seq,
            best_score,
        )
        print(f"[{time.time() - start_time:.2f}s] Hill‑climb finished – best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – final objective ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
