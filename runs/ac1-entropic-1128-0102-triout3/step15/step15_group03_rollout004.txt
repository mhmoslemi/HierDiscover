"""
Propose a step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search combines three ideas:

1.  Initialise from the best known sequence (global height_sequence_1) or from
    a uniform distribution.
2.  A *soft‑max gradient descent* that directly minimises a smooth
    approximation of the max‑convolution.  The temperature (beta) is raised
    gradually so that the optimisation first sees a smooth landscape and then
    focuses on the true max.
3.  A long simulated‑annealing / hill‑climbing phase that moves tiny amounts
    of mass from “over‑represented’’ indices to “under‑represented’’ ones.
    The temperature decays exponentially, allowing occasional uphill moves
    early on and only improvements later.

Both phases respect the 1000 s wall‑clock limit and keep the vector on the
probability simplex (non‑negative, sum = 1).  The final list is returned as
plain Python floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (linear, not circular)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w

# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


# ----------------------------------------------------------------------
# Soft‑max (smooth max) gradient descent
# ----------------------------------------------------------------------
def _softmax_gd(a0: np.ndarray, deadline: float,
                best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Perform many gradient steps on the smooth approximation
        f_β(a) = (1/β)·log Σ_k exp(β·conv_k(a))
    with β increasing from a small to a large value.
    The iterate is always projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    # hyper‑parameters
    beta = 5.0                     # start temperature for the soft‑max
    beta_target = 200.0            # final temperature
    beta_step = (beta_target - beta) / 4000   # linear schedule (~4000 iters)
    step = 0.2                     # initial step scale
    step_decay = 0.9997            # geometric decay of the step scale

    itr = 0
    while time.time() < deadline:
        itr += 1
        # raise β gradually
        if beta < beta_target:
            beta = min(beta_target, beta + beta_step)

        # convolution of the current iterate
        conv = _conv_self(a)
        max_val = conv.max()

        # soft‑max weights (numerically stable)
        w = np.exp(beta * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:                     # extremely unlikely, but guard
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of Σ_k w_k·conv_k w.r.t. a  (see analysis)
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)      # length 3·n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # normalise the step
        gnorm = np.linalg.norm(grad) + 1e-12
        alpha = step / gnorm

        # gradient step and projection
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # compute new (exact) objective
        new_score = _objective_score(a_new)

        # accept if improvement, otherwise keep the old iterate
        if new_score <= cur_score:
            a = a_new
            cur_score = new_score

        # keep the best overall sequence seen so far
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Soft‑max GD it={itr} "
                  f"β={beta:.1f} step={alpha:.3e} score={cur_score:.6f}")

        # decay the step size a little
        step = max(step * step_decay, 1e-6)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated annealing / tiny‑mass hill‑climb
# ----------------------------------------------------------------------
def _annealing(a0: np.ndarray, deadline: float,
               best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Move tiny amounts of mass from a source index to a destination index.
    Early in the run uphill moves are allowed with probability exp(-Δ/T).
    The temperature T decays exponentially.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_score = _objective_score(a)

    rng = np.random.default_rng()
    T = 0.02                     # initial temperature
    T_min = 1e-7
    decay = 0.99995              # temperature decay per iteration

    it = 0
    while time.time() < deadline:
        it += 1

        # ---- choose a move -------------------------------------------------
        if rng.random() < 0.2:                     # occasional larger move
            src = rng.integers(n)
            while a[src] < 1e-14:
                src = rng.integers(n)
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)
            eps = a[src] * rng.uniform(0.01, 0.15)  # up to 15 % of source
        else:                                      # tiny move (default)
            src = rng.integers(n)
            while a[src] < 1e-14:
                src = rng.integers(n)
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)
            eps = a[src] * rng.uniform(1e-6, 0.004)  # up to 0.4 % of source

        # perform the transfer – sum stays exactly 1, no projection needed
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # compute the new objective
        new_score = _objective_score(a_prop)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            a = a_prop
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Anneal it={it} "
                      f"T={T:.2e} score={cur_score:.6f}")

        # temperature schedule
        T = max(T * decay, T_min)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the provided evaluation function.  The algorithm runs for at most
    ``budget_s`` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # problem size – keep it moderate so that FFTs stay cheap
    n = 4096

    # ------------------------------------------------------------------
    # Starting point (global starter if available)
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n)   # uniform baseline
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    try:
        # ``height_sequence_1`` may have been defined by a previous run
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.size != n:
                # either truncate or pad with zeros
                if starter.size > n:
                    starter = starter[:n]
                else:
                    starter = np.pad(starter, (0, n - starter.size))
            starter_sum = starter.sum()
            if starter_sum > 0:
                starter /= starter_sum
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – Soft‑max gradient descent (≈ 40 % of the budget)
    # ------------------------------------------------------------------
    gd_deadline = start_time + 0.40 * (deadline - start_time)
    if time.time() < gd_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting soft‑max gradient descent")
        best_seq, best_score = _softmax_gd(best_seq, gd_deadline,
                                          best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After GD score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Simulated annealing / hill‑climb (use the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting annealing phase")
        best_seq, best_score = _annealing(best_seq, deadline,
                                          best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final annealing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise (just in case numerical drift happened) and return list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total <= 0:
        best_seq = np.full(n, 1.0 / n)
    else:
        best_seq /= total

    print(f"[{time.time() - start_time:.2f}s] Finished – best score≈{best_score:.6f}")
    return [float(x) for x in best_seq]
