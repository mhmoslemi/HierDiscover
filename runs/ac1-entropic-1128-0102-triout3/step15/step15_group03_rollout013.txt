def _adam_smooth_gradient_descent(a0, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lambda_val = 5.0
    while time.time() < deadline:
        t += 1
        conv = _conv_self(a)
        # smooth max via softmax
        max_val = conv.max()
        # shift for numerical stability
        shifted = conv - max_val
        exp_term = np.exp(lambda_val * shifted)
        p = exp_term / exp_term.sum()
        # compute gradient
        p_rev = p[::-1]
        grad_full = _conv_fft(p_rev, a)
        grad = 2.0 * grad_full[len(a)-1:len(a)-1+len(a)]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.1 / (np.sqrt(v_hat) + eps)  # base step scaling
        a = a - step * m_hat
        # projection onto simplex
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        # compute actual score
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam lambda={lambda_val:.2f} score={cur_score:.6f}")
        # slowly increase lambda (push towards true max)
        if t % 2000 == 0 and lambda_val < 30:
            lambda_val *= 1.2
    return best_a, best_score
```

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The search is split into four phases and respects a 1000 s wall‑clock budget.

1️⃣  Fast parametric sweep (geometric & power families) – gives a quick
    decent seed.
2️⃣  Random Dirichlet draws – occasional stochastic jumps.
3️⃣  Adam‑type gradient descent on a smooth approximation of
    ``max(conv)``.  The objective is
        L(a)=log‑sum‑exp( λ·conv(a) )/λ,
    whose gradient is easy to obtain via a second convolution.
    λ is increased gradually (annealing) so that the optimisation first
    flattens the whole convolution and later focuses on the true maximum.
    After each update the iterate is projected onto the probability simplex.
4️⃣  A final tiny‑mass hill‑climb (the same “move mass from the most
    over‑represented index to the most under‑represented one” that was
    used in the previous state‑of‑the‑art).

All intermediate best scores are printed.  The final list is normalised
to sum = 1 and consists only of non‑negative floats, ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real inputs)."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = rfft(x, n=L)
    Y = rfft(y, n=L)
    conv = irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (no circular wrap‑around)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    When several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # derivative contribution a_{k-idx} appears twice (except when idx==k-idx)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Very light parametric sweep (geometric & power families)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 80)          # reduced grid for speed
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 7):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 81):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet draws
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline, trials=120):
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)            # concentration between ~0.01 and 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Adam‑type gradient descent on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_smooth_gradient_descent(a0: np.ndarray, deadline, best_seq, best_score):
    """
    Optimises the smooth objective   L(a)=log‑sum‑exp(λ·conv(a))/λ
    with Adam updates.  λ starts small (≈5) and is increased
    periodically so that the optimisation finally focuses on the true max.
    After each step the iterate is projected onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t = 0
    lambda_val = 5.0                # start with a modest smoothing
    max_lambda = 30.0
    # schedule for λ (multiply every 2000 steps)
    lambda_growth = 1.2

    while time.time() < deadline:
        t += 1

        # ----- 1. compute current convolution  -------------------------
        conv = _conv_self(a)                     # length 2n‑1

        # ----- 2. smooth max via softmax -------------------------------
        max_val = conv.max()
        shifted = conv - max_val                 # avoid overflow
        exp_term = np.exp(lambda_val * shifted)
        p = exp_term / exp_term.sum()            # softmax probabilities, size 2n‑1

        # ----- 3. gradient of the smooth max ---------------------------
        # grad_j = 2 Σ_k p_k a_{k-j}
        p_rev = p[::-1]
        grad_full = _conv_fft(p_rev, a)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # slice of length n

        # ----- 4. Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = 0.05 / (np.sqrt(v_hat) + eps)     # base learning rate 0.05

        a = a - step * m_hat
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # ----- 5. actual score (non‑smooth) ----------------------------
        cur_score = _objective_score(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam λ={lambda_val:.2f} it={t} score={cur_score:.6f}")

        # ----- 6. increase λ slowly ------------------------------------
        if t % 2000 == 0 and lambda_val < max_lambda:
            lambda_val = min(max_lambda, lambda_val * lambda_growth)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (tiny mass transfers, as in the previous code)
# ----------------------------------------------------------------------
def _hill_climb(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        eps = a[src] * (rng.random() * 0.004 + 1e-6)   # up to 0.4 % of source
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill step score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
                a[i1] -= jitter
                a[i2] += jitter
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score

    # --------------------------------------------------------------
    rnd_deadline = start_time + min(170.0, budget_s * 0.17)
    if time.time() < rnd_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, rnd_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Adam‑type smooth‑max gradient descent
    # --------------------------------------------------------------
    grad_deadline = deadline - 150.0
    if time.time() < grad_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam‑smooth gradient descent")
        best_seq, best_score = _adam_smooth_gradient_descent(best_seq, grad_deadline,
                                                             best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After Adam score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final directed hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
