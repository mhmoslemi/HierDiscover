        conv = np.fft.irfft(Af*Af, n=L_fft)[:2*n-1]

        max_scaled = np.max(scaled)

        w /= w.sum()

        w_rev = w[::-1]
        Wa = np.fft.rfft(w_rev, n=L_corr)
        Aa = np.fft.rfft(a, n=L_corr)

        grad = 2.0 * g
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        # projection onto simplex
        a_candidate = _project_simplex(a_candidate)
        # evaluate
        cur_score = _eval_fast(a_candidate, L_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            print(f"[{time.time()-start_time:.2f}s] Adam it={t} alpha={alpha:.1f} score={cur_score:.6f}")
            a = a_candidate
        else:
            # optionally accept occasional uphill moves
            if rng.random() < 0.005:
                a = a_candidate
        # increase alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * growth)
        # jitter occasionally

                eps = 1e-4 * rng.standard_normal(n) * a

                a = _project_simplex(a)
                # reset Adam moments

                v = np.zeros_like(a)
        # for safety break time
        if time.time() > deadline - 0.1:
            break
    return best_a, best_score

"""
Optimize a non‑negative step function for the objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1️⃣  A short random parametric sweep (geometric × power‑tail family) to
    find a promising starting vector.

2️⃣  A smooth‑max Adam descent: we minimise a soft‑max approximation of
    the peak of the self‑convolution.  The temperature (α) is increased
    gradually, turning the problem into a true max‑convolution minimisation.
    Gradients are obtained via FFT‑based correlation and the iterates are
    projected onto the probability simplex.

3️⃣  A cheap final “mass‑transfer” hill‑climb that moves tiny amounts of
    probability mass between two coordinates.

All steps respect the 1000 s time budget, print progress information,
and finally return the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = np.fft.irfft(np.fft.rfft(a, n=L_fft) * np.fft.rfft(a, n=L_fft), n=L_fft)
    conv = conv[: 2 * a.shape[0] - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric‑power‑tail family)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components proportional to
    r**i * (i+1)**p * (n-i)**q (i = 0…n‑1).
    Computed in log‑space for stability.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return lam·uniform + (1‑lam)·shape  (both already sum to 1).
    Renormalise to guard against drift.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Final tiny hill‑climb (random mass transfers)
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two
    coordinates, keeping the vector on the simplex.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(
                f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}"
            )
        else:
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_sc


# ----------------------------------------------------------------------
# Adam descent on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------


def _adam_smoothing_descent(a0: np.ndarray,
                            L_fft: int,
                            L_corr: int,
                            deadline: float,
                            start_time: float,
                            best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam descent on  g(a) = (1/α)·log(∑ exp(α·conv(a,a)) ).
    The temperature α is increased gradually, turning g into max(conv).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = best_score_initial

    # Adam hyper‑parameters
    lr = 0.18
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature schedule
    alpha = 1.0
    alpha_max = 2000.0
    grow = 1.025

    beta1_pow = beta1
    beta2_pow = beta2
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1

        # ------------------- self‑convolution -------------------
        Af = np.fft.rfft(a, n=L_fft)
        conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]

        # ------------------- soft‑max weights -------------------
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()                     # now ∑ w_k = 1

        # ------------------- gradient of the smooth objective -------------------
        # gradient = 2 * correlation(w, a)
        w_rev = w[::-1]                    # length 2n‑1
        Wa = np.fft.rfft(w_rev, n=L_corr)
        Aa = np.fft.rfft(a, n=L_corr)
        corr_full = np.fft.irfft(Wa * Aa, n=L_corr)
        corr_full = corr_full[: len(w_rev) + n - 1]

        idx_start = len(w_rev) - 1          # = 2n‑2
        g = corr_full[idx_start - np.arange(n)]
        grad = 2.0 * g

        # ------------------- Adam update -------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        cur_score = _eval_fast(a_candidate, L_fft)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            # reset Adam moments after a genuine improvement
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta1
            beta2_pow = beta2
            print(
                f"[{time.time() - start_time:.2f}s] Adam it={it:,} α={alpha:.1f} score={cur_score:.6f}"
            )
        else:
            # occasional uphill accept to avoid stagnation
            if rng.random() < 0.006:
                a = a_candidate

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * grow)

        # update bias‑correction factors
        beta1_pow *= beta1
        beta2_pow *= beta2

        # occasional jitter to shake the iterate
        if it % 2500 == 0:
            jitter = 1e-4 * rng.standard_normal(n) * a
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta1
            beta2_pow = beta2

    return best_a, best_score

    # --------------------------------------------------------------
    n = 4096
    L_fft = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 0️⃣  Uniform baseline
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_fft)
    print(
        f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}"
    )

    # --------------------------------------------------------------
    # Load a globally supplied starter if it exists
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            if starter.sum() == 0.0:
                starter = uniform.copy()
            else:
                starter = starter / starter.sum()
            cur = _eval_fast(starter, L_fft)
            if cur < best_score:
                best_seq = starter.copy()
                best_score = cur
                print(
                    f"[{time.time() - start_time:.2f}s] global starter score={cur:.6f}"
                )
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Short random parametric sweep (geometric × power‑tail)
    # --------------------------------------------------------------
    sweep_deadline = start_time + 150.0
    while time.time() < sweep_deadline:
        # Sample parameters
        r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
        p = rng.uniform(-3.0, 3.0)
        q = rng.uniform(-3.0, 3.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start_time:.2f}s] random sweep r={r:.4f} "
                f"p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}"
            )

    # --------------------------------------------------------------
    # 2️⃣  Main Adam‑smooth‑max descent from the best candidate so far
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    main_deadline = time.time() + max(300.0, remaining * 0.55)
    a_opt, s_opt = _adam_smoothing_descent(
        best_seq, L_fft, L_corr, main_deadline, start_time, best_score
    )
    if s_opt < best_score:
        best_seq = a_opt.copy()
        best_score = s_opt
        print(
            f"[{time.time() - start_time:.2f}s] after Adam descent score={best_score:.6f}"
        )

    # --------------------------------------------------------------
    # 3️⃣  Optional short restarts (random starts + brief Adam runs)
    # --------------------------------------------------------------
    while True:
        remaining = deadline - time.time()
        if remaining < 60.0:
            break
        sub_deadline = time.time() + min(30.0, remaining * 0.5)

        # random fresh start
        r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
        p = rng.uniform(-3.0, 3.0)
        q = rng.uniform(-3.0, 3.0)
        lam = rng.uniform(0.0, 1.0)
        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        init_sc = _eval_fast(cand, L_fft)

        a_tmp, s_tmp = _adam_smoothing_descent(
            cand, L_fft, L_corr, sub_deadline, start_time, init_sc
        )
        if s_tmp < best_score:
            best_seq = a_tmp.copy()
            best_score = s_tmp
            print(
                f"[{time.time() - start_time:.2f}s] restart improved score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # 4️⃣  Final cheap mass‑transfer hill‑climb
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        a_final, sc_final = _final_local_search(
            best_seq, deadline, start_time, best_score
        )
        if sc_final < best_score:
            best_seq = a_final.copy()
            best_score = sc_final
            print(
                f"[{time.time() - start_time:.2f}s] final local best score={best_score:.6f}"
            )

    # --------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Log the official evaluation (security‑checked function)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(
            f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}"
        )
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
