assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\,\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the linear convolution of the non‑negative vector `a`.  
Scaling `a` does not change the score, therefore we can keep the vector
normalised to sum = 1 and only have to minimise  

\[
F(a)=2\,n\;\max\bigl(\text{conv}(a,a)\bigr).
\]

`F` is a non‑smooth function: the gradient is the sub‑gradient of the
maximum entry of the convolution.  A simple exponentiated‑gradient scheme
gets stuck after many iterations.  The new programme uses an **Adam‑type
mirror‑descent on the simplex**, which gives an adaptive step size
and momentum, together with a short random‑restarts phase and a final
mix‑with‑uniform sweep.

The algorithm works in three phases (time‑aware):

1. **Broad random search** – a few thousand Dirichlet samples with widely
   varying concentration parameters.  The best sample is kept.
2. **Adam‑mirror descent** – starting from the best sample we run
   Adam on the simplex using the exact sub‑gradient of `max(conv)`.
   The step size is kept relatively large at the beginning and decays
   automatically via Adam’s moment estimates.  Random jitter is added
   every few thousand iterations to escape shallow plateaus.
3. **Uniform mixing** – a cheap line‑search over convex combinations of
   the final vector and the uniform distribution, keeping the best
   mixture.

All operations are `O(n log n)` (FFT based convolution) and the whole
routine finishes well before the 1000 s budget, leaving a safe margin
for the external runner.

```python
"""
Step‑function optimiser – Adam‑mirror descent on the probability simplex.

The evaluator used in the contest is
    score = 2·n·max(conv(a,a)) / (sum(a)^2)
Since the score is homogeneous we keep the vector normalised to
sum(a)=1 and minimise   F(a)=2·n·max(conv(a,a)).

Algorithm
---------
1.  Random Dirichlet initialisation (wide range of concentration
    parameters) – keeps the best candidate.
2.  Adam‑type mirror descent on the simplex.
    * Sub‑gradient of max(conv) is computed exactly.
    * The Adam moments give an adaptive step‑size and momentum.
    * Every few thousand steps a tiny random jitter is added to help
      escape plateaus.
3.  Final sweep over convex combinations with the uniform distribution
    (mixing λ·a+(1‑λ)·uniform).

The optimiser is time‑aware (budget≈1000 s) and prints progress.
The final list contains non‑negative numbers and is normalised to sum≈1.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities (FFT based convolution, sub‑gradient)
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat values within 1e‑12 of the maximum as active
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]        # contribution a_{k‑i}
    if active.size:
        grad = 2.0 * grad / active.size   # factor 2 because each product appears twice
    return grad


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used during optimisation (a is already normalised)."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# 1️⃣  Random Dirichlet initialisation
# --------------------------------------------------------------

def _random_dirichlet_initialisation(n: int,
                                     deadline: float,
                                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Generate many Dirichlet samples until the deadline – keep the best."""
    best_a = np.full(n, 1.0 / n)                # uniform start
    best_score = _objective_score(best_a, n)

    while time.time() < deadline:
        # concentration drawn log‑uniformly from [0.02, 20]
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s == 0.0:
            continue
        a = sample / s
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_a, best_score


# --------------------------------------------------------------
# 2️⃣  Adam mirror‑descent on the simplex
# --------------------------------------------------------------

def _adam_descent(a0: np.ndarray,
                  n: int,
                  L_fft: int,
                  deadline: float,
                  rng: np.random.Generator,
                  best_a: np.ndarray,
                  best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex with the sub‑gradient of max(conv).
    A small Gaussian jitter is added every few thousand iterations.
    """
    a = a0.copy()
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.4                     # initial learning‑rate (quite aggressive)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t = 0
    while time.time() < deadline:
        t += 1

        conv = _conv_self(a, L_fft)
        cur_score = 2.0 * n * conv.max()

        # keep the best ever seen
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Adam it={t:,} score={cur_score:.6f}")

        # ---- Adam update -------------------------------------------------
        grad = _subgradient_maxconv(a, conv)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat

        # project back onto the simplex (clip and renormalise)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new / s
        else:
            a_new = a                     # should never happen

        # occasional jitter to escape flat regions
        if t % 5000 == 0:
            jitter = rng.normal(scale=0.002, size=n)   # tiny perturbation
            a_new = np.maximum(a_new + jitter, 0.0)
            total = a_new.sum()
            if total > 0.0:
                a_new = a_new / total
            else:
                a_new = a

        a = a_new

    return best_a, best_score


# --------------------------------------------------------------
# 3️⃣  Mixing with the uniform distribution
# --------------------------------------------------------------

def _mix_with_uniform(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Simple line‑search over λ·a+(1‑λ)·uniform."""
    uniform = np.full(n, 1.0 / n)

    # coarse grid
    for lam in np.linspace(0.0, 1.0, 41):
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        seq /= seq.sum()
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_a = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    return best_a, best_score

    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
```
