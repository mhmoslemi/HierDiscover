"""
search for a non‑negative step function minimizing
    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm (≈ 1000 s budget)

1️⃣ Baseline – uniform vector (score ≈ 2.0) and a short random
   exploration of the 4‑parameter geometric‑power‑tail family
   a_i ∝ r**i·(i+1)**p·(n‑i)**q mixed with a uniform background.
   The best candidate from this stage seeds the next phase.

2️⃣ Smooth Adam descent.
   We minimise a smooth approximation of the max‑convolution:
       Fτ(a) = τ·log Σ_k exp(conv_k(a)/τ) .
   When τ → 0, Fτ → max(conv).  The gradient of Fτ is
       ∇Fτ = 2· (w * a_rev)  with  w_k ∝ exp(conv_k/τ) .
   The gradient is obtained by a single FFT‑based convolution.
   An Adam optimiser (momentum & adaptive steps) is used, while the
   temperature τ is annealed geometrically from a relatively large value
   (≈ 0.2·max conv) down to a very small one (≈ 10⁻⁸·max conv).

3️⃣ Final tiny “mass‑transfer’’ hill‑climb that moves a minute amount of
   probability mass between two coordinates (kept on the simplex).

All vectors are projected onto the probability simplex after each
update.  The routine respects the 1000 s time limit and returns the
best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    n = a.shape[0]
    if L is None:
        L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_fft(u: np.ndarray, v: np.ndarray, L: int | None = None) -> np.ndarray:
    """
    Linear convolution of real vectors u and v via FFT.
    Returns the full length‑(len(u)+len(v)‑1) result.
    """
    n = u.shape[0] + v.shape[0] - 1
    if L is None:
        L = _next_pow2(n)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)[:n]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (assumes sum≈1)
# ----------------------------------------------------------------------


def _eval_fast(a: np.ndarray, L_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    conv = _conv_self_fft(a, L_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Shape generation (geometric‑power‑tail family)
# ----------------------------------------------------------------------


def _make_shape_general(n: int, r: float, p: float, q: float) -> np.ndarray:
    """
    Produce a normalised vector of length n with components
    proportional to r**i·(i+1)**p·(n‑i)**q   (i = 0…n‑1).
    Computed in log‑space for numerical stability.
    """
    i = np.arange(n, dtype=np.float64)
    log_shape = i * np.log(r) + p * np.log(i + 1.0) + q * np.log(n - i)
    log_shape -= np.max(log_shape)               # avoid overflow
    shape = np.exp(log_shape)
    total = shape.sum()
    if total == 0.0 or not np.isfinite(total):
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def _mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Return λ·uniform + (1‑λ)·shape  (both already sum to 1).
    The result is re‑normalised to guard against numerical drift.
    """
    n = shape.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * shape
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Sub‑gradient (active‑set) – fallback if softmax becomes numerically bad
# ----------------------------------------------------------------------


def _compute_subgrad_multi(a: np.ndarray,
                           conv: np.ndarray,
                           tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a, aggregated over every shift
    whose value lies within `tol` of the current maximum.
    """
    max_val = conv.max()
    act = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in act:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]                # ∂/∂a_i from term a_i·a_j
        mask = i_vals != j_vals
        if mask.any():
            g[j_vals[mask]] += a[i_vals[mask]]
    return g


# ----------------------------------------------------------------------
# Random parametric sweep (geometric‑power‑tail family)
# ----------------------------------------------------------------------


def _random_param_search(n: int,
                         deadline: float,
                         start_time: float) -> tuple[np.ndarray,
                                                     float,
                                                     tuple[float, float, float, float]]:
    """
    Uniform‑in‑log random sampling of (r,p,q,λ) and keep the best vector.
    Returns the best sequence, its score and the parameters that produced it.
    """
    rng = np.random.default_rng()
    L_fft = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = float("inf")
    best_params = None

    while time.time() < deadline:
        # r ∈ (0.85, 0.999999) – log‑uniform
        r = np.exp(rng.uniform(np.log(0.85), np.log(0.999999)))
        p = rng.uniform(-3.0, 3.0)
        q = rng.uniform(-3.0, 3.0)
        lam = rng.uniform(0.0, 1.0)

        shape = _make_shape_general(n, r, p, q)
        cand = _mix_with_uniform(shape, lam)
        sc = _eval_fast(cand, L_fft)

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            best_params = (r, p, q, lam)
            print(f"[{time.time() - start_time:.2f}s] random sweep r={r:.4f} "
                  f"p={p:.2f} q={q:.2f} λ={lam:.2f} score={sc:.6f}")

    return best_seq, best_score, best_params


# ----------------------------------------------------------------------
# Tiny random mass‑transfer hill‑climb (final polishing)
# ----------------------------------------------------------------------


def _final_local_search(a: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_score_initial: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a tiny fraction of probability mass between two coordinates,
    keeping the vector on the simplex.  Accepts strictly improving moves,
    occasionally accepts uphill moves to escape plateaus.
    """
    n = a.shape[0]
    L_fft = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = best_score_initial
    rng = np.random.default_rng()

    while time.time() < deadline:
        src = rng.integers(n)
        if a[src] < 1e-15:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer up to 0.2 % of the source mass
        eps = a[src] * rng.random() * 2e-3
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        new_sc = _eval_fast(a_prop, L_fft)
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_prop.copy()
            a = a_prop
            print(f"[{time.time() - start_time:.2f}s] final local score={new_sc:.6f}")
        else:
            # very occasional uphill acceptance
            if rng.random() < 0.001:
                a = a_prop

    return best_a, best_sc

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the target
    evaluation function.  The routine respects the 1000 s time budget and
    returns the best sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 8192 gives a finer resolution while still fast.
    # ------------------------------------------------------------------
    n = 8192
    L_conv = _next_pow2(2 * n - 1)                  # for self‑convolution
    L_grad = _next_pow2(3 * n - 2)                  # for gradient convolution

    # ------------------------------------------------------------------
    # 0️⃣ Uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = uniform.copy()
    best_score = _eval_fast(best_seq, L_conv)
    print(f"[{time.time() - start_time:.2f}s] uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Short random sweep in the 4‑parameter family
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 30.0
    rp_seq, rp_score, rp_params = _random_param_search(
        n, sweep_deadline, start_time)

    if rp_seq is not None and rp_score < best_score:
        best_seq = rp_seq.copy()
        best_score = rp_score
        best_params = rp_params
        print(f"[{time.time() - start_time:.2f}s] after random sweep best score={best_score:.6f}")
    else:
        # fallback parameters – they are not used later, but we keep a placeholder
        best_params = (0.97, 0.0, 0.0, 0.3)

    # ------------------------------------------------------------------
    # 2️⃣ Smooth Adam descent on the soft‑max of the convolution.
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    alpha = 0.5               # base learning‑rate (will be scaled per‑coordinate)

    cur_a = best_seq.copy()
    m = np.zeros_like(cur_a)
    v = np.zeros_like(cur_a)
    t_iter = 0

    # total remaining time for this stage (we leave a few seconds for final local search)
    grad_stage_deadline = deadline - 5.0

    while time.time() < grad_stage_deadline:
        t_iter += 1
        # ------------------------------------------------------------------
        # compute convolution and a temperature τ that is annealed geometrically
        # ------------------------------------------------------------------
        conv = _conv_self_fft(cur_a, L_conv)
        max_c = conv.max()
        # elapsed fraction of the whole budget (0 → 1)
        elapsed = time.time() - start_time
        total_budget = deadline - start_time
        frac = min(elapsed / total_budget, 1.0)

        # τ starts at ~20 % of max_c and ends at ~0.1 % of max_c
        tau_factor = 0.20 * (1.0 - frac) + 0.001
        tau = max(max_c * tau_factor, 1e-8)

        # ------------------------------------------------------------------
        # soft‑max weights w_k ∝ exp((conv_k–max_c)/τ) – numerically safe
        # ------------------------------------------------------------------
        exp_shift = (conv - max_c) / tau
        w = np.exp(exp_shift)
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0.0:
            # fallback to the active‑set sub‑gradient
            g = _compute_subgrad_multi(cur_a, conv, tol=1e-12)
        else:
            w /= w_sum
            # gradient = 2 * (w * a_rev)  ; a_rev = a reversed
            a_rev = cur_a[::-1]
            conv_wr = _conv_fft(w, a_rev, L_grad)
            g = 2.0 * conv_wr[n - 1:n - 1 + n]

        # ------------------------------------------------------------------
        # Adam update (projected onto the simplex after the step)
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = alpha / (np.sqrt(v_hat) + eps)

        a_new = cur_a - step * m_hat
        a_new = _project_simplex(a_new)

        cur_score = _eval_fast(a_new, L_conv)

        # keep track of the overall best vector
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            # reset the moments to avoid carrying stale velocity from a poor region
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{time.time() - start_time:.2f}s] adam it={t_iter:,} "
                  f"score={cur_score:.6f} τ={tau:.2e}")

        # Always move the optimisation state forward (even if the step was bad)
        cur_a = a_new

        # occasional tiny random jitter to shake out of flat plateaus
        if t_iter % 2500 == 0:
            jitter = 1e-4 * rng.random(n) * cur_a
            cur_a = np.maximum(cur_a + jitter, 0.0)
            cur_a = _project_simplex(cur_a)

    # ------------------------------------------------------------------
    # 3️⃣ Final tiny random mass‑transfer hill‑climb
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_score = _final_local_search(best_seq,
                                                   deadline,
                                                   start_time,
                                                   best_score)

    # ------------------------------------------------------------------
    # Normalise finally (sum → 1) and return a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Log the official evaluation (uses the provided security‑checked function)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time() - start_time:.2f}s] evaluate_sequence => {final_val:.6f}")
    except Exception as exc:
        print(f"Unable to call evaluate_sequence: {exc}")

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")

    # return as plain list of Python floats
    return [float(v) for v in best_seq]
