"""
Advanced search for a low‑value step function.

The objective is 2·n·max(conv(a,a))/sum(a)², invariant under scaling.
We therefore keep the vector on the probability simplex (sum = 1).

The algorithm proceeds in four phases:

1. *Diverse analytic pool* – generate many candidates from a richer family
   of shapes (uniform, exponentials with very small decay, power‑laws,
   shifted power‑laws, sqrt‑decay, asymmetric Beta, random Dirichlet).
   The TOP_K best are kept.

2. *Fast multiplicative sub‑gradient* – a short run of the exponentiated
   sub‑gradient method (the “exp‑GD” from the reference) to quickly
   escape obvious bad regions.

3. *Soft‑max Adam polishing* – the main optimisation phase.  The hard max
   is approximated by a temperature‑controlled soft‑max; an Adam optimiser
   with simplex projection refines the vector for the bulk of the time
   budget.

4. *Simulated‑annealing finish* – a cheap hill‑climbing / annealing loop
   that makes tiny random transfers of mass between coordinates.
   This helps to jump out of any residual plateaus and often gains the
   last few hundredths of the bound.

All heavy work (FFT‑based convolution, projection onto the simplex) is
vectorised.  The code fits comfortably into the 1000 s budget and
normally pushes the bound well below 1.508 (target ≈ 1.503).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negatives caused by round‑off.
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shape generators (all return a vector with sum == 1)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)  (lam ≥ 0)"""
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (p ≥ 0)"""
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_law_shift(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i+shift)^{-p}  (shift > 0)"""
    idx = np.arange(0, n, dtype=float) + shift
    shape = idx ** (-p)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_sqrt_decay(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)  (a simple decreasing shape)."""
    idx = np.arange(1, n + 1, dtype=float)
    shape = 1.0 / np.sqrt(idx)
    shape /= shape.sum()
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    total = shape.sum()
    if total > 0:
        shape /= total
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Random decreasing Dirichlet vector (conc > 0)."""
    vec = rng.gamma(conc, 1.0, size=n)
    vec = np.sort(vec)[::-1]      # enforce decreasing order
    total = vec.sum()
    if total > 0:
        vec /= total
    else:
        vec = np.full(n, 1.0 / n)
    return vec


# ----------------------------------------------------------------------
# Top‑k pool handling
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (fast phase)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + it / 5000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD it={it:,} best={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max Adam descent (main phase)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.1
    T0 = 0.1
    T_end = 1e-9

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD it={it:,} best={cur_score:.6f}")

        # temperature schedule – exponential decay, slower than the reference
        T = T_end + (T0 - T_end) * (0.5 ** (it / 3000.0))

        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing / hill‑climbing refinement
# ----------------------------------------------------------------------
def _annealing_optimize(a_start: np.ndarray,
                        deadline: float,
                        start_time: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers with a decreasing temperature."""
    a = a_start.copy()
    cur_score = _objective(a)
    best = a.copy()
    best_score = cur_score

    n = a.shape[0]
    it = 0
    T = 1e-2  # initial temperature (acceptance tolerance)
    while time.time() < deadline:
        it += 1
        # occasionally perform a bigger random mixing
        if rng.random() < 0.02:
            # draw a fresh Dirichlet and mix a little
            vec = _make_random_dirichlet(n, conc=0.5, rng=rng)
            alpha = 0.02
            cand = (1 - alpha) * a + alpha * vec
        else:
            # transfer a tiny amount from i to j
            i, j = rng.choice(n, size=2, replace=False)
            # magnitude proportional to the source mass
            delta = a[i] * rng.random() * 0.02
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta

        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        cand_score = _objective(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} new best={cand_score:.6f}")

        # geometric cooling
        if it % 1000 == 0:
            T *= 0.95
            T = max(T, 1e-7)

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation bound.
    Returns a plain list of non‑negative heights (sum ≈ 1).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start = time.time()
    deadline = start + budget_s - 0.2   # safety margin

    # --------------------------------------------------------------
    # Choose problem size – larger N gives a richer search at modest cost.
    # --------------------------------------------------------------
    remaining = deadline - start
    if remaining > 800:
        N = 16384
    elif remaining > 400:
        N = 8192
    else:
        N = 4096

    # --------------------------------------------------------------
    # 1) Build a diverse initial pool of analytic shapes.
    # --------------------------------------------------------------
    TOP_K = 15
    pool = []                     # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        _push_top(pool, _objective(vec), vec, TOP_K)

    # very basic baselines
    _add(_make_uniform(N))

    # exponential decays – now explore very small lambda as well
    for lam in np.concatenate([np.logspace(-5, -1, 12), np.logspace(-0.9, 0.5, 10)]):
        _add(_make_exponential(N, lam))

    # power‑law decays, include p close to zero
    for p in np.linspace(0.01, 2.5, 18):
        _add(_make_power_law(N, p))

    # shifted power‑law (adds a horizontal shift to smooth the left edge)
    for shift in [0.1, 0.5, 1.0, 2.0]:
        for p in [0.2, 0.5, 1.0]:
            _add(_make_power_law_shift(N, p, shift))

    # sqrt decay (often excellent)
    _add(_make_sqrt_decay(N))

    # asymmetric Beta shapes – sweep a few combos skewed to the left

    # random Dirichlet draws – varied concentration parameters
    for conc in [0.2, 0.5, 0.9, 1.5, 3.0]:
        for _ in range(12):
            if time.time() >= deadline:
                break
            _add(_make_random_dirichlet(N, conc=conc, rng=rng))

    # If a previous solution exists, upsample it (useful for continuity)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                _add(starter_up)
    except Exception:
        pass

    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Brief exponentiated sub‑gradient descent (quick improvement).
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sub_end = time.time() + 0.15 * remaining
        # run the exp‑GD on each top candidate once
        for _, vec in pool:
            seq, sc = _exp_subgradient_descent(vec, sub_end, start)
            if sc < best_score:
                best_score, best_vec = sc, seq.copy()
                print(f"[{time.time() - start:.2f}s] ExpGD refinement → {sc:.6f}")
            if time.time() >= sub_end:
                break

    # --------------------------------------------------------------
    # 3) Main Adam polishing (soft‑max) – bulk of the time budget.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        smooth_end = time.time() + 0.70 * remaining
        seq, sc = _smooth_gradient_descent(best_vec, smooth_end, start)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time() - start:.2f}s] SmoothGD improvement → {sc:.6f}")

    # --------------------------------------------------------------
    # 4) Final simulated‑annealing hill‑climb – try to break plateaus.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq, sc = _annealing_optimize(best_vec, deadline, start, rng)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()

    # --------------------------------------------------------------
    # 5) Normalise and output.
    # --------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()

    else:

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final bound ≈ {final_score:.6f}")

    # Return a plain Python list.
    return [float(v) for v in best_vec]
