def _projected_gradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_M = conv.max()
        cur_score = 2.0 * n * cur_M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # maybe print

        # compute subgradient of max conv
        grad = _subgradient_maxconv(a, conv)
        # make direction zero-sum
        d = -(grad - grad.mean())
        # compute max feasible step
        # for indices where d<0, we can only move at most a_i/(-d_i)
        # for d>0, no bound because increasing a_i may produce sum>1? but sum(d)=0, so overall remains constant.
        # compute max step:
        idx_neg = np.where(d < 0)[0]
        if idx_neg.size > 0:
            max_step = np.min(-a[idx_neg] / d[idx_neg])
        else:
            max_step = 1.0
        # start with step = max_step * 0.5 (or perhaps max_step * 0.9)
        step = max_step * 0.8
        # line search:
        while step > 1e-12:
            a_new = a + step * d
            # ensure nonneg and sum close to 1 (should be)
            if np.any(a_new < 0):
                step *= 0.5
                continue
            # compute score:
            conv_new = _conv_self(a_new)
            score_new = 2.0 * n * conv_new.max()
            if score_new <= cur_score:
                a = a_new
                break
            else:
                step *= 0.5
        # if step <= 1e-12 break maybe
        if step <= 1e-12:
            # maybe break out or just continue with small random perturb to escape
            # Random reinitialization occasional?
            # Continue with random jitter
            # random tiny perturb
            eps = 1e-6
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0, None)
            a /= a.sum()
    return best_seq, best_score

def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    rng = np.random.default_rng()
    cur_score = best_score
    cur_a = a.copy()
    # temperature schedule: T0 maybe (cur_score - 1.5) * 0.5? But we can set relative.
    # We'll compute total time left = deadline - time.time()
    # We'll set T0 = 0.005 * cur_score maybe.
    total_time = deadline - time.time()
    start = time.time()
    while time.time() < deadline:
        # compute progress fraction
        elapsed = time.time() - start
        frac = elapsed / total_time
        T = max(0.0, (1.0 - frac)) * cur_score * 0.001  # start T = 0.001*score, linearly decay
        # choose src-high grads: compute grad each few steps maybe
        # compute grad periodically e.g., every 100 steps
        if int(elapsed*1000) % 100 == 0:
            conv = _conv_self(a)
            grad = _subgradient_maxconv(a, conv)
        # choose src as weighted by grad (higher)
        src = rng.choice(n, p=np.maximum(grad,0)/np.maximum(grad,0).sum())
        # choose dst as weighted by inverse grad maybe or just random small weight
        dst = rng.integers(n)
        if src == dst:
            continue
        # transfer fraction
        max_frac = 0.3
        frac_transfer = rng.random() * max_frac
        eps = a[src] * frac_transfer
        if eps < 1e-12:
            continue
        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        # renormalize not needed because sum unchanged
        # compute objective
        conv_new = _conv_self(a_prop)
        score_new = 2.0 * n * conv_new.max()
        delta = score_new - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            a = a_prop
            cur_score = score_new
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                # print progress
        # occasional random restart after many non-improving steps
        # Not needed
    return best_seq, best_score

"""
Propose a candidate step‑function sequence that (almost) minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds in four stages (all time‑controlled to stay under the
1000 s budget)

1. Broad parametric sweep (geometric, power, (a)symmetric Beta, symmetric
   exponential, two‑block and a new skewed‑exponential family).

2. Random Dirichlet draws with a wide range of concentration parameters.

3. Projected gradient descent on the simplex using the exact sub‑gradient of
   max(conv).  A back‑tracking line‑search guarantees feasibility and
   monotone improvement.

4. Simulated‑annealing hill‑climb that repeatedly transfers a fraction of the
   mass from a high‑gradient index to a low‑gradient index; occasional
   uphill moves are accepted with a temperature that linearly cools.

All convolutions are performed by a fast FFT routine; the simplex projection
is O(n).  The best sequence found is normalised to sum = 1 and returned as a
plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad

def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 300)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.05, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_seq = (a - 1.0) * np.log(idx + 1.0) + (a - 1.0) * np.log(np.maximum(n - idx, 1))
        seq = np.exp(log_seq - np.max(log_seq))
        s = seq.sum()
        if s == 0 or not np.isfinite(s):
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymBeta α={a:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.1, 5.0, 21)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay: a_i ∝ exp(-λ·min(i, n‑1‑i))."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 100):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends with a tunable mass split."""
    for L_frac in np.linspace(0.01, 0.4, 40):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.1, 0.9, 9):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_skewed_exponential(n, best_seq, best_score, deadline):
    """
    Piecewise exponential: left side uses λ₁, right side uses λ₂.
    """
    mid = n // 2
    left_idx = np.arange(mid, dtype=np.float64)
    right_idx = np.arange(mid, n, dtype=np.float64)
    for lam_left in np.logspace(-4, -0.2, 30):
        for lam_right in np.logspace(-4, -0.2, 30):
            if time.time() >= deadline:
                break
            seq = np.empty(n, dtype=np.float64)
            seq[:mid] = np.exp(-lam_left * left_idx)
            seq[mid:] = np.exp(-lam_right * (n - 1 - right_idx))
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] SkewExp λL={lam_left:.1e} λR={lam_right:.1e} score={sc:.6f}")
    return best_seq, best_score

def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 800  # many more draws than the previous code
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)      # concentration ≈ 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:

        if sc < best_score:

    return best_seq, best_score

# 3️⃣ Projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _projected_gradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] PGD it={it:,} score={cur_score:.6f}")
            if cur_score <= 1.503:
                # reached target – can stop early
                break

        grad = _subgradient_maxconv(a, conv)
        # descent direction respecting sum‑zero
        d = -(grad - grad.mean())

        # largest feasible step without violating non‑negativity
        idx_neg = np.where(d < 0)[0]
        if idx_neg.size:
            max_step = np.min(-a[idx_neg] / d[idx_neg])
        else:
            max_step = 1.0
        # start with a fairly aggressive step
        step = max_step * 0.9

        # back‑tracking line search
        while step > 1e-14:
            a_new = a + step * d    # sum stays 1 by construction
            if np.any(a_new < 0):
                step *= 0.5
                continue
            conv_new = _conv_self(a_new)
            score_new = float(2.0 * n * conv_new.max())
            if score_new <= cur_score:
                a = a_new
                break
            step *= 0.5

        # if we cannot make progress, inject a tiny random jitter
        if step <= 1e-14:
            eps = 1e-7
            a = a + np.random.uniform(-eps, eps, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
    return best_seq, best_score

def _simulated_annealing(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_score = best_score
    cur_a = a.copy()

    # total time left for this phase
    total = max(deadline - time.time(), 1e-6)
    start = time.time()
    it = 0
    no_improve = 0

    # Pre‑compute a few gradients to guide source selection
    grad = _subgradient_maxconv(a, _conv_self(a))

    while time.time() < deadline:
        it += 1
        # linear cooling schedule
        elapsed = time.time() - start
        frac = elapsed / total
        T = (1.0 - frac) * cur_score * 0.001   # start temperature ~0.1 % of the score

        # pick source (high gradient) and destination (low gradient)
        # update the gradient only every ~100 iterations to save time
        if it % 100 == 0:
            grad = _subgradient_maxconv(a, _conv_self(a))
            pos = grad.max()
            neg = grad.min()
        # probability proportional to gradient magnitude (shifted to be positive)
        prob = np.maximum(grad, 0)
        prob_sum = prob.sum()
        if prob_sum > 0:
            src = rng.choice(n, p=prob / prob_sum)
        else:
            src = rng.integers(n)
        # destination – pick a low‑gradient index (could be uniform random)
        dst = rng.integers(n)
        if src == dst:
            continue

        # transfer a fraction of the source mass
        max_frac = 0.25                # allow larger jumps than the simple hill‑climb
        frac_transfer = rng.random() * max_frac
        eps = a[src] * frac_transfer
        if eps <= 1e-14:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps

        # sum unchanged, non‑negativity guaranteed by construction
        conv_new = _conv_self(a_prop)
        score_new = float(2.0 * n * conv_new.max())
        delta = score_new - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            a = a_prop
            cur_score = score_new
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                no_improve = 0
                print(f"[{time.time() - start_time:.2f}s] SA it={it:,} score={best_score:.6f}")
                if best_score <= 1.503:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random shake‑up if stuck
        if no_improve > 5000:
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps2 = a[i1] * (rng.random() * max_frac)
            a[i1] -= eps2
            a[i2] += eps2
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = float(2.0 * n * _conv_self(a).max())
            no_improve = 0

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine consists of a parametric sweep (including a new skewed‑exponential
    family), random Dirichlet draws, a projected gradient descent, and finally a
    simulated‑annealing hill‑climb.  The best vector (normalised to sum = 1) is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # --------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)  # type: ignore
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_score, best_seq = sc, starter.copy()
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣ Broad parametric sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0  # reserve ~5 min for later phases
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_skewed_exponential(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣ Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣ Projected gradient descent
    # --------------------------------------------------------------
    pgd_deadline = deadline - 150.0
    if time.time() < pgd_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting projected gradient descent")
        best_seq, best_score = _projected_gradient_descent(best_seq, pgd_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After PGD best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting simulated‑annealing hill‑climb")
        best_seq, best_score = _simulated_annealing(best_seq, deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] Final SA score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:

    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
