def _local_annealing(a, deadline, start):
    best = a.copy()
    best_score = _objective(best)
    n = len(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving epsilon from i to j
        epsilon = best[i] * rng.uniform(0.001, 0.01) # small fraction
        if epsilon > best[i]:
            continue
        candidate = best.copy()
        candidate[i] -= epsilon
        candidate[j] += epsilon
        # keep nonnegativity
        cur = _objective(candidate)
        if cur < best_score:
            best = candidate
            best_score = cur
        # optionally accept worse with small probability?
    return best, best_score

"""
Hybrid search for a low‑score step‑function.
-------------------------------------------------
1.  Build a diverse pool of analytic and random shapes on a modest
    size (N_small).  Keep the TOP_K candidates (by the true objective).
2.  Run a *soft‑max Adam* (smooth approximation of the max‑convolution)
    on each top candidate for as long as time allows.
    - larger initial temperature and a larger base learning rate broaden
      exploration compared with the reference code.
3.  Upsample the best small‑size vector to the target size N_final and
    polish it with a short Adam run followed by a tiny local annealing
    that swaps tiny amounts of mass between two coordinates.
4.  Return the final probability vector (non‑negative, sum = 1) as a plain
    Python list – the evaluator will convert it to the required format.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return np.maximum(conv, 0.0)          # eliminate tiny negative artefacts


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Analytic shape generators (work for any length n)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum() if shape.sum() > 0 else 1.0
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    shape /= shape.sum() if shape.sum() > 0 else 1.0
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    shape /= shape.sum() if shape.sum() > 0 else 1.0
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape /= shape.sum() if shape.sum() > 0 else 1.0
    return shape

# Optimisation primitives
# ----------------------------------------------------------------------
def _smooth_gradient_descent_custom(a0: np.ndarray,
                                    deadline: float,
                                    start_time: float,
                                    *,
                                    step0: float = 0.2,
                                    T0: float = 0.3,
                                    T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max approximation of max(conv(a,a)).
    Parameters ``step0``, ``T0`` and ``T_end`` control learning‑rate
    and temperature schedule; larger ``T0`` and ``step0`` give a
    more exploratory start.
    """
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD (T0={T0}) iter={it:,} best={cur_score:.6f}")

        # temperature annealing (exponential decay)
        T = T_end + (T0 - T_end) * (0.5 ** (it / 4000.0))

        # soft‑max weights over the convolution vector
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        # gradient = 2 * correlation(w, a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # keep on the simplex
        a = _project_simplex(a)

    return best, best_score


def _local_annealing(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     max_move_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: repeatedly move a tiny fraction of
    mass from a randomly chosen source index to a random target index.
    Accept only improvements (strict descent); this is sufficient for a
    final polishing phase lasting a few seconds.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a)
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] < 1e-12:
            continue
        # move a small fraction from i to j
        frac = rng.uniform(1e-4, max_move_frac)
        delta = frac * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        # numerical drift – renormalise (preserves sum ≈ 1)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum
        cur = _objective(cand)
        if cur < best_score:
            best_score = cur
            best = cand
            a = cand   # continue local search from the improved point
            # occasional log
            if it % 1000 == 0:
                print(f"[{time.time() - start_time:.2f}s] Anneal iter={it:,} best={cur:.6f}")
    return best, best_score

    for p in np.linspace(0.2, 3.5, 11):
        for plat in np.linspace(0.0, 0.4, 3):
            _add(_make_power_ramp(N_small, p, plat))

    for lam in np.linspace(0.5, 5.0, 12):
        _add(_make_exponential(N_small, lam))

    for p in np.linspace(0.5, 4.0, 12):
        _add(_make_power_law(N_small, p))

    for a in np.linspace(0.2, 4.0, 7):
        for b in np.linspace(0.2, 4.0, 7):
            _add(_make_beta_shape_asym(N_small, a, b))

    # Random Dirichlet draws – we keep generating while we have time.
    rng = np.random.default_rng(seed + 7777)
    while time.time() < deadline - 30.0:
        # draw a concentration parameter on a log‑scale
        conc = np.exp(rng.uniform(np.log(0.02), np.log(3.0)))
        vec = rng.gamma(conc, 1.0, size=N_small)
        # two flavours: sorted (monotone) and unsorted
        for sorted_flag in (True, False):
            cand = np.sort(vec)[::-1] if sorted_flag else vec.copy()
            s = cand.sum()
            if s > 0:
                cand /= s
                _add(cand)

    # Seed with the previously known construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != N_small:
                x_old = np.linspace(0, 1, prev.size, endpoint=False)
                x_new = np.linspace(0, 1, N_small, endpoint=False)
                prev_small = np.interp(x_new, x_old, prev)
            else:
                prev_small = prev.copy()
            prev_small = np.maximum(prev_small, 0.0)
            s = prev_small.sum()
            if s > 0:
                prev_small /= s
                _add(prev_small)
    except Exception:
        pass

    pool.sort(key=lambda x: x[0])
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Soft‑max Adam descent on the small problem.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # we devote about 70 % of the remaining budget to this phase
        descent_deadline = time.time() + 0.70 * remaining
        idx = 0
        while time.time() < descent_deadline:
            vec = pool[idx % len(pool)][1]
            seq, sc = _smooth_gradient_descent_custom(
                vec,
                descent_deadline,
                start,
                step0=0.3,   # a slightly larger base step than the reference
                T0=0.5,
                T_end=1e-9,
            )
            if sc < best_score:
                best_score, best_vec = sc, seq.copy()
                print(f"[{time.time() - start:.2f}s] Adam improvement → {sc:.6f}")
            idx += 1

    # ------------------------------------------------------------------
    # 3) Upsample to the final length and polish.
    # ------------------------------------------------------------------
    if N_final != N_small:
        x_small = np.linspace(0, 1, N_small, endpoint=False)
        x_final = np.linspace(0, 1, N_final, endpoint=False)
        best_up = np.interp(x_final, x_small, best_vec)
    else:
        best_up = best_vec

    best_up = np.maximum(best_up, 0.0)
    s = best_up.sum()
    best_up = best_up / s if s > 0 else np.full(N_final, 1.0 / N_final)

    # final Adam polishing (shorter horizon)
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq, sc = _smooth_gradient_descent_custom(
            best_up,
            deadline,
            start,
            step0=0.2,
            T0=0.3,
            T_end=1e-9,
        )
        if sc < best_score:
            best_score, best_up = sc, seq.copy()
            print(f"[{time.time() - start:.2f}s] Final Adam polishing → {sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Very cheap local annealing to possibly shave off the last
    #    fractions of the bound.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        seq, sc = _local_annealing(best_up, deadline, start, max_move_frac=0.015)
        if sc < best_score:
            best_score, best_up = sc, seq.copy()
            print(f"[{time.time() - start:.2f}s] Local annealing improvement → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_up = np.maximum(best_up, 0.0)
    total = best_up.sum()
    if total > 0:
        best_up /= total
    else:
        best_up = np.full(N_final, 1.0 / N_final)

    final_score = _objective(best_up)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    # Return the result as a Python list for the external evaluator.
    return [float(v) for v in best_up]
