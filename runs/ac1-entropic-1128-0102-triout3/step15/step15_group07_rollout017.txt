"""
Hybrid optimiser for the step‑function attack.

Key ideas
---------
*   Use a modest problem size (N≈8 k) – this lets many more
    optimisation steps be performed within the 1000 s budget.
*   Initialise a diverse pool (uniform, Beta, Hann, Gaussian,
    random Dirichlet) and keep the K best candidates.
*   First phase – fast sub‑gradient descent (exact gradient of the
    max‑convolution).  This is cheap because only the indices attaining
    the current maximum need to be examined.
*   Second phase – smooth‑max descent.  The true objective
    `max(conv)` is replaced by a log‑sum‑exp surrogate,
    providing a smooth gradient that moves mass away from the current
    peaks more effectively.  The surrogate temperature (β) is annealed
    from a small to a large value.
*   Final polishing – a light simulated‑annealing (mass‑swap) pass.

All updates stay on the probability simplex via Euclidean projection.
The optimiser respects the 1000 s budget, prints progress and finally
returns a plain Python list of non‑negative numbers.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]               # last true index
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic shapes used for the initial pool
# ----------------------------------------------------------------------
def _make_shape(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Power‑ramp with optional symmetric plateau."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= shape.sum()
    return shape


def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    shape /= shape.sum()
    return shape


def _make_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    shape /= shape.sum()
    return shape


def _make_hann_shape(n: int) -> np.ndarray:
    """Standard Hann window (cosine‑bell)."""
    x = np.arange(n, dtype=float)
    shape = 0.5 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
    shape[0] = shape[-1] = 0.0               # enforce zero at the ends
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Objective (true) – used for reporting and final evaluation
# ----------------------------------------------------------------------
def _true_objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – the value returned by evaluate_sequence when sum=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient descent (exact max‑convolution gradient)
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     rng: np.random.Generator,
                     step0: float = 200.0,
                     max_move_frac: float = 0.01) -> tuple[np.ndarray, float]:
    """Classical exponentiated‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _true_objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] sub‑grad iter={it:,} best={cur_score:.6f}")

        # --- sub‑gradient (only indices attaining the maximum) ---
        ks = np.where(conv >= cur_max - 1e-14)[0]

        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += 2.0 * a[k - idx]

        # exponentiated‑gradient update
        step = step0 / math.sqrt(it)
        expo = np.exp(-step * g)
        a = a * expo
        a = a / a.sum()

        # occasional random mass move (helps escape plateaus)
        if it % 3000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * max_move_frac * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑max descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    rng: np.random.Generator,
                    step0: float = 1.0,
                    beta0: float = 4.0,
                    beta_max: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Gradient descent on Sβ(a) = (1/β)·log Σ exp(β·conv_k).
    β is annealed from β0 to β_max.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _true_objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # ---- compute convolution (exact) ---------------------------------
        conv = _conv_self(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time() - start_time:.2f}s] smooth iter={it:,} best={cur_score:.6f}")

        # ---- temperature schedule ---------------------------------------
        t_frac = (time.time() - start_time) / (deadline - start_time)
        beta = beta0 + t_frac * (beta_max - beta0)

        # ---- soft‑max weights -------------------------------------------
        # shift by max to avoid overflow
        w = np.exp(beta * (conv - cur_max))
        w /= w.sum()

        # ---- gradient of the smooth objective ---------------------------
        # g = 2 * (a * w) , where * denotes linear convolution
        L = _next_pow2(n + (2 * n - 1) - 1)          # length for convolution a (n) + w (2n‑1) -1
        Af = np.fft.rfft(a, n=L)
        Wf = np.fft.rfft(w, n=L)
        conv_aw = np.fft.irfft(Af * Wf, n=L)
        # central part corresponds to indices 0…n‑1
        grad = 2.0 * conv_aw[n - 1: n - 1 + n]

        # ---- step size (simple 1/√t schedule) --------------------------
        step = step0 / math.sqrt(it)

        # ---- additive update + projection onto simplex -------------------
        a = a - step * grad
        a = _project_simplex(a)

        # ---- occasional random perturbation -------------------------------
        if it % 2000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * 0.02 * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polish
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.03) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _true_objective(a)
    best_a = a.copy()
    best_score = cur_score

    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        frac = min(1.0, (time.time() - start_time) / (deadline - start_time))
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta
        a_prop = _project_simplex(a_prop)

        new_score = _true_objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time() - start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to keep the K best candidates (sorted by score)
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a plain Python list of non‑negative floats (sum = 1).
    """
    rng = np.random.default_rng(seed + 12345)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # Choose a problem size – smaller N gives many more optimisation steps.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 700:
        N = 8192            # fast enough to do many iterations
    elif remaining > 300:
        N = 4096
    else:
        N = 2048

    # --------------------------------------------------------------
    # 1) Initialise a pool of diverse candidates
    # --------------------------------------------------------------
    TOP_K = 8
    pool = []   # (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _true_objective(uniform), uniform, TOP_K)

    # power‑ramps
    for p in np.linspace(0.2, 2.0, 7):
        for plat in np.linspace(0.0, 0.4, 5):
            a = _make_shape(N, p, plat)
            _push_top(pool, _true_objective(a), a, TOP_K)

    # symmetric Beta shapes
    for alpha in np.arange(0.05, 1.00, 0.10):
        a = _make_beta_shape(N, alpha)
        _push_top(pool, _true_objective(a), a, TOP_K)

    # Gaussian‑like bumps
    for sigma in [N * 0.04, N * 0.08, N * 0.16]:
        a = _make_gaussian_shape(N, sigma)
        _push_top(pool, _true_objective(a), a, TOP_K)

    # Hann window (cosine bell)
    hann = _make_hann_shape(N)
    _push_top(pool, _true_objective(hann), hann, TOP_K)

    # a few random Dirichlet draws
    concs = [0.02, 0.08, 0.2, 0.5]
    for conc in concs:
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = rng.gamma(conc, 1.0, size=N)
            a /= a.sum()
            _push_top(pool, _true_objective(a), a, TOP_K)

    # --------------------------------------------------------------
    # If a previous best is available, inject it (interpolate if needed)
    # --------------------------------------------------------------
    try:
        prev = globals().get('height_sequence_1')
        if prev is not None:
            prev = np.asarray(prev, dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                if prev.size != N:
                    x_old = np.linspace(0, 1, prev.size, endpoint=False)
                    x_new = np.linspace(0, 1, N, endpoint=False)
                    prev = np.interp(x_new, x_old, prev)
                prev = np.maximum(prev, 0.0)
                if prev.sum() > 0:
                    prev = prev / prev.sum()
                    _push_top(pool, _true_objective(prev), prev, TOP_K)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Pick current best candidate
    # --------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_score, best_seq = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start_time:.2f}s] Initial best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Short sub‑gradient descents on the whole pool (exploration)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # give each pooled vector a tiny budget (≈5 % of the remaining time)
        per_candidate = max(5.0, 0.05 * remaining / len(pool))
        for _, vec in pool:
            sub_dead = time.time() + per_candidate
            cand, cand_score = _subgrad_descent(vec, sub_dead, start_time, rng,
                                                step0=300.0, max_move_frac=0.02)
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"[{time.time() - start_time:.2f}s] After quick SG best = {best_score:.6f}")
            if time.time() >= deadline:
                break

    # --------------------------------------------------------------
    # 3) Main sub‑gradient descent on the current best (majority of time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_dead = time.time() + 0.65 * remaining
        refined, refined_score = _subgrad_descent(best_seq, sub_dead,
                                                  start_time, rng,
                                                  step0=250.0, max_move_frac=0.02)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined.copy()
            print(f"[{time.time() - start_time:.2f}s] After long SG best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Smooth‑max descent (fine‑grained polishing)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 4.0:
        smooth_dead = time.time() + 0.30 * remaining
        smooth, smooth_score = _smooth_descent(best_seq, smooth_dead,
                                               start_time, rng,
                                               step0=1.2,
                                               beta0=4.0,
                                               beta_max=250.0)
        if smooth_score < best_score:
            best_score, best_seq = smooth_score, smooth.copy()
            print(f"[{time.time() - start_time:.2f}s] After smooth descent best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Optional short SA polish (if any time left)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_score = _simulated_annealing(best_seq, deadline,
                                                    start_time, rng,
                                                    max_move_frac=0.035)

    # --------------------------------------------------------------
    # 6) Final normalisation and conversion to plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full(N, 1.0 / N)

    final_score = _true_objective(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Finished – final score ≈ {final_score:.6f}")

    # expose the final result for a possible next run
    globals()['height_sequence_1'] = best_seq.copy()

    return [float(v) for v in best_seq]
