"""
Evolutionary optimisation of a step‑function for the bound

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in three stages:

1. **Initial diverse pool** – a set of analytic shapes (uniform, ramps,
   exponentials, power‑laws, asymmetric Beta, random Dirichlet draws) is
   built and the best K are retained.

2. **Evolutionary search** – a simple (μ + λ) EA on the probability simplex:
   elite individuals are selected, crossed‑over and mutated (multiplicative
   log‑Gaussian noise), re‑normalised and evaluated.  The population size is
   modest (≈30) so that many generations fit into the time budget.
   The mutation step‑size is linearly annealed to focus the search.

3. **Local polishing** – the current best individual is refined first by a
   few hundred iterations of exponentiated sub‑gradient descent (active‑set
   ℓ∞ optimisation) and then by an Adam optimiser on a smooth‑max surrogate.

All convolutions are performed with a real FFT (O(n log n)) and the code
respects the 1000 s budget, keeping a small safety margin.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    # Clip tiny negatives caused by numerical noise.
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Analytic shape generators (all return a normalized probability vector)
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    """Symmetric power‑ramp with exponent p and plateau of relative length."""
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        elif shape.size > n:
            shape = shape[:n]
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    tot = shape.sum()
    if tot > 0:
        shape /= tot
    else:
        shape = np.full(n, 1.0 / n)
    return shape

# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent on max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    it = 0
    eta0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2 * (w ⋆ a_rev)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) *
                            np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1: n - 1 + n]

        eta = eta0 / (1.0 + it / 5000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best, best_score

# Smooth‑max (soft‑max) Adam descent
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.1
    T0 = 0.1
    T_end = 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        T = T_end + (T0 - T_end) * (0.5 ** (it / 4000.0))
        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) *
                            np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1: n - 1 + n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best, best_score

    Returns a list of non‑negative heights (normalised to sum = 1).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.05  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger N gives finer resolution, but slower)
    # ------------------------------------------------------------------
    remaining_total = deadline - start
    if remaining_total > 800:
        N = 16384
    elif remaining_total > 400:
        N = 8192
    else:
        N = 4096

    # ------------------------------------------------------------------
    # 1) Build an initial diverse pool of analytic shapes
    # ------------------------------------------------------------------
    POP_SIZE = 30            # total population size
    TOP_K = POP_SIZE         # keep the whole pool sorted
    pool = []                # list of (score, vector)

    def _add(vec: np.ndarray) -> None:
        # ensure positivity and normalise
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s > 0:
            vec /= s
        else:
            vec = np.full(N, 1.0 / N)
        _push_top(pool, _objective(vec), vec, TOP_K)

    # Uniform baseline
    _add(_make_uniform(N))

    # Power ramps (symmetric)
    for p in np.linspace(0.2, 4.0, 9):
        for plat in np.linspace(0.0, 0.4, 3):
            _add(_make_power_ramp(N, p, plat))

    # Exponential decays
    for lam in np.linspace(0.2, 5.0, 12):
        _add(_make_exponential(N, lam))

    # Power‑law decays
    for p in np.linspace(0.5, 5.0, 12):
        _add(_make_power_law(N, p))

    # Asymmetric Beta families
    alphas = np.linspace(0.2, 5.0, 6)
    betas = np.linspace(0.2, 5.0, 6)
    for a in alphas:
        for b in betas:
            _add(_make_beta_shape_asym(N, a, b))

    # Random Dirichlet draws (sorted decreasing to give a “convex” shape)
    rng = np.random.default_rng(seed + 777)
    for conc in [0.05, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(20):
            if time.time() >= deadline:
                break
            vec = rng.gamma(conc, 1.0, size=N)
            vec = np.sort(vec)[::-1]    # enforce a decreasing envelope
            _add(vec)

    # Incorporate the previous best sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != N:
                x_old = np.linspace(0.0, 1.0, prev.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, N, endpoint=False)
                prev_up = np.interp(x_new, x_old, prev)
            else:
                prev_up = prev.copy()
            _add(prev_up)
    except Exception:
        pass

    # Ensure we have a full population (fill with random Dirichlet if needed)
    while len(pool) < POP_SIZE:
        vec = rng.dirichlet(np.ones(N))
        _add(vec)

    # Store the current best
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Evolutionary search (simple (μ+λ) EA)
    # ------------------------------------------------------------------
    ELITE = max(5, POP_SIZE // 6)   # number of elites that survive unchanged
    sigma0 = 0.5                    # initial multiplicative noise scale
    # Keep some time for the final polishing
    polishing_reserved = 20.0
    while time.time() < deadline - polishing_reserved:
        # ----- parent selection (uniform among elites) -----
        elite_inds = np.random.choice(range(ELITE), size=2, replace=False)
        parent1 = pool[elite_inds[0]][1]
        parent2 = pool[elite_inds[1]][1]

        # ----- crossover -----
        t = np.random.rand()
        child = t * parent1 + (1.0 - t) * parent2

        # ----- mutation (multiplicative log‑Gaussian) -----
        progress = (time.time() - start) / budget_s
        sigma = sigma0 * (1.0 - progress)  # linear cooling
        if sigma < 1e-4:
            sigma = 1e-4
        noise = np.exp(sigma * np.random.randn(N))
        child *= noise

        # ----- renormalise to the simplex -----
        child = np.maximum(child, 1e-15)
        s = child.sum()
        if s > 0:
            child /= s
        else:
            child = np.full(N, 1.0 / N)

        # ----- evaluation -----
        child_score = _objective(child)

        # ----- insertion into population (replace worst if better) -----
        if child_score < pool[-1][0]:
            pool[-1] = (child_score, child.copy())
            pool.sort(key=lambda x: x[0])

        # ----- track global best -----
        if child_score < best_score:
            best_score, best_vec = child_score, child.copy()
            print(f"[{time.time() - start:.2f}s] EA new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Local polishing of the current best vector
    # ------------------------------------------------------------------
    time_left = deadline - time.time()

        # Allocate 60 % of the remaining time to exponentiated sub‑gradient descent
        sub_deadline = time.time() + 0.6 * time_left
        best_vec, best_score = _exp_subgradient_descent(best_vec, sub_deadline, start)

    time_left = deadline - time.time()
    if time_left > 3.0:
        # Use the remaining time for smooth‑max Adam polishing
        best_vec, best_score = _smooth_gradient_descent(best_vec, deadline, start)

    # ------------------------------------------------------------------
    # 4) Final normalisation and conversion to plain list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
