import cvxpy as cp

def solve_lp(a, M):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # define conv a * b (full length)
    # we can implement convolution via Toeplitz matrix: C * b where C is matrix of shape (2n-1, n) where C[k,i] = a[i] if 0 <= k-i < n else 0.
    # Use linear constraints: sum_i a_i * b_{k-i} <= M for each k.
    constraints = []
    for k in range(2*n - 1):
        # compute indices i where b_{k-i} defined: i from max(0,k-(n-1)) to min(n-1,k)
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # pts: sum_{i=lo}^{hi} a[i] * b[k-i] <= M
        expr = cp.sum(cp.multiply(a[lo:hi+1], b[k - np.arange(lo, hi+1)]))
        constraints.append(expr <= M)
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-8, reltol=1e-8)
    return b.value
```

"""
Improved step‑function search for ``evaluate_sequence``.

The algorithm works in three stages:

1️⃣  Initialise a probability vector (uniform, a provided starter,
    or a quick Dirichlet sweep).

2️⃣  *Linear‑programming descent*:  for the current distribution a,
    solve the LP
        maximise   Σ b_i
        s.t.      a * b ≤ M  (pointwise self‑convolution bound)
                   b_i ≥ 0
    where M = max(a∗a).  The solution b is then scaled so that
    max(a∗b) = M and mixed with a (convex combination).  This step
    provably never increases the bound and usually reduces it.

3️⃣  A short mirror‑descent (exponentiated‑gradient/sub‑gradient)
    polishing phase to clean up any remaining slack.

All work respects a 1000 s wall‑clock budget.  The returned sequence is
normalised to sum 1; ``evaluate_sequence`` is scale‑invariant so this
normalisation is harmless.
"""

import time
import numpy as np

# cvxpy is used for the linear program in the descent phase.
# If it cannot be imported the routine simply falls back to the
# original heuristic (the imports succeed on the evaluation platform).
try:
    import cvxpy as cp
    _has_cvxpy = True
except Exception:  # pragma: no cover
    _has_cvxpy = False


def _conv_full(a: np.ndarray, b: np.ndarray | None = None) -> np.ndarray:
    """Full linear convolution (a∗a if b is None).  Uses FFT for speed."""
    if b is None:
        b = a
    n = a.size + b.size - 1
    # small vectors – direct convolution is fine and avoids FFT overhead
    if n <= 4096:
        return np.convolve(a, b, mode='full')
    # otherwise use FFT (unlikely with n=1024)
    L = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(A * B, n=L)[:n]
    return conv


def _objective_score(seq: np.ndarray) -> float:
    """Score = 2·n·max(conv(seq,seq))  (assumes Σseq = 1)."""
    n = seq.shape[0]
    conv = _conv_full(seq)
    return float(2.0 * n * conv.max())


def _solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_i   s.t.   a∗b ≤ M   (pointwise),   b ≥ 0.

    Returns the optimal b as a dense numpy array, or ``None`` on failure.
    """
    if not _has_cvxpy:          # pragma: no cover
        return None

    n = a.shape[0]
    # Build the Toeplitz matrix T such that (T @ b)[k] = (a∗b)[k].
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        T[i:i + n, i] = a[i]

    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=3000, abstol=1e-9, reltol=1e-9, verbose=False)
    except Exception:          # pragma: no cover
        return None

    if b.value is None:
        return None
    return np.array(b.value, dtype=np.float64)


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) w.r.t. a (averaged over active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _final_refine(a: np.ndarray, score: float,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    A lightweight exchange‑move hill‑climb using the exact max‑convolution
    sub‑gradient.  Stops when time runs out or after a few thousand moves.
    """
    rng = np.random.default_rng()
    n = a.shape[0]
    while time.time() < deadline:
        conv = _conv_full(a)
        grad = _subgradient_maxconv(a, conv)

        # pick source (large grad) and destination (small grad)
        src = np.argmax(grad)
        dst = np.argmin(grad)

        if a[src] < 1e-12:
            # pick a random source if the best one is already zero
            src = rng.integers(n)
            if a[src] < 1e-12:
                continue

        # move a small fraction of mass
        frac = rng.random() * 0.005 + 1e-4
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total == 0:
            continue
        a_new /= total

        new_score = _objective_score(a_new)
        if new_score < score:
            a, score = a_new, new_score
            # print progress lightly
            print(f"[{time.time() - start_time:.2f}s] refine score={score:.6f}")
    return a, score


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a list of
    floats (the sum is normalised to 1).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.2    # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate n makes the LP cheap while preserving
    # the asymptotic constant (the objective does not depend on n).
    # ------------------------------------------------------------------
    n = 1024

    # ------------------------------------------------------------------
    # Initial distribution (uniform or a provided starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    a = uniform.copy()
    best_score = _objective_score(a)

    # If a previous construction is available use it (trim / pad as needed)
    starter = globals().get('height_sequence_1')
    if isinstance(starter, (list, tuple, np.ndarray)):
        s = np.asarray(starter, dtype=float).ravel()
        if s.size > 0:
            if s.size > n:
                s = s[:n]
            elif s.size < n:
                s = np.pad(s, (0, n - s.size), 'constant')
            s = np.maximum(s, 0.0)
            total = s.sum()
            if total > 0:
                s /= total
                sc = _objective_score(s)
                if sc < best_score:
                    a, best_score = s, sc
                    print(f"[{time.time() - start_time:.2f}s] starter score={sc:.6f}")

    print(f"[{time.time() - start_time:.2f}s] baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Quick Dirichlet sweep to get a decent starting point
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 1)
    dirichlet_trials = 300
    for _ in range(dirichlet_trials):
        if time.time() >= deadline:
            break
        log_c = rng.uniform(np.log(0.01), np.log(20.0))
        c = np.exp(log_c)
        cand = rng.gamma(shape=c, scale=1.0, size=n)
        if cand.sum() == 0:
            continue
        cand /= cand.sum()
        sc = _objective_score(cand)
        if sc < best_score:
            a, best_score = cand, sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet cand score={sc:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – LP descent (main optimisation loop)
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline - 4.0:           # reserve a few seconds
        it += 1
        M = _conv_full(a).max()
        b = _solve_lp(a, M)
        if b is None:
            # fall back to a simple sub‑gradient step if the LP fails
            break

        conv_ab = _conv_full(a, b)
        max_ab = conv_ab.max()
        if max_ab <= 0:
            break
        lam = M / max_ab
        g = b * lam
        g_norm = g / g.sum()

        # try a few mixing rates and keep the best improvement
        improved = False
        for t in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = (1.0 - t) * a + t * g_norm
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective_score(cand)
            if sc < best_score - 1e-12:
                a, best_score = cand, sc
                improved = True
                print(f"[{time.time() - start_time:.2f}s] LP it={it} t={t:.3f} score={sc:.6f}")
                break
        if not improved:
            # If none of the mixes helped we assume we are at a fixed point.
            # A small random perturbation may help escape a shallow plateau.
            if rng.random() < 0.05:
                perturb = rng.normal(scale=1e-3, size=n)
                a = np.maximum(a + perturb, 0.0)
                a /= a.sum()
                best_score = _objective_score(a)
                print(f"[{time.time() - start_time:.2f}s] random perturb after stall")
            else:
                # No progress – quit the LP phase.
                break

    # ------------------------------------------------------------------
    # Stage 3 – final refinement with a cheap exchange‑move hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a, best_score = _final_refine(a, best_score, deadline)

    # ------------------------------------------------------------------
    # Normalise and return the result as a plain Python list
    # ------------------------------------------------------------------
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = uniform.copy()
    final_score = _objective_score(a)
    print(f"[{time.time() - start_time:.2f}s] finished best score ≈ {final_score:.6f}")

    # The external evaluator will apply its own clamping, but we keep the
    # sequence in the allowed [0,1000] range.
    return [float(v) for v in a]
