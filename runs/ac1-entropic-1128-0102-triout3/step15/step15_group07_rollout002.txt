"""
Smooth‑gradient descent on a soft‑max approximation of max(conv(a,a)).

The candidate is a probability vector (non‑negative, sum = 1).  The true
objective is 2·n·max(conv(a,a)).  Because the max is non‑differentiable,
we replace it by the log‑sum‑exp (soft‑max) with an annealing
parameter β.  The gradient of the smoothed objective can be written as a
convolution, which we compute efficiently with FFTs.  A projected
gradient (Adam) step keeps the iterates on the simplex.  The algorithm
starts from a diverse pool of analytic shapes (Beta, Dirichlet,
uniform, etc.), refines the best one with the smooth descent for most
of the time budget and finishes with a short simulated‑annealing polish.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not less than x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using FFT (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _conv_full_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full convolution of two 1‑D arrays using FFT."""
    L = _next_pow2(len(u) + len(v) - 1)
    Uf = np.fft.rfft(u, n=L)
    Vf = np.fft.rfft(v, n=L)
    conv = np.fft.irfft(Uf * Vf, n=L)
    return conv[: len(u) + len(v) - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Maintain a sorted pool of at most k best (lowest‑score) candidates."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _make_beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    shape = np.exp(log_shape)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _make_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.gamma(conc, 1.0, size=n)
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent with Adam
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    rng: np.random.Generator,
                    beta_start: float = 5.0,
                    beta_max: float = 200.0,
                    step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv) via a smooth approximation.
    Returns best vector and its exact score (not the smoothed one).
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    beta = beta_start

    while time.time() < deadline:
        it += 1

        # exact objective (for recording)
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time()-start_time:.2f}s] smooth iter={it:,} new best={cur_score:.6f}")

        # convolution a*a
        conv = _conv_self(a)

        # soft‑max weights (stable)
        maxc = conv.max()
        exp_vals = np.exp(beta * (conv - maxc))
        w = exp_vals / exp_vals.sum()   # length 2n‑1, sum 1

        # gradient of smoothed objective:
        #   ∂f/∂a = 4·n·(w * rev(a)) shifted by n‑1
        rev_a = a[::-1]
        g_full = _conv_full_fft(w, rev_a)          # length 3n‑2
        g = 4.0 * n * g_full[n - 1: n - 1 + n]     # slice of length n

        # Adam step (learning rate scaled by it)
        lr = step0 / math.sqrt(it)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        adapt = lr / (np.sqrt(v_hat) + eps)
        a = a - adapt * m_hat

        # keep on the simplex
        a = _project_simplex(a)

        # anneal β slowly
        if it % 200 == 0:
            beta = min(beta_max, beta * 1.05)

        # occasional random jitter to escape plateaus
        if it % 4000 == 0:
            src = rng.choice(n, p=a)
            dst = rng.integers(n)
            if src != dst:
                delta = a[src] * 0.02 * rng.random()
                a[src] -= delta
                a[dst] += delta
                a = _project_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing polish (same as the previous code)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         rng: np.random.Generator,
                         max_move_frac: float = 0.04) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    T_start = 0.2
    T_end = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        T = T_start * ((T_end / T_start) ** frac)

        src = rng.choice(n, p=a)
        if a[src] < 1e-16:
            continue
        dst = rng.integers(n)
        if src == dst:
            continue
        delta = a[src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue

        a_prop = a.copy()
        a_prop[src] -= delta
        a_prop[dst] += delta

        new_score = _objective(a_prop)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time()-start_time:.2f}s] SA iter={it:,} new best={new_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.
    Returns a plain list of non‑negative floats that sum to 1.
    """
    rng = np.random.default_rng(seed + 12345)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose size N (larger N gives potentially lower scores)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 800:
        N = 65536
    elif remaining > 400:
        N = 32768
    else:
        N = 16384

    # ------------------------------------------------------------------
    # Build a small pool of diverse initial candidates
    # ------------------------------------------------------------------
    TOP_K = 6
    pool = []   # list of (score, vector)

    # uniform baseline
    uniform = np.full(N, 1.0 / N)
    _push_top(pool, _objective(uniform), uniform, TOP_K)

    # fine‑grained symmetric Beta shapes (U‑shaped)
    betas = np.concatenate((
        np.logspace(np.log10(0.001), np.log10(0.2), 12),
        np.arange(0.22, 0.98, 0.06)
    ))
    for alpha in betas:
        a = _make_beta_shape(N, float(alpha))
        _push_top(pool, _objective(a), a, TOP_K)

    # random Dirichlet draws with a broad range of concentrations
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.25, 0.5]
    for conc in concs:
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _make_dirichlet(N, conc, rng)
            _push_top(pool, _objective(a), a, TOP_K)

    # include the previous best if it exists
    try:
        starter = globals().get('height_sequence_1')
        if starter is not None:
            starter = np.asarray(starter, dtype=float)
            if starter.ndim == 1 and starter.size > 0:
                if starter.size != N:
                    x_old = np.linspace(0.0, 1.0, starter.size, endpoint=False)
                    x_new = np.linspace(0.0, 1.0, N, endpoint=False)
                    starter_up = np.interp(x_new, x_old, starter)
                else:
                    starter_up = starter.copy()
                starter_up = np.maximum(starter_up, 0.0)
                s = starter_up.sum()
                if s > 0:
                    starter_up /= s
                    _push_top(pool, _objective(starter_up), starter_up, TOP_K)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Pick the current best candidate
    # ------------------------------------------------------------------
    pool.sort(key=lambda x: x[0])
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time()-start:.2f}s] Initial best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main refinement: smooth gradient descent on the best candidate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sub_deadline = time.time() + remaining * 0.85
        refined_vec, refined_score = _smooth_descent(best_vec,
                                                     sub_deadline,
                                                     start,
                                                     rng,
                                                     beta_start=5.0,
                                                     beta_max=200.0,
                                                     step0=0.5)
        if refined_score < best_score:
            best_score, best_vec = refined_score, refined_vec.copy()
            print(f"[{time.time()-start:.2f}s] After smooth descent best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Short simulated‑annealing polish if any time left
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        best_vec, best_score = _simulated_annealing(best_vec,
                                                    deadline,
                                                    start,
                                                    rng,
                                                    max_move_frac=0.04)

    # ------------------------------------------------------------------
    # Final normalisation and conversion to plain Python list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time()-start:.2f}s] Finished – final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
