"""
Propose a step‑function sequence that (hopefully) beats the previous
best bound ≈ 1.508.  The algorithm proceeds as follows:

1. Choose a problem size N that fits the remaining budget (larger N gives
   a richer search space, but each convolution is more expensive).
2. Build a diverse initial pool of candidates:
   – analytic shapes (uniform, power‑ramps, exponentials, power‑laws,
     asymmetric beta)
   – many random Dirichlet draws with various concentration parameters
   – sparse vectors with a random subset of positions
   – (optionally) the previous best construction `height_sequence_1`
   The pool keeps the TOP_K best according to the exact objective.
3. Run a fast exponentiated sub‑gradient descent (multiplicative updates)
   on each pool element in a round‑robin fashion.  The active set of
   convolution indices attaining the current maximum supplies the sub‑gradient.
   A larger base step‑size (η₀=1.0) and a generous time budget (≈ 80 % of the
   total) give a strong descent.
4. Refine the current best with a smooth‑max Adam optimiser.  The hard max
   is replaced by a temperature‑annealed soft‑max and the iterate is
   projected onto the probability simplex after each step.
5. Return the best vector (non‑negative, sums to 1) as a plain list.

All heavy work is carried out with FFT‑based convolutions, and the routine
stops well before the 1000 s limit, guaranteeing a valid answer.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via real FFT, clipped to ≥0."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return np.maximum(conv, 0.0)


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    max_conv = float(conv.max())
    s = float(seq.sum())
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic (and random) shape generators
# ----------------------------------------------------------------------
def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _make_power_ramp(n: int, p: float, plateau_frac: float) -> np.ndarray:
    plateau_len = int(round(plateau_frac * n))
    if plateau_len % 2 == 1 and plateau_len > 0:
        plateau_len -= 1
    side_len = (n - plateau_len) // 2
    if side_len <= 0:
        shape = np.ones(n)
    else:
        idx = np.arange(1, side_len + 1, dtype=float) / side_len
        ramp = idx ** p
        shape = np.concatenate([ramp, np.ones(plateau_len), ramp[::-1]])
        if shape.size < n:
            shape = np.concatenate([shape, np.zeros(n - shape.size)])
        else:
            shape = shape[:n]
    shape /= max(shape.sum(), 1e-12)
    return shape


def _make_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=float)
    shape = np.exp(-lam * idx)
    shape /= max(shape.sum(), 1e-12)
    return shape


def _make_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=float)
    shape = idx ** (-p)
    shape /= max(shape.sum(), 1e-12)
    return shape


def _make_beta_shape_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_shape = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape /= max(shape.sum(), 1e-12)
    return shape


def _make_sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k distinct positions with equal weight 1/k (k≥2)."""
    k = max(2, min(k, n))
    idx = rng.choice(n, size=k, replace=False)
    vec = np.zeros(n, dtype=float)
    vec[idx] = 1.0 / k
    return vec


def _make_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet(α,…,α) vector and sort descending."""
    v = rng.gamma(conc, 1.0, size=n)
    v = np.sort(v)[::-1]
    v_sum = v.sum()
    if v_sum > 0:
        v /= v_sum
    else:
        v = np.full(n, 1.0 / n)
    return v

# Pool management utilities
# ----------------------------------------------------------------------
def _push_top(pool: list, score: float, vec: np.ndarray, k: int) -> None:
    """Insert (score,vec) into a sorted top‑k pool (ascending score)."""
    if len(pool) < k:
        pool.append((score, vec.copy()))
        pool.sort(key=lambda x: x[0])
    elif score < pool[-1][0]:
        pool[-1] = (score, vec.copy())
        pool.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent (fast multiplicative updates)
# ----------------------------------------------------------------------
def _exp_subgradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float,
                             eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent on max‑convolution.
    a0 must be a probability vector (non‑negative, sum = 1).
    """
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = float(conv.max())

        # active set (indices attaining the max, tolerance = 1e‑12 relative)
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            active = np.array([conv.argmax()], dtype=int)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient = 2·(w ★ a_rev)  (★ = correlation)
        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # decreasing step‑size
        eta = eta0 / (1.0 + it / 5000.0)

        factor = np.exp(np.clip(-eta * grad, -50.0, 50.0))
        a *= factor
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] ExpGD iter={it:,} best={cur_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max (soft‑max) Adam descent – fine polishing
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray,
                             deadline: float,
                             start_time: float) -> tuple[np.ndarray, float]:
    """Adam on a soft‑max surrogate of the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]

    best = a.copy()
    best_score = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.1
    T0, T_end = 0.1, 1e-8

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothGD iter={it:,} best={cur_score:.6f}")

        # temperature schedule – exponential decay
        T = T_end + (T0 - T_end) * (0.5 ** (it / 4000.0))

        w_raw = np.exp((conv - max_conv) / T)
        w = w_raw / w_raw.sum()

        a_rev = a[::-1]
        L = _next_pow2(3 * n - 2)
        grad = np.fft.irfft(np.fft.rfft(w, n=L) * np.fft.rfft(a_rev, n=L), n=L)
        grad = 2.0 * grad[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = step0 / math.sqrt(it)
        a -= step * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a)

    return best, best_score

# ----------------------------------------------------------------------

                     budget_s: int = 1000,
                     **kwargs) -> list[float]:

    The algorithm mixes analytic families, random Dirichlet/sparse draws,
    exponentiated sub‑gradient descent and a final Adam polishing.
    """

    rng = np.random.default_rng(seed + 12345)

    # Choose problem size (larger N gives a richer search space)
    # ------------------------------------------------------------------

    if remaining > 800:

    else:
        N = 8192

    # ------------------------------------------------------------------

    for p in np.linspace(0.2, 3.0, 7):
        for plat in np.linspace(0.0, 0.4, 2):
            _add(_make_power_ramp(N, p, plat))
    for lam in np.linspace(0.5, 4.0, 8):
        _add(_make_exponential(N, lam))
    for p in np.linspace(0.5, 3.0, 8):
        _add(_make_power_law(N, p))
    for a in np.linspace(0.2, 3.0, 4):
        for b in np.linspace(0.2, 3.0, 4):
            _add(_make_beta_shape_asym(N, a, b))

    # Random Dirichlet draws (different concentrations)
    for conc in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(12):
            if time.time() >= deadline:
                break
            _add(_make_dirichlet(N, conc, rng))

    # Random sparse vectors
    max_sparse_k = max(2, N // 32)   # up to about 3 % non‑zeros
    for _ in range(30):
        if time.time() >= deadline:
            break
        k = rng.integers(2, max_sparse_k + 1)
        _add(_make_sparse_random(N, k, rng))

    # Include the previous best construction if available
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            if starter.size != N:
                x_old = np.linspace(0, 1, starter.size, endpoint=False)
                x_new = np.linspace(0, 1, N, endpoint=False)
                starter_up = np.interp(x_new, x_old, starter)
            else:
                starter_up = starter.copy()
            starter_up = np.maximum(starter_up, 0.0)
            s = starter_up.sum()
            if s > 0:
                starter_up /= s
                _add(starter_up)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the current best
    # ------------------------------------------------------------------
    best_score, best_vec = pool[0][0], pool[0][1].copy()
    print(f"[{time.time() - start:.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Exponentiated sub‑gradient descent (major part of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        subdeadline = time.time() + 0.78 * remaining
        slice_time = 4.0                     # seconds per candidate slice
        while time.time() < subdeadline:
            for i in range(len(pool)):
                now = time.time()
                if now >= subdeadline:
                    break
                slice_dead = min(now + slice_time, subdeadline)
                seq, sc = _exp_subgradient_descent(pool[i][1], slice_dead, start, eta0=1.0)
                if sc < best_score:
                    best_score, best_vec = sc, seq.copy()
                    print(f"[{time.time() - start:.2f}s] ExpGD improvement → {sc:.6f}")
                # update the pool entry if we improved it
                if sc < pool[i][0]:
                    pool[i] = (sc, seq.copy())

    # ------------------------------------------------------------------
    # 3) Smooth‑max Adam polishing (use whatever time remains)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq, sc = _smooth_gradient_descent(best_vec, deadline, start)
        if sc < best_score:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time() - start:.2f}s] SmoothGD final improvement → {sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Normalise and return as plain Python list
    # ------------------------------------------------------------------
    best_vec = np.maximum(best_vec, 0.0)
    total = best_vec.sum()
    if total > 0:
        best_vec /= total
    else:
        best_vec = np.full(N, 1.0 / N)

    final_score = _objective(best_vec)
    print(f"[{time.time() - start:.2f}s] Finished, final score ≈ {final_score:.6f}")

    return [float(v) for v in best_vec]
