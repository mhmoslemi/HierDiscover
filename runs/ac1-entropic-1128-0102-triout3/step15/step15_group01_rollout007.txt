"""
Multi‑phase optimizer for the autocorrelation bound.

The objective (after normalising sum = 1) is   score = 2·n·max(conv(a,a)).
The optimiser proceeds through the following stages, all respecting a
global time budget (≈1000 s):

0️⃣  Initialise from the global starter (if any) or a uniform vector.
1️⃣  Broad random Dirichlet draws to get a decent starting point.
2️⃣  Smooth‑max (log‑sum‑exp) mirror‑descent:
    – a soft approximation of the max‑convolution is used,
    – the smoothing parameter β is increased gradually,
    – updates are multiplicative (exponentiated‑gradient) and stay on the
      probability simplex.
3️⃣  Classic sub‑gradient descent on the exact max‑convolution (as in the
    reference implementation) to polish the solution.
4️⃣  Local random “mass‑transfer’’ moves with an exact recomputation of the
    convolution – a cheap hill‑climber that can escape tiny plateaus.
5️⃣  Final mixing with the uniform distribution and a short λ‑grid search.

All updates keep the vector non‑negative and renormalise to sum = 1.
Progress (current best score and elapsed time) is printed throughout.
The returned list is a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, L_fft: int | None = None) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    if L_fft is None:
        L_fft = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(Af * Af, n=L_fft)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray, n: int) -> float:
    """Score used by the optimiser: 2·n·max(conv(a,a)).  a is normalised."""
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)).
    When several shifts attain the maximum, the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    active = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 0️⃣  Random Dirichlet draws (broad initial exploration)
# ----------------------------------------------------------------------
def _random_dirichlet_search(n: int,
                             best_seq: np.ndarray,
                             best_score: float,
                             deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    while time.time() < deadline:
        log_c = rng.uniform(np.log(0.02), np.log(20.0))
        c = np.exp(log_c)
        sample = rng.gamma(shape=c, scale=1.0, size=n)
        s = sample.sum()
        if s <= 0:
            continue
        a = sample / s
        sc = _objective_score(a, n)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} → score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 1️⃣  Smooth‑max exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _smooth_expgrad_descent(a0: np.ndarray,
                            n: int,
                            L_fft_self: int,
                            L_fft_grad: int,
                            start_time: float,
                            deadline: float,
                            best_seq: np.ndarray,
                            best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a smooth (log‑sum‑exp) surrogate
    for the max‑convolution.  The smoothing parameter β grows from a small
    value (very smooth) to a large value (sharp).  Updates are
    multiplicative (exponentiated‑gradient) with a 1/√t step‑size.
    """
    a = a0.copy()
    it = 0
    beta_start = 5.0
    beta_end = 3000.0
    max_iters_beta = 5000  # after this β is saturated

    while time.time() < deadline:
        it += 1
        # current convolution and score
        conv = _conv_self(a, L_fft_self)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Smooth‑GD it={it:,} β≈{beta_start + (beta_end-beta_start)*min(1,it/max_iters_beta):.1f} score={cur_score:.6f}")

        # increase β linearly with iteration (capped)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / max_iters_beta)

        # softmax weights over the convolution entries
        w_unnorm = np.exp(beta * (conv - cur_max))          # ≤ 1, max = 1 at active shifts
        w_sum = w_unnorm.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_unnorm / w_sum

        # gradient of the smooth objective:
        #   ∇ = 2 * (a_rev * w) evaluated at indices n‑1 … n‑1+n‑1
        rev_a = a[::-1]
        rev_fft = np.fft.rfft(rev_a, n=L_fft_grad)
        w_fft = np.fft.rfft(w, n=L_fft_grad)
        cross = np.fft.irfft(rev_fft * w_fft, n=L_fft_grad)
        grad = 2.0 * cross[n - 1:n - 1 + n]

        # multiplicative (exponentiated‑gradient) update
        eta = 0.7 / np.sqrt(it)
        exp_factor = np.exp(-eta * grad)
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a_new = a * exp_factor
        # renormalise onto the simplex
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s

        # evaluate new candidate
        conv_new = _conv_self(a_new, L_fft_self)
        new_score = 2.0 * n * conv_new.max()

        # accept if improvement; otherwise allow a small annealed uphill move
        if new_score < cur_score:
            a = a_new
        else:
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 1e-3 * (remaining / total)          # tiny temperature, decays to 0
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_new
        # loop continues
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Classic sub‑gradient descent (as in the reference code)
# ----------------------------------------------------------------------
def _expgrad_descent(a0: np.ndarray,
                     n: int,
                     L_fft_self: int,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the true sub‑gradient of max(conv).
    This phase refines the solution after the smooth‑max stage.
    """
    a = a0.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_self(a, L_fft_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it={it:,} score={cur_score:.6f}")

        grad = _subgradient_maxconv(a, conv)

        eta = 0.7 / np.sqrt(it)
        exp_factor = np.exp(-eta * grad)
        exp_factor = np.clip(exp_factor, 1e-12, 1e12)
        a_new = a * exp_factor
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s

        conv_new = _conv_self(a_new, L_fft_self)
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score:
            a = a_new
        else:
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 1e-3 * (remaining / total)
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_new
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Local random “mass‑transfer’’ search
# ----------------------------------------------------------------------
def _local_random_search(a0: np.ndarray,
                         n: int,
                         L_fft_self: int,
                         start_time: float,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Small stochastic hill‑climbing: move a tiny fraction of probability mass
    from a source index to a target index and keep the move if it improves
    the max‑convolution (or with a tiny uphill probability).
    """
    a = a0.copy()
    conv = _conv_self(a, L_fft_self)
    cur_score = 2.0 * n * conv.max()
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        # pick a source index proportionally to its mass (avoids useless moves)
        if a.sum() == 0.0:
            break
        src = rng.choice(n, p=a)                      # source of mass
        if a[src] <= 1e-12:
            continue
        tgt = rng.integers(0, n)
        if tgt == src:
            continue
        # transfer a small *relative* amount of mass
        eps = rng.uniform(0.001, 0.01)                 # 0.1 % – 1 % of src mass
        delta = min(eps * a[src], a[src])
        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        conv_candidate = _conv_self(a_candidate, L_fft_self)
        score_candidate = 2.0 * n * conv_candidate.max()
        if score_candidate < cur_score:
            a, conv, cur_score = a_candidate, conv_candidate, score_candidate
            if score_candidate < best_score:
                best_score, best_seq = score_candidate, a_candidate.copy()
                print(f"[{time.time() - start_time:.2f}s] Local move it={it:,} score={score_candidate:.6f}")
        else:
            # occasional uphill acceptance (tiny temperature)
            remaining = max(0.0, deadline - time.time())
            total = max(0.1, deadline - start_time)
            T = 1e-4 * (remaining / total)
            if np.random.rand() < np.exp(-(score_candidate - cur_score) / max(T, 1e-12)):
                a, conv, cur_score = a_candidate, conv_candidate, score_candidate
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final fine‑grained mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """
    Scan λ∈[0,1] for seq = λ·a + (1‑λ)·uniform.
    The best mixture is kept.
    """
    uniform = np.full(n, 1.0 / n)
    # coarse sweep
    lam_grid = np.linspace(0.0, 1.0, 41)      # 0.025 step
    for lam in lam_grid:
        if time.time() >= deadline:
            break
        seq = lam * a + (1.0 - lam) * uniform
        seq = seq / seq.sum()
        sc = _objective_score(seq, n)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Mix λ={lam:.3f} → score={sc:.6f}")

    # optional refinement around the best λ (simple local search)
    # (the remaining budget is usually tiny, so we skip a deeper search)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget and prints
    intermediate scores.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.25   # tiny safety margin

    # problem size – 4096 performed well historically
    n = 4096
    L_fft_self = _next_pow2(2 * n - 1)
    L_fft_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0️⃣  Initialise (uniform or global starter)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = _objective_score(best_seq, n)
    print(f"[0.00s] Uniform baseline score={best_score:.6f}")

    # Optional global starter (`height_sequence_1` may be present)
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            # adjust size if necessary
            if starter.size != n:
                if starter.size < n:
                    tmp = np.zeros(n)
                    tmp[:starter.size] = starter
                    starter = tmp
                else:
                    starter = starter[:n]
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0.0:
                starter = starter / s
                sc = _objective_score(starter, n)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[0.00s] Global starter improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad random Dirichlet search (≈5 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.05 * budget_s
    if time.time() < phase1_deadline:
        best_seq, best_score = _random_dirichlet_search(
            n, best_seq, best_score, phase1_deadline
        )

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max exponentiated‑gradient descent
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s
    if time.time() < phase2_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting smooth‑max GD")
        best_seq, best_score = _smooth_expgrad_descent(
            best_seq, n, L_fft_self, L_fft_grad,
            start_time, phase2_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After smooth‑GD best≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Classic sub‑gradient descent for polishing
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.90 * budget_s
    if time.time() < phase3_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting sub‑gradient descent")
        best_seq, best_score = _expgrad_descent(
            best_seq, n, L_fft_self,
            start_time, phase3_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After sub‑grad best≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Local random mass‑transfer hill climbing
    # ------------------------------------------------------------------
    phase4_deadline = start_time + 0.95 * budget_s
    if time.time() < phase4_deadline:
        print(f"[{time.time() - start_time:.2f}s] Local random search")
        best_seq, best_score = _local_random_search(
            best_seq, n, L_fft_self,
            start_time, phase4_deadline, best_seq, best_score
        )
        print(f"[{time.time() - start_time:.2f}s] After local search best≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Mixing with the uniform distribution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Mixing with uniform")
        best_seq, best_score = _mix_with_uniform(
            best_seq, n, deadline, start_time, best_seq, best_score
        )

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(v) for v in best_seq]
