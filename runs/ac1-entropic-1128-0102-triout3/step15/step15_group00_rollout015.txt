    a = a0.copy()
    n = len(a)
    it = 0
    beta = 5.0
    beta_max = 300.0
    eta = 500.0
    eta_min = 1.0
    # schedule: increase beta linearly, decrease eta linearly.
    total_iters_possible = 20000  # just for schedule length.

    while time.time() < deadline:
        it += 1
        # schedule updates
        t_ratio = min(it / 20000.0, 1.0)  # assume many iterations within time
        beta = 5.0 + t_ratio * (beta_max - 5.0)
        eta = 500.0 * (1 - t_ratio) + eta_min * t_ratio  # linearly decreasing

        # compute gradient
        conv = _conv_self(a)   # may use FFT for efficiency
        maxc = conv.max()
        diff = conv - maxc
        # Use stable scaling: shift to avoid overflow; diff <= 0.
        exp_term = np.exp(beta * diff)
        sum_exp = exp_term.sum()
        if sum_exp == 0:
            # fallback: maybe all diff zero
            p = np.full_like(exp_term, 1.0 / exp_term.size)
        else:
            p = exp_term / sum_exp

        # compute term1 term2 via FFT
        a_rev = a[::-1]
        term1_full = _conv_full_fft(p, a_rev)

        factor = np.exp(-eta * grad)

        if it % 1000 == 0:

            pass

def _smooth_gradient(a, beta):
    n = len(a)
    # Compute self-convolution via FFT (full).
    # Use length L for FFT: next power of two >= 3*n
    L = _next_pow2(3*n)  # ensures large enough for conv with p length 2n-1

    # FFT of a
    Fa = np.fft.rfft(a, n=L)
    # Self convolution
    conv_full = np.fft.irfft(Fa * Fa, n=L)[:2*n-1]  # shape (2n-1)
    maxc = conv_full.max()
    diff = conv_full - maxc
    # probabilities
    exp_term = np.exp(beta * diff)  # diff <=0
    sum_exp = exp_term.sum()
    if sum_exp == 0:
        p = np.full_like(exp_term, 1.0 / exp_term.size)
    else:
        p = exp_term / sum_exp

    # Prepare for gradient conv
    # FFT of p
    P = np.fft.rfft(p, n=L)

    # FFT of a_rev
    a_rev = a[::-1]
    Fa_rev = np.fft.rfft(a_rev, n=L)

    # term2_full = conv(p, a) = irfft(P*Fa)
    term2_full = np.fft.irfft(P * Fa, n=L)[: (2*n-1) + n -1]  # length 3n-2
    term2 = term2_full[:n]

    # term1_full = conv(p, a_rev)
    term1_full = np.fft.irfft(P * Fa_rev, n=L)[: (2*n-1) + n -1]
    term1 = term1_full[n-1:n-1+n]

    grad = term1 + term2
    return grad, maxc, conv_full

"""
Optimise step‑function coefficients for the evaluation function

    2 * n * max(conv(a,a)) / (sum(a)^2)

using three phases:

1.  fast parametric sweeps (geometric, power, beta‑shapes, exponential,
    two‑block) to obtain a decent starting point.
2.  a *smooth* descent – a continuation of the classic multiplicative‑weight
    scheme where the max‑convolution is replaced by a temperature‑controlled
    log‑sum‑exp of all convolution entries.  The gradient can be computed with
    two FFT‑based convolutions, giving an O(n log n) iteration.
3.  a short projected hill‑climb to polish the result.

All vectors are kept on the probability simplex (non‑negative, sum = 1).
The routine stops after the supplied time budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _conv_full_fft(x: np.ndarray, y: np.ndarray, L: int = None) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    if L is None:
        L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (sum is forced to 1)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 1️⃣  Parametric families (identical to the reference implementation)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)

    eps_grid = np.logspace(-5, -1, 300)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        s = geo.sum()
        if s == 0:
            continue
        geo /= s
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_symmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.05, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_seq = (a - 1.0) * np.log(idx + 1.0) + (a - 1.0) * np.log(np.maximum(n - idx, 1))
        seq = np.exp(log_seq - np.max(log_seq))
        s = seq.sum()
        if s == 0 or not np.isfinite(s):
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] SymBeta α={a:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_asymmetric_beta(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    alphas = np.linspace(0.1, 5.0, 21)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a_part = (a - 1.0) * np.log(idx + 1.0)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a_part + (b - 1.0) * np.log(np.maximum(n - idx, 1))
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] AsymBeta α={a:.2f} β={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_exponential(n, best_seq, best_score, deadline):
    """Symmetric exponential decay: a_i ∝ exp(-λ·min(i, n‑1‑i))."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 100):
        if time.time() >= deadline:
            break
        seq = np.exp(-lam * dist)
        s = seq.sum()
        if s == 0:
            continue
        seq /= s
        sc = _objective_score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time() - start_time:.2f}s] Exp λ={lam:.1e} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_blocks(n, best_seq, best_score, deadline):
    """Two uniform blocks at the ends with a tunable mass split."""
    for L_frac in np.linspace(0.01, 0.4, 40):
        L = int(L_frac * n)
        if L < 1:
            continue
        for w in np.linspace(0.1, 0.9, 9):
            if time.time() >= deadline:
                break
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet drawn initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 400
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)      # concentration ≈ 0.01 … 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smoothed descent (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float):
    """
    Returns
        grad   – gradient of  (1/beta)·log∑exp(beta·conv)  (scaled by 2)
        maxc   – the current max of the self‑convolution
        conv   – full self‑convolution (length 2·n‑1)
    """
    n = a.shape[0]
    L = _next_pow2(3 * n)                     # enough for all convolutions

    # FFT of a (used for self‑convolution and for the two weighted convolutions)
    Fa = np.fft.rfft(a, n=L)

    # ----- self‑convolution -------------------------------------------------
    conv_full = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    maxc = conv_full.max()

    # ----- softmax weights --------------------------------------------------
    diff = conv_full - maxc                    # ≤ 0
    exp_term = np.exp(beta * diff)            # stable because max diff = 0
    sum_exp = exp_term.sum()
    p = exp_term / sum_exp if sum_exp != 0 else np.full_like(exp_term, 1.0 / exp_term.size)

    # ----- gradient (2 * Σ_j p_{i+j} a_j) -----------------------------------
    # term = conv(p, a_rev)  ; a_rev is a reversed copy
    a_rev = a[::-1]
    Fa_rev = np.fft.rfft(a_rev, n=L)
    P = np.fft.rfft(p, n=L)

    term_full = np.fft.irfft(P * Fa_rev, n=L)[: (2 * n - 1) + n - 1]   # length 3n‑2
    term = term_full[n - 1:n - 1 + n]                                 # slice length n

    grad = 2.0 * term
    return grad, maxc, conv_full


def _smooth_descent(a0: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float):
    """Multiplicative‑weight descent on the smoothed objective."""
    a = a0.copy()
    n = a.shape[0]
    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    # schedule parameters
    eta_start = 250.0
    eta_end = 0.5
    beta_start = 1.0
    beta_end = 200.0
    max_it_est = 20000  # used only for the schedule, actual loop stops by time

    while time.time() < deadline:
        it += 1
        # linear continuation on [0,1]
        t = min(it / max_it_est, 1.0)
        beta = beta_start + t * (beta_end - beta_start)
        eta = eta_start * (1.0 - t) + eta_end * t

        grad, maxc, _ = _smooth_gradient(a, beta)

        # centred gradient → multiplicative update
        g_center = grad - grad.mean()
        factor = np.exp(-eta * g_center)
        factor = np.clip(factor, 1e-12, 1e12)
        a = a * factor
        a = _project_simplex(a)

        cur_score = float(2.0 * n * maxc)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(f"[{time.time() - start_time:.2f}s] SmoothDescent it={it:,} "
                  f"β={beta:.1f} η={eta:.3f} score={cur_score:.6f}")

        # occasional random shake‑up if we have been stuck for a while
        if it - last_improve > 5000:
            i, j = rng.choice(n, size=2, replace=False)
            frac = rng.random() * 0.02 + 1e-5
            eps = a[i] * frac
            a[i] -= eps
            a[j] += eps
            a = _project_simplex(a)
            last_improve = it

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Directed hill‑climb (same as reference)
# ----------------------------------------------------------------------
def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts)."""
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


def _hill_climb_enhanced(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = float(2.0 * n * _conv_self(a).max())
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        max_frac = 0.05
        frac = rng.random() * max_frac + 1e-5
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        if a_prop[src] < -1e-14:
            continue

        cur_score = float(2.0 * n * _conv_self(a_prop).max())
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill++ score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                i1, i2 = rng.choice(n, size=2, replace=False)
                eps2 = a[i1] * (rng.random() * max_frac + 1e-5)
                a[i1] -= eps2
                a[i2] += eps2
                a = _project_simplex(np.maximum(a, 0.0))
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search the space of non‑negative step functions to minimise
    ``evaluate_sequence``.  Returns a list of floats (normalised to sum = 1).
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # Problem size – a relatively large vector gives a good trade‑off
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (fast families)
    # ------------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_geometric(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_power(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_symmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_asymmetric_beta(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_exponential(n, best_seq, best_score, phase_deadline)
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_two_blocks(n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # ------------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed exponential‑weight descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print(f"[{time.time() - start_time:.2f}s] Starting smoothed descent")
        best_seq, best_score = _smooth_descent(best_seq, deadline - 150.0,
                                               best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After smoothed descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb_enhanced(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise and return as a plain Python list
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
