    a = a0.copy()

    iteration = 0

def _hill_climb_optimal(a0, deadline, best_seq, best_score):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = best_score
    rng = np.random.default_rng()
    # precompute index to quickly compute d vector contributions:
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        grad = _subgradient_maxconv(a, conv)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)  # may be same as src but we handle later
        if src == dst:
            # random pair
            src, dst = rng.choice(n, size=2, replace=False)

"""
A fast sub‑gradient optimizer for the max‑self‑convolution problem.

The algorithm:
1.  Initialise a probability vector (the step‑function) – uniform, a
    global starter, or a few random Dirichlet draws.
2.  Repeatedly minimise the convex objective  max(conv(a,a)) by a
    projected sub‑gradient method.  The sub‑gradient is given by the
    active set of lag indices where the convolution attains its maximum.
3.  Refine the solution with a deterministic hill‑climb that moves an
    optimally chosen amount of mass from the “most harmful’’ index to the
    “most helpful’’ one (exact line‑search by ternary search).
All stages respect a wall‑clock budget of 1000 s and the best sequence
found is returned as a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (using FFT, length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assume sum(a) == 1)
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    """Return the evaluation score = 2·n·max(conv(a,a))."""
    n = a.shape[0]
    return float(2.0 * n * _conv_self(a).max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several lags attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Simple random initialisers (uniform / Dirichlet)
# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, trials: int = 300) -> np.ndarray:
    """Return the best of a few Dirichlet samples (normalised to sum = 1)."""
    rng = np.random.default_rng()
    best = np.full(n, 1.0 / n)
    best_sc = _objective_score(best)
    for _ in range(trials):
        logc = rng.uniform(-2.0, 0.7)       # concentration ∈ [0.01, 5]
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        sc = _objective_score(a)
        if sc < best_sc:
            best, best_sc = a, sc
    return best


# ----------------------------------------------------------------------
# 2️⃣  Projected sub‑gradient descent on max(conv)
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray, deadline: float,
                     best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent minimising max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    step = 0.5               # initial step‑size scale
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()
        # sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)
        gnorm = np.linalg.norm(grad) + 1e-12
        step_sz = step / gnorm

        a_new = a - step_sz * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # enforce symmetry – the optimal solution is symmetric
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = _project_simplex(a_new)

        conv_new = _conv_self(a_new)
        sc_new = 2.0 * n * conv_new.max()

        if sc_new < best_score:
            best_score, best_seq = sc_new, a_new.copy()
            print(f"[{time.time() - start_time:.2f}s] Sub‑grad it {it} score={sc_new:.6f}")

        # accept if we really improved the max‑value, otherwise shrink step
        if conv_new.max() < max_val:
            a = a_new
            step = min(step * 1.02, 1.0)   # slowly enlarge step
        else:
            step *= 0.5
            if step < 1e-8:
                step = 1e-8
        it += 1
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Deterministic hill‑climb with optimal line‑search
# ----------------------------------------------------------------------
def _optimal_eps(conv: np.ndarray, d: np.ndarray, max_move: float) -> float:
    """
    Ternary search for epsilon ∈ [0, max_move] minimising
    max(conv + 2·epsilon·d).  The objective is convex in epsilon.
    """
    lo, hi = 0.0, max_move
    for _ in range(30):                     # enough for double precision
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = (conv + 2.0 * m1 * d).max()
        v2 = (conv + 2.0 * m2 * d).max()
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5


def _optimal_hill_climb(a0: np.ndarray, deadline: float,
                        best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Move mass from the worst to the best index with optimal epsilon."""
    a = a0.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        # if src==dst pick a random pair (still a feasible move)
        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
            if a[src] < 1e-15:
                continue

        # build the direction vector d such that conv' = conv + 2*eps*d
        d = np.zeros(2 * n - 1, dtype=np.float64)
        d[dst:dst + n] += a                # +a for destination shift
        d[src:src + n] -= a                # -a for source shift

        max_move = a[src]                  # cannot move more than we own
        eps_opt = _optimal_eps(conv, d, max_move)

        if eps_opt <= 0.0:
            # no beneficial move – random jitter to escape flat plateaus
            i1, i2 = rng.choice(n, size=2, replace=False)
            jitter = a[i1] * (rng.random() * 0.004 + 1e-6)
            a[i1] -= jitter
            a[i2] += jitter
            a = _project_simplex(np.maximum(a, 0.0))
            continue

        a_new = a.copy()
        a_new[src] -= eps_opt
        a_new[dst] += eps_opt
        a_new = _project_simplex(np.maximum(a_new, 0.0))

        sc_new = 2.0 * n * _conv_self(a_new).max()
        if sc_new < best_score - 1e-12:
            a = a_new
            best_seq, best_score = a_new.copy(), sc_new
            print(f"[{time.time() - start_time:.2f}s] Hill‑climb it {it} score={sc_new:.6f}")

        it += 1
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------

    # ------------------------------------------------------------------

    try:

    else:

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
