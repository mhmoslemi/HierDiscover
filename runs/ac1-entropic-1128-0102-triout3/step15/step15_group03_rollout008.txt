def _hill_climb_incremental(a, conv, deadline, best_seq, best_score):
    n = len(a)
    L = len(conv)
    rng = np.random.default_rng()
    no_improve = 0
    accepted = 0
    while time.time() < deadline:
        # compute subgradient to choose src and dst
        grad = _subgradient_maxconv(a, conv)
        # Choose src and dst using gradient
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size ==0: break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)
        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)
        # choose delta
        delta_frac = rng.uniform(0.01, 0.3)  # 1%-30% of source
        delta = a[src] * delta_frac
        if delta <= 0:
            continue
        # simulate move
        # use a copy for shift
        a_temp = a  # we will compute using a copy of conv
        # compute new conv
        new_conv = conv.copy()
        new_conv[src:src+n] -= 2*delta*a
        new_conv[dst:dst+n] += 2*delta*a
        # d*d contributions
        new_conv[2*src] += delta*delta
        new_conv[2*dst] += delta*delta
        new_conv[src+dst] -= 2*delta*delta
        new_score = 2.0 * n * new_conv.max()
        if new_score < best_score - 1e-12:
            # accept
            a[src] -= delta
            a[dst] += delta
            conv = new_conv
            best_score = new_score
            best_seq = a.copy()
            no_improve = 0
            accepted += 1
            if accepted % 1000 == 0:
                print(f"[{time.time()-start_time:.2f}s] Hill inc accepted={accepted} best={best_score:.6f}")

            no_improve += 1

            remaining = deadline - time.time()
            total = budget_s

            temp = 0.001 * remaining / total

"""
Improved search for a non‑negative step‑function that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in several phases (all respecting a 1000 s wall‑clock
budget):

1️⃣  Coarse parametric sweep on a modest size (n≈2048) to obtain a good
    starting shape (geometric, power‑law, Beta families, two‑exponential
    mixture).

2️⃣  Random Dirichlet draws (many concentrations) on the same size.

3️⃣  Sub‑gradient descent on the *maximum* of the self‑convolution
    (projected onto the probability simplex).  This rapidly flattens
    the peak of the convolution.

4️⃣  Upscaling to a larger size (n≈8192) by simple repetition – the shape
    is preserved while providing a finer search space.

5️⃣  A fast hill‑climb that moves a (potentially large) fraction of mass
    from a “high‑gradient’’ index to a “low‑gradient’’ one.  The
    convolution is updated incrementally (O(n) per move) and the exact
    quadratic update, including the d*d term, is applied.  A tiny simulated‑
    annealing component and periodic jitter help escape plateaus.

All intermediate best scores are printed.  The final list is normalised
to sum = 1 and returned as plain Python floats.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (real vectors)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = rfft(a, n=L)
    conv = irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Objective helpers (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max()), conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a.
    If several shifts attain the maximum the gradients are averaged.
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    if active.size:
        grad = 2.0 * grad / active.size
    return grad


# ----------------------------------------------------------------------
# 1️⃣  Parametric families on a moderate size (n_small)
# ----------------------------------------------------------------------
def _sweep_geometric(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    uniform = np.full(n, 1.0 / n)
    eps_grid = np.logspace(-5, -1, 200)
    for eps in eps_grid:
        if time.time() >= deadline:
            break
        r = 1.0 - eps
        geo = np.power(r, idx)
        if geo.sum() == 0:
            continue
        geo /= geo.sum()
        for lam in np.linspace(0.0, 1.0, 11):
            seq = lam * uniform + (1.0 - lam) * geo
            seq /= seq.sum()
            sc, _ = _score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time()-start_time:.2f}s] Geo eps={eps:.1e} lam={lam:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_power(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    for p in np.linspace(-2.0, 2.0, 151):
        if time.time() >= deadline:
            break
        seq = np.power(idx + 1.0, p)
        if seq.sum() == 0:
            continue
        seq /= seq.sum()
        sc, _ = _score(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time()-start_time:.2f}s] Power p={p:.3f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_beta(n, best_seq, best_score, deadline):
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1‑based
    rev = np.arange(n, 0, -1, dtype=np.float64)        # n,…,1
    alphas = np.linspace(0.05, 4.0, 81)
    for a in alphas:
        if time.time() >= deadline:
            break
        log_a = (a - 1.0) * np.log(idx)
        for b in alphas:
            if time.time() >= deadline:
                break
            log_seq = log_a + (b - 1.0) * np.log(rev)
            seq = np.exp(log_seq - np.max(log_seq))
            s = seq.sum()
            if s == 0 or not np.isfinite(s):
                continue
            seq /= s
            sc, _ = _score(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time()-start_time:.2f}s] Beta a={a:.2f} b={b:.2f} score={sc:.6f}")
    return best_seq, best_score


def _sweep_two_exponential(n, best_seq, best_score, deadline):
    idx = np.arange(n, dtype=np.float64)
    r_grid = np.linspace(0.90, 0.995, 30)
    for r in r_grid:
        if time.time() >= deadline:
            break
        left = np.power(r, idx)
        right = np.power(r, n - 1 - idx)
        left /= left.sum()
        right /= right.sum()
        uniform = np.full(n, 1.0 / n)
        for w_left in np.linspace(0.0, 0.6, 7):
            if time.time() >= deadline:
                break
            for w_right in np.linspace(0.0, 0.6, 7):
                w_unif = 1.0 - w_left - w_right
                if w_unif < 0.0:
                    continue
                seq = w_left * left + w_right * right + w_unif * uniform
                seq /= seq.sum()
                sc, _ = _score(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[{time.time()-start_time:.2f}s] TwoExp r={r:.3f} wL={w_left:.2f} wR={w_right:.2f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers (still on n_small)
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline, rng, trials=300):
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-2.0, 0.7)               # concentration between ~0.01 and 5
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc, _ = _score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time()-start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Sub‑gradient descent on max‑convolution (any n)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline, best_seq, best_score):
    """
    Projected sub‑gradient descent that directly minimises max(conv(a,a)).
    """
    a = a0.copy()
    n = a.shape[0]
    iteration = 0
    eta0 = 0.5                                 # initial step‑size
    while time.time() < deadline:
        iteration += 1
        cur_score, conv = _score(a)
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time()-start_time:.2f}s] Subgrad it={iteration} score={cur_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _subgradient_maxconv(a, conv)
        eta = eta0 / np.sqrt(iteration)        # diminishing step
        a_new = a - eta * grad
        a_new = _project_simplex(np.maximum(a_new, 0.0))
        a = a_new
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Fast hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------
def _hill_climb_incremental(a: np.ndarray, conv: np.ndarray,
                            deadline, best_seq, best_score, rng):
    """
    Move a (potentially large) fraction of mass from a high‑gradient index
    to a low‑gradient one.  The convolution is updated in O(n) using the
    exact quadratic formula (including the d*d term).  A tiny simulated‑
    annealing component and periodic jitter help escape plateaus.
    """
    n = a.shape[0]
    L = conv.shape[0]
    best_a = a.copy()
    no_improve = 0
    accepted = 0

    while time.time() < deadline:
        # choose src / dst using sub‑gradient information
        grad = _subgradient_maxconv(a, conv)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)
        if src == dst:
            src, dst = rng.choice(n, size=2, replace=False)

        # fraction of source mass to relocate – adaptively large
        frac = rng.uniform(0.01, 0.30)            # between 1 % and 30 %
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        # ----- simulate move (exact quadratic update) -----
        new_conv = conv.copy()
        # linear part 2*delta*(a shifted from dst – a shifted from src)
        new_conv[src:src + n] -= 2.0 * delta * a
        new_conv[dst:dst + n] += 2.0 * delta * a
        # d*d contributions
        new_conv[2 * src] += delta * delta
        new_conv[2 * dst] += delta * delta
        new_conv[src + dst] -= 2.0 * delta * delta

        new_score = float(2.0 * n * new_conv.max())

        # ----- acceptance test -----
        if new_score < best_score - 1e-12:
            # good move – accept
            a[src] -= delta
            a[dst] += delta
            conv = new_conv
            best_score, best_a = new_score, a.copy()
            no_improve = 0
            accepted += 1
            if accepted % 2000 == 0:
                print(f"[{time.time()-start_time:.2f}s] Hill inc acc={accepted} best={best_score:.6f}")
        else:
            # possibly accept a worsening move (simulated annealing)
            remaining = deadline - time.time()
            temp = 0.001 * remaining / (budget_s if 'budget_s' in globals() else 1000.0)
            if rng.random() < np.exp(-(new_score - best_score) / (temp + 1e-12)):
                a[src] -= delta
                a[dst] += delta
                conv = new_conv
            no_improve += 1

        # ----- occasional jitter & re‑normalisation -----
        if no_improve > 5000:
            # small random redistribution to shake out of plateaus
            i1, i2 = rng.choice(n, size=2, replace=False)
            eps = a[i1] * rng.uniform(0.001, 0.005)
            eps = min(eps, a[i1])
            a[i1] -= eps
            a[i2] += eps
            conv = _conv_self(a)                # recompute exact conv
            no_improve = 0

        # keep the simplex constraint numerically exact
        if (accepted % 4000) == 0:
            a_sum = a.sum()
            if abs(a_sum - 1.0) > 1e-12:
                a /= a_sum
                conv = _conv_self(a)

    return best_a, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises the evaluation function.
    The procedure follows:
        * small‑scale parametric sweep + random Dirichlet initialisation,
        * sub‑gradient descent on max‑convolution,
        * up‑scaling to a larger dimension,
        * fast hill‑climb with incremental updates.
    The returned list is ready for ``evaluate_sequence``.
    """
    global start_time, budget_s
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.25          # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1️⃣  Coarse search on a moderate size (n_small)
    # --------------------------------------------------------------
    n_small = 2048
    uniform_small = np.full(n_small, 1.0 / n_small)
    best_seq_small = uniform_small.copy()
    best_score_small, _ = _score(best_seq_small)
    print(f"[{time.time()-start_time:.2f}s] Uniform (n={n_small}) score={best_score_small:.6f}")

    # use a shared deadline for the “early” phases
    early_deadline = start_time + 180.0               # first 3 min for sweeps

    # parametric families
    if time.time() < early_deadline:
        best_seq_small, best_score_small = _sweep_geometric(
            n_small, best_seq_small, best_score_small, early_deadline)
    if time.time() < early_deadline:
        best_seq_small, best_score_small = _sweep_power(
            n_small, best_seq_small, best_score_small, early_deadline)
    if time.time() < early_deadline:
        best_seq_small, best_score_small = _sweep_beta(
            n_small, best_seq_small, best_score_small, early_deadline)
    if time.time() < early_deadline:
        best_seq_small, best_score_small = _sweep_two_exponential(
            n_small, best_seq_small, best_score_small, early_deadline)

    # random Dirichlet seeds
    dirichlet_deadline = start_time + 300.0
    if time.time() < dirichlet_deadline:
        best_seq_small, best_score_small = _random_dirichlet_trials(
            n_small, best_seq_small, best_score_small, dirichlet_deadline, rng)

    # --------------------------------------------------------------
    # 2️⃣  Sub‑gradient descent on the small problem
    # --------------------------------------------------------------
    subgrad_deadline = start_time + 450.0
    if time.time() < subgrad_deadline:
        print(f"[{time.time()-start_time:.2f}s] Starting sub‑gradient descent (n={n_small})")
        best_seq_small, best_score_small = _subgradient_descent(
            best_seq_small, subgrad_deadline, best_seq_small, best_score_small)
        print(f"[{time.time()-start_time:.2f}s] After sub‑grad score={best_score_small:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Upscale to a large size (n_large) – simple repetition
    # --------------------------------------------------------------
    n_large = 8192
    repeat_factor = n_large // n_small
    best_seq_large = np.repeat(best_seq_small, repeat_factor)
    # normalise again (numerical safety)
    best_seq_large = best_seq_large / best_seq_large.sum()
    best_score_large, conv_large = _score(best_seq_large)
    print(f"[{time.time()-start_time:.2f}s] Upscaled to n={n_large} score={best_score_large:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Sub‑gradient descent on the large problem (shorter budget)
    # --------------------------------------------------------------
    subgrad_large_deadline = start_time + 600.0
    if time.time() < subgrad_large_deadline:
        print(f"[{time.time()-start_time:.2f}s] Sub‑gradient on large n")
        best_seq_large, best_score_large = _subgradient_descent(
            best_seq_large, subgrad_large_deadline,
            best_seq_large, best_score_large)
        # recompute exact convolution after the descent
        best_score_large, conv_large = _score(best_seq_large)

    # --------------------------------------------------------------
    # 5️⃣  Fast hill‑climb with incremental updates (majority of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time()-start_time:.2f}s] Starting fast hill‑climb (n={n_large})")
        best_seq_large, best_score_large = _hill_climb_incremental(
            best_seq_large, conv_large, deadline, best_seq_large,
            best_score_large, rng)
        print(f"[{time.time()-start_time:.2f}s] Final hill‑climb score={best_score_large:.6f}")

    # --------------------------------------------------------------
    # Normalise and return plain Python list
    # --------------------------------------------------------------
    best_seq_large = np.maximum(best_seq_large, 0.0)
    total = best_seq_large.sum()
    if total > 0:
        best_seq_large = best_seq_large / total
    else:
        best_seq_large = np.full(n_large, 1.0 / n_large)

    print(f"[{time.time()-start_time:.2f}s] Finished – best score ≈ {best_score_large:.6f}")
    return [float(x) for x in best_seq_large]
