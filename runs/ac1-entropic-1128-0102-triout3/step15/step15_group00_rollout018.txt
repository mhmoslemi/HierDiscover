def _smooth_gradient_descent(a0, deadline, best_seq, best_score):
    a = a0.copy()
    z = np.log(a + 1e-12)  # log to avoid zeros
    n = len(a)
    iteration = 0
    lr0 = 0.5
    theta_max = 5000.0
    while time.time() < deadline:
        iteration += 1
        # compute a from z
        z_shift = z - np.max(z)
        a = np.exp(z_shift)
        a_sum = a.sum()
        a /= a_sum
        # compute conv
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.2f}s] SmoothDescent it={iteration} score={cur_score:.6f}")
        # temperature schedule
        theta = 1.0 + (theta_max - 1.0) * min(1.0, iteration / 2000.0)  # ramp up
        # compute weights
        w = np.exp(theta * conv - np.max(theta * conv))
        w /= w.sum()
        # compute gradient wrt a
        a_rev = a[::-1]
        grad_full = 2.0 * _conv_fft(w, a_rev)  # length len(w)+len(a_rev)-1
        grad = grad_full[n-1 : n-1 + n]   # slice
        # gradient wrt z (softmax)
        sum_ag = np.dot(a, grad)
        grad_z = a * (grad - sum_ag)
        # learning rate schedule
        lr = lr0 / np.sqrt(iteration)
        # optional momentum
        # update z
        z -= lr * grad_z
    return best_seq, best_score
```

"""
Smooth‐entropy optimisation of a non‑negative step function.

The objective (for a sequence a with Σ a=1) is

        F(a) = 2·n·max_k (a*a)[k] ,

i.e. minimise the peak of the full linear convolution.
The function is non‑convex because the max operator is non‑smooth.
We replace max by a smooth “soft‑max’’ (log‑sum‑exp) with a temperature
parameter θ that is slowly increased.  For a given temperature the
objective becomes differentiable and the gradient can be written in closed
form using convolutions, which we evaluate with FFTs.  The variables are
optimised in the log‑domain (z_i = log a_i) so that the simplex constraint
(a_i≥0, Σa_i=1) is satisfied automatically by a softmax.

The optimiser is an Adam‑style stochastic gradient descent
with a schedule for θ (from 1 up to a few thousand).  Several random
initialisations (Dirichlet draws) are tried and the best sequence found
within the 1000 s budget is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using the FFT."""
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective helpers (the sequence is always normalised to sum = 1)
# ----------------------------------------------------------------------
def _objective_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = _conv_self(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth optimisation (soft‑max of the convolution)
# ----------------------------------------------------------------------
def _adam_smooth_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective.
    a0       – starting probability vector (sum = 1, non‑negative)
    deadline – absolute time at which to stop
    best_seq / best_score – current global best, updated whenever an
                            improvement is found.
    Returns the (possibly) improved best_seq and best_score.
    """
    n = a0.shape[0]
    # latent variables – a = softmax(z)
    z = np.log(a0 + 1e-14)
    m = np.zeros_like(z)        # first moment
    v = np.zeros_like(z)        # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    # temperature schedule (goes from 1 to theta_max)
    theta_max = 4000.0
    # learning‑rate schedule (base step size)
    base_lr = 0.12

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # a = softmax(z)  – stays on the simplex automatically
        # --------------------------------------------------------------
        z_shift = z - np.max(z)               # numerical stability
        a = np.exp(z_shift)
        a /= a.sum()                          # sum‑to‑one

        # full self‑convolution and current (exact) score
        conv = _conv_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] AdamDescent it={it:,} "
                f"score={cur_score:.6f}"
            )

        # --------------------------------------------------------------
        # smooth approximation of the max: soft‑max with temperature θ
        # --------------------------------------------------------------
        # linear ramp of the temperature (first 2000 iters)
        theta = 1.0 + (theta_max - 1.0) * min(1.0, it / 2000.0)
        # stable soft‑max over the convolution values
        w = np.exp(theta * conv - np.max(theta * conv))
        w /= w.sum()          # w has length 2·n‑1 and sums to 1

        # --------------------------------------------------------------
        # gradient of the smoothed objective w.r.t. a
        #   ∂/∂a_i  Σ_k w_k (a * a)[k]  =  2 Σ_k w_k a_{k-i}
        # This is a cross‑correlation (w ⋆ a)[i]; we obtain it via FFT.
        # --------------------------------------------------------------
        a_rev = a[::-1]                     # reversed a
        grad_full = 2.0 * _conv_fft(w, a_rev)   # length 3·n‑2
        grad = grad_full[n - 1 : n - 1 + n]      # slice of length n

        # --------------------------------------------------------------
        # chain rule through the soft‑max (a = softmax(z))
        #   ∂F/∂z_i = a_i * (∂F/∂a_i - Σ_j a_j ∂F/∂a_j)
        # --------------------------------------------------------------
        avg = np.dot(a, grad)                # Σ_j a_j ∂F/∂a_j
        grad_z = a * (grad - avg)

        # --------------------------------------------------------------
        # Adam update on the latent variables
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / (np.sqrt(v_hat) + eps)
        z -= lr * m_hat                     # gradient step

    return best_seq, best_score


def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    """Generate many Dirichlet draws (different concentration parameters)."""
    rng = np.random.default_rng()
    # number of draws is calibrated to the remaining budget
    while time.time() < deadline:
        logc = rng.uniform(-3.0, 0.8)          # concentration ≈ 0.001 … 6.3
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(
                f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.4g} "
                f"score={sc:.6f}"
            )
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

            if s > 0.0:

    else:
