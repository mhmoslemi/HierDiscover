The objective  

\[
E(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

is *scale‑invariant*: after normalising the vector to a probability
distribution (`∑a = 1`) the denominator is constant and we only have to
minimise  

\[
F(a)=2\,n\;\max (a*a)\; .
\]

`a*a` is the full self‑convolution of the non‑negative vector `a`.
The maximal entry of the convolution is attained at a (small) set of
shifts `k`.  For each shift we can write the contribution as a linear
form  

\[
(a*a)[k]=\sum_i a_i\,a_{k-i} .
\]

Hence the sub‑gradient of the maximal entry w.r.t. `a` can be computed
very cheaply – it is just the (normalised) sum of the vectors
`a_{k-i}` over all active shifts `k`.  Using this sub‑gradient we move
mass from positions with a large influence on the maximal convolution
value to positions with a small influence.  The move is performed on
the probability simplex by a cheap Euclidean projection.

The search is split into three stages

1. **Diverse initialisers** – a few parametric families (geometric,
   exponential, two‑block) and many random Dirichlet draws are
   evaluated; the best among them becomes the starting point.
2. **Projected sub‑gradient descent** – repeatedly
   *‑move mass from the index with the largest sub‑gradient to the one
   with the smallest*.
   The step size (`α`) decays linearly from `0.2` to `0.001` during the
   run, guaranteeing ever finer adjustments.
3. **Exploration / random shake‑ups** – after a long streak without
   improvement a tiny random perturbation is added; occasionally a
   purely random transfer is tried and accepted if it improves the
   score.  Symmetry (`a_i = a_{n-1-i}`) is enforced after every
   modification, which reduces the dimensionality of the problem and
   speeds up convergence.

All operations are `O(n log n)` because the convolution is computed
with an FFT.  With `n = 4096` a single evaluation costs less than a
millisecond, allowing several hundred thousand iterations in the
available 1000 s budget.

The final vector is projected onto the simplex one last time,
clamped to the interval `[0,1000]` (the limits enforced by
`evaluate_sequence`) and returned as a plain Python list.

```python
"""Improved search for a non‑negative step function that minimises
   evaluate_sequence (i.e. 2·n·max(conv(a,a)) / (∑a)²).

   The algorithm consists of:
   1. a quick sweep over a few parametric families and many random
      Dirichlet draws,
   2. a projected sub‑gradient descent that repeatedly transfers mass
      from “high‑impact” to “low‑impact” coordinates,
   3. occasional random perturbations (simulated‑annealing style) to
      escape plateaus.

   All vectors are kept normalised (∑a = 1) and symmetric (a[i] = a[-1-i]),
   which strongly reduces the effective dimension and improves the
   quality of the final solution.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, simplex projection, …)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using an FFT."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (v≥0, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _objective_score(a: np.ndarray) -> float:
    """Score = 2·n·max(conv(a,a)) with ∑a = 1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _raw_gradient(a: np.ndarray, conv: np.ndarray, max_val: float,
                  tol: float = 1e-12) -> tuple[np.ndarray, np.ndarray]:
    """
    Return the *unscaled* sub‑gradient of the maximal convolution entry
    and the list of active shift indices.
    """
    n = a.shape[0]
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]          # contribution a_{k-i}
    return grad, active


# ----------------------------------------------------------------------
# Stage 1 – diverse initialisers
# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator,
                        deadline: float, best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """Try a handful of parametric families and many Dirichlet draws."""
    # --- simple parametric families ------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # geometric decay (a_i ∝ r^i)
    for eps in np.logspace(-5, -1, 30):
        if time.time() >= deadline:
            return best_seq, best_score
        r = 1.0 - eps
        geo = r ** idx
        if geo.sum() == 0:
            continue
        seq = geo / geo.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] geo eps={eps:.1e} score={sc:.6f}")

    # symmetric exponential decay a_i ∝ exp(-λ·min(i,n‑1‑i))
    dist = np.minimum(idx, n - 1 - idx)
    for lam in np.logspace(-4, -0.2, 30):
        if time.time() >= deadline:
            return best_seq, best_score
        seq = np.exp(-lam * dist)
        seq /= seq.sum()
        sc = _objective_score(seq)
        if sc < best_score:
            best_seq, best_score = seq, sc
            print(f"[{time.time() - start_time:.2f}s] exp λ={lam:.1e} score={sc:.6f}")

    # two uniform blocks at the ends
    for L_frac in np.linspace(0.01, 0.3, 20):
        L = max(1, int(L_frac * n))
        for w in np.linspace(0.1, 0.9, 9):
            if time.time() >= deadline:
                return best_seq, best_score
            seq = np.zeros(n)
            seq[:L] = w / L
            seq[-L:] = (1.0 - w) / L
            sc = _objective_score(seq)
            if sc < best_score:
                best_seq, best_score = seq, sc
                print(f"[{time.time() - start_time:.2f}s] TwoBlocks L={L} w={w:.2f} score={sc:.6f}")

    # --- random Dirichlet draws ----------------------------------------------------
    trials = 400
    for _ in range(trials):
        if time.time() >= deadline:
            break
        logc = rng.uniform(-3.0, 1.0)          # concentration in ≈[0.001, 10]
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        if a.sum() == 0:
            continue
        a /= a.sum()
        sc = _objective_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – projected sub‑gradient descent (mass transfer)
# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray, deadline: float,
                     best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of the mass from the coordinate with the
    largest (unscaled) sub‑gradient to the one with the smallest.
    The step size α decays linearly from 0.2 to 0.001.
    After a long period without improvement a tiny random shake‑up is
    applied (simulated‑annealing flavour).  Symmetry a[i]=a[-1-i] is
    enforced at every step.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.shape[0]
    no_improve = 0
    total_time = deadline - start_time

    while time.time() < deadline:
        conv = _conv_self(a)
        max_val = conv.max()

        # compute raw gradient and active set
        grad, _ = _raw_gradient(a, conv, max_val)

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # linear decay of the transfer fraction α
        time_left = deadline - time.time()
        α_max, α_min = 0.20, 0.001
        α = α_min + (α_max - α_min) * (time_left / total_time)

        # actual amount to move
        transfer = a[src] * α
        if transfer <= 0.0:
            # should not happen, fall back to a tiny random move
            candidates = np.where(a > 1e-12)[0]
            if candidates.size == 0:
                break
            src = int(rng.choice(candidates))
            dst = rng.integers(n)
            transfer = a[src] * α

        a_candidate = a.copy()
        a_candidate[src] -= transfer
        a_candidate[dst] += transfer

        # enforce symmetry and project onto the simplex
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
        a_candidate = _project_simplex(a_candidate)

        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score, best_seq = cand_score, a_candidate.copy()
            a = a_candidate
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Refine improved score={best_score:.6f}")
            continue

        # no improvement – try a random transfer with small probability
        no_improve += 1
        if rng.random() < 0.01:      # 1 % chance
            src2 = int(rng.choice(np.where(a > 1e-12)[0]))
            dst2 = rng.integers(n)
            α2 = rng.random() * α_max
            tr2 = a[src2] * α2
            a_rand = a.copy()
            a_rand[src2] -= tr2
            a_rand[dst2] += tr2
            a_rand = (a_rand + a_rand[::-1]) * 0.5
            a_rand = _project_simplex(a_rand)
            sc2 = _objective_score(a_rand)
            if sc2 < best_score:
                best_score, best_seq = sc2, a_rand.copy()
                a = a_rand
                no_improve = 0
                print(f"[{time.time() - start_time:.2f}s] Random move improved score={best_score:.6f}")
                continue

        # after a long dry spell inject a tiny Gaussian perturbation
        if no_improve > 6000:
            a = a + rng.normal(scale=0.001, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            a = (a + a[::-1]) * 0.5
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a list of floats) before the time
    budget expires.
    """
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin

    rng = np.random.default_rng(seed)

    # problem size; 4096 works well and was used in the reference code
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional global starter (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        starter = np.asarray(height_sequence_1, dtype=float)
        if starter.ndim == 1 and starter.size > 0:
            starter = np.maximum(starter, 0.0)
            s = starter.sum()
            if s > 0:
                starter = starter / s
                sc = _objective_score(starter)
                if sc < best_score:
                    best_seq, best_score = starter.copy(), sc
                    print(f"[{time.time() - start_time:.2f}s] Global starter score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Stage 1 – generate a good starting point
    # ------------------------------------------------------------------
    stage1_deadline = deadline - 300.0               # reserve time for refinement
    if time.time() < stage1_deadline:
        best_seq, best_score = _initial_candidates(
            n, rng, stage1_deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (mass transfer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting refinement")
        best_seq, best_score = _refine_sequence(
            best_seq, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # Final normalisation and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
```
