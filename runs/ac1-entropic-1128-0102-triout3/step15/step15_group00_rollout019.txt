def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:len(x) + len(y) - 1]
    return conv

def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    # Euclidean projection onto simplex.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
    else:
        w = np.full_like(v, 1.0 / n)
    return w

def _objective_score(a: np.ndarray) -> float:
    # assumes sum(a)=1, returns 2*n*max(conv(a,a))
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())

def _smooth_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective 2*n*softmax_beta(conv).
    Returns the gradient w.r.t. a (no projection).
    """
    n = a.shape[0]
    conv = _conv_self(a)
    # softmax weights (stable)
    max_val = conv.max()
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:

    idx = (2 * n - 2) - np.arange(n)          # select the needed slice
    term = full[idx]
    grad = term + a                           # because Σ_i w_i = 1 adds a_j

        # concentration sampled log‑uniformly; very small → sparse,

def _adam_descent(start_a, deadline, best_seq, best_score):
    """
    Projected Adam optimisation on the smoothed max‑convolution.
    beta grows from 5 to 80 during the run.
    """
    a = start_a.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta0, beta_max = 5.0, 80.0
    lr = 0.8                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0
    # pre‑compute a crude estimate of total iterations we can afford
    while time.time() < deadline:
        t += 1
        # linearly increase beta over the run
        progress = (time.time() - start_time) / (deadline - start_time)
        beta = beta0 + progress * (beta_max - beta0)

        grad = _smooth_gradient(a, beta)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _renorm_positive(a)

        # occasional evaluation (every 200 steps)
        if t % 200 == 0:
            sc = _objective_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={t} beta={beta:.1f} score={sc:.6f}")

        # occasional random shake‑up to avoid stagnation
        if t % 1500 == 0:
            rng = np.random.default_rng()
            mask = rng.random(a.shape) < 0.02
            a[mask] *= (rng.random(np.count_nonzero(mask)) * 0.5 + 0.75)
            a = _renorm_positive(a)

    return best_seq, best_score

    a = start_a.copy()

            best_score = cur_score
            no_improve = 0

                a[i2] += delta
                a = _renorm_positive(a)
                no_improve = 0
    return best_a, best_score

"""
Search for a non‑negative step‑function that minimises the
``evaluate_sequence`` objective.

The strategy mixes several ideas that differ from the reference
implementation:

1.  A dedicated parametric sweep over common signal‑processing windows
    (Hann, Hamming, Blackman, sine, triangular) – these shapes are
    smooth, symmetric and often give a flatter self‑convolution than a
    plain geometric or power law.

2.  A large batch of random Dirichlet draws with a wide range of
    concentration parameters to obtain diverse starting points.

3.  A *smooth‑max* Adam optimiser.  Instead of the sub‑gradient of the
    exact max‑convolution, we approximate the max by a soft‑max
    (log‑sum‑exp) with a temperature β that grows during the run.
    This yields a smooth gradient that can be followed with a
    projected Adam update, efficiently computed by FFT‑based convolutions.

4.  A final hill‑climb that transfers a modest fraction of mass from
    coordinates with large sub‑gradients to those with small ones,
    together with occasional random shake‑ups to escape plateaus.

All steps respect the 1000 s budget, normalise the sequence to unit
L₁‑norm (the evaluator rescales anyway) and keep every coefficient
non‑negative.  The best sequence discovered is returned as a plain
Python list ready for ``evaluate_sequence``.
"""

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:

    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[: len(x) + len(y) - 1]
    return conv

    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]

    Af = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Af * Af, n=L)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0

    else:

    return w

    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        v = v / s
    else:
        v = np.full_like(v, 1.0 / v.shape[0])
    return v

def _objective_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at a (averaged over all active shifts).
    """
    n = a.shape[0]
    max_val = conv.max()
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)

        idx = np.arange(lo, hi + 1, dtype=int)

    """
    Gradient of the smoothed objective

    conv = _conv_self(a)
    max_val = conv.max()

    # Blackman
    seq = (0.42 - 0.5 * np.cos(2.0 * np.pi * t / (n - 1))
           + 0.08 * np.cos(4.0 * np.pi * t / (n - 1)))
    seq /= seq.sum()
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Blackman window score={sc:.6f}")

    # Sine (raised‑sine)
    seq = np.sin(np.pi * (t + 0.5) / n)
    seq /= seq.sum()
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Sine window score={sc:.6f}")

    # Triangular (Bartlett)
    seq = 1.0 - np.abs(2.0 * t / (n - 1) - 1.0)
    seq /= seq.sum()
    sc = _objective_score(seq)
    if sc < best_score:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time() - start_time:.2f}s] Triangular window score={sc:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 2️⃣  Random Dirichlet initialisers
# ----------------------------------------------------------------------
def _random_dirichlet_trials(n, best_seq, best_score, deadline):
    rng = np.random.default_rng()
    trials = 400
    for _ in range(trials):
        if time.time() >= deadline:
            break
        # log‑uniform concentration in a wide range
        logc = rng.uniform(-2.5, 0.8)
        c = 10 ** logc
        a = rng.gamma(shape=c, scale=1.0, size=n)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        sc = _objective_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time() - start_time:.2f}s] Dirichlet c={c:.3f} score={sc:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def _adam_descent(start_a, deadline, best_seq, best_score):
    """
    Projected Adam optimisation on the smooth max‑convolution objective.
    The inverse temperature β grows from 5 to 80 during the run.
    """
    a = start_a.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta0, beta_max = 5.0, 80.0      # temperature schedule
    lr = 0.8                         # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        # linear increase of β with elapsed time
        prog = (time.time() - start_time) / (deadline - start_time)
        beta = beta0 + prog * (beta_max - beta0)

        grad = _smooth_gradient(a, beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _renorm_positive(a)

        # periodic evaluation
        if t % 200 == 0:
            sc = _objective_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.time() - start_time:.2f}s] Adam it={t} β={beta:.1f} score={sc:.6f}")

        # tiny random jitter every few thousand steps to keep exploration alive
        if t % 1500 == 0:
            rnd = np.random.default_rng()
            mask = rnd.random(a.shape) < 0.02
            a[mask] *= (rnd.random(np.count_nonzero(mask)) * 0.5 + 0.75)
            a = _renorm_positive(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4️⃣  Final hill‑climb (enhanced version of the reference)
# ----------------------------------------------------------------------
def _hill_climb_enhanced(start_a, deadline):
    a = start_a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective_score(a)
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline:
        conv = _conv_self(a)
        grad = _subgradient_maxconv(a, conv)

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break

        src = src_candidates[np.argmax(grad[src_candidates])]
        dst = np.argmin(grad)

        max_frac = 0.08
        frac = rng.random() * max_frac + 1e-5
        eps = a[src] * frac
        if eps <= 0:
            continue

        a_prop = a.copy()
        a_prop[src] -= eps
        a_prop[dst] += eps
        a_prop = _renorm_positive(a_prop)

        cur_score = _objective_score(a_prop)
        if cur_score < best_score:
            a = a_prop
            best_a = a.copy()
            best_score = cur_score
            no_improve = 0
            print(f"[{time.time() - start_time:.2f}s] Hill++ score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2500:
                i1, i2 = rng.choice(n, size=2, replace=False)
                delta = a[i1] * (rng.random() * max_frac + 1e-5)
                a[i1] -= delta
                a[i2] += delta
                a = _renorm_positive(a)
                no_improve = 0
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine runs for at most
    ``budget_s`` seconds, combines a window‑function sweep, random
    Dirichlet starts, a smooth‑max Adam optimiser and a final hill‑climb,
    and returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    global start_time
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size (FFT works best with powers of two)
    # --------------------------------------------------------------
    n = 4096
    uniform = np.full(n, 1.0 / n)

    best_seq = uniform.copy()
    best_score = _objective_score(best_seq)
    print(f"[{time.time() - start_time:.2f}s] Uniform baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Window‑function sweep
    # --------------------------------------------------------------
    phase_deadline = deadline - 300.0
    if time.time() < phase_deadline:
        best_seq, best_score = _sweep_windows(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 2️⃣  Random Dirichlet initialisers
    # --------------------------------------------------------------
    phase_deadline = deadline - 200.0
    if time.time() < phase_deadline:
        best_seq, best_score = _random_dirichlet_trials(n, best_seq, best_score, phase_deadline)

    # --------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam descent
    # --------------------------------------------------------------
    md_deadline = deadline - 120.0
    if time.time() < md_deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting Adam descent")
        best_seq, best_score = _adam_descent(best_seq, md_deadline, best_seq, best_score)
        print(f"[{time.time() - start_time:.2f}s] After Adam descent best score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time() - start_time:.2f}s] Starting final hill‑climb")
        best_seq, best_score = _hill_climb_enhanced(best_seq, deadline)
        print(f"[{time.time() - start_time:.2f}s] Final hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Normalise and return as plain list
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = uniform.copy()

    print(f"[{time.time() - start_time:.2f}s] Finished – best score ≈ {best_score:.6f}")
    return [float(x) for x in best_seq]
