"""
Optimisation of a non‑negative step‑function (a probability vector)
minimising

        2·n·max_i (a * a)_i        ( * = discrete convolution )

The objective is invariant under scaling, therefore we restrict the search
to the probability simplex {a_i ≥ 0, Σ a_i = 1}.  The non‑smooth max is
approximated by a smooth log‑sum‑exp (soft‑max) with an increasing
temperature parameter *c*.  For a given *c* the gradient is

        ∂/∂a_i  ( 1/c·log Σ_k e^{c·(a*a)_k} )
            = 2· Σ_k w_k a_{k‑i} ,

where w_k are the normalised soft‑max weights.
We perform projected gradient descent on the simplex,
starting from the best known construction (``height_sequence_1``) and from a
few random initialisations for several lengths.  The temperature *c* is
gradually increased so that the algorithm first smooths the landscape and
then focuses on the true maximum.  A simple Adam‑like step‑size schedule
and occasional random jitter are used to escape shallow plateaus.
The best sequence found before the time budget expires is returned.
"""

import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i ≥ 0, Σ x_i = 1 }.
    Implements the algorithm of [Chen & Ye 2011].
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1        # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_and_grad(seq: np.ndarray, c: float):
    """
    For a probability vector ``seq`` (sum = 1) compute

        smooth = (1/c)·log Σ_k exp(c·conv_k)   (≈ max(conv))
        grad   = ∂smooth/∂seq  (shape = seq.shape)
        max_conv = max(conv)

    where ``conv = np.convolve(seq, seq)``.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)               # length 2·n‑1
    # soft‑max weights, numerically stable
    max_c = conv.max()
    exp_vals = np.exp(c * (conv - max_c))
    w = exp_vals / exp_vals.sum()              # Σ w = 1

    # gradient = 2 * correlation(w, seq)   (see derivation in the docstring)
    # np.correlate returns length 3·n‑2, shift 0 is at index n‑1
    corr = np.correlate(w, seq, mode='full')
    grad = 2.0 * corr[n - 1 : n - 1 + n]       # slice the part corresponding to shifts 0…n‑1

    # smooth approximation of the max
    smooth = max_c + np.log(exp_vals.sum()) / c
    return smooth, grad, conv.max()


def _compute_score(seq: np.ndarray) -> float:
    """
    Direct computation of the evaluation function when Σ a_i = 1:
        score = 2·n·max(conv(seq, seq))
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights that approximately minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget,
    explores several sequence lengths, and uses a smoothed L∞ objective
    together with projected (Adam‑style) gradient descent.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Load the previously best known construction if available
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    if 'height_sequence_1' in globals():
        try:
            arr = np.array(list(height_sequence_1), dtype=float)
            if arr.sum() == 0.0:
                # fall back to uniform if the loaded vector is degenerate
                arr = np.full_like(arr, 1.0 / arr.size)
            else:
                arr = arr / arr.sum()          # normalise to a probability vector
            best_score = _compute_score(arr)
            best_seq = arr.tolist()
            print(f"[init] Loaded height_sequence_1 – score {best_score:.6f}")
        except Exception as exc:
            print("[init] Failed to load height_sequence_1:", exc)

    # ------------------------------------------------------------------
    # Parameter sweep over several lengths
    # ------------------------------------------------------------------
    lengths = [1200, 2000, 2800, 3500, 4200, 5000]
    max_total_iters = 40000                     # rough cap per length; early stop on time

    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # Generate a few initial seeds for this length
        # ------------------------------------------------------------------
        seeds = []

        # 1. uniform distribution (already 2.0, but provides a baseline)
        seeds.append(np.full(n, 1.0 / n))

        # 2. power‑law family – coarse sweep of β ∈ [0.2, 2.0]
        betas = np.linspace(0.2, 2.0, 9)
        for beta in betas:
            idx = np.arange(1, n + 1, dtype=float)
            seq = idx ** (-beta)
            seq /= seq.sum()
            seeds.append(seq)

        # 3. random Dirichlet draws (few of them)
        for _ in range(3):
            seq = np.random.dirichlet(np.ones(n))
            seeds.append(seq)

        # 4. the globally best known sequence, stretched/compressed to length n
        if best_seq is not None:
            old = np.array(best_seq, dtype=float)
            # simple linear interpolation to new length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, old)
            interp = np.maximum(interp, 0.0)
            if interp.sum() == 0.0:
                interp = np.full_like(interp, 1.0 / n)
            else:
                interp /= interp.sum()
            seeds.append(interp)

        # ------------------------------------------------------------------
        # Run the optimiser from each seed
        # ------------------------------------------------------------------
        for seed_id, init_seq in enumerate(seeds):
            if time.time() >= deadline:
                break

            seq = init_seq.copy()
            seq = np.maximum(seq, 0.0)
            if seq.sum() == 0.0:
                seq = np.full_like(seq, 1.0 / n)
            else:
                seq = seq / seq.sum()          # ensure simplex

            # Adam‑style accumulators
            m = np.zeros_like(seq)
            v = np.zeros_like(seq)

            # smoothing temperature schedule
            c = 1.0               # start very smooth
            c_max = 80.0
            c_mul = 1.018         # multiplicative increase per iteration

            # step‑size schedule (base learning rate scaled by 1/√t)
            lr0 = 0.35 / n        # initial base LR; tuned empirically
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8

            # track local improvement for possible restarts
            last_improve_iter = 0
            cur_score = _compute_score(seq)

            for it in range(max_total_iters):
                if time.time() >= deadline:
                    break

                # ---------- compute smooth objective and its gradient ----------
                _, grad, max_conv = _smooth_max_and_grad(seq, c)

                # ---------- Adam update ----------
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad ** 2)

                m_hat = m / (1.0 - beta1 ** (it + 1))
                v_hat = v / (1.0 - beta2 ** (it + 1))

                eta = lr0 / np.sqrt(it + 1)
                step = eta * m_hat / (np.sqrt(v_hat) + eps)

                # ---------- projected descent ----------
                new_seq = seq - step
                new_seq = _project_simplex(new_seq)

                # optional monotonicity (helps to keep a “nice” shape)
                new_seq = np.sort(new_seq)[::-1]

                # ---------- evaluate ----------
                new_score = _compute_score(new_seq)

                # ---------- accept if improvement ----------
                if new_score + 1e-12 < cur_score:
                    seq = new_seq
                    cur_score = new_score
                    last_improve_iter = it
                else:
                    # occasional stochastic acceptance to escape flat regions
                    if np.random.rand() < 0.005:
                        seq = new_seq
                        cur_score = new_score

                # ---------- global best tracking ----------
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = seq.tolist()
                    print(f"[global] New best {best_score:.6f} (n={n}, it={it}, seed={seed_id})")

                # ---------- temperature update ----------
                c = min(c_max, c * c_mul)

                # ---------- jitter to break symmetry ----------
                if (it + 1) % 500 == 0:
                    jitter = (np.random.rand(n) - 0.5) * eta * 0.12
                    seq = _project_simplex(np.maximum(seq + jitter, 0.0))
                    seq = np.sort(seq)[::-1]

                # ---------- restart if stagnated ----------
                if it - last_improve_iter > 3000:
                    # re‑initialise from a fresh Dirichlet sample
                    seq = np.random.dirichlet(np.ones(n))
                    m.fill(0.0)
                    v.fill(0.0)
                    last_improve_iter = it
                    # keep the temperature at its current value (no reset)
                    continue

    # ------------------------------------------------------------------
    # Fallback – if nothing was found return a modest uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 800
        best_seq = (np.full(n, 1.0 / n)).tolist()
        best_score = _compute_score(np.array(best_seq))
        print("[fallback] Returning uniform sequence, score", best_score)

    # Final sanity check (evaluate_sequence does the same scaling internally)
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"\n>>> Final candidate score (evaluate_sequence) = {final_score:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence failed on final candidate:", exc)

    return best_seq
