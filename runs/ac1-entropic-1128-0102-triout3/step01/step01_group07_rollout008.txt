def _eg_step(seq, eta):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    # gradient g_i = 2 * a_{k-i} if i in range.
    grad = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1)
    grad[idx] = 2 * seq[k - idx]  # all positive.
    # exponentiated update
    new_seq = seq * np.exp(-eta * grad)
    # renormalize onto simplex (sum to 1)
    new_seq = new_seq / new_seq.sum()
    # enforce monotonic decreasing condition optionally
    new_seq = np.sort(new_seq)[::-1]
    return new_seq

"""
Propose a candidate step‑function sequence.

Algorithm – smoothed‑max Adam optimisation
------------------------------------------------
* Work on a single length (default 3000, or the length of the
  previously known construction ``height_sequence_1``).
* Initialise the probability vector either from the known construction
  (with a tiny random perturbation) or from a random Dirichlet draw.
* The true objective is 2·n·max(conv(a,a)) (the ℓ₁‑norm of ``a`` is kept
  equal to one).  Direct optimisation of the max is difficult because
  the sub‑gradient jumps; instead we minimise a smooth approximation:
      f_λ(a) = (1/λ)·log Σ_k exp( λ·(a*a)_k )
  whose gradient is
      ∇f_λ(a)_i = 2·Σ_k w_k·a_{k-i},
  where w_k = exp(λ·conv_k)/Σ_j exp(λ·conv_j) is a softmax over the
  convolution entries.
* The gradient is obtained efficiently with a correlation operation.
* An Adam optimiser (first‑ and second‑moment adaptivity) is applied,
  followed by a Euclidean projection onto the probability simplex.
  No monotonicity constraint is imposed – the optimiser can freely
  discover non‑monotone shapes that flatten the self‑convolution.
* The smoothing parameter λ is increased slowly (continuation method)
  from a small start (λ≈0.1) up to a larger value (λ≈20).  Early iterations
  move the mass globally; later iterations behave like a sub‑gradient step
  on the true max.
* The best sequence seen (according to the exact ``evaluate_sequence``
  function) is stored and returned.  The whole search respects the
  1000‑second wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1          # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute softmax weights w_k = exp(lam*conv_k) / Σ exp(lam*conv_k)
    in a numerically stable way.
    """
    # shift by the maximum to avoid overflow
    cmax = conv.max()
    exp_vals = np.exp(lam * (conv - cmax))
    w = exp_vals / exp_vals.sum()
    return w


def _gradient_smoothed_max(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed maximum (log‑sum‑exp) of the convolution
    a*a with smoothing parameter lam.
    """
    # full convolution (length 2n‑1)
    conv = np.convolve(a, a)
    w = _softmax_weights(conv, lam)          # shape (2n‑1,)
    # correlation: grad_i = 2 Σ_k w_k a_{k-i}
    # np.correlate with mode='valid' yields exactly Σ_{j} w_{i+j} a_j
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def _adam_step(a: np.ndarray, grad: np.ndarray,
               m: np.ndarray, v: np.ndarray, t: int,
               lr: float = 0.05, beta1: float = 0.9,
               beta2: float = 0.999, eps: float = 1e-8) -> tuple:
    """
    Perform one Adam update on vector a given its gradient.
    Returns the updated (a, m, v).
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    return a_new, m, v


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The function respects the 1000 s wall‑clock
    limit and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Determine the working length and initialise from a known construction
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
        a = base / base.sum()
        # add a tiny random jitter to escape possible stagnation
        a += (np.random.rand(n) - 0.5) * 1e-6
        a = _project_simplex(a)
    else:
        # fall back to a modest length if no previous result exists
        n = 3000
        a = np.random.rand(n)
        a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Optimisation state
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_score = float('inf')   # will be updated after first evaluation

    # Adam moment vectors
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)

    # Continuation schedule for the smoothing parameter λ
    lam_start = 0.1
    lam_end = 20.0
    log_ratio = math.log(lam_end / lam_start)

    # Number of training steps – we will stop earlier if the time budget is hit.
    max_iters = 25000

    for it in range(1, max_iters + 1):
        # ------------------------------------------------------------------
        # Time check
        # ------------------------------------------------------------------
        if time.time() >= deadline:
            print("[time] Budget exhausted, terminating optimisation.")
            break

        # ------------------------------------------------------------------
        # Update smoothing parameter λ (continuation)
        # ------------------------------------------------------------------
        progress = it / max_iters
        lam = lam_start * math.exp(log_ratio * progress)

        # ------------------------------------------------------------------
        # Compute smoothed‑max gradient
        # ------------------------------------------------------------------
        grad = _gradient_smoothed_max(a, lam)

        # ------------------------------------------------------------------
        # Adam step + projection onto the simplex
        # ------------------------------------------------------------------
        a, m_vec, v_vec = _adam_step(a, grad, m_vec, v_vec, it,
                                     lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8)
        a = _project_simplex(a)

        # ------------------------------------------------------------------
        # Periodic evaluation of the true objective
        # ------------------------------------------------------------------
        if it % 200 == 0 or it == max_iters:
            # Exact score = 2·n·max(conv) because Σ a_i = 1
            conv = np.convolve(a, a)
            cur_score = float(2.0 * n * conv.max())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[iter {it:5d}] λ={lam:6.3f} → new best score {best_score:.6f}")

        # ------------------------------------------------------------------
        # Occasional random jitter to help escape shallow plateaus
        # ------------------------------------------------------------------
        if it % 1000 == 0:
            jitter = (np.random.rand(n) - 0.5) * 5e-5
            a = _project_simplex(np.maximum(a + jitter, 0.0))

    # ------------------------------------------------------------------
    # Final validation via the official evaluator (guards against hidden
    # edge‑cases in the local score computation).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        final_score = float('inf')
        print("[warning] evaluate_sequence raised an exception on the final candidate.")

    # If for some reason the local best is invalid, fall back to a safe uniform.
    if final_score == float('inf') or np.isnan(final_score):
        print("[fallback] Using uniform sequence as a safety net.")
        a = np.full(n, 1.0 / n)
        best_seq = a
        final_score = evaluate_sequence(best_seq.tolist())

    print(f"Final best score reported by evaluate_sequence: {final_score:.6f} "
          f"(length {len(best_seq)})")
    return best_seq.tolist()
