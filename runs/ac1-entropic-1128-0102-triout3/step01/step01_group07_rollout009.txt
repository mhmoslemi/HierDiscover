```
from scipy.optimize import linprog

def solve_lp_for_g(f, M):
    n = len(f)
    m = 2*n - 1
    # Build A matrix
    A = np.zeros((m, n))
    for j in range(n):
        A[j:j+n, j] = f
    b = np.full(m, M)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    # Normalize to sum 1 (since objective maximized sum; we can keep as is)
    total = g.sum()
    if total == 0:
        return None
    return g / total
```

```
def mix_optimal_alpha(f, g):
    ff = np.convolve(f, f)
    fg = np.convolve(f, g)
    gg = np.convolve(g, g)
    # Compute coefficients
    # For each k:
    A = ff + gg - 2*fg
    B = 2*(fg - ff)
    C = ff
    # For α in [0,1], h = A*α**2 + B*α + C.
    # Sup of h across k: max_k h_k(α). convex in α.
    # Use ternary search (or golden-section).
    def sup_at(alpha):
        return np.max(A * alpha**2 + B * alpha + C)
    # Ternary search because convex.
    lo, hi = 0.0, 1.0

        else:

    alpha_opt = (lo + hi) / 2.0
    # Return convex combination
    new_f = (1 - alpha_opt) * f + alpha_opt * g
    return new_f, sup_at(alpha_opt)
```

The evaluation function  

\[
E(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is homogeneous – scaling the whole vector does not change the score – therefore we can
always normalise the candidate to a probability vector (`∑a_i = 1`).  With this
normalisation the score simplifies to  

\[
E(a)=2\,n\;\max_k (a*a)_k .
\]

The problem is to make the maximal entry of the auto‑convolution as small as possible.
A direct optimisation is hard because the constraints are quadratic.
The following strategy avoids the quadratic programme completely:

1. **Start from a good probability vector** – we reuse the best sequence that is already
   available (`height_sequence_1`) or fall back to a simple power‑law distribution.
2. **Linear programme for a partner vector** –  
   For a fixed vector `f` and a scalar `M = max(f*f)` the convolution  
   \((f*g)_k = \sum_i f_i\,g_{k-i}\) is **linear** in `g`.  
   We therefore solve the LP  

   ```
   maximise   ∑ g_i
   subject to (f*g)_k ≤ M   for all k
              g_i ≥ 0
   ```

   The LP returns a vector `g` with the largest possible ℓ₁‑norm that still respects the
   same convolution bound `M`.  After normalising `g` to sum = 1 we obtain a second
   probability vector that can be safely mixed with `f`.
3. **Optimal convex mixing** –  
   For a mixing coefficient `α∈[0,1]` the auto‑convolution of  
   `h(α) = (1‑α)·f + α·g` expands to a quadratic polynomial in `α` for each index `k`:

   \[
   h(α)*h(α)_k = A_k α^2 + B_k α + C_k,
   \]
   where `A_k = ff_k + gg_k – 2·fg_k`,
         `B_k = 2·(fg_k – ff_k)`,
         `C_k = ff_k`.

   The maximum over all `k` of these convex quadratics is itself a convex
   function of `α`.  A cheap ternary search (≈30 iterations) yields the
   optimal `α*` that minimises `max_k h(α)*h(α)_k`.  The new vector
   `f ← h(α*)` is again a probability vector.
4. **Iterate** – the whole step (LP → optimal mix) is repeated until the time
   budget expires.  Occasionally a tiny random jitter is added and a few
   projected sub‑gradient steps are performed to escape possible plateaus.
5. **Keep the best ever seen** – after each iteration we evaluate the candidate
   with `evaluate_sequence`; if the score improves we store it as the current best
   answer.

The algorithm stays completely within the allowed 1000 s, needs only a single
CPU, and uses only NumPy + SciPy (which are part of the standard execution
environment.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Propose a step‑function sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation builds on the following ideas

* normalise every candidate to a probability vector (ℓ₁‑norm = 1);
* for a fixed f, the convolution f * g is linear in g – we can therefore
  maximise the ℓ₁‑norm of g under the linear constraints (f * g)_k ≤ M
  with a linear programme (scipy.optimize.linprog);
* after obtaining a partner vector g we mix f and g with the optimal
  convex coefficient α∈[0,1] (the objective is a convex piece‑wise quadratic
  function of α – a short ternary search finds the minimum);
* the pair (LP → optimal mix) is iterated until the time budget is exhausted,
  with occasional random jitter and a handful of projected sub‑gradient steps
  to avoid getting stuck.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Linear‑programming helper (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:          # pragma: no cover
    _SCIPY_AVAILABLE = False


def _lp_optimize_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise   sum(g)
        subject to (f * g)_k ≤ M   for every k = 0 … 2·n‑2
                  g_i ≥ 0
    where the convolution (f * g) is linear in g.
    The returned vector is normalised to have ℓ₁‑norm = 1.
    Returns None if the LP fails.
    """
    n = f.shape[0]
    m = 2 * n - 1                               # number of convolution entries

    # Build dense constraint matrix A (shape m × n) where
    #   A[k, i] = f_{k‑i}  if 0 ≤ k‑i < n, else 0.
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f                        # shift f downwards

    b = np.full(m, M, dtype=np.float64)          # same bound for all rows
    bounds = [(0.0, None) for _ in range(n)]

    # minimise -sum(g)  <=>  maximise sum(g)
    c = -np.ones(n, dtype=np.float64)

    # SciPy's 'highs' method is fast and does not need additional options.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None

    g = res.x
    total = g.sum()
    if total <= 0.0:
        return None
    return g / total


# ----------------------------------------------------------------------
# Convex mixing – optimal α for a pair (f, g)
# ----------------------------------------------------------------------
def _optimal_mix(f: np.ndarray, g: np.ndarray) -> np.ndarray:
    """
    Compute the convex combination h(α) = (1‑α)·f + α·g that minimises
    max_k (h(α) * h(α))_k.
    The function is convex in α, therefore a short ternary search finds the
    optimum to machine precision.
    """
    ff = np.convolve(f, f)          # length 2n‑1
    fg = np.convolve(f, g)
    gg = np.convolve(g, g)

    # Quadratic coefficients for each k
    A = ff + gg - 2.0 * fg          # α² term
    B = 2.0 * (fg - ff)             # α   term
    C = ff                          # constant term

    # ---------------------------------------------------------------
    # ternary search on [0, 1] (convex objective)
    # ---------------------------------------------------------------
    def _sup(alpha: float) -> float:
        return np.max(A * alpha * alpha + B * alpha + C)

    lo, hi = 0.0, 1.0
    for _ in range(30):            # 30 iterations → ~1e‑9 precision
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if _sup(m1) < _sup(m2):
            hi = m2
        else:
            lo = m1
    alpha = (lo + hi) / 2.0

    h = (1.0 - alpha) * f + alpha * g
    # Numerical drift may perturb the sum – renormalise explicitly.
    h = np.maximum(h, 0.0)
    s = h.sum()
    if s == 0.0:                    # should never happen, but guard against it
        return f.copy()
    h = h / s
    # (optional) enforce a monotone decreasing shape – helps stability
    h = np.sort(h)[::-1]
    return h


# ----------------------------------------------------------------------
# Project onto probability simplex (used for tiny random jitter)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optional: a few sub‑gradient steps (helps escaping plateaus)
# ----------------------------------------------------------------------
def _subgradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the quantity max_k (seq*seq)_k.
    Afterwards the vector is projected onto the simplex and forced to be
    monotone decreasing.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))               # active index
    grad = np.zeros_like(seq)

    # The gradient of (seq*seq)_k w.r.t. seq_i is 2*seq_{k-i} when the index is valid.
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1, dtype=int)
    grad[idx] = 2.0 * seq[k - idx]

    new_seq = seq - eta * grad
    new_seq = _project_simplex(new_seq)
    new_seq = np.sort(new_seq)[::-1]
    return new_seq


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves upon the
    current best known upper bound.  The routine works inside the given
    time budget, repeatedly solving a linear programme to obtain a partner
    vector and mixing it optimally with the current candidate.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # small safety margin

    # ------------------------------------------------------------------
    # 0) initialise from the best known construction, otherwise use a power‑law
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        f = np.asarray(height_sequence_1, dtype=np.float64)
        f = np.maximum(f, 0.0)
        if f.sum() == 0.0:
            f = np.full_like(f, 1.0 / f.size)
        else:
            f = f / f.sum()
        print(f"[init] Loaded height_sequence_1 (n={len(f)})")
    else:
        # a modest power‑law with β≈0.7 works well as a generic start
        n_start = 2000
        beta = 0.7
        idx = np.arange(1, n_start + 1, dtype=np.float64)
        f = idx ** (-beta)
        f = f / f.sum()
        print(f"[init] No previous sequence – using power‑law (n={n_start}, β={beta})")

    n = f.shape[0]
    best_f = f.copy()
    best_score = float('inf')
    try:
        best_score = evaluate_sequence(best_f.tolist())
    except Exception:                           # pragma: no cover
        best_score = float('inf')
    print(f"[score] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # ----- 1) current bound M -------------------------------------------------
        ff = np.convolve(f, f)
        M = ff.max()

        # ----- 2) linear programme for a partner vector g ---------------------------
        if not _SCIPY_AVAILABLE:                 # pragma: no cover
            print("[lp] SciPy not available – aborting LP phase.")
            break

        g = _lp_optimize_g(f, M)
        if g is None:
            print("[lp] LP failed – stopping.")
            break

        # ----- 3) optimal convex mixing -------------------------------------------
        h = _optimal_mix(f, g)

        # ----- 4) (optional) tiny random jitter every few iterations ---------------
        if it % 40 == 0:
            jitter = (np.random.rand(n) - 0.5) * 1e-4
            h = h + jitter
            h = np.maximum(h, 0.0)
            h = h / h.sum()
            h = np.sort(h)[::-1]

        # ----- 5) (optional) a couple of sub‑gradient refinements -----------------
        # we keep them cheap – a single step with a diminishing step size.
        if it % 25 == 0:
            eta = 0.3 / n
            h = _subgradient_step(h, eta)

        # ----- 6) evaluate ---------------------------------------------------------
        try:
            cur_score = evaluate_sequence(h.tolist())
        except Exception:                       # pragma: no cover
            cur_score = float('inf')

        if cur_score < best_score:
            best_score = cur_score
            best_f = h.copy()
            print(f"[iter {it:04d}] New best = {best_score:.6f} (n={n})")

        # ----- 7) update current iterate -------------------------------------------
        # We keep the new vector as the starting point for the next round.
        f = h

        # safety break if we already beat the target 1.503 (nice but not required)
        if best_score < 1.503:
            print(f"[target] Desired bound reached (score={best_score:.6f}).")
            break

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_f.tolist())
    print(f"Final best score: {final_score:.6f} (length {len(best_f)})")
    return best_f.tolist()
```
