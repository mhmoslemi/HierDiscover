def _grad_max_exact(a, k_max):
    # compute gradient of conv[k_max] w.r.t. a
    # which is 2 * a shifted.
    n = a.shape[0]
    # output gradient vector
    g = np.zeros_like(a)
    # For j in [0,n-1], compute idx = k_max - j
    # If idx in [0, n-1], then g[j] = 2*a[idx]
    # So we can compute overlapping region: j from max(0, k_max - (n-1)) to min(n-1, k_max)
    j_start = max(0, k_max - (n - 1))
    j_end = min(n - 1, k_max)
    # For j in this range:
    idx = k_max - np.arange(j_start, j_end + 1)
    g[j_start:j_end + 1] = 2 * a[idx]
    return g

"""
Propose a step‑function (non‑negative sequence) that minimizes the
`evaluate_sequence` value.

Algorithm
---------
* Choose a length `n` (the previous best length is used if available).
* Initialise the sequence from several strategies:
  – uniform, – a few random Dirichlet draws, – the previous best solution.
* Optimise locally with an *Adam*‑accelerated projected gradient descent.
  The objective is the soft‑max smoothed max‑convolution:
      τ·log Σ exp(conv/τ) .
  The gradient is computed in the frequency domain (`_grad_softmax`).
* To encourage a flatter autocorrelation we occasionally perform a
  *spectral flattening* step:
    – Keep the current Fourier phases, replace all magnitudes by the
      average magnitude (constant‑magnitude projection).
    – Invert the transform and project back onto the simplex.
* The iterate is always projected onto the ℓ₁‑simplex
  `{x ≥ 0, Σ x = √(2n)}`.
* If no improvement is seen for a long time the iterate is re‑initialised
  from a random Dirichlet point.
* The best sequence seen during the whole run is returned.

The routine respects the 1000 s budget, prints progress, and only uses
NumPy (no external files or networks).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (fully self‑contained, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the first (2*len(a)-1) entries as a real array.
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of  τ·log Σ exp(conv/τ)  w.r.t. a.
    Uses the identity  grad = 2·(w * a_rev) where w_k ∝ exp(conv_k/τ).
    """
    # stabilise the soft‑max (avoid overflow)
    shifted = conv - conv.max()
    w = np.exp(shifted / tau)
    w_sum = w.sum()
    if w_sum == 0:
        # degenerate case – fall back to uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # convolution of w with the reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1: n - 1 + n]   # the part that matches a
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex
    {x ≥ 0, Σ x = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w
