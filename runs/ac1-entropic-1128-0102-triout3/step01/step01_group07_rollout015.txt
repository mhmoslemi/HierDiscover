"""
Propose a step‑function sequence that tries to lower the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The objective is scale‑invariant, therefore we keep the ℓ₁‑norm of the
vector equal to one (probability simplex).  The algorithm works in three
phases:

1. **Exploration of parametric families** – for several lengths `n`
   we generate a handful of candidates (power‑law, geometric, arcsine‑like,
   sinusoidal) and keep the best one.

2. **Smooth‑max gradient descent** – starting from the best candidate we
   minimise a smooth approximation of the `max` (log‑sum‑exp) using a
   projected gradient step.  The gradient of the smooth max can be
   written as a convolution, which we evaluate with FFTs for speed.
   The step size decays, the smoothing parameter `γ` grows, and we
   occasionally add a tiny random jitter to escape plateaus.

3. **Multiple restarts & lengths** – the whole procedure is repeated for
   several values of `n` (up to a few thousand) and a few random
   restarts, always keeping the globally best sequence seen.

All vectors are kept on the simplex by the Euclidean projection
implemented in `_project_simplex`.  The implementation respects the
1000 s overall budget and prints progress information.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, sum_i x_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1          # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Convolution a * a using real FFT (length 2·len(a)‑1).
    """
    n = a.shape[0]
    L = 2 * n - 1
    Nfft = 1 << (L - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(Fa * Fa, n=Nfft)[:L]
    return conv


def _gradient_via_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max objective.
    For the smooth max with weights w (sum w = 1) the gradient is

        g_i = 2 * Σ_k w_k * a_{k‑i} ,

    which can be computed as a convolution of the reversed `a`
    with the weight vector `w`.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = len(a_rev) + len(w) - 1          # = 3·n‑2
    Nfft = 1 << (L - 1).bit_length()
    Fa = np.fft.rfft(a_rev, n=Nfft)
    Fw = np.fft.rfft(w, n=Nfft)
    conv = np.fft.irfft(Fa * Fw, n=Nfft)[:L]
    # slice that corresponds to indices i = 0 … n‑1
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad


def _powerlaw_sequence(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-β}  (monotone decreasing)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta)
    seq /= seq.sum()
    return seq


def _geometric_sequence(n: int, r: float) -> np.ndarray:
    """a_i ∝ r^{i}  (geometric decay, 0<r<1)."""
    idx = np.arange(n, dtype=np.float64)
    seq = r ** idx
    seq /= seq.sum()
    return seq


def _arcsine_sequence(n: int) -> np.ndarray:
    """
    Approximate the continuous arcsine law:
        a_i ∝ 1 / sqrt( (i+0.5)·(n‑0.5‑i) )
    This shape puts more mass near the ends and is known to flatten
    the self‑convolution.
    """
    eps = 1e-12
    idx = np.arange(n, dtype=np.float64) + 0.5
    seq = 1.0 / np.sqrt(idx * (n - 0.5 - idx) + eps)
    seq /= seq.sum()
    return seq


def _sine_sequence(n: int, k: int = 1) -> np.ndarray:
    """
    a_i ∝ sin( π·k·(i+1) / (n+1) ), clipped to non‑negative.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = np.sin(np.pi * k * idx / (n + 1))
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= seq.sum()
    return seq


def _compute_score(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a sequence that already sums to 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_self_fft(seq)
    return float(2.0 * n * conv.max())

    """

    deadline = time.time() + budget_s - 5.0   # a tiny safety margin

    # ------------------------------------------------------------------
    lengths = [1800, 2400, 3000, 3600, 4200]   # stay comfortably below the 1000‑s limit
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Exploring length n={n} ===")

        # ----- generate a pool of seed sequences --------------------
        pool = []

        # power‑law family
        for beta in np.linspace(0.0, 4.0, 25):
            pool.append(_powerlaw_sequence(n, beta))
