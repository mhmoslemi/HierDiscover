        f_sum = f.sum()   # =1
        best_local_val = cur_best_val
        best_local_seq = f.copy()

        # iterative LP improvements
        t = 0.3
        max_iter = 30
        for it in range(max_iter):
            if time.time() > deadline:
                break
            # compute current max convolution M
            conv_f = np.convolve(f, f)
            M = conv_f.max()
            # build matrix A
            size = 2*n - 1
            A = np.zeros((size, n), dtype=np.float64)
            # fill columns
            for i in range(n):
                A[i:i+n, i] = f
            # solve LP: maximize sum(g) => minimize -sum(g)
            c = -np.ones(n, dtype=np.float64)  # minimize -sum(g)
            bounds = [(0, None)] * n

            if not res.success:
                print(f"[iter {it}] LP failed: {res.message}")
                break
            g0 = res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 1e-12:
                # no improvement possible
                break
            # scale g to have same ℓ1 norm as f (which is 1)
            g = g0 / sum_g0
            # convex combination
            f_new = (1 - t) * f + t * g
            # re‑normalise to sum 1 (to suppress numerical drift)
            f_new /= f_new.sum()
            new_val = fast_score(f_new)
            # report
            print(f"[n={n} iter {it}] val {new_val:.6f}, M {M:.6f}, sum_g0 {sum_g0:.6f}")
            # accept if improved
            if new_val < best_local_val:
                best_local_val = new_val
                best_local_seq = f_new.copy()
                # also keep improvement as current
                f = f_new
                # optionally decrease t to refine
                t = max(t * 0.9, 0.05)
            else:
                # not improved, reduce step size maybe
                t = max(t * 0.7, 0.01)
                # keep current f unchanged
        # after LP stage, optionally hill‑climb a bit
        # Use small transfers similar to previous algorithm but on final f
        current = best_local_seq.copy()
        cur_val = best_local_val
        step = 0.1
        while time.time() < deadline:

            if i == j:

            delta = step * current[i] * np.random.rand()
            if delta > current[i]:
                delta = current[i]
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            cand /= cand.sum()
            cand_val = fast_score(cand)
            if cand_val < cur_val:
                current, cur_val = cand, cand_val
                if cur_val < best_local_val:

                    best_local_seq = current.copy()
            # adapt step size
            if np.random.rand() < 0.001:

            best_seq = best_local_seq.copy()
            print(f"[global] new best value {best_val:.6f} with n={n}")
    # final output
    if best_seq is None:
        # fallback: uniform sequence
        best_seq = np.ones(1000) / 1000
    return best_seq.tolist()

"""
Hybrid search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a two‑phase optimiser:

1. **Analytic start** – use the discretised kernel a_i = 1/√(i+offset) and
   pick the best offset on a few coarse values.

2. **Iterative linear‑programming refinement** – for a fixed length n,
   keep the current sequence f (ℓ¹‑norm = 1).  Compute M = max(f*f).
   Solve the linear program

        maximise   Σ g_i
        subject to A @ g ≤ M ,   g ≥ 0

   where A is the convolution matrix (A_{k,i}=f_{k‑i} if the index exists).
   The LP finds a direction g that increases the total mass while
   respecting the current convolution peak.  After normalising g to ℓ¹‑norm 1
   we move a fraction t towards it:
        f ← (1‑t)·f + t·g .
   Repeating this step steadily reduces the peak M and therefore the
   evaluation value.

3. **Fine‑grained hill climbing** – a short random‑mass‑transfer loop
   cleans up remaining local irregularities.

The whole routine respects the 1000 s wall‑clock budget and prints
intermediate progress.
"""

import numpy as np
import time
import scipy.optimize

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, offset: float) -> np.ndarray:
    """
    Initialise a positive sequence based on a_i = 1 / sqrt(i+1+offset)
    and normalise it to ℓ¹‑norm 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.sqrt(idx)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A (shape (2n‑1, n)) that maps a vector g to
    the linear convolution (f * g).  Entry A[k,i] = f[k‑i] if 0 ≤ k‑i < n,
    otherwise 0.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_i   s.t. A @ g ≤ M , g ≥ 0,
    where A is the convolution matrix built from f.
    Returns the optimal g (may be all zeros if no improvement is possible).
    """
    A = build_convolution_matrix(f)
    n = f.size
    c = -np.ones(n, dtype=np.float64)          # minimise –Σg_i  ⇔ maximise Σg_i
    bounds = [(0, None)] * n
    rhs = np.full(A.shape[0], M, dtype=np.float64)

    # ``highs`` is fast, deterministic and works with dense matrices.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=rhs, bounds=bounds,
                                method="highs", options={"presolve": True})
    if not res.success:
        # Fall back to a zero vector – the caller will simply skip the update.
        return np.zeros_like(f)
    return res.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget (seconds).  The function stops a fraction
        of a second early to guarantee a return value.

    Returns
    -------
    list[float]
        The best sequence discovered (ℓ¹‑norm = 1, non‑negative).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a handful of candidate lengths.  Longer sequences give a
    #     finer discretisation of the 1/√t kernel, but they also make the
    #     LP larger.  The set below comfortably fits into the time budget.
    candidate_lengths = [800, 1200, 1800, 2400]

    global_best_val = np.inf
    global_best_seq = None

    for n in candidate_lengths:
        if time.time() > deadline:
            break

        # ------------------------------------------------------------------
        # 2.  Scan a small grid of offsets for the analytical shape.
        best_local_seq = None
        best_local_val = np.inf
        for offset in (0.0, 0.2, 0.5, 1.0):
            seq = build_initial_sequence(n, offset)
            val = fast_score(seq)
            if val < best_local_val:
                best_local_val = val
                best_local_seq = seq
        print(f"[n={n}] initial best value = {best_local_val:.6f}")

        f = best_local_seq.copy()                # current candidate (ℓ¹‑norm = 1)
        cur_val = best_local_val

        # ------------------------------------------------------------------
        # 3.  Iterative LP refinement.
        t = 0.35          # mixing fraction towards the LP direction
        max_lp_iters = 28
        for it in range(max_lp_iters):
            if time.time() > deadline:
                break

            # current peak of the self‑convolution
            M = np.convolve(f, f).max()

            # solve the LP: maximise total mass while staying below the peak M
            g0 = solve_lp_max_sum(f, M)
            sum_g0 = g0.sum()
            if sum_g0 <= 1e-12:
                # No beneficial direction found – stop the LP phase.
                break

            # normalise g so that it has the same ℓ¹‑norm as f (which is 1)
            g = g0 / sum_g0

            # convex combination
            f_new = (1.0 - t) * f + t * g
            f_new /= f_new.sum()                     # guard against drift

            new_val = fast_score(f_new)

            # reporting
            print(f"[n={n} LP {it:02d}] value={new_val:.6f}  "
                  f"peak={M:.6e}  sum_g0={sum_g0:.6f}")

            # keep improvements
            if new_val < cur_val:
                f, cur_val = f_new, new_val
                if new_val < best_local_val:
                    best_local_val, best_local_seq = new_val, f_new.copy()
                # slowly reduce the step as we get closer to optimum
                t = max(t * 0.92, 0.05)
            else:
                # no improvement – decrease the step size to become more cautious
                t = max(t * 0.75, 0.02)

        # ------------------------------------------------------------------
        # 4.  Fine‑grained stochastic hill‑climbing (small mass transfers).
        #     This cleans up tiny irregularities left by the LP phase.
        hill_step = 0.07
        it = 0
        while time.time() < deadline:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            delta = hill_step * f[i] * np.random.rand()
            if delta > f[i]:
                delta = f[i]
            cand = f.copy()
            cand[i] -= delta
            cand[j] += delta
            cand /= cand.sum()
            cand_val = fast_score(cand)
            if cand_val < cur_val:
                f, cur_val = cand, cand_val
                if cand_val < best_local_val:
                    best_local_val, best_local_seq = cand_val, cand.copy()
            # adapt step occasionally
            if it % 4000 == 0 and it > 0:
                hill_step = max(hill_step * 0.9, 1e-4)
            it += 1
            if it % 2000 == 0:
                print(f"[n={n} hill {it}] cur={cur_val:.6f} best={best_local_val:.6f}")

        # ------------------------------------------------------------------
        # 5.  Keep the best across all lengths.
        if best_local_val < global_best_val:
            global_best_val, global_best_seq = best_local_val, best_local_seq.copy()
            print(f"[global] new overall best = {global_best_val:.6f} (n={n})")

    # ----------------------------------------------------------------------
    # 6.  Fallback – should never happen but guards against an empty result.
    if global_best_seq is None:
        # a tiny uniform sequence (guaranteed to be valid)
        global_best_seq = np.ones(1000, dtype=np.float64) / 1000.0

    print(f"[final] best value found = {global_best_val:.6f}")
    # Return a plain Python list of floats.
    return global_best_seq.tolist()
