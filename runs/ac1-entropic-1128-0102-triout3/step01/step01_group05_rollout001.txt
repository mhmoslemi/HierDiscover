"""
Propose a low‐value step‑function for the evaluation function.

The search proceeds in three stages:

1.  Analytic start – a discretised kernel a_i = (i+δ)^(-p) is tried for a
    small grid of offsets δ and exponents p.  The best of those is kept as
    the initial sequence (ℓ¹‑norm is forced to 1).

2.  Primal‑dual improvement – the current sequence `f` defines a
    convolution peak  M = max(f∗f).  Keeping M fixed we solve the linear
    program

        maximise   Σ g_j
        subject to A(g) ≤ M·1 ,  g ≥ 0

    where A(g) = f∗g.  This is exactly the programme described in the
    statement.  The optimum `g` typically has a larger ℓ¹‑norm than `f`.
    After normalising `g` to the same ℓ¹‑norm as `f`, the two
    sequences are mixed:  f ← (1‑τ)·f + τ·g with a small mixing factor τ.
    Because  max(f∗g) ≤ M, the mixed sequence strictly reduces the
    objective.  The linear programme is solved with CVXPY (ECOS) – the
    constraint matrix is a Toeplitz‐style dense matrix built from `f`.
    The procedure is repeated until the improvement stalls or the time
    budget is exhausted.

3.  Local refinement – a fast stochastic hill‑climber (mass transfer
    between two random coordinates) runs on the best sequence found so
    far, using the same `fast_score` that the official `evaluate_sequence`
    computes.  This fine‑tunes the solution and often yields a few extra
    10⁻³ improvements.

All numbers are clipped to the allowed range by the official checker,
and the sum is kept far above 0.01, so the returned sequence is always
valid.

The implementation respects the 1000 s time limit and prints progress
information.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective 2·n·max(conv(a,a)) / (sum(a)**2)
    (identical to evaluate_sequence without the security checks).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # use a size that is a power of two for an efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
def build_kernel(n: int, offset: float, exp: float) -> np.ndarray:
    """
    Return a non‑negative sequence a_i ∝ (i+1+offset)^(-exp) and normalised
    to ℓ¹‑norm 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-exp)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   subject to f∗g ≤ M,  g ≥ 0.
    Returns the optimal vector g (ℓ¹‑norm not normalised).
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz‑like convolution matrix A so that A @ g = f∗g.
    row_idx = np.arange(rows)[:, None]          # shape (rows,1)
    col_idx = np.arange(n)[None, :]             # shape (1,n)
    idx = row_idx - col_idx                     # broadcasting → (rows,n)
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((rows, n), dtype=np.float64)
    A[mask] = f[idx[mask]]

    # Linear programme
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(rows)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True,
               abstol=1e-9, reltol=1e-9, feastol=1e-9, max_iters=5000)

    if g.value is None:
        raise RuntimeError("LP solver failed to produce a solution")
    return np.asarray(g.value, dtype=np.float64)

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------- 1.   Choose a problem size --------------------
    # A few hundred entries are enough: the objective is (approximately)
    # independent of n, while the LP stays cheap.
    n = 800

    # ------------------- 2.   Analytic initialisation -----------------
    best_seq = None
    best_val = np.inf
    # search a small grid of offsets and exponents around the classical
    # 1/sqrt(i) shape (exp=0.5).  The grid is deliberately fine – it costs
    # virtually nothing compared with the later LP steps.
    offsets = np.linspace(0.0, 3.0, 7)           # 0,0.5,…,3
    exponents = [0.40, 0.45, 0.50, 0.55, 0.60]
    for off in offsets:
        for p in exponents:
            seq = build_kernel(n, off, p)
            val = fast_score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                best_off, best_exp = off, p
    print(f"[init] best start value {best_val:.6f} (offset={best_off:.2f}, exp={best_exp:.2f})")

    # ------------------- 3.   Primal‑dual LP improvement -------------
    f = best_seq.copy()                     # current sequence (ℓ¹‑norm = 1)
    best_f = f.copy()
    iteration = 0

    while time.time() < deadline:
        # current statistics
        cur_val = fast_score(f)
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # solve the LP for g
        try:
            g_raw = solve_lp_for_g(f, M)
        except Exception as e:
            print(f"[lp] solver error after {iteration} iterations: {e}")
            break

        sum_g_raw = g_raw.sum()
        if sum_g_raw <= 0:
            print("[lp] trivial solution, aborting LP loop")
            break

        # normalise g to the same ℓ¹‑norm as f (which is 1)
        g = g_raw / sum_g_raw

        #   try a few mixing parameters τ and keep the best improvement
        best_tau_val = cur_val
        best_tau = 0.0
        for tau in (0.05, 0.10, 0.20, 0.30, 0.40):
            candidate = (1 - tau) * f + tau * g
            candidate = np.maximum(candidate, 0.0)
            candidate /= candidate.sum()
            cand_val = fast_score(candidate)
            if cand_val < best_tau_val:
                best_tau_val = cand_val
                best_tau = tau
                best_candidate = candidate.copy()

        # accept the best mixing found (if any improvement)
        if best_tau > 0.0 and best_tau_val < cur_val - 1e-8:
            f = best_candidate
            cur_val = best_tau_val
            if cur_val < best_val:
                best_val = cur_val
                best_f = f.copy()
                print(f"[LP {iteration:04d}] improved to {best_val:.6f} (τ={best_tau:.2f})")
        else:
            # no improvement – reduce step size by moving a tiny amount
            # towards g to keep the dynamics alive
            f = 0.99 * f + 0.01 * g
            f = np.maximum(f, 0.0)
            f /= f.sum()

        iteration += 1
        # stop once the LP loop has converged sufficiently
        if iteration >= 50:
            # after a reasonable number of LP steps the gain becomes tiny
            if best_tau == 0.0:
                break

        # safety break
        if time.time() >= deadline:
            break

    # ------------------- 4.   Local stochastic hill‑climbing ---------
    # Use the remaining time for a cheap mass‑transfer hill‑climber.
    remaining = deadline - time.time()
    if remaining > 5.0:
        cur = best_f.copy()
        cur_val = best_val
        step = 0.05
        hill_it = 0
        while time.time() < deadline:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            delta = step * cur[i] * np.random.rand()
            if delta > cur[i]:
                delta = cur[i]

            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            # renormalise to keep ℓ¹‑norm exactly 1 (stability)
            new_seq /= new_seq.sum()
            new_val = fast_score(new_seq)

            if new_val < cur_val:
                cur, cur_val = new_seq, new_val
                if new_val < best_val:
                    best_val, best_f = new_val, new_seq.copy()
                    print(f"[hill {hill_it:07d}] new best {best_val:.6f}")
            else:
                # tiny uphill chance to escape flat regions
                if np.random.rand() < 1e-4:
                    cur, cur_val = new_seq, new_val

            hill_it += 1
            if hill_it % 2000 == 0:
                print(f"[hill {hill_it:07d}] cur {cur_val:.6f} best {best_val:.6f}")

            # slowly decay the step size
            if hill_it % 5000 == 0:
                step = max(step * 0.9, 1e-5)

    # ------------------- 5.   Return the best sequence ----------------
    print(f"[final] best objective {best_val:.6f}")
    # Convert to plain Python list (the external grader expects this)
    return best_f.tolist()
