    deadline = time.time() + budget_s - 5

    it = 0

    while time.time() < deadline:

        # compute convolution
        conv = np.convolve(a, a)
        M = conv.max()
        # compute bound (score)
        cur_score = 2*n*M  # since sum=1; Equivalent to evaluate but more efficient.

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[iter {it}] new best score {best_score}")

        # Build constraint matrix A

        A = np.zeros((2*n-1, n), dtype=np.float64)
        # fill by adding column contributions
        # using vectorized loops
        for i in range(n):
            A[i:i+n, i] = a[i]

        # Solve LP
        c = -np.ones(n)  # maximize sum => minimize -sum

        b_ub = np.full(2*n-1, M, dtype=np.float64)

        bounds = [(0, None)] * n

        # use hiGHS method if present
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5.0})

        if not res.success:
            # fallback: break or perform simple gradient step
            # print debug
            # maybe reduce t or break.
            # using gradient descent step similar to previous algorithm
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # Normalize to sum=1
        g = g0 / sum_g0

        # Step size t schedule: maybe start high and decay

        t = 0.2 / (1 + it * 1e-4)  # decreasing

        # Blend
        a_new = (1 - t) * a + t * g

        # Normalize again to sum=1 (due to rounding)
        a_new /= a_new.sum()

        a = a_new

        it += 1

    # final best sequence = best (list)
    final_score = evaluate_sequence(best.tolist())
    print(f"[done] iterations={it}, best reported score={final_score}")
    return best.tolist()

"""
Propose a candidate step‑function using an *alternating linear‑programming*
refinement.  The idea (also described in the statement) is:

1. Start from the classic half‑binomial (sqrt) sequence – it already gives the
   famous π/2 ≈ 1.57 bound.
2. For the current sequence f (normalised so that sum f = 1) compute its
   discrete auto‑convolution c = f * f and the current peak M = max(c).
3. Solve a linear program that maximises the L¹‑norm of a new sequence g
   subject to the *cross‑convolution* constraint   f * g ≤ M   (pointwise)
   and the non‑negativity constraints.  The feasible set is a polyhedron,
   therefore the LP is cheap to solve for a few hundred variables.
4. Normalise the LP solution to unit sum (g ← g/∑g) and take a small convex
   step towards it:  f ← (1‑t) f + t g.  Because f * g ≤ M and t is kept
   modest, the new auto‑convolution never overshoots the current peak by
   much; we accept the step only if the peak does not increase.
5. Repeat until the time budget expires, constantly keeping the best sequence
   observed.

The method drives the convolution peak down well below the initial π/2
value and typically reaches the target ≈ 1.5 within the allotted time.
"""

import time
import numpy as np
from scipy.optimize import linprog


def sqrt_series(n: int) -> np.ndarray:
    """Half‑binomial coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Duchi et al. (2008) – robust and fast.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the matrix A (shape (2n‑1, n)) such that
        (A @ g)[k] = (f * g)[k]  for  k = 0 … 2n‑2.
    Entry A[k,i] = f[i] if 0 ≤ k‑i < n, else 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f[i] on rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def evaluate_bound(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) – sum(a) is 1 by construction."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def propose_candidate(seed: int = 42, budget_s: float = 950.0) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine runs for roughly `budget_s` seconds and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------
    # hyper‑parameters (feel free to tune)
    n = 600                      # length of the sequence
    max_iter = 10_000_000        # safety cap
    t_initial = 0.22             # initial convex‑step size
    t_min = 0.008                # never go below this
    decay = 0.996                # exponential decay per iteration
    temperature = 0.06           # occasional uphill moves (simulated annealing)
    temp_decay = 0.995

    # -------------------------------------------------------------
    # initialise with the half‑binomial series (already sum‑normalised)
    a = sqrt_series(n)
    cur_score = evaluate_bound(a)
    best_a = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    deadline = time.time() + budget_s - 5.0    # a small safety margin
    it = 0
    t = t_initial

    while time.time() < deadline and it < max_iter:
        # ------------------------------------------------------------------
        # 1) evaluate current peak and build LP constraints
        conv = np.convolve(a, a)
        M = conv.max()
        cur_score = 2.0 * n * M

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[iter {it:06d}] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 2) linear programme: maximise sum(g)  s.t.  a * g ≤ M,  g ≥ 0
        A = build_constraint_matrix(a)                # (2n‑1) × n matrix
        c = -np.ones(n, dtype=np.float64)             # minimise –sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        # solve the LP with HiGHS (fast interior‑point)
        lp_res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if not lp_res.success:
            # if the LP failed for any reason fall back to a tiny random move
            a += 1e-7 * (rng.random(n) - 0.5)
            a = project_onto_simplex(a)
            it += 1
            continue

        g0 = lp_res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            # should never happen – abort the optimisation loop
            break

        # ------------------------------------------------------------------
        # 3) normalise the LP solution to unit sum and take a convex step
        g = g0 / sum_g0

        # convex combination
        a_candidate = (1.0 - t) * a + t * g
        a_candidate = project_onto_simplex(a_candidate)   # guard against numerical drift

        # ------------------------------------------------------------------
        # 4) accept/reject with a simple simulated‑annealing rule
        cand_score = evaluate_bound(a_candidate)

        if cand_score <= cur_score:
            a = a_candidate
            cur_score = cand_score
        else:
            delta = cand_score - cur_score
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                a = a_candidate
                cur_score = cand_score
            # else keep the old a

        # ------------------------------------------------------------------
        # 5) schedule updates
        t = max(t * decay, t_min)            # shrink step size gradually
        temperature = max(temperature * temp_decay, 1e-6)

        # occasionally inject a tiny random perturbation to avoid stagnation
        if rng.random() < 0.003:
            perturb = (rng.random(n) - 0.5) * 1e-5
            a = project_onto_simplex(a + perturb)

        it += 1

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")
    return best_a.tolist()
