"""
Propose a low‑score step‑function.

The algorithm mixes several analytic starters (sqrt‑derivative, its
symmetric version and a sine‑shaped bump) with a fast gradient‑guided
local optimiser.  For a current vector `a` the convolution
`c = a*a` is computed, the index `k` where `c` is maximal is found,
and the sub‑gradient of the objective at this point is
`∇ = 2·rev_k(a)`.  A small step against this gradient moves mass away
from the entries that most contribute to the peak, while a simple
simulated‑annealing acceptance rule and occasional random swaps help
escape local minima.  The whole search is performed under the given
time budget; the best sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Desired L1 norm (same normalisation used in the reference code)."""
    return np.sqrt(2.0 * n)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """
    Monotone decreasing sequence:
        a_i = √(i+1+δ) – √(i+δ)
    scaled so that ∑a = sqrt(2·n).
    """
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _symmetric_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """
    Build a symmetric sequence by mirroring the analytic sqrt shape.
    The left half has length ceil(n/2); the centre element (if any) is not
    duplicated.
    """
    half = (n + 1) // 2                         # length of the left part
    left = _analytic_sqrt_sequence(half, delta)
    if n % 2 == 0:                              # even length → perfect mirror
        seq = np.concatenate([left, left[::-1]])
    else:                                       # odd length → drop the duplicated centre
        seq = np.concatenate([left, left[-2::-1]])
    seq = seq * (_target_sum(n) / seq.sum())
    return seq


def _sin_shape(n: int) -> np.ndarray:
    """
    Sinusoidal shape vanishing at the ends: a_i = sin(π·(i+1)/(n+1)).
    Normalised to the required L1 norm.
    """
    i = np.arange(1, n + 1, dtype=float)
    base = np.sin(np.pi * i / (n + 1))
    scale = _target_sum(n) / base.sum()
    return base * scale


def _partner(seq: np.ndarray, k: int) -> np.ndarray:
    """
    For a given lag `k`, return the vector p such that
        (seq * seq)[k] = sum_i seq[i] * p[i]
    i.e. p[i] = seq[k-i] when the index is inside the array, otherwise 0.
    """
    n = len(seq)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    partner = np.zeros_like(seq)
    idx = np.arange(lo, hi + 1)
    partner[idx] = seq[k - idx]
    return partner


def _score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence)."""
    n = len(seq)
    max_conv = np.max(np.convolve(seq, seq))
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_seq ** 2))


# ----------------------------------------------------------------------
# Local optimiser – gradient step + simulated annealing + random swaps
# ----------------------------------------------------------------------


def _local_optimize(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine ``seq`` while the time limit is not exceeded.
    Returns the best sequence encountered and its score.
    """
    n = len(seq)
    # keep the normalised L1 norm (helps keep numbers moderate)
    cur = seq.copy()
    cur = cur * (_target_sum(n) / cur.sum())
    cur_score = _score(cur)

    best_seq = cur.copy()
    best_score = cur_score

    # initial step size (fraction of the mean entry) and temperature
    step = max(1e-9, cur.mean() * 0.2)
    temperature = 0.2

    while time.time() < deadline:
        # 1) find the offending lag
        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))

        # 2) gradient of the maximal convolution term
        grad = 2.0 * _partner(cur, k_max)

        # 3) propose a gradient step
        cand = cur - step * grad
        cand = np.clip(cand, 0.0, 1000.0)

        # renormalise to the target L1 norm (scale invariance of the score)
        s = cand.sum()
        if s < 1e-12:                     # safety – should never happen
            cand = cur.copy()
            cand_score = cur_score
        else:
            cand = cand * (_target_sum(n) / s)
            cand_score = _score(cand)

        # 4) acceptance test (simulated annealing)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            temperature *= 0.9995           # cool down slowly
            if cur_score < best_score:
                best_seq = cur.copy()
                best_score = cur_score

        # 5) decay step size
        step *= 0.9995

        # 6) occasional random mass‑transfer move (helps escape plateaux)
        if np.random.rand() < 0.005:
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and cur[i] > 0.0:
                delta = cur[i] * 0.05          # move 5 % of the source mass
                cur[i] -= delta
                cur[j] += delta
                cur = cur * (_target_sum(n) / cur.sum())
                cur_score = _score(cur)
                if cur_score < best_score:
                    best_seq = cur.copy()
                    best_score = cur_score

        # early stop when the step becomes numerically tiny
        if step < 1e-12:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    Multiple random restarts (different lengths and analytic families)
    are performed; each restart is refined by the gradient‑guided
    optimiser.  The best sequence found before the time budget expires
    is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin to avoid cutting off the very last iteration
    deadline = start_time + max(0.0, budget_s - 5.0)

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # lengths we try – a compromise between quality and speed
    possible_lengths = list(range(400, 2501, 200))
    init_kinds = ["sqrt", "sym", "sin", "rand"]

    # ------------------------------------------------------------------
    # Outer loop – launch new restarts until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = int(rng.choice(possible_lengths))
        kind = rng.choice(init_kinds)
        delta = rng.random() * 2.0                      # used for the sqrt families

        if kind == "sqrt":
            seq = _analytic_sqrt_sequence(n, delta)
        elif kind == "sym":
            seq = _symmetric_sqrt_sequence(n, delta)
        elif kind == "sin":
            seq = _sin_shape(n)
        else:  # random Dirichlet start
            seq = rng.dirichlet(np.ones(n)) * _target_sum(n)

        init_score = _score(seq)
        if init_score < best_score:
            best_score = init_score
            best_seq = seq.copy()
            print(
                f"[init] n={n:4d} kind={kind:<4} delta={delta: .3f} score={init_score:.6f}"
            )

        # allocate a modest inner budget (≈0.2 s) for the local optimiser
        remain = deadline - time.time()
        if remain < 0.1:
            break
        inner_deadline = time.time() + min(0.2, remain * 0.5)

        seq_opt, score_opt = _local_optimize(seq, inner_deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(
                f"[opt ] n={n:4d} kind={kind:<4} delta={delta: .3f} score={score_opt:.6f}"
            )

        # optional early continuation if we already beat an interesting target
        if best_score < 1.505:
            # keep searching a little longer to try to push below the target
            pass

    # ------------------------------------------------------------------
    # Final refinement on the current best vector (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 0.5:
            best_seq, best_score = _local_optimize(
                best_seq, time.time() + remaining
            )

    # fallback – should never be needed
    if best_seq is None:
        best_seq = np.ones(10) * (_target_sum(10) / 10.0)

    # ensure all entries obey the evaluator's range restrictions
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
