    2 * n * max(conv(a, a)) / (sum(a)**2)

The method combines three ideas:

1.  Analytic start – a discrete derivative of x**p (p≈0.5) with a shift δ.
    This family contains the classic √‑derivative that already beats the
    naïve constant sequence (score ≈ π/2).

2.  Stochastic “mass‑transfer’’ hill climbing.
    A move takes a fraction of a large coefficient (source) and transfers it
    to a small coefficient (target), keeping the total L¹ mass exactly constant.
    The move is accepted if the evaluation improves, otherwise it may be
    accepted with a small probability (simulated‑annealing style) to escape
    local minima.  Step size and temperature decay slowly.

3.  Multiple restarts with different lengths, exponents, shifts and a
    fallback to the best construction found so far (including the global
    variable `height_sequence_1` if it exists).

The search is lightweight (O(n log n) per move via FFT‑based convolution) and
fits comfortably into the 1000 s budget.  The best sequence encountered before
the timeout is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Required L¹‑norm used throughout the code: √(2 n)."""
    return np.sqrt(2.0 * n)


def _analytic_power_sequence(n: int, p: float = 0.5, delta: float = 0.0) -> np.ndarray:
    """
    Return a length‑n non‑negative sequence of the form

        a_i = (i+1+δ)**p – (i+δ)**p

    scaled so that its L¹‑norm equals √(2 n).
    """
    i = np.arange(n, dtype=float)
    base = np.power(i + 1.0 + delta, p) - np.power(i + delta, p)
    # numerical safety – base is already non‑negative
    scale = _target_sum(n) / base.sum()
    return base * scale


def _max_convolution(seq: np.ndarray) -> float:
    """
    Fast (FFT‑based) computation of max(conv(seq, seq)).
    The linear convolution length is 2·len(seq)‑1; only that part is examined.
    """
    n = seq.shape[0]
    m = 1 << (2 * n - 1).bit_length()          # next power of two
    # rfft works with real input and returns N/2+1 complex numbers
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)
    # keep only the first 2n‑1 entries (the linear convolution)
    return float(np.max(conv[: 2 * n - 1]))


def _eval_fast(seq: np.ndarray) -> float:
    """
    Direct evaluation of the objective, avoiding the heavyweight
    security wrapper of `evaluate_sequence`.  The formula is

        2·n·max(conv(seq,seq)) / (sum(seq)**2)

    The function returns `np.inf` for degenerate inputs (sum too small).
    """
    n = seq.shape[0]
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")
    max_conv = _max_convolution(seq)
    return float(2.0 * n * max_conv / (s * s))


def _mass_transfer_opt(seq: np.ndarray,
                       max_iter: int,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with mass‑transfer moves.
    The total L¹ mass stays constant; each move transfers a fraction of a
    large coefficient to a small one.
    Returns the best sequence seen during the run and its score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _eval_fast(seq)

    # current working copy (start from the input)
    cur_seq = seq.copy()
    cur_score = best_score

    # step‑size (fraction of the source entry) and temperature for SA
    step_frac = 0.25
    temperature = 0.05  # a modest initial temperature

    # small epsilon to avoid division by zero
    eps = 1e-12

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # --- choose a source (large entry) ----------
        # probability proportional to the magnitude – more likely to pick large values
        prob_src = cur_seq / (cur_seq.sum() + eps)
        src = np.random.choice(n, p=prob_src)

        if cur_seq[src] <= eps:
            continue

        # --- choose a target (small entry) ----------
        # probability inversely proportional to the magnitude
        inv = 1.0 / (cur_seq + eps)
        prob_tgt = inv / inv.sum()
        tgt = np.random.choice(n, p=prob_tgt)

        if src == tgt:
            continue

        delta = step_frac * cur_seq[src]          # amount to move
        cur_seq[src] -= delta
        cur_seq[tgt] += delta

        # safety – numerical wiggle‑room
        if cur_seq[src] < 0.0:
            # revert and skip this iteration
            cur_seq[tgt] -= delta
            cur_seq[src] += delta
            continue

        cand_score = _eval_fast(cur_seq)

        if cand_score < cur_score:
            # strict improvement – keep the move
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cur_seq.copy()
        else:
            # occasional uphill acceptance (simulated annealing)
            accept_prob = np.exp(-(cand_score - cur_score) / (temperature + eps))
            if np.random.rand() < accept_prob:
                cur_score = cand_score
            else:
                # reject – revert the move
                cur_seq[tgt] -= delta
                cur_seq[src] += delta

        # decay step size and temperature slowly
        step_frac *= 0.9995
        temperature *= 0.9995
        # keep a minimal step size / temperature to retain some exploration
        step_frac = max(step_frac, 1e-4)
        temperature = max(temperature, 1e-5)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the outer evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.
    Multiple random restarts (different lengths, exponents p,
    shifts δ, and an occasional reuse of the global `height_sequence_1`)
    are performed; each restart is refined by the mass‑transfer hill‑climb.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for outer bookkeeping
    deadline = start_time + max(0.0, budget_s - 5.0)

    best_seq = None
    best_score = float("inf")

    # candidate lengths – a few hundred up to a few thousand elements
    possible_lengths = list(range(400, 2501, 100))

    # outer restart loop
    while time.time() < deadline:
        # ----------------------------------------------------------
        # Choose an initial candidate
        # ----------------------------------------------------------
        use_known = ('height_sequence_1' in globals()
                     and np.random.rand() < 0.15)   # occasional reuse
        if use_known:
            # reuse the globally provided construction
            seq0 = np.array(height_sequence_1, dtype=float)
            n = seq0.shape[0]
            # re‑normalise to the required L¹ norm
            seq0 *= _target_sum(n) / seq0.sum()
            seq = seq0
        else:
            n = int(np.random.choice(possible_lengths))

            # random exponent p around the classic 0.5 (0.35 … 0.65)
            p = np.random.uniform(0.35, 0.65)

            # shift δ – exponential distribution favours small values but
            # still explores larger offsets
            delta = np.random.exponential(scale=0.5)

            seq = _analytic_power_sequence(n, p=p, delta=delta)

            # a tiny random jitter to break symmetry
            if np.random.rand() < 0.5:
                jitter = np.random.randn(n) * 0.005 * np.mean(seq)
                seq = np.clip(seq + jitter, 0.0, None)
                # re‑normalise after jitter
                seq *= _target_sum(n) / seq.sum()

        # ----------------------------------------------------------
        # Evaluate the initial candidate
        # ----------------------------------------------------------
        init_score = _eval_fast(seq)
        if init_score < best_score:
            best_score = init_score
            best_seq = seq.copy()
            print(f"[init] n={n:4d} p={p if not use_known else '—':>6} "
                  f"δ={delta if not use_known else '—':>6} score={init_score:.6f}")

        # ----------------------------------------------------------
        # Local optimisation via mass‑transfer hill‑climb
        # ----------------------------------------------------------
        inner_deadline = time.time() + 0.3  # each restart gets ~0.3 s
        seq_opt, score_opt = _mass_transfer_opt(seq,
                                                max_iter=5000,
                                                deadline=inner_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[impr] n={n:4d} p={p if not use_known else '—':>6} "
                  f"δ={delta if not use_known else '—':>6} score={score_opt:.6f}")

        # Early exit if we already reach a very good bound
        if best_score < 1.45:   # well below the target 1.503 – very unlikely but safe
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, _target_sum(n) / n)

    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
