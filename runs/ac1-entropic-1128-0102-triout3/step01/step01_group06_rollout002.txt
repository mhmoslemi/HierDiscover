Projection onto simplex algorithm:

```
def project_onto_simplex(v):
    # v: numpy array
    # returns x with sum(x)=1 and x>=0.
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho
    rho = np.nonzero(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate
-----------------

Iterative refinement using a projected sub‑gradient method.

* Start from the best known construction (the global `height_sequence_1`
  if it exists; otherwise a quick power‑law grid search is performed).
* The objective to minimise is  
        f(a) = 2·n·max_k (a * a)[k]               (sum(a)=1, a_i≥0)
  where `*` is the ordinary (non‑circular) convolution.
* At each iteration we locate a shift `k*` that attains the maximum,
  compute the sub‑gradient g = ∇(a*a)[k*] = 2·a shifted by `k*`,
  take a step opposite to g and project back onto the probability simplex
  (non‑negative entries summing to 1).  A simple back‑tracking line‑search
  is used to guarantee monotone improvement.
* Periodically a random “hill‑climb” move (mass transfer between two
  positions) and occasional random restarts from the power‑law family
  are performed to escape shallow local minima.
* The loop runs until the allotted time (~1000 s) expires and the best
  sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the last index where u_i > cssv_i / i
    rho = np.where(u > cssv / ind)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fft_max_convolution(a: np.ndarray, fft_len: int):
    """Return (max_value, argmax_index, full convolution) of a*a."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    return max_val, max_idx, conv


def convolution_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution term (a*a)[k] w.r.t. a.
    For a_i ≥ 0 the sub‑gradient is 2·a_{k−i} on the overlapping region.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        g[idx] = 2.0 * a[k - idx]
    return g


def score_from_max_conv(n: int, max_conv: float) -> float:
    """The evaluation score when the L1‑norm of a is 1."""
    return 2.0 * n * max_conv


def build_power_candidate(n: int, alpha: float, c: float) -> np.ndarray | None:
    """a_i = (i + c)^{-α}, normalised to unit L1.  Returns None on degenerate case."""
    i = np.arange(n, dtype=np.float64)
    a = (i + c) ** (-alpha)
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    finish_time = time.time() + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # 1. Determine problem size
    # --------------------------------------------------------------
    try:
        # `height_sequence_1` is supplied by the outer harness when available
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = base_seq.size
        # normalise to unit sum (scale does not matter for the objective)
        a = base_seq.copy()
        if a.sum() == 0.0:
            raise ValueError
        a = a / a.sum()
        best_seq = a.copy()
        max_conv, _, _ = fft_max_convolution(a, 1 << ((2 * n - 1).bit_length()))
        best_score = score_from_max_conv(n, max_conv)
        print(f"[init] using provided height_sequence_1 (n={n}) → score={best_score:.6f}")
    except Exception:
        # fallback: modest length and a quick power‑law scan
        n = 3000
        fft_len = 1 << ((2 * n - 1).bit_length())

        best_seq = None
        best_score = np.inf
        # very coarse grid, just to get a sensible starting point
        alphas = np.linspace(0.35, 0.80, 9)
        cs = np.linspace(1.0, n / 2.0, 9)
        for alpha in alphas:
            for c in cs:
                if time.time() > finish_time:
                    break
                cand = build_power_candidate(n, float(alpha), float(c))
                if cand is None:
                    continue
                max_cv, _, _ = fft_max_convolution(cand, fft_len)
                sc = score_from_max_conv(n, max_cv)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[grid] α={alpha:.3f} c={c:.1f} → score={sc:.6f}")

        if best_seq is None:
            # final fallback – uniform distribution
            best_seq = np.ones(n) / n
            best_score = 2.0
            print("[fallback] uniform distribution → score=2.0")

    # --------------------------------------------------------------
    # 2. Pre‑compute constant FFT length
    # --------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # 3. Sub‑gradient optimisation loop
    # --------------------------------------------------------------
    a = best_seq.copy()
    step = 0.1                     # initial step size for the descent
    min_step = 1e-6
    no_improve_cnt = 0
    iter_cnt = 0

    while time.time() < finish_time:
        iter_cnt += 1

        # ---- evaluate current sequence ----
        max_c, k_star, _ = fft_max_convolution(a, fft_len)
        cur_score = score_from_max_conv(n, max_c)

        # ---- compute sub‑gradient at the active shift ----
        g = convolution_gradient(a, k_star)

        # ---- try a back‑tracking line‑search ----
        eta = step
        improved = False
        while eta > min_step:
            cand = a - eta * g
            # project back onto the simplex (non‑negative, sum=1)
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s <= 0.0:
                cand = np.ones_like(cand) / n
            else:
                cand = cand / s

            max_c_cand, _, _ = fft_max_convolution(cand, fft_len)
            cand_score = score_from_max_conv(n, max_c_cand)

            if cand_score < cur_score - 1e-12:      # accept only genuine improvement
                a = cand
                best_seq = cand.copy()
                best_score = cand_score
                step = eta * 1.05                 # slightly increase step after success
                print(f"[sgd] it={iter_cnt} step={eta:.6f} → score={cand_score:.6f}")
                improved = True
                no_improve_cnt = 0
                break
            else:
                eta *= 0.5                         # shrink step and retry

        if not improved:
            # no improvement – shrink the global step size a little
            step *= 0.9
            no_improve_cnt += 1

        # --------------------------------------------------------------
        # 4. occasional random perturbations (hill‑climbing style)
        # --------------------------------------------------------------
        if np.random.rand() < 0.01 or no_improve_cnt > 200:
            # random transfer of a few percent of mass between two positions
            i1, i2 = np.random.choice(n, size=2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.02  # up to ±1 %
            # enforce feasibility before applying
            if a[i1] + delta < 0.0:
                delta = -a[i1] + 1e-12
            if a[i2] - delta < 0.0:
                delta = a[i2] - 1e-12
            a[i1] += delta
            a[i2] -= delta
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            # evaluate
            max_c2, _, _ = fft_max_convolution(a, fft_len)
            score2 = score_from_max_conv(n, max_c2)
            if score2 < best_score:
                best_seq = a.copy()
                best_score = score2
                print(f"[hill] it={iter_cnt} → score={score2:.6f}")
                no_improve_cnt = 0
            else:
                # revert to best if the random move was bad
                a = best_seq.copy()
                no_improve_cnt += 1

        # --------------------------------------------------------------
        # 5. occasional restart from the power‑law family (exploration)
        # --------------------------------------------------------------
        if np.random.rand() < 0.02:
            r_alpha = np.random.uniform(0.35, 0.80)
            r_c = np.random.uniform(1.0, n / 2.0)
            cand_restart = build_power_candidate(n, r_alpha, r_c)
            if cand_restart is not None:
                max_cr, _, _ = fft_max_convolution(cand_restart, fft_len)
                score_r = score_from_max_conv(n, max_cr)
                if score_r < best_score:
                    a = cand_restart.copy()
                    best_seq = cand_restart.copy()
                    best_score = score_r
                    print(f"[restart] α={r_alpha:.3f} c={r_c:.1f} → score={score_r:.6f}")
                    step = 0.1  # reset step after a successful jump
                    no_improve_cnt = 0

    # --------------------------------------------------------------
    # 6. Return the best construction found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n) / n
    return best_seq.tolist()
