"""
Propose a candidate step‑function using a smooth‑max approximation and
accelerated projected gradient descent on the probability simplex.

The evaluation metric (with sum(a)=1) is
    score = 2 * n * max(conv(a, a))

We minimise this convex objective by iteratively:
  * computing conv(a, a) via FFT,
  * forming a smooth approximation of the max using weights
        w_k ∝ exp(alpha·conv_k)  (log‑sum‑exp trick),
  * obtaining the gradient of the smooth max as a convolution of w
    with the reversed a,
  * applying a heavy‑ball momentum step,
  * projecting back onto the simplex (Euclidean projection).

The smoothing parameter alpha is increased geometrically during the run,
so the algorithm first smooths the landscape and later focuses on the
true maximum.  Several sequence lengths are tried; the best sequence found
before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive -> uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute softmax weights for the log‑sum‑exp smooth max.
    Prevent overflow by subtracting the maximal entry.
    """
    m = np.max(conv)
    ex = np.exp(alpha * (conv - m))
    s = np.sum(ex)
    if s == 0 or not np.isfinite(s):
        # fallback to uniform weighting
        return np.full_like(conv, 1.0 / conv.size)
    return ex / s

def _gradient_wrt_a(a: np.ndarray, w: np.ndarray,
                    grad_fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    It is 2 * (w * a_rev) evaluated on the indices n‑1 … n‑1+n‑1.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    # FFTs
    A_rev_fft = np.fft.rfft(a_rev, n=grad_fft_len)
    W_fft = np.fft.rfft(w, n=grad_fft_len)
    conv = np.fft.irfft(W_fft * A_rev_fft, n=grad_fft_len)
    # slice the needed part
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad

                         best_seq: np.ndarray) -> tuple:
    """
    Run the accelerated projected gradient descent for a given length `n`.
    Returns updated (best_score, best_seq, final_a).
    """
    a = init_a.copy()
    v = np.zeros_like(a)                # momentum buffer

    # hyper‑parameters (tuned empirically)
    step0 = 0.5                         # initial step size
    beta = 0.9                          # momentum coefficient
    alpha = 1.0                         # start smoothing temperature
    alpha_max = 200.0
    alpha_mul = 2.0                     # geometric increase
    alpha_inc_iter = 3000               # increase alpha every this many iters

    # FFT lengths (powers of two for speed)
    conv_fft_len = 1 << ((2 * n - 1).bit_length())      # for a*a
    grad_fft_len = 1 << ((3 * n - 2).bit_length())      # for w * a_rev

    it = 0
    max_iter = 200_000                # generous hard cap per length

    while time.time() < deadline and it < max_iter:
        it += 1

        # ---- convolution a * a (linear) ----
        Af = np.fft.rfft(a, n=conv_fft_len)
        conv = np.fft.irfft(Af * Af, n=conv_fft_len)[: 2 * n - 1]

        max_val = np.max(conv)
        cur_score = 2.0 * n * max_val      # sum(a) == 1

        # record improvement
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"Improved: n={n}, it={it}, score={cur_score:.6f}, alpha={alpha:.1f}")

        # ---- smooth max weights ----
        w = _smooth_weights(conv, alpha)

        # ---- gradient of smooth max ----
        grad = _gradient_wrt_a(a, w, grad_fft_len)

        # ---- heavy‑ball momentum update ----
        step = step0 / math.sqrt(it)      # diminishing step
        v = beta * v - step * grad
        a_new = a + v
        a = _project_simplex(a_new)

        # ---- increase smoothing temperature periodically ----
        if it % alpha_inc_iter == 0 and alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

    return best_score, best_seq, a


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights (a step function) that attains a
    low value of `evaluate_sequence`.  The routine tries several sequence
    lengths and runs a smooth‑max accelerated projected gradient descent
    within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for the caller

    best_score = float('inf')
    best_seq = None

    # Lengths to explore – larger n gives a lower theoretical bound.
    lengths = [1200, 1600, 2000, 2400, 2800, 3200, 3600, 4000, 5000]
    np.random.shuffle(lengths)

    # If a previous construction is available, keep it for reuse.
    prev_seq = None
    try:
        prev_seq = globals().get('height_sequence_1')
        if prev_seq is not None:
            prev_seq = np.array(prev_seq, dtype=float)
    except Exception:
        prev_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- initialise a starting point -----
        if prev_seq is not None and prev_seq.size <= n:
            # embed the previous construction at the beginning and pad with zeros
            a0 = np.zeros(n, dtype=float)
            a0[:prev_seq.size] = prev_seq
            a0 /= a0.sum()
        else:
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # ----- optimise for this length -----
        best_score, best_seq, _ = _optimize_for_length(
            n, a0, start_time, deadline, best_score, best_seq
        )

    # Fallback in the unlikely event that nothing was found
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Convert to a plain Python list for the evaluator
    return [float(x) for x in best_seq]
