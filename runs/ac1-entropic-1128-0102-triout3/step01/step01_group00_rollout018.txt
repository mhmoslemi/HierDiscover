def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    best_score = float('inf')
    best_seq = None

    # list of n values
    lengths = [256, 512, 1024, 2048, 3000, 4096]   # but only include n <= max where conv length <= 8000 perhaps?

    # maybe use 1200 as earlier but also try others.

    for n in lengths:
        if time.time() >= deadline: break

        # initialization: uniform on simplex
        a = np.full(n, 1.0 / n)   # uniform; could also random etc.

        # we store z = log(a) maybe
        # to avoid zeros, start with uniform and then variable not too large.
        z = np.log(a + 1e-12)   # small offset

        # Adam parameters
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        alpha = 0.1 # initial step size

        # parameters for beta (softmax smoothing)
        beta_soft = 0.5 # initially low
        beta_max = 10.0
        beta_inc = (beta_max - beta_soft) / 200000  # gradually increase

        it = 0
        while time.time() < deadline and it < 200000:
            it += 1

            # compute a from z via softmax
            # a = np.exp(z) / np.sum(np.exp(z))
            # Use stable scaling
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            a = exp_z / np.sum(exp_z)

            # convolution auto-correlation via FFT
            # conv length = 2n -1
            fft_len = 1 << ((2*n - 1).bit_length())
            Af = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]

            # evaluate exact score for best tracking
            sum_a = np.sum(a)
            max_conv = np.max(conv)
            cur_score = 2.0 * n * max_conv / (sum_a * sum_a)   # sum_a approx 1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start:.2f}s] n={n}, iter={it}, score={cur_score:.6f}")

            # compute softmax weights
            # increase beta gradually
            beta_s = beta_soft + beta_inc * it
            if beta_s > beta_max:
                beta_s = beta_max

            # compute w = exp(beta * conv) normalized (softmax)
            # to avoid overflow, subtract max
            max_conv_val = np.max(conv)
            w = np.exp(beta_s * (conv - max_conv_val))
            w_sum = np.sum(w)
            w = w / w_sum  # size 2n-1

            # compute gradient wrt a: g_a = 4 * n * r_rev where r = conv(w, a_rev)
            # compute convolution of w with a_rev
            a_rev = a[::-1]
            # conv length = len(w) + len(a_rev) - 1 = (2n-1) + n - 1 = 3n - 2
            conv_len = 1 << ((3*n - 2).bit_length())
            Wf = np.fft.rfft(w, n=conv_len)
            Af_rev = np.fft.rfft(a_rev, n=conv_len)
            r_full = np.fft.irfft(Wf * Af_rev, n=conv_len)
            # r_full length conv_len, need slice for indices 0..(2n-2)+(n-1)-??? Actually we need r indices n-1-i for i=0..n-1
            # So we take r_slice = r_full[0: (2*n-1 + n -1) ] = length 3n-2
            r_needed = r_full[:(3*n - 2)]  # early part contains required correlations

            # gradient g_i = 4 * n * r_needed[n-1 - i]
            g_a = 4.0 * n * r_needed[(n-1) - np.arange(n)]

            # compute gradient wrt z
            dot_ag = np.dot(a, g_a)
            g_z = a * (g_a - dot_ag)

            # Adam update
            m = beta1 * m + (1 - beta1) * g_z
            v = beta2 * v + (1 - beta2) * (g_z ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            z = z - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # after while loop for this n, continue to next length

    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # return as list of floats
    return [float(x) for x in best_seq]

"""
Propose a candidate step‑function using a soft‑max (smooth max) surrogate
and Adam‑style optimization on the probability simplex.

The evaluation metric is
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

Because the score is scale‑invariant we fix sum(a)=1 and work with a
probability vector a.  The non‑smooth max is replaced by a soft‑max
with a temperature β that is slowly increased (β : 0.1 → 20).  The
gradient of the surrogate can be written in closed form:
    ∂max̂/∂a_i = 2 * Σ_k w_k a_{k-i},
where w are the soft‑max weights of the convolution entries.
Using a log‑parameterisation a_i = exp(z_i) / Σ_j exp(z_j) we obtain
unconstrained variables z and apply Adam with projection (via the
soft‑max) after each step.  The algorithm is run for several lengths
(increasing from 200 up to ~3000) and a few random restarts; the best
sequence seen (lowest true score) is returned.

The implementation respects the 1000 s budget, uses only NumPy and
standard library, and prints intermediate improvements.
"""

import time
import numpy as np

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (empirically) gives a low
    value of `evaluate_sequence`.  The routine runs for at most
    `budget_s` seconds and prints progress.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_score = float("inf")
    best_seq = None

    # lengths to explore – larger n gives a lower theoretical bound
    lengths = [200, 400, 800, 1200, 2000, 3000]
    # make them ascending so that we can warm‑start the next size
    lengths.sort()

    # optional warm‑start from a previously computed sequence
    prev_best = None
    if "height_sequence_1" in globals():
        try:
            hs = list(height_sequence_1)   # type: ignore[name-defined]
            if all(isinstance(v, (int, float)) for v in hs):
                prev_best = np.array([float(v) for v in hs], dtype=float)
        except Exception:
            prev_best = None

    # Adam hyper‑parameters (shared across all runs)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.1            # base learning rate for Adam
    beta_start, beta_end = 0.1, 20.0   # soft‑max temperature schedule

    for n in lengths:
        if time.time() >= deadline:
            break

        # number of random restarts for this length
        restarts = 3

        # pre‑compute FFT lengths (powers of two) to avoid repeated work
        fft_len_conv = 1 << ((2 * n - 1).bit_length())       # for a*a
        fft_len_grad = 1 << ((3 * n - 2).bit_length())       # for w * a_rev

        for restart in range(restarts):
            # ----- initialise candidate -----
            if prev_best is not None and restart == 0:
                # interpolate previous best to the new size
                x_old = np.linspace(0, len(prev_best) - 1, len(prev_best))
                x_new = np.linspace(0, len(prev_best) - 1, n)
                a = np.interp(x_new, x_old, prev_best)
                a = np.clip(a, 0, None)
                a += np.random.rand(n) * 1e-4  # tiny jitter
                a /= a.sum()
            else:
                a = np.random.rand(n)
                a /= a.sum()

            # log‑parameterisation (soft‑max will enforce the simplex)
            z = np.log(a + 1e-12)

            # Adam moment estimates
            m = np.zeros_like(z)
            v = np.zeros_like(z)

            it = 0
            # run until time runs out or a generous per‑size iteration cap
            max_iter = 200_000
            while time.time() < deadline and it < max_iter:
                it += 1

                # ---- probability vector from logits ----
                max_z = np.max(z)
                exp_z = np.exp(z - max_z)
                a = exp_z / np.sum(exp_z)            # sum(a) == 1

                # ---- auto‑convolution a * a (linear, not circular) ----
                Af = np.fft.rfft(a, n=fft_len_conv)
                conv = np.fft.irfft(Af * Af, n=fft_len_conv)[: 2 * n - 1]

                # ---- true score (no smoothing) ----
                cur_score = 2.0 * n * np.max(conv)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[{time.time() - start_time:6.2f}s] "
                        f"Improved: n={n}, restart={restart}, iter={it}, "
                        f"score={cur_score:.6f}"
                    )

                # ---- soft‑max weights for the surrogate max ----
                beta = beta_start + (beta_end - beta_start) * min(1.0, it / max_iter)
                c_max = np.max(conv)
                w = np.exp(beta * (conv - c_max))
                w /= np.sum(w)                       # shape (2n‑1,)

                # ---- gradient wrt a (using correlation) ----
                a_rev = a[::-1]
                Wf = np.fft.rfft(w, n=fft_len_grad)
                Af_rev = np.fft.rfft(a_rev, n=fft_len_grad)
                corr_full = np.fft.irfft(Wf * Af_rev, n=fft_len_grad)

                # slice that gives Σ_k w_k a_{k‑i} for each i
                # indices n‑1 … n‑1+n‑1 (inclusive) → length n
                corr_slice = corr_full[n - 1 : n - 1 + n]
                g_a = 4.0 * n * corr_slice          # ∂score/∂a

                # ---- gradient wrt logits z (soft‑max Jacobian) ----
                dot_ag = np.dot(a, g_a)
                g_z = a * (g_a - dot_ag)

                # ---- Adam update on logits ----
                m = beta1 * m + (1.0 - beta1) * g_z
                v = beta2 * v + (1.0 - beta2) * (g_z * g_z)

                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)

                z -= base_lr * m_hat / (np.sqrt(v_hat) + eps)

            # after a restart keep the best sequence as warm‑start for larger n
            prev_best = best_seq

    # Fallback – should never happen but guarantees a valid return value
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # convert to plain Python list (the evaluator accepts any list of floats)
    return [float(v) for v in best_seq]
```
