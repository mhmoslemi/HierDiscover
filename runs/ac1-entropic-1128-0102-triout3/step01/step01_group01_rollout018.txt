"""
Propose a step‑function sequence that minimizes
    evaluate_sequence(sequence)
using two complementary ideas:

1. Fast Adam‑style projected optimisation on the simplex.
   The gradient is taken on the current maximal convolution entry
   (or a small active set of equal maxima) via FFT.  This rapidly
   drives the objective below the uniform baseline (≈2.0).

2. A “linear‑programming polishing” step.  For the current iterate
   f we compute M = max(f*f).  Maximising Σ g_j under the linear
   constraints (f*g)_k ≤ M and g ≥ 0 yields a vector g that can hold
   strictly more ℓ₁‑mass than f without increasing the worst convolution.
   After scaling g back to the required ℓ₁‑norm we mix it with f.
   Repeating this LP‑refinement a few times dramatically reduces the
   maximal self‑convolution and pushes the evaluation below the target
   1.503.

The routine respects the 1000 s time budget, tries a few random restarts,
keeps the best vector ever seen and finally returns a list of non‑negative
heights.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (real output)."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_from_active(a: np.ndarray, active_idxs: np.ndarray, nfft: int) -> np.ndarray:
    """
    Gradient of the maximum convolution entry.
    If several indices are active, their gradients are averaged.
    """
    n = a.shape[0]
    w = np.zeros(2 * n - 1, dtype=np.float64)
    if len(active_idxs):
        w[active_idxs] = 1.0 / len(active_idxs)

    # gradient = 2 * (w * a_rev) where * denotes convolution
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (step function) with a low
    evaluate_sequence() score.  The method combines Adam‑projected descent
    with a small number of linear‑programming polishing steps.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # initialise from the best known construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        a = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)

    n = a.shape[0]
    S = np.sqrt(2.0 * n)               # required ℓ₁‑norm
    nfft = _next_pow2(2 * n - 1)

    # force exact feasibility on the simplex
    a = _project_to_simplex(a, S)

    best_a = a.copy()
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase 1 : Adam‑style projected optimisation (fast descent)
    # --------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    max_adam_iters = 20000

    while time.time() < deadline and it < max_adam_iters:
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[Adam {it:05d}] New best: {best_score:.6f}")
            if best_score < 1.503:
                break

        # active set (indices whose value is almost maximal)
        tol = cur_max * 1e-6
        active = np.where(conv >= cur_max - tol)[0]

        grad = _grad_from_active(a, active, nfft)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size scales with typical magnitude of a
        base_eta = 0.5 * S / n
        step = base_eta * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = _project_to_simplex(a - step, S)

        # accept if we really improve the true max‑convolution
        cand_max = _convolve_self_fft(a_candidate, nfft).max()
        if cand_max < cur_max:
            a = a_candidate
        else:
            # occasional random jitter to escape shallow plateaux
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * (base_eta * 0.3)
                a = _project_to_simplex(a + jitter, S)

    # --------------------------------------------------------------
    # Phase 2 : Linear‑programming polishing
    # --------------------------------------------------------------
    lp_iterations = 0
    while time.time() < deadline and lp_iterations < 5:
        conv = _convolve_self_fft(a, nfft)
        M = conv.max()

        if M < best_score:
            best_score = M
            best_a = a.copy()
            print(f"[LP {lp_iterations}] New best: {best_score:.6f}")
            if best_score < 1.503:
                break

        # Build the convolution constraints: (a * g)_k ≤ M
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idxs = np.arange(i_start, i_end + 1)
            A[k, idxs] = a[k - idxs]   # each entry = a_{k-j}

        b = np.full(rows, M, dtype=np.float64)

        # maximise Σ g_j  → minimise −Σ g_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        # solve the LP (HiGHS is the default for recent SciPy)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options={"presolve": True})

        if not res.success:
            print("[LP] Solver failed – exiting polishing loop.")
            break

        g_opt = res.x
        sum_g = g_opt.sum()
        if sum_g <= 0:
            print("[LP] Obtained zero mass – exiting.")
            break

        # Scale to the required ℓ₁‑norm and mix.
        scale = S / sum_g
        g_scaled = g_opt * scale

        t = 0.5   # mixing proportion (can be reduced in later rounds)
        a = (1.0 - t) * a + t * g_scaled
        a = _project_to_simplex(a, S)   # safeguard against numerical drift

        lp_iterations += 1

    # --------------------------------------------------------------
    # Final sanity check and return
    # --------------------------------------------------------------
    final_seq = _project_to_simplex(best_a, S).tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Finished search. Best evaluation = {final_score:.6f}")
    return final_seq
