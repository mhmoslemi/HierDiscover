def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2 # safety

    best_score = float('inf')
    best_seq = None

    # lengths to test
    lengths = [2000, 3000, 4000, 5000]  # bigger lengths if time permits
    # maybe include earlier lengths for quick improvements
    lengths += [800, 1200, 1600] etc. but shuffle.

    for n in lengths:
        if time.time() >= deadline: break

        # picks number of restarts based on n and time budget
        restarts = 3 if n <= 3000 else 2

        for restart in range(restarts):
            if time.time() >= deadline: break

            if restart == 0 and 'height_sequence_1' in globals():
                # use seeded previous best and adjust length
                seq0 = globals()['height_sequence_1']
                # may not have same length; if longer, trim; if shorter, pad with small random noise.
                if len(seq0) >= n:
                    a = np.array(seq0[:n])
                else:
                    # pad with random small values
                    a = np.concatenate([np.array(seq0), np.random.rand(n - len(seq0)) * 1e-3])
                a = np.maximum(a, 0)
                a = a / a.sum()
            else:
                a = np.random.rand(n)
                a = a / a.sum()

            # ADAM parameters
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            m = np.zeros_like(a)
            v = np.zeros_like(a)

            # parameters for smooth max
            alpha = 1.0  # initial smoothing parameter
            alpha_inc = 0.001  # increase per iteration (or adaptively)
            max_alpha = 40.0

            # Step size for Adam
            eta = 0.05

            # Precompute fft length L for this n for correlation: L = next_pow2(3n-2)
            L = 1 << ((3 * n - 2).bit_length())

            # iteration loop
            it = 0
            while time.time() < deadline and it < 200000:
                it += 1

                # compute FFT of a padded to length L
                A = np.fft.rfft(a, n=L)
                # auto-convolution
                conv_full = np.fft.irfft(A * A, n=L)
                conv = conv_full[:2 * n - 1]

                # compute current score using max conv
                cur_score = 2.0 * n * np.max(conv)
                # keep best overall
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start:6.2f}s] New best: n={n} iter={it} score={cur_score:.6f}")

                # update alpha gradually
                if alpha < max_alpha:
                    alpha = min(max_alpha, alpha + alpha_inc)

                # compute softmax weights
                # for stability subtract max
                z = alpha * conv
                z_max = np.max(z)
                exp_z = np.exp(z - z_max)
                w = exp_z / np.sum(exp_z)  # length 2n-1

                # FFT of weights
                W = np.fft.rfft(w, n=L)

                # gradient = 2 * correlation a with w
                grad_full = np.fft.irfft(A * np.conj(W), n=L)
                grad = 2.0 * grad_full[:n]   # shape (n,)

                # Adam update
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad ** 2)
                m_hat = m / (1 - beta1 ** it)
                v_hat = v / (1 - beta2 ** it)
                step = eta * m_hat / (np.sqrt(v_hat) + eps)

                # Mirror descent: multiplicative update using step as gradient?
                # We'll just do exponentiated gradient: a = a * exp(-step)
                a = a * np.exp(-step)
                # Ensure positivity and renorm
                a_sum = a.sum()
                if a_sum == 0 or np.isnan(a_sum):
                    a = np.random.rand(n)
                    a = a / a.sum()
                else:
                    a = a / a_sum

                # occasionally add small random jitter to escape local minima
                if it % 5000 == 0:
                    # perturb a small fraction
                    perturb = np.random.rand(n) * 1e-4
                    a = a + perturb
                    a = np.maximum(a, 0)
                    a = a / a.sum()

            # end while
        # end restarts
    # end lengths

    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
