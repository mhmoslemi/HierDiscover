def convolve_fft(x, y):
    L = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[:len(x)+len(y)-1]
```

def _simplex_normalize(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implemented by simple clipping and renormalisation – sufficient for
    gradient descent and far cheaper than the full Duchi et al. sort.
    """

    s = v.sum()
    if s <= 0.0:                     # degenerate case – fall back to uniform

        return np.full_like(v, 1.0 / n)
    return v / s


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using real FFT (fast for repeated calls)."""
    n = len(x) + len(y) - 1

    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:n]
    return conv

    returned by ``evaluate_sequence`` (apart from the tiny clamping that
    ``evaluate_sequence`` performs).

    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * conv.max())

    seq = idx ** (-beta)

    """a_i ∝ r**i  (geometric decay)."""
    seq = r ** np.arange(n, dtype=np.float64)

    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = (idx ** (-beta)) * (r ** np.arange(n, dtype=np.float64))

            r = np.random.uniform(0.85, 0.999)
            families.append(("both", beta, r))

        # (iv) a few purely random Dirichlet vectors
        for _ in range(10):
            rand = np.random.rand(n)
            families.append(("rand", None, None))

        # ----------------------------------------------------------------------
        # 2.  Evaluate every candidate, keep the best for this length.
        # ----------------------------------------------------------------------
        local_best_seq = None
        local_best_score = float("inf")

        for typ, beta, r in families:
            if time.time() >= deadline:
                break

            if typ == "power":
                seq = _powerlaw_sequence(n, beta)
            elif typ == "exp":
                seq = _exp_sequence(n, r)
            elif typ == "both":
                seq = _power_exp_sequence(n, beta, r)
            else:   # random Dirichlet
                seq = np.random.rand(n)
                seq = _simplex_normalize(seq)

            sc = _compute_score(seq)
            if sc < local_best_score:
                local_best_score, local_best_seq = sc, seq.copy()
                # also update global best
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(f"[grid]  n={n:4d}  type={typ:5s}  score={sc:.6f}")

        # ----------------------------------------------------------------------
        # 3.  Refine the locally best sequence with smoothed projected GD.
        # ----------------------------------------------------------------------
        if local_best_seq is None:
            continue

        seq = local_best_seq.copy()
        # hyper‑parameters (chosen empirically)
        step = 1.0 / (5.0 * n)      # initial learning rate
        alpha = 1.0                 # soft‑max temperature
        max_iter = 8000

        for it in range(max_iter):
            if time.time() >= deadline:
                break

            # --- objective and early‑stop check ---------------------------------
            conv = _conv_fft(seq, seq)
            max_conv = conv.max()
            score = float(2.0 * n * max_conv)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"[sgd]   n={n:4d}  it={it:4d}  new best {score:.6f}")

            # --- soft‑max weights -----------------------------------------------
            shifted = conv - max_conv                # for numerical stability
            # increasing α slowly concentrates the mass on the true max
            w = np.exp(alpha * shifted)
            w /= w.sum()                             # normalised weights

            # --- gradient (smooth approximation of max) -------------------------
            # grad_i = 2 * Σ_k w_k * a_{k-i}
            # which is 2 * (w ⋆ a)[i] where ⋆ denotes cross‑correlation.
            # Using FFT: cross‑correlation = conv(w, a[::-1]) with appropriate slice.
            grad_full = _conv_fft(w, seq[::-1])
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # --- projected gradient step -----------------------------------------
            seq = seq - step * grad
            seq = _simplex_normalize(seq)          # keep on the simplex

            # --- schedule updates ------------------------------------------------
            step *= 0.9997                         # gentle decay
            alpha = min(alpha * 1.015, 250.0)      # anneal temperature

            # --- occasional jitter to escape shallow plateaus --------------------
            if (it + 1) % 400 == 0:
                jitter = (np.random.rand(n) - 0.5) * step * 0.2
                seq = _simplex_normalize(np.maximum(seq + jitter, 0.0))

    # ----------------------------------------------------------------------
    # 4.  Fallback – uniform distribution (should never be needed).
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score(best_seq)

    # ----------------------------------------------------------------------
    # Final report (the scoreboard will call ``evaluate_sequence`` itself)
    # ----------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
