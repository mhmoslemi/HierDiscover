"""
Propose a candidate step‑function sequence.

The algorithm works in three stages:

1. **Exploratory families** –  For a set of lengths `n` we generate
   several parametric families (uniform, decreasing power‑law,
   edge‑concentrated power‑law, symmetric exponential and mixtures with
   the uniform distribution).  The best sequence of each length is
   stored as a starting point.

2. **Smoothed‑max gradient descent** –  The objective is
   ``obj(a) = max_k (a*a)_k`` (the ℓ∞ norm of the self‑convolution).
   As ``max`` is non‑smooth, we minimise the smooth approximation

        S_γ(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)

   whose gradient can be written as a convolution and computed by FFT.
   We start with a small smoothing factor `γ` and increase it slowly,
   while projecting onto the probability simplex (∑a_i = 1, a_i ≥ 0)
   after each step.  The sequence is kept symmetric
   (a_i = a_{n‑1‑i}) which respects the symmetry of the convolution
   and reduces the optimisation dimension.

3. **Random local jitter** –  After the descent a small amount of
   random perturbation is added and the candidate is kept if it yields
   a better objective.  Several short jitter cycles are performed to
   escape shallow plateaus.

The whole search respects a 1000 s time budget; it prints intermediate
best scores so that progress can be monitored.  The returned sequence
has non‑negative entries and is automatically normalised by the
evaluation function, therefore the absolute scale is irrelevant.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Linear‑algebra / optimisation helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1                # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a power‑of‑two FFT (real inputs)."""
    m = len(x) + len(y) - 1
    L = 1 << (m - 1).bit_length()            # next power of two
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    Z = X * Y
    z = np.fft.irfft(Z, n=L)
    return z[:m]


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (same length as np.convolve)."""
    return _fft_convolve(a, a)


def _compute_score(seq: np.ndarray) -> float:
    """
    Objective value for a sequence whose ℓ₁ ‑norm is 1:
    2·n·max( a * a ).
    """
    n = seq.shape[0]
    conv = _full_convolution(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _uniform_sequence(n: int) -> np.ndarray:
    """Uniform distribution (baseline)."""
    return np.full(n, 1.0 / n)


def _powerlaw_sequence(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing power‑law a_i ∝ (i+1)^{-β}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta)
    seq /= seq.sum()
    return seq


def _edge_powerlaw_sequence(n: int, beta: float) -> np.ndarray:
    """
    Edge‑concentrated power‑law:
    a_i ∝ ((i+1)*(n-i))^{-β}, i.e. large values near the ends,
    small values near the centre.
    """
    i = np.arange(1, n + 1, dtype=np.float64)            # i = 1 … n
    prod = i * (n + 1 - i)                               # (i)*(n+1-i)
    seq = prod ** (-beta)
    seq /= seq.sum()
    return seq


def _sym_exponential_sequence(n: int, beta: float) -> np.ndarray:
    """
    Symmetric exponential decay from the edges:
    a_i ∝ exp(-β·dist_to_edge(i)),   dist_to_edge = min(i, n-1-i).
    """
    i = np.arange(n)
    dist = np.minimum(i, n - 1 - i).astype(np.float64)
    seq = np.exp(-beta * dist)
    seq /= seq.sum()
    return seq


def _mix_sequences(base: np.ndarray, uniform: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·uniform + (1‑λ)·base."""
    seq = lam * uniform + (1.0 - lam) * base
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (symmetric variant)
# ----------------------------------------------------------------------


def _sym_smooth_max_optimize(start_seq: np.ndarray,
                             deadline: float,
                             max_iters: int = 20000) -> np.ndarray:
    """
    Perform projected gradient descent on the smooth approximation of
    max_k (a*a)_k, keeping the sequence symmetric and on the simplex.
    """
    a = start_seq.copy()
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)

    # initial hyper‑parameters
    gamma = 0.5                # smoothing factor
    eta = 0.8 / n              # learning rate
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_score = _compute_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ----- convolution and smooth max -----
        conv = _full_convolution(a)                     # length 2n‑1
        cmax = conv.max()
        # stabilise the exponentials
        shifted = conv - cmax
        exp_shift = np.exp(gamma * shifted)
        w = exp_shift / exp_shift.sum()                 # softmax weights

        # ----- gradient (2·C_k·a) averaged with softmax -----
        a_rev = a[::-1]
        conv_w = _fft_convolve(w, a_rev)                # length 3n‑2
        grad = 2.0 * conv_w[n - 1:n - 1 + n]             # slice of length n

        # ----- momentum update -----
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        # ----- gradient step + projection + symmetrisation -----
        a_new = a - eta * momentum
        a_new = _project_simplex(a_new)                 # enforce sum=1, non‑neg
        a_new = (a_new + a_new[::-1]) / 2.0             # enforce symmetry
        a_new = _project_simplex(a_new)                 # project again (tiny drift)

        # ----- evaluate -----
        cur_score = _compute_score(a_new)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[gd] iter={it:5d}  score={best_score:.6f}")

        a = a_new

        # ----- schedules -----
        eta *= 0.9995                  # slowly decay the step size
        gamma = min(gamma * 1.001, 8.0)   # increase smoothing → focus on true max

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0       # small safety margin

    # ------------------------------------------------------------------
    # Use a known construction as a starting point if available
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        try:
            seq = list(height_sequence_1)
            s = evaluate_sequence(seq)
            if s < best_score:
                best_score = s
                best_seq = seq
                print(f"[init] known height_sequence_1 score={best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) Scan parametric families for several lengths
    # ------------------------------------------------------------------
    lengths = [1200, 2000, 2800, 3500]   # moderate sizes fitting the time budget
    for n in lengths:
        if time.time() >= deadline:
            break

        uniform_seq = _uniform_sequence(n)

        # Containers for the best candidate of this length
        local_best_seq = None
        local_best_score = float("inf")

        # ----- power‑law (decreasing) -----
        betas = np.linspace(0.0, 3.0, 61)          # step 0.05
        for beta in betas:
            seq = _powerlaw_sequence(n, beta)
            sc = _compute_score(seq)
            if sc < local_best_score:
                local_best_score, local_best_seq = sc, seq

        # ----- edge‑concentrated power‑law -----
        for beta in betas:
            seq = _edge_powerlaw_sequence(n, beta)
            sc = _compute_score(seq)
            if sc < local_best_score:
                local_best_score, local_best_seq = sc, seq

        # ----- symmetric exponential (edge‑peaked) -----
        betas_exp = np.linspace(0.0, 8.0, 41)      # step 0.2
        for beta in betas_exp:
            seq = _sym_exponential_sequence(n, beta)
            sc = _compute_score(seq)
            if sc < local_best_score:
                local_best_score, local_best_seq = sc, seq

        # ----- mixtures of edge‑concentrated + uniform -----
        lambdas = [0.0, 0.25, 0.5, 0.75, 1.0]
        for beta in [0.5, 1.0, 1.5, 2.0]:
            edge = _edge_powerlaw_sequence(n, beta)
            for lam in lambdas:
                seq = _mix_sequences(edge, uniform_seq, lam)
                sc = _compute_score(seq)
                if sc < local_best_score:
                    local_best_score, local_best_seq = sc, seq

        # Update global best if a new record appears
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.tolist()
            print(f"[grid] n={n} new global best {best_score:.6f}")

        # ------------------------------------------------------------------
        # 2) Refine the locally best candidate with smoothed‑max GD
        # ------------------------------------------------------------------
        seq_ref = local_best_seq.copy()
        seq_opt = _sym_smooth_max_optimize(seq_ref, deadline, max_iters=15000)

        opt_score = _compute_score(seq_opt)
        if opt_score < best_score:
            best_score = opt_score
            best_seq = seq_opt.tolist()
            print(f"[refine] n={n} improved to {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3) Random jitter phase (few quick escapes)
        # ------------------------------------------------------------------
        jitter_iters = 3000
        a = np.array(best_seq, dtype=np.float64)
        for it in range(jitter_iters):
            if time.time() >= deadline:
                break
            # small random transfer between two random positions
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            delta = (np.random.rand() - 0.5) * 1e-4   # tiny tweak
            a_candidate = a.copy()
            a_candidate[i] = max(a_candidate[i] + delta, 0.0)
            a_candidate[j] = max(a_candidate[j] - delta, 0.0)
            a_candidate = _project_simplex(a_candidate)
            a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
            a_candidate = _project_simplex(a_candidate)

            sc = _compute_score(a_candidate)
            if sc < best_score:
                best_score = sc
                best_seq = a_candidate.tolist()
                a = a_candidate
                print(f"[jitter] iter={it:4d} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing was discovered (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        seq = np.full(n, 1.0 / n)
        best_seq = seq.tolist()
        best_score = evaluate_sequence(best_seq)
        print("[fallback] using uniform sequence")

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
