def _solve_lp_for_g(a):
    import scipy.sparse as sp
    from scipy.optimize import linprog
    n = a.shape[0]
    m = 2*n - 1
    # Build A matrix
    data = [np.full(n, a[d]) for d in range(n)]
    offsets = -np.arange(n)
    A = sp.diags(data, offsets, shape=(m, n), format='csr')
    # M column:
    Mcol = sp.csr_matrix(-np.ones((m, 1)))
    A_ub = sp.hstack([A, Mcol], format='csr')
    # equality constraint sum(g)=1
    # Build as dense row vector
    eq_row = np.concatenate([np.ones(n), np.zeros(1)])
    A_eq = sp.csr_matrix(eq_row.reshape(1, n+1))
    b_eq = np.array([1.0])

    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M

    bounds = [(0, None)] * n + [(0, None)]

    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    return res

"""
Iterative linear‑programming + convex‑combination optimizer.

The algorithm follows the classic “alternating LP” scheme:
1.  Keep a non‑negative step function `a` with ℓ¹‑norm `S = sqrt(2·n)`.
2.  Solve a linear program that, for the current `a`,
    finds a vector `g` (ℓ¹‑norm = 1) minimizing the smallest
    constant `M` such that `a * g ≤ M` (convolution is linear in `g`).
3.  Scale `g` to have ℓ¹‑norm `S` and mix it with `a`:
    `a_new = (1‑t)·a + t·g`.  The mixing coefficient `t` is chosen by a
    cheap line‑search that keeps the maximal self‑convolution as low as
    possible (the self‑convolution of the mixture is a quadratic form in
    `t`).  `t` is limited to stay safely away from values that would make
    the `g·g`‑term dominate.
4.  Project the result back onto the simplex `{x ≥ 0, Σx = S}`.
5.  Keep the best sequence seen so far and restart from a random Dirichlet
    point if no improvement is observed for many iterations.

The procedure is repeated until the allotted time (≈ 1000 s) expires or the
target bound (`1.503`) is reached.  All heavy work (FFT convolutions and
LP construction) is vectorised; the LP has `n+1` variables and `2·n‑1`
constraints and is solved with SciPy’s HiGHS interior‑point routine.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using FFT; returns the first 2·len(a)-1 values."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _convolve_fft(a: np.ndarray, b: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*b using FFT."""
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(A * B, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix `A` such that A @ g = a * g (linear convolution).
    Row i, column j contains a[i‑j] if 0 ≤ i‑j < n, otherwise 0.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # each sub‑diagonal d (0 ≤ d < n) is filled with the constant a[d]
    data = [np.full(n, a[d]) for d in range(n)]
    offsets = -np.arange(n)                # negative offsets = sub‑diagonals
    A = sp.diags(data, offsets, shape=(m, n), format='csr')
    return A


def _solve_lp_for_g(a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Solve:
        minimise   M
        subject to A @ g - M ≤ 0
                   Σ g = 1
                   g ≥ 0 , M ≥ 0
    where A implements convolution with `a`.
    Returns (g, M_opt).  If the LP fails, raises RuntimeError.
    """
    n = a.shape[0]
    m = 2 * n - 1

    A = _build_conv_matrix(a)                     # shape (m, n)
    Mcol = -sp.csr_matrix(np.ones((m, 1)))        # column for –M
    A_ub = sp.hstack([A, Mcol], format='csr')     # (m, n+1)

    # equality Σ g = 1
    eq_row = np.concatenate([np.ones(n), np.zeros(1)])
    A_eq = sp.csr_matrix(eq_row.reshape(1, n + 1))

    c = np.zeros(n + 1)
    c[-1] = 1.0                                    # minimise M

    bounds = [(0, None)] * n + [(0, None)]        # g_i ≥0 , M ≥0

    # Linprog with HiGHS, honour a per‑iteration time budget
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=np.zeros(m),
        A_eq=A_eq,
        b_eq=np.array([1.0]),
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': time_limit},
    )

    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    g = np.maximum(res.x[:n], 0.0)                 # numerical safety
    M_opt = max(res.x[-1], 0.0)
    return g, M_opt

            # jitter a little and continue
            jitter = np.random.randn(n) * (S / n) * 0.05
            a = _project_to_simplex(a + jitter, S)
            iteration += 1
            continue

        # explore a modest set of t values
        t_candidates = np.linspace(0.0, t_limit, 11)[1:]   # exclude 0

        best_local_a = a
        best_local_score = cur_score

        for t in t_candidates:
            w1 = (1.0 - t) ** 2
            w2 = 2.0 * t * (1.0 - t)
            w3 = t * t
            # convex combination of the three convolutions
            conv_mix = w1 * conv_self + w2 * conv_ag + w3 * conv_gg
            new_max = conv_mix.max()
            new_score = new_max                     # same scaling as above

            if new_score < best_local_score - 1e-12:
                best_local_score = new_score
                cand = (1.0 - t) * a + t * g
                best_local_a = _project_to_simplex(cand, S)

        # ------------------------------------------------------------------
        #  accept the best candidate (if any improvement)
        # ------------------------------------------------------------------
        if best_local_score < cur_score - 1e-12:
            a = best_local_a
        else:
            # occasional random jitter to escape plateaus
            if np.random.rand() < 0.02:
                jitter = np.random.randn(n) * (S / n) * 0.2
                a = _project_to_simplex(a + jitter, S)

        # ------------------------------------------------------------------
        #  restart if stagnation persists
        # ------------------------------------------------------------------
        if no_improve >= max_no_improve:
            alpha = np.full(n, 0.5)                     # Dirichlet concentration
            a = np.random.dirichlet(alpha) * S
            a = _project_to_simplex(a, S)
            no_improve = 0
            print(f"[{iteration:06d}] Restarted after stagnation.")

        iteration += 1

    # final projection and score check
    final_seq = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
