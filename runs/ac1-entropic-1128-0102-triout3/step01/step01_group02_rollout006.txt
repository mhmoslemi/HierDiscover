def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, cvxpy as cp

    start = time.time()
    # Set problem size
    n = 500
    # generate initial sequence as Catalan truncated
    seq = _catalan_sequence(n)
    # scale to have sum = sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    seq = np.array(seq, dtype=float)
    seq = seq * (target_sum / seq.sum())
    # ensure nonnegatives already.

    # utils
    def convolve_max(a):
        b = np.convolve(a, a)
        return b.max()

    M = convolve_max(seq)
    best_seq = seq.copy()
    best_M = M
    t = 0.2  # mixing step size
    max_iter = 200
    # precompute matrix A
    # Build convolution matrix where each column i is seq shifted by i
    # shape (2*n-1, n)
    def build_A(a):
        # a is length n
        rows = 2 * len(a) - 1
        cols = len(a)
        A = np.zeros((rows, cols))
        for i in range(cols):
            A[i:i+len(a), i] = a
        return A

    # initial A
    A = build_A(seq)  # shape (2n-1,n)
    while time.time() - start < budget_s - 1.0:
        # solve LP: maximize sum(g) s.t. A @ g <= M, g >= 0
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2*n-1)]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        # solve
        prob.solve(solver=cp.ECOS, warm_start=True)
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            print("LP not solved optimally:", prob.status)
            break
        g_opt = np.array(g.value, dtype=float)
        sum_g = g_opt.sum()
        if sum_g <= 0:
            print("Zero sum g, break")
            break
        # rescale g to match target sum sqrt(2n)
        scale = target_sum / sum_g
        g_rescaled = g_opt * scale
        # convex combination
        f_new = (1 - t) * seq + t * g_rescaled
        M_new = convolve_max(f_new)
        if M_new < M or (M_new < best_M):
            seq = f_new
            M = M_new
            A = build_A(seq)  # update A for next iteration
            if M < best_M:
                best_seq = seq.copy()
                best_M = M
                print(f"Iter improved M={M:.6f}")
        else:
            # no improvement, maybe reduce t?
            t *= 0.9
            if t < 1e-4:
                break
    # final seq: ensure nonnegative, clamp values to [0,1000]
    final_seq = [float(min(max(v, 0.0), 1000.0)) for v in best_seq.tolist()]
    print(f"Final evaluation: {evaluate_sequence(final_seq):.6f}")
    return final_seq

"""
Propose a low‑score step function using a simple convex‑relaxation /
alternating‑optimization scheme (the “LP‑mix” method).

The idea is to keep the ℓ₁‑norm of the vector fixed (∑aᵢ = √(2n)) so that
the evaluation reduces to the maximal autocorrelation value M =
maxₖ (a∗a)[k].  Starting from the Catalan (central‑binomial) vector,
which yields M = π/2 ≈ 1.571, we repeatedly:

1. Solve a linear program  
   max ‖g‖₁   s.t.   (f∗g)ₖ ≤ M  ∀k,  g ≥ 0.
   Because (f∗g)ₖ is linear in g, this is a genuine LP.
2. Rescale g so that it also has ℓ₁‑norm √(2n) and mix it with the
   current vector f (convex combination).  The mixing strictly lowers M
   unless we are already at a fixed point.
3. Occasionally perform cheap random perturbations to escape stagnation.

All operations respect the constraints enforced by ``evaluate_sequence``.
The routine runs well within the 1000 s budget for n≈500 (≈10 k LP
variables, 1 k constraints) and usually drives M below the target
1.503.  If ``cvxpy`` is unavailable the code falls back to the plain
Catalan construction.
"""

import time
import math
import random
import numpy as np

# try to import cvxpy; it may be missing on some platforms
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _catalan_sequence(n: int) -> list[float]:
    """Return the first n Catalan‑type numbers  c_i = C(2i,i)/4**i."""
    seq = [1.0]
    for i in range(1, n):
        prev = seq[-1]
        # recurrence  c_{i} = c_{i-1} * (2*i-1)/(2*i)
        seq.append(prev * (2 * i - 1) / (2 * i))
    return seq


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2*n‑1)×n matrix A such that (A @ g) == np.convolve(a, g).
    Each column i of A is the vector a shifted by i positions.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _max_convolution(a: np.ndarray) -> float:
    """Return max entry of the full self‑convolution a∗a."""
    return np.max(np.convolve(a, a))


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a candidate step‑function using the LP‑mix optimisation.
    If cvxpy is not available a plain Catalan sequence is returned.
    """

    start = time.time()
    TIME_ALLOW = budget_s - 2.0               # keep a small safety margin

    # ------------------------------------------------------------------
    # 1) Choose a moderate problem size (larger n -> finer bound but
    #    slower LP).  n≈500 gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 500

    # ------------------------------------------------------------------
    # 2) Initialise with the Catalan sequence, scaled so that
    #    sum(a) = sqrt(2*n).  With this scaling the evaluation score
    #    equals the maximal autocorrelation value M.
    # ------------------------------------------------------------------
    base = np.array(_catalan_sequence(n), dtype=float)
    target_sum = math.sqrt(2.0 * n)
    a = base * (target_sum / base.sum())          # current vector f
    M = _max_convolution(a)                       # current maximal convolution

    # keep the best sequence seen so far
    best_a = a.copy()
    best_M = M

    # ------------------------------------------------------------------
    # 3) If the optimisation infrastructure (cvxpy) is unavailable,
    #    simply return the Catalan construction.
    # ------------------------------------------------------------------
    if cp is None:
        # fall back – conform to the required limits and return
        res = [float(min(max(v, 0.0), 1000.0)) for v in a.tolist()]
        print("[fallback] cvxpy not installed – using plain Catalan")
        print(f"[fallback] evaluation = {evaluate_sequence(res):.6f}")
        return res

    # ------------------------------------------------------------------
    # 4) Main optimisation loop (LP‑mix + occasional random tweaks).
    # ------------------------------------------------------------------
    t = 0.25                     # initial mixing weight
    max_iter = 200
    iteration = 0

    # Pre‑compute the convolution matrix for the current a
    A = _build_conv_matrix(a)

    while time.time() - start < TIME_ALLOW and iteration < max_iter:
        iteration += 1

        # --------------------------------------------------------------
        # 4a) Solve the linear program:
        #     maximise   sum(g)
        #     subject to A @ g ≤ M * 1   and   g ≥ 0
        # --------------------------------------------------------------
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2 * n - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception as e:                # pragma: no cover
            print("[LP] solver error:", e)
            break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            print(f"[LP] iteration {iteration}: not optimal (status={prob.status})")
            break

        g_opt = np.asarray(g.value).reshape(-1)
        sum_g = g_opt.sum()
        if sum_g <= 1e-12:
            print("[LP] zero‑sum solution – aborting")
            break

        # --------------------------------------------------------------
        # 4b) Rescale g to the same ℓ₁‑norm as a (i.e. target_sum) and mix.
        # --------------------------------------------------------------
        scale = target_sum / sum_g
        g_scaled = g_opt * scale

        # convex combination
        a_new = (1.0 - t) * a + t * g_scaled
        M_new = _max_convolution(a_new)

        # --------------------------------------------------------------
        # 4c) Accept if we made progress; otherwise shrink the mixing step.
        # --------------------------------------------------------------
        if M_new < M - 1e-8:
            a = a_new
            M = M_new
            best_a = a.copy()
            best_M = M
            A = _build_conv_matrix(a)          # update matrix for next LP
            print(f"[iter {iteration:03d}] improved M = {M:.6f}")
            # slowly increase step size to speed up convergence
            t = min(0.9, t * 1.03)
        else:
            # no improvement – reduce mixing weight to stay closer to the
            # current point (helps to overcome possible numerical noise)
            t = max(0.01, t * 0.8)

        # --------------------------------------------------------------
        # 4d) Occasionally perform a tiny random perturbation to escape
        #     shallow local minima.
        # --------------------------------------------------------------
        if iteration % 10 == 0 and time.time() - start < TIME_ALLOW:
            pert = a.copy()
            # pick a small set of indices and move mass around
            for _ in range(3):
                i = random.randrange(n)
                delta = (random.random() - 0.5) * 0.02 * pert[i]   # ≤ 1 % change
                pert[i] = max(0.0, pert[i] + delta)
            # renormalise to keep the ℓ₁‑norm unchanged
            pert = pert * (target_sum / pert.sum())
            M_pert = _max_convolution(pert)
            if M_pert < M:
                a = pert
                M = M_pert
                best_a = a.copy()
                best_M = M
                A = _build_conv_matrix(a)
                print(f"[perturb {iteration:03d}] improved M = {M:.6f}")

    # ------------------------------------------------------------------
    # 5) Post‑processing: clamp to the limits required by evaluate_sequence.
    # ------------------------------------------------------------------
    final_seq = [float(min(max(v, 0.0), 1000.0)) for v in best_a.tolist()]

    # sanity check – report the final score (should equal best_M)
    try:
        final_score = evaluate_sequence(final_seq)
    except Exception:               # pragma: no cover
        final_score = float('nan')
    print(f"[final] length = {len(final_seq)}, evaluation = {final_score:.6f}")

    return final_seq
