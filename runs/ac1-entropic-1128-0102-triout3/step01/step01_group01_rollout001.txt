"""
Hybrid Adam‑projected‑gradient + simulated‑annealing search.

The objective that `evaluate_sequence` finally returns (for a fixed ℓ₁‑norm)
is simply `max(conv(a,a))`.  We therefore minimise the peak value of the
self‑convolution while keeping the vector non‑negative and its ℓ₁‑norm
constant.  The algorithm combines

* an Adam‑style projected gradient descent on a smooth log‑sum‑exp
  approximation of the max,
* an exponential temperature schedule that drives the approximation to a
  hard max,
* occasional random “jitter’’ moves and full restarts when progress stalls,
* a final cheap hill‑climbing phase.

All operations are performed with FFT‑based convolutions (O(n log n))
and a Euclidean projection onto the probability simplex, which guarantees
the ℓ₁‑constraint without ever leaving the feasible region.

The routine respects the 1000 s budget, prints intermediate best scores,
and finally returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real‑valued). Returns the full
    length‑(2*len(a)-1) array."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries forced to zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Stable soft‑max weights w ∝ exp(conv/τ).  Returns a probability vector."""
    # For numerical stability subtract the maximum first.
    maxc = conv.max()
    exp_arg = (conv - maxc) / tau
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0:
        # fall back to a hard‑max (single 1 at the max index)
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
    else:
        w /= w_sum
    return w


def _grad_from_weights(a: np.ndarray, w: np.ndarray, nfft: int) -> np.ndarray:
    """Gradient of Σ w_k·conv_k (i.e. of τ·log‑sum‑exp) with respect to a."""
    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a small maximum
    self‑convolution value.

    The routine runs for up to ``budget_s`` seconds and returns the
    best sequence (as a Python list) discovered.  It starts from the
    globally provided ``height_sequence_1`` when available, otherwise
    from a uniform distribution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise the design vector a
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        n = a.size
        # enforce non‑negativity and clip to the evaluator's hard limits
        a = np.clip(a, 0.0, 1000.0)
        target_sum = a.sum()
        if target_sum <= 0.01:                     # sanity fallback
            a = np.full(n, 1.0)
            target_sum = a.sum()
    else:
        # a reasonable fallback: uniform over 2000 entries
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        a = np.full(n, target_sum / n, dtype=np.float64)

    # Ensure we are exactly on the simplex (important for the scaling‑invariance)
    a = _project_to_simplex(a, target_sum)

    # 2. Pre‑compute FFT length
    # ------------------------------------------------------------------
    nfft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3. Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # base learning‑rate – roughly a modest fraction of the typical entry size
    base_alpha = 0.5 * target_sum / n

    # ------------------------------------------------------------------
    # 4. Temperature schedule for the log‑sum‑exp smoothing
    # ------------------------------------------------------------------
    tau = 1.0          # start fairly smooth
    tau_min = 1e-5
    # we aim for ~500 k outer iterations (adjusted dynamically by time)
    max_est_iters = 500_000
    tau_decay = (tau_min / tau) ** (1.0 / max_est_iters)

    # ------------------------------------------------------------------
    # 5. Book‑keeping
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    iteration = 0
    no_improve = 0
    stagnation_limit = 15_000          # restart after this many non‑improving steps
    jitter_prob = 0.01                 # probability of performing a random jitter

    # ------------------------------------------------------------------
    # 6. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # a) compute current convolution and objective value
        # --------------------------------------------------------------
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()                 # equal to evaluate_sequence when sum fixed
        cur_score = cur_max

        # --------------------------------------------------------------
        # b) keep the best ever seen

        if cur_score < best_score - 1e-12:

            best_seq = a.copy()

        else:
            no_improve += 1

        # early termination if we already beat the target bound
        if best_score < 1.503:
            print(f"[{iteration:07d}] Target achieved (score={best_score:.6f})")
            break

        # --------------------------------------------------------------
        # c) compute gradient of the smoothed max (log‑sum‑exp)
        # --------------------------------------------------------------
        w = _softmax_weights(conv, tau)
        grad = _grad_from_weights(a, w, nfft)

        # --------------------------------------------------------------
        # d) Adam update + projection onto the ℓ₁‑simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        step = base_alpha * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step                     # descent step
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        # --------------------------------------------------------------
        # e) Accept if it truly improves the hard max, otherwise keep a.
        #    Occasionally apply a random jitter to escape plateaus.
        # --------------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()

        if cand_max < cur_max:
            a = a_candidate
        else:
            if np.random.rand() < jitter_prob:
                # move a tiny amount from a random donor to a random receiver
                i, j = np.random.choice(n, size=2, replace=False)
                if a[i] > 0:
                    epsilon = 0.1 * a[i]          # up to 10 % of the donor mass
                    a_jitter = a.copy()
                    a_jitter[i] -= epsilon
                    a_jitter[j] += epsilon
                    a_jitter = _project_to_simplex(a_jitter, target_sum)
                    a = a_jitter

        # --------------------------------------------------------------
        # f) temperature (soft‑max smoothing) annealing
        # --------------------------------------------------------------
        tau = max(tau * tau_decay, tau_min)

        # --------------------------------------------------------------
        # g) restart if we have been stagnant for a while
        # --------------------------------------------------------------
        if no_improve >= stagnation_limit:
            # draw a fresh point on the simplex (Dirichlet)
            alpha = np.full(n, 0.5)                 # moderate concentration
            a = np.random.dirichlet(alpha) * target_sum
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{iteration:07d}] Restarted (stagnation).")

        iteration += 1

    # ------------------------------------------------------------------
    # 7. Optional fine‑grained hill climbing on the best candidate
    # ------------------------------------------------------------------
    a = best_seq.copy()
    while time.time() < deadline:
        # propose a tiny local move (donor→receiver) and accept if it improves
        i, j = np.random.choice(n, size=2, replace=False)
        if a[i] <= 1e-12:
            continue
        eps = 0.02 * a[i]                     # 2 % of donor mass
        a_tmp = a.copy()
        a_tmp[i] -= eps
        a_tmp[j] += eps
        a_tmp = _project_to_simplex(a_tmp, target_sum)

        conv_tmp = _convolve_self_fft(a_tmp, nfft)
        if conv_tmp.max() < best_score:
            best_score = conv_tmp.max()
            best_seq = a_tmp.copy()
            a = a_tmp
            print(f"[refine] Improved to {best_score:.6f}")
        else:
            # occasional random perturbation to keep the search alive
            if np.random.rand() < 0.001:
                a = a_tmp

    # ------------------------------------------------------------------
    # 8. Final verification & return
    # ------------------------------------------------------------------
    final_seq = _project_to_simplex(best_seq, target_sum)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
