def simplex_projection(v):
    # sorts descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    def conv_full(a):
        A = np.fft.rfft(a, n=fft_len)
        c = np.fft.irfft(A * A, n=fft_len)[: 2*n - 1]

    def compute_score(a):
        s = a.sum()
        if s < 0.01:
            return np.inf
        c = conv_full(a)
        max_c = c.max()
        return float(2.0 * n * max_c / (s*s))

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            return a
        return a / s

    # Initial candidate
    if init_seq is not None:
        a = normalize(init_seq.copy())
    else:
        i = np.arange(n, dtype=float)

2. **Projected gradient descent on the max‑convolution** – repeatedly
   find the lag `k_max` where the self‑convolution `a*a` attains its
   maximum, compute the sub‑gradient of that single term,
   take a small step opposite to the gradient, clip negatives and
   re‑normalise to the probability simplex.  The step size decays
   linearly with the remaining time budget.

3. **Polishing phase** – occasional random mass transfers (the
   “hill‑climbing” idea from the previous solution) and occasional
   random restarts from the power‑law family help escape shallow local
   minima.

All operations respect the constraints of `evaluate_sequence`
(non‑negative entries, implicit clipping, sum > 0).  The routine runs
for at most ``budget_s`` seconds and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------
def _full_convolution(seq: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution of `seq` with itself using an FFT of length `fft_len`."""
    A = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(seq) - 1]
    return conv


def _objective(seq: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute the exact value returned by ``evaluate_sequence`` for a
    non‑negative `seq`.  The sequence is *not* assumed to be normalised;
    the formula is 2·n·max(conv) / (sum(seq)²).
    """
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = _full_convolution(seq, fft_len)
    max_conv = conv.max()
    return 2.0 * n * max_conv / (s * s)


def _normalise(seq: np.ndarray) -> np.ndarray:
    """Clip to non‑negative, then scale to sum = 1 (probability simplex)."""
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        return seq
    return seq / total


def _gradient_max_lag(seq: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the term (a*a)[k_max] with respect to `seq`.
    For indices i where j = k_max‑i lies inside the array,
    the gradient component is 2·seq[j]; otherwise 0.
    """
    n = len(seq)
    grad = np.zeros_like(seq)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    j = k_max - idx
    grad[idx] = 2.0 * seq[j]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Produce a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence`.  The routine respects a 1000 s time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    finish_time = start_time + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # 1. Determine problem size and initialise a candidate
    # --------------------------------------------------------------
    try:
        # reuse the best known construction if it is available
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
    except Exception:
        # fallback length – big enough to give a good asymptotic bound
        n = 2000
        init = None

    # FFT length – next power of two that can hold a full convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    if init is not None:
        a = _normalise(init.copy())
    else:
        # simple power‑law start, normalised to unit sum
        i = np.arange(n, dtype=np.float64)
        a = _normalise((i + 1.0) ** (-0.5))

    best_a = a.copy()
    best_score = _objective(best_a, n, fft_len)
    print(f"[init] length={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Quick random exploration of the power‑law family
    # --------------------------------------------------------------
    phase1_end = start_time + min(12.0, budget_s * 0.07)   # ~7 % of total time
    while time.time() < phase1_end:
        alpha = np.random.uniform(0.30, 0.80)
        c = np.random.uniform(1.0, max(1.0, n / 2.0))
        i = np.arange(n, dtype=np.float64)
        cand = _normalise((i + c) ** (-alpha))
        sc = _objective(cand, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand.copy()
            print(f"[rand] α={alpha:.4f} c={c:.1f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 3. Projected gradient descent on the active (maximum) lag
    # --------------------------------------------------------------
    step0 = 0.5          # initial step multiplier
    min_step = 1e-6
    it = 0

    while time.time() < finish_time:
        # current convolution and the lag with the largest entry
        conv_cur = _full_convolution(a, fft_len)
        max_val = conv_cur.max()
        k_max = int(conv_cur.argmax())
        cur_score = 2.0 * n * max_val          # sum(a)==1, so denominator =1

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[grad] it={it:6d} score={cur_score:.6f}")

        # sub‑gradient of the active term
        g = _gradient_max_lag(a, k_max)

        # linearly decaying step size (remaining‑time proportion)
        remaining = finish_time - time.time()
        proportion = max(0.0, remaining) / (budget_s - 2.0)
        step = max(min_step, step0 * proportion)

        # projected gradient step
        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() == 0.0:
            # degenerate step – shrink and continue
            step *= 0.5
            continue
        a_new = _normalise(a_new)

        new_score = _objective(a_new, n, fft_len)

        # simple acceptance rule (always accept improvements,
        # occasionally accept worse moves to escape shallow traps)
        if new_score < best_score:
            a = a_new
        else:
            # simulated‑annealing‑style acceptance with a tiny temperature
            temperature = max(1e-4, proportion)
            prob = np.exp(-(new_score - best_score) / (temperature * 0.01))
            if np.random.rand() < prob:
                a = a_new

        # ----------------------------------------------------------
        # 4. Light‑weight random hill‑climbing steps for extra diversity
        # ----------------------------------------------------------
        if np.random.rand() < 0.015:          # about one in ~70 iterations
            i1, i2 = np.random.choice(n, size=2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.02   # up to ±1 % of a donor
            # enforce non‑negativity
            if a[i1] + delta < 0.0:
                delta = -a[i1] + 1e-12
            if a[i2] - delta < 0.0:
                delta = a[i2] - 1e-12
            a[i1] += delta
            a[i2] -= delta
            a = _normalise(a)

        # ----------------------------------------------------------
        # 5. Occasional restart from the parametric family
        # ----------------------------------------------------------
        if np.random.rand() < 0.008:
            r_alpha = np.random.uniform(0.35, 0.75)
            r_c = np.random.uniform(1.0, n / 2.0)
            i = np.arange(n, dtype=np.float64)
            cand = _normalise((i + r_c) ** (-r_alpha))
            cand_score = _objective(cand, n, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand.copy()
                print(f"[restart] α={r_alpha:.4f} c={r_c:.1f} → score={cand_score:.6f}")

        it += 1

    # --------------------------------------------------------------
    # 6. Final polishing with a few pure hill‑climbing passes
    # --------------------------------------------------------------
    while time.time() < finish_time:
        i1, i2 = np.random.choice(n, size=2, replace=False)
        delta = (np.random.rand() - 0.5) * 0.015
        if best_a[i1] + delta < 0.0:
            delta = -best_a[i1] + 1e-12
        if best_a[i2] - delta < 0.0:
            delta = best_a[i2] - 1e-12
        best_a[i1] += delta
        best_a[i2] -= delta
        best_a = _normalise(best_a)

        # quick check – if this tiny move improves we keep it
        sc = _objective(best_a, n, fft_len)
        if sc < best_score:
            best_score = sc
            print(f"[final hill] score={sc:.6f}")

    print(f"[done] best score={best_score:.6f} after {it} gradient iterations")
    return best_a.tolist()
