def build_Q_matrices(n):
    Q_list = []
    for k in range(2*n-1):
        rows = []
        cols = []
        data = []
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            if i > j: continue   # handle symmetric later
            if i == j:
                rows.append(i)
                cols.append(i)
                data.append(1.0)
            else:
                rows.append(i)
                cols.append(j)
                data.append(0.5)
                rows.append(j)
                cols.append(i)
                data.append(0.5)
        Q = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(n, n)).tocsc()
        Q_list.append(Q)
    return Q_list
```

process reduces the evaluation E = 2 n·max(f∗f)/ (∑f)².

def _catalan_sequence(n):
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i-1] * (2*i-1) / (2*i)
    return seq

def _build_conv_matrix(f):
    """Return matrix A with A[k, j] = f[k-j] if 0≤k-j < len(f) else 0.
    Shape (2n-1, n)."""
    n = len(f)
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _solve_lp_max_sum(f, M):
    """Solve linear program maximize sum(b) s.t. b≥0 and A b ≤ M,
    where A is the convolution matrix of f.
    Returns b (numpy array). Raises RuntimeError on failure."""
    if linprog is None:
        raise RuntimeError("scipy.linprog not available")

"""
Iterative linear‑programming improvement of a step‑function.

The functional to minimise is  

    E(a) = 2·n·max_k (a∗a)[k] / (∑ a_i)²

for a non‑negative vector `a` of length `n`.  The construction starts from the
Catalan (central‑binomial) coefficients, normalises the ℓ¹‑norm to 1 and then
repeatedly solves a linear programme

    maximise   ∑ b_j
    subject to   b_j ≥ 0
                 (f ∗ b)[k] ≤ M   for all k

where `f` is the current vector and `M = max (f ∗ f)`.  The optimiser finds a
vector `b` that has a larger ℓ¹‑norm while preserving the same convolution
bound.  After rescaling `b` to unit ℓ¹‑norm we form several convex mixtures of
`f` and the scaled vector and keep the mixture with the smallest evaluation
score.  This procedure strictly reduces the maximal self‑convolution and
converges quickly to a sequence whose evaluation is well below the Catalan
value (~1.571).  A short random local‑search polishing phase is run with any
remaining budget.

The implementation uses only `numpy` and `scipy.optimize.linprog`.  If SciPy is
unavailable the routine falls back to the plain Catalan construction.
"""

import time
import numpy as np

# SciPy is used for the linear programme; fall back if it cannot be imported.
try:
    from scipy.optimize import linprog
except Exception:   # pragma: no cover
    linprog = None   # type: ignore


def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients c_i = C(2i,i)/4**i."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A that represents the linear map
    b ↦ f ∗ b.
    Row k, column j contains f[k‑j] when 0 ≤ k‑j < n, otherwise 0.
    """
    n = f.shape[0]
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    for j in range(n):
        # rows j … j+n‑1 receive a copy of f
        A[j:j + n, j] = f
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear programme

        maximise   ∑ b_j
        subject to b ≥ 0  and  (f ∗ b)[k] ≤ M  for all k.

    Returns the optimiser `b` as a dense NumPy array.
    Raises RuntimeError on failure.
    """
    if linprog is None:                     # pragma: no cover
        raise RuntimeError("scipy.linprog is not available")

    A = _build_conv_matrix(f)
    K = A.shape[0]
    b_ub = np.full(K, M, dtype=np.float64)   # RHS of the inequality constraints
    c = -np.ones(f.shape[0], dtype=np.float64)   # minimise –∑b  ⇔ maximise ∑b
    bounds = [(0, None)] * f.shape[0]

    # ``highs`` is a fast, robust LP solver shipped with SciPy ≥1.6.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    # Numerical noise may produce tiny negative entries – clip them.
    return np.maximum(0.0, res.x)


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a step‑function with a low evaluation score.

    The algorithm proceeds in three stages:
      1. Initialise with a Catalan sequence of modest length (≤ 800) and
         normalise the ℓ¹‑norm to 1.
      2. Run a small number of LP‑based improvement iterations – each iteration
         solves a linear programme, rescales the optimiser, forms a handful of
         convex mixtures, and keeps the mixture with the smallest evaluation.
      3. If time permits, perform a lightweight random local‑search to polish
         the result.

    The final list is clipped to the interval [0, 1000] as required by
    `evaluate_sequence`.
    """
    start = time.time()

    # ----------------------------------------------------------------------
    # Choose a length that remains tractable for the LP solver.
    # ----------------------------------------------------------------------
    try:   # type: ignore[name-defined]
        n = len(height_sequence_1)          # use the length of the previous best
        n = min(n, 800)                    # but cap it for speed
    except Exception:                       # pragma: no cover
        n = 600  # default when no previous sequence is available

    # ----------------------------------------------------------------------
    # Initialise with the Catalan construction and renormalise.
    # ----------------------------------------------------------------------
    f = _catalan_sequence(n).astype(np.float64)
    f /= np.sum(f)                # now ∑ f_i = 1
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[propose_candidate] start length={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Main improvement loop (LP + mixing).
    # ----------------------------------------------------------------------
    max_iters = 30
    iteration = 0
    while iteration < max_iters and (time.time() - start) < budget_s - 5:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        # Solve the LP: maximise ℓ¹‑norm under the current convolution bound.
        try:
            b_opt = _solve_lp_max_sum(f, M)
        except Exception as e:
            print(f"[iteration {iteration}] LP failed: {e}")
            break

        sum_b = float(np.sum(b_opt))
        if sum_b <= 1.0 + 1e-12:
            print(f"[iteration {iteration}] no ℓ¹‑norm increase (sum_b={sum_b:.6f})")
            break

        # Rescale to keep ℓ¹‑norm equal to 1.
        scale = 1.0 / sum_b
        g = b_opt * scale

        # Form a small pool of candidates (pure g and several mixtures).
        candidates = [f, g]
        for t in (0.2, 0.4, 0.6, 0.8):
            candidates.append((1.0 - t) * f + t * g)

        # Pick the candidate with the lowest evaluation score.
        best_local = f
        best_local_score = best_score
        for cand in candidates:
            score = evaluate_sequence(cand.tolist())
            if score < best_local_score:
                best_local = cand
                best_local_score = score

        # Update the global best if we have improved.
        if best_local_score < best_score - 1e-12:
            best_f = best_local.copy()
            best_score = best_local_score
            print(f"[iteration {iteration}] improved score to {best_score:.6f}")
        else:
            print(f"[iteration {iteration}] no improvement (best local {best_local_score:.6f})")

        # Prepare for the next iteration.
        f = best_local
        iteration += 1

        # Early stop if we are already very close to the theoretical lower bound.
        theoretic_min = 2.0 * n / (2 * n - 1)   # corresponds to max ≈ 1/(2n‑1)
        if best_score <= theoretic_min * 1.02:
            print("[propose_candidate] near theoretical lower bound – stopping early")
            break

    # ----------------------------------------------------------------------
    # Light random local search (uses any remaining time budget).
    # ----------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 10.0:
        rng = np.random.default_rng(seed)
        n = best_f.shape[0]
        cur_seq = best_f.copy()
        cur_score = best_score
        max_local_iters = 20000
        for i in range(max_local_iters):
            if (time.time() - start) > budget_s - 2.0:
                break
            i1, i2 = rng.integers(0, n, size=2)
            if i1 == i2:
                continue
            # Transfer a small random fraction from i1 to i2.
            delta = cur_seq[i1] * rng.random() * 0.05
            if delta <= 0:
                continue
            new_seq = cur_seq.copy()
            new_seq[i1] -= delta
            new_seq[i2] += delta
            if new_seq[i1] < -1e-12:          # safeguard against rounding artefacts
                continue
            score = evaluate_sequence(new_seq.tolist())
            if score < cur_score:
                cur_seq = new_seq
                cur_score = score
                if score < best_score:
                    best_f = new_seq
                    best_score = score
                if i % 5000 == 0:
                    print(f"[local search] iter {i}, new best score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check and return.
    # ----------------------------------------------------------------------
    final_seq = [float(min(max(v, 0.0), 1000.0)) for v in best_f.tolist()]
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[propose_candidate] final score = {final_eval:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[propose_candidate] final evaluation failed:", exc)

    return final_seq
