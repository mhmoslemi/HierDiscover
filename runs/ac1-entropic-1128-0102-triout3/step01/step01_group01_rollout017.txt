"""
Propose a new step‐function sequence.

The implementation combines two complementary ideas:

1. **Projected sub‑gradient descent** on the exact objective
   f(a)=max(a*a).  The gradient of the max‑convolution is obtained
   analytically (no soft‑max smoothing).  The iterate is projected
   onto the ℓ₁–simplex of radius S=√(2n) to keep the evaluation
   function equal to the maximal convolution value.

2. **A global LP improvement step**.  For a given sequence a with
   current bound M = max(a*a) we solve the linear program

        maximise  Σ g_j
        subject   T(a)·g ≤ M ,   g ≥ 0

   where T(a) is the Toeplitz matrix that implements the linear
   convolution a*g.  The optimal g has a larger ℓ₁‑norm than a;
   after scaling it back to the required norm S we obtain a new
   candidate that usually lowers the maximal convolution dramatically.
   The LP is solved with SciPy's HiGHS interior‑point solver using a
   sparse representation of T(a).  If the LP does not improve the
   bound we fall back to the sub‑gradient step.

The algorithm runs for at most ``budget_s`` seconds (default 1000 s)
and returns the best feasible sequence found.  It starts from the
global variable ``height_sequence_1`` when present, otherwise from a
random Dirichlet point.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT (first 2*len(a)-1 entries)."""
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_toeplitz(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix T that implements the linear map
    g ↦ a * g (convolution).  Row k, column j contains a_{k-j} if the
    index is in range, otherwise 0.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # each diagonal -d (d=0…n‑1) consists of constant a[d]
    data = [np.full(n, a[d]) for d in range(n)]
    offsets = [-d for d in range(n)]
    T = sp.diags(data, offsets, shape=(m, n), format="csr")
    return T


def _lp_max_sum(a: np.ndarray, M: float, S: float):
    """
    Solve   max Σ g_j   s.t.   T(a)·g ≤ M ,   g ≥ 0.
    Returns (g_opt, sum_opt).  If the LP fails, returns (None, None).
    """
    n = a.shape[0]
    T = _build_toeplitz(a)

    c = -np.ones(n)                     # maximise sum(g)  -> minimise -sum(g)
    b_ub = np.full(2 * n - 1, M)        # T·g ≤ M
    bounds = [(0, None)] * n

    # HiGHS handles sparse matrices efficiently
    try:
        res = scipy.optimize.linprog(
            c,
            A_ub=T,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 5.0},
        )
    except Exception:  # pragma: no cover
        return None, None

    if not res.success:
        return None, None
    g_opt = res.x
    return g_opt, g_opt.sum()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    The routine mixes a fast LP based global improvement with a simple
    projected sub‑gradient descent.  It stops after ``budget_s`` seconds
    and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Initialise a candidate (use the best known construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2401                     # a reasonably large default
        a = np.full(n, 1.0, dtype=np.float64)

    # Normalise to the required ℓ₁‑norm S = √(2n)
    S = np.sqrt(2.0 * n)
    a = a * (S / a.sum())

    # bookkeeping
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    nfft = _next_pow2(2 * n - 1)

    # parameters for the sub‑gradient fallback
    eta = 0.5                           # base step size
    no_improve = 0
    max_no_improve = 2000

    iteration = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1.  current objective
        # --------------------------------------------------------------
        conv = _convolve_self_fft(a, nfft)
        M = conv.max()                  # because S² = 2n, evaluate = M
        if M < best_score:
            best_score = M
            best_a = a.copy()
            print(f"[{iteration:06d}] New best  M={best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # early exit if target already reached
        if best_score < 1.503:
            print(f"[{iteration:06d}] Target reached (M={best_score:.6f})")
            break

        # --------------------------------------------------------------
        # 2.  LP global improvement step
        # --------------------------------------------------------------
        g_opt, sum_g = _lp_max_sum(a, M, S)

        # If the LP succeeded and gave a larger ℓ₁‑norm, use it.
        if g_opt is not None and sum_g > S * (1.0 + 1e-6):
            # scale g back to the required norm S (still feasible)
            g_scaled = g_opt * (S / sum_g)

            # evaluate the candidate (cheap – one FFT)
            conv_g = _convolve_self_fft(g_scaled, nfft)
            M_g = conv_g.max()

            if M_g < M:
                # A strict improvement – accept g_scaled as the new iterate
                a = g_scaled
                iteration += 1
                continue
            else:
                # No improvement after scaling – fall back to a convex combo
                t_mix = 0.15
                a_candidate = (1 - t_mix) * a + t_mix * g_scaled
                a_candidate = _project_to_simplex(a_candidate, S)
                conv_cand = _convolve_self_fft(a_candidate, nfft)
                if conv_cand.max() < M:
                    a = a_candidate
                    iteration += 1
                    continue
                # otherwise fall back to sub‑gradient (below)

        # --------------------------------------------------------------
        # 3.  Sub‑gradient descent on the exact max‑convolution
        # --------------------------------------------------------------
        # arg‑max index of the convolution
        k = int(conv.argmax())
        # gradient of (a*a)_k w.r.t. a
        grad = np.zeros_like(a)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            step = eta * grad / gnorm
            a_candidate = _project_to_simplex(a - step, S)

            conv_cand = _convolve_self_fft(a_candidate, nfft)
            if conv_cand.max() < M:
                a = a_candidate
            else:
                # occasional random jitter to escape shallow plateaux
                if np.random.rand() < 0.02:
                    jitter = np.random.randn(n) * (0.02 * S)
                    a = _project_to_simplex(a + jitter, S)

        # slowly decay the step size
        eta = max(0.01, eta * 0.9995)

        # --------------------------------------------------------------
        # 4.  Random restart if we are stuck for a long time
        # --------------------------------------------------------------
        if no_improve >= max_no_improve:
            alpha = np.full(n, 0.3)
            a = np.random.dirichlet(alpha) * S
            print(f"[{iteration:06d}] Restarted from random simplex point.")
            no_improve = 0

        iteration += 1

    # --------------------------------------------------------------
    # Finalisation
    # --------------------------------------------------------------
    final_seq = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
