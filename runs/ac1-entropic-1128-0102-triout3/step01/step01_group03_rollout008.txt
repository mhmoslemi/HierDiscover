"""
Iterative LP‑based optimizer for the step‑function bound.

The routine starts from a reasonable seed (the classic half‑binomial
distribution or a previously found construction) and repeatedly solves a
linear program

    maximize   sum(g_i)
    subject to g_i ≥ 0,
               (f * g)_k ≤ M   for all convolution indices k,

where M = max_k (f * f)_k for the current sequence f.  The LP finds a
distribution g that can carry more total mass without increasing any
convolution value beyond the current maximum M.  After scaling g to
unit L¹‑norm we move a small step towards it (convex combination) and
repeat.  The maximal convolution value – and therefore the evaluation
score – monotonically decreases.  The algorithm stops after the allotted
time or when progress stalls and returns the best sequence seen.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions

def sqrt_series(n: int) -> np.ndarray:
    """
    Half‑binomial (central binomial) coefficients:
        a_i = C(2i,i) / 4**i,
    normalised to sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Autoconvolution x * x using real FFT.
    Result length = 2*len(x)-1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    return np.fft.irfft(X * X, n=m)


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def build_conv_ub_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the matrix A (size (2n‑1) × n) representing the linear map
        g ↦ f * g
    where * is discrete convolution.  Each row k of A contains
    the coefficients f_i that multiply the variables g_{k-i}.
    The matrix is very sparse (exactly n² non‑zeros) and is built
    using vectorised numpy operations.
    """
    n = f.shape[0]
    # column index for each variable (repeated n times)
    cols = np.repeat(np.arange(n, dtype=np.int64), n)

    # row indices: for each column j we need rows j, j+1, …, j+n‑1
    rows = (np.arange(n, dtype=np.int64)[:, None] + np.arange(n, dtype=np.int64)).flatten(order='F')
    # data = f repeated for each column
    data = np.tile(f, n)

    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP described in the docstring.
    Returns the optimal g (un‑scaled) or None if the solver fails.
    """
    n = f.shape[0]
    A = build_conv_ub_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # HiGHS (default for scipy ≥1.6) is fast for medium‑size LPs.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the evaluation
    metric.  The algorithm iteratively improves the sequence by solving a
    linear program that enlarges the feasible L¹‑mass without increasing
    the maximal autocorrelation, then moves a convex combination towards
    the solution.  The best sequence discovered before the time limit is
    returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine starting point and length
    try:
        # Use the globally supplied construction if present
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
        a = np.maximum(init_seq, 0.0)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        print(f"[init] Using supplied height_sequence_1 of length {n}.")
    except NameError:
        # Fallback: half‑binomial seed
        n = 2000                     # moderately large but still fast enough
        a = sqrt_series(n)
        print(f"[init] No external seed – using half‑binomial of length {n}.")

    # ------------------------------------------------------------------
    # Initial evaluation
    cur_conv = conv_fft(a)
    cur_M = cur_conv.max()
    cur_score = 2.0 * n * cur_M
    best_a = a.copy()
    best_score = cur_score
    print(f"[init] Initial score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation loop
    deadline = time.time() + budget_s - 5.0   # small safety margin
    it = 0
    t = 0.4                     # initial mixing factor (0 < t ≤ 1)
    no_improve_counter = 0

    while time.time() < deadline and it < 5000:
        # 1. compute current bound M
        cur_conv = conv_fft(a)
        cur_M = cur_conv.max()

        # 2. solve the LP to obtain a new direction g
        g_raw = solve_lp_step(a, cur_M)
        if g_raw is None:
            # fallback – small random perturbation and continue
            noise = (rng.random(n) - 0.5) * 1e-5
            a = project_onto_simplex(a + noise)
            it += 1
            continue

        sum_g = g_raw.sum()
        if sum_g <= 1e-12:          # degenerate solution – abort early
            break

        # Scale to unit L¹
        g = g_raw / sum_g

        # 3. convex combination
        a_candidate = (1.0 - t) * a + t * g
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = project_onto_simplex(a_candidate)

        # 4. evaluate candidate
        cand_conv = conv_fft(a_candidate)
        cand_M = cand_conv.max()
        cand_score = 2.0 * n * cand_M

        # 5. book‑keeping
        if cand_score < best_score - 1e-8:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[iter {it:06d}] New best score = {best_score:.6f}")
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # Accept only if we make progress (guarantees monotone decrease)
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
        else:
            # no progress – shrink the step size to be more cautious
            t = max(0.01, t * 0.85)

        # Occasionally inject a tiny random jitter to escape flat plateaux
        if rng.random() < 0.003:
            jitter = (rng.random(n) - 0.5) * 1e-4
            a = project_onto_simplex(a + jitter)

        # Decay the mixing factor slowly
        if (it + 1) % 500 == 0:
            t = max(0.01, t * 0.97)

        # Stop early if we have not improved for many iterations
        if no_improve_counter > 1200:
            print("[stop] No improvement for many iterations – terminating early.")
            break

        it += 1

    # ------------------------------------------------------------------
    # Final safety evaluation using the provided wrapper
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] Iterations = {it}, final reported score = {final_score:.6f}")

    return best_a.tolist()
