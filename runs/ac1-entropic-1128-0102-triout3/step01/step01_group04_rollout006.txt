import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp_max_sum(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        # j from max(0, k - (n-1)) to min(k, n-1)
        jmin = max(0, k - (n - 1))
        jmax = min(k, n - 1)
        # coefficient for b_j is f_{k - j}
        for j in range(jmin, jmax + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    # bounds: b_i >= 0
    bounds = [(0, None)] * n
    # maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    # inequality b_ub = M vector
    b_ub = np.full(2*n - 1, M)
    # solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        # fallback: return original f maybe
        return None
```

```python
"""
Propose a non‑negative step‑function (list of heights) minimising
the score = max(conv(a,a)).
The algorithm combines an analytic √‑derivative start with a linear
programming refinement that monotonically reduces the max‑convolution.
For a current sequence f we solve:

    maximise ∑ b_i
    s.t.   b_i ≥ 0
           (f * b)_k ≤ M  for all k

where M = max(f*f).  The optimiser yields a new shape `g`.  We mix
f←(1‑t)f+t g with a modest step‑size t; the mixing strictly lowers the
max‑convolution, and the process is repeated until no improvement or
the time budget expires.  Several lengths n and random Δ‑parameters are
tried; the best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _target_sum(n: int) -> float:
    """L1‑norm we enforce (same normalisation as the reference code)."""
    return np.sqrt(2.0 * n)

def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """Discrete derivative of sqrt(x+δ) and scaled to the required L1 norm."""
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale

def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq.tolist())

def _max_conv(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)_k."""
    return np.max(np.convolve(seq, seq))

def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A such that (A @ b) == f * b (full linear convolution).
    Shape of A is (2*len(f)-1, len(f)).
    Row k contains f_{k-j} in column j whenever 0 ≤ k‑j < len(f).
    """
    n = len(f)
    rows = []
    cols = []
    data = []
    # iterate over all possible output indices k
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        # for each j in [j_min, j_max] the coefficient is f[k‑j]
        # note that f index (k - j) is guaranteed to be in [0, n-1]
        coeffs = f[k - np.arange(j_min, j_max + 1)]
        idxs = np.arange(j_min, j_max + 1)
        rows.extend([k] * (j_max - j_min + 1))
        cols.extend(idxs.tolist())
        data.extend(coeffs.tolist())
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise   ∑ b_i
        subject to b_i ≥ 0
                   (f * b)_k ≤ M   for all k
    Returns the optimal b as a dense NumPy array, or None if the solver fails.
    """
    n = len(f)
    # Build the (2n‑1) × n sparse constraint matrix.
    A = _build_conv_matrix(f)
    # Right‑hand side – a tiny epsilon prevents numerical infeasibility.
    b_ub = np.full(2 * n - 1, M + 1e-12, dtype=float)
    # Objective = -∑ b_i (linprog minimises).
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n
    # Use HiGHS – works well for moderate size LPs.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    # Numerical noise may produce tiny negatives – clip.
    b = np.maximum(res.x, 0.0)
    return b

def _lp_refine(initial_seq: np.ndarray,
               deadline: float,
               t_initial: float = 0.5,
               t_min: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Iteratively improve `initial_seq` with the linear‑programming step.
    Returns the best sequence seen before the deadline and its score.
    """
    n = len(initial_seq)
    target = _target_sum(n)
    f = initial_seq.copy()
    best_f = f.copy()
    best_M = _max_conv(f)  # this is the evaluation score
    # Main refinement loop
    while time.time() < deadline:
        M = best_M
        # Solve the LP for the current f and bound M
        b = _solve_lp_max_sum(f, M)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        # Scale b to exact L1‑norm
        scale = target / sum_b
        g = b * scale
        # Try to mix f and g.  We start with a relatively aggressive step‑size
        # and halve it until we see a genuine reduction of the max‑convolution.
        t = t_initial
        improved = False
        while t >= t_min and time.time() < deadline:
            candidate = (1.0 - t) * f + t * g
            cand_M = _max_conv(candidate)
            if cand_M < best_M - 1e-12:
                # Accept the improvement
                f = candidate
                best_f = f.copy()
                best_M = cand_M
                improved = True
                break
            t *= 0.5
        if not improved:
            # No improvement could be obtained with the current LP solution;
            # we stop the refinement.
            break
        # If the improvement is extremely small we also stop.
        if M - best_M < 1e-7:
            break
    # Compute final score using the official wrapper (guarantees clipping etc.)
    final_score = _evaluate(best_f)
    return best_f, final_score

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low max‑convolution.
    Multiple lengths and δ‑initialisations are tried; each start is refined
    by the LP‑based monotone improvement described above.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Reserve a few seconds for final output handling
    deadline = start_time + max(0.0, budget_s - 5.0)

    best_seq = None
    best_score = float('inf')

    # Candidate lengths – we focus on the region where the previous run
    # performed best (≈1500‑2000).  A few extra lengths are added for robustness.
    lengths = list(range(1200, 2101, 100))

    # Outer loop: keep launching restarts until time runs out.
    while time.time() < deadline:
        n = int(np.random.choice(lengths))
        # Random delta ∈ [0, 0.8) – the analytic start is sensitive to δ.
        delta = np.random.rand() * 0.8
        seq0 = _analytic_sqrt_sequence(n, delta)

        # Quick sanity: compute the starting score (should be ≈π/2 ≈1.57)
        init_score = _evaluate(seq0)
        if init_score < best_score:
            best_score = init_score
            best_seq = seq0.copy()
            print(f"[init] n={n:4d} δ={delta:.3f} score={init_score:.6f}")

        # Allocate a modest inner budget – we leave about 0.3 s per restart
        # (the LP solve dominates the time anyway).
        inner_deadline = time.time() + 0.35
        seq_opt, score_opt = _lp_refine(seq0, deadline=inner_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[impr] n={n:4d} δ={delta:.3f} score={score_opt:.6f}")

        # Early exit: if we have already beaten the target by a comfortable margin,
        # we keep searching a little longer but we do not waste time on very long
        # sequences.
        if best_score < 1.51:
            # continue briefly to see if we can push below 1.503
            pass

    # Fallback: if for whatever reason nothing was found, use the analytic shape.
    if best_seq is None:
        n = 1500
        best_seq = _analytic_sqrt_sequence(n, delta=0.0)

    # Ensure the output is a plain Python list as required.
    return best_seq.tolist()

"""
Propose a non‑negative step‑function (list of heights) that minimises
the score returned by `evaluate_sequence`.  The score equals the
maximum value of a * a (auto‑convolution) when the L¹‑norm of a is
fixed to √(2 n).

The procedure consists of two phases:

1. **Analytic start** – a deterministic shape of the form
   a_i = (i+1+δ)^p – (i+δ)^p (√‑derivative corresponds to p=½) is
   generated and scaled so that Σa = √(2 n).  Several lengths n and
   random parameters (p, δ) are tried.

2. **LP‑based monotone refinement** – for a current sequence f with
   max‑convolution M = max(f∗f) we solve the linear program

        maximise   Σ b_i
        subject to b_i ≥ 0
                   (f ∗ b)_k ≤ M for all k∈[0,2n‑2]

   The optimiser yields a new shape g (after scaling to the required
   L¹‑norm).  A convex combination f←(1‑t)·f + t·g is accepted only
   if it strictly lowers the max‑convolution; the step size t starts
   at 0.5 and is halved until improvement is observed.  Repeating the
   LP‑solve/mix step monotonically reduces the bound.

Multiple random restarts are launched until the time budget (≈1000 s)
expires; the best sequence seen is returned.  If SciPy is not
available the algorithm falls back to a simple stochastic hill‑climb
similar to the baseline code.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – the LP‑based refinement needs it.
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Basic helpers (same for both LP and hill‑climb paths)
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Required L¹ norm (same normalisation as the reference code)."""
    return np.sqrt(2.0 * n)


def _analytic_power_sequence(n: int,
                             p: float = 0.5,
                             delta: float = 0.0) -> np.ndarray:
    """
    Analytic family a_i = (i+1+δ)^p – (i+δ)^p.
    Returned vector is already scaled to have L¹ norm = √(2 n).
    """
    i = np.arange(n, dtype=float)
    base = np.power(i + 1.0 + delta, p) - np.power(i + delta, p)
    # Guard against the pathological case p=0 (constant 0); replace with tiny.
    if base.sum() == 0.0:
        base = np.full_like(base, 1e-12)
    scale = _target_sum(n) / base.sum()
    return base * scale


# The original sqrt‑derivative is a special case of the power family
def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """Alias for p = ½ (the classic starting shape)."""
    return _analytic_power_sequence(n, p=0.5, delta=delta)


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq.tolist())


def _max_conv(seq: np.ndarray) -> float:
    """Maximum of the discrete auto‑convolution of `seq`."""
    return np.max(np.convolve(seq, seq))


# ----------------------------------------------------------------------
# Linear‑programming helpers (used only when SciPy is present)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A such that (A @ b) == f ∗ b (full linear
    convolution).  Shape of A is (2·len(f)‑1, len(f)).
    Column j of A contains the vector f placed from row j to j+len(f)-1.
    """
    n = f.size
    # Number of non‑zero entries is exactly n².
    rows = np.empty(n * n, dtype=int)
    cols = np.empty(n * n, dtype=int)
    data = np.empty(n * n, dtype=float)

    idx = 0
    for j in range(n):
        rows[idx:idx + n] = np.arange(j, j + n, dtype=int)
        cols[idx:idx + n] = j
        data[idx:idx + n] = f
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve:
        maximise Σ b_i
        s.t.   b_i ≥ 0
               (f ∗ b)_k ≤ M  for all k.
    Returns the optimal b as a dense NumPy array, or None if the
    optimizer fails.
    """
    n = f.size
    A = _build_conv_matrix(f)
    # Slightly relax the RHS to avoid numerical infeasibility.
    b_ub = np.full(2 * n - 1, M + 1e-12, dtype=float)
    c = -np.ones(n, dtype=float)            # linprog minimises → maximise Σb_i
    bounds = [(0.0, None)] * n

    # HiGHS is the default high‑performance LP solver in SciPy ≥1.6.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    # Clip tiny negatives that sometimes appear because of numerical noise.
    return np.maximum(res.x, 0.0)


def _lp_refine(initial_seq: np.ndarray,
               deadline: float,
               t_initial: float = 0.5,
               t_min: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based monotone refinement described in the header.
    Returns the best sequence obtained before the deadline and its score.
    """
    n = initial_seq.size
    target = _target_sum(n)

    f = initial_seq.copy()
    best_f = f.copy()
    best_M = _max_conv(f)                     # same as evaluate_sequence for valid L¹ norm

    while time.time() < deadline:
        M = best_M
        b = _solve_lp_max_sum(f, M)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 1e-12:                    # degenerate solution – stop
            break

        # Scale so that Σg = √(2 n)
        scale = target / sum_b
        g = b * scale

        # Try a full replacement first – it is the most aggressive step.
        cand_M = _max_conv(g)
        if cand_M < best_M - 1e-12:
            f = g
            best_f = f.copy()
            best_M = cand_M
            continue

        # Otherwise blend f and g, decreasing the mix factor until improvement.
        t = t_initial
        improved = False
        while t >= t_min and time.time() < deadline:
            candidate = (1.0 - t) * f + t * g
            cand_M = _max_conv(candidate)
            if cand_M < best_M - 1e-12:
                f = candidate
                best_f = f.copy()
                best_M = cand_M
                improved = True
                break
            t *= 0.5

        if not improved:
            # No mix gives a lower max‑convolution → stop iterating.
            break

        # Stop if the progress is negligible.
        if M - best_M < 1e-7:
            break

    final_score = _evaluate(best_f)
    return best_f, final_score


# ----------------------------------------------------------------------
# Simple hill‑climb (fallback when SciPy/Lp is unavailable)
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray,
                max_iter: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very lightweight stochastic hill‑climb (single‑coordinate Gaussian
    perturbations) that respects the L¹‑norm constraint.
    """
    n = seq.shape[0]
    target = _target_sum(n)

    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    # Initial step size – 20 % of the average entry.
    step_sigma = max(1e-8, seq.mean() * 0.2)

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        idx = np.random.randint(n)
        delta = np.random.randn() * step_sigma
        old_val = seq[idx]
        new_val = max(old_val + delta, 0.0)
        seq[idx] = new_val

        # Renormalise to keep Σa = √(2 n)
        seq *= target / seq.sum()

        cand_score = _evaluate(seq)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = seq.copy()
        else:
            # revert
            seq[idx] = old_val
            seq *= target / seq.sum()

        step_sigma *= 0.9995   # slow decay

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low max‑convolution.
    Multiple random restarts with different lengths, exponent p and shift
    δ are launched; each restart is refined either by the LP‑based
    monotone improvement (when SciPy is available) or by a fallback
    hill‑climb.  The best sequence found within the time budget is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for final I/O.
    deadline = start_time + max(0.0, budget_s - 5.0)

    best_seq = None
    best_score = float('inf')

    # Lengths that performed well in earlier runs (≈1500‑2000).  A few
    # extra values are tried for robustness.
    lengths = list(range(1200, 2101, 100))

    while time.time() < deadline:
        n = int(np.random.choice(lengths))

        # Randomly decide on the analytic family.
        if np.random.rand() < 0.6:                # bias towards sqrt‑derivative
            delta = np.random.rand() * 0.8
            seq0 = _analytic_sqrt_sequence(n, delta)
        else:
            p = np.random.rand() * 0.9 + 0.1      # p ∈ [0.1, 1.0]
            delta = np.random.rand() * 0.8
            seq0 = _analytic_power_sequence(n, p, delta)

        init_score = _evaluate(seq0)
        if init_score < best_score:
            best_score = init_score
            best_seq = seq0.copy()
            print(f"[init] n={n:4d} p={p if 'p' in locals() else 0.5:.3f}"
                  f" δ={delta:.3f} score={init_score:.6f}")

        # Allocate a short inner budget for the refinement of this restart.
        inner_deadline = time.time() + 0.35

        if _SCIPY_AVAILABLE:
            seq_opt, score_opt = _lp_refine(seq0, deadline=inner_deadline)
        else:
            seq_opt, score_opt = _hill_climb(seq0,
                                             max_iter=2500,
                                             deadline=inner_deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[impr] n={n:4d} score={score_opt:.6f}")

        # Very early exit when we have already broken the target bound.
        if best_score < 1.51:   # a comfortable margin above the target 1.503
            # keep searching a little to see if we can get even lower
            pass

    # ------------------------------------------------------------------
    # Final quick hill‑climb on the best sequence (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_deadline = deadline
        seq_refined, score_refined = _hill_climb(best_seq.copy(),
                                                 max_iter=5000,
                                                 deadline=final_deadline)
        if score_refined < best_score:
            best_score = score_refined
            best_seq = seq_refined
            print(f"[final] improved to {best_score:.6f}")

    # Fallback – should never happen, but guarantees a legal return value.
    if best_seq is None:
        best_seq = _analytic_sqrt_sequence(1500, delta=0.0)

    return best_seq.tolist()
