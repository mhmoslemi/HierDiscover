def project_simplex(v):
    # ensure non-negative and sum=1
    # Implementation from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def smoothed_max_and_grad(a, lam):
    # compute conv
    n = a.size
    size = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    max_c = conv.max()
    # weights
    # subtract max to avoid overflow
    exp_vals = np.exp(lam * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    # compute gradient
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)
    L = conv.shape[0] # 2*n-1
    g = 2.0 * conv_rev[L-1 - np.arange(n)]
    # gradient of smoothed max: g
    return max_c, g
```

def project_to_simplex(v):
   n = len(v)
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   ind = np.arange(1, n+1)
   cond = u - cssv / ind > 0
   if not np.any(cond):
       # all components zero, return uniform?
       return np.full_like(v, 1.0/n)
   rho = ind[cond][-1]
   theta = cssv[cond][-1] / rho
   w = np.maximum(v - theta, 0)
   return w

"""
Hybrid search for a low‑value step function.

The algorithm proceeds in three stages

1. **Parametric grid search** –  a family a_i ∝ (i+c)^(-p) is
   examined for several exponents `p` and offsets `c`.  The
   best candidate from this cheap scan is taken as a starting point.

2. **Smoothed‑max gradient descent** –  we minimise the quantity
   `max(conv(a,a))` (the denominator of `evaluate_sequence` is fixed by
   normalising `a` to ℓ¹‑norm 1).  The max is approximated by the
   log‑sum‑exp smoothed function  

        Φ_λ(a) = 1/λ·log Σ_k exp(λ·c_k) ,  c = a∗a

   Its gradient can be written in closed form and is computed by FFT
   in O(n log n).  After each gradient step the vector is projected onto
   the probability simplex (non‑negative, sum = 1).

3. **Annealing of the smoothing parameter** –  the smoothing factor `λ`
   starts small (smooth landscape) and is multiplied by a factor
   `1.05` every 1000 iterations up to a ceiling `λ_max`.  This guides the
   optimisation from a global to a local search.

The whole optimisation runs for the supplied time budget (default
1000 s).  The best sequence found is returned as a plain list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (used internally – the final answer will be checked by
# the provided `evaluate_sequence` function).
def fast_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)**2) – a is a 1‑D float array."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # FFT size large enough for the linear convolution
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def conv_and_gradient(a: np.ndarray, lam: float):
    """
    Return (max_convolution, gradient_of_smoothed_max, full_convolution).

    - a  : non‑negative vector with sum ≈ 1
    - lam: smoothing factor λ ≥ 0
    """
    n = a.shape[0]
    L = 2 * n - 1               # length of a*a
    # FFT length sufficient for both a*a and a*rev(w)
    size = 1 << ((3 * n - 2).bit_length())

    # FFT of a (pad once, reuse for both convolutions)
    A = np.fft.rfft(a, size)

    # ---- convolution a*a -------------------------------------------------
    conv_full = np.fft.irfft(A * A, size)[:L]          # linear conv, length L
    max_c = conv_full.max()

    # ---- smoothed max (log‑sum‑exp) --------------------------------------
    # shift by max to avoid overflow
    shifted = conv_full - max_c
    exp_vals = np.exp(lam * shifted)
    w = exp_vals / exp_vals.sum()                     # weights, sum = 1

    # ---- gradient of Φ_λ -----------------------------------------------
    # gradient_i = 2 * Σ_k w_k * a_{k-i}
    w_rev = w[::-1]                                   # reverse order
    W_rev = np.fft.rfft(w_rev, size)
    grad_full = np.fft.irfft(A * W_rev, size)[: (n + L - 1)]
    # pick the appropriate slice: indices L‑1, L‑2, …, L‑n
    idx = L - 1 - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return max_c, grad, conv_full


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation value.
    The function respects the 1000 s time budget and returns the best
    sequence discovered as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length that balances quality and speed.
    n = 3000

    # ------------------------------------------------------------------
    # 2. Parametric grid search: a_i ∝ (i + c)^(-p)
    best_seq = None
    best_val = np.inf

    p_vals = np.linspace(0.30, 0.70, 41)   # 0.30 … 0.70 in steps of 0.01
    c_vals = np.linspace(0.0, 5.0, 51)    # 0 … 5 in steps of 0.1

    idx_base = np.arange(1, n + 1, dtype=np.float64)   # i+1 part, later shifted by c

    for p in p_vals:
        for c in c_vals:
            # a_i = (i + c)^{-p}
            a = (idx_base + c) ** (-p)
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = fast_score(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
        # occasional progress report
        if time.time() > deadline:
            break
    print(f"[grid] best value after param sweep: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑max gradient descent starting from the best grid point.
    a = best_seq.copy()                        # current iterate
    current_val = best_val

    # optimisation hyper‑parameters
    eta = 0.12                                  # base step‑size (adjusted later)
    lam = 0.5                                   # initial smoothing
    lam_max = 30.0
    it = 0

    while time.time() < deadline:
        # -------- compute max convolution, gradient (smoothed) ----------
        max_c, grad, _ = conv_and_gradient(a, lam)

        # current objective (exact, not the smoothed surrogate)
        current_val = 2.0 * n * max_c / (a.sum() ** 2)

        # -------- simple adaptive step ---------------------------------
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break
        step = eta / grad_norm
        a_new = a - step * grad
        a_new = project_to_simplex(a_new)       # keep feasibility

        # -------- evaluate new candidate -------------------------------
        new_val = fast_score(a_new)

        # keep the globally best sequence
        if new_val < best_val:
            best_val = new_val
            best_seq = a_new.copy()
            print(f"[iter {it:6d}] new best {best_val:.6f}")

        # accept the move if it improves (otherwise keep the old point;
        # occasionally accept a worsening move to escape plateaus)
        if new_val < current_val:
            a = a_new
            current_val = new_val
        elif np.random.rand() < 0.001:
            a = a_new
            current_val = new_val

        # -------- schedule updates --------------------------------------
        it += 1
        if it % 2000 == 0:
            eta *= 0.9                     # slowly shrink step size
        if it % 1000 == 0 and lam < lam_max:
            lam = min(lam_max, lam * 1.05)   # increase smoothing → sharper max

    print(f"[final] best value found: {best_val:.6f}")
    # return a plain Python list as required by the evaluation harness
    return best_seq.tolist()
