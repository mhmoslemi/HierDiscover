def simplex_projection(v):
    # ensure non-negative, sum to 1
    # Standard algorithm: sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    if v.sum() == 1 and np.all(v >= 0):

def _fft_max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the self‑convolution a*a using an FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def _score_from_array(a: np.ndarray, fft_len: int) -> float:
    """Score used during search: 2·n·max_conv (n = len(a))."""
    n = len(a)
    max_conv = _fft_max_convolution(a, fft_len)
    return 2.0 * n * max_conv


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex (non‑negative, sum = 1).
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_decreasing(n: int) -> np.ndarray:
    """
    Produce a random decreasing non‑negative vector of length n,
    then normalise it to have sum = 1.
    """
    r = np.random.rand(n)
    r.sort()
    a = r[::-1]                      # decreasing
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n


def _make_power_law(n: int, alpha: float, c: float) -> np.ndarray:
    """
    Return a normalised power‑law vector a_i = (i + c)^{-α}.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + c) ** (-alpha)
    s = a.sum()
    return a / s if s > 0 else np.ones(n) / n
