"""
propose_candidate
-----------------
Implements a two‑phase continuous optimisation of the step‑function heights.
Phase 1: a smoothed “soft‑max” objective is minimised by projected gradient
descent.  The smoothing parameter λ is increased geometrically, turning the
smooth objective into the true max‑convolution as optimisation proceeds.
Phase 2: a classic sub‑gradient step (active at the current worst convolution
entry) fine‑tunes the solution.  Multiple sequence lengths are tried, the
best sequence (lowest score) is kept, and the algorithm stops automatically
when the allotted time budget is exhausted.

The method stays on the probability simplex (non‑negative entries summing to
one) by Euclidean projection after each additive update.  All heavy work is
performed with FFT‑based linear convolutions, so each iteration costs
O(n log n).  The implementation respects the competition’s constraints
(no I/O, no closures, top‑level helpers only) and finishes well within the
1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, sum(x) = 1} using the algorithm of Duchi et al."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – fall back to the centre of the simplex
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: resize an existing sequence to a new length via linear interpolation
# ----------------------------------------------------------------------
def _resize_sequence(seq, new_n: int) -> np.ndarray:
    old = np.asarray(seq, dtype=float)
    old_n = old.shape[0]
    if old_n == new_n:
        return old / old.sum()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, old)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        new_seq[:] = 1.0 / new_n
    else:
        new_seq /= s
    return new_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds,
    trying several sequence lengths and applying a smoothed‑max gradient
    descent followed by a sub‑gradient polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # If a previous construction is available, use it as a warm start.
    # ------------------------------------------------------------------
    try:
        init_seq = height_sequence_1          # type: ignore[name-defined]
    except NameError:
        init_seq = None

    # Lengths to explore – larger n yields a lower theoretical bound.
    lengths = [600, 800, 1000, 1200, 1500, 1800, 2100, 2500, 3000, 3500]
    np.random.shuffle(lengths)

    # ------------------------------------------------------------------
    # Main loop over candidate lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initial point on the simplex
        # --------------------------------------------------------------
        if init_seq is not None:
            a = _resize_sequence(init_seq, n)
        else:
            a = np.random.rand(n)
            a /= a.sum()

        # --------------------------------------------------------------
        # Pre‑compute FFT length (next power of two ≥ 2n‑1)
        # --------------------------------------------------------------
        conv_len = 2 * n - 1
        fft_len = 1 << ((conv_len).bit_length())

        # --------------------------------------------------------------
        # Phase 1 – smoothed‑max gradient descent (continuation on λ)
        # --------------------------------------------------------------
        lam = 0.5                 # initial smoothing parameter
        lam_max = 20.0
        lam_factor = 1.5
        step0 = 0.08              # base step size for this phase
        it_total = 0

        while lam <= lam_max and time.time() < deadline:
            # a modest number of iterations per λ; the total number will be
            # automatically limited by the overall time budget.
            max_iter_here = 2500
            for it in range(max_iter_here):
                it_total += 1
                # ----- convolution a*a (linear) -----
                A_f = np.fft.rfft(a, fft_len)
                conv = np.fft.irfft(A_f * A_f, fft_len)[:conv_len]

                # ----- keep the best score seen so far -----

                # shift for numerical stability

                p = exp_vals / exp_vals.sum()          # p sums to 1

                p_f = np.fft.rfft(p, fft_len)
                a_rev_f = np.fft.rfft(a_rev, fft_len)

        # --------------------------------------------------------------
        # Phase 2 – classic sub‑gradient polishing on the worst entry

            k_max = int(conv.argmax())
            max_val = conv[k_max]
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Improved (subgrad): n={n}, it={it_total + it + 1}, "
                      f"score={cur_score:.6f}")

            # ----- sub‑gradient for the active max entry -----
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            i_idx = np.arange(low, high + 1)
            j_idx = k_max - i_idx
            g = np.zeros_like(a)
            g[i_idx] = 2.0 * a[j_idx]

            # ----- step and simplex projection -----
            step = step0_sg / math.sqrt(it + 1)
            a = _project_simplex(a - step * g)

        # --------------------------------------------------------------
        # Store the result of the current length as a candidate for the next
        # length (helps the multiscale strategy).
        # --------------------------------------------------------------
        init_seq = a.copy()

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Return a plain Python list – evaluate_sequence accepts any list of floats.
    return [float(x) for x in best_seq]
