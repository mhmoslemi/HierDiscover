def improve_sequence(seq, iterations=10, t=0.1):
    f = np.array(seq, dtype=float)
    for it in range(iterations):
        # compute M
        conv_ff = np.convolve(f, f)
        M = np.max(conv_ff)
        # Build constraints A_ub
        n = len(f)
        # Use ints for row indexes
        # Build using np.indices maybe.
        i = np.arange(n, dtype=np.int32)
        j = np.arange(n, dtype=np.int32)
        I = np.repeat(i, n)
        J = np.tile(j, n)
        rows = I + J
        data = f[I]  # f repeated per I
        # create sparse matrix
        from scipy.sparse import csr_matrix
        A = csr_matrix((data, (rows, J)), shape=(2*n-1, n))
        # linear program
        from scipy.optimize import linprog
        c = -np.ones(n)  # maximize sum g
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
        if not res.success:
            print(f"LP failed at iteration {it}: {res.message}")
            break
        g = res.x
        sum_f = np.sum(f)
        sum_g = np.sum(g)
        if sum_g == 0:
            print("Zero sum g")
            break
        scale = sum_f / sum_g
        g_scaled = g * scale
        # update f
        f = (1 - t) * f + t * g_scaled
        # compute new score
        score = evaluate_sequence(f.tolist())
        print(f"Iter {it+1}: M={M:.6f}, sum_f={sum_f:.6f}, sum_g={sum_g:.6f}, score={score:.6f}")
    return f.tolist()

The functional  

\[
E(a)=\frac{2\,n\;\max_k (a*a)[k]}{\bigl(\sum_i a_i\bigr)^2}
\]

is *scale‑invariant*: multiplying all coefficients by a constant does not change the
value.  
Hence we can treat the **shape** of the step‑function as the optimisation
variable.

A classical construction (Catalan coefficients) gives a constant convolution
value `M = 1` and `E → π/2 ≈ 1.571`.  
To beat this bound we iteratively improve the shape using the linear‑program
step described in the statement:

* fix a current non‑negative vector `f`;
* let `M = max_k (f*f)[k]`;
* maximise `∑ g_i` under the linear constraints  

  \[
  (f*g)[k]\le M,\qquad k=0,\dots ,2n-2,\qquad g_i\ge 0 .
  \]

  This is a linear program (LP); the optimal `g` **adds the most $\ell_1$‑mass
  while respecting the same convolution bound**.
* rescale `g` so that `∑ g = ∑ f` and replace  

  \[
  f\;\leftarrow\;(1-t)f + t\,g
  \]

  with a modest step size `t`.  The new vector has the same $\ell_1$‑norm but a
  (typically) smaller `max_k (f*f)[k]`.  Repeating the step drives the value of
  `E` down.

The LP is built with a **sparse convolution matrix** – each row `k` contains the
coefficients `f_i` that appear in the constraint  
\(\sum_i f_i\,g_{k-i}\le M\).  For a length‑`n` vector the matrix has
`(2n‑1) × n` dimensions and `n²` non‑zero entries, which are generated
efficiently with NumPy broadcasting and stored as a `csr_matrix`.

The algorithm runs while there is time left (budget ≈ 1000 s).  It keeps the
best sequence seen so far and finally returns it.

"""Iterative LP improvement of a step‑function.

The algorithm starts from the Catalan construction (known ∼π/2 bound) and
repeatedly solves a linear program

    max  Σ g_i       s.t.   (f * g)[k] ≤ M  (∀k),   g_i ≥ 0,

where M = max_k (f * f)[k] and * denotes the ordinary discrete convolution.
The optimal g adds the largest possible ℓ₁‑mass while preserving the same
convolution bound.  After rescaling g to the same ℓ₁‑norm as f we move a small
step towards g.  This non‑increasing update reduces the quantity

    E(f) = 2·n·max_k (f*f)[k] / (Σ f_i)²

and converges to a better bound (empirically below 1.503).

The LP constraints are built with a sparse Toeplitz matrix; SciPy’s
high‑performance ‘highs’ solver handles the problem efficiently for
n up to a few thousand.
"""

def _catalan_sequence(n: int) -> list[float]:
    """Return the first n Catalan‑type coefficients c_i = C(2i,i)/4ⁱ."""
    seq = [1.0]
    for i in range(1, n):
        # c_{i} = c_{i-1} * (2*i-1) / (2*i)
        prev = seq[-1]
        seq.append(prev * (2 * i - 1) / (2 * i))
    return seq


def _conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
        (f * g)[k] = (A g)[k]
    for all k.
    """
    n = f.shape[0]
    # Indices of the non‑zero entries: all pairs (i,j) with 0≤i,j<n
    I = np.repeat(np.arange(n, dtype=np.int32), n)      # i repeated n times
    J = np.tile(np.arange(n, dtype=np.int32), n)        # j tiled n times
    rows = I + J                                         # k = i + j
    data = f[I].astype(np.float64)                       # weight f_i
    return csr_matrix((data, (rows, J)), shape=(2 * n - 1, n))


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    For a fixed non‑negative vector f and a bound M,
    solve the LP  max Σ g_i   s.t.   (f * g)[k] ≤ M,   g ≥ 0.
    Returns the optimal g (as a dense 1‑d array) or None on failure.
    """
    n = f.shape[0]
    A = _conv_matrix(f)                 # (2n‑1) × n, sparse
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximisation → minimise –Σg_i
    bounds = [(0, None)] * n

    # Use the HiGHS solver (fast, handles large sparse problems)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    # printing the reason can be useful for debugging
    print("[LP] failed:", res.message)
    return None


def _improve_sequence(f: np.ndarray,
                      max_iter: int = 12,
                      t_start: float = 0.5,
                      time_budget: float = 900.0) -> np.ndarray:
    """
    Perform a few iterations of the LP‑based improvement.
    Returns the best vector encountered.
    """
    best = f.copy()
    best_score = evaluate_sequence(best.tolist())
    start = time.time()

    for it in range(max_iter):
        # Stop if we run out of time (reserve a small margin)
        if time.time() - start > time_budget:
            break

        # current bound M
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # solve the auxiliary LP
        g = _solve_lp_for_g(f, M)
        if g is None:
            break

        sum_f = float(f.sum())
        sum_g = float(g.sum())
        if sum_g <= 1e-12:          # safety
            break

        # Rescale g so that it has the same ℓ₁‑norm as f
        scale = sum_f / sum_g
        g_scaled = g * scale

        # Try a decreasing sequence of step sizes until M improves
        accepted = False
        t = t_start
        while t > 1e-3:
            f_candidate = (1.0 - t) * f + t * g_scaled
            conv_candidate = np.convolve(f_candidate, f_candidate)
            M_cand = float(conv_candidate.max())
            # accept only if the bound strictly decreases
            if M_cand <= M * (1.0 - 1e-8):
                f = f_candidate
                accepted = True
                break
            t *= 0.5

        # If no improvement possible, abort the loop
        if not accepted:
            break

        # Evaluate the new sequence and keep the best one
        score = evaluate_sequence(f.tolist())
        print(f"[iter {it+1:2d}] M={M:.6f} → {M_cand:.6f}, "
              f"score={score:.6f}, t={t:.4f}")

        if score < best_score:
            best_score = score
            best = f.copy()

    elapsed = time.time() - start
    print(f"[improve] finished in {elapsed:.2f}s, best score={best_score:.6f}")
    return best


def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Generate a step‑function with a low evaluation score.

    The routine starts from the Catalan construction (the current state‑of‑the‑art)
    and performs a small number of LP‑based improvement steps (see the module
    docstring for details).  The total running time is kept well below the
    provided budget, and the best sequence found is returned.
    """
    start_total = time.time()

    # ------------------------------------------------------------------
    # 1) Choose the length of the sequence.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 1500  # a modest size that keeps the LP tractable

    # ------------------------------------------------------------------
    # 2) Initialise with the Catalan coefficients (they already give
    #    a very good bound ≈π/2).
    # ------------------------------------------------------------------
    seq = np.array(_catalan_sequence(n), dtype=np.float64)

    # ------------------------------------------------------------------
    # 3) Iteratively improve the shape using the LP step.
    #    The remaining time (budget_s – elapsed) is passed to the inner loop.
    # ------------------------------------------------------------------
    time_left = max(0.0, budget_s - (time.time() - start_total) - 0.5)
    improved = _improve_sequence(seq,
                                 max_iter=20,
                                 t_start=0.5,
                                 time_budget=time_left)

    # ------------------------------------------------------------------
    # 4) Final sanity check (the evaluator clamps values to [0,1000] anyway).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(improved.tolist())
    print(f"[propose_candidate] final length={len(improved)}, score={final_score:.6f}")

    # Make sure we return before the overall budget expires
    elapsed_total = time.time() - start_total
    if elapsed_total > budget_s:
        print("[propose_candidate] warning: exceeded total budget")
    return improved.tolist()
