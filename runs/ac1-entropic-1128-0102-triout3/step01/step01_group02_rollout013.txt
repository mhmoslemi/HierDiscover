import numpy as np
from scipy.optimize import linprog

def _catalan_sequence(n):
    seq = np.empty(n, dtype=float)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i-1] * (2*i - 1) / (2*i)
    return seq

def _build_A_active(a, active_ks):
    n = len(a)
    rows = []
    data = []
    cols = []
    for row_idx, k in enumerate(active_ks):
        # row for constraint sum_i a_i g_{k-i} <= M
        # For each j where 0 <= k - j < n => j in [max(0, k - (n-1)), min(n-1, k)]
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # coefficient for g_j is a_{k - j}
        coeffs = a[k - np.arange(j_start, j_end+1)]
        # Append to sparse matrix data (we can build dense matrix for few rows)
        rows.extend([row_idx] * (j_end - j_start + 1))
        cols.extend(list(range(j_start, j_end+1)))
        data.extend(coeffs.tolist())
    # Build dense matrix
    A = np.zeros((len(active_ks), n), dtype=float)
    A[rows, cols] = data
    return A

"""
Propose a near‑optimal step‑function.

The algorithm starts from the Catalan construction (which yields the
asymptotic bound π/2 ≈ 1.571) and then performs two refinement phases:

1️⃣  **Active‑set linear programming** –  
    For the current sequence `a` we compute the worst convolution value
    `M = max(a*a)`.  Only the indices where this maximum is attained are
    kept as linear constraints (they are the “active set”).  Solving a
    tiny LP maximises the ℓ¹‑norm of a new vector `g` under the
    constraint `a*g ≤ M`.  Mixing `a` a little towards `g` increases the
    total sum while preserving (or even lowering) the maximal
    convolution.  Repeating this step gives a monotone improvement of the
    evaluation score.

2️⃣  **Fast stochastic hill‑climbing** –  
    After the LP phase a light‑weight simulated‑annealing loop makes
    random mass‑transfer moves between coordinates and accepts them if
    they improve the objective (or with a temperature‑controlled
    probability).  The convolution is evaluated with an FFT for speed.

Both phases respect the constraints of `evaluate_sequence`
(non‑negative, ≤ 1000).  The whole routine runs inside the allotted
budget (default 1000 s) and returns the best sequence found.
"""

import time

from scipy.optimize import linprog

# Helper: Catalan‑type coefficients  c_i = C(2i,i)/4**i
def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients."""
    seq = np.empty(n, dtype=float)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


# ----------------------------------------------------------------------
# Helper: build the small constraint matrix for the active set
def _build_A_active(a: np.ndarray, active_ks: np.ndarray) -> np.ndarray:
    """
    For a given vector `a` and a list of convolution indices `active_ks`,
    build the matrix A where each row corresponds to the linear constraint

            sum_j a[k-j] * g_j  ≤  M      (for k in active_ks)

    Only the columns that actually appear are filled; the rest stay zero.
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []

    for row_idx, k in enumerate(active_ks):
        # j must satisfy 0 ≤ k-j < n  →  j ∈ [max(0,k-n+1), min(n-1,k)]
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # coefficients are a[k-j] for those j's
        js = np.arange(j_start, j_end + 1, dtype=int)
        coeffs = a[k - js]
        rows.extend([row_idx] * len(js))
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())

    # dense matrix – tiny because the active set is usually very small
    A = np.zeros((len(active_ks), n), dtype=float)
    A[rows, cols] = data
    return A

        iter_cnt += 1

    # ------------------------------------------------------------------
    # 3️⃣  Fast stochastic refinement (simulated annealing style)
    #    Only a few thousand cheap moves are attempted.
    a = best_seq.copy()
    n = a.shape[0]

    # FFT size for fast convolution (next power of two)
    fft_len = 1 << (2 * n - 1).bit_length()

    def max_conv(x: np.ndarray) -> float:
        """Return max_k (x*x)[k] using an FFT (real‑valued)."""
        fx = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(fx * fx)[: 2 * n - 1]
        return conv.max()

    cur_max = max_conv(a)
    cur_score = 2.0 * n * cur_max / (a.sum() ** 2)
    temperature = 0.1

    while time.time() - start < budget_s - 0.5:
        i, j = rng.choice(n, size=2, replace=False)

        # Move a small random fraction of the mass from i to j.
        delta = a[i] * rng.random() * 0.02   # ≤ 2 % of a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = max_conv(a_new)
        new_score = 2.0 * n * new_max / (a_new.sum() ** 2)

        # Accept if better, otherwise with Metropolis probability.
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - new_score) / max(temperature, 1e-8))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            cur_max = new_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[refine] new best score={best_score:.6f}")

        # simple exponential cooling
        temperature *= 0.9999
        if temperature < 1e-4:
            temperature = 1e-4

    print(f"[final] best score obtained: {best_score:.6f}")

    # Ensure compliance with the limits enforced in evaluate_sequence.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
