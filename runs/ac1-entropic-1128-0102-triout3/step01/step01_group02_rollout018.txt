def solve_lp(f, slack):
    # f length n; slack vector length 2n-1
    active_indices = np.where(slack > 1e-12)[0]
    # Build sparse matrix for constraints: for each k in active_indices:
    # For i in max(0, k-(n-1)) to min(k, n-1):
    # column i has value f[k-i]
    rows = []
    cols = []
    data = []
    for idx, k in enumerate(active_indices):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            rows.append(idx)
            cols.append(i)
            data.append(f[k - i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(len(active_indices), n))
    b_ub = slack[active_indices]  # upper bounds
    c = -np.ones(n)  # maximize sum means minimize -sum
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')

"""
Propose a candidate step‐function using an iterative linear‑programming scheme.

The functional to minimise is  

    E(a) = 2·n·max_k (a∗a)[k] / (∑ a_i)² .

If we normalise the vector to have ∑a_i = √(2n) then  E(a) = max_k (a∗a)[k] .
Hence the problem is equivalent to: keep the ℓ1‑norm at its maximal feasible
value √(2n) and drive the infinity‑norm of the self‑convolution down to 1.
The algorithm starts from the uniform vector (which already has the maximal
ℓ1‑norm) and repeatedly:

1. solves a linear programme that finds a vector g0 with the largest possible
   ℓ1‑norm under the constraint   f∗g0 ≤ M   (M = max(f∗f)),
2. rescales g0 to the normalisation √(2n) giving g,
3. mixes f and g with a small step size t – this never increases the maximal
   convolution value,
4. repeats until the maximal convolution drops below the target (≈1.5).

The LP is tiny (≈1000 variables) and is solved with SciPy’s HiGHS solver.
If SciPy is unavailable we fall back to the Catalan construction.
The whole routine respects the 1000 s budget and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: build the constraint matrix for  (f * b)[k] ≤ M
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray) -> "csr_matrix":  # type: ignore
    """Return A (as a CSR matrix) such that A @ b = f * b (full discrete convolution)."""
    from scipy.sparse import csr_matrix

    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # for each output index k = 0 .. 2n-2
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over the indices of f that contribute to (f*b)[k]
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[k - i])          # coefficient f_{k‑i}
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

    The method keeps the ℓ1‑norm at the theoretical maximum √(2n) and
    repeatedly reduces the infinity‑norm of the self‑convolution by solving
    a linear programme.  The final sequence is returned as a plain list.
    """

    # while still allowing a good bound.

    try:

            n = 800

    #   sum(a) = sqrt(2n)   and   max(a∗a) = 2 .
    # --------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    f = np.full(n, target_sum / n, dtype=np.float64)   # uniform
    conv_f = np.convolve(f, f)
    M = float(conv_f.max())

    # store the best vector seen (initially the uniform one)
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, sum={np.sum(f):.6f}, max_conv={M:.6f}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Iterative improvement
    # --------------------------------------------------------------
    max_iters = 30
    for it in range(1, max_iters + 1):
        # time check
        if time.time() - start > budget_s - 5:
            print("[budget] approaching time limit, stopping early.")
            break

        # --- Linear programme: maximise Σ b_i  subject to  f * b ≤ M , b ≥ 0
        # This is a standard LP: maximise -cᵀb  with c = -1.
        # ------------------------------------------------------------------

        A_ub = _build_constraint_matrix(f)                # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)   # all constraints ≤ M

        if M <= 1.503 + 1e-6:
            print(f"[target] reached M ≤ 1.503 after {it} iterations.")
            break

    # Final safety: clamp to the limits required by evaluate_sequence
    best_f = np.clip(best_f, 0.0, 1000.0)

    elapsed = time.time() - start
    print(f"[done] elapsed={elapsed:.1f}s, final length={n}, final score={best_score:.6f}")
    return best_f.tolist()
