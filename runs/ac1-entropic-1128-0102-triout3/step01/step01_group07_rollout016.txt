def solve_linearized_lp(seq):
    n = len(seq)
    conv0 = np.convolve(seq, seq)  # length 2n-1
    # Build coefficient matrix for constraints: dim (2n-1, n+1)
    # Use numpy for speed.
    M = np.zeros((2*n-1, n), dtype=np.float64)
    # fill each column
    for i in range(n):
        M[i:i+n, i] = seq
    # 2*M for coefficients of a_i
    A_ub = np.hstack((2*M, -np.ones((2*n-1, 1), dtype=np.float64)))
    b_ub = conv0.astype(np.float64)
    # equality for sum
    A_eq = np.zeros((1, n+1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    # bounds for a_i and t
    bounds = [(0, None)] * n + [(0, None)]
    # solve LinProg
    res = linprog(c=np.concatenate([np.zeros(n), [1.0]]), A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        # return None
        return None
    a_opt = res.x[:n]
    t_opt = res.x[-1]
    # ensure sum approx 1
    return a_opt, t_opt

def _solve_linearized_lp(seq):
    n = seq.shape[0]
    conv0 = np.convolve(seq, seq)
    rows = 2*n-1
    # Build sparse M
    row_idx = []
    col_idx = []
    data = []
    # For each i column
    for i in range(n):
        # rows i..i+n-1
        r = np.arange(i, i+n, dtype=np.int64)
        row_idx.append(r)
        col_idx.append(np.full(n, i, dtype=np.int64))
        data.append(2*seq)  # note same seq for each column
    # flatten lists
    row_idx = np.concatenate(row_idx)
    col_idx = np.concatenate(col_idx)
    data = np.concatenate(data)
    M = sparse.coo_matrix((data, (row_idx, col_idx)), shape=(rows, n)).tocsr()
    # column for t variable
    t_col = sparse.coo_matrix((-np.ones(rows), (np.arange(rows), np.full(rows, n))), shape=(rows, n+1)).tocsr()
    # Combine
    A_ub = sparse.hstack([M, t_col], format='csr')
    b_ub = conv0.astype(np.float64)

    # equality
    A_eq = np.zeros((1, n+1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    # bounds
    bounds = [(0, None)] * n + [(0, None)]

    # objective: minimize t -> c vector zeros for a_i and 1 for t
    c = np.zeros(n+1, dtype=np.float64)
    c[-1] = 1.0

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'presolve': True})

        print(f"Linprog failed: {e}")
        return None

    if not res.success:
        print(f"Linprog not successful: {res.message}")
        return None

    a_opt = res.x[:n]
    t_opt = res.x[-1]
    # Normalize to sum 1 (may drift due to numerics)
    a_opt = a_opt / a_opt.sum()
    return a_opt, t_opt

"""
Propose a step‑function sequence that (approximately) minimises

        2 * n * max(a*a) / (sum(a)**2)

where ``*`` denotes discrete convolution.
The search consists of three stages:

1. **Simple parametric families** – power‑law sequences a_i ∝ (i+1)^(-β)
   are tried on several lengths; the best among them is kept.

2. **Iterated linear‑programming improvement** – given a current candidate
   `f`, the following LP is solved:

        maximise   Σ b_j
        subject to (f * b)_k ≤ M   for all k
                    b_j ≥ 0

   where `M = max(f*f)`.  The solution `b` is normalised to sum‐1 and
   combined convexly with `f` (step size α).  The process is repeated
   a few times.  This is the “majorisation‑minimisation’’ scheme described
   in the statement and it typically reduces the supremum of the
   self‑convolution more efficiently than plain sub‑gradient steps.

3. **Short projected sub‑gradient polishing** – a few sub‑gradient
   steps are carried out to escape possible plateaus.

All operations respect the 1000 s wall‑clock budget, use only a single
CPU, and keep the sequence non‑negative (the simplex projection guarantees
`sum = 1`, which makes the evaluation scale‑invariant).
"""
import time
import numpy as np
from typing import List, Optional, Tuple

# ------------------------- helper utilities -------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(seq: np.ndarray) -> float:
    """Score = 2 * n * max(conv(seq, seq))   (seq is assumed to sum to 1)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


def _powerlaw_sequence(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)^(-β)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta)
    seq /= seq.sum()
    return seq


def _subgradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """One projected sub‑gradient step for max_k (seq*seq)_k."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    grad = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1, dtype=int)
    grad[idx] = 2.0 * seq[k - idx]          # 2·a_{k−i}
    new_seq = seq - eta * grad
    new_seq = _project_simplex(new_seq)
    return new_seq


def _solve_lp_for_b(f: np.ndarray, M: float) -> Optional[np.ndarray]:
    """
    Solve   maximise Σ b_j   s.t. (f * b)_k ≤ M,  b ≥ 0.
    Returns the optimal b (not normalised) or None on failure.
    """
    from scipy.optimize import linprog
    from scipy import sparse

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A where (A @ b)_k = (f * b)_k.
    # Each diagonal offset p (0 ≤ p < n) contains the constant f[p].
    diag_data = [np.full(n, f[p], dtype=np.float64) for p in range(n)]
    A = sparse.diags(diag_data, offsets=range(n), shape=(rows, n), format='csr')

    # Inequality: A @ b ≤ M
    A_ub = A
    b_ub = np.full(rows, M, dtype=np.float64)

    # Objective: maximise Σ b_j  <=> minimise -Σ b_j
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception as e:
        print("LP solve exception:", e)
        return None

    if not res.success:
        # print("LP failed:", res.message)
        return None
    return res.x
