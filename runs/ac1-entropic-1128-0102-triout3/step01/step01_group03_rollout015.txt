        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)

        a_new[target] += eps

        # optionally add small Gaussian noise to all entries (maybe not needed)

        # ensure non-negative and sum=1 automatically

        # Evaluate
        conv_new = conv_fft(a_new)
        max_new = conv_new.max()
        new_score = 2 * n * max_new

        # Accept? (lower always)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):

    # final evaluation using evaluate_sequence
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best score after search {final_score:.6f}")

"""
propose_candidate – Simulated‑annealing + weighted‑gradient search.

The algorithm builds a probability vector `a` (non‑negative, sum = 1) of length
`n`.  The objective to minimise is

    score(a) = 2 * n * max(conv(a, a))

where `conv` is the discrete convolution.  This is exactly the value returned
by `evaluate_sequence` after the internal clipping.

The search consists of several independent restarts.  Each restart follows a
simple annealing loop:

1.  Compute the convolution `c = conv(a, a)`.
2.  Pick the `K` indices with the largest convolution values (they are the
    current “bottlenecks’’).
3.  Build a weight vector `w` that places the convolution magnitude on those
    `K` indices and is zero elsewhere.
4.  The gradient of the weighted‑max (a smooth proxy for the true max) with
    respect to `a` is

        grad_i = a_i * sum(w) + Σ_{k} w_k * a_{k‑i}

    The second term is a correlation of `w` with `a`, which can be obtained
    with `np.correlate`.
5.  Perform a projected gradient step `a ← a - η·grad`.  The step size `η`
    and an annealing temperature decay smoothly during the run.
6.  Occasionally a random “mass‑swap’’ between two indices is tried to help
    escape shallow local minima.
7.  The best vector seen overall (across all restarts) is kept.

After the time budget expires the best vector is finally evaluated with the
provided `evaluate_sequence` and returned.

The method is deliberately simple, uses only NumPy (no external LP solver)
and respects the 1‑CPU limit.  It improves on the baseline half‑binomial
construction by allowing entries to become zero and by moving mass away from
the indices that currently dominate the convolution peak.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be defined at top level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First `n` coefficients of the series a_i = C(2i,i) / 4**i
    (the half‑binomial / central‑binomial coefficients).  The vector is
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
        {x | x_i ≥ 0,  Σ_i x_i = 1 }.
    Implementation of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Compute the discrete convolution x * x using real FFT.
    Length of the result is 2*len(x)-1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def weighted_gradient(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    """
    Compute a gradient that concentrates on the `top_k` largest entries of
    `conv`.  The weight on each selected index is its convolution value,
    which yields a smooth approximation of the L∞ norm.
    """
    n = a.shape[0]
    L = conv.shape[0]                     # = 2*n-1

    # indices of the largest values (ties are broken arbitrarily)
    if top_k >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]

    # build weight vector w (length L); non‑selected entries stay zero
    w = np.zeros(L, dtype=np.float64)
    w[idx] = conv[idx]

    # sum of the weights – appears together with a_i in the derivative
    w_sum = w.sum()

    # correlation of w with a (valid mode gives length n)
    # correlation_i = Σ_j w_{i+j} * a_j   (exactly Σ_k w_k * a_{k-i})
    corr = np.correlate(w, a, mode='valid')   # shape (n,)

    # gradient = a * w_sum + corr
    grad = a * w_sum + corr
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises the evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Amount of time (seconds) allocated to the search.  The routine stops
        a few seconds before this limit to avoid being terminated.
    **kwargs
        Unused – kept for compatibility with the external harness.

    Returns
    -------
    list[float]
        The best sequence (normalised to sum = 1) found before the time limit.
    """
    rng = np.random.default_rng(seed)

    # -------------------- configuration --------------------
    n = 2500                     # length of the candidate vector
    top_k = 10                   # how many peak indices influence the gradient
    max_iters_per_restart = 5_000_000   # safety cap (real stop is time)

    # annealing schedule parameters (they decay during a run)
    eta_start = 0.12             # initial step size multiplier
    temp_start = 0.05            # initial temperature for uphill moves
    decay_factor = 0.99995       # multiplicative decay per iteration
    random_swap_prob = 0.006     # probability of a random mass‑swap each iter
    random_swap_frac = 0.03      # fraction of source mass moved in a random swap

    # number of independent restarts – each gets roughly equal share of time
    n_restarts = 3

    # -------------------- bookkeeping --------------------
    best_a = None
    best_score = np.inf

    start_time = time.time()
    global_deadline = start_time + budget_s - 5.0   # safety margin

    for restart in range(n_restarts):
        # allocate time slices (simple round‑robin)
        remaining_time = global_deadline - time.time()
        if remaining_time < 1.0:
            break
        # initialise a different starting point for each restart
        if restart == 0:
            a = sqrt_series(n)                     # classic half‑binomial
        elif restart == 1:
            a = np.full(n, 1.0 / n)                # uniform
        else:
            # random Dirichlet (concentration 0.8 gives slightly spiky vectors)
            a = rng.dirichlet(np.full(n, 0.8))
        a = a / a.sum()                            # ensure exact sum = 1

        cur_score = 2.0 * n * conv_fft(a).max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        eta = eta_start
        temperature = temp_start

        it = 0
        while time.time() < global_deadline and it < max_iters_per_restart:
            # ----- core evaluation -----
            conv = conv_fft(a)
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv

            # ----- gradient step based on top‑k peaks -----
            grad = weighted_gradient(a, conv, top_k)

            step = eta * grad
            # ensure we do not step beyond the available mass (allow zeros)
            step = np.minimum(step, a)               # never subtract more than present
            a_trial = a - step
            # project back onto the simplex (non‑negative, sum = 1)
            a_trial = project_onto_simplex(a_trial)

            # ----- occasional random swap to increase diversity -----
            if rng.random() < random_swap_prob:
                src = rng.choice(np.where(a > 1e-12)[0])
                dst = rng.integers(n)
                delta = min(a[src] * random_swap_frac, a[src] * 0.9)
                a_swap = a_trial.copy()
                a_swap[src] -= delta
                a_swap[dst] += delta
                a_swap = project_onto_simplex(a_swap)
                a_trial = a_swap

            # ----- evaluate candidate -----
            conv_trial = conv_fft(a_trial)
            trial_score = 2.0 * n * conv_trial.max()

            # ----- accept / reject (simulated annealing) -----
            accept = False
            if trial_score <= cur_score:
                accept = True
            else:
                # Metropolis acceptance
                if temperature > 0 and rng.random() < np.exp(-(trial_score - cur_score) / max(temperature, 1e-12)):
                    accept = True

            if accept:
                a = a_trial
                cur_score = trial_score

            # ----- keep global best -----
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[restart {restart} iter {it:07d}] new best score = {best_score:.6f}")

            # ----- schedule decay -----
            eta *= decay_factor
            temperature *= decay_factor
            eta = max(eta, 1e-6)
            temperature = max(temperature, 1e-9)

            it += 1

    # ------------------------------------------------------------------
    # final evaluation using the provided security‑checked routine
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best reported score = {final_score:.6f} (iterations elapsed ≈ {it})")
    return best_a.tolist()
