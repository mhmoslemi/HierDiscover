Linear‑programming + hill‑climbing optimizer for the step‑function task.

The routine works as follows:

1. Choose a working size `n` (default 2000).  If a reference sequence
   ``height_sequence_1`` is available it is down‑sampled to this size,
   otherwise a simple power‑law (i+1)^‑0.5 is used.  The vector is
   normalised to have L1‑norm 1 (the objective is scale‑invariant).

2. Repeatedly:
   * compute the current maximal self‑convolution  M = max(f*f);
   * build the dense convolution matrix A such that  (f*g)[k] = (A g)_k;
   * solve the linear program
         maximise  sum(g)   (equivalently minimise –sum(g))
         subject to   A g ≤ M   and   g ≥ 0.
     (``scipy.optimize.linprog`` with the HiGHS backend.)
   * scale the LP solution down so that its total mass does not exceed
     the current mass of f;
   * blend it with the current vector using a small step size `t`
     (trying 0.05, 0.10, 0.15, 0.20) and accept the first improvement.
   The blend is renormalised to unit L1‑norm.  The process stops
   when no improvement can be found or the time budget is exhausted.

3. If the original reference sequence existed and had a different
   length, the best vector is linearly interpolated back to that length
   (preserving the shape) and renormalised.

4. Finally a cheap random hill‑climbing phase moves a tiny amount of
   mass between two random positions until the time budget expires.

All numbers stay non‑negative, are well below the clipping threshold
(1000), and the implementation respects the 1000 s time limit.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _conv_max(a: np.ndarray) -> float:
    """Maximum value of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


def _score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector normalised to sum = 1."""
    return 2.0 * a.shape[0] * _conv_max(a)


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the dense matrix A (size (2·n‑1) × n) such that
    (A @ g) == np.convolve(f, g) for any vector g of length n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f
    return A

    minimises ``evaluate_sequence``.
    """

    # ------------------------------------------------------------------
    np.random.seed(seed)
    finish_time = time.time() + budget_s - 2.0   # small safety margin

    # 1. Determine a working size and an initial vector
    # ------------------------------------------------------------------
    try:

        ref_len = len(ref)
    except Exception:
        ref = None
        ref_len = 0

    # work size – keep it modest for the LP step
    n_work = 2000
    if ref is not None and ref_len <= n_work:
        n_work = ref_len

    if ref is not None:
        # down‑sample or truncate the reference to n_work points
        if ref_len != n_work:
            x_old = np.arange(ref_len, dtype=np.float64)
            x_new = np.linspace(0, ref_len - 1, n_work)
            init = np.interp(x_new, x_old, np.asarray(ref, dtype=np.float64))
        else:
            init = np.asarray(ref, dtype=np.float64).copy()
    else:
        # simple power‑law starting point
        i = np.arange(n_work, dtype=np.float64)
        init = (i + 1.0) ** (-0.5)

    init = np.maximum(init, 0.0)
    s = init.sum()
    if s == 0.0:
        init = np.ones(n_work, dtype=np.float64) / n_work
    else:
        init = init / s                     # normalise to L1 = 1

    best = init.copy()
    best_score = _score(best)
    print(f"[init] n={n_work}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement loop
    # ------------------------------------------------------------------
    while time.time() < finish_time - 5.0:
        n_cur = best.shape[0]
        M = _conv_max(best)                 # current max convolution

        # construct convolution matrix A for the constraints A g ≤ M
        A = _build_conv_matrix(best)
        ub = np.full(A.shape[0], M, dtype=np.float64)

        # linear programme: maximise sum(g)  ⇔ minimise –sum(g)
        c = -np.ones(n_cur, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n_cur)]

        if not res.success:

            break

        sum_f = best.sum()

        for t in (0.05, 0.10, 0.15, 0.20):
            cand = (1.0 - t) * best + t * g
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                continue
            cand = cand / cand_sum          # renormalise to unit L1
            sc = _score(cand)
            if sc < best_score - 1e-12:
                best = cand
                best_score = sc
                improved = True
                print(f"[blend] t={t:.2f}, score={sc:.6f}")
                break
        if not improved:
            print("[blend] no improvement – leaving LP loop")
            break

    # ------------------------------------------------------------------
    # 3. Upscale back to the original reference length (if it existed)
    # ------------------------------------------------------------------
    if ref is not None and ref_len != best.shape[0]:
        x_src = np.linspace(0, 1, best.shape[0])
        x_dst = np.linspace(0, 1, ref_len)
        up = np.interp(x_dst, x_src, best)
        up = np.maximum(up, 0.0)
        up = up / up.sum()
        up_score = _score(up)
        print(f"[upscale] score={up_score:.6f}")
        if up_score < best_score:
            best = up
            best_score = up_score

    # ------------------------------------------------------------------
    # 4. Final cheap hill‑climbing phase (random mass transfers)
    # ------------------------------------------------------------------
    while time.time() < finish_time:
        a = best.copy()
        n = a.shape[0]
        i1, i2 = np.random.choice(n, size=2, replace=False)
        delta = (np.random.rand() - 0.5) * 0.02   # ±1 % of donor mass
        if a[i1] + delta < 0.0:
            delta = -a[i1] + 1e-12
        if a[i2] - delta < 0.0:
            delta = a[i2] - 1e-12
        a[i1] += delta
        a[i2] -= delta
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            continue
        a = a / s
        sc = _score(a)
        if sc < best_score - 1e-12:
            best = a
            best_score = sc
            print(f"[hill] score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found
    # ------------------------------------------------------------------
    return best.tolist()
