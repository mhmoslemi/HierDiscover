def max_conv_for_r(n, S, r):
    # Input r in (0,1)
    log_r = np.log(r)
    # compute denominator
    r_pow_n = np.exp(n * log_r)
    denom = 1 - r_pow_n
    if denom <= 0:
        return np.inf
    A = S * (1 - r) / denom
    # compute coefficients
    # create array of k powers
    k = np.arange(2 * n - 1, dtype=np.float64)
    log_r_pow = k * log_r
    r_pow = np.exp(log_r_pow)
    # coefficient array:
    coeff = np.empty(2 * n - 1, dtype=np.float64)
    coeff[:n] = np.arange(1, n + 1)
    coeff[n:] = np.arange(n - 1, 0, -1)
    conv_vals = (A * A) * coeff * r_pow
    max_val = conv_vals.max()
    return max_val

def find_best_r(n, S):
    # coarse grid over epsilon
    eps_vals = np.logspace(-6, -2, num=2000)  # epsilon = 1 - r
    best_score = np.inf
    best_r = None
    for eps in eps_vals:
        r = 1 - eps
        if r <= 0:
            continue
        score = max_conv_for_r(n, S, r)
        if score < best_score:
            best_score = score
            best_r = r
    # refine around best_r
    # Define interval [best_r - delta, best_r + delta] where delta small.
    # Ensure within (0,1)
    delta = max(best_r * 0.5, 1e-4)  
    left = max(0.0, best_r - delta)
    right = min(0.999999999, best_r + delta)
    # perform golden-section search
    phi = (np.sqrt(5) - 1) / 2
    for _ in range(80):
        r1 = right - phi * (right - left)
        r2 = left + phi * (right - left)
        score1 = max_conv_for_r(n, S, r1)
        score2 = max_conv_for_r(n, S, r2)
        if score1 < score2:
            right = r2
            if score1 < best_score:
                best_score = score1
                best_r = r1
        else:
            left = r1
            if score2 < best_score:
                best_score = score2
                best_r = r2
    return best_r, best_score

def build_sequence(n, S, r):
    log_r = np.log(r)
    # compute A
    r_pow_n = np.exp(n * log_r)
    A = S * (1 - r) / (1 - r_pow_n)
    # compute a_i
    i = np.arange(n, dtype=np.float64)
    a = A * np.exp(i * log_r)  # equivalent to A * r**i
    # adjust sum to be exactly S (project onto simplex)
    sum_a = a.sum()
    if sum_a != 0:
        a = a * (S / sum_a)
    # ensure nonnegative
    a = np.maximum(a, 0.0)
    return a.tolist()

def propose_candidate(...):
    import numpy as np, time

    # Choose length
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2401

    S = np.sqrt(2.0 * n)

    # define function compute_score
    def score_of_r(r):
        # r must be in (0,1)
        log_r = np.log(r)
        # compute r^n
        r_pow_n = np.exp(n * log_r)
        denom = 1.0 - r_pow_n
        if denom <= 0.0:
            return np.inf
        A = S * (1.0 - r) / denom
        # compute coeff array
        k_arr = np.arange(2*n - 1, dtype=np.float64)
        # compute r^k
        r_pow_k = np.exp(k_arr * log_r)  # may underflow
        # coefficient for convolution
        coeff = np.empty_like(k_arr)
        coeff[:n] = np.arange(1, n+1, dtype=np.float64)
        coeff[n:] = np.arange(n-1, 0, -1, dtype=np.float64)
        conv_vals = (A * A) * coeff * r_pow_k
        max_val = conv_vals.max()
        return max_val  # equal to evaluate sequence score

    # search over epsilon
    # produce epsilon range
    eps_min = 1e-5
    eps_max = 0.01
    num_grid = 5000
    eps_grid = np.logspace(np.log10(eps_min), np.log10(eps_max), num=num_grid)
    best_score = np.inf
    best_r = None
    for eps in eps_grid:
        r = 1.0 - eps
        if not (0 < r < 1):
            continue
        sc = score_of_r(r)
        if sc < best_score:
            best_score = sc
            best_r = r
    # refine around best_r with golden section
    if best_r is None:
        best_r = 0.999  # fallback
        best_score = score_of_r(best_r)

    # define refined interval
    delta = max(0.5 * best_r, 0.001) # allow bigger interval but limited
    left = max(0.0, best_r - delta)
    right = min(0.999999999, best_r + delta)

    phi = (np.sqrt(5) - 1) / 2
    for _ in range(60):
        r1 = right - phi * (right - left)
        r2 = left + phi * (right - left)
        s1 = score_of_r(r1)
        s2 = score_of_r(r2)
        if s1 < s2:
            right = r2
            if s1 < best_score:
                best_score = s1
                best_r = r1
        else:
            left = r1
            if s2 < best_score:
                best_score = s2
                best_r = r2

    # build final sequence
    r = best_r
    log_r = np.log(r)
    r_pow_n = np.exp(n * log_r)
    A = S * (1.0 - r) / (1.0 - r_pow_n)
    i = np.arange(n, dtype=np.float64)
    a = A * np.exp(i * log_r)  # A * r**i
    # enforce sum = S (small rounding errors)
    sum_a = a.sum()
    if sum_a > 0:
        a *= (S / sum_a)
    a = np.maximum(a, 0.0)
    # final check:
    # verify using evaluate_sequence (debug)
    try:
        val = evaluate_sequence(a.tolist())
        print(f"Final candidate score: {val:.6f} (r={best_r:.9f})")
    except Exception as e:
        print("Evaluation failed:", e)

    return a.tolist()

"""
Propose a new step‑function using a two‑phase search:

1.  **Geometric seed** –  Scan a one‑parameter family  
    a_i = A·r^i (r∈(0,1)) that obeys the ℓ₁‑norm  Σa_i = √(2·n).  
    For each r we compute the exact maximal entry of the self‑convolution
    (no soft‑max approximation).  The best r gives a very good starting
    point (the score is already around 1.8 for n≈2400).

2.  **Projected sub‑gradient descent** –  Starting from the geometric seed we
    minimise the true objective  
    `max_k (a*a)[k]`  by iterative projected sub‑gradient steps.
    The sub‑gradient of the max‑convolution is obtained analytically:
    for every shift k attaining the current maximum the gradient contribution
    is 2·a_rev shifted by k.  After a step we project back onto the
    ℓ₁‑simplex `{x≥0, Σx = √(2·n)}`.  Occasional random jitter and
    restarts avoid stagnation.

The routine respects the 1000 s time budget, prints progress, and finally
returns the best sequence encountered as a plain Python list of floats.
"""

import time
import numpy as np

# Utility functions (no closures, no lambdas)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returning the first 2·len(a)-1 entries."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _geom_sequence(n: int, r: float) -> np.ndarray:
    """Return the geometric sequence a_i = A·r^i with Σa_i = √(2n)."""
    S = np.sqrt(2.0 * n)
    # avoid overflow / under‑flow for extreme r
    log_r = np.log(r)
    r_pow_n = np.exp(n * log_r)
    denom = 1.0 - r_pow_n
    if denom == 0.0:
        raise ValueError("Denominator vanished – r too close to 1")
    A = S * (1.0 - r) / denom
    i = np.arange(n, dtype=np.float64)
    a = A * np.exp(i * log_r)        # A·r^i
    # correct any tiny rounding error on the ℓ₁‑norm
    a_sum = a.sum()
    if a_sum != 0.0:
        a *= (S / a_sum)
    return a


def _max_conv_of_sequence(a: np.ndarray) -> float:
    """Return max(a*a) – the exact quantity minimised (sum is already √(2n))."""
    conv = np.convolve(a, a)        # small n, direct convolution is fine
    return float(conv.max())


def _score_geom(r: float, n: int) -> float:
    """Fast score for a geometric shape (no allocation of the full vector)."""
    S = np.sqrt(2.0 * n)
    log_r = np.log(r)
    r_pow_n = np.exp(n * log_r)
    denom = 1.0 - r_pow_n
    if denom <= 0.0:
        return np.inf
    A = S * (1.0 - r) / denom
    # coefficients of the convolution (k+1) for k<n, (2n-1-k) otherwise
    k = np.arange(2 * n - 1, dtype=np.float64)
    coeff = np.empty_like(k)
    coeff[:n] = np.arange(1, n + 1, dtype=np.float64)
    coeff[n:] = np.arange(n - 1, 0, -1, dtype=np.float64)
    r_pow_k = np.exp(k * log_r)
    conv_vals = (A * A) * coeff * r_pow_k
    return float(conv_vals.max())


def _grad_max(a: np.ndarray, conv: np.ndarray, nfft: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    Takes the average over all indices attaining the maximal value.
    """
    max_val = conv.max()
    mask = np.isclose(conv, max_val, atol=1e-12)
    w = mask.astype(np.float64)
    if w.sum() == 0.0:
        return np.zeros_like(a)
    w /= w.sum()                     # normalise – any scaling works with step‑size
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The method first scans a simple exponential family,
    then refines the best candidate with a projected sub‑gradient descent.
    It stops after `budget_s` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2401                     # a sensible default (used by the previous run)

    S = np.sqrt(2.0 * n)            # required ℓ₁‑norm
    nfft = _next_pow2(2 * n - 1)    # FFT length (power of two)

    # ------------------------------------------------------------------
    # 2) Phase 1 – geometric seed (grid + tiny refinement)
    # ------------------------------------------------------------------
    print("[phase‑1] Searching geometric family (n = %d)" % n)

    # coarse logarithmic grid in ε = 1‑r
    eps_grid = np.logspace(-5, -2, num=2000)   # ε ∈ [1e‑5, 1e‑2]
    best_r = None
    best_score = np.inf

    for eps in eps_grid:
        r = 1.0 - eps
        if r <= 0.0 or r >= 1.0:
            continue
        sc = _score_geom(r, n)
        if sc < best_score:
            best_score = sc
            best_r = r

    # tiny local refinement around the best point (golden‑section)
    if best_r is None:
        best_r = 0.999
        best_score = _score_geom(best_r, n)

    left = max(0.0, best_r - 5e-4)
    right = min(0.9999999, best_r + 5e-4)
    phi = (np.sqrt(5) - 1) / 2          # 0.618…

    for _ in range(50):
        r1 = right - phi * (right - left)
        r2 = left + phi * (right - left)
        s1 = _score_geom(r1, n)
        s2 = _score_geom(r2, n)
        if s1 < s2:
            right = r2
            if s1 < best_score:
                best_score, best_r = s1, r1
        else:
            left = r1
            if s2 < best_score:
                best_score, best_r = s2, r2

    print(f"[phase‑1] Best geometric r = {best_r:.10f}, score = {best_score:.6f}")

    # construct the seed vector
    a = _geom_sequence(n, best_r)

    # bookkeeping for the best solution found overall
    best_a = a.copy()
    best_score = _max_conv_of_sequence(best_a)

    # ------------------------------------------------------------------
    # 3) Phase 2 – projected sub‑gradient descent on the exact max
    # ------------------------------------------------------------------
    print("[phase‑2] Starting sub‑gradient descent")
    base_eta = 0.3 * S / n       # initial stepsize (≈0.01 for n≈2400)
    iteration = 0
    no_improve = 0
    max_no_improve = 1500
    target = 1.503

    while time.time() < deadline:
        # 3.1 current convolution
        conv = _convolve_self_fft(a, nfft)
        cur_max = float(conv.max())

        # update global best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{iteration:06d}] New best score = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        if best_score < target:
            print(f"[{iteration:06d}] Target reached (score < {target})")
            break

        # 3.2 sub‑gradient of the max‑convolution
        grad = _grad_max(a, conv, nfft)

        # 3.3 descent step (project back onto the ℓ₁‑simplex)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0.0:
            eta = base_eta / (1.0 + iteration * 1e-4)
            step = eta * grad / grad_norm
        else:
            step = np.zeros_like(a)

        a_candidate = _project_to_simplex(a - step, S)

        # 3.4 accept if we truly improve the objective
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = float(cand_conv.max())

        if cand_max < cur_max:
            a = a_candidate
        else:
            # occasional random perturbation to escape plateaux
            if np.random.rand() < 0.02:
                jitter = np.random.randn(n) * (eta * 0.5)
                a = _project_to_simplex(a + jitter, S)

        # 3.5 restart if we have been stuck for a long time
        if no_improve >= max_no_improve:
            alpha = np.full(n, 0.3)
            a = np.random.dirichlet(alpha) * S
            no_improve = 0
            print(f"[{iteration:06d}] Restarted from a random simplex point.")

        iteration += 1

    # ------------------------------------------------------------------
    # 4) Final verification and return
    # ------------------------------------------------------------------
    final_seq = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")

    return final_seq.tolist()
