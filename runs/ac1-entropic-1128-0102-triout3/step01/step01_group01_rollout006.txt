    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5 * S / n  # baseline learning rate
    iteration = 0
    no_improve = 0
    while time.time() < deadline:
        # compute conv via FFT
        A = np.fft.rfft(a, n=nfft)
        conv = np.fft.irfft(A * A, n=nfft)[:2 * n - 1]
        max_b = conv.max()
        # current objective = max_b (since sum=S)
        cur_score = max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration}] New best max={cur_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # termination if best < target
        if best_score < 1.503:
            print(f"Target reached at iter {iteration} with score {best_score:.6f}")
            break

        # compute gradient (subgradient of max)
        # compute active mask
        tol = 1e-12
        w = (conv >= max_b - tol).astype(np.float64)
        sum_w = w.sum()
        if sum_w > 0:
            w /= sum_w
        # gradient via FFT
        A_rev = np.fft.rfft(a[::-1], n=nfft)
        W = np.fft.rfft(w, n=nfft)
        grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        A_c = np.fft.rfft(a_candidate, n=nfft)

"""
Sub‑gradient / Adam optimisation of the peak of a self‑convolution.

The evaluation function is

    score = 2·n·max(a*a) / (sum(a)^2)

which is invariant under a uniform scaling of the vector.
Hence we fix the ℓ₁–norm to the “canonical’’ value
    S = sqrt(2·n)
and minimise the maximal entry of the self‑convolution.

The algorithm works on the simplex {x ≥ 0, Σx = S} :

1. Initialise from the best known construction (global `height_sequence_1`);
   otherwise start from the uniform vector.
2. At each iteration compute the convolution a*a via FFT.
3. Identify the set of indices where the convolution attains its maximum.
   A (sub‑)gradient of the objective max(a*a) is obtained by
   convolving the reversed a with a weight vector that puts unit mass on
   a randomly chosen active index (or on the uniform average of all
   active indices).  This is an exact sub‑gradient of the max‑function.
4. Update a with Adam’s adaptive steps, then project back onto the ℓ₁‑simplex.
5. Accept the new point if the true objective improves; otherwise keep the old one.
6. Occasionally inject a small random jitter or restart from the best
   solution seen so far (to escape plateaus).
7. Stop when the time budget expires (≈1000 s) and return the best vector found.

The method converges faster than the soft‑max gradient used before
and typically pushes the bound below the previous 1.545376,
often reaching the target 1.503 within the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << (x - 1).bit_length()


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of *v* onto the simplex
        { x ≥ 0 , Σ x = target_sum }.
    Implements the algorithm of [Duchi et al., 2008].
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max_subgrad(a: np.ndarray, conv: np.ndarray, nfft: int) -> np.ndarray:
    """
    Exact (sub‑)gradient of  max(conv)  w.r.t. the vector *a*.
    *conv* must be the linear convolution a*a (length 2·len(a)–1).
    The sub‑gradient is obtained by selecting either a single active index
    or the uniform average over all active indices.
    """
    n = a.size
    max_val = conv.max()
    # indices where the maximum is attained (within a tiny tolerance)
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]

    # choose a weight vector w of total mass 1
    if active.size == 0:                # sanity – should never happen
        w = np.zeros_like(conv)
        w[0] = 1.0
    else:
        # with 50 % probability pick a single active index,
        # otherwise spread uniformly over all of them
        if np.random.rand() < 0.5:
            w = np.zeros_like(conv)
            w[np.random.choice(active)] = 1.0
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size

    # convolution w * a_rev, where a_rev = a[::-1]
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * n - 1]
    # the relevant slice corresponds to the entries of the gradient
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Uses a projected Adam optimiser with exact sub‑gradients of the max‑convolution.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        # fallback: uniform vector of length 2400
        n = 2400
        S = np.sqrt(2.0 * n)
        a = np.full(n, S / n, dtype=np.float64)

    n = a.size
    S = np.sqrt(2.0 * n)          # ℓ₁‑norm that makes the score == max(conv)
    a = _project_to_simplex(a, S)

    nfft = _next_pow2(2 * n - 1)   # FFT length (power of two)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5 * S / n        # base learning rate (roughly the size of a component)

    # bookkeeping
    best_a = a.copy()
    # evaluate via the provided function (ensures all caps/filters are applied)
    best_score = evaluate_sequence(best_a.tolist())
    iteration = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --- current convolution & objective --------------------------------
        A = np.fft.rfft(a, n=nfft)
        conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
        cur_max = conv.max()                     # with Σa = S, score == cur_max
        cur_score = cur_max

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:06d}] New best max = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # early exit if target is reached
        if best_score < 1.503:
            print(f"Target reached after {iteration} iterations (score={best_score:.6f}).")
            break

        # --- sub‑gradient of max(conv) ------------------------------------
        grad = _grad_max_subgrad(a, conv, nfft)

        # --- Adam update ---------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        lr = base_lr / np.sqrt(iteration)       # slowly decaying stepsize
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- candidate & acceptance -----------------------------------------
        a_candidate = _project_to_simplex(a - step, S)

        # compute its true objective (fast via FFT)
        A_cand = np.fft.rfft(a_candidate, n=nfft)
        conv_cand = np.fft.irfft(A_cand * A_cand, n=nfft)[: 2 * n - 1]
        cand_score = conv_cand.max()

        if cand_score < cur_score:
            a = a_candidate
        else:
            # occasional random jitter to escape shallow plateaux
            if np.random.rand() < 0.02:
                jitter = np.random.randn(n) * (lr * 0.3)
                a = _project_to_simplex(a + jitter, S)

        # --- restart if stuck for a long time -------------------------------
        if no_improve >= 5000:
            # restart near the best solution with a small perturbation
            a = best_a + np.random.randn(n) * 0.01
            a = _project_to_simplex(a, S)
            no_improve = 0
            print(f"[{iteration:06d}] Restarted from best (noise added).")

        # occasional rotation – does not change the objective but may help
        if iteration % 8000 == 0:
            shift = np.random.randint(1, n)
            a = np.roll(a, shift)

    # ------------------------------------------------------------------
    # finalisation
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
