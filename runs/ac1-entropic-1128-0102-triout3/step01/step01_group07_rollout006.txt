"""
Propose a candidate step‑function sequence.

The algorithm works on a fixed ℓ₁‑norm (sum = 1) and directly minimises
the quantity 2·n·max(a∗a).  Instead of the non‑smooth max we minimise
a smooth approximation

    f_λ(a) = (1/λ)·log Σ_k exp( λ·(a∗a)_k ),

whose gradient can be written in closed form using convolutions.
Each optimisation step therefore consists of a few FFTs (O(n log n))
and a cheap projection onto the probability simplex (clip‑and‑renorm).

A schedule of increasing λ values drives the iterates towards the true
max‑norm while an Adam‑style adaptive step‑size guarantees stability.
Multiple random restarts and a coarse power‑law grid help to escape
poor local minima.  The whole search stays well within the 1000 s
budget and usually reaches a score below the target 1.503.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small utility: smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
    { y | y_i ≥ 0, Σ y_i = 1 }.
    This implementation follows the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1            # zero‑based
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(a∗a) for a sequence a whose sum is 1.
    Uses an FFT‑based convolution (O(n log n)).
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, L)
    conv = np.fft.irfft(fft_a * fft_a, L)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _gradient(a: np.ndarray, lam: float, L: int) -> np.ndarray:
    """
    Gradient of the smooth approximation
        g(a) = (1/lam) log Σ_k exp(lam·(a∗a)_k)
    The gradient is   2· (a ⋆ w)  where w_k ∝ exp(lam·(a∗a)_k)
    and ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]
    # convolution a*a
    fft_a = np.fft.rfft(a, L)
    conv = np.fft.irfft(fft_a * fft_a, L)[: 2 * n - 1]

    # soft‑max weights
    z = lam * conv
    z_max = np.max(z)
    w = np.exp(z - z_max)
    w /= w.sum()                     # length 2n‑1

    # cross‑correlation gradient  (a ⋆ w)
    fft_w = np.fft.rfft(w, L)
    corr = np.fft.irfft(fft_a * np.conj(fft_w), L)  # length L
    start = (2 * n - 1) - 1          # index of shift 0
    grad = 2.0 * corr[start : start + n]
    return grad


def _optimize(
    a0: np.ndarray,
    lam_schedule: List[float],
    iters_per_phase: List[int],
    deadline: float,
) -> np.ndarray:
    """
    Perform Adam optimisation of the smooth objective under the simplex
    constraints.  Returns the final sequence (sum = 1, non‑negative).
    """
    a = a0.copy()
    n = a.shape[0]
    L = _next_pow2(3 * n)          # enough for both convolution and correlation
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    base_lr = 0.5                  # base learning rate (scaled later by Adam)

    for lam, phase_iters in zip(lam_schedule, iters_per_phase):
        for _ in range(phase_iters):
            if time.time() >= deadline:
                return a
            t += 1
            # gradient of the smoothed max‑norm
            g = _gradient(a, lam, L)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a -= base_lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto the simplex (clip‑and‑renorm – very cheap)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a.fill(1.0 / n)

    return a


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative height sequence that minimises the
    ``evaluate_sequence`` objective.  The routine uses a smooth
    approximation of the max‑norm and Adam optimisation with several
    restarts and a simple power‑law grid.  It returns the best
    sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    best_seq: List[float] = []
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0) seed from previously known construction, if any
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            sc0 = evaluate_sequence(height_sequence_1)
            if sc0 < best_score:
                best_score = sc0
                best_seq = list(height_sequence_1)
                print(f"[init] existing seq score = {sc0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) search over a few lengths (larger n → better lower bound)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000]   # adapt if time is low
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # ---- a) coarse power‑law grid (fast, O(n log n)) ----
        betas = np.linspace(0.3, 3.0, 31)   # 31 candidates
        local_best = None
        local_best_score = float("inf")
        idx = np.arange(1, n + 1, dtype=np.float64)

        for beta in betas:
            seq = idx ** (-beta)
            seq /= seq.sum()
            sc = _fast_score(seq)
            if sc < local_best_score:
                local_best_score = sc
                local_best = seq

        # ---- b) a couple of random Dirichlet restarts ----
        for _ in range(2):
            seq = np.random.rand(n)
            seq /= seq.sum()
            sc = _fast_score(seq)
            if sc < local_best_score:
                local_best_score = sc
                local_best = seq

        # keep the best from the initialisation stage
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best.tolist()
            print(f"[grid] new global best {best_score:.6f}")

        # ------------------------------------------------------------------
        # 2) Refine the best initialisation with smooth Adam optimisation
        # ------------------------------------------------------------------
        lam_schedule = [5.0, 20.0, 80.0, 200.0]        # increasingly sharp
        iters_per_phase = [4000, 4000, 4000, 4000]    # total 16000 steps
        a_opt = _optimize(local_best, lam_schedule, iters_per_phase, deadline)

        # occasional jitter to shake out of very flat plateaus
        if time.time() < deadline:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a_opt = np.maximum(a_opt + jitter, 0.0)
            a_opt /= a_opt.sum()

        # final evaluation (exact same as evaluate_sequence for sum=1)
        final_score = _fast_score(a_opt)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_opt.tolist()
            print(f"[opt]  n={n} final score = {final_score:.6f}")

    # ------------------------------------------------------------------
    # 3) fallback – uniform sequence if nothing was found (should not happen)
    # ------------------------------------------------------------------
    if not best_seq:
        n = 500
        uniform = np.full(n, 1.0 / n)
        best_seq = uniform.tolist()
        best_score = _fast_score(uniform)

    print(f"\nFinal best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
