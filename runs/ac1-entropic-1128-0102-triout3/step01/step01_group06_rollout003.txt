Pseudo:

def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def subgradient_minimize(a_start, fft_len, time_limit):
    a = a_start.copy()
    best = a.copy()
    best_score = score(a, fft_len)
    start = time.time()
    iter = 0
    eta0 = 0.5
    while time.time() < time_limit:
        max_val, k_max = max_conv_and_index(a, fft_len)
        # compute gradient g: zero array
        g = np.zeros_like(a)
        low = max(0, k_max - (len(a) - 1))
        high = min(k_max, len(a) - 1)
        idx = np.arange(low, high+1)
        g[idx] = 2 * a[k_max - idx]
        # step size schedule
        eta = eta0 / np.sqrt(iter + 1)
        # exponentiated update
        a = a * np.exp(-eta * g)
        # renorm
        s = a.sum()
        if s == 0:
            a = best.copy()
            continue
        a = a / s
        # compute new score
        sc = score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best = a.copy()

        if np.random.rand() < 0.01:

            i1, i2 = np.random.choice(len(a), 2, replace=False)

            a[i1] = max(a[i1] + delta, 0)
            a[i2] = max(a[i2] - delta, 0)
            a = a / a.sum()
        iter += 1
    return best, best_score

        if a_new[i1] < 0:
            a_new[i1] = 0.0
        if a_new[i2] < 0:

def subgradient_minimize(a_start, fft_len, deadline):
    a = a_start.copy()
    n = len(a)
    best_a = a.copy()
    best_score = score(a, fft_len)
    eta0 = 0.5
    iter = 0
    while time.time() < deadline:
        max_val, k_max = max_conv_and_index(a, fft_len)
        # gradient
        g = np.zeros_like(a)
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        # Ensure indexing works: a[k_max - idx] yields array
        g[idx] = 2.0 * a[k_max - idx]
        eta = eta0 / np.sqrt(iter + 1)
        # exponentiated gradient step
        a = a * np.exp(-eta * g)
        # Avoid underflow: add epsilon
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        sc = score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[sgd] iter={iter} score={sc:.6f}")
        # occasional random perturb
        if np.random.rand() < 0.01:
            i1, i2 = np.random.choice(n, size=2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.01
            a[i1] += delta
            a[i2] -= delta
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        iter += 1
    return best_a, best_score

Search for a non‑negative step‑function that minimises the objective used in
`evaluate_sequence`.  The method consists of three stages:

1. **Analytic seeding** – try a variety of classic window families
   (power‑law, exponential, half‑Gaussian, Hann/Hamming/Blackman,
   Kaiser, DPSS) and keep the best normalised vector.
2. **Sub‑gradient / exponentiated‑gradient descent** – repeatedly
   compute the self‑convolution, locate the index where it attains its
   maximum and move the probability vector in the opposite direction
   of the sub‑gradient (the gradient of the active convolution term).
   The update is an exponentiated‑gradient step followed by a simplex
   projection (renormalisation).  The step size decays as 1/√iter.
3. **Short hill‑climbing polish** – random pairwise mass transfers
   with acceptance only when the score improves.

The whole search respects a user supplied time budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# Optional SciPy import for DPSS windows

    _SCIPY_AVAILABLE = False

def _normalize(arr: np.ndarray) -> np.ndarray:
    """Scale the vector so that its entries sum to one."""
    s = arr.sum()
    if s == 0.0:
        return arr
    return arr / s


def _max_conv_and_index(a: np.ndarray, fft_len: int):
    """
    Return (max_value, argmax_index) of the linear self‑convolution a*a.
    Uses a real FFT for speed.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(a*a) (input vector is assumed normalised)."""
    n = len(a)
    max_val, _ = _max_conv_and_index(a, fft_len)
    return 2.0 * n * max_val

# Simple hill‑climbing polish (random pairwise mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, fft_len: int, deadline: float):
    n = len(a)
    best_a = a.copy()
    best_score = _score(best_a, fft_len)

    while time.time() < deadline:
        i1, i2 = np.random.choice(n, size=2, replace=False)
        delta = (np.random.rand() - 0.5) * 0.02  # ±1 % of donor mass

        # try moving mass from i2 to i1 (sign of delta decides direction)
        a_new = best_a.copy()

        a_new[i2] -= delta

        if a_new[i1] < 0.0:
            a_new[i1] = 0.0
        if a_new[i2] < 0.0:
            a_new[i2] = 0.0

        a_new = _normalize(a_new)
        sc = _score(a_new, fft_len)

        if sc < best_score:
            best_score = sc
            best_a = a_new
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient / exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _exponentiated_gradient_descent(
    a_start: np.ndarray, fft_len: int, deadline: float
):
    """Minimise max‑convolution using a simple EG sub‑gradient method."""
    a = a_start.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _score(best_a, fft_len)

    eta0 = 0.5
    iteration = 0

    while time.time() < deadline:
        # current worst autocorrelation index
        _, k_max = _max_conv_and_index(a, fft_len)

        # sub‑gradient of (a*a)_{k_max}
        g = np.zeros_like(a)
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = 2.0 * a[k_max - idx]          # gradient = 2·a_{k‑max−i}

        # diminishing step size
        eta = eta0 / np.sqrt(iteration + 1.0)

        # exponentiated‑gradient update (keeps entries positive)
        a = a * np.exp(-eta * g)

        # avoid pathological under‑flow
        a = np.maximum(a, 1e-15)

        # re‑project onto the simplex (just renormalise – entries are already ≥0)
        a = a / a.sum()

        sc = _score(a, fft_len)
        if sc < best_score:

            a[i2] -= delta
            a = np.maximum(a, 0.0)
