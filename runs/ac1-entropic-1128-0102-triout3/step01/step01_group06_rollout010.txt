def propose_candidate(seed=123,budget_s=1000,**kwargs):
    set random seeds.
    determine n.
    # Build initial candidate:
    best_seq = init_sequence (use height_sequence_1 if present else param search)
    best_seq = best_seq / best_seq.sum()
    current_seq = best_seq.copy()
    current_score = compute_score(current_seq)
    best_score = current_score

    finish_time = time.time() + budget_s - 2.0
    total_time = finish_time - start_time

    # Annealing parameters:
    max_temp = best_score * 0.05 # e.g. ~0.08
    min_temp = best_score * 0.001 # ~0.0015
    max_frac = 0.2
    min_frac = 0.001

    while time.time() < finish_time:
        elapsed = time.time() - start_time
        t_frac = elapsed / total_time
        temp = max_temp * (1 - t_frac) + min_temp * t_frac
        step_frac = max_frac * (1 - t_frac) + min_frac * t_frac

        # select donor weighted by mass
        donor = np.random.choice(n, p=current_seq / current_seq.sum())
        # ensure donor != same index
        recv = np.random.choice(n)
        while recv == donor:
            recv = np.random.choice(n)

        transfer = step_frac * current_seq[donor] * np.random.rand()
        # apply
        new_seq = current_seq.copy()
        new_seq[donor] -= transfer
        new_seq[recv] += transfer
        # no renorm needed as sum unchanged
        # evaluate
        new_score = _score(new_seq, fft_len)

        delta = new_score - current_score
        if delta < 0 or np.random.rand() < math.exp(-delta / (temp + 1e-12)):
            # accept
            current_seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[SA] new best score={best_score:.6f}")

        # occasional random restart (global):
        if np.random.rand() < 0.001:
            # sample new param candidate
            init = random_param_candidate()
            # compute score
            init_score = _score(init, fft_len)
            if init_score < best_score:
                best_seq = init.copy()
                best_score = init_score
                current_seq = init.copy()
                current_score = init_score
                print("restart from param")

    return best_seq.tolist()

"""
Simulated‑annealing optimizer for the min‑max self‑convolution problem.
...
"""
import time, math, numpy as np

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return conv

def _score_from_conv(n: int, conv: np.ndarray) -> float:
    return 2.0 * n * float(conv.max())

def _random_powerlaw(n: int, rng: np.random.RandomState):
    alpha = rng.uniform(0.30, 0.85)
    c = rng.uniform(1.0, max(1.0, n/2.0))
    a = (np.arange(n, dtype=np.float64) + c) ** (-alpha)
    s = a.sum()
    if s == 0:
        return None
    return a / s

def _random_exponential(n: int, rng: np.random.RandomState):
    # λ chosen so that the last element is in [1e-6, 0.1] after normalisation
    # Choose λ uniformly in a useful range.
    max_lambda = 7.0 / max(1, n-1)   # exp(-λ*(n-1)) ≈ e^{-7} ~ 1e-3
    lam = rng.uniform(0.0, max_lambda)
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    if s == 0:
        return None
    return a / s

def _random_candidate(n: int, rng: np.random.RandomState):
    if rng.rand() < 0.5:
        return _random_powerlaw(n, rng)
    else:
        return _random_exponential(n, rng)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The search consists of a quick random
    exploration of a few parametric families to obtain a good seed,
    followed by a long simulated‑annealing phase that repeatedly moves
    a small amount of mass between two positions (sometimes targeting
    the indices that contribute most to the current maximal
    autocorrelation).  The algorithm respects all constraints of
    ``evaluate_sequence`` and stops before the time budget expires.
    """
    rng = np.random.RandomState(seed)
    start_time = time.time()
    finish_time = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Problem size (use reference construction if provided)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined globally by the runner
        ref = globals().get("height_sequence_1")
        n = int(len(ref))
    except Exception:
        n = 2000

    # FFT length – next power of two that can hold a full convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Obtain an initial candidate
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # If a reference sequence is present, evaluate it first
    ref_seq = globals().get("height_sequence_1")
    if isinstance(ref_seq, (list, tuple, np.ndarray)):
        a0 = np.array(ref_seq, dtype=np.float64)
        if a0.size == n:
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0:
                a0 /= s
                conv = _full_convolution(a0, fft_len)
                sc = _score_from_conv(n, conv)
                best_seq = a0.copy()
                best_score = sc
                print(f"[init] using provided reference → score={sc:.6f}")

    # Quick random exploration of the two families (≈30 s or 5000 trials)
    explore_deadline = start_time + min(30.0, budget_s * 0.05)
    trials = 0
    while time.time() < explore_deadline and trials < 10000:
        trials += 1
        cand = _random_candidate(n, rng)
        if cand is None:
            continue
        conv = _full_convolution(cand, fft_len)
        sc = _score_from_conv(n, conv)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[explore] trial {trials:5d} → score={sc:.6f}")

    # If nothing was found (unlikely) fall back to uniform.
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = _score_from_conv(n, _full_convolution(best_seq, fft_len))
        print("[fallback] uniform sequence")

    # ------------------------------------------------------------------
    # 3. Simulated annealing with occasional targeted moves
    # ------------------------------------------------------------------
    a = best_seq.copy()
    conv = _full_convolution(a, fft_len)
    cur_score = _score_from_conv(n, conv)

    # Temperature schedule (initial ≈5 % of current best, final ≈0.05 %)
    max_temp = best_score * 0.05
    min_temp = best_score * 0.0005

    # Step‑size schedule (initial transfers up to 30 % of donor mass)
    max_frac = 0.30
    min_frac = 0.001

    iteration = 0
    while time.time() < finish_time:
        iteration += 1
        # Compute elapsed fraction for schedules
        elapsed = time.time() - start_time
        total = finish_time - start_time
        frac = min(1.0, elapsed / total)

        temperature = (1 - frac) * max_temp + frac * min_temp
        step_frac = (1 - frac) * max_frac + frac * min_frac

        # ------------------------------------------------------------------
        # 3a. Choose a move – 70 % random, 30 % targeted (tunable)
        # ------------------------------------------------------------------
        if rng.rand() < 0.30:
            # TARGETED: pick the index that contributes most to the current peak
            peak_idx = int(conv.argmax())
            # contribution of each position i to this peak
            contrib = np.zeros(n, dtype=np.float64)
            # For i where 0 ≤ peak_idx-i < n we have a_i * a_{peak_idx-i}
            lo = max(0, peak_idx - (n - 1))
            hi = min(n - 1, peak_idx)
            i_idx = np.arange(lo, hi + 1, dtype=int)
            j_idx = peak_idx - i_idx
            contrib[i_idx] = a[i_idx] * a[j_idx]
            donor = int(i_idx[np.argmax(contrib[i_idx])])
            # Choose a receiver that currently has the smallest contribution
            receiver = int(i_idx[np.argmin(contrib[i_idx])])
            if donor == receiver:
                # fallback to random pick if degenerate
                donor = rng.choice(n, p=a / a.sum())
                receiver = rng.choice(n)
                while receiver == donor:
                    receiver = rng.choice(n)
        else:
            # RANDOM move: donor proportional to its mass, receiver uniform
            donor = rng.choice(n, p=a / a.sum())
            receiver = rng.integers(0, n)
            while receiver == donor:
                receiver = rng.integers(0, n)

        # Amount to move (positive, never exceeds donor mass)
        transfer = step_frac * a[donor] * rng.rand()
        if transfer <= 0.0:
            continue

        a_new = a.copy()
        a_new[donor] -= transfer
        a_new[receiver] += transfer

        # Numerical safety – keep non‑negative and renormalise
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s

        conv_new = _full_convolution(a_new, fft_len)
        new_score = _score_from_conv(n, conv_new)

        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis criterion
            if temperature > 0:
                prob = math.exp(-delta / temperature)
                if rng.rand() < prob:
                    accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[SA {iteration:6d}] improvement → score={new_score:.6f}")

        # ------------------------------------------------------------------
        # 3b. Rare random restart from a fresh parametric candidate
        # ------------------------------------------------------------------
        if rng.rand() < 0.001:
            cand = _random_candidate(n, rng)
            if cand is not None:
                conv_cand = _full_convolution(cand, fft_len)
                sc_cand = _score_from_conv(n, conv_cand)
                if sc_cand < best_score:
                    best_seq = cand.copy()
                    best_score = sc_cand
                    a = cand.copy()
                    conv = conv_cand
                    cur_score = sc_cand
                    print(f"[restart]  score={sc_cand:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n) / n
    return best_seq.tolist()

sequence) is

    score = 2·n·max_k (a * a)_k ,

i.e. minimise the largest entry of the aperiodic autocorrelation
(self‑convolution).  The algorithm works in two stages:

1. **Fast random exploration** of two simple parametric families
   (power‑law and exponential) to obtain a decent seed.  If a reference
   construction ``height_sequence_1`` is supplied it is used as an
   additional starting point.

2. **Long simulated‑annealing phase** (≈ budget – few seconds).  In each
   iteration a small amount of mass is transferred from one index to
   another.  30 % of the iterations use a *targeted* move that tries to
   reduce the current maximal autocorrelation peak: the donor index is
   the one contributing most to that peak, the receiver the one
   contributing least.  The remaining moves are random (donor chosen
   proportionally to its mass, receiver uniformly).  Acceptance follows
   a Metropolis rule with a temperature that linearly cools from about
   5 % of the current best score down to 0.05 % of that score.  A tiny
   fraction of the mass (initially up to 30 % of the donor) is moved,
   and the step size also decays.

All numbers stay non‑negative, the sequence is always re‑normalised to
sum = 1, and values never exceed the 0–1000 clipping range of
``evaluate_sequence``.  The routine respects a 1000 s time budget and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full discrete self‑convolution a*a using a real FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return conv


def _score_from_conv(n: int, conv: np.ndarray) -> float:
    """Score = 2·n·max(conv) for a normalised sequence."""
    return 2.0 * n * float(conv.max())


def _random_powerlaw(n: int, rng: np.random.RandomState) -> np.ndarray | None:
    """a_i ∝ (i + c)^{-α}   with random α∈[0.30,0.85] and c∈[1,n/2]."""
    alpha = rng.uniform(0.30, 0.85)
    c = rng.uniform(1.0, max(1.0, n / 2.0))
    a = (np.arange(n, dtype=np.float64) + c) ** (-alpha)
    s = a.sum()
    return a / s if s > 0 else None


def _random_exponential(n: int, rng: np.random.RandomState) -> np.ndarray | None:
    """a_i ∝ exp(-λ·i)   with λ chosen so that the tail is not too short."""
    # λ in [0, 7/(n‑1)] gives a_n‑1 ≈ e^{-7} ≈ 0.001 before normalisation.
    max_lambda = 7.0 / max(1, n - 1)
    lam = rng.uniform(0.0, max_lambda)
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s > 0 else None


def _random_candidate(n: int, rng: np.random.RandomState) -> np.ndarray | None:
    """Choose randomly between the two families."""
    if rng.rand() < 0.5:
        return _random_powerlaw(n, rng)
    return _random_exponential(n, rng)


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The function respects the time budget and
    prints progress messages.
    """
    rng = np.random.RandomState(seed)
    start_time = time.time()
    # leave a short safety margin
    finish_time = start_time + budget_s - 2.0
    total_time = finish_time - start_time

    # ------------------------------------------------------------------
    # 1. Problem size (prefer the length of the supplied reference)
    # ------------------------------------------------------------------
    ref = globals().get("height_sequence_1")
    if isinstance(ref, (list, tuple, np.ndarray)):
        n = len(ref)
    else:
        n = 2000

    # FFT length – next power of two that can contain the full convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Obtain an initial candidate (reference → random exploration)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # 2a) use the reference sequence if it exists
    if isinstance(ref, (list, tuple, np.ndarray)):
        a0 = np.asarray(ref, dtype=np.float64)
        if a0.shape[0] == n:
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0:
                a0 /= s
                conv = _full_convolution(a0, fft_len)
                sc = _score_from_conv(n, conv)
                best_seq = a0.copy()
                best_score = sc
                print(f"[init] reference → score={sc:.6f}")

    # 2b) quick random exploration of the two families (≈30 s or 10 k trials)
    explore_deadline = start_time + min(30.0, budget_s * 0.05)
    trials = 0
    while time.time() < explore_deadline and trials < 10000:
        trials += 1
        cand = _random_candidate(n, rng)
        if cand is None:
            continue
        conv = _full_convolution(cand, fft_len)
        sc = _score_from_conv(n, conv)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[explore] trial {trials:5d} → score={sc:.6f}")

    # 2c) fallback to uniform if nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = _score_from_conv(n, _full_convolution(best_seq, fft_len))
        print("[fallback] uniform sequence")

    # ------------------------------------------------------------------
    # 3. Simulated annealing with occasional targeted moves
    # ------------------------------------------------------------------
    a = best_seq.copy()
    conv = _full_convolution(a, fft_len)
    cur_score = _score_from_conv(n, conv)

    # Temperature schedule (initial ≈5 % of best, final ≈0.05 % of best)
    max_temp = best_score * 0.05
    min_temp = best_score * 0.0005

    # Step‑size schedule (initial up to 30 % of donor mass, ending at 0.1 %)
    max_frac = 0.30
    min_frac = 0.001

    iteration = 0
    while time.time() < finish_time:
        iteration += 1
        # fraction of elapsed time – controls cooling and step size
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / total_time)

        temperature = (1.0 - frac) * max_temp + frac * min_temp
        step_frac = (1.0 - frac) * max_frac + frac * min_frac

        # --------------------------------------------------------------
        # 3a) Choose a move: 30 % targeted, 70 % random
        # --------------------------------------------------------------
        if rng.rand() < 0.30:
            # TARGETED move – focus on the current maximal convolution entry
            peak_idx = int(conv.argmax())
            # indices that actually appear in this peak
            lo = max(0, peak_idx - (n - 1))
            hi = min(n - 1, peak_idx)
            i_idx = np.arange(lo, hi + 1, dtype=int)
            j_idx = peak_idx - i_idx
            contrib = a[i_idx] * a[j_idx]                         # contribution to the peak
            donor = int(i_idx[np.argmax(contrib)])
            receiver = int(i_idx[np.argmin(contrib)])
            if donor == receiver:                                 # degenerate – fall back to random
                donor = rng.choice(n, p=a / a.sum())
                receiver = rng.randint(0, n)
                while receiver == donor:
                    receiver = rng.randint(0, n)
        else:
            # RANDOM move – donor proportional to its mass, receiver uniform
            donor = rng.choice(n, p=a / a.sum())
            receiver = rng.randint(0, n)
            while receiver == donor:
                receiver = rng.randint(0, n)

        # Transfer a positive amount (never larger than the donor's mass)
        transfer = step_frac * a[donor] * rng.rand()
        if transfer <= 0.0:
            continue

        a_new = a.copy()
        a_new[donor] -= transfer
        a_new[receiver] += transfer
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s

        # --------------------------------------------------------------
        # 3b) Evaluate the candidate
        # --------------------------------------------------------------
        conv_new = _full_convolution(a_new, fft_len)
        new_score = _score_from_conv(n, conv_new)

        delta = new_score - cur_score
        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis acceptance
            if temperature > 0.0:
                if rng.rand() < math.exp(-delta / temperature):
                    accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[SA {iteration:6d}] improvement → score={new_score:.6f}")

        # --------------------------------------------------------------
        # 3c) Very occasional restart from a fresh parametric candidate
        # --------------------------------------------------------------
        if rng.rand() < 0.001:
            cand = _random_candidate(n, rng)
            if cand is not None:
                conv_cand = _full_convolution(cand, fft_len)
                sc_cand = _score_from_conv(n, conv_cand)
                if sc_cand < best_score:
                    best_seq = cand.copy()
                    best_score = sc_cand
                    a = cand.copy()
                    conv = conv_cand
                    cur_score = sc_cand
                    print(f"[restart] new candidate → score={sc_cand:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n) / n
    return best_seq.tolist()
