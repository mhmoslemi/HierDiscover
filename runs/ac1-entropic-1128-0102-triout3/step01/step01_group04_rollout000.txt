def solve_lp_max_sum(given_f, M):
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz
    n = len(given_f)
    c = -np.ones(n)   # minimize -sum(g)
    # Build Toeplitz matrix
    ccol = np.concatenate([given_f, np.zeros(n-1)])
    crow = np.concatenate([given_f[0:1], np.zeros(n-1)])
    A = toeplitz(ccol, crow)  # shape (2n-1, n)
    # Convert to sparse
    A_sparse = scipy.sparse.csr_matrix(A)
    # b vector
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # solve
    res = linprog(c, A_ub=A_sparse, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return np.maximum(res.x,0)

"""
LP‑based alternating optimisation for the step‑function problem.

The algorithm works by repeatedly:
  1. Starting from an analytic shape a_i = (i+1+δ)^p – (i+δ)^p,
     scaled so that Σa = √(2 n).
  2. Computing M = max(conv(a, a)).
  3. Solving a linear program that maximises Σg subject to
        a * g ≤ M   (convolution bound)   and   g ≥ 0.
  4. Rescaling the LP‑solution g to the required ℓ¹‑norm and mixing it
     with the current a (a ← (1‑t) a + t g).  The mixing factor t decays
     over the refinement steps.
  5. Repeating steps 2‑4 a few times, then restarting with new random
     parameters (length n, shift δ, exponent p).

All candidates are evaluated with the provided `evaluate_sequence`
function (the objective reduces to max(conv) because the ℓ¹‑norm is
fixed).  The best sequence found before the time budget expires is
returned.

No closures or lambdas are used; all helper functions are top‑level.
"""

import time
import math
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog


def _target_sum(n: int) -> float:
    """Normalization used by the reference code: Σa = √(2 n)."""
    return math.sqrt(2.0 * n)


def _analytic_power_sequence(n: int, delta: float = 0.0, power: float = 0.5) -> np.ndarray:
    """
    Analytic non‑negative shape:
        a_i = (i+1+δ)^p – (i+δ)^p,
    scaled to the required ℓ¹‑norm.
    """
    i = np.arange(n, dtype=float)
    base = np.power(i + 1.0 + delta, power) - np.power(i + delta, power)
    # safeguard against pathological parameters
    if base.sum() == 0.0:
        base = np.ones_like(base)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the self‑convolution of `seq`."""
    return np.max(np.convolve(seq, seq))


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the external evaluate_sequence."""
    # evaluate_sequence is assumed to be available in the global namespace
    return evaluate_sequence(seq.tolist())


def _solve_lp(f_vec: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise   Σ g_j
        subject to A·g ≤ M,   g ≥ 0,
    where A is the (2n‑1)×n convolution matrix of the current `f_vec`.
    Returns the optimal g (as a dense array) or None on failure.
    """
    n = f_vec.shape[0]

    # Construct the lower‑triangular Toeplitz matrix that encodes (f*g)[k].
    # First column: f followed by zeros, first row: f[0] then zeros.
    first_col = np.concatenate([f_vec, np.zeros(n - 1)])
    first_row = np.concatenate([f_vec[:1], np.zeros(n - 1)])
    A = toeplitz(first_col, first_row)          # shape (2n‑1, n)

    # Linear programme data
    c = -np.ones(n)                             # minimise –Σg  ⇔ maximise Σg
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # Using HiGHS (default for method='highs') – works with dense A.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None
    return np.maximum(res.x, 0.0)


def _lp_refine_step(f_vec: np.ndarray, target_sum: float, mix: float) -> tuple[np.ndarray, float]:
    """
    One refinement step:
      * compute M = max(conv(f, f))
      * solve the LP for g (maximise Σg under f*g ≤ M)
      * rescale g to the target ℓ¹‑norm
      * convex‑combine f and g with mixing factor `mix`
    Returns the new f and its max‑convolution value.
    """
    M = _conv_max(f_vec)
    g0 = _solve_lp(f_vec, M)
    if g0 is None:
        # failure – keep the current vector unchanged
        return f_vec, M

    sum_g0 = g0.sum()
    if sum_g0 <= 1e-12:
        return f_vec, M

    # Rescale g to the required ℓ¹‑norm.
    g = g0 * (target_sum / sum_g0)

    # Convex combination (still respects the ℓ¹‑norm).
    new_f = (1.0 - mix) * f_vec + mix * g
    # Re‑normalise to eliminate tiny numerical drift.
    new_f = new_f * (target_sum / new_f.sum())
    new_M = _conv_max(new_f)
    return new_f, new_M


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,

    """

            if sc < best_score:
                best_score = sc
                best_seq = prior_arr.copy()
                print(f"[baseline] score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter ranges for random restarts.
    # ------------------------------------------------------------------
    n_min, n_max = 800, 2500          # length of the sequence
    max_refine = 6                    # LP steps per restart (tuned for speed)

    while time.time() < deadline:
        # Randomly pick parameters for a fresh start.
        n = np.random.randint(n_min, n_max + 1)
        delta = np.random.rand() * 1.0               # δ ∈ [0, 1)

        f = _analytic_power_sequence(n, delta=delta, power=power)
        target = _target_sum(n)

        cur_score = _conv_max(f)      # equal to evaluate_sequence because Σa is fixed
        if cur_score < best_score:

        # ------------------------------------------------------------------
        # LP‑based refinement loop.
        # ------------------------------------------------------------------
        for step in range(max_refine):
            if time.time() >= deadline:
                break

            # Decaying mixing factor – larger at the beginning for faster change.
            mix = 0.45 * (0.85 ** step)

            f, new_score = _lp_refine_step(f, target, mix)
