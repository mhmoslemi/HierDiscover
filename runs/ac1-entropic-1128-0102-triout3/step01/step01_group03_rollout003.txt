"""
Propose a step‑function sequence that minimizes the evaluation function.

The algorithm follows a simple projected (sub)‑gradient descent on the
probability simplex.  Starting from the classic “half‑binomial’’ (the
coefficients of (1‑z)^‑½) we repeatedly:

1.  Compute the discrete self‑convolution a*a via a real FFT.
2.  Locate the index k where the convolution attains its maximum.
3.  The sub‑gradient of the maximum is g_i = 2·a_{k‑i} for all i that
    satisfy 0 ≤ k‑i < n (0 otherwise).  This is the exact gradient of the
    peak (the previous implementation used an incorrect approximation).
4.  Take a step a ← a − η·g, then project back onto the probability
    simplex {x ≥ 0, Σx = 1} using the Duchi et al. Euclidean projection.
5.  Accept the move if it improves the score; otherwise accept it with a
    Metropolis probability that slowly cools (simulated‑annealing).
6.  Occasionally add tiny random perturbations to escape shallow
    plateaus and, after a long stretch without improvement, restart from
    a random Dirichlet draw.

All operations are vectorised; the dominant cost per iteration is the
FFT‑based convolution (O(n log n)).  With n≈2500 the loop can perform
hundreds of thousands of iterations within the allotted time, typically
finding a lower bound than the naïve half‑binomial (~1.5709) and often
approaching the target 1.503.

The function returns the best sequence found (as a plain Python list of
floats).  The caller will evaluate it with the provided ``evaluate_sequence``
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of a_i = C(2i,i) / 4**i (half‑binomial series),
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries become equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Fast convolution x * x using a real FFT.
    Length of result is 2*len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The routine performs a projected sub‑gradient descent with simulated
    annealing, random perturbations and occasional restarts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # configurable parameters
    n = 2500                     # length of the sequence
    max_iter = 10_000_000        # safety upper bound
    eta_start = 0.12             # initial step size
    temp_start = 0.04            # initial annealing temperature
    decay_every = 2500           # parameter decay interval
    restart_threshold = 120_000  # iterations without improvement before restart
    noise_prob = 0.008           # probability of adding random noise each iteration
    noise_scale = 1e-4           # magnitude of random noise

    # ------------------------------------------------------------------
    # initialise from the half‑binomial series (good starting point)
    a = sqrt_series(n)           # already sums to 1
    conv = conv_fft(a)
    max_val = conv.max()
    cur_score = 2.0 * n * max_val
    best_a = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation loop
    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin
    eta = eta_start
    temperature = temp_start
    it = 0
    last_improve = 0

    # momentum variables (simple Nesterov style) – optional
    a_prev = a.copy()
    momentum = 0.0      # we keep it zero for now (can be tuned later)

    while time.time() < deadline and it < max_iter:
        # compute convolution and locate the peak index
        conv = conv_fft(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient of the max (exact)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        jdx = k - idx
        # g_i = 2 * a_{k-i}
        grad[idx] = 2.0 * a[jdx]

        # ---------------------------------------
        # optional Nesterov momentum (simple version)
        # y = a + momentum * (a - a_prev)
        # here we keep momentum == 0 (can be enabled later)
        y = a

            a_trial += noise

        # project back onto the simplex (sum=1, non‑negative)

            best_a = a_trial.copy()
            last_improve = it

        # acceptance rule
        if trial_score <= cur_score:
            a_prev = a
            a = a_trial
            cur_score = trial_score
        else:
            # Metropolis acceptance for occasional uphill moves
            delta = trial_score - cur_score
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                a_prev = a
                a = a_trial
                cur_score = trial_score
            # else reject – keep current a

        # decay step size and temperature periodically
        if (it + 1) % decay_every == 0:
            eta = max(eta * 0.93, 1e-5)
            temperature = max(temperature * 0.92, 1e-7)

        # restart if stuck for a long time

            last_improve = it

        it += 1

    # ------------------------------------------------------------------
    # final safety evaluation using the provided routine
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
