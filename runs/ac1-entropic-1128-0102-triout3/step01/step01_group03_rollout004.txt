   M = cur_conv.max()
   cur_score = 2.0 * n * M
   best_a = a.copy()

   print(f"[init] n={n}, initial score={cur_score:.6f}")

   it = 0

   while time.time() < deadline:

       T = np.empty((2*n-1, n), dtype=np.float64)
       T.fill(0.0)

       # Solve LP: maximize sum(g) subject to T @ g <= M, g >= 0
       c = -np.ones(n)  # minimize -sum(g)

"""
Propose a step‑function sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

using an iterative *conditional‑gradient* (Frank–Wolfe) scheme.
Starting from the classic half‑binomial (central‑binomial) vector we
solve, at each iteration, a linear programme

    maximise   sum(g)
    subject to  a * g ≤ M        (component‑wise)
                g ≥ 0

where `M` is the current maximal self‑convolution of `a`.
The LP finds a feasible direction `g` that can increase the total
mass while respecting the same peak bound `M`.  Mixing `a` with the
normalised `g` (convex combination) strictly reduces the peak,
and the process is repeated until the time budget expires or no
further improvement is observed.

The algorithm works entirely on the simplex (non‑negative, sum = 1),
so the evaluation function reduces to `2·n·max(conv)`.  It is very
fast for moderate lengths (≈ 2000–2500) and typically drives the
score well below the half‑binomial value ≈ 1.571.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First `n` coefficients of  a_i = C(2i,i) / 4**i  (the (1‑z)^‑1/2 series),
    normalised to sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    return np.fft.irfft(X * X, n=m)


def build_toeplitz(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n Toeplitz matrix `T` such that
        (T @ g) = f * g   (linear convolution)
    for any vector `g` of length n.
    """
    n = f.shape[0]
    L = 2 * n - 1
    T = np.zeros((L, n), dtype=np.float64)
    # each column i contains the full vector `f` shifted by `i` rows
    for i in range(n):
        T[i:i + n, i] = f
    return T


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    (Used only as a safety net after numerical mixing.)
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation function.  The routine runs a conditional‑gradient
    loop (linear‑programming step + convex mixing) until the time
    budget is exhausted.  The best sequence encountered is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------ configurable parameters --------------------
    n = kwargs.get("n", 2000)            # length of the sequence
    max_iter = kwargs.get("max_iter", 200)   # hard cap on LP steps
    mix_step = kwargs.get("mix_step", 0.5)   # default convex‑mix weight
    min_improve = 1e-6                     # stop if improvement tiny

    # --------------------------------------------------------------------
    # initialise from the half‑binomial series (already very good)
    a = sqrt_series(n)                     # sum(a) == 1
    conv_a = conv_fft(a)
    M = conv_a.max()                       # current peak
    cur_score = 2.0 * n * M
    best_a = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    deadline = time.time() + budget_s - 2.0   # small safety margin
    it = 0
    last_impr = np.inf

    while time.time() < deadline and it < max_iter:
        # ----- linear programming step ---------------------------------
        # Build the Toeplitz matrix for the constraint a * g ≤ M
        T = build_toeplitz(a)                     # shape (2n‑1, n)

        # linprog: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)         # objective
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]

        res = scipy.optimize.linprog(c, A_ub=T, b_ub=b_ub,
                                    bounds=bounds, method="highs")
        if not res.success:
            print(f"[iter {it}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:            # nothing useful found
            print("[iter] LP returns zero mass – terminating.")
            break

        # normalise g to stay on the simplex
        g_norm = g / sum_g

        # ----- mixing step ---------------------------------------------
        # Try the default mixing weight; if it does not improve, fall back
        # to a smaller weight (helps near the end of the search).
        for t in (mix_step, mix_step * 0.5, 0.25):
            a_candidate = (1.0 - t) * a + t * g_norm
            # guard against tiny negative values from floating error
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate /= a_candidate.sum()          # re‑project onto simplex

            conv_cand = conv_fft(a_candidate)
            M_cand = conv_cand.max()
            score_cand = 2.0 * n * M_cand

            if score_cand < cur_score - 1e-9:
                # accept improvement
                a = a_candidate
                M = M_cand
                cur_score = score_cand
                if score_cand < best_score - 1e-9:
                    best_score = score_cand
                    best_a = a_candidate.copy()
                    print(f"[iter {it}] new best score = {best_score:.6f}")
                break   # exit the inner t‑loop
        else:
            # No improvement from mixing – inject a very small random perturbation
            # to possibly escape a flat region.
            delta = (rng.random(n) - 0.5) * 1e-5
            a_pert = a + delta
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
            conv_pert = conv_fft(a_pert)
            M_pert = conv_pert.max()
            score_pert = 2.0 * n * M_pert
            if score_pert < cur_score:
                a = a_pert
                M = M_pert
                cur_score = score_pert
                if score_pert < best_score:
                    best_score = score_pert
                    best_a = a_pert.copy()
                    print(f"[iter {it}] new best (pert) score = {best_score:.6f}")

        # ----- convergence test -----------------------------------------
        improvement = best_score - cur_score
        if improvement < min_improve:
            # no appreciable progress – we can stop early
            print(f"[iter {it}] improvement {improvement:.2e} below tolerance – stop.")
            break

        it += 1

    # --------------------------------------------------------------------
    # final safety evaluation with the provided function
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
