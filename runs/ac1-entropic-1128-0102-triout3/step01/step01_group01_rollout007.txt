def propose_candidate(seed=42, budget_s=1000, **kwargs):

    import numpy as np, time, math, random
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2000
        S = math.sqrt(2*n)
        a = np.full(n, S/n)

    # Ensure sum = sqrt(2n)
    n = a.size
    S = math.sqrt(2.0 * n)
    a = a * (S / a.sum())

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    deadline = time.time() + budget_s - 0.5 # margin
    iteration = 0
    max_iters = 5000 # spec may limit

    # Parameter t for mixing step
    t = 0.2 # can adapt

    # We'll keep track of times we do LP
    while time.time() < deadline:
        # compute current convolution and max
        conv = np.convolve(a, a)
        cur_max = conv.max()
        cur_score = cur_max # because sum fixed
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration}] New best: {best_score:.6f}")

            # early exit
            if best_score < 1.503:
                print("Goal reached")
                break

        # Build convolution matrix for LP
        # Use sparse representation: each column j corresponds to f shifted by j
        rows = np.arange(n* n, dtype=np.int64) # not length; generate differently to avoid memory?

        # Actually we can generate using list then convert to csr.
        row_inds = []
        col_inds = []
        data_vals = []
        # Use vectorized loops: for shift j from 0..n-1
        for j in range(n):
            start = j
            end = j + n
            row_inds.append(np.arange(start, end))
            col_inds.append(np.full(n, j, dtype=np.int64))
            data_vals.append(a) # will broadcast
        # combine
        row_inds = np.concatenate(row_inds)
        col_inds = np.concatenate(col_inds)
        data_vals = np.concatenate(data_vals)

        A_ub = csr_matrix((data_vals, (row_inds, col_inds)), shape=(2*n-1, n))

        # Constraint RHS is cur_max
        b_ub = np.full(2*n-1, cur_max, dtype=np.float64)

        # Solve LP maximize sum of g (c) subject to A_ub @ g <= b_ub, g >=0
        c = -np.ones(n)  # negative for minimization to maximize sum
        # Use bounds (0, None)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if not res.success:
            # fallback: maybe gradient descent
            print("LP failed, fallback to gradient step")
        else:
            g = res.x
            # res.x may not sum to S, compute sum_g
            sum_g = g.sum()
            if sum_g > 1e-12:
                # scale g to have same L1 S
                g = g * (S / sum_g)
            else:
                g = np.full(n, S/n) # fallback uniform
            
            # mixture
            a = (1 - t) * a + t * g

        iteration += 1

"""
Propose a candidate sequence using an alternating convex‑linear programme
strategy. Starting from the current best construction (global variable
height_sequence_1) we repeatedly

1. compute the peak M of the self‑convolution,
2. solve a linear programme that maximises the ℓ₁‑norm of a partner
   sequence g under the linear constraints (f∗g)_k ≤ M,
3. mix the current sequence with the partner (a←(1‑t)a+tg) and project
   back onto the ℓ₁‑simplex (∑a=S, a_i≥0).

Because the constraints are linear in g the LP is cheap to solve and,
as proven in the editorial, each successful mixing step never increases
the peak.  The algorithm is essentially the one described in the
statement but implemented directly, with adaptive step size, restarts
and a simple line‑search for the mixing coefficient.

The function respects the 1000 s time limit and prints progress.
"""

import time
import numpy as np

# helper to project onto the simplex {x≥0, sum(x)=target}
def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT for speed."""
    n = a.size
    nfft = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Uses the alternating LP + mixing scheme described above.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 1. initialise the current point on the ℓ₁‑simplex
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
    else:
        # fall back to a moderately large uniform start
        n = 2000
        a = np.full(n, 1.0, dtype=np.float64)

    n = a.size
    S = np.sqrt(2.0 * n)               # target ℓ₁‑norm (the normalisation used
                                        # in the reference solution)
    a = a * (S / a.sum())              # project onto the simplex

    # bookkeeping
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial length {n}, initial score {best_score:.6f}")

    # parameters
    max_iters = 5000                     # just a safety bound
    t_init = 0.3                         # initial mixing coefficient
    t_min = 1e-4
    mix_decay = 0.99                     # slowly shrink the step size
    no_improve = 0
    max_no_improve = 500                # after this many iterations we restart

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        # ---------------------------------------------------------
        # 2. evaluate current peak
        # ---------------------------------------------------------
        conv_a = _convolve_self(a)
        cur_max = conv_a.max()
        cur_score = cur_max                # because we keep sum == S
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration:05d}] New best: {best_score:.6f} (max={cur_max:.6f})")
            no_improve = 0
            if best_score < 1.503:
                print("Target reached – stop early")
                break
        else:
            no_improve += 1

        # ---------------------------------------------------------
        # 3. build the linear programme A g ≤ cur_max
        #    where A is the (2n‑1)×n Toeplitz matrix generated by a.
        # ---------------------------------------------------------
        # Use a sparse matrix with a constant diagonal for each offset.
        from scipy.sparse import diags

        # each diagonal d (0 ≤ d < n) contains the constant a[d]
        # shape is (2n‑1, n). For every offset the diagonal length is n.
        offsets = np.arange(n, dtype=int)
        # stacked rows: each row of `data` corresponds to one diagonal
        data = np.vstack([np.full(n, a[d]) for d in range(n)])
        A_ub = diags(data, offsets, shape=(2 * n - 1, n), format='csr')

        b_ub = np.full(2 * n - 1, cur_max, dtype=np.float64)

        # ---------------------------------------------------------
        # 4. solve the LP: maximise 1ᵀ g  subject to A g ≤ cur_max, g ≥ 0,
        #    with a trivial upper bound to keep the problem bounded.
        # ---------------------------------------------------------
        from scipy.optimize import linprog

        c = -np.ones(n)                     # linprog minimises, so negate
        bounds = [(0.0, S) for _ in range(n)]   # we never need more than S per entry

        # Give the solver a short time‑limit per call (∼10 s) to stay within budget.
        # The "highs" method respects the `time_limit` option.
        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'time_limit': 10.0, 'presolve': True})

        if not res.success:
            # Something went wrong – fall back to a tiny random perturbation.
            print(f"[{iteration:05d}] LP failed ({res.message}), jittering.")
            jitter = np.random.randn(n) * (t_init * 0.1)
            a = _project_to_simplex(a + jitter, S)
            iteration += 1
            continue

        g = res.x

        # ---------------------------------------------------------
        # 5. mixing step – line search for the best coefficient t.
        # ---------------------------------------------------------
        t = t_init
        improved = False
        # pre‑compute mixed‑convolution components to avoid repeated FFTs:
        conv_g = _convolve_self(g)
        conv_ag = np.convolve(a, g)  # linear convolution a * g (length 2n‑1)

        # try a few decreasing t's until we see an improvement
        while t >= t_min:
            a_candidate = (1.0 - t) * a + t * g
            # project back onto the simplex to keep the ℓ₁‑norm exact
            a_candidate = _project_to_simplex(a_candidate, S)

            conv_cand = _convolve_self(a_candidate)
            cand_max = conv_cand.max()
            if cand_max < cur_max:
                a = a_candidate
                improved = True
                break
            # try a smaller step
            t *= 0.5

        if not improved:
            # no reduction – perform a tiny random jitter to escape possible plateaux
            jitter = np.random.randn(n) * (t_init * 0.05)
            a = _project_to_simplex(a + jitter, S)

        # ---------------------------------------------------------
        # 6. schedule updates
        # ---------------------------------------------------------
        t_init = max(t_init * mix_decay, t_min)   # slowly shrink the step size

        # occasional restart if we have been stuck for long
        if no_improve >= max_no_improve:
            print(f"[{iteration:05d}] Restart after {no_improve} iterations without improvement.")
            # random Dirichlet point on the simplex (α = 0.3 gives a fairly
            # spread distribution).  This gives a fresh starting region.
            alpha = np.full(n, 0.3)
            a = np.random.dirichlet(alpha) * S
            no_improve = 0

        iteration += 1

    # --------------------------------------------------------------
    # 7. final check – make sure the returned sequence respects the
    #    simplex constraint (tiny rounding errors are removed).
    # --------------------------------------------------------------
    final_seq = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()

"""
Alternating linear‑program / mixing search.

The algorithm follows the scheme sketched in the statement:
starting from the current best step‑function (global variable
`height_sequence_1`) we repeatedly

1. compute the current peak `M = max(a*a)`,
2. solve a linear programme that maximises the ℓ₁‑norm of a partner
   sequence `g` under the *linear* constraints `(a * g)_k ≤ M`,
3. mix the current sequence with the partner,
   `a ← (1‑t)·a + t·g`, re‑project onto the ℓ₁‑simplex (∑a = √(2n)),
   and keep the change only if it reduces the peak.

The LP is cheap (≈ 2·n constraints, n variables) and each successful
mixing step never increases the peak.  A simple line‑search for the
mixing coefficient `t` and occasional random restarts avoid stagnation.

The routine respects a 1000 s time limit, prints progress and finally
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = target}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the simplex with non‑negative entries and a given L1‑sum."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear convolution using FFT (fast for n≈2000)
# ----------------------------------------------------------------------
def _convolve_self(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (real valued)."""
    n = a.size
    nfft = 1 << ((2 * n - 2).bit_length())  # next power of two
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:

    Search for a non‑negative step function minimising `evaluate_sequence`.
    Uses the alternating LP + mixing scheme described in the module docstring.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # -------------------------------------------------------------

    n = a.size
    S = np.sqrt(2.0 * n)               # target ℓ₁‑norm (same normalisation as the baseline)
    a = a * (S / a.sum())               # ensure exact sum = S

    # bookkeeping
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial length {n}, initial score {best_score:.6f}")

    # algorithmic parameters
    max_iters = 5000
    t_init = 0.3                         # initial mixing weight
    t_min = 1e-4
    mix_decay = 0.99                     # slowly shrink the mixing coefficient
    no_improve = 0
    max_no_improve = 500                 # restart if stuck too long

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        # ---------------------------------------------------------
        # 1. current peak of the self‑convolution (the objective)
        # ---------------------------------------------------------
        conv_a = _convolve_self(a)
        cur_max = conv_a.max()
        cur_score = cur_max                # because ∑a = S ⇒ evaluate = cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

            no_improve = 0
            if best_score < 1.503:

                break
        else:
            no_improve += 1

        # ---------------------------------------------------------
        # 2. Build the linear programme A·g ≤ cur_max,
        #    where A is the Toeplitz matrix generated by a.
        #    For each offset d the corresponding diagonal has the constant a[d].
        # ---------------------------------------------------------
        from scipy.sparse import diags

        offsets = np.arange(n, dtype=int)                     # 0 … n‑1

        # ---------------------------------------------------------

        # ---------------------------------------------------------

            no_improve = 0

    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
