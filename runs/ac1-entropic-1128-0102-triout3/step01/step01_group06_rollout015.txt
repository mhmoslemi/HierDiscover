def param_to_seq(params):
    alpha = params[0]
    c = params[1]
    w0 = params[2]
    # K = (len(params)-3)//2
    K = (len(params)-3)//2
    i = np.arange(n)
    base = (i + c) ** (-alpha)
    g = w0 * np.ones(n)
    for k in range(K):
        w_cos = params[3 + 2*k]
        w_sin = params[4 + 2*k]
        angle = 2*np.pi*(k+1)*i / n
        g += w_cos * np.cos(angle) + w_sin * np.sin(angle)
    a = base * np.exp(g)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0/n) # fallback uniform
    a /= a_sum
    return a

import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    conv = cp.convolve(f, b)   # length 2n-1
    constraints = [conv <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    if b.value is None:
        return None
    return np.maximum(b.value, 0.0)

        conv = np.fft.irfft(A * A, n=fft_len)[:2*n-1]
        return float(conv.max())

        return 2.0 * n * max_conv(a)

        f = f / f.sum()

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """...
    """
    import time, numpy as np, math
    try:
        import scipy.optimize as opt
        import scipy.sparse as sp
    except Exception:
        opt = None
        sp = None

    np.random.seed(seed)
    start = time.time()
    end = start + budget_s - 1.0  # margin

    # Choose length (reasonable for LP)
    # Use reference length if available, otherwise fallback to 2000.
    ref_seq = globals().get('height_sequence_1')
    if ref_seq is not None:
        L_ref = len(ref_seq)
    else:
        L_ref = None

    # Limit n for LP solve to at most 2500 to keep memory manageable.
    n = min(2500, L_ref) if L_ref is not None else 2000
    if n < 500:
        n = max(500, n)

    print(f"[setup] using n={n}")

    # FFT length for convolution
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Helper functions
    def max_conv(a):
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
        return float(conv.max())

    def score(a):
        return 2.0 * n * max_conv(a)

    # Initialise f
    if ref_seq is not None:
        # Truncate or pad to length n
        seq = np.array(ref_seq, dtype=np.float64)
        if len(seq) >= n:
            f = seq[:n]
        else:
            pad = np.full(n - len(seq), 1e-12, dtype=np.float64)
            f = np.concatenate([seq, pad])
    else:
        i = np.arange(n, dtype=np.float64)
        alpha, c = 0.5, 1.0
        f = (i + c) ** (-alpha)
    # Normalise to unit sum (scaling does not affect objective)
    f_sum = f.sum()
    if f_sum == 0:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    best_seq = f.copy()
    best_sc = score(f)
    print(f"[init] score={best_sc:.6f}")

    if opt is not None and sp is not None:
        i_vec = np.arange(n)[:, None]               # shape (n,1)

        col_idx = np.repeat(np.arange(n), n)         # length n*n

        def solve_lp(f_vec, M):

            # Data for sparse matrix = tiled f_vec

    """Return max of the discrete self‑convolution a*a using a real FFT."""
    A = np.fft.rfft(a, n=fft_len)
