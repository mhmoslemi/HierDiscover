Define LP solve:

def _solve_lp(f, M):
    n = len(f)
    A = _build_conv_matrix(f)
    # Convolution constraints: A @ b <= M
    b_ub = np.full(2 * n - 1, M, dtype=float)
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n, dtype=float)
    bounds = [(0, None)] * n
    # Solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # If fails, fallback to zeros
        return np.zeros(n, dtype=float)
    b_opt = res.x
    # Clip small negatives due to numeric
    b_opt = np.maximum(b_opt, 0.0)
    return b_opt
```

Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The evaluation function returns exactly `max(conv(a,a))` when the
sequence is normalised to sum = √(2n).  The algorithm works as follows:

1.  Start from the analytically‑derived decreasing shape
        a_i ∝ √(i+1+δ) – √(i+δ)

2.  Repeatedly improve the shape by solving a linear programme:
        maximise   Σ b_j
        subject to   b_j ≥ 0                     (non‑negative)
                     (f * b)_k ≤ M    for all k (pointwise bound)
    where f is the current sequence and M = max(conv(f,f)).

    The solution b is then scaled to the required L¹‑norm and mixed into f
    (convex combination).  This step spreads the mass away from the
    peak of the autocorrelation and typically reduces M.

3.  After a few LP steps a cheap stochastic hill‑climbing refinement
    (single‑coordinate Gaussian perturbations with renormalisation) is
    performed.

4.  The whole process is repeated with many random restarts until the
    allotted time (~1000 s) is exhausted.  The best sequence seen is

If SciPy is not available the code falls back to the hill‑climbing
only (the previous state‑of‑the‑art).  All helper functions are top‑level

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the LP step)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:          # SciPy may be missing in the execution env
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Required L¹‑norm: √(2 n)."""
    return np.sqrt(2.0 * n)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """
    Analytic decreasing sequence a_i = √(i+1+δ) – √(i+δ),
    rescaled so that its L¹‑norm equals √(2 n).
    """
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _evaluate(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq.tolist())


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return np.max(np.convolve(seq, seq))


# ----------------------------------------------------------------------
# Linear‑programming helpers
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build matrix A (shape (2n‑1, n)) such that A @ b == conv(f, b).
    Each column j of A contains the vector f shifted by j positions.
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    # column j contributes f[0 … n‑1] to rows j … j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve
        maximise   Σ b_j
        subject to b ≥ 0,
                   conv(f, b) ≤ M   (pointwise)
    Returns the optimal b (may be all zeros if the LP fails).
    """
    if not SCIPY_AVAILABLE:
        return np.zeros_like(f)

    n = f.shape[0]
    A = _build_conv_matrix(f)                     # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=float)    # RHS of all constraints

    # Objective: maximise Σ b  →  minimise –Σ b
    c = -np.ones(n, dtype=float)

    # Simple finite upper bound for each variable to keep the LP bounded.
    # The worst possible contribution of any b_j to a convolution entry is
    # f_max * b_j, therefore b_j ≤ M / f_max is safe.
    f_max = np.max(f)
    if f_max <= 0:
        ub = M * 1e6
    else:
        ub = M / (f_max + 1e-12)
    bounds = [(0.0, ub) for _ in range(n)]

    res = linprog(c,
                  A_ub=A,
                  b_ub=b_ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30.0})
    if not res.success:
        # In rare cases the LP may be numerically problematic.
        return np.zeros_like(f)

    b_opt = np.maximum(res.x, 0.0)   # clip tiny negatives
    return b_opt


def _lp_improve(seq: np.ndarray,
                max_iters: int,
                t_initial: float,
                t_decay: float,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP step described above.
    Returns the best sequence encountered and its score.
    """
    seq = seq.copy()
    n = seq.shape[0]
    target = _target_sum(n)

    best_seq = seq.copy()
    best_score = _evaluate(seq)
    cur_score = best_score
    t = t_initial

    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        M = _conv_max(seq)
        b_opt = _solve_lp(seq, M)

        sum_b = b_opt.sum()
        if sum_b == 0.0:
            break

        # Scale to the required L¹‑norm
        b_scaled = b_opt * (target / sum_b)

        # Convex update
        new_seq = (1.0 - t) * seq + t * b_scaled
        new_seq = new_seq * (target / new_seq.sum())   # renormalise (numerical safety)

        new_score = _evaluate(new_seq)

        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()

        # Accept only if the move improves the current state
        if new_score < cur_score:
            seq = new_seq
            cur_score = new_score
        else:
            # No improvement – shrink the step size
            t *= 0.5

        t *= t_decay
        if t < 1e-5:      # step too tiny → stop early
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (single‑coordinate Gaussian moves)
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray,
                max_iter: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing: perturb one coordinate by a Gaussian step,
    renormalise to the target L¹‑norm, keep the change if the evaluation
    score improves.  Returns the best sequence and its score.
    """
    seq = seq.copy()
    n = seq.shape[0]
    target = _target_sum(n)

    best_seq = seq.copy()
    best_score = _evaluate(seq)

    # initial step size: a fraction of the mean entry
    step_sigma = max(1e-8, seq.mean() * 0.2)

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        idx = np.random.randint(n)
        delta = np.random.randn() * step_sigma

        # propose a new candidate
        cand = seq.copy()
        cand[idx] = max(0.0, cand[idx] + delta)
        cand = cand * (target / cand.sum())   # enforce the L¹‑norm

        cand_score = _evaluate(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand                 # accept the move
        # else: reject – keep `seq` unchanged

        # slowly decay the proposal width
        step_sigma *= 0.9995

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the external evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation metric.  The function respects the given time budget
    (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds free for the outer housekeeping
    deadline = start_time + max(0.0, budget_s - 5.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional: start from the globally supplied construction, if present
    # ------------------------------------------------------------------
    global_seq = globals().get('height_sequence_1')
    if isinstance(global_seq, (list, np.ndarray)):
        gs = np.array(global_seq, dtype=float)
        if gs.ndim == 1 and gs.size > 0:
            n0 = gs.size
            gs = gs * (_target_sum(n0) / gs.sum())
            score0 = _evaluate(gs)
            best_seq = gs.copy()
            best_score = score0
            print(f"[global] using supplied height_sequence_1 (n={n0}) score={score0:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop (random restarts)
    # ------------------------------------------------------------------
    possible_lengths = list(range(400, 2100, 200))   # 400, 600, …, 2000

    while time.time() < deadline:
        # pick a random length and a random delta for the analytic start
        n = int(np.random.choice(possible_lengths))
        delta = np.random.rand() * 0.8                # δ ∈ [0, 0.8)

        seq = _analytic_sqrt_sequence(n, delta=delta)

        # tiny epsilon to avoid exact zeros (helps the LP bounds)
        seq = np.maximum(seq, 1e-12)

        # --------------------------------------------------------------
        # 1) evaluate the analytic start
        # --------------------------------------------------------------
        init_score = _evaluate(seq)
        if init_score < best_score:
            best_score = init_score
            best_seq = seq.copy()
            print(f"[init] n={n:4d} δ={delta:.3f} score={init_score:.6f}")

        # --------------------------------------------------------------
        # 2) try the LP‑based improvement (if SciPy is available)
        # --------------------------------------------------------------
        if SCIPY_AVAILABLE and time.time() < deadline - 0.5:
            # give the LP phase at most ~0.2 s per restart
            lp_deadline = time.time() + 0.2
            seq_lp, score_lp = _lp_improve(seq,
                                            max_iters=15,
                                            t_initial=0.5,
                                            t_decay=0.95,
                                            deadline=lp_deadline)
            if score_lp < best_score:
                best_score = score_lp
                best_seq = seq_lp.copy()
                print(f"[lp ] n={n:4d} δ={delta:.3f} score={score_lp:.6f}")

            # ----------------------------------------------------------
            # 3) final stochastic hill‑climb for fine tuning
            # ----------------------------------------------------------
            if time.time() < deadline - 0.1:
                hc_deadline = time.time() + 0.15
                seq_hc, score_hc = _hill_climb(seq_lp, max_iter=2000, deadline=hc_deadline)
                if score_hc < best_score:
                    best_score = score_hc
                    best_seq = seq_hc.copy()
                    print(f"[hc ] n={n:4d} δ={delta:.3f} score={score_hc:.6f}")

        else:
            # SciPy missing – fall back to hill‑climbing only
            if time.time() < deadline - 0.1:
                hc_deadline = time.time() + 0.25
                seq_hc, score_hc = _hill_climb(seq, max_iter=3000, deadline=hc_deadline)
                if score_hc < best_score:
                    best_score = score_hc
                    best_seq = seq_hc.copy()
                    print(f"[hc ] n={n:4d} δ={delta:.3f} score={score_hc:.6f}")

        # early stop if we already beat a reasonable target
        if best_score < 1.52 and (time.time() - start_time) > 0.2 * budget_s:
            # keep searching a little longer but avoid wasting too much time
            pass

    # ------------------------------------------------------------------
    # Fallback in the extreme (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _analytic_sqrt_sequence(800, delta=0.0)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
