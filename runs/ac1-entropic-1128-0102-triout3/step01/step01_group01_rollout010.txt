"""
Adam‑based soft‑max gradient descent for minimizing the maximal entry of
a self‑convolution on the simplex.

The algorithm works with the natural probability parametrisation
    p_i = exp(z_i) / sum_j exp(z_j)      (p_i ≥ 0, Σ p_i = 1)

and minimises the smoothed objective
    τ·log Σ exp( (p*p)_k / τ )
whose gradient w.r.t. p is obtained analytically via FFT.
The gradient is then back‑propagated to the logits z and an Adam
optimizer is applied.  The temperature τ is annealed to zero so that
the true ℓ∞‑norm of the convolution is approached.

The search starts from a carefully chosen “arcsine‑type’’ distribution
p_i ∝ 1/√((i+½)(n‑½‑i)) which is known to flatten the self‑convolution.
Occasionally the optimiser is restarted from a fresh Dirichlet draw if
no improvement is observed for a long time.  The best sequence found
within the allotted time is returned (scaled to the required ℓ₁‑norm
√(2n) so it can be fed directly to `evaluate_sequence`).
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part)."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float, nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    Returns a vector of length len(a).
    """
    # soft‑max weights (stable)
    w = np.exp(conv / tau - np.max(conv / tau))
    w /= w.sum()
    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad

    deadline = time.time() + budget_s - 0.5  # small safety margin

        n = a0.size

    nfft = _next_pow2(2 * n - 1)             # length for FFT (power of two)

    # initial probability vector p (sum = 1)
    # -----------------------------------------------------------------
    i = np.arange(n, dtype=float)

    start = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))

    tau = 0.8
    tau_min = 0.001
    # we anticipate a few million iterations; decay is exponential
    total_iters_est = 2_000_000
    tau_decay = (tau_min / tau) ** (1.0 / total_iters_est)

    # bookkeeping
    iteration = 0
    no_improve = 0
    max_no_improve = 3000          # restart threshold
    best_p = p.copy()
    best_score = evaluate_sequence((best_p * S).tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # -----------------------------------------------------------------
    # main optimisation loop
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # ----- (a) current probabilities -----
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()   # guarantees Σp = 1 exactly

        # ----- (b) convolution and gradient (soft‑max) -----
        conv = _convolve_self_fft(p, nfft)          # p * p
        grad_p = _grad_softmax(p, conv, tau, nfft)  # d/dp τ·log Σexp(conv/τ)

        # ----- (c) back‑propagation to logits -----
        dot = np.dot(grad_p, p)                     # scalar ⟨grad, p⟩
        grad_z = p * (grad_p - dot)                 # ∂J/∂z_i

        # ----- (d) Adam update -----
        iteration += 1
        lr = 0.05 * (1.0 - iteration / total_iters_est)  # linear decay
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- (e) temperature annealing -----
        tau = max(tau * tau_decay, tau_min)

        # ----- (f) periodic evaluation of the true objective -----
        if iteration % 80 == 0:
            a_candidate = p * S
            cur_score = evaluate_sequence(a_candidate.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                no_improve = 0
                print(f"[{iteration:07d}] new best: {best_score:.6f}")
            else:
                no_improve += 1

            # early exit if we reach the target bound
            if best_score < 1.503:
                print(f"Target reached after {iteration} iterations.")
                break

            # occasional restart to escape stagnation
            if no_improve >= max_no_improve:
                # fresh random Dirichlet point (moderate concentration)
                alpha = np.full(n, 0.4)
                new_p = np.random.dirichlet(alpha)
                logits = np.log(np.maximum(new_p, eps))
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[{iteration:07d}] restart (stagnation)")

    # -----------------------------------------------------------------
    # final result (project back to the ℓ₁‑simplex for safety)
    # -----------------------------------------------------------------
    final_seq = (best_p * S).tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq
