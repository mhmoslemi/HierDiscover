"""Subgradient projected gradient descent for minimizing max self-convolution.

Starts from analytically motivated power-law initializations, then iteratively
reduces the peak of the linear convolution using a sub‑gradient step (the
gradient of the active convolution entry).  Each step is projected onto the
simplex to keep the coefficients non‑negative and normalised.  Occasional
random perturbations and hill‑climbing moves are added to escape local minima.
The method runs for the allotted time budget and returns the best sequence
found.
"""
import numpy as np
import time

def fast_score(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]  # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def build_initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Power‑law initialisation a_i = (i+offset)^{-p} (i starts at 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

def subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max_{k} (a*a) at the active index k."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    # j = k - i  => i = k - j
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    # gradient_i = 2 * a_{j}
    grad[mask] = 2.0 * a[j[mask]]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation value.

    Parameters
    ----------
    seed : int
        Random seed.
    budget_s : int
        Time budget in seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a small safety margin

    # --------------------------------------------------------------
    # 1.   Choose length.  A few thousand gives a good trade‑off.
    n = 3000

    # --------------------------------------------------------------
    # 2.   Scan a few analytically‑motivated initial families.
    best_seq = None
    best_val = np.inf
    # power‑law exponent candidates & small offsets
    exponents = [0.3, 0.4, 0.5, 0.6, 0.8]
    offsets = [0.0, 0.1, 0.5, 1.0, 2.0]

    for p in exponents:
        for off in offsets:
            seq = build_initial_sequence(n, p, off)
            val = fast_score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
    print(f"[init] best value after power‑law scan: {best_val:.6f}")

    # --------------------------------------------------------------
    # 3.   Optionally use the previously found construction if available.
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(height_sequence_1, dtype=np.float64)
            if prev.size >= n:
                seq = prev[:n].copy()
            else:
                # pad with zeros if shorter
                seq = np.zeros(n, dtype=np.float64)
                seq[:prev.size] = prev
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0:
                seq /= seq.sum()
                val = fast_score(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                print(f"[init] using height_sequence_1 gave value {val:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 4.   Main optimisation loop – subgradient descent with projection.
    a = best_seq.copy()
    best_seq = a.copy()
    best_val = fast_score(a)

    # Parameters for the sub‑gradient scheme
    eta0 = 0.3               # initial step size
    eta_decay = 1e-5         # decay factor per iteration
    random_move_prob = 0.03  # probability of a random hill‑climb move
    random_move_step = 0.2   # relative amount of mass transferred

    # Pre‑compute FFT size for the convolution (power of two)
    conv_size = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        # ---- 4.1 compute the self‑convolution and locate the maximal entry
        fft_a = np.fft.rfft(a, conv_size)
        conv = np.fft.irfft(fft_a * fft_a, conv_size)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # ---- 4.2 sub‑gradient step (projected)
        grad = subgradient(a, k_max)
        eta = eta0 / (1.0 + it * eta_decay)
        v = a - eta * grad
        a = proj_simplex(v)

        # ---- 4.3 occasional random perturbation (small hill‑climb)
        if np.random.rand() < random_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j and a[i] > 0:
                delta = random_move_step * a[i] * np.random.rand()
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = proj_simplex(cand)  # keep feasibility
                cand_val = fast_score(cand)
                if cand_val < best_val:
                    best_val, best_seq = cand_val, cand.copy()
                    a = cand  # accept the improvement
                else:
                    # tiny uphill acceptance to keep exploration
                    if np.random.rand() < 0.001:
                        a = cand

        # ---- 4.4 periodic reporting and record‑keeping
        if it % 5000 == 0:
            cur_val = fast_score(a)
            if cur_val < best_val:
                best_val, best_seq = cur_val, a.copy()
                print(f"[iter {it:6d}] new best {best_val:.6f}")
            else:
                print(f"[iter {it:6d}] current {cur_val:.6f} best {best_val:.6f}")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    # Return as a plain Python list – evaluate_sequence will perform its own checks.
    return best_seq.tolist()

"""
Sub‑gradient projected descent for the max‑self‑convolution problem.

The algorithm works as follows:

1.  Choose a reasonable length (n≈3000).  Initialise the sequence with several
    power‑law families a_i ∝ (i+offset)^{-p} and keep the best according to
    the fast evaluation routine.

2.  Optionally seed the start from the previously found global variable
    ``height_sequence_1`` (if present).

3.  Main optimisation loop (runs until the time budget expires):
       * Compute the linear self‑convolution a∗a via FFT.
       * Locate the index k where the convolution attains its maximum.
       * The sub‑gradient of the objective at this active index is
         g_i = 2·a_{k‑i} (zero when the index is out of range).
       * Perform a projected gradient step:
                 a ← Π_{simplex}(a − η·g)
         where Π_{simplex} projects onto the probability simplex
         (non‑negative entries that sum to 1).  The step size η decays
         slowly with the iteration count.
       * With a small probability we also try a random “hill‑climbing”
         move that transfers a fraction of mass between two coordinates;
         improvements are kept, and rare uphill moves are occasionally
         accepted to escape plateaus.
       * Every few thousand iterations we evaluate the true objective
         (using the fast version) and keep the best sequence seen so far.

4.  Return the best sequence as a plain Python list; the external
    ``evaluate_sequence`` function will perform the final checks.

The method mixes deterministic sub‑gradient descent (fast and directly
targeting the active convolution peak) with occasional stochastic moves,
which together give a good chance of beating the previous best bound
≈1.5758 within the 1000 s budget.

import numpy as np
import time

# Fast evaluation – identical to the provided `evaluate_sequence` but
# without the security checks.  Used internally for speed.
def fast_score(a: np.ndarray) -> float:
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum_i x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)          # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, p: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """
    Power‑law initialisation:
        a_i = (i + offset)^{-p},   i = 1,…,n
    The vector is normalised to have ℓ¹‑norm equal to 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.power(idx, p)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
def subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry with respect to `a`.
    For the active index k the gradient is
        g_i = 2·a_{k-i}   (if 0 ≤ k‑i < n, else 0).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad
