def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    # choose length n = 3401 (or bigger like 4000)
    n = 3401
    # build Catalan
    seq = catalytic(n)
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq)  # using provided evaluate for safety
    # precompute FFT length
    L = 1 << ((2*n - 1).bit_length())
    # compute conv via FFT function
    def compute_score(a):
        a = np.asarray(a, dtype=np.float64)
        # could clamp values to [0,1000] just in case
        a = np.clip(a, 0.0, 1000.0)
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        maxb = conv.max()
        sum_a = a.sum()
        if sum_a < 0.01:
            return np.inf
        return 2 * n * maxb / (sum_a * sum_a)
    # main loop
    iteration = 0
    while time.time() - start < budget_s * 0.95:
        iteration += 1
        # random index selection
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # ensure i < j for consistency

def compute_score(arr):

    maxb = conv.max()

    if sum_a < 0.01:
        return np.inf
    return 2 * n * maxb / (sum_a * sum_a)
