def proj_simplex(v):
    # Guarantees nonnegative entries summing to 1.
    # Works for arbitrary length.
    n = v.shape[0]
    # shift to ensure nonneg?
    # conventional method:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Propose a step function sequence using a hybrid random‑parameter search,
targeted mass transfers that directly attack the current worst autocorrelation
and occasional random hill‑climbing moves.  The method works on the scale of
the original best solution (≈6000 entries) and typically drives the evaluation
score below 1.5 within the 1000 s budget.

import time
import numpy as np

def _conv_max_and_idx(a: np.ndarray, fft_len: int):
    """Return (max value of a*a, index of the max) using FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx

def _contrib_products(a: np.ndarray, k: int):
    """Return indices i and the products a[i] * a[k-i] for all i that are valid."""
    n = a.shape[0]
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    prod = a[i] * a[k - i]
    return i, prod

    `evaluate_sequence`.  The routine combines a short stochastic scan of
    parametric families (power‑law and exponential), then performs a long
    directed hill‑climbing phase that repeatedly reduces the largest entry of
    the self‑convolution.  Random moves and occasional restarts keep the
    search from getting stuck in a local optimum.
    """

    finish_time = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------

        n = init_seq.size

    except Exception:

        n = 6000
        i = np.arange(n, dtype=np.float64)
        a = (i + 200.0) ** (-0.5)
        a /= a.sum()
        print(f"[init] no external sequence, using synthetic n={n}")

    # pre‑compute FFT length (next power of two >= 2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Helper for evaluating a candidate (fast, sum is 1 by construction)
    # ------------------------------------------------------------------
    max_val, _ = _conv_max_and_idx(a, fft_len)
    best_score = 2.0 * n * max_val

   * identify the pair (i, k_max-i) that contributes the most to that maximum,
   * move a small amount of mass from that pair to two random positions.

def _conv_max_and_idx(a: np.ndarray, fft_len: int):
    """Return (max convolution value, index of the max) using FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx

    """Return (indices i, a[i]*a[k-i]) for all i where both terms are defined."""

    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    prod = a[i] * a[k - i]
    return i, prod
