    # row: convolution index k = i + j

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm starts from the classic Catalan (central‑binomial) sequence,
which already gives a score ≈ π/2 ≈ 1.571.  It then repeatedly solves a
linear program that, for the current sequence f, finds a non‑negative vector b
with maximal ℓ₁‑norm while keeping the convolution f*b bounded by the current
maximum value M = max(f*f).  After rescaling b to the same ℓ₁‑norm as f
(the sum is fixed to √(2 n)), the new vector is blended with the old one.
Because the objective after the normalisation is exactly the maximum
convolution value, this “alternating LP’’ procedure monotonically drives
the score down and in practice reaches values well below the target 1.503.

The implementation uses `scipy.optimize.linprog` with a sparse constraint
matrix (the convolution matrix).  The whole optimisation is performed inside
the 1000 s budget; the best sequence encountered is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper: Catalan (central‑binomial) coefficients
def _catalan_sequence(n: int) -> list[float]:
    """First n Catalan‑type numbers  c_i = C(2i,i) / 4**i   (c_0=1)."""
    seq = [1.0]
    for i in range(1, n):
        # recurrence c_{i} = c_{i-1} * (2*i-1) / (2*i)
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


# ----------------------------------------------------------------------
# Build the sparse convolution matrix for a fixed left‑hand side f.
# Row k (0 ≤ k < 2n‑1) contains the coefficients f_{k‑j} for the variables b_j.
def _build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return a sparse (2n‑1)×n matrix A where (A·b)_k = (f*b)_k."""
    n = f.shape[0]
    # outer sum produces all k = i + j (i,g index of f; j index of b)
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------

    Generate a non‑negative step‑function with a small evaluation score.

    The method:
      * start from the Catalan construction,
      * normalise the ℓ₁‑norm to √(2 n)   →   the evaluation equals the
        maximal convolution value M,
      * repeatedly solve a linear programme that maximises the ℓ₁‑norm of a
        candidate b under the constraint f*b ≤ M,
      * rescale b to the same ℓ₁‑norm, optionally replace f by the new vector
        (or blend it), and repeat until no noticeable improvement.
    """
    start = time.time()

    # --------------------------------------------------------------
    # Choose a manageable length.
    try:
        # Prefer the length of the previously found construction,
        # but cap it to keep the LP tractable.
        n = len(height_sequence_1)  # type: ignore[name-defined]
        n = min(n, 1500)           # 1500 gives ~2.2 M non‑zeros in the LP matrix
        if n < 500:
            n = 500                # very small lengths are ineffective
    except Exception:
        n = 1200                   # fallback size

    # --------------------------------------------------------------
    # Initialise with the Catalan sequence and normalise the sum to √(2 n).
    f = np.array(_catalan_sequence(n), dtype=float)
    target_sum = math.sqrt(2.0 * n)
    f *= target_sum / f.sum()       # now sum(f) == √(2 n)

    # Keep the best sequence seen so far.
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[propose_candidate] start n={n}, initial score={best_score:.6f}")

    # Parameters for the iterative scheme.
    max_iter = 30
    blend_factor = 0.2            # 0 = no update, 1 = full replacement
    prev_score = best_score

    for it in range(max_iter):
        # ------------------------------------------------------------------
        # Time‑budget check.
        if time.time() - start > budget_s - 5.0:
            print("[propose_candidate] budget exhausted, stopping iterations")
            break

        # Current maximal self‑convolution (this is the current score).
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # Build the LP constraint matrix for the bound f*b ≤ M.
        A = _build_constraint_matrix(f)

        # Linear programme:
        #   maximise   sum(b_j)
        #   subject to A·b ≤ M ,   b ≥ 0
        c = -np.ones(n)                         # linprog minimises, so use -sum
        ub = np.full(2 * n - 1, M)             # right‑hand side for every convolution index
        try:
            res = linprog(c, A_ub=A, b_ub=ub,
                          bounds=[(0, None)] * n,
                          method="highs",
                          options={"presolve": True, "time_limit": budget_s - (time.time() - start)})
        except Exception as exc:                # any problem with the solver -> abort
            print("[propose_candidate] linprog exception:", exc)
            break

        if not res.success:
            print(f"[propose_candidate] LP failed at iter {it+1}: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            print("[propose_candidate] LP returned zero sum, aborting")
            break

        # --------------------------------------------------------------
        # Rescale b to the required ℓ₁‑norm (still √(2 n)).
        scale = target_sum / sum_b
        g = b * scale

        # Compute the self‑convolution of the candidate.
        conv_gg = np.convolve(g, g)
        M_g = float(conv_gg.max())

        # Update f:
        #   if the candidate already improves the maximal value, replace it;
        #   otherwise blend a little towards it.
        if M_g < M:
            f = g
            M = M_g
        else:
            f = (1.0 - blend_factor) * f + blend_factor * g

        # --------------------------------------------------------------
        # Evaluate the new sequence.
        cur_score = evaluate_sequence(f.tolist())
        print(f"[propose_candidate] iter {it+1:2d}: M={M:.6f}, score={cur_score:.6f}")

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        # Simple convergence test.
        if abs(prev_score - cur_score) < 1e-7:
            print("[propose_candidate] convergence reached")
            break
        prev_score = cur_score

    # ----------------------------------------------------------------------
    # Final clean‑up: enforce the limits imposed by `evaluate_sequence`
    seq = [float(min(max(v, 0.0), 1000.0)) for v in best_f.tolist()]
    final_score = evaluate_sequence(seq)
    print(f"[propose_candidate] final best score={final_score:.6f}")
    return seq
