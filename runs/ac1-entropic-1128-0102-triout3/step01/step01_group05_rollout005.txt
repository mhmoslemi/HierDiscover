"""
Hybrid random‑parameter search + projected sub‑gradient descent.

1.  Randomly sample a few simple parametric families 
    a_i ∝ (i+offset)^{-p} (p≈0.5) and keep the best‑scoring shape.
2.  Starting from that shape run a sub‑gradient method that at each
    iteration reduces the mass that contributes most to the current
    maximal convolution value.
    The step is projected back to the probability simplex (non‑negative,
    ℓ¹‑norm = 1) by zero‑clipping and renormalisation.
3.  The loop runs until the time budget (≈1000 s) is exhausted,
    continuously keeping the best sequence seen.

The algorithm is completely self‑contained and only uses NumPy.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Compute the evaluation value 2·n·max(conv(a,a)) / (sum(a))²
    using an FFT based convolution.  No security checks – used only
    internally.
    """
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, offset: float) -> np.ndarray:
    """Analytical start‑point a_i = 1/√(i+1+offset), ℓ¹‑normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.sqrt(idx)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def param_shape(n: int, p: float, offset: float) -> np.ndarray | None:
    """
    Return a normalised shape a_i ∝ (i+1+offset)^{-p}.
    Returns ``None`` if the parameters lead to an invalid (non‑positive) array.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    if np.any(idx <= 0):
        return None
    a = idx ** (-p)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain
    Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # 1) Choose a problem size.
    # Larger n does not improve the asymptotic π/2 bound but gives us
    # more degrees of freedom to beat it.  n≈3000 offers a good trade‑off.
    n = 3000

    # --------------------------------------------------------------
    # 2) Quick parametric sweep (≈10 % of the budget) to obtain a good
    #    starting point.  The analytic 1/√(·) shape is always tried.
    param_time = min(30.0, budget_s * 0.15)          # at most 30 s
    param_end = time.time() + param_time

    best_seq = build_initial_sequence(n, 0.5).copy()
    best_val = fast_score(best_seq)

    # keep the analytical shape as a baseline
    seq0 = build_initial_sequence(n, 0.0)
    val0 = fast_score(seq0)
    if val0 < best_val:
        best_seq, best_val = seq0, val0

    # random sampling of (p, offset) pairs
    while time.time() < param_end:
        p = np.random.uniform(0.3, 0.8)          # exponent close to 0.5
        off = np.random.uniform(-0.9, 2.0)      # offset to shift the kernel
        cand = param_shape(n, p, off)
        if cand is None:
            continue
        val = fast_score(cand)
        if val < best_val:
            best_seq, best_val = cand, val
        # early exit if we already beat a reasonable target
        if best_val < 1.53:
            break

    print(f"[init] best value after param sweep: {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Projected sub‑gradient descent on the peak of the self‑convolution.
    current = best_seq.copy()
    # enforce exact ℓ¹‑norm = 1 (helps numerical stability)
    current = current / current.sum()
    current_val = fast_score(current)

    step = 0.02          # initial step size for the gradient move
    it = 0
    while time.time() < deadline:
        # --- convolution and location of the current maximum -----------------
        size = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(current, size)
        conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
        max_conv = conv.max()
        p_idx = int(conv.argmax())          # index where the peak occurs

        # --- sub‑gradient of the peak (2·a_{p-i}) ---------------------------
        grad = np.zeros_like(current)
        lo = max(0, p_idx - (n - 1))
        hi = min(p_idx, n - 1)
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = p_idx - j
        grad[i] = 2.0 * current[j]

        # --- take a step and project back onto the simplex -----------------
        cand = current - step * grad
        cand = np.where(cand > 0.0, cand, 0.0)   # clip negatives to zero
        s = cand.sum()
        if s == 0.0:               # degenerate – reduce step and continue
            step *= 0.5
            continue
        cand /= s

        cand_val = fast_score(cand)

        # --- accept improvements; occasionally accept uphill moves ----------
        if cand_val < current_val:
            current, current_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_seq = cand_val, cand.copy()
        else:
            if np.random.rand() < 0.001:   # tiny chance to escape plateaus
                current, current_val = cand, cand_val

        # --- adapt the step size -------------------------------------------
        if it % 5000 == 0 and it > 0:
            step = max(step * 0.9, 1e-5)

        # --- progress report ------------------------------------------------
        if it % 2000 == 0:
            print(f"[iter {it:6d}] best={best_val:.6f}, cur={current_val:.6f}, step={step:.6e}")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    return best_seq.tolist()
