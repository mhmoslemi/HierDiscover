"""
Exponential‐decreasing step function.

The evaluation function equals the maximal entry of the self‑convolution
because the L¹‑norm is fixed to √(2·n).  For a geometric sequence

    a_i = c·r^i ,   r = exp(‑α/n),

the convolution values can be written analytically and the largest one is
approximately

    max conv ≈ 2·α²·e^{‑α} / (1‑e^{‑α})²          (central peak)
    a₀²   = 2·α² / (n·(1‑e^{‑α})²)                (edge peak)

The two peaks are equal when α ≈ ln n, yielding a value about
2·(ln n)²/n, which quickly drops below 0.1 for n≈2 000.  The algorithm
therefore tries a few lengths n and the corresponding α≈ln n (plus a
small neighbourhood) and returns the best sequence found.
"""

import time
import numpy as np

def _exp_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Return a length‑n non‑negative geometric sequence
        a_i = c·r**i ,   r = exp(‑alpha/n),

    scaled so that the L¹‑norm equals √(2·n).
    """
    # geometric ratio
    r = np.exp(-alpha / n)
    # r**n = exp(-alpha)
    r_n = np.exp(-alpha)  # more stable than r**n for large n
    # target L1 norm
    target = np.sqrt(2.0 * n)
    # scaling factor c so that sum_i a_i = target
    # sum = c·(1‑rⁿ)/(1‑r)
    c = target * (1.0 - r) / (1.0 - r_n)
    # build the sequence
    idx = np.arange(n, dtype=np.float64)
    seq = c * (r ** idx)
    return seq


def _evaluate(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence (expects a Python list)."""
    return evaluate_sequence(seq.tolist())


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a low‑score step function using the exponential family.
    Several lengths and alphas are tried; the best sequence found before the
    time budget expires is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the outer loop
    deadline = start + max(0.0, budget_s - 2.0)

    best_seq = None
    best_score = float("inf")

    # candidate lengths (small to moderate – fast to evaluate)
    lengths = [200, 500, 1000, 2000, 5000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # optimal α is roughly ln n; we explore a small neighbourhood
        alphas = [np.log(n) + delta for delta in (-2.0, -1.0, 0.0, 1.0, 2.0)]
        # ensure α stays positive
        alphas = [max(0.1, a) for a in alphas]

        for alpha in alphas:
            if time.time() >= deadline:
                break

            seq = _exp_sequence(n, alpha)
            score = _evaluate(seq)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[improved] n={n:4d} α={alpha:6.3f} score={score:.6f}")

    # As a safety net, fall back to a very simple uniform sequence.
    if best_seq is None:
        n = 200
        uniform = np.full(n, np.sqrt(2.0 * n) / n)
        best_seq = uniform
        print("[fallback] using uniform sequence")

    # Convert to plain Python list for the external evaluator.
    return best_seq.tolist()
