"""
Hybrid LP‑+‑subgradient optimizer for the step‑function attack.

The score (with the sum of the coefficients fixed to 1) is
    score = 2·n·max(conv(a, a))

We start from a good known construction (`height_sequence_1` if it exists),
or from a random simplex point.  Two complementary phases are run:

1. **Linear‑programming phase**  (inspired by the paper’s “active‑set” step)
   For the current f we solve
        max  Σ b_j
        s.t. f * b ≤ M   (M = max(conv(f,f))) ,   b ≥ 0
   This yields a feasible distribution g that pushes many coefficients
   up while keeping the convolution bound.  A convex combination
        f ← (1‑t)·f + t·g
   with a modest step size t reduces the maximal auto‑convolution.
   The LP is built with a sparse Toeplitz matrix (size (2n‑1)×n) and
   solved by SciPy’s HiGHS solver.

2. **Projected sub‑gradient phase** – a fast refinement.
   On the probability simplex we repeatedly move opposite to the
   sub‑gradient of the active convolution entry, using a diminishing
   step and Euclidean projection onto the simplex.

Both phases keep the best sequence seen (lowest score) and stop
as soon as the allotted time (≈1000 s) is exhausted.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}

    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries non‑positive – return the uniform point.
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """
    Return a list of non‑negative heights whose evaluation score is as low
    as possible within the given time budget.

    The algorithm mixes a linear‑programming step (to obtain a global
    direction that respects the current max‑convolution bound) with a
    projected sub‑gradient descent (to finely smooth the worst site).
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin to avoid hitting the hard limit
    deadline = start_time + budget_s - 2.0

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    # Choose the problem size.
    # If a previously found construction is available, reuse its length.
    # Otherwise fall back to a modest default.
    # ------------------------------------------------------------------
    base_seq = globals().get('height_sequence_1')
    if isinstance(base_seq, (list, tuple)) and len(base_seq) > 0:
        n = len(base_seq)
        init_vec = np.array(base_seq, dtype=float)
    else:
        # a reasonable default size – larger sizes give a lower theoretical bound
        n = 1200
        init_vec = np.random.rand(n).astype(float)

    # Normalise to the probability simplex (sum = 1)
    f = init_vec / init_vec.sum()

    # ------------------------------------------------------------------
    # Helper: fast convolution (full, non‑circular) using FFT.
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len).bit_length())   # next power of two ≥ conv_len

    def max_convolution(u: np.ndarray) -> float:
        """Return max(conv(u,u)) for a vector with sum(u)=1."""
        Uf = np.fft.rfft(u, n=fft_len)
        conv_full = np.fft.irfft(Uf * Uf, n=fft_len)[:conv_len]
        return float(conv_full.max())

    # ------------------------------------------------------------------
    # Record the initial candidate.
    # ------------------------------------------------------------------
    M = max_convolution(f)
    cur_score = 2.0 * n * M
    best_score = cur_score
    best_seq = f.copy()
    print(f"[{time.time() - start_time:6.2f}s] initial n={n}, score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Linear‑programming phase
    # ------------------------------------------------------------------
    # Pre‑compute sparse matrix index pattern (independent of the current f)
    i = np.arange(n)
    j = np.arange(n)
    rows = (i[:, None] + j[None, :]).ravel()          # shape (n*n,)
    cols = np.broadcast_to(j, (n, n)).ravel()        # shape (n*n,)
    # shape of the convolution matrix
    A_shape = (conv_len, n)

    # Parameters for the LP loop
    t = 0.2                     # starting convex‑combination weight
    max_lp_iters = 300          # enough iterations for a noticeable drop
    lp_iter = 0

    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1

        # Current bound M (max of auto‑convolution)
        M = max_convolution(f)

        # Build the constraint matrix A with the *current* f as coefficients.
        # A @ b gives the linear convolution f * b.
        data = np.tile(f, n)                     # size n*n
        A = csr_matrix((data, (rows, cols)), shape=A_shape)

        # Linear program: maximise Σ b_j  ⇔ minimise -Σ b_j
        c = -np.ones(n)                          # objective
        ub = np.full(conv_len, M)                # RHS of A b ≤ M

        res = linprog(c, A_ub=A, b_ub=ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            print(f"[{time.time() - start_time:6.2f}s] LP iter {lp_iter} failed: {res.message}")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            # No positive direction can be taken.
            break

        # Normalise to keep sum = 1
        g = b_opt / sum_b

        # Convex combination with step size t
        f_candidate = (1.0 - t) * f + t * g
        f_candidate /= f_candidate.sum()   # re‑project onto simplex (numerical safety)

        # Evaluate the candidate
        M_new = max_convolution(f_candidate)
        score_new = 2.0 * n * M_new

        if score_new < best_score:
            best_score = score_new
            best_seq = f_candidate.copy()
            print(f"[{time.time() - start_time:6.2f}s] LP improve: iter={lp_iter}, t={t:.4f}, score={score_new:.6f}")

        # Accept the move if it actually improves the max‑convolution.
        if M_new < M:
            f = f_candidate
            # slowly increase the step size to accelerate convergence,
            # but keep it bounded.
            t = min(t * 1.05, 0.5)
        else:
            # Otherwise shrink the step and try again.
            t *= 0.5
            if t < 1e-5:
                break

    # ------------------------------------------------------------------
    # 2️⃣  Projected sub‑gradient refinement (uses the same simplex constraint)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:   # need a few seconds to make it worthwhile
        a = f.copy()                     # start from the last LP result
        step0 = 0.6                      # initial step magnitude
        sub_it = 0

        while time.time() < deadline:
            sub_it += 1

            # Full auto‑convolution via FFT
            Af = np.fft.rfft(a, n=fft_len)
            conv_full = np.fft.irfft(Af * Af, n=fft_len)[:conv_len]
            k_max = int(np.argmax(conv_full))
            max_val = conv_full[k_max]
            score = 2.0 * n * max_val

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] SG improve: iter={sub_it}, score={score:.6f}")

            # Sub‑gradient of the active quadratic term (2·a_{k‑i})
            g_sub = np.zeros_like(a)
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            i_idx = np.arange(low, high + 1)
            j_idx = k_max - i_idx
            g_sub[i_idx] = 2.0 * a[j_idx]

            # Diminishing step size and projection onto the simplex
            step = step0 / math.sqrt(sub_it)
            a = _project_simplex(a - step * g_sub)

    # ------------------------------------------------------------------
    # Defensive fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Convert to plain Python list (the evaluator accepts any Sequence[float])
    return [float(x) for x in best_seq]
