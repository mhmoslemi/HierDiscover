"""
propose_candidate – improve the Catalan (π/2) bound by alternating linear
programming and convex mixing.

The evaluation of a sequence a of length n is

    E(a) = 2·n·max_k (a∗a)[k] / (∑ a_i)² .

If we normalise a so that   ∑ a_i = √(2 n)   then the denominator equals 2 n
and the score is exactly the maximal convolution value M = max_k (a∗a)[k].

Starting from the Catalan sequence (which gives M = π/2 ≈ 1.571) we
repeatedly solve a linear programme

    maximise   1ᵀ g
    subject to  f∗g ≤ M,   g ≥ 0 ,

where f is the current sequence and M = max(f∗f).  Because f is fixed,
the constraints are linear in g.  The LP pushes mass into indices where
f is small, thereby increasing the L¹‑norm while keeping the convolution
with f bounded by M.

The solution g₀ is rescaled to have the same L¹‑norm as f,
and we replace f by a convex combination
    f ← (1‑t)·f + t·g   (t≈0.5).
After each mix the maximal convolution M strictly decreases
(the “upper bound” improves).  Repeating a dozen times brings M below
the target 1.503.

If the optimisation libraries are not available the function falls back
to the plain Catalan construction.

All numbers are non‑negative and well inside the [0, 1000] limits required
by `evaluate_sequence`.
"""

from __future__ import annotations
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Catalan (central‑binomial) coefficients:  c_i = C(2i,i)/4**i
def _catalan_sequence(n: int) -> list[float]:
    seq = [1.0]                     # c₀
    for i in range(1, n):
        # c_i = c_{i‑1} * (2*i‑1) / (2*i)
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


# ----------------------------------------------------------------------
# Pre‑compute the sparsity pattern of the convolution matrix:
#   (f∗g)[k] = Σ_i f[i]·g[k‑i]  (0≤i,n‑1 ; 0≤k‑i,n‑1)
def _convolution_pattern(n: int):
    """Return (row_idx, col_idx, i_idx) for the matrix representing f·g ≤ M."""
    rows = []
    cols = []
    i_vals = []
    K = 2 * n - 1
    for k in range(K):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)          # row of the constraint = k
            cols.append(k - i)      # column (variable index) = j = k‑i
            i_vals.append(i)        # the coefficient is f[i]
    return (
        np.array(rows, dtype=np.int32),
        np.array(cols, dtype=np.int32),
        np.array(i_vals, dtype=np.int32),
    )


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function whose evaluation score is (hopefully) < 1.503.
    The function runs a few LP‑based improvement steps starting from the
    Catalan construction.  If SciPy is missing it returns the Catalan sequence.
    """
    start_time = time.time()

    # ------------------------------------------------------------------
    # Try to import the optimisation tools; fall back to Catalan if unavailable.
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp
    except Exception:
        # No SciPy – use the simple Catalan construction.
        n = 3401 if "height_sequence_1" in globals() else 2000
        seq = _catalan_sequence(n)
        seq = [float(v) for v in seq]
        print("[fallback] returning Catalan sequence, n =", n)
        return seq

    # ------------------------------------------------------------------
    # Choose a length.  Use the previous best length if it exists, otherwise cap at 2500
    # to keep the LP size reasonable.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        n = max(200, min(n, 2500))
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Initialise with the Catalan sequence, normalised so that Σa_i = √(2n).
    base = _catalan_sequence(n)
    sum_target = math.sqrt(2.0 * n)          # desired L¹‑norm
    base_sum = sum(base)
    f = np.array(base, dtype=np.float64) * (sum_target / base_sum)

    # Pre‑compute the sparsity pattern of the convolution operator.
    row_idx, col_idx, i_idx = _convolution_pattern(n)
    K = 2 * n - 1

    # ------------------------------------------------------------------
    # Main improvement loop.
    max_iters = 12
    t = 0.5                     # mixing parameter (0 < t ≤ 1)
    prev_M = None

    for it in range(max_iters):
        # 1. current maximal self‑convolution value
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # stop if no meaningful improvement
        if prev_M is not None and M >= prev_M - 1e-6:
            print(f"[iter {it}] no improvement, M = {M:.6f}")
            break
        print(f"[iter {it}] M = {M:.6f}")
        prev_M = M

        # 2. Build the linear constraints  f ∗ g ≤ M .
        data = f[i_idx]                     # coefficients specific to current f
        A_ub = sp.csr_matrix((data, (row_idx, col_idx)), shape=(K, n))
        b_ub = np.full(K, M, dtype=np.float64)

        # 3. Linear programme: maximise Σ g_i  (i.e. minimise –Σ g_i)
        c_obj = np.full(n, -1.0, dtype=np.float64)   # minimise –sum(g)
        bounds = [(0, None)] * n                     # g_i ≥ 0

        # Solve – the HiGHS interior‑point solver is fast for this size.
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs')

        if not res.success:
            print("[lp] failure:", res.message)
            break

        g0 = res.x
        sum_g0 = float(g0.sum())
        if sum_g0 < 1e-9:
            print("[lp] trivial solution, aborting")
            break

        # 4. Rescale g₀ to the same L¹‑norm as f ( = √(2n) )
        scale_g = sum_target / sum_g0
        g = g0 * scale_g

        # 5. Convex combination – this reduces the maximal convolution.
        f = (1.0 - t) * f + t * g
        f = np.maximum(f, 0.0)  # safeguard against tiny negatives

        # 6. Check remaining budget.
        if time.time() - start_time > budget_s - 0.5:
            print("[budget] out of time after iteration", it)
            break

    # ------------------------------------------------------------------
    # Convert to plain Python list, enforce the evaluate_sequence limits.
    final_seq = [float(min(max(v, 0.0), 1000.0)) for v in f.tolist()]

    # Log the final score (use the provided evaluator if it works).
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] length = {len(final_seq)}, score = {final_score:.6f}")
    except Exception as e:
        print("[result] evaluation error:", e)

    return final_seq
