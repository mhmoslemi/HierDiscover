def project_simplex(v):
    """Project onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑guided stochastic search for a low‑value step function.

The objective (after fixing the ℓ¹‑norm to 1) is:
    F(a) = 2 n · max(conv(a,a))

where conv(a,a) is the linear convolution of the non‑negative vector a.
The search keeps the vector on the probability simplex (non‑negative,
sum = 1).  At each iteration we

1. compute the full convolution via an FFT,
2. form a smooth approximation of the max using
      w_k ∝ exp(α·(conv_k – max_conv)),
   with α slowly increased during the run,
3. obtain the (sub‑)gradient of this smooth max:
      ∇F ≈ 2·corr(w, a)               (corr = cross‑correlation),
4. move a small amount of mass from the index with the largest
   gradient to the index with the smallest gradient,
5. re‑project onto the simplex and keep the move if it improves the
   true objective (computed with the fast_score routine).

Occasional random perturbations and a tiny uphill‑acceptance probability
help the search escape plateaus.  The algorithm is purely NumPy‑based,
runs in O(n log n) per iteration and easily fits the 1000 s budget.
"""
import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: fast evaluation (same formula as evaluate_sequence but without
# the security checks).  The vector is assumed to be non‑negative.
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, offset: float) -> np.ndarray:
    """Initialize a positive decreasing sequence a_i = 1/√(i+1+offset) and
    renormalise it to ℓ¹‑norm 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.sqrt(idx)
    a /= a.sum()
    return a

            if i == j:
                continue

        # ----- propose a mass transfer -----
        delta = step_frac * current[i] * np.random.rand()
        if delta <= 0.0:
            continue
        new_seq = current.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # re‑project onto the simplex (keeps sum = 1)
        new_seq = project_simplex(new_seq)

        # ----- evaluate the true objective -----
        new_val = fast_score(new_seq)

        # ----- acceptance test -----
        if new_val < current_val - 1e-12:           # strict improvement
            current, current_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < uphill_prob:
                current, current_val = new_seq, new_val

        # ----- occasional random jitter (helps escape deeper local minima) -----
        if iter_count % 30000 == 0 and iter_count > 0:
            jitter_scale = 0.02 * step_frac
            noise = (np.random.rand(n) - 0.5) * jitter_scale
            trial = current + noise
            trial = project_simplex(trial)
            trial_val = fast_score(trial)
            if trial_val < current_val:
                current, current_val = trial, trial_val
                if trial_val < best_val:
                    best_val, best_seq = trial_val, trial.copy()

        # ----- schedule adaptation -----
        if iter_count % adapt_interval == 0 and iter_count > 0:
            step_frac = max(step_frac * 0.9, min_step)
            alpha = min(alpha * alpha_inc, alpha_max)

        # ----- progress report -----
        if iter_count % 2000 == 0:
            elapsed = time.time() - (deadline - budget_s + 0.5)
            print(f"[iter {iter_count:6d}] best={best_val:.6f}, "
                  f"cur={current_val:.6f}, max_conv={max_conv:.6e}, "
                  f"time={elapsed:.1f}s")

        # early stopping if target already reached
        if best_val <= 1.503:
            print("[stop] target reached")
            break

        iter_count += 1

    print(f"[final] best value found: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list
    return best_seq.tolist()
