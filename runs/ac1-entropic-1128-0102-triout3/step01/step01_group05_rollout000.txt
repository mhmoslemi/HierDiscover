"""
Gradient‑guided stochastic search for a non‑negative step function
with a small value of

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases

1.  A cheap grid search over the two‑parameter family
        a_i ∝ (i+1+offset)^{-p}
    (offset ∈ [−0.5,2.0], p ∈ [0.40,0.60]) to get a good
    starting point.

2.  A projected gradient‑descent loop.
    At each iteration we
        * compute the linear convolution a∗a,
        * locate the active indices where the maximum is attained,
        * compute the gradient of the (local) max‑convolution,
        * move a a small amount opposite to that gradient,
        * project back onto the probability simplex (non‑negative,
          unit ℓ¹‑norm).
    The step‑size is adapted automatically and a tiny uphill‑move
    probability (simulated‑annealing style) helps to escape plateaus.

3.  A final light‑weight hill‑climbing phase (mass transfer between two
    random coordinates) to polish the result.

All heavy operations use FFT‑based convolutions (O(n log n)).  The
search stops a short time before the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation without the security checks (used internally)
def _fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using FFT."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
def _linear_conv(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    return np.fft.irfft(fx * fy, size)[:n]


# ----------------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the active maximum of conv(a,a) w.r.t. a.
    The returned array has the same shape as a.
    """
    # mask : 0/1 vector of length 2n‑1 marking active positions
    a_rev = a[::-1]
    # convolution mask * a_rev  -> length 3n‑2
    full = _linear_conv(mask.astype(np.float64), a_rev)
    n = a.size
    # slice the part that corresponds to the gradient (see analysis)
    g = full[n - 1 : n - 1 + n]
    return 2.0 * g


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        # ----- gradient of the active maximum -----
        grad = _gradient_of_max(cur, mask)

        # ----- decide a step size -------------------------------------------------
        # we limit the L∞ change to step_factor * max(cur)
        max_allowed = step_factor * np.max(cur)
        if max_allowed <= 0:
            max_allowed = step_factor / n
        # scaling factor so that max|eta * grad| = max_allowed
        eta = max_allowed / (np.max(np.abs(grad)) + 1e-16)

        # ----- propose a new point -------------------------------------------------
        cand = cur - eta * grad
        # projection onto the simplex (non‑negative, sum = 1)
        cand = _project_to_simplex(cand)

        cand_val = _fast_score(cand)

        # ----- accept / reject ----------------------------------------------------
        improved = cand_val < cur_val
        if improved:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_seq = cand_val, cand.copy()
        else:
            # simulated‑annealing style uphill acceptance
            # temperature slowly decays to zero
            remaining = max(deadline - time.time(), 0.0)
            T = 0.02 * (remaining / budget_s)   # tiny temperature
            if T > 0 and np.random.rand() < np.exp(-(cand_val - cur_val) / T):
                cur, cur_val = cand, cand_val
                # slightly shrink step factor when we accept a worse move
                step_factor = max(step_factor * 0.9, 1e-4)

        # adapt step factor (make it smaller when we stall)
        if it % 2000 == 0 and it > 0:
            step_factor = max(step_factor * 0.95, 1e-4)

        # ----- occasional random mass‑transfer move (helps escape flat regions) -----
        if np.random.rand() < 0.01:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.1 * cur[i] * np.random.rand()
                tmp = cur.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = _project_to_simplex(tmp)
                tmp_val = _fast_score(tmp)
                if tmp_val < cur_val:
                    cur, cur_val = tmp, tmp_val
                    if tmp_val < best_val:
                        best_val, best_seq = tmp_val, tmp.copy()

        # ----- progress report -----
        if it % 1000 == 0:
            print(f"[iter {it:6d}] cur={cur_val:.6f}  best={best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    # 4) Final polishing: short hill‑climbing using tiny transfers
    cur = best_seq.copy()
    cur_val = best_val
    step = 0.04
    it2 = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = step * cur[i] * np.random.rand()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand)
        cand_val = _fast_score(cand)
        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_seq = cand_val, cand.copy()
        if it2 % 2000 == 0:
            print(f"[polish {it2:6d}] cur={cur_val:.6f}  best={best_val:.6f}")
        it2 += 1

    print(f"[final] best value found: {best_val:.6f}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
