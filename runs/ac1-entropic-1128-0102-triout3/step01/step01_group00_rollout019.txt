                           b ≥ 0 .

        - Normalise the solution to have sum 1 and mix it with f
          (f ← (1‑t)f + t b,  t≈0.2).  This monotone step reduces the
          maximal convolution value.

3.  Refine the result with a *projected sub‑gradient* descent on the
    probability simplex (the same idea as the previous best solution
    but with momentum and many more iterations).  The sub‑gradient of the
    max‑convolution at index k is
        g_i = 2·f_{k‑i}  (zero when the index is out of range).

        f ← Π_S (f − η·v)          (Π_S = Euclidean projection onto the simplex).

import time
import math
import numpy as np
from scipy.optimize import linprog


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * a.size - 1]
    return conv

        mix_t = 0.18                     # mixing coefficient (adjustable)
        for lp_iter in range(max_lp_iters):
            if time.time() >= deadline:
                break

            # current convolution and score
            conv = _self_convolution(a, fft_len_self)
            M = conv.max()
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"LP‑phase n={n} iter={lp_iter} score={cur_score:.6f}")

            # Build dense constraint matrix A (2n‑1)×n  : (f * b)_k ≤ M
            rows = 2 * n - 1
            A = np.zeros((rows, n), dtype=float)
            # each column i contributes a[i] to rows i … i+n‑1
            for i in range(n):
                A[i:i + n, i] = a[i]

            # Linear programme: maximise sum(b)  ⇔ minimise -∑b
            c = -np.ones(n)
            b_ub = np.full(rows, M)
            bounds = [(0, None)] * n

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
            if not res.success:
                # LP failed – abort this length early
                print(f"  LP failure at n={n} iter={lp_iter}")
                break
            b_vec = res.x
            s = b_vec.sum()
            if s <= 0:
                break
            b_vec = b_vec / s                     # re‑normalise to sum = 1

            # convex combination, keep sum‑1 invariant
            a = (1.0 - mix_t) * a + mix_t * b_vec
            a = np.maximum(a, 0.0)
            a /= a.sum()                          # safeguard against numeric drift

        # --------------------------------------------------------------
        # 5. Sub‑gradient refinement (many cheap iterations)
        # --------------------------------------------------------------
        max_sg_iters = 200_000                     # plenty of room under the budget
        step0 = 0.6                                 # initial step size
        beta = 0.85                                 # momentum coefficient
        v = np.zeros_like(a)                       # momentum vector

        it = 0
        while time.time() < deadline and it < max_sg_iters:
            it += 1

            # convolution and current worst index
            conv = _self_convolution(a, fft_len_self)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"SG‑refine n={n} iter={it} score={cur_score:.6f}")

            k_max = int(conv.argmax())

            # sub‑gradient of the active max entry
            g = np.zeros_like(a)
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            i_idx = np.arange(low, high + 1, dtype=int)
            j_idx = k_max - i_idx
            g[i_idx] = 2.0 * a[j_idx]

            # momentum update
            v = beta * v + (1.0 - beta) * g

            # diminishing step size
            step = step0 / math.sqrt(it)

            # projected gradient step
            a = _project_simplex(a - step * v)

            # occasional tiny random perturbation to escape tiny plateaus
            if it % 50000 == 0:
                eps = 1e-4
                a += eps * (np.random.rand(n) - 0.5)
                a = np.maximum(a, 0.0)
                a /= a.sum()

        # end of one length ------------------------------------------------

    # ------------------------------------------------------------------
    # 6. Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Convert to plain Python list – evaluate_sequence accepts any list of floats
    return [float(x) for x in best_seq]
