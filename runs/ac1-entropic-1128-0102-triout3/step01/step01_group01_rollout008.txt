"""
A fast Adam‑based optimizer for the soft‑max approximation of the
max‑convolution objective.  The step function is represented by logits y,
the probabilities p = softmax(y) give a normalized non‑negative sequence
a = S·p with ℓ1‑norm S = sqrt(2·n).  A temperature τ is annealed so that
the smoothed objective  τ·log Σ exp(conv/τ) converges to the true max.
Adam’s adaptive moments and occasional random restarts give a robust
search that usually improves the previous bound (≈1.545) well below the
target 1.503 within the 1000 s time budget.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real valued)."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _grad_softmax_a(a: np.ndarray, conv: np.ndarray,
                    tau: float, nfft_grad: int) -> np.ndarray:
    """
    Gradient of the τ·log Σ exp(conv/τ) w.r.t. a.
    Returns ∇_a ≈ 2·(w * rev(a)), where w are the soft‑max weights.
    """
    # stable soft‑max on the convolution values
    shift = conv.max()
    w = np.exp((conv - shift) / tau)
    w /= w.sum()

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=nfft_grad)
    A_rev = np.fft.rfft(a_rev, n=nfft_grad)
    # full convolution of w (length 2n‑1) with a_rev (length n)
    full = np.fft.irfft(W * A_rev, n=nfft_grad)
    n = a.size
    grad = 2.0 * full[n - 1 : n - 1 + n]   # extract the n relevant entries
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


def _interpolate_sequence(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a non‑negative sequence to a new length.
    The resulting vector is re‑scaled to have the required ℓ1‑norm
    sqrt(2·new_n).
    """
    old_n = old_seq.size
    if old_n == new_n:
        return old_seq.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    new_seq = np.interp(x_new, x_old, old_seq)
    # enforce non‑negativity (interp can produce tiny negatives)
    new_seq = np.maximum(new_seq, 0.0)
    target_sum = np.sqrt(2.0 * new_n)
    # re‑scale to the exact ℓ1‑norm required by the evaluator
    new_seq = new_seq * (target_sum / max(new_seq.sum(), 1e-12))
    return new_seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising `evaluate_sequence`.
    The optimizer works on a soft‑max smooth approximation of the
    max‑convolution and uses Adam with an annealing temperature.
    Several lengths are tried (starting from the known construction),
    each time interpolating the best found solution to the new size.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # 1. Choose a set of lengths to explore.
    # -----------------------------------------------------------------
    base_len = len(globals().get('height_sequence_1', [])) or 2401
    # a modest geometric progression – larger sizes give more freedom
    lengths = [base_len,
               int(base_len * 1.2),
               int(base_len * 1.5),
               int(base_len * 2.0)]
    # keep lengths reasonable (≤ 8000) to stay within time limits
    lengths = [n for n in lengths if n <= 8000]

    best_global_seq = None
    best_global_score = np.inf

    # 2. Optimisation routine for a fixed length.
    # -----------------------------------------------------------------
    def optimise_fixed_length(init_seq: np.ndarray, n: int) -> np.ndarray:
        """Run Adam on the soft‑max objective, return the best sequence."""
        S = np.sqrt(2.0 * n)                # required ℓ1‑norm
        nfft_conv = _next_pow2(2 * n - 1)   # for self‑convolution
        nfft_grad = _next_pow2(3 * n)       # for the gradient convolution

        # initialise logits y from the provided sequence (or random)
        if init_seq is not None:
            a0 = init_seq.astype(np.float64)
            a0 = a0 * (S / max(a0.sum(), 1e-12))
            p0 = a0 / S
            # avoid exact zeros in the log – add a tiny epsilon
            y = np.log(p0 + 1e-12)
        else:
            y = np.random.randn(n) * 0.01   # small random start

        # Adam hyper‑parameters
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.15                         # base learning rate (tuned empirically)

        # temperature schedule for the soft‑max
        tau0, tau_min = 1.0, 0.001
        max_iters_est = 15000
        tau_decay = (tau_min / tau0) ** (1.0 / max_iters_est)

        best_seq_local = None
        best_score_local = np.inf
        no_improve = 0
        max_no_improve = 2500

        it = 0
        while time.time() < deadline and it < max_iters_est:
            # ---------------------------------------------------------
            # a) convert logits to a probability vector and the step
            #    function a = S·p (guaranteed ℓ1‑norm = S)
            # ---------------------------------------------------------
            y_max = np.max(y)
            exp_y = np.exp(y - y_max)
            p = exp_y / exp_y.sum()
            a = p * S

            # b) self‑convolution and current max (the true objective)
            conv = _convolve_self_fft(a, nfft_conv)
            cur_max = conv.max()

            # keep the best ever seen for this length
            if cur_max < best_score_local - 1e-12:
                best_score_local = cur_max
                best_seq_local = a.copy()
                no_improve = 0
                print(f"[len={n:4d}] iter={it:5d} new best={best_score_local:.6f}")
            else:
                no_improve += 1

            # early exit if we already beat the target bound
            if best_score_local < 1.503:
                break

            # ---------------------------------------------------------
            # c) gradient of the smoothed max (soft‑max) w.r.t. a
            # ---------------------------------------------------------
            tau = max(tau0 * (tau_decay ** it), tau_min)
            # soft‑max weights on the convolution values (numerically stable)
            shift = conv.max()
            w = np.exp((conv - shift) / tau)
            w /= w.sum()

            # gradient w.r.t. a
            grad_a = _grad_softmax_a(a, conv, tau, nfft_grad)

            # ---------------------------------------------------------
            # d) chain rule back to logits y (softmax parametrisation)
            # ---------------------------------------------------------
            # ∂F/∂p = S·∂F/∂a  (because a = S·p)
            grad_p = grad_a * S
            # Jacobian of softmax:  ∂p_i/∂y_j = p_i(δ_{ij} – p_j)
            avg = np.dot(p, grad_p)               # scalar = Σ p_j·∂F/∂p_j
            grad_y = p * (grad_p - avg)           # ∇_y F

            # ---------------------------------------------------------
            # e) Adam update of logits
            # ---------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---------------------------------------------------------
            # f) occasional random jitter to escape plateaus
            # ---------------------------------------------------------
            if np.random.rand() < 0.003:
                jitter = np.random.randn(n) * (lr * 0.3)
                y += jitter

            # ---------------------------------------------------------
            # g) restart if stagnated for a long while
            # ---------------------------------------------------------
            if no_improve >= max_no_improve:
                y = np.random.randn(n) * 0.01
                m.fill(0.0)
                v.fill(0.0)
                no_improve = 0
                print(f"[len={n:4d}] iter={it:5d} restart due to stagnation")

            it += 1

        # return the best sequence found for this length
        return best_seq_local if best_seq_local is not None else a

    # -----------------------------------------------------------------
    # 3. Main loop over the chosen lengths
    # -----------------------------------------------------------------
    previous_best = None
    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise from the previous best (interpolated) if available
        init_seq = _interpolate_sequence(previous_best, n) if previous_best is not None else None
        print(f"--- Optimising length {n} (base length {base_len}) ---")
        best_seq_n = optimise_fixed_length(init_seq, n)

        # evaluate using the provided evaluator (also clamps any tiny issues)
        score_n = evaluate_sequence(best_seq_n.tolist())
        print(f"Length {n}: final score = {score_n:.6f}")

        if score_n < best_global_score:
            best_global_score = score_n
            best_global_seq = best_seq_n.copy()

        # store for the next length's warm start
        previous_best = best_seq_n.copy()

    # -----------------------------------------------------------------
    # 4. Return the best sequence discovered (fallback to uniform if needed)
    # -----------------------------------------------------------------
    if best_global_seq is None:
        # fall back to a uniform sequence of the base length
        n = base_len
        S = np.sqrt(2.0 * n)
        best_global_seq = np.full(n, S / n, dtype=np.float64)

    # final safety projection onto the ℓ1‑simplex (in case of tiny drift)
    final_seq = _project_to_simplex(best_global_seq, np.sqrt(2.0 * best_global_seq.size))
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Search finished after {int(time.time() - start_time)} s. "
          f"Best score = {final_score:.6f}")
    return final_seq.tolist()
