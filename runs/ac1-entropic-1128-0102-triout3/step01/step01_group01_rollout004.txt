"""
Hybrid optimizer for the max‑self‑convolution problem.

The routine combines three ideas:

1️⃣  Adam‑based projected gradient descent on a smooth (log‑sum‑exp)
    surrogate of the maximum convolution entry.

2️⃣  Every ``lp_interval`` iterations we solve a linear program:
    maximise the ℓ₁‑norm of a non‑negative vector ``b`` under the
    linear constraints   conv(f,b) ≤ current_max .
    This is the “LP‑step” described in the statement and yields a new
    direction that strictly respects the current worst‑case bound.

3️⃣  Simple stochastic restarts and occasional jitter to escape
    plateaus.

The algorithm works for any length (default: the length of the
existing best construction ``height_sequence_1``).  It respects the
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT convolution, soft‑max gradient, simplex proj.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a via FFT (real part only)."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    The derivative is 2·(w * a_rev) where w_k ∝ exp(conv_k/τ).
    """
    w = np.exp(conv / tau)
    w /= w.sum()
    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]
    n = a.size
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

                    if sum_g > 0:
                        # Rescale to the required ℓ₁‑norm
                        g = g * (S / sum_g)
                    else:
                        g = a.copy()
                else:
                    # LP failed – fall back to current vector
                    g = a.copy()
            except Exception as e:  # pragma: no cover
                print(f"LP solver exception: {e}")
                g = a.copy()

            # Convex combination (mixing parameter)
            mix = 0.5
            a = (1.0 - mix) * a + mix * g
            # Numerical drift → re‑project onto the simplex
            a = _project_to_simplex(a, S)

        else:
            # ------------------- Adam gradient step -------------------
            grad = _grad_softmax(a, conv, tau, nfft)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** (t_iter + 1))
            v_hat = v / (1.0 - beta2 ** (t_iter + 1))

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
            a_candidate = _project_to_simplex(a - step, S)

            # Accept only if the true max‑convolution improves
            cand_conv = _convolve_self_fft(a_candidate, nfft)
            cand_M = cand_conv.max()
            if cand_M < cur_M:
                a = a_candidate
            else:
                # occasional random jitter to shake out of flat regions
                if np.random.rand() < 0.02:
                    jitter = np.random.randn(n) * (base_lr * 0.4)
                    a = _project_to_simplex(a + jitter, S)

        # ---- 3) temperature annealing for the soft‑max surrogate ----
        tau = max(tau * tau_decay, tau_min)

        # ---- 4) simple stagnation handling (random restart) ----
        # If no improvement for a long stretch, inject a random point.
        if t_iter % 5000 == 0 and t_iter > 0:
            if best_score - cur_score > -1e-6:  # essentially no progress
                alpha = np.full(n, 0.2)  # highly sparse Dirichlet
                a = np.random.dirichlet(alpha) * S
                print(f"[{t_iter:06d}] Random restart (Dirichlet).")

        t_iter += 1

    # --------------------------------------------------------------
    # Return the best sequence discovered (projected once more for safety)
    # --------------------------------------------------------------
    final_seq = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {t_iter} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
