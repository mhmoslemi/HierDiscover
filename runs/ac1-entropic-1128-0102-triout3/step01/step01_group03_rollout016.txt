def propose_candidate(seed=42, budget_s=950., **kwargs):
    rng = np.random.default_rng(seed)
    n = 3000  # choose tradeoff
    # start with sqrt_series
    a = sqrt_series(n)  # sum=1
    # compute initial score
    conv = conv_fft(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    # parameters
    eta = 0.05
    gamma = 0.1
    gamma_inc = 1.001  # inc per iteration
    eta_decay = 0.9995
    # loop
    deadline = time.time() + budget_s - 5.0
    it = 0
    while time.time() < deadline:
        conv = conv_fft(a)
        max_val = conv.max()
        # compute softmax weights
        # apply temperature scaling gamma
        w = np.exp(gamma * (conv - max_val))  # subtract max to avoid overflow
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum
        # gradient of smoothed max
        grad = 2.0 * np.correlate(a, w, mode='full')[n-1:2*n-1]
        # take descent step
        step = eta * grad
        # avoid overshoot: clip step so a - step stays nonnegative after projection
        a_trial = a - step
        # occasional random perturbation
        if rng.random() < 0.01:
            noise = (rng.random(n) - 0.5) * 1e-4
            a_trial += noise
        # project onto simplex
        a_trial = project_onto_simplex(a_trial)
        # evaluate
        conv_trial = conv_fft(a_trial)
        trial_score = 2.0 * n * conv_trial.max()
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            print(f"[iter {it}] new best {best_score:.6f}")
        # accept if improvement else maybe probabilistic
        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            # Metropolis based on increase
            delta = trial_score - cur_score
            temperature = max(0.001, 0.02 * (1.0 - it / 1e6))
            if rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_score = trial_score
        # decay parameters
        eta *= eta_decay
        gamma *= gamma_inc
        it += 1
    # final evaluation via provided evaluate_sequence
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] best score {final_score:.6f}")
    return best_a.tolist()

"""
Propose a better sequence for the evaluation function.

The algorithm works in two stages:

1.  Start from the classic half‑binomial (sqrt) sequence which already gives a
    score close to π/2.  The sequence is normalized to unit L¹‑norm.

2.  Perform a long projected‑gradient descent where the objective is a smooth
    approximation of the L∞‑norm of the auto‑convolution.  For a temperature‑like
    parameter γ we form a soft‑max weight vector
        w_k = exp(γ·(conv_k – max(conv))) / Σ exp(⋯) .
    The gradient of the smoothed maximum w.r.t. the coefficients a_i is

        ∇_i  = 2· Σ_j a_j·w_{i+j}
             = 2·correlation(a, w)[i]               (full correlation)

    After taking a step opposite to this gradient we project back onto the
    probability simplex (non‑negative, ∑a_i = 1).  Both the step size η and the
    smoothing parameter γ are decayed gradually so that the optimisation first
    moves gently and later focuses more sharply on the true maximum.

3.  To avoid being trapped in shallow minima a small amount of random mass
    exchange is performed from time to time and a Metropolis‑type acceptance rule
    is applied for uphill moves.

The whole loop runs until the allocated time (default 950 s, leaving a safety
margin) expires.  The best sequence found is finally evaluated with the
provided `evaluate_sequence` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must stay at top level (no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1 - x)^(-1/2) :
        a_i = C(2i,i) / 4**i .
    The returned vector is normalised to sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
        { x | x_i ≥ 0, Σ x_i = 1 }.
    Implementation of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Compute the discrete linear convolution x * x using a real FFT.
    Result length is 2*len(x) - 1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def smooth_max_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Soft‑max weights for a given convolution vector and smoothing parameter γ.
    The weights sum to 1 and put most mass on the largest entries.
    """
    # shift by the current maximum to avoid overflow
    max_val = conv.max()
    shifted = conv - max_val
    w = np.exp(gamma * shifted)
    s = w.sum()
    if s == 0.0:
        # fallback – uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return w / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The routine starts from the half‑binomial series, then runs a long
    projected gradient descent on a smoothed version of the L∞‑norm of the
    auto‑convolution.  The best sequence encountered before the time limit is
    returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # configuration (feel free to tweak)
    n = 3000                       # length of the sequence
    max_iter = 1_000_000_000       # huge safety bound – the real stop is time
    eta = 0.08                     # initial step size
    gamma = 0.15                   # initial smoothing (soft‑max) parameter
    eta_decay = 0.9997             # gradual reduction of step size
    gamma_inc = 1.0015            # gently increase focus on the true maximum
    noise_prob = 0.008             # frequency of a tiny random perturbation
    noise_scale = 1e-4             # amplitude of the random perturbation

    # ------------------------------------------------------------------
    # initialise – half‑binomial (sqrt) : already L1‑normalised
    a = sqrt_series(n)
    conv = conv_fft(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # a small safety margin
    it = 0
    no_improve = 0          # increments when we see no improvement
    report_every = 5000

    while time.time() < deadline and it < max_iter:
        # ----- convolution and soft‑max weights -----
        conv = conv_fft(a)
        max_val = conv.max()

        # soft‑max over all convolution entries
        w = smooth_max_weights(conv, gamma)

        # gradient of the smoothed maximum:
        #   ∇_i = 2 * Σ_j a_j * w_{i+j}
        # compute via full correlation (np.correlate)
        grad_full = np.correlate(a, w, mode='full')
        grad = 2.0 * grad_full[n - 1:2 * n - 1]   # take lag 0 … n‑1

        # ----- take a descent step -----
        step = eta * grad

        # avoid a huge step that would make many entries negative;
        # clip to 95 % of the current mass at each coordinate
        step = np.minimum(step, a * 0.95)

        a_trial = a - step

        # occasional tiny random perturbation to keep diversity
        if rng.random() < noise_prob:
            noise = (rng.random(n) - 0.5) * noise_scale
            a_trial += noise

        # project back onto the probability simplex
        a_trial = project_onto_simplex(a_trial)

        # ----- evaluate candidate -----
        conv_trial = conv_fft(a_trial)
        trial_score = 2.0 * n * conv_trial.max()

        # keep the best ever seen
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            print(f"[iter {it:07d}] new best score = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # ----- acceptance rule (always accept improvements) -----
        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            # Metropolis style acceptance for uphill moves
            # a simple temperature schedule that cools down slowly
            temp = max(0.001, 0.04 * (1.0 - it / (budget_s * 0.8)))
            if rng.random() < np.exp(-(trial_score - cur_score) / temp):
                a = a_trial
                cur_score = trial_score
            # otherwise reject

        # ----- parameter decay -----
        eta = max(eta * eta_decay, 1e-6)
        gamma = gamma * gamma_inc

        # ----- occasional restart if stuck -----
        if no_improve > 200_000:
            # blend the best solution with a fresh Dirichlet draw
            fresh = rng.random(n)
            fresh = fresh / fresh.sum()
            a = project_onto_simplex(0.6 * best_a + 0.4 * fresh)
            cur_score = 2.0 * n * conv_fft(a).max()
            no_improve = 0
            print(f"[iter {it:07d}] restart – new score {cur_score:.6f}")

        # ----- periodic progress report -----
        if (it + 1) % report_every == 0:
            elapsed = time.time() - (deadline - budget_s + 5.0)
            print(f"[iter {it:07d}] best={best_score:.6f}, cur={cur_score:.6f}, "
                  f"eta={eta:.2e}, gamma={gamma:.3f}, elapsed={elapsed:.1f}s")

        it += 1

    # ------------------------------------------------------------------
    # final evaluation with the supplied function (adds the clipping etc.)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
