```
    t = 0.4    # mixing factor
    while time.time() < deadline:
        # compute conv and current bound
        conv_f = conv_fft(f)
        M = conv_f.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_f = f.copy()
            print(f"[iter {iteration}] new best score {best_score:.6f}")

        # Use LP step if available else gradient descent
        if use_lp:
            # build matrix
            A = np.empty((2*n-1, n), dtype=np.float64)
            # fill using broadcasting
            for i in range(n):
                A[i:i+n, i] = f
            b = np.full(2*n-1, M, dtype=np.float64)
            # solve LP: maximize sum(g)
            c = -np.ones(n, dtype=np.float64)  # minimize -sum => maximize sum
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
            if res.success:
                g0 = res.x
                sum_g0 = g0.sum()
                if sum_g0 > 1e-12:
                    g = g0 / sum_g0
                else:
                    g = f.copy()
            else:
                # fallback to small random perturbation
                g = f.copy()
        else:
            # fallback: small random perturbation
            g = f + (rng.random(n)-0.5)*1e-3
            g = np.maximum(g, 0)
            g /= g.sum()

        # convex combination
        f_new = (1 - t) * f + t * g
        # optionally small random noise
        if rng.random() < 0.02:
            noise = (rng.random(n) - 0.5) * 1e-4
            f_new += noise

        # acceptance rule
        if new_score <= cur_score:

                cur_score = new_score
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}= \frac{2n\;\max (a*a)}{(\sum a)^2}\;,
\]

where `a` is a non‑negative sequence (the step heights) and `a*a` is its discrete
auto‑convolution.  
The score is invariant under scaling of the sequence, therefore we fix
`∑a = 1` and simply minimise `2·n·max(a*a)`.

The algorithm combines three ideas:

1. **Start from a good analytic seed** – the “half‑binomial’’ (central‑binomial)
   sequence `a_i = C(2i,i)/4^i`, normalised to sum = 1.
2. **Linear‑programming improvement** – for a fixed `f` we solve  

   \[
   \max \;\sum_j g_j \quad\text{s.t.}\quad g_j\ge 0,\; (f*g)_k \le M\ \forall k,
   \]

   where `M = max(f*f)`.  The constraint matrix is linear in `g`,
   therefore a standard LP (`scipy.optimize.linprog`) gives the feasible
   `g` with the largest total mass.  After normalising `g` we move a small
   amount towards it: `f←(1‑τ)f+τg`.  Repeating this step monotonically
   decreases `M`.
3. **Simulated‑annealing fine‑tuning** – occasional random perturbations
   and a Metropolis acceptance rule help to escape shallow minima.

All heavy work is done inside the time budget (≈ 950 s).  The best sequence
encountered is returned.

```python
"""
Propose a step‑function sequence with a low evaluation score.

The algorithm works as follows
  1. Initialise with the half‑binomial (central‑binomial) sequence,
     normalised to L1‑norm = 1.
  2. Repeatedly:
        – compute the current maximal auto‑convolution value M,
        – solve a linear program that maximises the L1‑mass of a new
          sequence g under the linear constraints (f * g)_k ≤ M,
        – normalise g to sum = 1 and move a fraction τ towards it
          (convex combination keeps the L1‑norm fixed),
        – occasionally add a tiny random perturbation and accept
          uphill moves with a Metropolis rule (simulated annealing).
  3. Keep the best sequence seen so far and return it after the time
     budget expires.

The LP is small (≈ 2 n × n constraints) – for n≈600 the matrix fits easily
into memory and `scipy.optimize.linprog` solves it in a fraction of a second.
If SciPy is unavailable the code falls back to a simple simulated‑annealing
search.

The returned list can be fed directly to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities -----------------------------------------------------

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of the series a_i = C(2i,i) / 4**i,
    i.e. the coefficients of (1 - x)^(-1/2).  The vector is normalised
    to sum‑to‑1 (probability simplex).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.
    Implementation of Duchi‑et‑al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Discrete convolution x*x using real FFT.  Result length is 2·len(x)-1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    return np.fft.irfft(X * X, n=m)


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ g) == f * g   (discrete convolution).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A
```
