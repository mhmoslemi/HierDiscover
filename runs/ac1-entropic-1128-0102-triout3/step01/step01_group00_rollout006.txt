def gradient_from_weights(a, w):
    n = a.shape[0]
    # compute convolution of w_rev and a
    w_rev = w[::-1]
    L = len(w_rev) + n - 1  # = (2n-1) + n -1 = 3n-2
    fft_len = 1 << ((L).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    Wf = np.fft.rfft(w_rev, n=fft_len)
    conv_rev = np.fft.irfft(Af * Wf, n=fft_len)
    conv_rev = conv_rev[:L]  # length 3n-2
    offset = (len(w_rev) - 1)  # = 2n-2
    g = 2 * conv_rev[offset:offset + n]
    return g
```

"""
Projected Adam optimizer with a smooth‐max approximation.

The evaluation function is scale–invariant, so we work on the probability
simplex (∑a_i = 1, a_i ≥ 0).  The objective to minimise is

    F(a) = 2·n·max_k (a * a)_k ,

where * denotes linear convolution.  To obtain a smooth gradient we replace
the hard max by a soft‑max with temperature β:

    w_k(β) = exp(β·(c_k – max c)) / Σ_j exp(β·(c_j – max c)),
    ∇F_β(a) = 2·(w_rev * a)_{2n‑2 … 2n‑2+n-1},

where c = a*a is the auto‑convolution and w_rev is the reversed weight
vector.  For β → ∞ the soft‑max concentrates on the active index and the
gradient coincides with the sub‑gradient of the original problem.

The optimiser proceeds in stages:
  * several problem sizes n are tried (larger n give lower theoretical
    bounds);
  * for each n a warm‑start is built (uniform, interpolation from a previously
    found sequence or the provided “height_sequence_1”);
  * Adam with Euclidean projection onto the simplex updates the vector;
  * β is increased geometrically (0.5 → … → 80) so that the method first
    smooths the landscape and later refines the true max;
  * a short pure sub‑gradient phase finishes the optimisation;
  * the best sequence seen (lowest score) is kept and returned.

The implementation uses only NumPy and runs entirely within the provided
time budget (default 1000 s).  Progress is printed so you can see the
improvements as they happen.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1} using the algorithm of Duchi et al."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            break

        # ---------- Initialise a ----------
        if prev_seq is not None and len(prev_seq) == n:
            a = np.array(prev_seq, dtype=float)
            a = np.maximum(a, 0.0)
            if a.sum() == 0:
                a = np.random.rand(n)
        elif prev_seq is not None:
            # interpolate the previous best to the new length
            old_n = len(prev_seq)
            old_x = np.linspace(0.0, 1.0, old_n)
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, prev_seq)
            a = np.maximum(a, 0.0)
            if a.sum() == 0:
                a = np.random.rand(n)
        else:
            a = np.random.rand(n)

        a = a / a.sum()                     # project onto the simplex

        # ---------- Adam state ----------
        m = np.zeros(n)                     # first moment
        v = np.zeros(n)                     # second moment
        t_iter = 0
        lr = 0.03                           # base learning rate
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # ---------- FFT planning ----------
        # Length needed for convolution a * a (linear, not circular) is 2n‑1.
        # For the smooth gradient we need a convolution with w_rev of length 3n‑2.
        conv_len = 2 * n - 1
        grad_len = 3 * n - 2
        fft_len = 1 << ((grad_len).bit_length())   # power of two ≥ 3n‑2

        # ---------- β schedule (smooth‑max temperature) ----------
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

        # Number of Adam iterations per β (adjusted later if time runs out).
        iters_per_beta = 6000

        # ------------------------------------------------------------------
        # Adam optimisation with increasing β.
        # ------------------------------------------------------------------
        for beta in beta_schedule:
            if time.time() >= deadline:
                break

            for _ in range(iters_per_beta):
                if time.time() >= deadline:
                    break
                t_iter += 1

                # ----- FFT of a (single transform reused for both conv & grad) -----
                Af = np.fft.rfft(a, n=fft_len)

                # ----- Linear auto‑convolution a*a (first 2n‑1 entries) -----
                conv_full = np.fft.irfft(Af * Af, n=fft_len)
                conv = conv_full[:conv_len]                # length 2n‑1

                # ----- Current score (sum a = 1 ⇒ score = 2·n·max(conv)) -----
                max_val = conv.max()
                cur_score = 2.0 * n * max_val
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] "
                          f"Improved: n={n}, it={t_iter}, β={beta}, score={cur_score:.6f}")

                # ----- Soft‑max weights for the smooth max -----
                if beta == 0:
                    w = np.full(conv_len, 1.0 / conv_len)
                else:
                    # subtract max for numerical stability
                    exp_vals = np.exp(beta * (conv - max_val))
                    w = exp_vals / exp_vals.sum()

                # ----- Gradient of the smooth objective -----
                # w_rev is the reversed weight vector.
                w_rev = w[::-1]
                Wf = np.fft.rfft(w_rev, n=fft_len)

                # convolution of w_rev with a (linear)
                conv_rev_full = np.fft.irfft(Af * Wf, n=fft_len)
                # needed slice starts at offset = len(w_rev)-1 = 2n‑2
                offset = len(w_rev) - 1
                g = 2.0 * conv_rev_full[offset:offset + n]   # shape (n,)

                # ----- Adam update with projection onto the simplex -----
                m = beta1 * m + (1.0 - beta1) * g
                v = beta2 * v + (1.0 - beta2) * (g * g)

                m_hat = m / (1.0 - beta1 ** t_iter)
                v_hat = v / (1.0 - beta2 ** t_iter)

                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                a = a - step
                a = project_simplex(a)

        # ------------------------------------------------------------------
        # Final pure sub‑gradient phase (β → ∞) to polish the solution.
        # ------------------------------------------------------------------
        subgrad_iters = 4000
        for _ in range(subgrad_iters):
            if time.time() >= deadline:
                break
            t_iter += 1

            Af = np.fft.rfft(a, n=fft_len)
            conv_full = np.fft.irfft(Af * Af, n=fft_len)
            conv = conv_full[:conv_len]

            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Improved (subgrad): n={n}, it={t_iter}, score={cur_score:.6f}")

            # Sub‑gradient corresponding to the maximal convolution entry.
            k_max = int(np.argmax(conv))
            g_sub = np.zeros_like(a)
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            i_idx = np.arange(low, high + 1, dtype=int)
            j_idx = k_max - i_idx
            g_sub[i_idx] = 2.0 * a[j_idx]

            # Diminishing step size (similar to the previous baseline).
            step_sz = 0.5 / math.sqrt(t_iter)
            a = a - step_sz * g_sub
            a = project_simplex(a)

        # If we already beat the target, we can stop early.
        if best_score < 1.503:
            print(f"[{time.time() - start_time:6.2f}s] Target reached: n={n}, score={best_score:.6f}")
            break

    # ------------------------------------------------------------------
    # Fallback in the (unlikely) case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Return a plain Python list – the evaluator accepts any sequence of floats.
    return [float(x) for x in best_seq]
