Pseudo:

def solve_lp(f, M):
    n = len(f)
    # Build matrix A: shape (2n-1, n)
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # for i from i_min to i_max: row[k-i] = f[i]
        # Equivalent row[j] = f[k-j] if valid
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
    A = np.vstack(rows)
    # inequality: A @ g <= M
    b_ub = np.full(2*n-1, M)
    # variable bounds: nonnegative
    bounds = [(0, None)] * n
    # objective: maximize sum(g) equivalent to minimize negative sum
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

def lp_mix_step(f, t): # returns new f after one mixing step if improves, else same.
    M = np.convolve(f, f).max()
    g = solve_lp(f, M)
    if g is None: return f
    # Compute sum(g), g_norm = g / sum(g)
    s = g.sum()
    if s <= 1e-12:
        return f
    g_norm = g / s
    f_new = (1 - t) * f + t * g_norm
    # Ensure nonnegative, renormalize to sum=1
    f_new = np.maximum(f_new, 0.0)

def solve_lp(f, M):
    n = len(f)
    # Build sparse A in csr format.
    import numpy as np
    from scipy.sparse import lil_matrix

    rows = 2 * n - 1
    A = lil_matrix((rows, n), dtype=np.float64)

    # fill
    for i in range(n):
        fi = f[i]
        if fi == 0:
            continue
        # row indices for this i are i + np.arange(n)
        # column indices are np.arange(n)
        # fill
        # Use for loop for speed but simple
        for j in range(n):
            row_idx = i + j
            A[row_idx, j] = fi

    # Convert to csr
    A_csr = A.tocsr()
    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)

    # Solve LP with bounds >=0
    res = linprog(c, A_ub=A_csr, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    else:
        return None

def subgradient_step(seq, eta):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    # gradient: For i from max(0, k-(n-1)) .. min(n-1, k): grad_i = 2 * seq[k-i]
    grad = np.zeros_like(seq)
    start = max(0, k-(n-1))
    end = min(n-1, k)
    # indices i such that j = k - i within [0, n-1]
    i = np.arange(start, end+1)
    grad[i] = 2.0 * seq[k - i]
    # step
    new_seq = seq - eta * grad
    # project onto simplex -> sum=1, non-negative
    new_seq = _project_simplex(new_seq)
    return new_seq

def _solve_lp(f, M):
    try:
        import scipy.sparse as sp
        from scipy.optimize import linprog
    except Exception:
        return None
    n = len(f)
    A = _build_conv_matrix(f)
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs-ds')
    # high-ds? use method='highs' which selects best.
    # fallback:
    if not res.success:
        # try alternative method
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm proceeds in four stages:

1. **Coarse families** – explore simple power‑law families a_i ∝ (i+1)^(-β)
   for a selection of lengths.  The best candidate from this grid is kept.

2. **Projected sub‑gradient descent** – starting from the best grid point,
   minimise the non‑smooth objective max_k (a*a)_k with a Euclidean
   projection onto the probability simplex (∑a_i = 1, a_i ≥ 0).  The step
   size is decayed and occasional random jitter is added.

3. **Linear‑programming mixing** – for the current vector f compute a
   feasible vector g that maximises its ℓ₁‑norm under the same convolution
   bound (conv(f,g) ≤ M, M = max conv(f,f)).  A convex combination of f and
   the normalised g is then searched (simple line search) for the best
   reduction of the objective.  Only modest sizes (n ≤ 1500) are treated
   with the LP to keep memory modest.

4. **Hill‑climbing refinement** – small random transfers of mass between
   two coordinates are accepted whenever they improve the objective.

All stages respect the time budget (default 1000 s).  The best sequence
found is returned as a plain Python list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring utilities (sum of the vector is assumed to be 1)
# ----------------------------------------------------------------------


def _compute_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a sequence with ℓ₁‑norm = 1."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Power‑law family generator
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, beta: float) -> np.ndarray:
    """Monotone decreasing power‑law a_i ∝ (i+1)^(-β), normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta)
    seq = seq / seq.sum()
    return seq


# ----------------------------------------------------------------------
# Sub‑gradient step (projected)
# ----------------------------------------------------------------------


def _subgradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """One projected sub‑gradient step for max_k (seq*seq)_k."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))                     # active index
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1)
    grad = np.zeros_like(seq)
    grad[idx] = 2.0 * seq[k - idx]               # 2·a_{k‑i}
    new_seq = seq - eta * grad
    new_seq = _project_simplex(new_seq)
    return new_seq


# ----------------------------------------------------------------------
# Linear programming utilities (only used for modest n)
# ----------------------------------------------------------------------


def _build_convolution_matrix(f: np.ndarray):
    """
    Build the dense matrix A (size (2n‑1)×n) such that A @ g = conv(f,g).
    Entry A[k, j] = f[k‑j] when 0 ≤ k‑j < n, otherwise 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i, fi in enumerate(f):
        if fi == 0.0:
            continue
        # fill the i‑th diagonal (offset i) with fi
        # row indices i + j, column j
        A[i:i + n, np.arange(n)] = fi
    return A


def _solve_lp(f: np.ndarray, M: float):
    """
    Solve   max 1ᵀg
    s.t.   conv(f,g) ≤ M   (component‑wise)
           g ≥ 0
    Returns the optimal g (numpy array) or None on failure.
    """
    try:
        from scipy.optimize import linprog
    except Exception:  # SciPy not available
        return None

    n = f.shape[0]
    A = _build_convolution_matrix(f)          # dense, but n is kept moderate
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)         # maximise sum(g) → minimise -sum(g)
    bounds = [(0, None)] * n

    # ``highs`` can handle dense matrices; it is fast and robust.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None


def _lp_improve(seq: np.ndarray, max_iters: int, deadline: float) -> np.ndarray:
    """
    Apply a few LP‑based mixing steps.
    The LP step is only attempted when the length is ≤ 1500 (memory
    considerations).  A simple line‑search over blending weights is used
    to obtain a new vector with a lower max‑convolution value.
    """
    n = seq.shape[0]
    if n > 1500:
        return seq

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        M = np.convolve(seq, seq).max()
        g = _solve_lp(seq, M)
        if g is None:
            break
        s = g.sum()
        if s <= 0.0:
            break
        g_norm = g / s                # bring back to ℓ₁‑norm = 1

        # line‑search on the convex combination weight t ∈ (0,1)
        best_local = seq
        best_score = _compute_score(seq)
        for t in np.linspace(0.0, 1.0, 11)[1:-1]:   # exclude 0 and 1
            new_seq = (1.0 - t) * seq + t * g_norm
            new_seq = _project_simplex(new_seq)    # enforce sum=1
            sc = _compute_score(new_seq)
            if sc < best_score:
                best_score = sc
                best_local = new_seq
        if best_score < _compute_score(seq) - 1e-12:
            seq = best_local
        else:
            break
    return seq


# ----------------------------------------------------------------------
# Tiny hill‑climbing refinement
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray, max_steps: int, deadline: float) -> np.ndarray:
    """
    Randomly move a small amount of mass between two coordinates.
    Accept the move only if it reduces the objective.
    """
    n = seq.shape[0]
    cur_score = _compute_score(seq)

    for _ in range(max_steps):
        if time.time() >= deadline:
            break
        i, j = np.random.choice(n, 2, replace=False)
        # move at most a modest fraction of the source mass
        max_eps = min(seq[i], 0.1 * seq[i])
        if max_eps <= 0.0:
            continue
        eps = max_eps * np.random.rand()
        new_seq = seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps
        new_seq = _project_simplex(new_seq)
        sc = _compute_score(new_seq)
        if sc < cur_score:
            seq = new_seq
            cur_score = sc
    return seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Global best bookkeeping
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    if 'height_sequence_1' in globals():
        try:
            best_seq = list(height_sequence_1)
            best_score = float(evaluate_sequence(best_seq))
            print(f"[init] Using provided height_sequence_1, score {best_score:.6f}")
        except Exception:
            best_seq = None
            best_score = float('inf')

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 2000, 2800]
    for n in lengths:

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        betas = np.linspace(0.0, 3.0, 61)          # step 0.05
        local_best_seq = None
