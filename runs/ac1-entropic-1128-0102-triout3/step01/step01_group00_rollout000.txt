def compute_conv_max(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[:2*n-1]
    k = int(np.argmax(conv))
    return conv[k], k, conv

We also need to compute gradient for given k.

def compute_subgradient(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - (n - 1))
    high = min(n - 1, k)
    i_idx = np.arange(low, high + 1)
    j_idx = k - i_idx
    g[i_idx] = 2.0 * a[j_idx]
    return g

def eg_update(a, g, eta):

    total = a.sum()
    if total <= 0:

"""
Optimise a non‑negative step function for the evaluation metric
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds in three phases:

1️⃣  Initialise from the best known construction (``height_sequence_1``) if
    available, otherwise start from a random probability vector.

2️⃣  For a sequence of increasing lengths we
    * up‑sample the current best vector (linear interpolation) to the new length,
    * run an **exponentiated‑gradient** descent on the probability simplex.
      The sub‑gradient of the active max‑convolution index k is
          g_i = Σ_{k active} 2·a_{k−i}  (zero when the index is out of range).
      The step size decays as η₀/√t.
    * keep the best sequence seen (lowest score).

3️⃣  Perform a short local random‑swap refinement: pick two indices (i,j) and
    transfer a small amount of mass from i to j; keep the change if it improves
    the score.

All operations respect the security checks of ``evaluate_sequence``:
the vector stays non‑negative, is normalised to sum‑to‑1 (scale‑invariant), and
is finally returned as a plain Python list of floats.

The method is fast (FFT based convolution) and, thanks to the up‑sampling
strategy, explores much larger dimensions than the reference implementation
while staying comfortably within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def compute_convolution(a: np.ndarray):
    """
    Compute the linear convolution a * a using a real FFT.
    Returns the full convolution of length 2*len(a)-1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
    return conv


def max_convolution_info(a: np.ndarray):
    """
    Return (max_value, max_indices, full_convolution) for the auto‑convolution
    of a.  ``max_indices`` contains all indices whose value is within a tiny
    tolerance of the maximum (to handle ties).
    """
    conv = compute_convolution(a)
    max_val = conv.max()
    # tolerance relative to magnitude, plus an absolute floor
    tol = max(1e-12 * max_val, 1e-15)
    max_inds = np.where(conv >= max_val - tol)[0]
    return max_val, max_inds, conv


def subgradient_from_active(a: np.ndarray, active_inds: np.ndarray):
    """
    Compute a sub‑gradient of the objective at ``a``.
    For each active convolution index k we add the contribution
        g_i += 2 * a_{k-i}
    (zero when k-i is out of range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_inds:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k - i_idx
        # add 2*a[j] to g[i]
        g[i_idx] += 2.0 * a[j_idx]
    return g


def exp_gradient_step(a: np.ndarray, g: np.ndarray, eta: float):
    """
    Exponentiated‑gradient update on the simplex:
        a_i ← a_i * exp(-η·g_i) , then renormalise.
    """
    # avoid under‑/overflow: clipping the exponent is unnecessary here
    a = a * np.exp(-eta * g)
    total = a.sum()
    if total <= 0.0:          # should not happen, but guard against numerical issues
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / total
    return a


def upsample_sequence(old: np.ndarray, new_len: int):
    """
    Linear interpolation from ``old`` (length m) to a new vector of length
    ``new_len``.  Result is normalised to sum‑to‑1.
    """
    m = old.shape[0]
    if m == new_len:
        return old.copy()
    # map old indices to [0,1]
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, new_len)
    new = np.interp(x_new, x_old, old)
    new = np.clip(new, 0.0, None)
    s = new.sum()
    if s == 0.0:
        new = np.full(new_len, 1.0 / new_len)
    else:
        new = new / s
    return new

# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget,
    prints progress, and uses a mixture of exponentiated‑gradient descent,
    progressive up‑sampling, and a short random‑swap refinement.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 1. Starting vector (try to reuse the previous best if available)
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)   # provided globally
        if np.any(np.isnan(init)) or np.any(np.isinf(init)):
            raise ValueError
        init = np.clip(init, 0.0, None)
        if init.sum() == 0.0:
            raise ValueError
        init = init / init.sum()
    except Exception:
        init = None  # fall back to random initialisation later

    # ------------------------------------------------------------------
    # 2. Sequence length schedule (progressively larger)
    # ------------------------------------------------------------------
    lengths = [200, 400, 800, 1200, 2400, 4800, 9600, 19200]

    # keep the best vector from the previous stage for up‑sampling
    previous_vec = None

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- initialise vector for this length -----------------------
        if previous_vec is not None:
            a = upsample_sequence(previous_vec, n)
        elif init is not None and init.shape[0] == n:
            a = init.copy()
        else:
            a = np.random.rand(n)
            a = a / a.sum()

        # ----- optimisation parameters ---------------------------------
        eta0 = 0.8                     # initial step size for EG
        # rough iteration budget inversely proportional to n (so total work stays bounded)
        max_iter = int(5e5 / n) + 5000
        it = 0

        # ----- main exponentiated‑gradient loop -----------------------
        while time.time() < deadline and it < max_iter:
            it += 1

            max_val, active_inds, _ = max_convolution_info(a)
            cur_score = 2.0 * n * max_val   # sum(a) == 1 by construction

            # keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start:6.2f}s] Improved: n={n}, iter={it}, score={cur_score:.6f}")

            # sub‑gradient of the active max‑convolution entry/entries
            g = subgradient_from_active(a, active_inds)

            # diminishing step size
            eta = eta0 / math.sqrt(it)

            # exponentiated‑gradient update
            a = exp_gradient_step(a, g, eta)

        # store for the next length
        previous_vec = a

        # ------------------------------------------------------------------
        # 3. Local random‑swap refinement (small, cheap perturbations)
        # ------------------------------------------------------------------
        # Number of local trials is modest; each trial still needs an FFT.
        local_trials = 2000
        for _ in range(local_trials):
            if time.time() >= deadline:
                break
            # pick two distinct indices
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # propose a tiny transfer of mass (up to ±1% of a[i])
            delta = a[i] * 0.02 * (np.random.rand() - 0.5)
            if delta == 0.0:
                continue
            new_i = a[i] - delta
            new_j = a[j] + delta
            if new_i < 0.0 or new_j < 0.0:
                continue
            a_candidate = a.copy()
            a_candidate[i] = new_i
            a_candidate[j] = new_j
            # sum unchanged, no renormalisation required

            max_val_cand, _, _ = max_convolution_info(a_candidate)
            cand_score = 2.0 * n * max_val_cand
            if cand_score < cur_score - 1e-12:          # strict improvement
                a = a_candidate
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start:6.2f}s] Local improve: n={n}, iter={it}, score={cand_score:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # Return plain Python list for the evaluator
    return [float(x) for x in best_seq]
