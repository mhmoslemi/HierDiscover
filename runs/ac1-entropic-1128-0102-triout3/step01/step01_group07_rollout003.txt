"""
Propose a step‑function sequence that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

The objective is homogeneous, therefore we keep the ℓ₁‑norm of *a* equal
to one and work on the simplex.  The optimisation consists of three
phases:

1. **Diverse initialisations** – several parametric families (power‑law,
   exponential, Gaussian, triangular) are sampled for a set of lengths.
   The best few seeds are kept.

2. **Annealed Adam descent** – for each seed an Adam optimiser with a
   soft‑max approximation of the ``max`` is run.  The temperature of the
   soft‑max is slowly decreased (annealing) so that the gradient initially
   sees many convolution entries and later focuses on the true peak.
   After each step the iterate is projected back onto the probability
   simplex.

3. **Best‑ever tracking** – the true score (using the provided
   ``evaluate_sequence``) is monitored and the best sequence seen over
   the whole budget is returned.

The routine respects the 1000 s time limit, prints progress and never
writes to disk.  It can start from the previously discovered global
``height_sequence_1`` if available.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1                # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper families for initialisation
# ----------------------------------------------------------------------


def _powerlaw_sequence(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-beta)   (monotonically decreasing)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta)
    seq /= seq.sum()
    return seq


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    seq /= seq.sum()
    return seq


def _gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-0.5 * ((i-μ)/σ)^2)   centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    seq /= seq.sum()
    return seq


def _triangular_sequence(n: int, width_frac: float) -> np.ndarray:
    """
    Symmetric triangular shape.
    width_frac ∈ (0, 1] controls the support width (1 ⇒ full support).
    """
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    width = width_frac * n
    seq = np.maximum(0.0, 1.0 - np.abs(idx - centre) / (width / 2.0))
    seq /= seq.sum()
    return seq


def _interpolate_sequence(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an old simplex vector to a new length."""
    if old.size == new_n:
        return old.copy()
    x_old = np.arange(old.size, dtype=np.float64)
    x_new = np.linspace(0, old.size - 1, new_n)
    new = np.interp(x_new, x_old, old)
    # clip tiny negatives caused by numerical noise, then renormalise
    new = np.maximum(new, 0.0)
    total = new.sum()
    if total == 0:
        new = np.full(new_n, 1.0 / new_n)
    else:
        new /= total
    return new


# ----------------------------------------------------------------------
# Core optimisation routine (annealed Adam)
# ----------------------------------------------------------------------


def _annealed_adam(
    a0: np.ndarray,
    deadline: float,
    best_score: float,
    best_seq: List[float],
    n: int,
) -> tuple[float, List[float]]:
    """
    Run a projected Adam optimiser on the soft‑max–approximated max‑conv.
    Returns the (possibly) improved best_score and best_seq.
    """

    a = a0.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    step = 0
    # initial temperature proportional to the current max–convolution
    init_temp = max(np.convolve(a, a).max() * 0.5, 1e-6)
    T = init_temp
    # base learning rate – tuned for simplex geometry
    eta0 = 0.08

    while True:
        if time.time() >= deadline:
            break
        if step >= 5000:          # safeguard against endless loops
            break
        step += 1

        # ----- convolution & current score -----
        conv = np.convolve(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(
                f"[adam] New global best {best_score:.6f} (n={n}, step={step})"
            )

        # ----- soft‑max weights over convolution indices -----
        # temperature annealing (geometric decay)
        T = max(T * 0.998, 1e-8)
        logits = conv / T
        logits -= np.max(logits)               # numeric stability
        exp_logits = np.exp(logits)
        lam = exp_logits / exp_logits.sum()    # λ_k ≥ 0, Σ λ_k = 1

        # ----- gradient of the smoothed objective -----
        # g_i = 2 * Σ_k λ_k a_{k-i}
        lam_rev = lam[::-1]
        conv_rev = np.convolve(lam_rev, a)    # length = 3n‑2
        L = len(lam)                          # = 2n‑1
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        a_new = a - eta0 * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection back to the simplex -----
        a_new = _project_simplex(a_new)

        # ----- occasional jitter to escape shallow plateaus -----
        if step % 400 == 0:
            jitter = (np.random.rand(n) - 0.5) * 2e-4
            a_new = _project_simplex(np.maximum(a_new + jitter, 0.0))

        a = a_new

    return best_score, best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the supplied
    ``evaluate_sequence`` function.  The routine respects a 1000 s wall‑time
    budget, prints progress, and returns the best sequence discovered.
    """

    np.random.seed(seed)
    # leave a small safety margin for the harness
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initialise best candidate (if a previous construction exists)
    # ------------------------------------------------------------------
    best_seq: List[float] | None = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        try:
            candidate = list(height_sequence_1)
            # Normalise to ℓ₁‑norm = 1 (the optimiser works on the simplex)
            total = sum(candidate)
            if total <= 0:
                raise ValueError
            candidate = [x / total for x in candidate]
            best_seq = candidate
            best_score = evaluate_sequence(best_seq)
            print(f"[init] Using height_sequence_1, score = {best_score:.6f}")
        except Exception:
            best_seq = None
            best_score = float("inf")

    # ------------------------------------------------------------------
    # Configuration of the search
    # ------------------------------------------------------------------
    # lengths we will explore – longer lengths give a finer granularity
    lengths = [1800, 2800, 4000, 5000]

    # for each length we will try a handful of generic seeds
    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Build a pool of seed vectors for the current length
        # ------------------------------------------------------------------
        seeds: List[np.ndarray] = []

        # 1) Interpolate the current best solution (if any) to this length
        if best_seq is not None:
            old = np.array(best_seq, dtype=np.float64)
            interp = _interpolate_sequence(old, n)
            seeds.append(interp)

        # 2) Parametric families (coarse grids – cheap to generate)
        for beta in np.linspace(0.0, 3.0, 5):
            seeds.append(_powerlaw_sequence(n, beta))
        for lam in np.linspace(0.0, 1.2, 4):
            seeds.append(_exponential_sequence(n, lam))
        for sigma in (n / 20.0, n / 10.0, n / 6.0):
            seeds.append(_gaussian_sequence(n, sigma))
        for width in (0.35, 0.55, 0.80):
            seeds.append(_triangular_sequence(n, width))

        # 3) A few pure random Dirichlet draws (highly diverse)
        for _ in range(3):
            seeds.append(np.random.dirichlet(np.ones(n)))

        # ------------------------------------------------------------------
        # Evaluate each seed and launch a short Adam walk from it
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline:
                break

            # quick score on the simplex (sum == 1)
            conv = np.convolve(seed_vec, seed_vec)
            seed_score = 2.0 * n * conv.max()
            if seed_score < best_score:
                best_score = seed_score
                best_seq = seed_vec.tolist()
                print(
                    f"[seed] Length {n}, seed {idx}, raw score {seed_score:.6f} → new best"
                )

            # Run a bounded Adam optimisation starting from this seed
            # (the optimiser itself respects the remaining deadline)
            best_score, best_seq = _annealed_adam(
                a0=seed_vec,
                deadline=deadline,
                best_score=best_score,
                best_seq=best_seq if best_seq is not None else [],
                n=n,
            )

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        uniform = np.full(n, 1.0 / n)
        best_seq = uniform.tolist()
        best_score = evaluate_sequence(best_seq)
        print("[fallback] Uniform sequence used.")

    # Final sanity check with the provided evaluator
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final reported score: {final_score:.6f} (length {len(best_seq)})")
    except Exception as exc:
        # In case the evaluator misbehaves (should not happen)
        print(f"Evaluation error: {exc}")

    return best_seq
