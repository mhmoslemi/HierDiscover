def propose_candidate(seed=0,budget_s=1000, **kwargs):
    start = time.time()
    # Determine block length n0
    try:
        m = len(height_sequence_1)
    except NameError:
        m = 500  # fallback
    # Build Catalan block
    block = _catalan_sequence(m)
    # Zero block of length m
    zeros = [0.0]*m
    # Number of repetitions B: choose maybe 3 or based on budget approx length < 50000.
    B = 3
    seq = []
    for i in range(B):
        seq.extend(block)
        if i != B-1:
            seq.extend(zeros)
    # Ensure within value bounds [0,1000] already satisfied
    # Evaluate
    try:
        score = evaluate_sequence(seq)
        print(f"[propose_candidate] total_len={len(seq)}, B={B}, block_len={m}, score={score:.6f}")
    except Exception as exc:
        print("[propose_candidate] evaluation failed:", exc)
    elapsed = time.time() - start
    if elapsed > budget_s:
        print("[propose_candidate] warning: budget exceeded")
    return seq

"""
A simple yet powerful improvement over the pure Catalan construction.

The convolution‐max of the Catalan sequence is exactly 1, and its ℓ₁‑norm
grows like 2·√(n/π).  By placing two copies of this optimal block far
enough apart (with a gap of the same length as the block) the three
convolution regions – self‑convolution of the first block, cross‑convolution
between the blocks, and self‑convolution of the second block – never
overlap.  Consequently the maximal convolution entry stays equal to 1,
while the total ℓ₁‑norm is doubled.  The evaluation functional

    E(a) = 2·n·max(a*a) / (∑a)²

then becomes

    E = 3·π/8 ≈ 1.1781

independent of the block size, which is well below the target 1.503.
The implementation builds a Catalan block of length *m* (taken from the
previous candidate if available, otherwise a safe fallback), inserts a
zero gap of the same length, and repeats the block twice.

The function returns the full list ready for `evaluate_sequence`.
"""

import time

def _catalan_sequence(n: int) -> list[float]:
    """First n Catalan‑type numbers  c_i = C(2i,i) / 4**i  (c₀ = 1)."""
    seq = [1.0]               # c₀
    for i in range(1, n):
        # recurrence: c_i = c_{i-1} * (2*i-1) / (2*i)
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq
