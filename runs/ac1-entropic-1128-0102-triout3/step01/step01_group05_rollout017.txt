def project_simplex(v):
    # ensure sum = 1, all entries >=0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that (empirically)
minimises the evaluation function

    2·n·max(conv(a,a)) / (sum(a)**2)

The code uses three ideas:

1. **Analytic start** – the discretised kernel a_i = 1/√(i+off) is known to be
   close to optimal (its continuous analogue gives the constant π/2 ≈ 1.57).
   A fine one‑dimensional search over the offset yields a very good initial
   point.

2. **Projected gradient descent on a smoothed max‑convolution** – the objective
   is the maximum of a set of quadratic forms.  By replacing `max` with a
   smooth “soft‑max” (exp‑weighted average) we obtain a differentiable surrogate.
   The gradient of a weighted sum of convolutions is itself a convolution and can
   be evaluated efficiently with FFTs.  After a gradient step we project the
   vector back onto the probability simplex (non‑negative entries summing to
   one).  This drives the peak of the self‑convolution down.

3. **Final hill‑climbing polish** – occasional random mass transfers
   (preserving the ℓ¹‑norm) are used to escape tiny plateaus and to refine the
   solution.

The whole search runs for the allotted time (default 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence (no security checks).
    Returns 2*n*max(conv(a,a)) / (sum(a)**2).
    """
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())          # power‑of‑two for FFT
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, offset: float) -> np.ndarray:
    """
    a_i = 1/√(i+offset)  (i = 1 … n)   and ℓ¹‑normalise to sum = 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.sqrt(idx)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a small evaluation value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a length.  A few thousand entries give a good trade‑off.
    n = 3000

    # ------------------------------------------------------------------
    # 2.  Scan a fine grid of offsets for the analytic kernel.
    best_seq = None
    best_val = np.inf
    offsets = np.linspace(0.0, 2.0, 401)      # step = 0.005
    for off in offsets:
        seq = build_initial_sequence(n, off)
        val = fast_score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
    print(f"[init] best value after offset scan: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3.  Projected gradient descent on a smooth approximation of max‑conv.
    #    We minimise   f(a) = softmax_λ(conv(a,a))   with the simplex constraint.
    current = best_seq.copy()
    current_val = best_val
    best_seq = current.copy()
    best_val = current_val

    # hyper‑parameters
    lam = 30.0                 # smoothing parameter for soft‑max
    step = 0.08                # initial gradient step size
    min_step = 5e-5
    it = 0
    size_conv = 1 << ((2 * n - 1).bit_length())
    size_grad = 1 << ((3 * n - 2).bit_length())   # for a * w_rev

    while time.time() < deadline:
        # ----- convolution of the current iterate --------------------
        fft_a = np.fft.rfft(current, size_conv)
        conv = np.fft.irfft(fft_a * fft_a, size_conv)[: 2 * n - 1]
        max_b = conv.max()
        # evaluate (no extra FFT needed)
        current_val = float(2.0 * n * max_b / (current.sum() ** 2))
        if current_val < best_val:
            best_val, best_seq = current_val, current.copy()
            print(f"[iter {it:7d}] new best {best_val:.6f}")

        # ----- soft‑max weights ---------------------------------------
        w = np.exp(lam * (conv - max_b))          # ≤ 1, max weight = 1
        # ----- gradient of the weighted sum of convolutions ----------
        # Gradient:  g_j = 2 * Σ_i a_i * w_{i+j}
        # This is a (linear) convolution of a with the reversed w.
        w_rev = w[::-1]                           # length = 2n‑1
        fft_a2 = np.fft.rfft(current, size_grad)
        fft_w2 = np.fft.rfft(w_rev, size_grad)
        conv_full = np.fft.irfft(fft_a2 * fft_w2, size_grad)

        # the slice we need is conv_full[2n‑2‑j] for j=0…n‑1
        offset = 2 * n - 2
        grad = 2.0 * conv_full[offset - np.arange(n)]

        # ----- gradient step + projection onto the simplex ------------
        proposed = current - step * grad
        proposed = project_onto_simplex(proposed)

        # quick acceptance test (no extra FFT)
        val_proposed = fast_score(proposed)
        if val_proposed < current_val:
            current, current_val = proposed, val_proposed
        else:
            # occasional uphill move to escape plateaus
            if np.random.rand() < 0.001:
                current = proposed

        # ----- step‑size schedule ------------------------------------
        if it % 3000 == 0 and it > 0:
            step = max(step * 0.9, min_step)

        it += 1

    # ------------------------------------------------------------------
    # 4.  Final polish with simple random mass‑transfer hill‑climbing.
    #    This phase can extract a few more hundredths of improvement.
    step_ratio = 0.06
    while time.time() + 0.1 < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = step_ratio * current[i] * np.random.rand()
        new_seq = current.copy()
        new_seq[i] = max(0.0, new_seq[i] - delta)
        new_seq[j] = new_seq[j] + delta
        new_seq = project_onto_simplex(new_seq)
        val_new = fast_score(new_seq)
        if val_new < best_val:
            best_val, best_seq, current = val_new, new_seq.copy(), new_seq.copy()
            print(f"[hill {it:7d}] improved to {best_val:.6f}")
        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    # Return a plain Python list – the caller will feed it to evaluate_sequence().
    return best_seq.tolist()
