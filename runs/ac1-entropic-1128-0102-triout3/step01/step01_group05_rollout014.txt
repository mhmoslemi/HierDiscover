def lp_refine(initial_seq, max_iter, deadline):
    a = initial_seq.copy()
    n = a.size
    sum_a = a.sum()
    for i in range(max_iter):
        if time.time() >= deadline: break
        max_b, conv = compute_conv_max(a)
        M = max_b
        # Build matrix A
        A = np.zeros((2*n-1, n), dtype=np.float64)
        # Fill using broadcasting: for each j, A[j:j+n, j] = a
        for j in range(n):
            A[j:j+n, j] = a
        # Setup LP
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        if g.value is None:

        sum_g = np.sum(g.value)
        if sum_g <= sum_a * (1 + 1e-12):
            break
        # rescale g to have sum equal sum_a
        scale = sum_a / sum_g
        a = g.value * scale
        # ensure positivity
        a = np.maximum(a, 0.0)
    return a

def hill_climb(a_initial, step_rel_start=0.1, iters=50000, deadline):
    a = a_initial.copy()
    current_val = fast_score(a)
    best_a = a.copy()
    best_val = current_val
    step_rel = step_rel_start
    it = 0
    while time.time() < deadline and it < iters:
        # compute conv and max index
        n = a.size
        # compute full conv via FFT
        size = 1 << ((2*n -1).bit_length())
        fft = np.fft.rfft(a, size)
        conv = np.fft.irfft(fft*fft, size)[:2*n-1]
        max_idx = np.argmax(conv)
        # compute contributions for each i
        idx = max_idx - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contr = np.zeros_like(a)
        contr[mask] = a[mask] * a[idx[mask]]
        # choose i with probability proportional to contr (if sum>0)
        total_contr = contr.sum()
        if total_contr > 0 and np.random.rand() < 0.9:
            probs = contr / total_contr
            i = np.random.choice(n, p=probs)
        else:
            i = np.random.randint(0, n)
        # choose j uniformly random not equal to i
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        # amount to move
        delta = step_rel * a[i] * np.random.rand()
        if delta > a[i]:
            delta = a[i]
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # re-normalize sum (should be same)
        # sum unchanged but due to floating errors we renorm
        new_a /= new_a.sum()
        new_val = fast_score(new_a)
        if new_val < current_val:
            a, current_val = new_a, new_val
            if new_val < best_val:
                best_val, best_a = new_val, new_a
        else:
            # accept uphill with small probability
            if np.random.rand() < 0.01:
                a, current_val = new_a, new_val
        if it % 2000 == 0:
            print(f"[hill {it}] cur={current_val:.6f}, best={best_val:.6f}")
        # adapt step size
        if it % 5000 == 0 and it > 0:
            step_rel = max(step_rel * 0.9, 1e-5)
        it += 1
    return best_a

"""Candidate generator for minimal convolution peak.

The algorithm proceeds in three phases:
1. A dense parametric search over power‑law sequences a_i ∝ (i+Δ)^{-p}
   (with p ≈ 0.5) to obtain a good analytic start.
2. A few iterations of a linear‑programming refinement
   (maximising Σ g_i under the convolution bound of the current
   sequence) using cvxpy.  The solution is rescaled to keep the ℓ¹‑norm
   constant, which provably reduces the peak of the self‑convolution.
3. A gradient‑guided stochastic hill‑climbing that repeatedly moves
   mass away from indices that contribute most to the current maximal
   convolution value.  The move set includes directed transfers,
   occasional global exponent perturbations and a tiny uphill acceptance
   probability.

All phases respect the 1000 s budget and keep the sequence non‑negative.
The best sequence found during the run is returned as a plain Python
list of floats.
"""

import numpy as np
import time

try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:
    _CVXPY_AVAILABLE = False

def fast_score(a: np.ndarray) -> float:
    """Compute the evaluation value 2*n*max(conv(a,a))/sum(a)^2 using FFT."""
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def compute_conv_and_max(a: np.ndarray):
    """Return full linear convolution a*a and its maximum."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return conv, conv.max()

def build_power_law_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """Return a normalized sequence a_i ∝ (i+1+offset)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-p)
    a /= a.sum()
    return a

def lp_refine(seq: np.ndarray, max_iter: int, deadline: float):
    """Improve `seq` by linear programming as described in the statement."""
    if not _CVXPY_AVAILABLE:
        return seq  # fallback – cvxpy not present
    a = seq.copy()
    n = a.size
    sum_a = a.sum()
    prev_g_val = None
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # 1) current maximal convolution value M
        _, M = compute_conv_and_max(a)
        # 2) build Toeplitz matrix A where A[k,i] = a[i] if k-i ∈ [0,n-1]
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        # each column i contributes a vertically shifted copy of a
        for i in range(n):
            A[i:i + n, i] = a
        # 3) solve LP: maximise sum(g) s.t. g ≥ 0 and A@g ≤ M
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        # warm‑start if we have a previous solution
        if prev_g_val is not None:
            g.value = prev_g_val
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-9, reltol=1e-9)
        if g.value is None:
            # solver failed – abort further LP steps
            break
        sum_g = float(np.sum(g.value))
        # if the sum cannot be increased, we have reached a fixed point
        if sum_g <= sum_a * (1.0 + 1e-12):
            break
        # 4) rescale back to original ℓ¹‑norm
        scale = sum_a / sum_g
        a = np.maximum(g.value * scale, 0.0)
        prev_g_val = a  # use the rescaled vector as warm start for next round
        # small sanity check – keep sum exactly sum_a
        a /= a.sum()
        a *= sum_a
        # (optional) print progress
        # print(f"[LP iter {it}] sum_g={sum_g:.6f} → after scaling max={fast_score(a):.6f}")
    return a

def hill_climb(initial_seq: np.ndarray, deadline: float):
    """Stochastic hill climbing guided by the current maximal convolution index."""
    n = initial_seq.size
    current = initial_seq.copy()
    current_val = fast_score(current)
    best_seq = current.copy()
    best_val = current_val
    step_rel = 0.08   # relative amount of mass moved each step
    it = 0
    while time.time() < deadline:
        # compute convolution and location of the current maximum
        conv, _ = compute_conv_and_max(current)
        max_idx = int(np.argmax(conv))
        # contributions a_i * a_{max_idx-i}
        idx = max_idx - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(current)
        contrib[mask] = current[mask] * current[idx[mask]]
        # pick source index i
        if contrib.sum() > 0 and np.random.rand() < 0.85:
            probs = contrib / contrib.sum()
            i = np.random.choice(n, p=probs)
        else:
            i = np.random.randint(0, n)
        # pick destination j different from i, preferring far positions
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        # amount to transfer
        delta = step_rel * current[i] * np.random.rand()
        if delta > current[i]:
            delta = current[i]
        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        # renormalise sum (tiny numerical drift)
        cand /= cand.sum()
        # occasional global exponent perturbation
        if np.random.rand() < 0.005:
            q = np.random.uniform(0.95, 1.05)
            cand = np.power(cand, q)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
        new_val = fast_score(cand)
        if new_val < current_val:
            current, current_val = cand, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, cand.copy()
        else:
            # tiny uphill acceptance to escape plateaus
            if np.random.rand() < 0.001:
                current, current_val = cand, new_val
        # adapt step size slowly
        if it % 5000 == 0 and it > 0:
            step_rel = max(step_rel * 0.85, 1e-5)
        if it % 2000 == 0:
            print(f"[hill {it:6d}] cur={current_val:.6f}, best={best_val:.6f}")
        it += 1
    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a step‑function as a list of non‑negative heights whose
    evaluation value is as low as possible within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1. Choose sequence length.
    # Larger n gives finer resolution; 1200 offers a good trade‑off
    # for the LP phase (quadratic memory) while still being fast.
    n = 1200

    # --------------------------------------------------------------
    # 2. Coarse power‑law grid‑search to obtain a solid start.
    best_seq = None
    best_val = np.inf
    p_vals = np.linspace(0.42, 0.58, 9)          # around the optimal 0.5
    offset_vals = np.linspace(0.0, 5.0, 51)     # dense sweep 0.1 steps
    for p in p_vals:
        for off in offset_vals:
            seq = build_power_law_sequence(n, p, off)
            val = fast_score(seq)
            if val < best_val:
                best_val, best_seq = val, seq
    print(f"[init] best after power‑law scan: {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. LP‑based refinement (few iterations, uses cvxpy if available).
    if _CVXPY_AVAILABLE:
        lp_deadline = deadline - 150.0          # reserve time for hill climb
        if lp_deadline > time.time():
            refined = lp_refine(best_seq, max_iter=12, deadline=lp_deadline)
            refined_val = fast_score(refined)
            print(f"[LP] value after refinement: {refined_val:.6f}")
            if refined_val < best_val:
                best_val, best_seq = refined_val, refined
    else:
        print("[LP] cvxpy not available – skipping LP refinement.")

    # --------------------------------------------------------------
    # 4. Gradient‑guided stochastic hill climbing.
    hill_deadline = deadline
    final_seq = hill_climb(best_seq, hill_deadline)
    final_val = fast_score(final_seq)
    print(f"[final] best value after hill climbing: {final_val:.6f}")

    # Return a plain Python list (the caller will run evaluate_sequence on it)
    return final_seq.tolist()

        if j >= i:
            j += 1

        # transfer a small amount of mass from i to j
        delta = step * cur[i] * np.random.rand()
        if delta > cur[i]:
            delta = cur[i]
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # occasional global exponent perturbation (helps escape plateaus)
        if np.random.rand() < 0.005:
            q = np.random.uniform(0.95, 1.05)
            cand = np.power(cand, q)
            cand = np.maximum(cand, 0.0)

        # re‑normalise to keep the ℓ¹‑norm exactly 1 (numerical safety)
        cand /= cand.sum()

        new_val = fast_score(cand)

        # deterministic improvement, otherwise tiny uphill probability
        if new_val < cur_val:
            cur, cur_val = cand, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, cand.copy()
        else:
            if np.random.rand() < 0.001:
                cur, cur_val = cand, new_val

        # adapt step size slowly
        if it % 5000 == 0 and it > 0:
            step = max(step * 0.85, 1e-5)

        # occasional progress report
        if it % 2000 == 0:
            print(f"[hill {it:6d}] cur={cur_val:.6f}, best={best_val:.6f}")

        it += 1

    return best_seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation metric.  Returns the best sequence found as a list of
    floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # -----------------------------------------------------------------
    # 1. Length of the sequence.
    #    1200 keeps the LP matrices moderate (≈23 MiB) while still
    #    providing a fine enough discretisation.
    n = 1200

    # -----------------------------------------------------------------
    # 2. Grid‑search over power‑law parameters (p, offset) for a solid start.
    best_seq = None
    best_val = np.inf
    p_vals = np.linspace(0.42, 0.58, 9)               # around the theoretical 0.5
    off_vals = np.linspace(0.0, 5.0, 51)             # step 0.1
    for p in p_vals:
        for off in off_vals:
            seq = power_law_sequence(n, p, off)
            val = fast_score(seq)
            if val < best_val:
                best_val, best_seq = val, seq
    print(f"[init] best after power‑law scan: {best_val:.6f}")

    # -----------------------------------------------------------------
    # 3. Linear‑programming refinement (if possible).  Reserve enough
    #    time for the final hill‑climb.
    if _CVX_AVAILABLE:
        lp_deadline = deadline - 150.0
        if lp_deadline > time.time():
            refined = lp_refine(best_seq, max_iter=12, deadline=lp_deadline)
            refined_val = fast_score(refined)
            print(f"[LP] value after refinement: {refined_val:.6f}")
            if refined_val < best_val:
                best_val, best_seq = refined_val, refined
    else:
        print("[LP] cvxpy not available – skipping LP refinement.")

    # -----------------------------------------------------------------
    # 4. Gradient‑guided stochastic hill climbing for the remaining time.
    final_seq = hill_climb(best_seq, deadline)
    final_val = fast_score(final_seq)
    print(f"[final] best value after hill climbing: {final_val:.6f}")

    # Return a plain Python list – the outer harness will call evaluate_sequence.
    return final_seq.tolist()
