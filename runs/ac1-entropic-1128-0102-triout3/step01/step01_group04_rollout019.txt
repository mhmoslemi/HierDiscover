def _gradient_descent(seq, max_iter, deadline):
    n = len(seq)
    target_sum = _target_sum(n)
    best_seq = seq.copy()
    conv = np.convolve(seq, seq)
    best_score = float(2*n*conv.max()/(seq.sum()**2))

    step_size = seq.mean() * 0.1  # maybe adjust
    alpha = 2.0

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute conv
        conv = np.convolve(seq, seq)
        # compute smooth max weights
        # choose alpha dynamics: increase linearly: alpha = 2.0 + 8.0*it/max_iter (max 10)
        alpha = 2.0 + 8.0*it/max_iter
        w = np.exp(alpha*conv)
        denom = w.sum()
        # gradient
        a_rev = seq[::-1]
        # conv_rev = np.convolve(w, a_rev)
        conv_rev = np.convolve(w, a_rev)  # length ~ 2n-1 + n -1
        # extract S for each k
        S = conv_rev[(n-1)-np.arange(n)]
        grad = 2.0 * S / denom
        # gradient descent update
        seq = seq - step_size * grad
        # clip and renormalize
        seq = np.maximum(seq, 0.0)
        # avoid sum==0
        seq_sum = seq.sum()
        if seq_sum == 0:
            # reset to uniform
            seq = np.full(n, target_sum/n)
            seq_sum = target_sum
        seq *= target_sum / seq_sum

        # optional adjust step size
        step_size *= 0.9995

        # compute current score
        conv = np.convolve(seq, seq)
        score = float(2*n*conv.max()/(seq.sum()**2))

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    return best_seq, best_score

def _simulated_annealing(seq, max_steps, target_sum, deadline):
    n = len(seq)
    cur_seq = seq.copy()
    cur_score = _calc_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    T0 = 1.0
    T_end = 0.001
    T = T0

    for step in range(max_steps):
        if time.time() >= deadline:
            break
        # propose move: transfer δ
        i, j = np.random.choice(n, 2, replace=False)
        # choose δ relative to a_i
        max_delta = cur_seq[i]
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta * 0.5  # up to 50% of a_i
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate
        new_score = _calc_score(new_seq)

        delta_score = new_score - cur_score
        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / T):
            # accept
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
        # temperature schedule
        T = T0 * (1 - step / max_steps) ** 3  # or linear
        # optional early exit if T small and no improvement for long time

    return best_seq, best_score

def _refine_sequence(seq, deadline):
    n = len(seq)
    target_sum = _target_sum(n)
    cur_seq = seq.copy()
    cur_score = _calc_score(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score

    T_initial = 1.0
    T_final = 1e-4
    # number of steps approx
    step = 0
    # we'll compute total allowed steps as:
    max_steps = int(1e5) # large but we break by time.
    while time.time() < deadline and step < max_steps:
        step += 1
        # temperature schedule linear
        T = max(T_final, T_initial * (1 - step / max_steps))
        # propose move
        i, j = np.random.choice(n, 2, replace=False)
        if cur_seq[i] <= 0:
            continue
        # delta as fraction of i's value
        delta = cur_seq[i] * (np.random.rand() * 0.1)  # up to 10% of i's value
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure nonnegative
        if new_seq[i] < 0:
            new_seq[i] = 0
        # compute score
        new_score = _calc_score(new_seq)
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / T):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
    return best_seq, best_score

def _transfer_sa(seq, deadline, max_iter=5000):
    n = len(seq)
    cur_seq = seq.copy()
    cur_score = _calc_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    # temperature schedule parameters
    T_initial = 1.0
    T_final = 1e-4
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # linear schedule
        T = T_initial * (1 - it / max_iter) + T_final * (it / max_iter)
        # propose transfer
        i, j = np.random.choice(n, 2, replace=False)
        if cur_seq[i] <= 0.0:
            continue
        # propose delta as random fraction of cur_seq[i]
        delta = cur_seq[i] * np.random.rand() * 0.3  # up to 30% of i value
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # compute score
        new_score = _calc_score(new_seq)
        # accept if better or with probability
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score)/T):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
    return best_seq, best_score

def _gradient_descent(seq, max_iter, deadline):
    n = len(seq)
    target_sum = _target_sum(n)
    best_seq = seq.copy()
    best_score = _calc_score(seq)
    step = seq.mean() * 0.1 if seq.mean()>0 else 0.01
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute conv and grad
        conv = np.convolve(seq, seq)
        max_conv = conv.max()
        # compute smooth weights
        alpha = 5.0 + 15.0 * it / max_iter   # from 5 to 20
        # subtract max to avoid overflow
        cm = conv.max()
        w = np.exp(alpha * (conv - cm))
        weight_sum = w.sum()
        # gradient
        a_rev = seq[::-1]
        corr = np.convolve(w, a_rev)
        # extracts S_k for k=0..n-1
        S = corr[(n-1)-np.arange(n)]
        grad = 2.0 * S / weight_sum
        # update
        seq = seq - step * grad
        # clip and renorm
        seq = np.maximum(seq, 0.0)
        ssum = seq.sum()
        if ssum == 0:
            seq = np.full(n, target_sum / n)
        else:
            seq *= target_sum / ssum
        # evaluate
        score = _calc_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        # decay step
        step *= 0.9995
    return best_seq, best_score

"""
Optimise a non‑negative step‑function to minimise

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The evaluation inside the competition normalises the L¹‑norm to
√(2·n); consequently the score is simply the maximal convolution
value.  The optimiser works in three stages

1. **Diverse initialisations** – several parametric families are used:
   * √‑derivative (a_i ∝ √(i+1+δ)−√(i+δ))
   * Power‑law (a_i ∝ (i+δ)^(−α))
   * Symmetric product (a_i ∝ ((i+δ)(n‑1‑i+δ))^p)

   Random parameters (δ, α, p) are drawn and the sequence is scaled
   to the required L¹‑norm.

2. **Stochastic transfer annealing** – the current sequence `a` is
   repeatedly modified by moving a random amount of mass from one
   index to another (the total sum stays unchanged).  Moves that
   improve the score are always accepted; worsening moves are accepted
   with a temperature‑controlled probability (simulated annealing).
   This can lower the maximal convolution far beyond a simple
   single‑coordinate hill‑climb.

3. **Smooth‑max gradient descent** – after the annealing stage the
   non‑smooth objective is approximated by a log‑sum‑exp of the
   convolution values.  The gradient is computed analytically
   (∂/∂a_k  max ≈ 2·Σ_i w_i a_{i−k} / Σ_i w_i) and a projected
   gradient step (non‑negativity + L¹‑renormalisation) is performed.
   The step size decays gently and the best point encountered is kept.

The outer loop launches many restarts with different lengths (up to a
few thousand entries) until the time budget expires.  The best sequence
seen overall is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Required L¹‑norm: Σ a_i = √(2·n)."""
    return np.sqrt(2.0 * n)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """Derivative of √(x+δ) (the classic starting shape)."""
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _power_law_sequence(n: int, alpha: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^(−α)   (α≈0.5 reproduces the √‑derivative)."""
    i = np.arange(n, dtype=float)
    base = (i + delta) ** (-alpha)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _sym_product_sequence(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ ((i+δ)(n‑1‑i+δ))^p  – symmetric, centrally peaked when p>0."""
    i = np.arange(n, dtype=float)
    base = ((i + delta) * (n - 1 - i + delta)) ** p
    scale = _target_sum(n) / base.sum()
    return base * scale


def _score(seq: np.ndarray) -> float:
    """Exact score used by the evaluator (avoids the extra checks)."""
    conv = np.convolve(seq, seq)
    max_conv = np.max(conv)
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    n = seq.size
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Stochastic transfer annealing (keeps the L¹‑norm exactly)
# ----------------------------------------------------------------------


def _transfer_annealing(seq: np.ndarray,
                        deadline: float,
                        max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Move a random amount of mass from one index to another.
    Accept with Metropolis rule (temperature decays linearly).
    Returns the best sequence found and its score.
    """
    n = seq.size
    cur = seq.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    T_start, T_end = 1.0, 1e-4

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iter)

        i, j = np.random.choice(n, 2, replace=False)

        if cur[i] == 0.0:
            continue

        # transfer a random fraction of the donor entry
        delta = cur[i] * np.random.rand() * 0.3  # up to 30 % of a_i
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp((cur_score - cand_score) / T):
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score

    return best, best_score


# ----------------------------------------------------------------------
# Smooth‑max gradient descent (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def _smooth_max_descent(seq: np.ndarray,
                        max_iter: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimises the max‑convolution using a smooth log‑sum‑exp surrogate.
    Each step projects back onto the non‑negative simplex (fixed L¹‑norm).
    """
    n = seq.size
    target = _target_sum(n)

    best = seq.copy()
    best_score = _score(seq)

    step = max(1e-8, seq.mean() * 0.1)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # convolution of the current sequence
        conv = np.convolve(seq, seq)

        # smooth‑max parameters – start gentle, finish hard
        alpha = 5.0 + 15.0 * (it / max_iter)  # 5 → 20

        # avoid overflow: shift by the maximum value
        cmax = conv.max()
        w = np.exp(alpha * (conv - cmax))
        w_sum = w.sum()

        # gradient of the smooth max w.r.t. the coefficients
        rev = seq[::-1]
        corr = np.convolve(w, rev)                      # length 3n‑2
        S = corr[(n - 1) - np.arange(n)]                # needed slice
        grad = 2.0 * S / w_sum

        # gradient step
        seq = seq - step * grad

        # projection: non‑negative & correct L¹‑norm
        np.maximum(seq, 0.0, out=seq)
        total = seq.sum()
        if total == 0.0:
            seq = np.full(n, target / n)
        else:
            seq *= target / total

        cur_score = _score(seq)
        if cur_score < best_score:
            best, best_score = seq.copy(), cur_score

        # gentle decay of the step size
        step *= 0.9995

    return best, best_score

            seq = _sym_product_sequence(n, p=p, delta=delta)

        # ---- quick evaluation of the raw start ----
        init_score = _score(seq)
        if init_score < best_score:
            best_seq, best_score = seq.copy(), init_score
            print(f"[init] family={family:4s} n={n:4d} score={init_score:.6f}")

        # ---- refinement by simulated‑annealing (transfer moves) --------------
        inner_deadline = time.time() + 0.4   # ~0.4 s per restart
        seq_opt, score_opt = _transfer_annealing(seq,
                                                 deadline=inner_deadline,
                                                 max_iter=8000)
        if score_opt < best_score:
            best_seq, best_score = seq_opt.copy(), score_opt
            print(f"[anneal] family={family:4s} n={n:4d} score={score_opt:.6f}")

        # ---- occasional extra gradient descent if time permits -------------
        remaining = deadline - time.time()
        if remaining > 2.0 and best_seq is not None:
            gd_deadline = time.time() + min(1.5, remaining * 0.5)
            seq_gd, score_gd = _smooth_max_descent(best_seq,
                                                   max_iter=15000,
                                                   deadline=gd_deadline)
            if score_gd < best_score:
                best_seq, best_score = seq_gd.copy(), score_gd
                print(f"[gd]    n={n:4d} score={score_gd:.6f}")

        # early stop if we already beat the informal target
        if best_score < 1.52:
            # keep searching a bit longer to see if we can push below 1.5
            pass

    # ------------------------------------------------------------------
    # Fallback – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 100
        best_seq = np.full(n, _target_sum(n) / n)

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
