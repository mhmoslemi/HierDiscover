def propose_candidate(seed=0,budget_s=1000):
    import time, numpy as np, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    best_seq = None
    best_score = float('inf')

    # helper: compute convolution (linear) via FFT
    def conv_linear(x):
        n = x.shape[0]
        L = 1 << ((2*n - 1).bit_length())
        X = np.fft.rfft(x, n=L)
        c = np.fft.irfft(X * X, n=L)[:2*n-1]
        return c

    # helper: gradient of smoothed max
    def gradient_softmax(conv, a, p):
        # conv length = 2n-1
        n = a.shape[0]
        max_val = conv.max()
        # compute softmax weights, shift by max to avoid overflow
        w = np.exp(p * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/ w.size)
        else:
            w = w / w_sum
        # compute gradient = 2 * conv(w_rev, a) at appropriate indices
        w_rev = w[::-1]
        L2 = 1 << (( (2*n-1) + n - 1).bit_length())
        W = np.fft.rfft(w_rev, n=L2)
        A = np.fft.rfft(a, n=L2)

        idx = 2*n - 2 - np.arange(n)
        grad = 2.0 * conv_rev[idx]
        return grad

    # set multiscale schedule
    lengths = [200, 400, 800, 1200, 1800, 2400, 3000, 4000]
    # If we have time, we can go up to 5000
    np.random.shuffle(lengths)
    # start with a basic sequence to upsample
    base_a = None

    for n in lengths:
        if time.time() >= deadline:
            break
        # initialise a
        if base_a is not None:
            # upsample base_a from prev_n to n via linear interpolation
            prev_n = base_a.shape[0]
            # simple linear interpolation
            x_old = np.linspace(0, 1, prev_n)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, base_a)
            a = np.maximum(a, 0)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = np.random.rand(n)
            a = a / a.sum()
        else:
            # random Dirichlet
            a = np.random.rand(n)
            a = a / a.sum()

        # schedule for smoothing p: start small, increase linearly toward p_max

            p = p0 + (p_max - p0) * min(1.0, (it / max_iter))

            grad = gradient_softmax(conv_val, a, p)

            # mirror descent update
            eta = eta0 / math.sqrt(it)   # diminishing

            a = a * np.exp(-eta * grad)
            # renormalize to sum 1
            s = a.sum()

        base_a = a.copy()
    # final fallback
    if best_seq is None:
        best_seq = np.full(100, 1.0/100)
    # return list of floats
    return [float(x) for x in best_seq]

            # renormalize to simplex
            s = a.sum()

            # occasional jitter to escape flat region

                a = a + jitter
                a = np.maximum(a, 0)
                a = a / a.sum()

        base_a = a.copy()

    # final fallback
    if best_seq is None:
        best_seq = np.full(100, 1.0/100)
    # Convert to Python list
    return [float(x) for x in best_seq]

"""
Propose a step‑function using a multiscale mirror‑descent on a smoothed
∞‑norm of the auto‑convolution.

The evaluation function is scale‑invariant, therefore we fix the sum of the
coefficients to 1 (simplex).  The objective  max_k (a∗a)_k is approximated by a
smooth “soft‑max’’:
    f_p(a) = (1/p)·log Σ_k exp(p·(a∗a)_k) .
Its gradient w.r.t. a is
    ∇f_p(a)_i = 2·(w_rev ∗ a)_{2n‑2‑i},
where w_k = exp(p·(a∗a)_k)/Σ_j exp(p·(a∗a)_j) and w_rev is w reversed.
For small p the landscape is smooth and the algorithm can move far; as p grows
the gradient approaches the sub‑gradient of the true max‑function.
We use an exponential (mirror‑descent / KL‑projection) update
    a_i ← a_i·exp(‑η·∇f_p(a)_i) ,   a ← a / Σ a,
with a diminishing step η = η₀/√t.

The optimisation proceeds on a schedule of lengths (coarse → fine).  A good
solution for a smaller length is up‑sampled (linear interpolation) and used as
the start for the next, larger length.  If a previous best sequence is
available in the global variable `height_sequence_1` it is used as a warm start.
Occasional tiny random jitter helps to escape plateaus.
The routine runs for at most `budget_s` seconds and returns the best sequence
found.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The search respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ----------------------------------------------------------------------
    # keep the best sequence seen so far
    best_seq = None
    best_score = float('inf')

    # If a previous construction is present, start from it
    prev = globals().get('height_sequence_1')
    if isinstance(prev, (list, tuple)) and prev:
        try:
            sc = evaluate_sequence(prev)          # type: ignore[name-defined]
            best_seq = [float(x) for x in prev]
            best_score = sc
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"Starting from previous seq (len={len(prev)}) score={sc:.6f}")
        except Exception:
            prev = None
    else:
        prev = None

    # ----------------------------------------------------------------------
    # lengths to explore – start coarse, finish fine
    lengths = [200, 400, 800, 1200, 1800, 2400, 3000, 4000, 5000]
    if prev:
        n0 = len(prev)
        if n0 not in lengths:
            lengths.append(n0)

    # make the order random to avoid systematic bias
    np.random.shuffle(lengths)

    # ----------------------------------------------------------------------
    # helper: full linear convolution via FFT (real‑valued)
    def conv_linear(x: np.ndarray, L_fft: int) -> np.ndarray:
        """Return x∗x (linear) using FFT of length L_fft (power of two)."""
        X = np.fft.rfft(x, n=L_fft)
        c = np.fft.irfft(X * X, n=L_fft)
        return c[: 2 * x.shape[0] - 1]

    # ----------------------------------------------------------------------
    # main loop over lengths
    base_a = None                # will hold the solution of the previous length
    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- initialise a -------------------------------------------------
        if base_a is not None and base_a.shape[0] != n:
            # up‑sample previous solution by linear interpolation
            x_old = np.linspace(0.0, 1.0, base_a.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, base_a)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = np.random.rand(n)
                a_sum = a.sum()
            a = a / a_sum
        elif base_a is not None and base_a.shape[0] == n:
            a = base_a.copy()
        elif prev is not None and len(prev) == n:
            a = np.array(prev, dtype=float)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = np.random.rand(n)
                a_sum = a.sum()
            a = a / a_sum
        else:
            a = np.random.rand(n)
            a = a / a.sum()

        # ---- hyper‑parameters (tuned empirically) -------------------------
        max_iter = 20000                     # per length
        eta0 = 0.7                           # initial mirror‑descent step
        p0 = 5.0                             # start smoothing (soft‑max) factor
        p_max = 200.0                        # final smoothing factor
        p_transition = 30.0                  # below this we use the soft gradient
        fft_len_conv = 1 << ((2 * n - 1).bit_length())          # for a∗a
        fft_len_grad = 1 << ((3 * n - 2).bit_length())          # for w_rev∗a

        # main optimisation loop for this length
        for it in range(1, max_iter + 1):
            if time.time() >= deadline:

            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:

            p = p0 + (p_max - p0) * min(1.0, it / max_iter)

                grad = 2.0 * conv_rev[idx]
            else:
                # ----- pure sub‑gradient (active max entry) ---------------
                grad = np.zeros_like(a)
                low = max(0, k_max - (n - 1))
                high = min(n - 1, k_max)
                i_idx = np.arange(low, high + 1)
                j_idx = k_max - i_idx
                grad[i_idx] = 2.0 * a[j_idx]

            # ----- mirror‑descent update (KL‑projection) ------------------
            eta = eta0 / math.sqrt(it)          # diminishing step size
            a = a * np.exp(-eta * grad)         # element‑wise
            s = a.sum()
            if s <= 0.0 or not np.isfinite(s):
                a = np.full(n, 1.0 / n)
            else:
                a = a / s

                a = a + jitter

                a = a / a.sum()

        # store the result as the starting point for the next (larger) length
        base_a = a.copy()

    # ----------------------------------------------------------------------
    # fallback – should never happen, but keeps the contract safe.
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # Convert to plain Python list (the evaluator accepts any sequence of floats)
    return [float(x) for x in best_seq]
