def solve_lp_direction(a, conv, M):
    n = len(a)
    # Build B matrix of shape (2n-1, n)
    n2 = 2*n -1
    # First column: a padded with zeros length n2
    c = np.concatenate([a, np.zeros(n2 - n, dtype=np.float64)])
    # First row: a[0] then zeros
    r = np.zeros(n, dtype=np.float64)
    r[0] = a[0]
    B = scipy.linalg.toeplitz(c, r) # shape (n2, n)
    # Compute C = 2 * B
    C = 2.0 * B

    # Build A_ub and b_ub
    # A_ub shape (n2, n+1)
    A_ub = np.hstack([C, np.ones((n2, 1), dtype=np.float64)])
    b_ub = M - conv  # shape (n2,)

    # Equality for sum(delta)=0
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([0.0], dtype=np.float64)

    # Bounds: delta_j >= -a_j, delta_j <= None
    bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]  # epsilon >=0

    # objective: maximize epsilon -> minimize -epsilon
    c_obj = np.zeros(n+1, dtype=np.float64)
    c_obj[-1] = -1.0

    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'presolve': True})
    if not result.success:
        return None, 0.0
    delta = result.x[:n]
    epsilon = result.x[-1]
    return delta, epsilon

def solve_direction(a, conv, M):
    n = len(a)
    n2 = 2 * n - 1
    # Build Toeplitz matrix B
    c = np.concatenate((a, np.zeros(n2 - n, dtype=np.float64)))
    r = np.zeros(n, dtype=np.float64)
    r[0] = a[0]
    B = scipy.linalg.toeplitz(c, r)   # shape (n2, n)
    C = 2.0 * B    # derivative coefficient

    # Build constraints
    A_ub = np.hstack((C, np.ones((n2, 1), dtype=np.float64)))
    b_ub = M - conv   # shape n2

    A_eq = np.hstack((np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)))
    b_eq = np.array([0.0], dtype=np.float64)

    bounds = [( -a[i], None) for i in range(n)] + [(0.0, None)]

    c_obj = np.zeros(n+1, dtype=np.float64)
    c_obj[-1] = -1.0

    result = scipy.optimize.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                   bounds=bounds, method='highs')
    if not result.success:
        return None, 0.0
    delta = result.x[:n]
    eps = result.x[-1]
    return delta, eps

We keep a non‑negative vector `a` with ∑a = 1 (the L1‑normalisation used in the
evaluation function).  In each iteration we solve a small linear program

                ε ≥ 0

convolution peak decreases.  The best vector seen during the run is returned.

longer improve the bound.  The implementation uses only NumPy, SciPy and the
provided `evaluate_sequence` function.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑x)^(-1/2) – the classical “half‑binomial”
    sequence, normalised to have L1‑norm = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x│x_i ≥ 0, Σx_i = 1}.  Implements the O(n) algorithm of Duchi et al.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """Discrete convolution x*x using a real FFT (length 2·len(x)-1)."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    return np.fft.irfft(X * X, n=m)


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine

def _lp_direction(a: np.ndarray, conv: np.ndarray, M: float):
    """
    Solve the LP that yields the best feasible direction δ and the guaranteed
    reduction ε of the current maximal convolution value M.

    Returns
    -------
    delta : np.ndarray of shape (n,)
        Feasible direction (sum(delta)=0, a+delta≥0).
    eps   : float
        Guaranteed reduction of the maximal convolution (non‑negative).
    """
    n = a.shape[0]
    n2 = 2 * n - 1

    # Toeplitz matrix that implements (a * δ) for all 2n‑1 indices.
    # First column: a followed by zeros; first row: a[0] then zeros.
    c = np.concatenate([a, np.zeros(n2 - n, dtype=np.float64)])
    r = np.zeros(n, dtype=np.float64)
    r[0] = a[0]
    B = scipy.linalg.toeplitz(c, r)          # shape (2n‑1, n)
    C = 2.0 * B                              # derivative of conv w.r.t. δ

    # Inequalities:  sum_j C[i, j]·δ_j + ε ≤ M - conv[i]   (i=0..2n‑2)
    A_ub = np.hstack([C, np.ones((n2, 1), dtype=np.float64)])
    b_ub = M - conv

    # Equality: Σ δ_j = 0
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64),
                      np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([0.0], dtype=np.float64)

    # Variable bounds: δ_j ≥ –a_j   (to keep a+δ ≥ 0),   ε ≥ 0
    bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]

    # Objective: maximise ε  ⇒ minimise –ε
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = -1.0

    # Solve with HiGHS (fast dense LP solver)
    res = scipy.optimize.linprog(c_obj,
                                 A_ub=A_ub, b_ub=b_ub,
                                 A_eq=A_eq, b_eq=b_eq,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        return None, 0.0
    delta = res.x[:n]
    eps = res.x[-1]
    return delta, eps


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises
        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The algorithm:
    * start from the half‑binomial (sqrt) sequence (∑a = 1);
    * repeatedly solve a small linear program that gives a feasible descent
      direction for the peak of the auto‑convolution;
    * apply the direction with a back‑tracking line‑search;
    * keep the best vector ever seen;
    * stop when the time budget runs out or when the LP can no longer improve.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Parameters – they can be tweaked if more time is available
    n = 1500                     # length of the sequence (adjustable)
    max_iters = 5000            # safety bound; loop also stops by deadline
    min_eps = 1e-12             # stop if LP reports negligible improvement
    line_search_factor = 0.5    # geometric reduction when a step is not accepted

    # initialise from the classic half‑binomial series (L1‑norm = 1)
    a = sqrt_series(n)          # already sums to 1
    conv = conv_fft(a)
    M = conv.max()
    cur_score = 2.0 * n * M
    best_a = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # small safety margin
    it = 0

    while time.time() < deadline and it < max_iters:
        # recompute convolution and peak
        conv = conv_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[iter {it:06d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 1) LP that returns a feasible descent direction δ and the guaranteed
        #    reduction ε of the current peak.
        delta, eps = _lp_direction(a, conv, M)

        # If the LP fails or can’t improve, fall back to a tiny random move.
        if delta is None or eps < min_eps:
            # random perturbation and projection onto the simplex – helps to
            # escape possible degenerate situations.
            noise = (rng.random(n) - 0.5) * 1e-6
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = project_onto_simplex(a)
            it += 1
            continue

        # --------------------------------------------------------------
        # 2) Back‑tracking line search on λ ∈ (0,1] to obtain an actual improvement.
        lam = 1.0
        accepted = False
        while lam > 1e-4:
            a_trial = a + lam * delta
            # a_trial is guaranteed to be non‑negative because delta ≥ –a
            conv_trial = conv_fft(a_trial)
            M_trial = conv_trial.max()
            trial_score = 2.0 * n * M_trial
            if trial_score < cur_score - 1e-12:
                a = a_trial
                accepted = True
                break
            lam *= line_search_factor

        # If even the smallest step does not improve, perform a tiny random nudge.
        if not accepted:
            noise = (rng.random(n) - 0.5) * 1e-7
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = project_onto_simplex(a)

        it += 1

    # ------------------------------------------------------------------
    # Final evaluation with the official routine (ensures all the security
    # checks are applied) and return the best sequence found.
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
