def build_conv_matrix(f):
    n = len(f)
    zeros_left = np.zeros(n-1, dtype=np.float64)
    # column j: zeros_left of length j then f then zeros_left of length n-1-j yields length 2n-1.
    # We can allocate A = np.zeros((2*n-1, n))
    A = np.empty((2*n-1, n), dtype=np.float64)
    for j in range(n):
        start = j
        end = j+n
        A[start:end, j] = f
    return A

"""
Propose a step‑function sequence that minimizes `evaluate_sequence`.

Algorithm (alternating linear‑programming refinement)
----------------------------------------------------
1. Fix the ℓ¹‑norm  S = √(2·n)  (the objective is scale‑invariant).
2. Start from an initial non‑negative sequence `f` on the simplex
   (uniform or the previously found `height_sequence_1`).
3. Loop while time remains
   a) Compute the self‑convolution `c = f * f` (FFT) and its peak M.
   b) Build the linear convolution matrix A such that  A·g = f * g .
      For each column j, A contains a copy of `f` shifted by j positions.
   c) Solve the linear program
         maximise    1ᵀ g
         subject to  A·g ≤ M ,   g ≥ 0 .
      (cvxpy + ECOS, sparse matrix)
   d) Rescale the LP solution to have ℓ¹‑norm S:
          g ← g·(S/∑g) .
   e) Mix the new vector with the current one:
          f ← (1‑t)·f + t·g   (t starts at 0.5 and slowly decays).
      Project the mixture back onto the simplex to correct rounding.
   f) Keep the best `f` seen so far (lowest value of
          2·n·max(f*f) / S² = evaluate_sequence ).
   g) If no improvement for many iterations, restart from a random Dirichlet
      point on the simplex.
4. Return the best sequence found (as a plain ``list[float]``).

The LP step directly respects the active convolution constraints,
often giving a stronger descent direction than the smooth‑max
gradient descent used previously.  The method is fast enough for
n≈2400 and fits well inside the 1000 s budget.
"""

import time
import numpy as np
import cvxpy as cp
from scipy.sparse import diags, csr_matrix


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real output)."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_matrix_sparse(f: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that A @ g = f * g.
    Each column j of A contains the vector f shifted by j rows.
    """
    n = f.shape[0]
    # create a list with n copies of f – one for each diagonal offset
    data = [f.copy() for _ in range(n)]
    offsets = np.arange(n)           # 0 … n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low `evaluate_sequence` value.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -----------------------------------------------------------------
    # Problem size and ℓ¹‑norm
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = init_seq.size
        # normalise to the required ℓ¹‑norm
        S = np.sqrt(2.0 * n)
        cur_sum = init_seq.sum()
        if cur_sum > 0:
            f = init_seq * (S / cur_sum)
        else:
            f = np.full(n, S / n, dtype=np.float64)
    else:
        n = 2401                     # a good default length (odd → symmetric)
        S = np.sqrt(2.0 * n)
        f = np.full(n, S / n, dtype=np.float64)   # uniform start

    # -----------------------------------------------------------------
    # bookkeeping
    # -----------------------------------------------------------------
    best_f = f.copy()
    best_score = 2.0 * n * _convolve_self_fft(f, _next_pow2(2 * n - 1)).max() / (S ** 2)
    print(f"Initial score (norm‑scaled) = {best_score:.6f}")

    # mixing schedule
    t = 0.5                # initial mixing weight
    t_min = 0.02
    decay_t = 0.9999

    # stagnation handling
    no_improve = 0
    max_no_improve = 200

    iteration = 0
    while time.time() < deadline:
        # -------------------------------------------------------------
        # 1) current self‑convolution and its maximum M
        # -------------------------------------------------------------
        nfft = _next_pow2(2 * n - 1)
        conv_f = _convolve_self_fft(f, nfft)
        M = conv_f.max()
        cur_score = 2.0 * n * M / (S ** 2)

        # -------------------------------------------------------------
        # 2) build sparse convolution matrix for f * g
        # -------------------------------------------------------------
        A = _build_conv_matrix_sparse(f)      # shape (2n‑1, n)

        # -------------------------------------------------------------
        # 3) solve LP: maximise 1ᵀ g  s.t.  A·g ≤ M , g ≥ 0
        # -------------------------------------------------------------
        g_var = cp.Variable(n)
        lp = cp.Problem(cp.Maximize(cp.sum(g_var)),
                         [A @ g_var <= M,
                          g_var >= 0])
        try:
            lp.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception as e:
            print(f"[{iteration}] LP solve failed: {e}")
            # fallback to a random restart
            f = np.random.dirichlet(np.full(n, 0.3)) * S
            no_improve = 0
            iteration += 1
            continue

        if g_var.value is None:
            # infeasible or solver gave no solution → random restart
            f = np.random.dirichlet(np.full(n, 0.3)) * S
            no_improve = 0
            iteration += 1
            continue

        g_opt = np.maximum(g_var.value, 0.0)   # enforce non‑negativity numerically
        sum_g = g_opt.sum()
        if sum_g <= 0:
            # degenerate solution → random restart
            f = np.random.dirichlet(np.full(n, 0.3)) * S
            no_improve = 0
            iteration += 1
            continue

        # -------------------------------------------------------------
        # 4) rescale g to have ℓ¹‑norm S and mix with current f
        # -------------------------------------------------------------
        g_opt = g_opt * (S / sum_g)
        mixed = (1.0 - t) * f + t * g_opt
        mixed = _project_to_simplex(mixed, S)   # correct possible numeric drift

        # -------------------------------------------------------------
        # 5) evaluate mixed candidate
        # -------------------------------------------------------------
        conv_mixed = _convolve_self_fft(mixed, nfft)
        mixed_score = 2.0 * n * conv_mixed.max() / (S ** 2)

        # keep the best ever seen
        if mixed_score < best_score:
            best_score = mixed_score
            best_f = mixed.copy()
            print(f"[{iteration:05d}] New best score = {best_score:.6f} (t={t:.4f})")
            no_improve = 0
        else:
            no_improve += 1

        # accept the mixed point only if it does not increase the true peak
        if mixed_score <= cur_score:
            f = mixed
        # otherwise keep the old f (the LP step is guaranteed not to worsen M
        # for the theoretical mixing scheme, but numerical noise may happen)

        # -------------------------------------------------------------
        # 6) schedule updates
        # -------------------------------------------------------------
        t = max(t * decay_t, t_min)
        iteration += 1

        # -------------------------------------------------------------
        # 7) restart on stagnation
        # -------------------------------------------------------------
        if no_improve >= max_no_improve:
            f = np.random.dirichlet(np.full(n, 0.3)) * S
            no_improve = 0
            print(f"[{iteration}] Stagnation – random restart.")

    # -----------------------------------------------------------------
    # final verification and return
    # -----------------------------------------------------------------
    final_seq = _project_to_simplex(best_f, S).tolist()
    final_obj = evaluate_sequence(final_seq)   # <-- external function
    print(f"Finished after {iteration} iterations. Best objective = {final_obj:.6f}")
    return final_seq
