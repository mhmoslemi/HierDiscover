def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5 # margin
    best_seq = None
    best_score = float('inf')
    # list of lengths
    lengths = [2000, 3000, 4000, 5000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break
        # compute FFT size
        conv_len = 2*n-1
        nfft_conv = 1 << ((conv_len).bit_length())
        conv2_len = 3*n-2
        nfft_full = 1 << ((conv2_len).bit_length())
        nfft = max(nfft_conv, nfft_full)
        # start with multiple restarts
        for restart in range(3):
            if time.time() >= deadline:
                break
            # init x: maybe random normal
            x = np.random.randn(n) * 0.1
            # optionally start from powerlaw
            # but we can compute a powerlaw and transform to x via log
            # We'll also try start from previous best if lengths match.
            # Optionally adjust with beta exponent
            # For restart=0, use powerlaw beta=1
            if restart == 0:
                beta = 1.0
                a0 = np.arange(1, n+1) ** (-beta)
                a0 = a0 / a0.sum()
                # convert to x via log
                x = np.log(a0 + 1e-12)
            elif restart == 1:
                # uniform
                a0 = np.full(n, 1.0/n)
                x = np.log(a0)
            else:
                # random
                a0 = np.random.rand(n)
                a0 /= a0.sum()
                x = np.log(a0 + 1e-12)
            # Adam parameters
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            lr = 0.5  # base learning rate for x
            # schedule for alpha
            alpha = 5.0
            alpha_max = 200.0
            # iteration loop
            it = 0
            while True:
                if time.time() >= deadline or it >= 20000:
                    break
                # produce probability vector a via softmax
                # stable softmax
                maxx = np.max(x)
                ex = np.exp(x - maxx)
                a = ex / ex.sum()

                fa = np.fft.rfft(a, n=nfft)

                inner = np.dot(grad_a, a)

                v = beta2 * v + (1 - beta2) * (grad_x ** 2)

"""
Propose a candidate step‑function.

The method works directly on the probability simplex using a soft‑max
parameterisation a_i = exp(x_i) / Σ exp(x).  The objective
    2·n·max_k (a*a)_k
is approximated by a smooth “log‑sum‑exp’’ of the convolution,
which yields a differentiable surrogate.  An Adam optimiser
updates the underlying x‑vector; the smoothing parameter α is
increased slowly so that the optimisation eventually targets the
true max‑norm.  Multiple sequence lengths and random restarts are
tried within the given time budget; the best sequence (according to
the exact evaluate_sequence function) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    ex = np.exp(x - mx)
    return ex / ex.sum()


def _convolve_full(a: np.ndarray, b: np.ndarray, nfft: int) -> np.ndarray:
    """
    Full convolution of a and b using a common FFT length nfft.
    The result length is len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(fa * fb, n=nfft)
    return conv[: len(a) + len(b) - 1]


def _conv_self(a: np.ndarray, nfft: int) -> np.ndarray:
    """Self‑convolution a*a (full) using the supplied FFT size."""
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)
    # length is exactly 2*len(a)-1, the remaining tail is zero‑padded
    return conv[: 2 * len(a) - 1]


def _grad_wrt_a(p: np.ndarray, a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Gradient of Σ_k p_k (a*a)_k w.r.t. a.
    Uses the identity ∂(a*a)_k / ∂a_i = 2·a_{k-i}.
    """
    a_rev = a[::-1]
    # convolution p * a_rev, then extract the central n entries
    full = _convolve_full(p, a_rev, nfft)
    n = len(a)
    # slice corresponds to indices n-1 … n-1+n-1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) minimises
    evaluate_sequence.  The routine runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # small safety margin

    best_seq: list[float] | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # (optional) start from the previous construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = list(height_sequence_1)
            sc = evaluate_sequence(prev)
            best_seq = prev
            best_score = sc
            print(f"[init] Loaded previous best = {best_score:.6f}")
        except Exception as exc:
            print("[init] Failed to evaluate previous best:", exc)

    # ------------------------------------------------------------------
    # Sequence lengths to explore (larger lengths give slightly better
    # asymptotics but also higher per‑iteration cost)
    # ------------------------------------------------------------------
    lengths = [8000, 5000, 4000, 2800, 2000]

    # ------------------------------------------------------------------
    # Main loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # FFT size that can accommodate both required convolutions
        conv_len = 2 * n - 1          # a*a
        grad_len = 3 * n - 2          # p * a_rev
        nfft = _next_pow2(max(conv_len, grad_len))

        # ----------------------------------------------------------------
        # Several random restarts for each length
        # ----------------------------------------------------------------
        for restart in range(3):
            if time.time() >= deadline:
                break

            # ---------- initialisation of x (log‑probabilities) ----------
            if restart == 0:
                # power‑law start
                beta = 1.0
                a0 = np.arange(1, n + 1, dtype=np.float64) ** (-beta)
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
            elif restart == 1:
                # uniform start
                a0 = np.full(n, 1.0 / n)
                x = np.log(a0)
            else:
                # pure random start
                a0 = np.random.rand(n)
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)

            # Adam state
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            lr = 0.5          # base learning rate for x
            it = 0

            # smoothing parameter (log‑sum‑exp)
            alpha = 5.0
            alpha_target = 200.0

            # ----------------------------------------------------------------
            # Optimisation loop
            # ----------------------------------------------------------------
            while True:
                if time.time() >= deadline or it >= 25000:
                    break

                # ---- probabilities a ----
                a = _softmax(x)

                # ---- self‑convolution (exact) ----
                conv = _conv_self(a, nfft)          # length 2n-1

                # ---- smooth approximation of the max ----
                tmp = alpha * conv
                tmax = np.max(tmp)
                w = np.exp(tmp - tmax)              # stability
                sum_w = np.sum(w)
                # smooth max (not needed for the objective itself, but needed for gradient)
                # smooth_max = (np.log(sum_w) + tmax) / alpha

                # ---- probability distribution over convolution indices ----
                p = w / sum_w

                # ---- gradient w.r.t. a (using convolution) ----
                grad_a = _grad_wrt_a(p, a, nfft)

                # ---- gradient w.r.t. x (soft‑max Jacobian) ----
                inner = np.dot(grad_a, a)
                grad_x = a * (grad_a - inner)

                # ---- Adam update ----
                it += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---- increase smoothing gradually ----
                if alpha < alpha_target:
                    alpha *= 1.0015

                # ---- periodic exact evaluation (real objective) ----
                if it % 400 == 0:
                    max_conv = conv.max()
                    score = float(2.0 * n * max_conv)          # exact objective for sum(a)=1
                    if score < best_score - 1e-12:
                        best_score = score
                        best_seq = a.tolist()
                        print(f"[n={n:5d}, rs={restart}] New best {best_score:.6f}  it={it}")

                # ---- occasional jitter to escape plateaus ----
                if it % 5000 == 0:
                    x += (np.random.rand(n) - 0.5) * 0.01

    # ------------------------------------------------------------------
    # Fallback – uniform distribution of moderate length
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        a = np.full(n, 1.0 / n)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)

    # Final verification (sometimes the exact value differs slightly due
    # to rounding).  This also guarantees a proper return if the
    # evaluation function applies extra clamping.
    try:
        final_score = evaluate_sequence(best_seq)
        if final_score < best_score:
            best_score = final_score
    except Exception:
        pass

    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
