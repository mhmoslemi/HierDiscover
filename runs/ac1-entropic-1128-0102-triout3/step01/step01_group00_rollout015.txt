def compute_softmax_gradient(a, conv, beta):
    # conv: array length 2n-1
    n = len(a)
    # compute softmax weights
    # shift conv for numerical stability
    shift = conv.max()
    exp_beta = np.exp(beta * (conv - shift))
    w = exp_beta / exp_beta.sum()
    # compute cross-correlation g_i = 2 * sum_j a_j w_{i + j}
    # compute using FFT
    # Pad a_rev and w to length L2 >= n + (2n-1) -1 = 3n-2

    # cr length L2, we need slice n-1:n-1+n
    start = n - 1
    grad = 2.0 * cr[start:start+n]
    return grad

def upsample(seq, new_n):
    old_n = len(seq)
    # simple nearest neighbor interpolation
    indices = np.linspace(0, old_n, num=new_n, endpoint=False, dtype=int)
    # ensure indices in [0, old_n-1]
    indices = np.clip(indices, 0, old_n-1)
    return seq[indices].copy()

"""
Projected Adam optimizer with soft‑max smoothing of the auto‑convolution.

The objective (with sum(a)=1) is  score = 2·n·max(conv(a,a)).
We replace max by a smooth “log‑sum‑exp” (soft‑max) with an
inverse temperature β that is increased during the run.
The gradient of the smooth objective is a weighted sum of the
individual sub‑gradients; it can be computed efficiently by a
single cross‑correlation (FFT) per iteration.

A simple Adam update is performed, then the vector is projected
onto the probability simplex (non‑negative, sum=1).  The algorithm
is run on a sequence of increasing lengths: we start with a modest
n, optimise, up‑sample the result and continue for larger n.  The
best sequence found (lowest score) is returned.

The code respects the 1000 s budget, prints improvements and
uses only NumPy and the standard library.
"""

import time
import math
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive -> uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Nearest‑neighbour up‑sampling of a probability vector."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    # map new indices to old indices linearly
    idx = (np.arange(new_n) * old_n / new_n).astype(int)
    idx = np.clip(idx, 0, old_n - 1)
    return seq[idx].copy()


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max weights w_k ∝ exp(beta·conv_k)."""
    # Use the max for numerical stability
    shift = conv.max()
    w_raw = np.exp(beta * (conv - shift))
    s = w_raw.sum()
    if s == 0 or np.isnan(s):
        # fallback to uniform weights (should not happen)
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / s


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float,
                     L_grad: int) -> np.ndarray:
    """Gradient of the soft‑max objective w.r.t. a (vector of length n)."""
    n = a.shape[0]
    # soft‑max weighting of the convolution entries
    w = _softmax_weights(conv, beta)
    # cross‑correlation a_rev * w  (g_i = 2 * Σ_j a_j w_{i+j})
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, n=L_grad)
    Wr = np.fft.rfft(w, n=L_grad)
    cross = np.fft.irfft(Ar * Wr, n=L_grad)
    start = n - 1                 # index corresponding to i = 0
    grad = 2.0 * cross[start:start + n]   # shape (n,)
    return grad


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (heuristically) minimises
    `evaluate_sequence`.  The method runs for at most `budget_s` seconds
    and combines a soft‑max Adam optimiser with incremental up‑sampling
    of the step‑function length.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final return

    # ------------------------------------------------------------------
    # initialise from a previously found construction if it exists
    init_seq = globals().get('height_sequence_1')
    if init_seq is not None:
        a0 = np.array(init_seq, dtype=float)
        a0 = np.clip(a0, 0.0, None)
        total = a0.sum()
        if total < 1e-12:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / total
    else:
        a0 = None

    # list of lengths to explore (ascending, more freedom for larger n)
    lengths = [800, 1200, 1800, 2400, 3000, 3500]
    # Shuffle a bit to avoid deterministic bias when time runs out early
    np.random.shuffle(lengths)

    best_score = float('inf')
    best_seq = None
    cur_a = a0  # may be None → random initialisation later

    # ------------------------------------------------------------------
    for n in sorted(lengths):
        if time.time() >= deadline:
            break

        # initialise a for this length
        if cur_a is None:
            a = np.random.rand(n)
            a = a / a.sum()
        else:
            if cur_a.shape[0] < n:
                a = _upsample(cur_a, n)
            else:
                a = cur_a[:n].copy()
            a = a / a.sum()

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.05                     # base learning rate
        m = np.zeros_like(a)          # first moment
        v = np.zeros_like(a)          # second moment

        # scheduling of soft‑max temperature β
        beta_start = 1.0
        beta_end = 200.0
        max_iter = 25000              # generous cap per length

        # FFT lengths (cached for the whole inner loop)
        L_conv = _next_pow2(2 * n - 1)          # for a*a
        L_grad = _next_pow2(3 * n - 2)          # for gradient cross‑correlation

        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1

            # ---- compute linear convolution a * a (FFT) -----------------
            Af = np.fft.rfft(a, n=L_conv)
            conv_full = np.fft.irfft(Af * Af, n=L_conv)
            conv = conv_full[: 2 * n - 1]          # length 2n‑1
            max_val = conv.max()
            cur_score = 2.0 * n * max_val

            # update global best if improved
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Improved: n={n}, iter={it}, score={cur_score:.6f}")

            # ---- soft‑max temperature (linear schedule) ----------------
            beta = beta_start + (beta_end - beta_start) * (it / max_iter)
            # ---- compute smooth gradient --------------------------------
            grad = _softmax_gradient(a, conv, beta, L_grad)

            # ---- Adam update -------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a = _project_simplex(a_new)

        # keep the solution as starting point for the next (larger) size
        cur_a = best_seq if best_seq is not None else a

        # Early exit if we have already reached the target bound
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep contract safe.
    if best_seq is None:
        # very small uniform vector as a safe default
        best_seq = np.full(100, 1.0 / 100.0)

    # Return a plain Python list (the evaluator expects any list of floats)
    return [float(x) for x in best_seq.tolist()]
