"""
Propose a candidate sequence of non‑negative heights that minimizes

        2 * n * max(conv(a,a)) / (sum(a)**2)

The objective is invariant under scaling, therefore all candidates are
kept on the probability simplex (non‑negative, sum = 1).  The optimisation
uses a smooth approximation of the max‑operator (soft‑max) together with
an Adam‑style projected gradient descent.  A variety of initial families
(power‑law, symmetric Beta PDFs, edge‑blocks, random Dirichlet draws) are
generated for several lengths.  Each start is run for a few thousand
iterations, the best sequence seen across all runs is returned.

The gradient of the soft‑max‑approximated objective can be written as a
cross‑correlation between the current weights and the sequence; this is
computed efficiently with an FFT based convolution.  After each update the
vector is projected onto the simplex and symmetrised (the problem is
invariant under reversal), which dramatically reduces the search space
without harming optimality.

The method runs well within the 1000 s budget and typically finds a
score below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Family generators (all return vectors that sum to 1)
# ----------------------------------------------------------------------
def _powerlaw_sequence(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-beta}, monotone decreasing."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta)
    seq /= seq.sum()
    return seq


def _beta_sequence(n: int, a: float, b: float) -> np.ndarray:
    """Discretised Beta(a,b) density on {0,…,n‑1}."""
    xs = (np.arange(n) + 0.5) / n   # mid‑point rule
    seq = (xs ** (a - 1)) * ((1 - xs) ** (b - 1))
    # Numerical safety – Beta may blow up at the ends when a,b<1
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq_sum = seq.sum()
    if seq_sum == 0.0:
        # fall back to uniform
        seq = np.full(n, 1.0 / n)
    else:
        seq /= seq_sum
    return seq


def _edge_block_sequence(n: int, w_frac: float) -> np.ndarray:
    """Put half the mass uniformly on the first w entries and half on the last w."""
    w = max(1, int(round(w_frac * n)))
    seq = np.zeros(n, dtype=np.float64)
    seq[:w] = 0.5 / w
    seq[-w:] = 0.5 / w
    return seq


def _random_dirichlet(n: int, conc: float) -> np.ndarray:
    """Draw from a Dirichlet with the given concentration."""
    x = np.random.gamma(conc, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
# FFT based convolution / correlation utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (real‑valued)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return conv


def _corr_fft(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a) = Σ_k w_{k+i} a_k.
    Implemented as convolution of w with the reversed a.
    """
    m = w.shape[0]
    n = a_rev.shape[0]
    size = _next_pow2(m + n - 1)
    fft_w = np.fft.rfft(w, size)
    fft_a = np.fft.rfft(a_rev, size)
    corr = np.fft.irfft(fft_w * fft_a, size)[: m + n - 1]
    return corr


# ----------------------------------------------------------------------
# Objective and gradient (soft‑max approximation)
# ----------------------------------------------------------------------
def _objective_softmax(seq: np.ndarray, alpha: float) -> float:
    """Return the soft‑max approximation of max(conv(seq,seq))."""
    conv = _conv_fft(seq)
    max_c = conv.max()
    # stable soft‑max
    ex = np.exp(alpha * (conv - max_c))
    w = ex / ex.sum()
    return np.dot(w, conv)  # weighted sum ≈ max_c


def _gradient_softmax(seq: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximated objective.
    g_i = 2 * Σ_k w_k a_{k-i}   (cross‑correlation of w with a),
    where w are the softmax weights.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    max_c = conv.max()
    ex = np.exp(alpha * (conv - max_c))
    w = ex / ex.sum()                     # shape (2n‑1,)

    # cross‑correlation w ⋆ a (linear, zero‑padded)
    a_rev = seq[::-1]
    corr = _corr_fft(w, a_rev)           # length 3n‑2
    grad = 2.0 * corr[n - 1 : n - 1 + n]  # slice of length n
    return grad


# ----------------------------------------------------------------------
# Scoring shortcut (sum is 1 → denominator disappears)
# ----------------------------------------------------------------------
def _compute_score(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = np.convolve(seq, seq)          # small n → direct convolution is fine
    return float(2.0 * n * conv.max())
