def propose_candidate(...):
    import time, numpy as np, math, random
    start = time.time()
    deadline = start + budget_s - 0.5

    # initialize best
    best_seq = None
    best_score = float('inf')
    # seeds for random restarts
    seeds = [0, 1, 2, 3, 4]  # we can shuffle.

    # lengths list
    lengths = [800, 1200, 2000, 3000, 4000]  # maybe up to 5000 if time.

    for n in lengths:
        # if time low break
        if time.time() >= deadline: break

        # number of restarts per length: maybe 3
        for restart in range(3):
            if time.time() >= deadline: break

            # Starting sequence:
            if restart == 0 and 'height_sequence_1' in globals():
                seq = np.array(height_sequence_1, dtype=float)
                if len(seq) != n:
                    # resample/interpolate to proper length
                    # as weight per index: use linear interpolation
                    x_old = np.linspace(0,1,len(seq),endpoint=False)
                    x_new = np.linspace(0,1,n,endpoint=False)
                    seq = np.interp(x_new, x_old, seq)
                a = seq
            else:
                a = np.random.rand(n)
            a = np.abs(a)  # ensure nonnegative
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, 1.0/n)
            else:
                a = a / a_sum

            # parameters
            step0 = 0.6  # or maybe 0.5
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            m = np.zeros_like(a)
            v = np.zeros_like(a)

            # FFT length
            fft_len = 1 << ((2*n - 1).bit_length())
            iter_count = 0
            max_iter = 200000  # high but time bound cuts off

            while time.time() < deadline and iter_count < max_iter:
                iter_count += 1

                # compute convolution
                Af = np.fft.rfft(a, n=fft_len)
                conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]

                # get max value and index
                k_max = int(np.argmax(conv))
                max_val = conv[k_max]
                cur_score = 2.0 * n * max_val
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print("[{:.2f}s] Improved: n={}, iter={}, score={:.6f}".format(time.time()-start, n, iter_count, cur_score))

                # gradient for max index (subgradient)
                g = np.zeros_like(a)
                low = max(0, k_max - (n - 1))
                high = min(n - 1, k_max)
                i_idx = np.arange(low, high + 1)
                j_idx = k_max - i_idx
                # apply g[i] = 2* a[j]
                g[i_idx] = 2.0 * a[j_idx]

                # Adam update
                m = beta1 * m + (1 - beta1) * g
                v = beta2 * v + (1 - beta2) * (g * g)
                m_hat = m / (1 - beta1**iter_count)
                v_hat = v / (1 - beta2**iter_count)
                step = step0 / (1 + 0.0001 * iter_count)  # decreasing step (optional)
                a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

                # projection onto simplex
                a = _project_simplex(a_new)

    if best_seq is None:
        # fallback uniform
        best_seq = np.full(100, 1.0/100)

    return [float(x) for x in best_seq]

"""
Propose a step‑function that minimizes `evaluate_sequence`.

The algorithm works in three stages:
1.  Choose a sequence length (1200 → 3000 → 5000).  For each length we start
    from a smooth Hann window (already fairly flat after auto‑convolution) or,
    if a better sequence is already known, we upscale that sequence by linear
    interpolation.
2.  Run a **projected gradient descent with momentum** on the probability
    simplex.  The gradient is the sub‑gradient of the current worst
    auto‑convolution entry (the index where the convolution attains its
    maximum).  Step size decays as 1/√t, momentum helps to traverse flat
    plateaus faster, and after each update the vector is projected back onto
    the simplex using the O(n log n) algorithm of Duchi et al.
3.  Keep the best sequence (lowest score) seen across all stages and return it.

The routine respects a 1000 s time budget, prints intermediate improvements,
and works with any non‑negative list of floats (the evaluator will clip and
normalise if needed).  No filesystem or network access is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
        {x | x_i >= 0, sum(x) = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: Upscale a sequence to a new length by linear interpolation
# ----------------------------------------------------------------------
def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    # linear interpolation on the interval [0,1)
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0:
        up = np.full(new_n, 1.0 / new_n)
    else:
        up = up / total
    return up

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # try to grab the previous construction if it exists
    try:
        height_sequence_1  # may raise NameError
        prev_seq = np.array(height_sequence_1, dtype=float)
        if prev_seq.sum() == 0:
            prev_seq = None
        else:
            prev_seq = np.maximum(prev_seq, 0.0)
            prev_seq = prev_seq / prev_seq.sum()
    except Exception:
        prev_seq = None

    best_score = float('inf')
    best_seq = None

    # Stages – increasing length gives a lower theoretical bound
    stage_lengths = [1200, 3000, 5000]

    # Hyper‑parameters for the optimizer
    step0 = 0.6          # initial step size (will be divided by sqrt(t))
    mu = 0.9             # momentum coefficient
    jitter_interval = 20000   # iterations between random perturbations
    jitter_sigma = 0.005      # scale of the perturbation

    # ------------------------------------------------------------------
    # Iterate over stages
    # ------------------------------------------------------------------
    for n in stage_lengths:
        if time.time() >= deadline:
            break

        # Initialise the working vector `a`
        if prev_seq is not None and len(prev_seq) <= n:
            # Upscale the best sequence from the previous stage
            a = _upscale(prev_seq, n)
        else:
            # Start from a smooth Hann window (all values non‑negative)
            a = np.hanning(n).astype(float)
            if a.sum() == 0:
                a = np.ones(n, dtype=float)
            # Tiny random noise to avoid exact symmetry
            a += np.random.rand(n) * 1e-6
            a = a / a.sum()

        # Ensure we are on the simplex
        a = _project_simplex(a)

        # Momentum buffer
        v = np.zeros_like(a)

        # Pre‑compute FFT length (next power of two ≥ 2n‑1)
        fft_len = 1 << ((2 * n - 1).bit_length())

        it = 0
        no_improve_counter = 0

        # ----------------------------------------------------------------
        # Main optimisation loop for the current length
        # ----------------------------------------------------------------
        while time.time() < deadline:
            it += 1

            # ---- Compute linear auto‑convolution via FFT ----
            Af = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]

            # ---- Locate the worst (maximum) entry ----
            k_max = int(np.argmax(conv))
            max_val = conv[k_max]
            cur_score = 2.0 * n * max_val

            # ---- Keep the global best -------------------------------------------------
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] "
                      f"Improved: n={n}, iter={it}, score={cur_score:.6f}")
                no_improve_counter = 0
            else:
                no_improve_counter += 1

            # ---- Sub‑gradient of the active maximum entry ----
            g = np.zeros_like(a)
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            i_idx = np.arange(low, high + 1)
            j_idx = k_max - i_idx
            g[i_idx] = 2.0 * a[j_idx]

            # ---- Momentum step (projected gradient descent) ----
            step = step0 / math.sqrt(it)          # diminishing step
            v = mu * v - step * g                 # momentum update
            a_new = a + v
            a = _project_simplex(a_new)

            # ---- Occasional random jitter to escape plateaus ----
            if it % jitter_interval == 0:
                a += np.random.normal(scale=jitter_sigma, size=n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                v = np.zeros_like(a)   # reset momentum after jitter

            # ---- Hard iteration cap per stage (safety) ----
            if it >= 500000:   # more than enough for the time budget
                break

        # after finishing this stage keep the best sequence as the seed for the next
        if best_seq is not None:
            prev_seq = best_seq.copy()
        else:
            prev_seq = a.copy()

    # ----------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # Return a plain Python list – the evaluator works with any list of floats
    return [float(x) for x in best_seq]
