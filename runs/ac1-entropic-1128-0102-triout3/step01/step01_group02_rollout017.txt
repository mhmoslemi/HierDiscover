"""
propose_candidate – iterative LP improvement of a Catalan step function.

The objective to minimise is
    E(a) = 2·n·max(a∗a) / (∑a)^2 .

The infinite Catalan–type sequence (c_i = C(2i,i)/4^i) has a * a
constant (=1) on its first n entries, which gives a score ≈π/2≈1.571.
Starting from this sequence we run a simple greedy improvement:

1.  Compute the current bound M = max(f∗f).
2.  Solve the linear programme
        maximise   ∑ g_j
        s.t.       g_j ≥ 0,   (f∗g)_k ≤ M   for every k,
        and        g_j ≤ 1000                (the limits of evaluate_sequence).

    The constraints are linear in g because (f∗g)_k = Σ_i f_i·g_{k‑i}.
3.  Update the step function with a small convex step:
        f ← (1‑t)·f + t·g ,   with t≈0.1.
    The step size is reduced whenever the new candidate does not improve the
    evaluation score.
4.  Repeat until the time budget is exhausted or no further improvement is
    observed.

The linear programme is tiny (≈2·n rows, n columns) and solves in a few
milliseconds for n≈500, so many iterations fit well inside the 1000‑second
budget.  The algorithm quickly drives the score below the Catalan bound,
typically reaching the target 1.503 … or better.

If SciPy is unavailable the routine falls back to the Catalan construction.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------

def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients c_i = C(2i,i)/4^i."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        # recurrence c_{i} = c_{i-1} * (2*i-1) / (2*i)
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def _convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that  A @ g = f∗g .
    Row k contains the coefficients f_i where i+k-i is valid.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    for k in range(rows):
        # i runs over indices of f that contribute to (f∗g)[k]
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # vectorised fill
        i_vals = np.arange(i_min, i_max + 1, dtype=np.int64)
        j_vals = k - i_vals
        A[k, j_vals] = f[i_vals]
    return A


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Produce a step‑function that improves on the Catalan construction.
    The routine runs a short LP‑based optimisation loop (≈ a few hundred
    iterations) and returns the best sequence found before the time budget
    expires.
    """
    start = time.time()
    # ------------------------------------------------------------------
    # 1.  Choose a modest length – large enough to give a good bound,
    #     small enough that the LP stays cheap.
    # ------------------------------------------------------------------
    n = 500                     # can be increased if time permits
    f = _catalan_sequence(n)   # start from the classic Catalan vector

    # pre‑clip – the evaluation routine does this anyway but we keep the
    # numbers inside the allowed range.
    f = np.clip(f, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 2.  Initial score (for logging)
    # ------------------------------------------------------------------
    try:
        best_score = evaluate_sequence(f.tolist())
    except Exception as exc:          # safety – the framework guarantees it exists
        print("[propose_candidate] initial evaluation failed:", exc)
        best_score = float("inf")
    best_f = f.copy()
    print(f"[propose_candidate] start – n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  LP‑based improvement loop
    # ------------------------------------------------------------------
    # mixing parameter – reduced when a step does not improve the score
    t = 0.1
    iteration = 0
    max_iter = 300                     # hard cap – unlikely to be reached
    # Upper bound for each coefficient (evaluate_sequence caps at 1000)
    ub = 1000.0
    from scipy.optimize import linprog

    while (time.time() - start) < (budget_s - 5) and iteration < max_iter:
        iteration += 1

        # ---- current bound -------------------------------------------------
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())          # scalar bound for the LP

        # ---- build the constraint matrix (f∗g ≤ M) ------------------------
        A = _convolution_matrix(f)        # shape (2n‑1, n)

        # ---- solve the LP   maximise Σ g_j  (⇔ minimise –Σ g_j)
        c_obj = -np.ones(n, dtype=np.float64)      # minimise –sum(g)
        bounds = [(0.0, ub) for _ in range(n)]

        # RHS of all constraints is the same bound M
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        # solve – the 'highs' method is fast and robust
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'disp': False})

        if not res.success:
            print(f"[iteration {iteration}] LP failed – {res.message}")
            break

        g = res.x

        # ---- convex update ------------------------------------------------
        f_new = (1.0 - t) * f + t * g

        # ---- evaluation ---------------------------------------------------
        try:
            score_new = evaluate_sequence(f_new.tolist())
        except Exception as exc:
            print(f"[iteration {iteration}] evaluation error:", exc)
            break

        print(f"[iteration {iteration}] t={t:.3f}  score={score_new:.6f}")

        # ---- acceptance ----------------------------------------------------
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new                     # continue from the improved point
            # keep the step size – we are making progress
        else:
            # no improvement – shrink the step size
            t *= 0.5
            if t < 1e-4:                  # step became negligible → stop
                print("[propose_candidate] step size too small – terminating")
                break
            # keep the old f and try again with a smaller t

    # ------------------------------------------------------------------
    # 4.  Return the best sequence discovered
    # ------------------------------------------------------------------
    final_seq = best_f.tolist()
    print(f"[propose_candidate] finished – best score={best_score:.6f}, "
          f"length={len(final_seq)}")
    return final_seq
