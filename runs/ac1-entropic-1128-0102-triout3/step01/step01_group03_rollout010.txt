   already beats the half‑binomial baseline (≈1.571).

2. **Local refinement** –  Starting from the best grid point we run a
   short projected gradient descent that directly minimises
   ``2·n·max(conv(a,a))``.  The gradient of the active convolution entry
   k is exactly 2·a_{k‑i} for each i that contributes, and after each
   step the vector is projected onto the probability simplex
   (non‑negative, sum = 1).

3. **Return the best found vector** –  The final sequence is returned
   as a plain ``list``; a safety call to ``evaluate_sequence`` is made
   before returning to guarantee that the reported score matches the
   evaluator used by the competition.

The whole routine respects the 1000 s budget (default 950 s is used
internally to leave a safety margin) and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at module level (no closures)

def power_law_sequence(n: int, p: float, offset: float = 1.0) -> np.ndarray:
    """
    Return a non‑negative sequence a_i = (i+offset)^(-p)  (i = 0…n‑1)
    normalised so that sum(a)=1.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    total = a.sum()
    if total == 0.0:          # should never happen for the ranges we use
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Compute the discrete self‑convolution x * x using a real FFT.
    Result length is 2·len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_peak(a: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the peak convolution entry conv[peak_idx] with respect
    to the vector a (self‑convolution).  The exact gradient is
        g_i = 2 * a_{peak_idx-i}   if 0 ≤ i ≤ n‑1 and 0 ≤ peak_idx-i ≤ n‑1
    otherwise g_i = 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    lo = max(0, peak_idx - (n - 1))
    hi = min(peak_idx, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    jdx = peak_idx - idx
    # each valid pair contributes 2*a[jdx]
    grad[idx] = 2.0 * a[jdx]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 950.0) -> list[float]:
    """
    Search for a non‑negative sequence minimising the evaluation function.
    The routine first sweeps a simple power‑law family, then refines the
    best candidate with a few thousand steps of projected gradient descent.
    The best sequence found before the time limit is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------- configuration ---------------------------
    n = 2500                     # length of the candidate (adjustable)
    max_iter = 10_000_000        # hard upper bound for the gradient loop

    # coarse grid parameters
    p_grid = np.arange(0.50, 0.81, 0.010)   # p ∈ [0.5, 0.8] step 0.01
    offsets = [0.5, 1.0, 2.0]               # a few offset choices

    # refinement grid (will be set later around the best p)
    fine_step = 0.001
    fine_range = 0.02    # ± around best p

    # gradient‑descent hyper‑parameters
    eta_start = 0.08          # step‑size multiplier
    decay_factor = 0.93
    decay_every = 1500
    temperature_start = 0.03  # for occasional uphill moves

    # --------------------------- time management ------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a safety margin

    best_a = None
    best_score = np.inf
    best_p = None
    best_offset = None

    # --------------------------- phase 1: coarse grid ------------------
    print("[search] starting coarse grid")
    for offset in offsets:
        for p in p_grid:
            if time.time() > deadline:
                break
            a = power_law_sequence(n, p, offset=offset)
            conv = conv_fft(a)
            max_val = conv.max()
            score = 2.0 * n * max_val          # sum(a) = 1, so denominator = 1
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_p = p
                best_offset = offset
                print(f"[grid] new best: p={p:.4f}, offset={offset}, score={score:.6f}")

    if best_a is None:
        # very unlikely, fallback to the classic half‑binomial seed
        from math import sqrt
        def sqrt_series(m: int) -> np.ndarray:
            a = np.empty(m, dtype=np.float64)
            a[0] = 1.0
            for i in range(1, m):
                a[i] = a[i-1] * (2.0 * i - 1.0) / (2.0 * i)
            a /= a.sum()
            return a
        best_a = sqrt_series(n)
        best_score = 2.0 * n * conv_fft(best_a).max()
        best_p = 0.5
        best_offset = 1.0
        print("[fallback] using half‑binomial seed")

    # --------------------------- phase 2: fine grid --------------------
    # explore a narrow band around the best p discovered
    if best_p is not None:
        p_low = max(0.01, best_p - fine_range)
        p_high = min(0.99, best_p + fine_range)
        fine_ps = np.arange(p_low, p_high + fine_step, fine_step)

        print("[search] starting fine grid")
        for offset in offsets:
            for p in fine_ps:
                if time.time() > deadline:
                    break
                a = power_law_sequence(n, p, offset=offset)
                conv = conv_fft(a)
                max_val = conv.max()
                score = 2.0 * n * max_val
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    best_p = p
                    best_offset = offset
                    print(f"[fine] new best: p={p:.4f}, offset={offset}, score={score:.6f}")

    # --------------------------- phase 3: gradient refinement ----------
    print("[grad] starting projected gradient descent")
    a = best_a.copy()
    cur_score = best_score
    eta = eta_start
    temperature = temperature_start
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = conv_fft(a)
        max_val = conv.max()
        peak_idx = int(conv.argmax())

        # exact sub‑gradient of the active convolution entry
        grad = gradient_of_peak(a, peak_idx)

        # take a step opposite to the gradient
        step = eta * grad
        # avoid creating negatives – keep at least 1 % of each component
        step = np.minimum(step, a * 0.99)
        a_trial = a - step

        # occasional small random jitter to escape plateaus
        if rng.random() < 0.005:
            jitter = (rng.random(n) - 0.5) * 1e-5
            a_trial += jitter

        # project back onto the probability simplex
        a_trial = project_onto_simplex(a_trial)

        # evaluate
        conv_trial = conv_fft(a_trial)
        max_val_trial = conv_trial.max()
        trial_score = 2.0 * n * max_val_trial

        # keep the globally best vector
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            print(f"[iter {it:08d}] new global best score = {best_score:.6f}")

        # accept improvements unconditionally, otherwise Metropolis rule
        if trial_score <= cur_score:
            a = a_trial
            cur_score = trial_score
        else:
            delta = trial_score - cur_score
            if temperature > 0.0 and rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_score = trial_score
            # else reject

        # decay learning rate and temperature periodically
        if (it + 1) % decay_every == 0:
            eta = max(eta * decay_factor, 1e-6)
            temperature = max(temperature * decay_factor, 1e-8)

        it += 1

    # --------------------------- final sanity check -------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")
    print(f"[info] best parameters: p={best_p:.5f}, offset={best_offset}")

    return best_a.tolist()
```
