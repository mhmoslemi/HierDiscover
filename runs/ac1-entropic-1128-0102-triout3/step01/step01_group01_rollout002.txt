"""
Propose a sequence that minimizes `evaluate_sequence`.

Algorithm (Frank‑Wolfe style):
1. Start from a feasible non‑negative sequence (the global variable
   `height_sequence_1` if present) and renormalise it so that
   Σ a_i = √(2·n).  For any such normalisation the objective equals the
   maximal entry of the self‑convolution, therefore we can focus on
   minimising that peak.
2. Given the current sequence f, compute M = max(f∗f).
   Solve the linear program

        max   Σ b_i
        s.t.  f∗b ≤ M   (element‑wise)
              b_i ≥ 0

   The feasible point b = f is always admissible, therefore the optimal
   value Σb_i is at least Σf_i.  After the LP we scale
   g = (√(2·n) / Σb)·b so that Σg_i = √(2·n) and
   f∗g ≤ (Σf / Σb)·M < M.
3. Perform a cheap line‑search on the convex combination
        f_t = (1−t)·f + t·g ,   t∈[0,1]
   and pick the t that yields the smallest peak of f_t∗f_t.
   Replace f by f_t.
4. Keep the best sequence seen so far.  If no improvement occurs for a
   few iterations, restart from the best known point with a small random
   perturbation to escape plateaus.
5. Stop when the time budget expires or when the target bound (1.503)
   is reached.

The LP is solved with SciPy’s HiGHS interior‑point solver; the
convolution is evaluated via FFT for speed.  The method converges
quickly (a few LP solves) and stays well within the 1000 s budget.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution a*a (real) using FFT, returning the first (2·len(a)−1) entries."""
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _build_conv_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build a sparse matrix A (shape (2n-1, n)) such that A @ b = f * b
    (discrete linear convolution).  Each diagonal offset i (0 ≤ i < n) is
    filled with the constant value f[i].
    """
    n = f.size
    # Every diagonal contains a constant f[i]; create n arrays of length n.
    diags = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = list(range(n))
    A = scipy.sparse.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A
