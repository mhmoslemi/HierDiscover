"""
Hybrid LP‑gradient optimizer for the convolution bound.

The routine starts from the classic half‑binomial (central‑binomial)
sequence a_i = C(2i,i)/4**i, which already gives a near‑flat
autoconvolution (score ≈ π/2).  To push the maximum of the
self‑convolution below this value we iteratively solve a linear program
that, for the current iterate f, finds a feasible vector g with the
largest possible ℓ₁‑norm under the same peak constraint
‖f*g‖_∞ ≤ M (where M = max(f*f)).  After normalising g to sum = 1 we
move a short step towards g (Frank‑Wolfe style) and repeat.  The LP
step breaks the symmetry of the half‑binomial fixed point and typically
lowers the score by a few percent.  A short projected‑gradient
refinement is performed at the end.

The algorithm respects the 1000 s time budget, prints progress and
returns the best sequence seen (as a plain list of floats).
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
#  Helper utilities (no closures, top‑level)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of a_i = C(2i,i)/4**i (the half‑binomial series).
    The vector is normalised to have ℓ₁‑norm equal to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Autoconvolution of a real vector using FFT.
    Length of the result is 2*len(x)-1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def build_convolution_matrix(a: np.ndarray) -> scipy.sparse.csc_matrix:
    """
    Build the sparse matrix A (shape (2n-1, n)) representing the linear map
    g ↦ a * g, i.e. each row k contains the coefficients a_{k-j} that
    multiply g_j (zero where the index is out of range).
    The matrix is in CSC format – suitable for scipy.optimize.linprog
    with method='highs'.
    """
    n = a.shape[0]
    # each diagonal with offset i contains a[i] repeated n times
    data = [np.full(n, a_i, dtype=np.float64) for a_i in a]
    offsets = np.arange(n)               # 0,1,…,n‑1
    A = scipy.sparse.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def evaluate(seq: list[float]) -> float:
    """
    Shortcut to the externally provided evaluate_sequence.  The function is
    assumed to be already present in the execution environment.
    """
    # no import – the evaluation function is supplied by the outer framework
    return evaluate_sequence(seq)   # type: ignore[name-defined]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) minimising

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The method combines a linear‑programming (LP) update that maximises the
    ℓ₁‑norm of a feasible partner vector, with a short projected gradient
    refinement.  It returns the best sequence found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------- parameters ------------------------------------
    # Length of the sequence – moderate size keeps the LP tractable.
    n = 1200

    # LP loop control
    max_lp_iters = 45                # max number of LP updates
    step_frac   = 0.30               # initial convex‑combination weight
    step_decay  = 0.95               # geometric decay of step_frac

    # Gradient‑refinement phase (after LP)
    grad_iters   = 120_000
    eta_start    = 0.09
    temp_start   = 0.07
    decay_every  = 3000

    # -----------------------------------------------------------------
    # initialise from the half‑binomial sequence (already ℓ₁‑normalised)
    a = sqrt_series(n)
    cur_score = evaluate(a.tolist())
    best_a    = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    deadline = time.time() + budget_s - 5.0   # small safety margin

    # -----------------------------------------------------------------
    # Phase 1 – LP driven Frank‑Wolfe style updates
    for it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        # current peak of the autocorrelation
        conv = conv_fft(a)
        M = conv.max()

        # build constraints a * g ≤ M  (elementwise)
        A = build_convolution_matrix(a)                     # (2n‑1) × n sparse
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # linear program: maximise sum(g)  ↔ minimise –sum(g)
        c = -np.ones(n, dtype=np.float64)

        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': max(0, deadline - time.time())}
        )

        if not res.success:
            print(f"[LP {it}] linprog failed (status {res.status}); abort LP phase")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            print(f"[LP {it}] zero ℓ₁‑norm returned; abort")
            break

        # normalise g to ℓ₁‑norm = 1 (the same as a)
        g /= sum_g

        # convex combination – keep the sequence on the simplex
        a_new = (1.0 - step_frac) * a + step_frac * g
        a_new /= a_new.sum()          # re‑normalise to mitigate numerical drift

        # evaluate the candidate (cheap because sum is 1)
        cand_score = evaluate(a_new.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_new.copy()
            print(f"[LP {it:03d}] new best score = {best_score:.6f}")

        # accept the new iterate (always – we are decreasing the bound)
        a = a_new

        # decay step size
        step_frac = max(step_frac * step_decay, 0.02)

    # -----------------------------------------------------------------
    # Phase 2 – lightweight projected gradient descent for polishing
    # The gradient is the sub‑gradient of the peak of the autocorrelation.
    eta = eta_start
    temperature = temp_start
    it = 0

    while time.time() < deadline and it < grad_iters:
        conv = conv_fft(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient of the active peak (see original code for derivation)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        jdx = k - idx
        grad[idx] = a[jdx] + a[idx]

        # Take a step opposite to the gradient, then project back onto the simplex
        step = eta * grad
        step = np.minimum(step, a * 0.94)   # keep a small safety margin
        a_trial = a - step

        # occasional random jitter to shake out of flat regions
        if rng.random() < 0.006:
            a_trial += (rng.random(n) - 0.5) * 5e-5

        # Euclidean projection onto the probability simplex
        # (simple implementation based on Duchi et al. 2008)
        v = np.sort(a_trial)[::-1]
        cssv = np.cumsum(v) - 1.0
        rho_arr = np.nonzero(v - cssv / (np.arange(1, n + 1)) > 0)[0]
        if rho_arr.size == 0:
            a_proj = np.full_like(a_trial, 1.0 / n)
        else:
            rho = rho_arr[-1]
            theta = cssv[rho] / (rho + 1)
            a_proj = np.maximum(a_trial - theta, 0.0)

        # evaluate trial
        trial_score = evaluate(a_proj.tolist())

        # keep the best ever seen
        if trial_score < best_score:
            best_score = trial_score
            best_a = a_proj.copy()
            print(f"[grad {it:07d}] new best score = {best_score:.6f}")

        # acceptance rule (always accept improvements)
        if trial_score <= cur_score:
            a = a_proj
            cur_score = trial_score
        else:
            # Metropolis‑type acceptance for occasional uphill moves
            delta = trial_score - cur_score
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                a = a_proj
                cur_score = trial_score
            # otherwise reject

        # decay step size and temperature periodically
        if (it + 1) % decay_every == 0:
            eta = max(eta * 0.92, 1e-5)
            temperature = max(temperature * 0.90, 1e-6)

        it += 1

    # -----------------------------------------------------------------
    # final verification (the evaluate call also guarantees compliance)
    final_score = evaluate(best_a.tolist())
    print(f"[done] iterations LP={max_lp_iters}, GD={it}, final score={final_score:.6f}")

    # Return a plain Python list of floats as required by the caller
    return best_a.tolist()
