import time
import math
import numpy as np

# Try to import SciPy; if unavailable we fall back to the Catalan sequence.
try:
    from scipy import optimize, sparse
    _SCIPY_OK = True
except Exception:                     # pragma: no cover
    _SCIPY_OK = False


def _catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first ``n`` Catalan‑type coefficients
    c_i = C(2i,i) / 4**i  (i = 0,1,…).
    The recurrence c_{i+1} = c_i * (2*i+1)/(2*i+2) is used.
    """
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def _conv_matrix(f: np.ndarray) -> sparse.csc_matrix:
    """
    Build the sparse matrix M such that   M @ g = np.convolve(f, g)
    for any vector ``g`` of the same length as ``f``.
    M has shape (2*len(f)-1, len(f)) and contains the Toeplitz pattern
    with the entries of ``f`` on the anti‑diagonals.
    """
    n = f.size
    # each diagonal offset i (0 ≤ i < n) contains the constant f[i]
    offsets = np.arange(n, dtype=int)
    # data[i, j] = f[i] for all columns j
    data = np.tile(f[:, np.newaxis], (1, n))
    return sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csc")

    The method starts from the Catalan coefficients and repeatedly solves
    a linear program that maximises the ℓ₁‑norm while keeping the
    convolution bound unchanged.  The optimiser is mixed with the current
    sequence, producing a monotone decrease of the evaluation score.
    The whole search stops when the time budget expires or no further
    improvement is possible.
    """
    start = time.time()

    # 1️⃣ Choose a problem size.

        n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:                         # pragma: no cover
        n = 1500                               # a size that is fast for LPs

    # -------------------------------------------------------------
    a = _catalan_sequence(n)
    a = np.clip(a, 0.0, 1000.0)                # safe guard, though not needed

    except Exception as exc:                  # pragma: no cover
        print("[propose_candidate] initial evaluation failed:", exc)
        best_score = float("inf")
    best_seq = a.copy()

        M = conv.max()

        # ---- b) build the matrix for the constraint f * g ≤ M --------------------------------
        A = _conv_matrix(a)                       # shape (2n‑1, n) sparse

        c = -np.ones(len(a), dtype=float)         # objective for minimisation
