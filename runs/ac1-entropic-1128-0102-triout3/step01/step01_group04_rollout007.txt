"""
Propose a step‑function (non‑negative sequence) that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

The evaluation reduces to minimising the maximum entry of the self‑convolution
once the L¹‑norm is fixed to √(2 n).  The algorithm works as follows:

1.  Initialise a sequence from the well‑known analytic shape
        a_i ∝ √(i+1+δ) – √(i+δ)
    and scale it so that ∑a = √(2 n).  Several lengths *n* and deltas *δ*
    are tried.

2.  Refine the sequence with a *projected sub‑gradient descent* on the
    objective  max(conv(a,a)).  For the current maximal shift *k* the
    sub‑gradient is
        g_i = 2 a_{k‑i}   (indices outside the support are ignored).
    An additive update `a ← a – η g` is performed and the resulting vector
    is projected back onto the simplex {a ≥ 0, ∑a = √(2 n)}.

3.  A short stochastic hill‑climb (single‑coordinate Gaussian perturbations)
    finishes the refinement.

The whole routine is time‑bounded (≈1000 s); many restarts are performed
and the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """L¹‑norm that the optimiser keeps (the normalisation used in the reference)."""
    return np.sqrt(2.0 * n)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """
    Return a length‑n non‑negative sequence of the form
        a_i = √(i+1+δ) – √(i+δ)
    scaled so that its L¹‑norm equals √(2 n).
    """
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _evaluate(seq: np.ndarray) -> float:
    """Wrap the provided evaluate_sequence (expects a Python list)."""
    # evaluate_sequence is already available in the execution environment
    return evaluate_sequence(seq.tolist())


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        { w | w_i ≥ 0 , Σ w_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return the zero vector
        theta = cssv[-1] / n
        w = np.maximum(v - theta, 0.0)
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical guard – enforce exact sum (may be off by <1e‑12)
    excess = w.sum() - s
    if abs(excess) > 1e-12:
        w += excess / n
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on max(conv(a,a))
# ----------------------------------------------------------------------


def _subgradient_descent(seq: np.ndarray,
                         max_iter: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected (additive) sub‑gradient descent that directly minimises
    max(conv(seq,seq)).  Returns the best sequence encountered and its score.
    """
    n = seq.shape[0]
    target = _target_sum(n)

    # keep track of the best sequence found
    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    # initial learning‑rate (fraction of the mean entry)
    step_sigma = max(1e-9, seq.mean() * 0.3)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # full self‑convolution
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        max_idx = int(conv.argmax())          # shift that attains the maximum

        # record improvement
        if max_val < best_score:
            best_score = max_val
            best_seq = seq.copy()
            # (the evaluation function coincides with max_val when the L¹‑norm is fixed)
            # print progress occasionally
            if it % 200 == 0:
                print(f"[subgrad] it={it:5d} n={n:4d} max={max_val:.6f}")

        # compute sub‑gradient for the active shift only
        # g_i = 2 * a_{k‑i}   (if the index is inside the support)
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(seq)
        g[mask] = 2.0 * seq[j[mask]]

        # normalise the gradient direction to keep steps comparable across iterations
        grad_norm = np.linalg.norm(g) + 1e-12
        direction = g / grad_norm

        # decaying additive step size
        eta = step_sigma / (1.0 + it * 0.001)

        # tentative update (may become negative)
        candidate = seq - eta * direction

        # project back onto the simplex (preserves non‑negativity and Σ = target)
        seq = _project_onto_simplex(candidate, target)

        # slowly decay the base step size
        step_sigma *= 0.9995

    # final evaluation of the last sequence
    final_score = _evaluate(seq)
    if final_score < best_score:
        best_score = final_score
        best_seq = seq.copy()

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (a lighter refinement)
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray,
                max_iter: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a lightweight stochastic hill‑climbing optimisation.
    Returns the best sequence encountered and its score.
    """
    n = seq.shape[0]
    target = _target_sum(n)

    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    # initial step size – a fraction of the mean entry
    step_sigma = max(1e-9, seq.mean() * 0.2)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # propose moving a single coordinate (Gaussian perturbation)
        idx = np.random.randint(n)
        delta = np.random.randn() * step_sigma
        old_val = seq[idx]
        new_val = old_val + delta
        if new_val < 0.0:
            new_val = 0.0
        seq[idx] = new_val

        # renormalise to keep the required L¹‑norm
        seq = _project_onto_simplex(seq, target)

        cand_score = _evaluate(seq)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = seq.copy()
        else:
            # reject – revert the change (the projection restores feasibility)
            seq[idx] = old_val
            seq = _project_onto_simplex(seq, target)

        # decay the step size slowly
        step_sigma *= 0.9996

    # final check
    final_score = _evaluate(best_seq)
    if final_score < best_score:
        best_score = final_score

    return best_seq, best_score

    return best_seq.tolist()
