"""
Propose a step‑function that minimises `evaluate_sequence`.

The algorithm works in three phases:

1. **Initialisation** – start from the best known construction
   (`height_sequence_1` if it exists, otherwise a ½‑power law
   a_i ∝ (i+1)^‑0.5).  The vector is normalised to have sum 1.

2. **Smoothed gradient descent** – minimise a smooth approximation of the
   max‑convolution  (log‑sum‑exp of the convolution).  For a current
   distribution `a` we

   * compute the full self‑convolution `c = a*a` with an FFT,
   * build soft‑max weights `w_k ∝ exp(κ·(c_k−max(c)))`,
   * obtain the gradient  
     `∇L_i = 2·Σ_k w_k · a_{k−i}`  (computed by another FFT),
   * take a step `a ← a – η·∇L` and project the result onto the probability
     simplex (non‑negative, sum 1).

   The smoothing parameter `κ` is slowly increased (the objective
   becomes sharper) while the learning rate `η` is decreased.

3. **Polishing (random hill‑climbing)** – after the gradient phase we
   perform cheap random mass transfers between two positions (accepting
   only improvements) to squeeze a little more performance.

The whole search runs for at most `budget_s` seconds (default 1000 s)
and returns the best sequence found.  All operations respect the
constraints of `evaluate_sequence` (non‑negative, clipping at 0 / 1000,
and normalisation, which cancels out of the objective)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Return the full (length 2·len(a)‑1) self‑convolution a*a using an FFT.
    `fft_len` must be a power of two ≥ 2·len(a)‑1.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[: 2 * len(a) - 1]


def _softmax_weights(conv: np.ndarray, kappa: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) weights for a convolution vector."""
    max_val = conv.max()
    # subtract max to avoid overflow, then exponentiate
    exp_arg = kappa * (conv - max_val)
    w = np.exp(exp_arg)
    return w / w.sum()


def _gradient_via_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smoothed max‑convolution objective.
    The formula (ignoring the negligible diagonal factor) is

        grad_i = 2 * Σ_k w_k * a_{k-i} .

    It is obtained by a single convolution of `w` with the reversed
    vector `a[::-1]`.
    """
    n = len(a)
    # length of w is 2*n-1 (full convolution length)
    L = w.shape[0]

    # convolution length needed = L + n - 1
    fft_len = _next_pow2(L + n - 1)

    # FFT of the weight vector
    W = np.fft.rfft(w, n=fft_len)

    # FFT of the reversed a vector
    A_rev = np.fft.rfft(a[::-1], n=fft_len)

    # linear convolution (full)
    tmp = np.fft.irfft(W * A_rev, n=fft_len)[: L + n - 1]

    # gradient_i = 2 * tmp[n-1-i]  (see derivation in docstring)
    grad = 2.0 * tmp[n - 1 :: -1][:n]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implements the algorithm of Duchi et al. (2008) – O(n log n).
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # Find the rho (largest index where condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → return uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_array(a: np.ndarray, conv_len: int) -> float:
    """
    Fast computation of the evaluation score (without calling
    `evaluate_sequence`).  Because the objective is homogeneous,
    the sum of `a` is assumed to be 1.
    """
    conv = _full_convolution_fft(a, conv_len)
    max_conv = conv.max()
    return float(2.0 * len(a) * max_conv)

# Main entry point required by the challenge
# ----------------------------------------------------------------------

    # ------------------------------------------------------------------

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

            print(f"[restart] α={r_alpha:.3f} c={r_c:.1f}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence found (as plain Python list)

    print(f"[final] best score = {best_score:.6f} (length={n})")
    return best_a.tolist()
