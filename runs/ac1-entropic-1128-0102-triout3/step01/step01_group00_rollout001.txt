"""
Propose a step‑function with a low score for `evaluate_sequence`.

Algorithm
---------
The score is scale‑invariant, therefore we restrict the coefficients to the
probability simplex (non‑negative and sum = 1).  For a fixed length ``n`` we
minimise the squared deviation of the auto‑convolution from the uniform
target

    m = 1/(2·n−1)          # the average value of a*a on the simplex
    L(a) = Σ_k ( (a*a)[k] – m )² .

The gradient of ``L`` can be written as a linear convolution, which we
evaluate efficiently by FFTs.  After each gradient step the vector is
projected back onto the simplex (Duchi et al. 2008).  This drives the whole
convolution towards a flat shape, reducing the maximum entry.

When the gradient phase has stalled we switch to a classic sub‑gradient
descent that directly minimises the maximum convolution entry; this
polishes the solution and often lowers the final score further.

Multiple sequence lengths and several random restarts are tried within the
time budget (default 1000 s).  The best sequence found is returned as a plain
list of floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x | x_i ≥ 0, Σ x_i = 1} (Duchi et al. 2008)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution of two real vectors using a power‑of‑two FFT length.
# ----------------------------------------------------------------------
def _linear_conv(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear (full) convolution of x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << (n - 1).bit_length()  # next power of two
    Xf = np.fft.rfft(x, n=L)
    Yf = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(Xf * Yf, n=L)[:n]
    return conv

                # sub‑gradient of the active entry
                g = np.zeros_like(sub_a)
                low = max(0, k_max - (n - 1))
                high = min(n - 1, k_max)
                i_idx = np.arange(low, high + 1)
                j_idx = k_max - i_idx
                g[i_idx] = 2.0 * sub_a[j_idx]

                step = sub_step0 / math.sqrt(it)
                sub_a = _project_simplex(sub_a - step * g)

    # ------------------------------------------------------------------
    # Final polishing: one more short sub‑gradient run on the overall best.
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.asarray(best_seq, dtype=float)
        n = a.shape[0]
        sub_step0 = 0.4
        max_iter_sub = 4000
        for it in range(1, max_iter_sub + 1):
            if time.time() >= deadline:
                break
            conv = _linear_conv(a, a)
            k_max = int(np.argmax(conv))
            max_val = conv[k_max]
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.2f}s] Final polish improve: n={n}, "
                      f"iter={it}, score={cur_score:.6f}")

            # sub‑gradient step
            g = np.zeros_like(a)
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            i_idx = np.arange(low, high + 1)
            j_idx = k_max - i_idx
            g[i_idx] = 2.0 * a[j_idx]

            step = sub_step0 / math.sqrt(it)
            a = _project_simplex(a - step * g)

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep the contract safe.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100)

    # Return a plain Python list (the evaluator works with any list of floats)
    return [float(x) for x in best_seq]
