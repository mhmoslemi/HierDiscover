"""
propose_candidate
-----------------

A two‑phase optimizer for the step‑function inequality.

Phase 1 – *LP refinement*:
    For a non‑negative sequence f (∑f=1) we compute the current worst
    self‑convolution value M = max (f*f).  Keeping M fixed we solve a
    linear program that maximises the ℓ¹‑norm of a new vector g under the
    linear constraints (f*g) ≤ M.  Because the constraints are linear in g,
    the LP can be solved very quickly (O(n³) in the worst case, but for
    n≈1500 the HiGHS solver finishes in < 1 s).  If the optimum sum(g) is
    > 1 we normalise g to unit ℓ¹‑norm and replace f by a convex combination
    of f and the normalised g.  This operation provably never increases M
    and typically makes it smaller – it is the same alternating scheme
    described in the problem statement.

Phase 2 – *local search*:
    After the LP stage the sequence is already close to the optimum.
    A short stochastic hill‑climbing pass (mass transfer between two
    random coordinates) fine‑tunes the solution.

Both phases respect the non‑negativity and normalisation constraints,
run entirely within the 1000 s budget and return a list of floats ready
for ``evaluate_sequence``.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_score(a: np.ndarray) -> float:
    """Fast version of the supplied evaluation function (sum of a is assumed ≠ 0)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def conv_max(a: np.ndarray) -> float:
    """Return the maximal entry of the linear convolution a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())


def build_initial(n: int, alpha: float, offset: float) -> np.ndarray:
    """
    Produce a non‑negative sequence a_i ∝ (i+1+offset)^(-alpha) and normalise
    it to ℓ¹‑norm 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-alpha)
    a /= a.sum()
    return a


def constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (n × n) lower‑triangular Toeplitz matrix A such that
    (A @ g)[i] = (f * g)[i] for i = 0,…,n‑1.
    """
    n = f.shape[0]
    rows = np.arange(n)[:, None]
    cols = np.arange(n)[None, :]
    diff = rows - cols                     # diff[i,j] = i‑j
    mask = diff >= 0
    A = np.empty((n, n), dtype=f.dtype)
    A[mask] = f[diff[mask]]
    A[~mask] = 0.0
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑evaluation step function within the allotted time."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a safety margin

    # --------------------------------------------------------------
    # 1.  Problem size.
    #     n≈1500 is a good compromise between quality and LP size.
    n = 1500

    # --------------------------------------------------------------
    # 2.  Choose a good analytic starting point.
    best_seq = None
    best_val = np.inf
    alphas = [0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60]
    offsets = [0.0, 0.2, 0.5, 0.8, 1.0, 1.5, 2.0]
    for alpha in alphas:
        for off in offsets:
            cand = build_initial(n, alpha, off)
            val = fast_score(cand)
            if val < best_val:
                best_val = val
                best_seq = cand
    # If a global variable ``height_sequence_1`` exists and matches the size,
    # use it as a potential starting point as well.
    hs1 = globals().get("height_sequence_1")
    if isinstance(hs1, list) and len(hs1) == n:
        cand = np.array(hs1, dtype=np.float64)
        cand /= cand.sum()
        val = fast_score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand

    print(f"[init] best value after analytic scan: {best_val:.6f}")

    # --------------------------------------------------------------
    # 3.  Alternating LP refinement.
    f = best_seq.copy()
    best_seq = f.copy()
    best_val = fast_score(f)

    lp_iter = 0
    max_lp_iter = 30                     # safety limit
    while time.time() < deadline and lp_iter < max_lp_iter:
        M = conv_max(f)                  # current worst convolution value
        # Build constraint matrix A (size n×n) and scale by M to keep the RHS = 1.
        A = constraint_matrix(f) / M
        b = np.ones(n, dtype=np.float64)

        # Linear program: maximise sum(g)  ⇔ minimise –sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve using HiGHS – fast and robust for this size.
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            print("[lp] optimisation failed, exiting LP phase")
            break

        g = res.x
        S = g.sum()
        # If the LP cannot increase the ℓ¹‑norm we have reached a fixed point.
        if S <= 1.0 + 1e-12:
            print(f"[lp] no ℓ¹‑norm increase (S={S:.6f}), stopping LP phase")
            break

        # Normalise g to unit ℓ¹‑norm.
        g_norm = g / S

        # Try to replace f by a convex combination of f and g_norm.
        # We attempt several mixing factors, taking the first that improves M.
        updated = False
        for t in (0.9, 0.5, 0.2, 0.1):
            f_candidate = (1.0 - t) * f + t * g_norm
            f_candidate /= f_candidate.sum()          # re‑normalise numerically
            M_new = conv_max(f_candidate)
            if M_new < M - 1e-12:                      # strict improvement
                f = f_candidate
                M = M_new
                updated = True
                break

        if not updated:
            # No improvement even with mixing – stop the LP iterations.
            print("[lp] mixing did not improve M, terminating LP loop")
            break

        val = 2.0 * n * M
        if val < best_val:
            best_val = val
            best_seq = f.copy()
            print(f"[lp iter {lp_iter:02d}] new best {best_val:.6f}")
        else:
            if lp_iter % 5 == 0:
                print(f"[lp iter {lp_iter:02d}] current {val:.6f}")

        lp_iter += 1

    # --------------------------------------------------------------
    # 4.  Final stochastic hill‑climbing (fine‑tuning).
    cur = best_seq.copy()
    cur_val = fast_score(cur)
    step = 0.05
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = step * cur[i] * np.random.rand()
        if delta > cur[i]:
            delta = cur[i]
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand /= cand.sum()
        cand_val = fast_score(cand)

        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_seq = cand_val, cand.copy()
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                cur, cur_val = cand, cand_val

        # slowly decay the step size
        if it % 5000 == 0 and it > 0:
            step = max(step * 0.9, 1e-5)

        if it % 2000 == 0:
            print(f"[hill {it:6d}] best={best_val:.6f}, cur={cur_val:.6f}")

        it += 1

    print(f"[final] best value discovered: {best_val:.6f}")
    return best_seq.tolist()
