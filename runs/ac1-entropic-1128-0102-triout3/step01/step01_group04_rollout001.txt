"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1.   Initialise a candidate using the analytically motivated
     “discrete derivative of √x” shape:
          a_i ∝ √(i+1+δ) – √(i+δ)   (δ ≥ 0)
     and rescale it so that  Σ a_i = √(2·n)  (the normalisation used by the
     evaluator).

2.   Refine the candidate with a projected sub‑gradient descent on the
     objective
          F(a) = 2·n·max(conv(a,a)) / (Σ a)² .
     For a fixed L¹‑norm this is equivalent to minimising
          max(conv(a,a)).
     The sub‑gradient of the max‑convolution is obtained by
     summing the gradients of all convolution entries that attain the
     current maximum (or lie within a tiny tolerance).

3.   One iteration:
        * compute the full convolution via FFT (O(n log n));
        * locate the active indices where the convolution attains its
          maximum;
        * build the sub‑gradient g_i = Σ_{k∈active,0≤k−i<n}
          (a_{k−i}+a_i);
        * take a step a ← a – η·g and project back onto the simplex
          {a ≥ 0, Σ a = √(2·n)} (a standard “projection onto a scaled
          simplex”).

     The step size η is initialised as a fraction of the mean entry,
     multiplied by a small factor when a step is successful and reduced
     otherwise; it also decays slowly.

4.   To avoid getting stuck in a local minimum a small random
     perturbation of a random coordinate is added every few hundred
     iterations.

5.   The outer loop performs many independent restarts with
     different lengths (n) and different δ‑values until the time budget
     (≈ 1000 s) is exhausted.  The best sequence seen across all
     restarts is returned.

All helper functions are top‑level, there are no closures, and the code
uses only NumPy and the standard library.  The implementation evaluates
the objective directly (instead of calling `evaluate_sequence` on every
iteration) for speed, but the final candidate is returned as a plain
Python list so that the external evaluator can check it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """L¹‑norm used by the reference: Σ a_i = sqrt(2·n)."""
    return np.sqrt(2.0 * n)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """
    Initialise a length‑n sequence a_i = √(i+1+δ) – √(i+δ),
    then scale it to have the target L¹‑norm.
    """
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector v onto the scaled simplex {x ≥ 0, Σ x = s}.
    Implementation of the O(n log n) algorithm from
    "Efficient Projections onto the L1‑ball".
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Shift to make the problem a standard simplex projection.
    # The algorithm works for any non‑negative s.
    # Sort v in descending order.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical cleanup – ensure the sum is exactly s (within floating error)
    w_sum = w.sum()
    if w_sum > 0:
        w *= s / w_sum
    return w


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of `seq` with itself, computed via FFT.
    The returned array has length 2·n−1 and all entries are non‑negative
    (tiny negative values from round‑off are truncated to 0).
    """
    n = seq.shape[0]
    # Next power of two for efficient FFT.
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    prod = fft_seq * fft_seq
    conv = np.fft.irfft(prod, n=size)[: 2 * n - 1]
    # Remove spurious negative eps due to floating‑point noise.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` (without the safety
    clipping, which never triggers for the sequences we generate).
    """
    n = seq.shape[0]
    conv = _conv_full_fft(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    # Guard against pathological sums – should never happen.
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _subgradient_max_convolution(seq: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of  max(conv(seq,seq))  with respect to `seq`.
    The sub‑gradient is the sum of the gradients of all convolution
    entries whose value lies within `tol` of the maximum.
    """
    n = seq.shape[0]
    conv = _conv_full_fft(seq)
    M = conv.max()
    # Choose a tolerance that is generous enough for floating‑point noise.
    active = np.where(conv >= M - max(tol, 1e-12 * M))[0]

    grad = np.zeros_like(seq)
    # For each active convolution index k, add the contribution
    #   ∂/∂a_i conv_k = a_{k-i} + a_i  (if 0 ≤ k-i < n)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # Vectorised version for the current k:
        i_range = np.arange(i_min, i_max + 1, dtype=int)
        j_range = k - i_range
        grad[i_range] += seq[j_range] + seq[i_range]
    return grad


def _subgradient_descent(
    seq: np.ndarray,
    max_iter: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the L∞‑norm of the autocorrelation.
    Returns the best sequence encountered and its objective value.
    """
    n = seq.shape[0]
    target = _target_sum(n)

    # Current state
    cur_seq = seq.copy()
    cur_score = _objective(cur_seq)

    # Initial step size – a modest fraction of the average entry.
    step = max(1e-8, cur_seq.mean() * 0.2)

    best_seq = cur_seq.copy()
    best_score = cur_score

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # Sub‑gradient of the max‑convolution.
        grad = _subgradient_max_convolution(cur_seq)

        # Candidate step opposite to the gradient.
        cand = cur_seq - step * grad
        cand = _project_simplex(cand, target)

        cand_score = _objective(cand)

        # Accept if we improve; otherwise shrink the step size.
        if cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            # Successful step – try to be a bit more aggressive.
            step *= 1.02
        else:
            step *= 0.5

        # Keep track of the overall best.
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()

        # Controlled decay ensures eventual convergence.
        step *= 0.999

        # Every few hundred iterations inject a small random jitter
        # to help escape flat regions.
        if (it + 1) % 500 == 0:
            idx = np.random.randint(n)
            jitter = np.random.randn() * step * 5.0
            tmp = cur_seq.copy()
            tmp[idx] = max(0.0, tmp[idx] + jitter)
            tmp = _project_simplex(tmp, target)
            tmp_score = _objective(tmp)
            if tmp_score < cur_score:
                cur_seq = tmp
                cur_score = tmp_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                step *= 1.01
            else:
                step *= 0.9

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.
    Multiple random restarts with different lengths and δ‑parameters are
    performed; each restart is refined by a projected sub‑gradient descent.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Reserve a few seconds for finalisation.
    deadline = start_time + max(0.0, budget_s - 5.0)

    best_seq = None
    best_score = float("inf")

    # Candidate lengths – we allow a fairly wide range; longer sequences
    # tend to give smaller max‑convolution values.
    possible_lengths = list(range(600, 2601, 200))  # 600, 800, …, 2600

    # Main restart loop.
    while time.time() < deadline:
        n = int(np.random.choice(possible_lengths))
        # Random δ ∈ [0, 0.9); larger δ reduces the left‑edge spike.
        delta = np.random.rand() * 0.9

        # Initialise from the analytic shape.
        seq = _analytic_sqrt_sequence(n, delta=delta)

        init_score = _objective(seq)
        if init_score < best_score:
            best_score = init_score
            best_seq = seq.copy()
            print(f"[init] n={n:4d}  δ={delta:.3f}  score={init_score:.6f}")

        # Allocate a modest inner budget (≈0.4 s) for the descent.
        inner_deadline = time.time() + 0.45
        seq_opt, score_opt = _subgradient_descent(
            seq, max_iter=20000, deadline=inner_deadline
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[impr] n={n:4d}  δ={delta:.3f}  score={score_opt:.6f}")

        # Early‑stop when we already beat a tentative target (helps the
        # competition script to finish a bit earlier).
        if best_score < 1.55:
            # Keep searching a little longer in case we can push even lower.
            pass

    # Fallback – should never be needed.
    if best_seq is None:
        n = 50
        best_seq = _analytic_sqrt_sequence(n, delta=0.0)

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
