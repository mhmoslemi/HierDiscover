"""
Propose a low‑score step‑function.

The algorithm starts from a good known construction (if available) or
from a power‑law family.  It then performs many iterations of a
projected mirror‑descent update that targets the indices where the
self‑convolution attains its current maximum.  In each iteration we

1. compute the full convolution by FFT,
2. pick the few (default 5) indices that realise the maximal value,
3. form a sub‑gradient “weight’’ for every position i by counting how
   much a_i contributes to those peak indices,
4. update a_i multiplicatively with a decreasing step‑size
   (exponentiated‑gradient / mirror descent) and renormalise onto the
   probability simplex,
5. keep the best sequence seen so far.

Occasional random restarts from the parametric family and a small
probability of accepting a worsening move (simulated‑annealing) help
to escape local minima.  The loop runs until the 1000 s budget is
exhausted and the best sequence found is returned as a plain Python
list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – they are top‑level (no closures)
# ----------------------------------------------------------------------


def _fft_max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Return the maximum value of the discrete self‑convolution a*a
    using a real FFT of length ``fft_len`` (which must be a power of two
    and at least 2*len(a)-1).
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the full linear convolution a * a using an FFT of length
    ``fft_len`` and return the (2*len(a)-1)‑element result.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return conv


def _score_from_array(a: np.ndarray, fft_len: int) -> float:
    """
    Return the exact value that ``evaluate_sequence`` would compute
    for a normalised array ``a`` (i.e. sum(a)=1).  The factor 2·n·max
    is used because the denominator cancels after normalisation.
    """
    n = len(a)
    max_conv = _fft_max_convolution(a, fft_len)
    return 2.0 * n * max_conv

# Main entry point required by the harness
# ----------------------------------------------------------------------

    minimises ``evaluate_sequence``.  The routine runs for at most

    """
