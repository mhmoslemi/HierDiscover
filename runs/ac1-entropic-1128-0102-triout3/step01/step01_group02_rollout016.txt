import time, math, numpy as np, scipy.optimize as opt, scipy.sparse as sp

def _catalan_sequence(n):
    seq = np.empty(n)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i-1] * (2*i - 1) / (2*i)
    return seq.tolist()

def _conv_matrix(a):
    """Return sparse matrix A such that A @ b = np.convolve(a, b)"""
    n = len(a)
    # Create list of data arrays: each is constant a[i] repeated n times
    # Offsets are negative: -i for i in 0..n-1
    data = [np.full(n, a[i], dtype=float) for i in range(n)]
    offsets = -np.arange(n, dtype=int)
    # shape: (2*n-1, n)
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    return A

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Improve on the Catalan construction by a linear‑programming refinement.
    The algorithm:
    1. Start from the Catalan sequence of length n.
    2. Repeatedly solve a LP that maximises the ℓ₁‑norm of a new vector b
       while keeping the convolution f∗b bounded by the current maximal
       convolution value M = max(f∗f).  The LP is linear because f is fixed.
    3. Scale b so that max(b∗b) equals M, then move the current vector
       towards b (convex combination).  The move reduces the ratio
       2 n·max_conv/ (ℓ₁)² and therefore the evaluation score.
    4. Stop after a time budget or when no further improvement is observed.
    """
    start = time.time()
    # Determine length
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 1500  # modest size that fits comfortably in time‑budget

    # initialise with Catalan sequence
    f = np.array(_catalan_sequence(n), dtype=float)
    # store best
    best_f = f.copy()
    try:
        best_score = evaluate_sequence(best_f.tolist())
    except Exception:
        best_score = float('inf')
    print(f"[propose_candidate] start length={n}, initial score={best_score:.6f}")

    # main refinement loop
    max_iters = 20
    it = 0
    t = 0.4  # mixing factor
    while it < max_iters and time.time() - start < budget_s * 0.95:
        # current maximal convolution
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        # Build sparse constraint matrix A such that (A @ b)[k] = (f * b)[k]
        A = _conv_matrix(f)
        b_ub = np.full(conv_f.shape, M, dtype=float)

        # Linear program: maximize sum(b)  <=> minimize -sum(b)
        c = -np.ones(n, dtype=float)
        bounds = [(0.0, None) for _ in range(n)]

        # Solve LP with HiGHS (fast and reliable)
        lp_res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True})
        if not lp_res.success:
            print(f"[iter {it}] LP failed: {lp_res.message}")
            break

        b = lp_res.x
        sum_b = b.sum()
        # scale b so that its self‑convolution has the same peak M
        conv_b = np.convolve(b, b)
        M_b = np.max(conv_b)

The algorithm starts from the Catalan construction (which gives a score
≈ π/2).  For a fixed vector f we compute

    M = max_k (f∗f)[k].

Keeping M fixed we solve the linear program

    maximise   Σ b_j
    subject to   (f∗b)[k] ≤ M   for all k
                 b_j ≥ 0.

Because (f∗b)[k] is linear in the unknown b, this is a genuine LP.
The solution b usually has a larger ℓ₁‑norm than f – the slack in the
late convolution entries can be exploited.  We then scale b so that
its own convolution attains the same peak M, and move the current
vector towards it (convex combination).  Repeating the step reduces the
ratio 2·n·max / (ℓ₁)², i.e. the value returned by `evaluate_sequence`.

The implementation uses `scipy.optimize.linprog` with the HiGHS solver

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> list[float]:
    """First n Catalan‑type coefficients  c_i = C(2i,i)/4ⁱ."""
    seq = np.empty(n, dtype=float)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq.tolist()


def _conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Return a sparse matrix A (shape (2*n‑1, n)) such that
    A @ b == np.convolve(a, b)   for any b of length n.
    The k‑th row corresponds to the k‑th entry of the convolution.
    """
    n = a.shape[0]
    # each diagonal `‑i` (i = 0 … n‑1) contains the constant a[i]
    data = [np.full(n, a[i], dtype=float) for i in range(n)]
    offsets = -np.arange(n, dtype=int)            # ‑0, ‑1, … ‑(n‑1)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A
