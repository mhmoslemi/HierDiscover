import time, numpy as np

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _fft_convolve(a, b, fft_len):
    # Always real arrays
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(A * B, fft_len)
    # as real, take only needed part
    return conv[: len(a) + len(b) - 1]

def _self_convolution(a, fft_len):
    return _fft_convolve(a, a, fft_len)

def _grad_from_weights(w, a_rev, fft_len):
    full = _fft_convolve(w, a_rev, fft_len)
    n = len(a_rev)
    start = n - 1
    return full[start:start + n]

def _project_simplex(v):
    # projection onto simplex {x >= 0, sum x = 1}
    # Implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # If all cond false, return zeros (should not happen)
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _max_convolution(a, fft_len):
    conv = _self_convolution(a, fft_len)
    return float(conv.max())

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Projected gradient descent on a smooth approximation of the L∞
    auto‑convolution norm, followed by a short hill‑climbing phase.
    The sequence is kept on the probability simplex (non‑negative,
    sum = 1).  No element ever exceeds the clip‑limit 1000, so the
    exact `evaluate_sequence` score equals 2·n·max(a*a).
    """
    np.random.seed(seed)
    t_end = time.time() + budget_s - 2.0

    # Determine size and starting point
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        n = len(init)
        a = np.maximum(init, 0.0)
    except Exception:
        n = 2000
        a = np.ones(n, dtype=np.float64)

    # Normalise to sum 1
    s = a.sum()
    if s == 0.0:
        a = np.ones(n) / n
    else:
        a = a / s

    # Pre‑compute FFT sizes
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # Initial best
    best_a = a.copy()
    best_score = 2.0 * n * _max_convolution(a, fft_len_self)
    print(f"[init] n={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Smooth projected gradient descent
    # ------------------------------------------------------------------
    # Parameters
    eta = 0.2           # step size
    gamma = 5.0         # smoothness parameter
    grad_iters = 0
    grad_time_limit = time.time() + (budget_s - 2.0) * 0.6  # use 60% of time

    while time.time() < grad_time_limit:
        grad_iters += 1
        conv = _self_convolution(a, fft_len_self)
        max_c = conv.max()
        # Soft weighting
        w = np.exp(gamma * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.ones_like(conv) / conv.size
            w_sum = 1.0
        w = w / w_sum

        # Gradient of the smoothed max
        rev_a = a[::-1]
        g = 2.0 * _grad_from_weights(w, rev_a, fft_len_grad)

        # Gradient step
        a_new = a - eta * g
        a_new = _project_simplex(a_new)

        # Accept
        a = a_new

        # Track best score
        cur_score = 2.0 * n * _max_convolution(a, fft_len_self)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[grad {grad_iters}] new best {best_score:.6f}")

        # Adapt parameters every few iterations
        if grad_iters % 30 == 0:
            eta *= 0.9          # decrease step size
            gamma *= 1.05       # sharpen the approximation

        # Occasionally inject a small random perturbation to avoid stagnation
        if np.random.rand() < 0.02:
            perturb = np.random.randn(n) * 0.001
            a = _project_simplex(a + perturb)

    # ------------------------------------------------------------------
    # 2. Final hill‑climbing polishing (same style as previous code)
    # ------------------------------------------------------------------
    # Re‑use the FFT length for max convolution
    while time.time() < t_end:
        # propose a random transfer of mass between two distinct positions
        i1, i2 = np.random.choice(n, size=2, replace=False)
        # transfer up to 1 % of the donor entry
        delta = (np.random.rand() - 0.5) * 0.02
        # keep non‑negativity
        if a[i1] + delta < 0.0:
            delta = -a[i1] + 1e-12
        if a[i2] - delta < 0.0:
            delta = a[i2] - 1e-12

"""
Proposed algorithm
------------------

The objective of `evaluate_sequence` for a vector whose sum is 1 is
    score = 2·n·max_k (a*a)_k .
Thus we have to minimise the infinity‑norm of the self‑convolution while
keeping the vector non‑negative and summing to 1.

The optimiser works in two phases:

1. **Projected gradient descent** on a smooth approximation of the
   max‑norm.  For a smoothing factor γ we use the soft‑max weights
   w_k ∝ exp(γ·(conv_k – max(conv))) and compute the gradient of the
   weighted sum  Σ w_k·conv_k .  The gradient is obtained by a second
   convolution and the update is projected onto the probability simplex
   (non‑negative, sum = 1).

   γ is slowly increased so that the approximation becomes tighter,
   while the step size η is decreased to keep the iterations stable.

2. **Hill‑climbing polishing** (the same small random mass‑transfer
   moves as in the reference code).  Random restarts from a simple
   power‑law family are also performed occasionally.

Both phases respect the clipping limits of `evaluate_sequence`; because
the vector is normalised to sum = 1 all entries stay far below 1000, so
the score computed as `2·n·max(conv)` is exact.

The whole routine stops after the allotted budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Small helper returning the next power‑of‑two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Convolution of two real vectors using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(A * B, fft_len)
    # keep the exact linear‑convolution length
    return conv[: len(a) + len(b) - 1]


def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a*a (full self‑convolution) via FFT."""
    return _fft_convolve(a, a, fft_len)


def _grad_from_weights(w: np.ndarray, a_rev: np.ndarray, fft_len: int) -> np.ndarray:
    """
    For a weight vector w (length 2·n‑1) and a_rev = a[::-1] (length n)
    return the vector
        g_i = Σ_t w_{i+t} a_t  (i = 0…n‑1)
    which is the needed gradient component.
    """
    full = _fft_convolve(w, a_rev, fft_len)
    n = len(a_rev)
    start = n - 1               # index of the first relevant element
    return full[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { x | x_i ≥ 0, Σ_i x_i = 1 }.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the self‑convolution a*a."""
    conv = _self_convolution(a, fft_len)
    return float(conv.max())
