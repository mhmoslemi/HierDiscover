def upscale_seq(a):
    n = len(a)

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Hybrid local‑search + multiscale refinement.
    Starts from the supplied `height_sequence_1` (if present) or from a
    simple exponential decay, then repeatedly performs a greedy
    “move‑mass” hill‑climbing step that explicitly reduces the current
    maximal autocorrelation, and finally upsamples the sequence (linear
    interpolation) to give the algorithm more degrees of freedom.
    The whole process runs for at most ``budget_s`` seconds and returns the
    best sequence found.
    """
    import time
    import numpy as np

    np.random.seed(seed)
    finish_time = time.time() + budget_s - 2.0   # safety margin

"""
Hybrid local‑search + multiscale refinement for the step‑function
evaluation problem.

The algorithm works as follows

1. **Initialisation** – use the provided `height_sequence_1` if it
   exists, otherwise start from a simple exponential decay.  The
   vector is normalised to L¹‑norm = 1 (the objective is scale‑invariant).

2. **Greedy mass‑transfer hill‑climbing** – repeatedly move a small
   amount of mass from a “donor’’ index *i* to a “receiver’’ index *j*.
   The move is chosen so that the current *max* entry of the
   self‑convolution (the quantity appearing in the objective) is reduced
   as much as possible.  The effect of a transfer on the maximal
   convolution is known analytically:
   Δconvₖ = 2 ε (a_{k‑j} − a_{k‑i}),
   therefore the algorithm explicitly computes the improvement and
   accepts the best move.  If no improving move is found a tiny random
   perturbation is performed to escape flat regions.

3. **Multiscale up‑sampling** – when the local search stalls, the
   sequence is up‑sampled by inserting linear interpolants
   (`…, a_i, (a_i+a_{i+1})/2, a_{i+1}, …`).  The length roughly doubles,
   providing many more degrees of freedom.  The optimisation then
   continues on the finer grid.

4. The whole process is bounded by the supplied time budget
   (`budget_s` ≈ 1000 s).  The best sequence found is returned as a plain
   Python list; it satisfies all constraints of `evaluate_sequence`.

The code uses only NumPy and the standard library (no external LP
solvers) and stays well within the 1 CPU‑core limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x≥1)."""
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def _max_and_second_conv(a: np.ndarray, fft_len: int):
    """
    Compute the full self‑convolution a*a via FFT, return
        max_val   – the largest entry,
        second_val– the second‑largest entry (strictly smaller, or equal if all equal),
        conv      – the whole convolution array.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    max_val = conv.max()
    # second maximum (ignore entries equal to the maximum up to a tiny tolerance)
    mask = conv < max_val - 1e-15
    if np.any(mask):
        second_val = conv[mask].max()
    else:
        second_val = max_val
    return max_val, second_val, conv


def _upscale_sequence(a: np.ndarray) -> np.ndarray:
    """
    Linear up‑sampling: turn a length‑n vector into length 2n‑1 by
    inserting the average of neighbouring entries.  The result is
    re‑normalised to keep L¹ = 1.
    """
    n = len(a)
    new_len = 2 * n - 1
    b = np.empty(new_len, dtype=np.float64)
    b[0::2] = a
    b[1::2] = (a[:-1] + a[1:]) / 2.0
    b /= b.sum()
    return b


def _greedy_local_optimize(a0: np.ndarray, deadline: float):
    """
    Greedy hill‑climbing that explicitly reduces the current maximal
    autocorrelation.  The routine works on a copy of ``a0`` and stops
    when the time limit ``deadline`` is reached (or after a long streak
    without improvement).
    Returns the best sequence found and its corresponding score.
    """
    a = a0.copy()
    n = len(a)
    fft_len = _next_pow2(2 * n - 1)

    # initial evaluation
    max_val, second_val, conv = _max_and_second_conv(a, fft_len)
    best_score = 2.0 * n * max_val
    best_a = a.copy()

    # counters for early termination when progress stalls
    no_improve = 0
    max_no_improve = 5000

    while time.time() < deadline:
        if no_improve > max_no_improve:
            break

        # index of the current maximal convolution entry
        k_max = int(np.argmax(conv))

        # reflected array r_i = a_{k_max - i} (0 if out of range)
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        r = np.zeros_like(a)
        r[valid] = a[idx[valid]]

        # donors – indices with large r (i.e. they contribute heavily to the max)
        donor_candidates = np.where(a > 1e-12)[0]
        if donor_candidates.size == 0:
            break
        donor_order = donor_candidates[np.argsort(-r[donor_candidates])]

        # receivers – indices with small r
        receiver_order = np.argsort(r)[: min(20, n)]

        improved = False
        # try a few promising (donor,receiver) pairs
        for i in donor_order[:10]:
            if a[i] <= 1e-15:
                continue
            for j in receiver_order[:10]:
                if i == j or r[i] <= r[j] + 1e-12:
                    continue

                # needed reduction of the current peak
                gap = max_val - second_val
                if gap <= 0:
                    continue

                # epsilon that would bring the peak down to the second‑largest value
                eps_needed = gap / (2.0 * (r[i] - r[j]))
                # limit epsilon to something realistic (donor cannot give more than it has)
                eps = min(a[i] * 0.5, eps_needed * 0.99)
                if eps <= 1e-15:
                    continue

                a_new = a.copy()
                a_new[i] -= eps
                a_new[j] += eps

                # evaluate the new configuration
                max_new, second_new, conv_new = _max_and_second_conv(a_new, fft_len)
                new_score = 2.0 * n * max_new

                if new_score < best_score - 1e-12:
                    # accept the move
                    best_score = new_score
                    best_a = a_new.copy()
                    a = a_new
                    max_val, second_val, conv = max_new, second_new, conv_new
                    improved = True
                    no_improve = 0
                    print(f"[greedy] score={best_score:.6f}")
                    break
            if improved:
                break

        if not improved:
            # fallback: tiny random transfer (helps to escape flat plateaus)
            i1, i2 = np.random.choice(n, 2, replace=False)
            eps = (np.random.rand() - 0.5) * 0.02 * a[i1]  # up to ±1 % of donor
            # keep non‑negativity
            if eps > 0:
                eps = min(eps, a[i1])
                a[i1] -= eps
                a[i2] += eps
            else:
                eps = -eps
                eps = min(eps, a[i2])
                a[i2] -= eps
                a[i1] += eps

            max_val, second_val, conv = _max_and_second_conv(a, fft_len)
            new_score = 2.0 * n * max_val
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                print(f"[random] score={best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1

    return best_a, best_score

    """

    """
    np.random.seed(seed)
    finish_time = time.time() + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # 1. Load an initial vector (prefer the reference if available)

    init = np.maximum(init, 0.0)

        phase_deadline = min(finish_time, time.time() + 30.0)
        cand_seq, cand_score = _greedy_local_optimize(best_seq.copy(),
                                                      phase_deadline)

        if cand_score < best_score - 1e-12:

        if time.time() >= finish_time:

    # 4. Final normalisation & return
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.ones_like(best_seq) / len(best_seq)

    return best_seq.tolist()
