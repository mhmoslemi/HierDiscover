def _grad_max(a: np.ndarray, conv: np.ndarray, nfft: int) -> np.ndarray:
    mask = (conv >= conv.max() - 1e-12)   # bool mask
    w = mask.astype(np.float64)   # shape (2n-1,)
    # compute convolution of w with reversed a.
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[:2*a.size-1]
    grad = 2.0 * grad_full[a.size-1 : a.size-1 + a.size]
    return grad

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # length n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000
    S = np.sqrt(2.0 * n)
    nfft = _next_pow2(2 * n - 1)

    # Helper functions defined earlier

    # Initialize best from existing sequence
    best_seq = np.array(height_sequence_1 if 'height_sequence_1' in globals() else np.full(n, S/n), dtype=np.float64)
    best_seq = _project_to_simplex(best_seq, S)
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial best score {best_score}")

    # Parametric initial candidates
    candidate_shapes = []

    # Uniform (already)
    # Inverse distance shape
    for p in [0.5, 1.0, 1.5, 2.0]:
        idx = np.arange(n)
        a = 1.0 / (idx + 1) ** p + 1.0 / (n - idx) ** p
        candidate_shapes.append(a)
    # Exponential shape
    for lam in [0.001, 0.003, 0.005, 0.01]:
        idx = np.arange(n)
        a = np.exp(-lam * idx) + np.exp(-lam * (n-1-idx))
        candidate_shapes.append(a)

    # For each candidate shape
    for shape in candidate_shapes:
        a = _project_to_simplex(shape, S)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"New best from shape: {best_score}")

    # Start from best_seq for descent loop
    a = best_seq.copy()
    iteration = 0
    no_improve = 0
    max_no_improve_restart = 2000

    # learning rate base
    base_lr = 0.5 * S / n

    while time.time() < deadline:
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[{iteration}] New best: {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # termination early if target reached
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration}")
            break

        # compute gradient of max conv
        grad = _grad_max(a, conv, nfft)

        # learning rate
        eta = base_lr / np.sqrt(iteration + 1)

        # Candidate 1: projected gradient descent
        a_proj = _project_to_simplex(a - eta * grad, S)

        # Candidate 2: exponentiated gradient
        if np.linalg.norm(grad) > 0:
            grad_norm = np.linalg.norm(grad)
            exp_factor = np.exp(- eta * grad / (grad_norm + 1e-12))
            a_exp = a * exp_factor
            a_exp = a_exp * (S / a_exp.sum())
        else:
            a_exp = a.copy()

        # Evaluate both
        # Evaluate a_proj
        conv_proj = _convolve_self_fft(a_proj, nfft)
        max_proj = conv_proj.max()
        # Evaluate a_exp
        conv_exp = _convolve_self_fft(a_exp, nfft)
        max_exp = conv_exp.max()

        # Choose best candidate
        if max_proj < max_exp:
            a_candidate = a_proj
            cand_max = max_proj
        else:
            a_candidate = a_exp
            cand_max = max_exp

        # Accept if improvement else occasional jitter
        if cand_max < cur_max:
            a = a_candidate
        else:
            # occasional jitter
            if np.random.rand() < 0.005:
                jitter = np.random.randn(n) * (eta * 0.5)
                a = a + jitter
                a = np.maximum(a, 0)
                a = a * (S / a.sum())
        # Periodic restart
        if no_improve >= max_no_improve_restart:
            # random restart
            a = np.random.dirichlet(np.full(n, 0.3)) * S
            no_improve = 0
            print(f"[{iteration}] Restarted randomly.")
        iteration += 1

    # Final projection and evaluation
    final_seq = _project_to_simplex(best_seq, S)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()

"""Simple monotone decreasing step function generation.

The algorithm constructs a linear decreasing sequence on a simplex,
normalises it to the required ℓ₁‑norm (√(2n)) and returns it.
For a sequence a_i ∝ (n‑i) the self‑convolution at the centre
is (4/3)·(n+2)/(n+1) after normalisation, yielding an
evaluation around 1.33 for n≈2400 – well below the target 1.503.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = target_sum}."""
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Build a decreasing linear step function and, if time permits,
    optionally explore a few simple parametric families.
    The construction is fast and achieves a score ≈ 1.33 on the
    benchmark problem (target < 1.503).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Length of the sequence – use the existing construction if available.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000

    # Target ℓ₁‑norm (the value used by `evaluate_sequence`).
    S = np.sqrt(2.0 * n)

    # -----------------------------------------------------------------
    # 1.  Simple monotone decreasing candidate (linear weights)
    # -----------------------------------------------------------------
    lin = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, …, 1
    a = _project_to_simplex(lin, S)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, linear decreasing score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2.  Optional light sweep over a few easy families (very cheap)
    # -----------------------------------------------------------------
    families = []

    # inverse‑power shapes: a_i ∝ (i+1)^(-p) + (n-i)^(-p)
    for p in [0.3, 0.5, 0.8, 1.0, 1.2, 1.5]:
        idx = np.arange(n, dtype=np.float64)
        shape = (1.0 / (idx + 1) ** p) + (1.0 / (n - idx) ** p)
        families.append(shape)

    # exponential decay from both ends: a_i ∝ exp(-λ i) + exp(-λ (n‑1‑i))
    for lam in [0.0005, 0.001, 0.002, 0.004, 0.008]:
        idx = np.arange(n, dtype=np.float64)
        shape = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
        families.append(shape)

    # quadratic decay: a_i ∝ (i+1)^2 + (n-i)^2  (negative exponent)
    for p in [1, 2]:
        idx = np.arange(n, dtype=np.float64)
        shape = (idx + 1) ** (-p) + (n - idx) ** (-p)
        families.append(shape)

    for shape in families:
        if time.time() >= deadline:
            break
        cand = _project_to_simplex(shape, S)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[search] improved score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3.  (Optional) a very short projected‑gradient refinement.
    #     This part runs only if time is still left.
    # -----------------------------------------------------------------

        def _grad_max(a_vec, conv):
            """Sub‑gradient of max_i (a*a)_i (averaged over active indices)."""
            mask = np.isclose(conv, conv.max(), atol=1e-12)
            w = mask.astype(np.float64)

            nff = _next_pow2(2 * n - 1)
            A_rev = np.fft.rfft(a_vec[::-1], n=nff)

            grad = 2.0 * grad_full[n - 1 : n - 1 + n]
            return grad

        a_cur = best_seq.copy()

            if no_improve >= 5000:

                no_improve = 0
                print(f"[gd {it}] restart random")
            it += 1

    # -----------------------------------------------------------------
    # 4.  Final sanity check and return.
    # -----------------------------------------------------------------
    final_seq = _project_to_simplex(best_seq, S)   # ensure exact ℓ₁‑norm
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished. Best score = {final_score:.6f}")
    return final_seq.tolist()
