def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    start=time.time()
    deadline=start+budget_s-2

    best_score=inf; best_seq=None

    # List of lengths to try
    lengths = [200,400,600,800,1200,1600,2000,2500,3000,3500,4000]
    # Random shuffle
    np.random.shuffle(lengths)

    for n in lengths:
        if time.time()>=deadline: break

        # initialize using either uniform random or upsampled best_seq if exists.
        if best_seq is not None and len(best_seq)==n:
            a = np.array(best_seq)
        elif best_seq is not None and len(best_seq) < n:
            # upsample by interpolation
            a = np.interp(np.linspace(0,1,len(best_seq)), np.linspace(0,1,n), best_seq) # not correct direction; but we can use np.interp with x values.
        else:
            a = np.random.rand(n)
            a = a / a.sum()

        # apply exponentiated gradient descent
        eta0 = 5.0 # initial learning rate; tune
        max_iter = 50000

        # precomputes for FFT length
        fft_len = 1 << ((2*n-1).bit_length())
        it=0
        while time.time()<deadline and it<max_iter:
            it+=1

            # conv via FFT
            Af = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]

            max_val = conv.max()
            # compute current score
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time()-start:6.2f}s] Improved: n={n}, iter={it}, score={cur_score:.6f}")

            # compute set of near-maximum indices
            eps = max_val * 0.005 # 0.5% relative tolerance
            # find indices where conv >= max_val - eps
            K = np.where(conv >= max_val - eps)[0]
            # compute gradient contributions
            g = np.zeros_like(a)
            for k in K:
                low = max(0, k-(n-1))
                high = min(n-1, k)
                i_idx = np.arange(low, high+1)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]

            # exponentiated gradient step
            eta = eta0 / math.sqrt(it)
            # Avoid overflow/underflow:
            a = a * np.exp(-eta * g)
            # renormalize to sum=1
            s = a.sum()
            if s <= 0:
                a = np.full_like(a, 1.0/n)
            else:
                a = a / s

        # after loops, iterate a few more times with simple projection subgrad to fine-tune
        # see below
    # after lengths loop
    ...

    return best_seq.tolist()

"""
Propose a candidate step‑function using a multi‑peak projected sub‑gradient
method with progressive up‑sampling.

The evaluation function is scale‑invariant, so we keep the coefficients on
the probability simplex (sum = 1, a_i ≥ 0).  For a given length n we minimise

    score = 2 * n * max(conv(a, a))

where conv is the linear auto‑convolution.  The objective is convex: it is the
maximum of a family of quadratic forms.  Its sub‑gradient at a is the sum of
the gradients of all near‑maximal convolution entries, i.e.

    g_i = 2 * Σ_{k∈K} a_{k-i}       (indices out of range are ignored)

with K = {k : conv[k] is within a small tolerance of the maximum}.
After each sub‑gradient step we project back onto the simplex (Euclidean
projection of Duchi et al.).  Starting from the best known sequence we
iteratively up‑sample (linear interpolation) to larger lengths (≈1.5× each
step) and continue the optimisation.  The algorithm stops after the allotted
time (≈1000 s) and returns the best sequence found.

This approach differs from the previous submission by:
* using several near‑maximal indices instead of only the single worst one;
* progressively increasing the problem size while keeping a good shape;
* a simple time‑adaptive iteration budget per size.

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of_seq(a: np.ndarray) -> float:
    """Return the evaluation score for a sequence that already sums to 1."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) with a low
    `evaluate_sequence` score.  The routine runs for at most ``budget_s``
    seconds and prints intermediate improvements.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Initialise from the previously found construction if available.
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the global namespace
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 0:
            init = init / s
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"Initialised from height_sequence_1: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        # No previous construction – start from a random simplex point.
        init_len = 1200
        init = np.random.rand(init_len)
        init = init / init.sum()
        best_seq = init.copy()
        best_score = _score_of_seq(best_seq)
        print(f"Random initialisation: n={init_len}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Progressive up‑sampling loop.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    cur_n = current.shape[0]

    # target maximum length (adjustable – keep it reasonable for the time budget)
    MAX_LEN = 6000

    while True:
        # decide the next length (≈1.5× bigger, but cap at MAX_LEN)
        next_n = int(cur_n * 1.5)
        if next_n > MAX_LEN:
            next_n = MAX_LEN
        if next_n <= cur_n:
            break  # cannot increase further

        # ----- up‑sample current shape to the new size -----------------
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, next_n)
        a = np.interp(x_new, x_old, current)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(next_n, 1.0 / next_n)
        else:
            a = a / a_sum

        # ----- optimisation parameters for this size -----------------
        # iteration budget scales inversely with n so that total runtime stays balanced
        base_iters = 50000
        max_iter = int(base_iters * 1200 / next_n) + 20000
        max_iter = max(max_iter, 15000)   # at least a few thousand iterations

        step0 = 0.6                       # initial step size (empirically good)
        eps_rel = 0.01                    # tolerance for “near‑max” indices
        fft_len = 1 << ((2 * next_n - 1).bit_length())

        # ----- main optimisation loop for this length ----------------
        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1

            # linear auto‑convolution via FFT
            Af = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * next_n - 1]

            max_val = conv.max()
            cur_score = 2.0 * next_n * max_val

            # record improvement
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Improved: n={next_n}, "
                      f"iter={it}, score={cur_score:.6f}")

            # ----- multi‑peak sub‑gradient ---------------------------------
            eps = max_val * eps_rel
            K = np.where(conv >= max_val - eps)[0]   # indices of near‑max entries
            g = np.zeros_like(a)

            # accumulate contributions of all active convolution entries
            for k in K:
                low = max(0, k - (next_n - 1))
                high = min(next_n - 1, k)
                i_idx = np.arange(low, high + 1)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]

            # ----- projected sub‑gradient step -----------------------------
            step = step0 / math.sqrt(it)          # diminishing step size
            a_new = a - step * g
            a = _project_simplex(a_new)

        # ------------------------------------------------------------------
        # 3. Prepare for the next up‑sampling stage
        # ------------------------------------------------------------------
        current = a
        cur_n = next_n

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final short refinement on the best length found (optional)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # One last refinement pass on the best length (few thousand iters)
        n = best_seq.shape[0]
        a = best_seq.copy()
        fft_len = 1 << ((2 * n - 1).bit_length())
        step0 = 0.4
        eps_rel = 0.008
        for it in range(1, 20001):
            if time.time() >= deadline:
                break
            Af = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(Af * Af, n=fft_len)[: 2 * n - 1]
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:6.2f}s] Final refine: n={n}, "
                      f"iter={it}, score={cur_score:.6f}")

            eps = max_val * eps_rel
            K = np.where(conv >= max_val - eps)[0]
            g = np.zeros_like(a)
            for k in K:
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                i_idx = np.arange(low, high + 1)
                j_idx = k - i_idx
                g[i_idx] += 2.0 * a[j_idx]

            step = step0 / math.sqrt(it)
            a_new = a - step * g
            a = _project_simplex(a_new)

        # keep the refined version if it improved
        if a is not None and _score_of_seq(a) < best_score:
            best_seq = a.copy()
            best_score = _score_of_seq(best_seq)

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python floats).
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely fallback – uniform 100‑element sequence
        best_seq = np.full(100, 1.0 / 100)

    return [float(x) for x in best_seq]
