"""
Propose a low‑score step‑function.

Algorithm
---------
* Several random restarts are performed (different lengths, different
  analytic “seed” shapes such as the sqrt‑derivative, power‑law, linear,
  arcsine or a random Dirichlet vector).  Each seed is normalised to
  unit ℓ¹‑norm (the evaluation is homogeneous, so any normalisation is
  fine).
* For a fixed seed a projected sub‑gradient descent is run on the
  objective 2·n·max(conv(a,a)).  The gradient of the ‑max‑term at the
  current maximal convolution index k is
        ∂/∂a_j  Σ_{i+i′=k} a_i a_{i′} = 2·a_{k‑j}  (if the index lies in [0,n)).
  The step is taken, then the vector is projected onto the probability
  simplex (non‑negative entries with sum 1) to respect the constraints.
  Step size decays slowly; if an iteration does not improve the score the
  step is reduced, otherwise it is slightly increased.
* The best sequence over all restarts is kept and finally returned as a
  plain Python list.  Convolutions are computed with FFT for speed, the
  inner optimisation typically performs a few thousand cheap iterations in
  <0.3 s, leaving the full 1000 s budget for many restarts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex: v >= 0 and sum(v) = 1."""
    # Duchi et al. (2008) – O(n log n)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _target_sum(n: int) -> float:
    """The normalisation used in the reference code (sum = √(2n))."""
    return np.sqrt(2.0 * n)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """
    a_i ∝ √(i+1+δ) – √(i+δ)    (discrete derivative of √x)
    Scaled to unit ℓ¹‑norm.
    """
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    base = np.maximum(base, 0.0)
    base /= base.sum()
    return base


def _power_sequence(n: int, delta: float, p: float) -> np.ndarray:
    """
    Generalised power‑law seed:
        a_i ∝ (i+1+δ)^p – (i+δ)^p   ,  0 < p < 1
    Normalised to unit ℓ¹‑norm.
    """
    i = np.arange(n, dtype=float)
    base = (i + 1.0 + delta) ** p - (i + delta) ** p
    base = np.maximum(base, 0.0)
    base /= base.sum()
    return base


def _linear_sequence(n: int) -> np.ndarray:
    """Decreasing linear shape a_i ∝ (n‑i)."""
    base = np.arange(n, 0, -1, dtype=float)
    base /= base.sum()
    return base


def _arcsine_sequence(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density:
        a_i ∝ 1 / sqrt{ (i+½)·(n+½‑i) } .
    Normalised to unit ℓ¹‑norm.
    """
    i = np.arange(n, dtype=float) + 0.5
    base = 1.0 / np.sqrt(i * (n + 0.5 - i))
    base = np.maximum(base, 0.0)
    base /= base.sum()
    return base


def _random_dirichlet_sequence(n: int) -> np.ndarray:
    """Simple random positive vector (Dirichlet(1,…,1))."""
    base = np.random.rand(n)
    base /= base.sum()
    return base


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute full convolution a*a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()          # next power of two
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)
    return conv[: 2 * n - 1]


def _max_conv_and_index(a: np.ndarray) -> tuple[float, int]:
    """
    Return (max_convolution_value, index_of_max) for self‑convolution.
    The index is in the range [0, 2·n‑2] (i.e. the raw convolution index).
    """
    conv = _conv_fft(a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    init_step: float,
    decay: float,
    time_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Returns the best sequence encountered and its score.
    """
    n = a0.shape[0]
    a = a0.copy()
    # current objective (sum is 1 → denominator = 1)
    cur_max, _ = _max_conv_and_index(a)
    cur_score = 2.0 * n * cur_max

    best_seq = a.copy()
    best_score = cur_score

    step = init_step

    for it in range(max_iters):
        if time.time() >= time_deadline:
            break

        # full convolution and arg‑max
        conv = _conv_fft(a)
        max_val = conv.max()
        max_idxs = np.where(conv >= max_val - 1e-14)[0]   # ties are rare

        # average sub‑gradient over all tying indices
        grad = np.zeros_like(a)
        for k in max_idxs:
            idx = k - np.arange(n)                # candidate partner index
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        grad /= float(len(max_idxs))

        # propose a step
        a_proposed = a - step * grad
        a_proposed = _project_simplex(a_proposed)

        # evaluate proposal
        prop_max, _ = _max_conv_and_index(a_proposed)
        prop_score = 2.0 * n * prop_max

        # simple back‑tracking: keep improvement, otherwise shrink step
        if prop_score < cur_score:
            a = a_proposed
            cur_score = prop_score
            # slightly increase step when we move downhill
            step = step * 1.01
        else:
            # reject – reduce step size
            step = step * 0.90

        # track global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # gradual decay
        step = step * decay

        # avoid step becoming absurdly small
        if step < 1e-12:
            step = 1e-12

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:

    """

    # Lengths we will explore – a reasonable spread from a few hundred to a few thousand
    possible_lengths = list(range(300, 2501, 200))

    # Parameters for the sub‑gradient optimisation
    inner_max_iters = 3000
    init_step = 0.02
    decay = 0.9995

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) pick a length and a seed shape
        # ------------------------------------------------------------------
        n = int(np.random.choice(possible_lengths))

        shape_type = np.random.choice(
            ["sqrt", "power", "linear", "arcsine", "random"]
        )
        if shape_type == "sqrt":
            delta = np.random.rand() * 4.0            # δ ∈ [0,4)
            seq = _analytic_sqrt_sequence(n, delta=delta)
        elif shape_type == "power":
            delta = np.random.rand() * 4.0
            p = np.random.uniform(0.15, 0.90)         # exponent 0.15–0.90
            seq = _power_sequence(n, delta=delta, p=p)
        elif shape_type == "linear":
            seq = _linear_sequence(n)
        elif shape_type == "arcsine":
            seq = _arcsine_sequence(n)
        else:  # "random"
            seq = _random_dirichlet_sequence(n)

        # make sure we have a valid ℓ¹‑normalised vector
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0:
            continue
        seq /= seq.sum()  # normalise to unit sum

        # ------------------------------------------------------------------
        # 2) quick sanity check (optional)
        # ------------------------------------------------------------------
        init_score = 2.0 * n * _max_conv_and_index(seq)[0]
        if init_score < best_score:
            best_score = init_score
            best_seq = seq.copy()
            print(f"[init] n={n:4d} type={shape_type:8s} score={init_score:.6f}")

        # ------------------------------------------------------------------
        # 3) run the projected sub‑gradient descent
        # ------------------------------------------------------------------
        inner_deadline = time.time() + 0.32  # roughly 300 ms per restart
        seq_opt, score_opt = _subgradient_descent(
            seq,
            max_iters=inner_max_iters,
            init_step=init_step,
            decay=decay,
            time_deadline=inner_deadline,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[impr] n={n:4d} type={shape_type:8s} score={score_opt:.6f}")

        # early exit if we already beat the target by a comfortable margin
        if best_score < 1.51:
            # keep searching for a few more candidates but we are already very good
            pass

    # Fallback – extremely unlikely to be used
    if best_seq is None:
        best_seq = np.ones(10, dtype=float)
        best_seq /= best_seq.sum()

    # Convert to a plain list for the external evaluator
    return best_seq.tolist()
