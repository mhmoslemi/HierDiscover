"""
Propose a candidate step‑function that minimises the provided evaluation
function.

The algorithm consists of two phases:

1. **LP‑phase** – A linear program is solved that, for the current
   sequence `a`, finds a second sequence `b` of maximal ℓ₁‑mass while
   keeping every entry of the cross‑convolution `a*b` below the current
   worst self‑convolution value `M = max(a*a)`.  The solution `b` is
   scaled to the required ℓ₁‑norm `√(2n)` and mixed with the current
   sequence.  This dramatically reduces the maximal convolution value
   in only a few iterations.

2. **Gradient‑phase** – A projected (sub‑)gradient descent on the true
   ℓ∞‑objective is performed.  The gradient is taken only with respect
   to the currently active (maximal) convolution entries, which yields
   the exact sub‑gradient of the max‑function.  After each step the
   iterate is projected onto the simplex `{x≥0, Σx = √(2n)}`.  Momentum,
   occasional random jitter and restarts are used to escape plateaus.

Both phases respect the time budget (default 1000 s).  The best sequence
found during the run is returned.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real output)."""
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex
    {x ≥ 0, Σ x = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_matrix_from_f(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that A @ b = np.convolve(f, b).
    Each column i consists of f[i] repeated n times on the i‑th diagonal.
    """
    n = f.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    # Every diagonal gets the same value f[i] on all its n entries.
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _grad_max(a: np.ndarray, conv: np.ndarray, nfft: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    The gradient of a single active entry k is
        2 * (rev(a) * e_k)  (convolution).
    If several entries are tied for the maximum we average their gradients.
    """
    n = a.shape[0]
    max_val = conv.max()
    # treat all entries within a tiny tolerance as active
    active = np.isclose(conv, max_val, atol=1e-12)
    w = active.astype(np.float64)
    if w.sum() == 0:
        return np.zeros_like(a)
    w /= w.sum()                # turn into a probability vector

    # convolution of w with reversed a (same trick as in the soft‑max code)
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad

# Main routine
# ----------------------------------------------------------------------

                a = a_candidate
                cur_max = cand_max
                print(f"[LP-{lp_iter}] Mixed solution improved max to {cand_max:.6f}")
            else:
                # No progress – shrink the mixing factor for the next LP iteration
                t_mix = max(0.1, t_mix * 0.8)

        # Record best ever
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[LP-{lp_iter}] New global best: {best_score:.6f}")

        if best_score < 1.503:
            print("[LP] Target bound reached.")
            break

        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true max‑objective
    # ------------------------------------------------------------------
    nfft = _next_pow2(2 * n - 1)
    iteration = 0
    no_improve = 0
    max_no_improve = 1500
    base_eta = 0.3 * S / n            # baseline step size

    while time.time() < deadline:
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()

        # keep global best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
            print(f"[GD-{iteration:05d}] New best: {best_score:.6f}")
        else:
            no_improve += 1

        if best_score < 1.503:
            print("[GD] Target bound reached.")
            break

        # --------------------------------------------------------------
        # Sub‑gradient of the ℓ∞‑norm (max) of the convolution
        # --------------------------------------------------------------
        grad = _grad_max(a, conv, nfft)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            eta = base_eta / (1.0 + iteration * 1e-4)
            step = eta * grad / grad_norm
        else:
            step = np.zeros_like(a)

        a_candidate = _project_to_simplex(a - step, S)

        # accept the step only if it really reduces the max‑convolution
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        if cand_max < cur_max:
            a = a_candidate
        else:
            # occasional jitter to escape shallow plateaus
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * (eta * 0.5)
                a = _project_to_simplex(a + jitter, S)

        # occasional restart if we are stuck for a long time
        if no_improve >= max_no_improve:
            alpha = np.full(n, 0.3)
            a = np.random.dirichlet(alpha) * S
            no_improve = 0
            print(f"[GD-{iteration:05d}] Restarted from a random simplex point.")

        iteration += 1

    # ------------------------------------------------------------------
    # Finalise and return the best sequence discovered
    # ------------------------------------------------------------------
    final_seq = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} GD iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
