It removes the monotone‐sorting step of the previous implementation, uses a soft‑max parametrisation (so the non‑negativity and ℓ₁‑normalisation are always satisfied), and optimises a *smoothed* version of the “peak of the convolution’’ with an Adam‑style optimiser.  
A few different lengths are tried, a small number of the largest convolution entries are used to build a richer gradient, and occasional random jitter helps escape plateaus.

```python
"""
Propose a candidate step‑function sequence.

This implementation works with a soft‑max parametrisation
    a_i = exp(z_i) / Σ_j exp(z_j)   (a_i ≥ 0, Σ a_i = 1).

The objective is 2·n·max_k (a*a)_k.
We approximate the non‑differentiable max with a weighted sum over the
few largest convolution entries and apply Adam optimisation in the
log‑space (z).  The method is fast (FFT‑based convolution) and
explores several lengths (1000–6000).  No monotone‑sorting is performed,
so the optimiser can discover non‑monotone shapes that give a flatter
auto‑correlation and a lower bound than the earlier code.
"""

import time
import numpy as np
from typing import List

# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using FFT.  Returns a vector of length 2·n‑1.
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad)
    C = A * A                         # |A|²
    conv_full = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return conv_full


def _score_from_a(a: np.ndarray) -> float:
    """
    Compute the evaluation score (2·n·max(conv)) for a vector a with
    Σ a_i = 1.  The scaling invariant objective of evaluate_sequence
    equals this quantity.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max())


def _adam_update(z: np.ndarray,
                 grad_z: np.ndarray,
                 m: np.ndarray,
                 v: np.ndarray,
                 t: int,
                 lr: float,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray]:
    """
    Classic Adam update for the parameter vector z.
    Returns the updated (z, (m, v)).
    """
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    z_new = z - step
    return z_new, (m, v)


def _gradient_wrt_a(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Build a sub‑gradient w.r.t. a from the (few) largest convolution
    entries whose indices are given in top_idxs.
    For each selected offset k we add the contribution
        ∂ (a*a)_k / ∂ a_i = a_{k-i} + a_{i-k} .
    Since a is real and the convolution is symmetric, this reduces to
    2·a_{k-i} for the admissible i.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in top_idxs:
        # valid i satisfy 0 ≤ i < n and 0 ≤ k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]

    return grad


def _logits_to_a(z: np.ndarray) -> np.ndarray:
    """Soft‑max (numerically stable)."""
    z_shift = z - np.max(z)               # avoid overflow
    e = np.exp(z_shift)
    return e / e.sum()

            break

        print(f"\n=== Optimising length n={n} ===")

        # hyper‑parameters that scale with n
        lr0 = 0.05 / np.sqrt(n)            # base learning rate
        top_k_initial = max(1, n // 800)   # number of active offsets (≈1‑5)

        # initialise logits – start from a soft‑max of small Gaussian noise
        z = np.random.randn(n) * 0.02
        a = _logits_to_a(z)

        # Adam state
        m = np.zeros_like(z)
        v = np.zeros_like(z)
        t = 0

        # schedule for top_k (start with a few indices, then focus on the max)
        top_k = top_k_initial
        top_k_decay_iter = 20000   # after this many steps we shrink to 1

        while time.time() < deadline:
            # ----------------------------------------------------------------
            # 1) convolution & current score
            # ----------------------------------------------------------------
            conv = _conv_fft(a)
            cur_max = conv.max()
            cur_score = float(2.0 * n * cur_max)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{n}] New global best {best_score:.6f}")

            # ----------------------------------------------------------------
            # 2) build gradient from the largest convolution entries
            # ----------------------------------------------------------------
            if top_k == 1:
                ks = np.array([int(np.argmax(conv))], dtype=int)
            else:
                # ``np.argpartition`` is O(L) and gives the indices of the top‑k
                ks = np.argpartition(conv, -top_k)[-top_k:]

            grad_a = _gradient_wrt_a(a, ks)

            # ----------------------------------------------------------------
            # 3) convert to gradient w.r.t. logits (soft‑max chain rule)
            # ----------------------------------------------------------------
            dot = a.dot(grad_a)
            grad_z = a * (grad_a - dot)

            # ----------------------------------------------------------------
            # 4) Adam update in log‑space
            # ----------------------------------------------------------------
            t += 1
            # learning‑rate decays slowly with the iteration count
            lr_t = lr0 / (1.0 + t * 1e-5)
            z, (m, v) = _adam_update(z, grad_z, m, v, t, lr_t)

            # recompute a
            a = _logits_to_a(z)

            # ----------------------------------------------------------------
            # 5) occasional random jitter to avoid getting stuck
            # ----------------------------------------------------------------
            if t % 1500 == 0:
                jitter = np.random.randn(n) * 1e-3
                z += jitter

            # ----------------------------------------------------------------
            # 6) gradually focus the gradient on the single maximum entry
            # ----------------------------------------------------------------
            if t == top_k_decay_iter:
                top_k = 1   # after a while we only optimise the true max

    # ------------------------------------------------------------------
    # Fallback – uniform sequence if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform seq score {best_score:.6f}")

    # Final sanity check (optional – evaluate_sequence is cheap)
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"Final best score (official): {final_score:.6f} "
              f"(length {len(best_seq)})")
    except Exception:
        print("Warning: evaluation of final sequence failed.")

    return best_seq
```
