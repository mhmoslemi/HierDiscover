def fast_score_seq(a):
    n = a.size
    s = a.sum()
    if s < 1e-12: return np.inf
    size = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

"""
A two‑stage optimizer for the convolution‑peak metric.

Stage 1 – Analytic grid search
    Search the two‑parameter family a_i = (i+1+δ)^{-p} (p≈0.5) for a short
    sequence (default n=200).  The shape is normalised to ℓ¹‑norm √(2 n) so
    that the evaluation value equals the maximal self‑convolution, i.e.
    score = max(conv(a,a)).  This gives a good starting point (≈1.55).

Stage 2 – Linear‑programming improvement
    Given the current f, let M = max(f*f).  Solve the LP
        maximise   Σ g_i
        subject to f * g ≤ M   (convolution bound, one constraint per shift)
                   g_i ≥ 0 .
    The LP computes a direction g that can carry *more* ℓ¹‑mass without
    increasing any convolution entry above M.  Mixing f with the normalised
    g (f←(1‑t)f + t·g) strictly reduces the peak M; iterating this step
    drives the score down.  The LP is tiny (n≈200 → ≤ 400 constraints) and
    solved with SciPy’s HiGHS optimiser.

Optional fine‑tuning
    After the LP loop a few projected‑gradient steps are applied to
    escape possible plateaus.

The whole routine respects the 1000 s budget, prints progress and finally
returns the best sequence as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    """
    Compute max of the linear convolution a*a using an FFT.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(conv.max())


def _make_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A of size (2n‑1)×n such that (f*g)[k] = (A·g)[k].
    Row k contains the coefficients f[k‑i] for all i where the index is
    in range.
    """
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        A[k, idx] = f[k - idx]
    return A


def _build_power_law(n: int, delta: float, p: float) -> np.ndarray:
    """
    a_i = (i+1+δ)^{-p}, normalised to ℓ¹‑norm √(2n).
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + delta
    a = idx ** (-p)
    # protect against negative or NaN (should not happen for the chosen range)
    a = np.maximum(a, 0.0)
    target = np.sqrt(2.0 * n)
    a *= target / a.sum()
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a score
    below the target 1.503.  The function runs for at most ``budget_s``
    seconds and prints intermediate progress.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a modest dimension for the LP phase.
    n = 200                       # small enough for fast LPs, large enough to be meaningful

    # ------------------------------------------------------------------
    # 2.  Coarse grid search over the (δ,p) family.
    best_f = None
    best_score = np.inf
    deltas = np.linspace(-0.99, 1.5, 81)          # include strong negative offsets
    ps = np.linspace(0.40, 0.60, 11)              # around the analytic exponent 0.5

    print("[grid] start scanning (delta, p) …")
    for delta in deltas:
        for p in ps:
            f = _build_power_law(n, delta, p)
            cur = _conv_max(f)          # because f is already normalised
            if cur < best_score:
                best_score = cur
                best_f = f.copy()
        # keep an eye on the clock
        if time.time() > deadline:
            break

    print(f"[grid] best after scan: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Linear‑programming refinement.
    #    f is already normalised to sum √(2n) → the metric equals max(conv).
    f = best_f.copy()
    target_sum = np.sqrt(2.0 * n)

    iteration = 0
    while time.time() < deadline:
        M = _conv_max(f)                       # current peak
        # Build constraints for (f * g) ≤ M
        A = _make_constraint_matrix(f)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # Linear programme: maximise Σ g_i   ⇔   minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)

        # Solve the LP (HiGHS is fast for these sizes)
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if res.status != 0:
            # Solver failed – exit the LP loop
            print("[LP] solver failed or timed‑out, status:", res.status)
            break

        g = res.x
        sum_g = g.sum()
        sum_f = f.sum()

        # If we cannot increase the ℓ¹‑mass, we are at a (local) optimum
        if sum_g <= sum_f * (1.0 + 1e-9):
            print("[LP] no mass increase, terminating LP phase.")
            break

        # Normalise g to the same ℓ¹‑norm as f (required before mixing)
        g = g * (sum_f / sum_g)

        # Try a few mixing rates; pick the first that lowers the peak.
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            f_test = (1.0 - t) * f + t * g
            M_test = _conv_max(f_test)
            if M_test < M:
                f = f_test
                M = M_test
                improved = True
                print(f"[LP iter {iteration:3d}] t={t:.2f} → M={M:.6f}")
                break
        if not improved:
            # No improvement with any step size → stop.
            print("[LP] mixing failed to improve, exiting LP loop.")
            break

        # keep track of global best
        if M < best_score:
            best_score = M
            best_f = f.copy()
            print(f"[LP] new global best M={best_score:.6f}")

        iteration += 1
        if iteration >= 25:   # prevent endless looping; 25 LP steps are more than enough
            break

    # ------------------------------------------------------------------
    # 4.  Light projected‑gradient fine‑tuning (optional, uses remaining time).
    #    Works directly on the original metric (2·n·max/∑²) but because we keep
    #    the ℓ¹‑norm fixed, the objective reduces to max(conv).
    step = 0.5                # initial step size (relative to current component)
    cur_best = best_score
    best_seq = best_f.copy()
    cur_f = f.copy()
    cur_M = _conv_max(cur_f)

    it = 0
    while time.time() < deadline:
        # compute conv and the index of the current maximum
        n_local = cur_f.size
        size = 1 << ((2 * n_local - 1).bit_length())
        fft_cur = np.fft.rfft(cur_f, size)
        conv = np.fft.irfft(fft_cur * fft_cur, size)[: 2 * n_local - 1]
        k_max = int(conv.argmax())

        # gradient of the active convolution entry
        g = np.zeros_like(cur_f)
        lo = max(0, k_max - (n_local - 1))
        hi = min(n_local - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] = 2.0 * cur_f[k_max - idx]            # ∂(f*f)[k_max]/∂f_i

        # project gradient to the simplex (preserve ℓ¹‑norm)
        avg = g.mean()
        delta = -step * (g - avg)

        trial = cur_f + delta
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            # fallback: skip this iteration
            continue
        trial *= (target_sum / s)                     # re‑normalise ℓ¹‑norm

        new_M = _conv_max(trial)

        # accept if improvement, otherwise occasional uphill move
        if new_M < cur_M:
            cur_f, cur_M = trial, new_M
            if new_M < cur_best:
                cur_best = new_M
                best_seq = trial.copy()
                print(f"[grad] new best M={cur_best:.6f} (it {it})")
        else:
            if np.random.rand() < 1e-4:
                cur_f, cur_M = trial, new_M

        # adapt step size slowly
        if it % 2000 == 0 and it > 0:
            step = max(step * 0.9, 1e-5)

        if it % 5000 == 0:
            print(f"[grad it {it:5d}] cur M={cur_M:.6f} best M={cur_best:.6f}")

        it += 1
        # stop early if we already beat the target
        if cur_best < 1.503:
            break

    # ------------------------------------------------------------------
    # 5.  Return the best sequence discovered.
    print(f"[final] best metric (max conv) = {cur_best:.6f}")
    # Convert to plain Python list; the external evaluator will apply its
    # clipping & normalisation, so we keep the raw floating‑point numbers.
    return best_seq.tolist()
