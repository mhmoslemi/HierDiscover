"""
Optimization of a non‑negative step function.
------------------------------------------------
The score returned by `evaluate_sequence` becomes simply the maximum value
of the self‑convolution of the vector once the vector is normalised so that
∑a_i = √(2 n) (because the factor 2·n/(∑a_i)² cancels out).

The algorithm below works in three stages:

1. **Initialisation** – a quick parametric grid search over the family
   a_i = (i + c)^(-α) (α≈0.5) is performed (or the supplied global
   `height_sequence_1` is used) and the best candidate is normalised.

2. **Projected sub‑gradient descent** – at each iteration the lags that
   realise the current maximum of the self‑convolution are identified.
   Their indicator vector is a (sub‑)gradient of the max‑convolution.
   A step is taken in the opposite direction, the vector is projected back
   onto the non‑negative orthant and re‑normalised to the target sum.
   The step size is chosen so that at most a few percent of the total mass
   is moved; a short back‑tracking line search guarantees monotone
   improvement.

3. **Polishing** – every few hundred iterations a tiny random mass transfer
   (hill‑climbing) and an occasional random restart from the parametric
   family are tried, allowing the method to escape shallow local minima.

All heavy work (self‑convolution and the gradient computation) is done
with FFTs, making each iteration cheap enough to perform thousands of
steps inside the 1000 s budget.  The best sequence found is returned as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level, no closures
# ----------------------------------------------------------------------


def _fft_self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Return the full (real) self‑convolution a*a using an FFT of length fft_len.
    The result is truncated to length 2*len(a)-1.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[: 2 * len(a) - 1]


def _fft_convolution(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Return the full (real) convolution x*y using an FFT of length fft_len.
    The result is truncated to length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def _gradient_from_active_mask(a: np.ndarray,
                               active_mask: np.ndarray,
                               fft_len_grad: int) -> np.ndarray:
    """
    Compute the sub‑gradient of max{a*a}_k at the current vector `a`.
    `active_mask` is a 0/1 vector of length 2*n-1 indicating the lags
    where the self‑convolution attains its maximum.
    The returned gradient has length n.
    """
    n = len(a)
    # correlation a ⋆ active_mask = (active_mask * a_rev) sliced appropriately
    a_rev = a[::-1]
    full = _fft_convolution(active_mask, a_rev, fft_len_grad)
    # gradient_i = 2 * (active_mask ⋆ a)[-i] = 2 * full[n-1 + i]
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _build_power_law_candidate(alpha: float,
                               c: float,
                               n: int,
                               target_sum: float) -> np.ndarray:
    """
    Return a normalised vector a_i = (i + c)^(-α) of length n with
    ∑a_i = target_sum.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + c) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        return None
    a *= target_sum / s
    return a

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a step‑function (list of non‑negative heights) that
    minimises `evaluate_sequence`.  The routine respects the time budget,
    runs a small parametric pre‑search, then performs projected
    sub‑gradient descent with occasional hill‑climbing and random restarts.
    """
    np.random.seed(seed)
    finish_time = time.time() + budget_s - 1.0   # small safety margin

    # --------------------------------------------------------------
    # 1. Problem size and target normalisation
    # --------------------------------------------------------------
    try:
        # use the reference solution if it exists
        init_vec = np.asarray(height_sequence_1, dtype=np.float64)
        n = len(init_vec)
    except Exception:
        # fall back to a modest length that still gives a good search space
        n = 4000
        init_vec = None

    target_sum = np.sqrt(2.0 * n)          # after normalisation the score equals max conv
    fft_len_self = 1 << ((2 * n - 1).bit_length())
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # 2. Obtain a decent starting point
    # --------------------------------------------------------------
    best_start = None
    best_start_max = np.inf

    # If we have a supplied vector, normalise it and compare
    if init_vec is not None and init_vec.size == n:
        a = np.maximum(init_vec, 0.0)
        if a.sum() == 0.0:
            a = np.full(n, target_sum / n)
        else:
            a = a * (target_sum / a.sum())
        conv = _fft_self_convolution(a, fft_len_self)
        max_conv = conv.max()
        best_start, best_start_max = a, max_conv

    # Quick coarse grid over (α, c) – cheap but often gives a better start
    alphas = np.linspace(0.35, 0.80, 10)          # 10 values
    c_vals = np.geomspace(1.0, n / 2.0, 10)      # geometric spacing

    for alpha in alphas:
        if time.time() > finish_time:
            break
        for c in c_vals:
            a = _build_power_law_candidate(alpha, float(c), n, target_sum)
            if a is None:
                continue
            conv = _fft_self_convolution(a, fft_len_self)
            max_conv = conv.max()
            if max_conv < best_start_max:
                best_start, best_start_max = a, max_conv
                print(f"[grid] α={alpha:.3f} c={c:.1f} → max_conv={max_conv:.6f}")

    # If everything failed, fall back to a uniform vector
    if best_start is None:
        best_start = np.full(n, target_sum / n)
        best_start_max = _fft_self_convolution(best_start, fft_len_self).max()

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient descent
    # --------------------------------------------------------------
    current = best_start.copy()
    cur_max = best_start_max
    best_seq = current.copy()
    best_score = cur_max

    # parameters for the descent
    move_fraction = 0.02          # at most 2 % of total mass moved per step
    iteration = 0
    stagnation = 0                # counts iterations without improvement

    while time.time() < finish_time:
        # self‑convolution and maximum
        conv = _fft_self_convolution(current, fft_len_self)
        cur_max = conv.max()

        # active set (lags achieving the maximum, up to a tiny tolerance)
        tol = 1e-12 * cur_max
        active_mask = (conv >= cur_max - tol).astype(np.float64)

        # sub‑gradient from the active set
        grad = _gradient_from_active_mask(current, active_mask, fft_len_grad)

        # L1 norm of the gradient – decides the step size
        grad_l1 = np.abs(grad).sum()
        if grad_l1 == 0.0:
            break

        # step size: move at most `move_fraction * target_sum` total mass
        target_move = move_fraction * target_sum
        eta = target_move / grad_l1

        # simple back‑tracking line search (max 5 reductions)
        improved = False
        for _ in range(5):
            cand = current - eta * grad
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                break
            cand = cand * (target_sum / s)          # re‑normalise

            new_conv = _fft_self_convolution(cand, fft_len_self)
            new_max = new_conv.max()
            if new_max + 1e-12 < cur_max:           # strict improvement
                current = cand
                cur_max = new_max
                improved = True
                if new_max < best_score:
                    best_score = new_max
                    best_seq = cand.copy()
                    print(f"[subgrad] iter={iteration:04d} max={new_max:.6f}")
                break
            else:
                eta *= 0.5                           # shrink step and retry

        if not improved:
            stagnation += 1
        else:
            stagnation = 0

        # ----------------------------------------------------------
        # 4. Periodic random hill‑climbing move (small mass transfer)
        # ----------------------------------------------------------
        if iteration % 100 == 0:

            hc[i1] -= delta
            hc[i2] += delta
            hc = np.maximum(hc, 0.0)
            hc = hc * (target_sum / hc.sum())
            hc_max = _fft_self_convolution(hc, fft_len_self).max()
            if hc_max + 1e-12 < cur_max:
                current = hc
                cur_max = hc_max
                if hc_max < best_score:
                    best_score = hc_max
                    best_seq = hc.copy()
                    print(f"[hill] iter={iteration:04d} max={hc_max:.6f}")

        # ----------------------------------------------------------
        # 5. Occasional random restart from the power‑law family
        # ----------------------------------------------------------
        if iteration % 500 == 0 and np.random.rand() < 0.2:
            r_alpha = np.random.uniform(0.35, 0.80)
            r_c = np.random.uniform(1.0, n / 2.0)
            cand = _build_power_law_candidate(r_alpha, r_c, n, target_sum)
            cand_max = _fft_self_convolution(cand, fft_len_self).max()
            if cand_max + 1e-12 < cur_max:
                current = cand
                cur_max = cand_max
                if cand_max < best_score:
                    best_score = cand_max
                    best_seq = cand.copy()
                    print(f"[restart] iter={iteration:04d} α={r_alpha:.3f} c={r_c:.1f} max={cand_max:.6f}")

        # ----------------------------------------------------------
        # 6. Adaptive reduction of step magnitude if we have stalled
        # ----------------------------------------------------------
        if stagnation >= 500:
            # become more conservative – halve the move fraction
            move_fraction = max(0.001, move_fraction * 0.5)
            stagnation = 0
            print(f"[adapt] iter={iteration:04d} reduced move_fraction to {move_fraction:.5f}")

        iteration += 1

    # --------------------------------------------------------------
    # 7. Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    return best_seq.tolist()
