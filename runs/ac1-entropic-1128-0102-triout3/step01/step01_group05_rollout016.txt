def project_to_simplex(v):
    """Project vector v onto the probability simplex (sum=1, nonnegative)."""
    # Based on algorithm from Duchi et al., 2008.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

        size = 1 << ((2*n-1).bit_length())

        c = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]  # conv

Define project_simplex(v):

```
def project_simplex(v):
    """Projection onto the probability simplex (sum=1, nonnegative)."""
    # Using algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        # all non-positive, project to uniform?
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑value step‑function using a few parametric families,
heavy‑weight projected gradient descent on a smooth approximation of the
max‑convolution and a short random hill‑climber for final polishing.
The algorithm works entirely with NumPy/FFT, stays within the 1 000 s
budget and respects the non‑negative sum‑one constraint required by
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast version of the provided evaluation (no security checks)
def fast_score(a: np.ndarray) -> float:
    """2 * n * max(conv(a, a)) / (sum(a) ** 2) using FFT (no checks)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # length for a linear convolution (next power of two for speed)
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (non‑negative, sum = 1)
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety – renormalise
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Family 1 – simple power‑law (the classic 1/√(i) shape)
def build_power_sequence(n: int, exponent: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-exponent}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = np.power(idx, -exponent)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Family 2 – symmetric sum of two power‑laws (U‑shaped)
def build_symmetric_power_sequence(n: int, exponent: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-exponent} + (n−i+offset)^{-exponent}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = np.power(i + offset, -exponent)
    right = np.power((n - i + 1) + offset, -exponent)
    a = left + right
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max gradient descent on the quantity
#   F(a) = 2 * max_t ∫ f(t)f(t+τ) dτ  (discretised via convolution)
def smooth_max_gradient_descent(start: np.ndarray,
                               deadline: float,
                               beta_start: float = 150.0,
                               step_start: float = 0.8,
                               max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed L∞ norm of the self‑convolution.
    Returns the best sequence found and its fast_score.
    """
    a = start.copy()
    n = a.size
    best_a = a.copy()
    best_val = fast_score(a)

    beta = beta_start
    step = step_start

    it = 0
    while time.time() < deadline and it < max_iter:
        # ---- convolution a*a -------------------------------------------------
        size_conv = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(fft_a * fft_a, size_conv)[: 2 * n - 1]

        # ---- smooth max (log‑sum‑exp) ---------------------------------------
        m = conv.max()
        # shift to avoid overflow
        exp_weights = np.exp(beta * (conv - m))
        w = exp_weights / exp_weights.sum()          # soft‑max weights (sum=1)

        # ---- gradient = 2 * (a ⋆ w) ----------------------------------------
        size_corr = 1 << ((len(w) + n - 1).bit_length())
        fft_w = np.fft.rfft(w, size_corr)
        fft_a_rev = np.fft.rfft(a[::-1], size_corr)
        corr = np.fft.irfft(fft_w * fft_a_rev, size_corr)
        grad = 2.0 * corr[n - 1 : n - 1 + n]          # gradient vector length n

        # ---- take a step opposite to the gradient ---------------------------
        a_candidate = a - step * grad
        a_candidate = project_simplex(a_candidate)

        cand_val = fast_score(a_candidate)

        # ---- accept if improvement -------------------------------------------
        if cand_val < best_val:
            best_val = cand_val
            best_a = a_candidate.copy()
            a = a_candidate
            # slowly enlarge step when we are making progress
            step = min(step * 1.02, 2.0)
        else:
            # occasional uphill moves to escape tiny plateaus
            if np.random.rand() < 0.005:
                a = a_candidate
            else:
                # otherwise shrink the step size
                step *= 0.98

        # ---- anneal beta (sharpen the approximation of max) ------------------
        if (it + 1) % 3000 == 0:
            beta = min(beta * 1.3, 1200.0)

        # ---- occasional diagnostics -------------------------------------------
        if (it + 1) % 2000 == 0:
            print(f"[GD iter {it+1:5d}] best={best_val:.6f}, step={step:.4f}, beta={beta:.1f}")

        it += 1

    return best_a, best_val

# Simple random hill‑climber used for a final polish
def random_hill_climb(start: np.ndarray,
                      deadline: float,
                      step_factor: float = 0.05,
                      max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Very thin random mass‑transfer hill‑climbing (similar to the reference code).
    """
    cur = start.copy()
    cur_val = fast_score(cur)
    best = cur.copy()
    best_val = cur_val
    n = cur.size
    step = step_factor
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = step * cur[i] * np.random.rand()
        delta = min(delta, cur[i])               # keep non‑negative
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand /= cand.sum()                       # stay on the simplex
        cand_val = fast_score(cand)

        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best = cand_val, cand.copy()
        else:
            if np.random.rand() < 0.001:        # tiny uphill probability
                cur, cur_val = cand, cand_val

        # slowly shrink the transferable mass as we converge
        if it % 5000 == 0 and it > 0:
            step = max(step * 0.92, 5e-6)

        if it % 2000 == 0:
            print(f"[HC iter {it:6d}] best={best_val:.6f}, cur={cur_val:.6f}")

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the supplied
    `evaluate_sequence` metric.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # work length – a few thousand entries give a good trade‑off
    n = 3000

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1 – explore a modest grid of power‑law families (both plain and symmetric)
    exponents = np.linspace(0.35, 0.65, 7)          # 0.35 … 0.65
    offsets = np.linspace(0.0, 2.5, 11)            # 0 … 2.5

    for exp in exponents:
        for off in offsets:
            if time.time() >= deadline:
                break
            # plain power law
            seq = build_power_sequence(n, exp, off)
            val = fast_score(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
            # symmetric variant
            seq_sym = build_symmetric_power_sequence(n, exp, off)
            val_sym = fast_score(seq_sym)
            if val_sym < best_val:
                best_val, best_seq = val_sym, seq_sym.copy()

    print(f"[grid] best after param sweep: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2 – run a few gradient‑descent refinements from promising starts
    # we start from the current best and also from a few random parameter choices
    starts = [best_seq]
    # add a handful of random starts
    for _ in range(4):
        exp = np.random.uniform(0.3, 0.7)
        off = np.random.uniform(0.0, 3.0)
        if np.random.rand() < 0.5:
            starts.append(build_power_sequence(n, exp, off))
        else:
            starts.append(build_symmetric_power_sequence(n, exp, off))

    for start in starts:
        if time.time() >= deadline:
            break
        print("[GD] starting gradient descent …")
        seq_opt, val_opt = smooth_max_gradient_descent(start, deadline,
                                                       beta_start=150.0,
                                                       step_start=0.8,
                                                       max_iter=15000)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"[GD] new global best: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – final polish with a focused random hill‑climber (few seconds)
    if time.time() < deadline:
        print("[HC] final hill‑climbing …")
        seq_final, val_final = random_hill_climb(best_seq, deadline,
                                                 step_factor=0.04,
                                                 max_iter=5_000_000)
        if val_final < best_val:
            best_val, best_seq = val_final, seq_final.copy()
            print(f"[HC] final best improved to: {best_val:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] best value discovered: {best_val:.6f}")
    # convert to plain Python list for the caller
    return best_seq.tolist()
