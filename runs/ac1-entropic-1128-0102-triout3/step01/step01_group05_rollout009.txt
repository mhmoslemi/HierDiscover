"""
Propose a step‑function sequence using an alternating linear‑program
+ mixing scheme.

We start from a discretised 1/√t shape (the analytically‑motivated
kernel).  In each iteration we:
  * compute the current peak M = max((f*f))
  * solve a linear programme:
        maximise Σ g_j
        s.t. (f * g)_k ≤ M   for all k,
             g_j ≥ 0.
    The matrix of the convolution (f*g) is linear in g, therefore the
    problem is a standard LP.  The optimiser prefers to move mass to
    positions that create little extra convolution, i.e. towards the
    ends of the interval.
  * rescale the LP‑solution to the same ℓ¹–norm as f and blend the two
    sequences with a small mixing coefficient.
  * keep the best sequence seen so far.

The loop runs until the time budget (≈ 1000 s) expires.  A light‑weight
hill‑climbing step is performed occasionally to escape tiny plateaus.
The final sequence is returned as a plain list of floats; it will be
checked by the provided `evaluate_sequence` function.
"""
import numpy as np
import time
import scipy.optimize as opt

def fast_score(a: np.ndarray) -> float:
    """Return the value of ``evaluate_sequence`` without the security checks."""
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def build_initial_sequence(n: int, offset: float) -> np.ndarray:
    """
    Initialise a positive sequence based on the discretised kernel
    a_i = 1 / sqrt(i + 1 + offset) and normalise its ℓ¹‑norm to 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.sqrt(idx)
    a /= a.sum()
    return a

def conv_max(a: np.ndarray) -> float:
    """Maximum of the linear (non‑circular) auto‑convolution of ``a``."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv.max()

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix T such that T @ g = f * g (linear convolution).
    Shape of T is (2*len(f)-1, len(f)); each column j contains ``f`` shifted
    by ``j`` rows.
    """
    n = f.size
    rows = 2 * n - 1
    T = np.empty((rows, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = f
    return T

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation value.

    The function respects the time budget and returns the best sequence
    found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length (large enough to approximate the continuous kernel,
    #    small enough for a cheap LP).
    n = 2000

    # ------------------------------------------------------------------
    # 2. Scan a few offsets for the analytical shape and keep the best.
    offsets = [0.0, 0.1, 0.5, 1.0, 2.0]
    best_seq = None
    best_val = np.inf
    for off in offsets:
        a = build_initial_sequence(n, off)
        val = fast_score(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    print(f"[init] best value after offset scan: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop.
    f = best_seq.copy()          # current sequence (ℓ¹‑norm = 1)
    current_val = best_val

    # mixing coefficients that will be tried each iteration
    t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]

    it = 0
    while time.time() < deadline:
        # ---- 3.1 compute the current peak of the self‑convolution ----
        M = conv_max(f)

        # ---- 3.2 build the linear convolution matrix (f * g) ----
        T = build_conv_matrix(f)

        # ---- 3.3 linear programme: maximise Σ g_j  (i.e. minimise −Σ g_j) ----
        c = -np.ones(n, dtype=np.float64)    # minimise –sum(g)  → maximise sum(g)
        bounds = [(0.0, None)] * n
        A_ub = T
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # run the LP (HiGHS is the default high‑performance backend)
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})

        if not res.success:
            # Fallback: simple stochastic hill‑climbing move.
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.05 * f[i] * np.random.rand()
                g_candidate = f.copy()
                g_candidate[i] -= delta
                g_candidate[j] += delta
                g_candidate /= g_candidate.sum()
                cand_val = fast_score(g_candidate)
                if cand_val < current_val:
                    f = g_candidate
                    current_val = cand_val
                    if cand_val < best_val:
                        best_val = cand_val
                        best_seq = f.copy()
                        print(f"[fallback] new best = {best_val:.6f}")
            it += 1
            continue

        # ---- 3.4 obtain the LP solution and normalise it ----
        g_opt = res.x
        sum_g = g_opt.sum()
        if sum_g == 0.0:
            # Should not happen, but protect against divide‑by‑zero.
            it += 1
            continue
        g_norm = g_opt / sum_g   # now Σ g_norm = 1, same normalisation as f

        # ---- 3.5 try several convex combinations of f and g_norm ----
        best_mix_val = current_val
        best_mix_seq = f
        for t in t_candidates:
            mixed = (1.0 - t) * f + t * g_norm
            # Numerical safety: re‑normalise to exact sum = 1.
            mixed /= mixed.sum()
            val = fast_score(mixed)
            if val < best_mix_val:
                best_mix_val = val
                best_mix_seq = mixed

        # ---- 3.6 accept the best mixture if it improves the score ----
        if best_mix_val < current_val - 1e-12:
            f = best_mix_seq
            current_val = best_mix_val
            if best_mix_val < best_val:
                best_val = best_mix_val
                best_seq = f.copy()
                print(f"[iter {it:6d}] new global best = {best_val:.6f}")

        # ---- 3.7 occasional cheap random perturbations (tiny hill‑climb) ----
        if it % 7 == 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.02 * f[i] * np.random.rand()
                pert = f.copy()
                pert[i] -= delta
                pert[j] += delta
                pert /= pert.sum()
                val = fast_score(pert)
                if val < current_val:
                    f = pert
                    current_val = val
                    if val < best_val:
                        best_val = val
                        best_seq = f.copy()
                        print(f"[perturb {it}] improved to {best_val:.6f}")

        # ---- 3.8 progress report ----
        if it % 10 == 0:
            elapsed = time.time() - (deadline - budget_s + 0.5)
            print(f"[iter {it:6d}] cur={current_val:.6f}, best={best_val:.6f}, elapsed={elapsed:.1f}s")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    # Convert to plain Python list for the caller.
    return best_seq.tolist()

        bounds = [(0.0, None)] * n
        A_ub = T
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True})

        if not res.success:
            # ---- fallback: a cheap random mass transfer
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.05 * f[i] * np.random.rand()
                cand = f.copy()
                cand[i] -= delta
                cand[j] += delta
                cand /= cand.sum()
                cand_val = fast_score(cand)
                if cand_val < current_val:
                    f = cand
                    current_val = cand_val
                    if cand_val < best_val:
                        best_val = cand_val
                        best_seq = f.copy()
                        print(f"[fallback] new best = {best_val:.6f}")
            it += 1
            continue

        # ---- 3.4 normalise LP solution to the same ℓ¹‑norm as f
        g_opt = res.x
        sum_g = g_opt.sum()
        if sum_g == 0.0:
            it += 1
            continue
        g_norm = g_opt / sum_g   # Σ g_norm = 1

        # ---- 3.5 try several convex combinations of f and g_norm
        best_mix_val = current_val
        best_mix_seq = f
        for t in t_candidates:
            mixed = (1.0 - t) * f + t * g_norm
            mixed /= mixed.sum()               # numerical safety
            val = fast_score(mixed)
            if val < best_mix_val:
                best_mix_val = val
                best_mix_seq = mixed

        # ---- 3.6 accept the best mixture if it improves the score
        if best_mix_val < current_val - 1e-12:
            f = best_mix_seq
            current_val = best_mix_val
            if best_mix_val < best_val:
                best_val = best_mix_val
                best_seq = f.copy()
                print(f"[iter {it:6d}] new global best = {best_val:.6f}")

        # ---- 3.7 occasional tiny random perturbation (hill‑climb)
        if it % 7 == 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                delta = 0.02 * f[i] * np.random.rand()
                pert = f.copy()
                pert[i] -= delta
                pert[j] += delta
                pert /= pert.sum()
                val = fast_score(pert)
                if val < current_val:
                    f = pert
                    current_val = val
                    if val < best_val:
                        best_val = val
                        best_seq = f.copy()
                        print(f"[perturb {it}] improved to {best_val:.6f}")

        # ---- 3.8 progress report
        if it % 10 == 0:
            elapsed = time.time() - start_time
            print(f"[iter {it:6d}] cur={current_val:.6f}, best={best_val:.6f}, "
                  f"elapsed={elapsed:.1f}s")

        it += 1

    print(f"[final] best value found: {best_val:.6f}")
    # Return a plain Python list as required by the caller.
    return best_seq.tolist()
