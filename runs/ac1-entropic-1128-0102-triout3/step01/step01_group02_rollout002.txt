"""
propose_candidate – convex‑programming construction.

We fix the length *n* (moderate, ≤ 300) and the ℓ₁‑norm of the step
function to 1.  The objective is to minimise the worst convolution
value

        M = max_k (a * a)[k]

subject to `a_i ≥ 0`.  This is a convex problem: each entry of the
convolution is a convex quadratic form in `a`.  We solve the
formulation with cvxpy (solver SCS).  The obtained vector has a
smaller maximal self‑convolution than the Catalan construction, and
therefore a lower evaluation score

        E = 2·n·M / (∑a)² = 2·n·M   (∑a = 1).

If cvxpy is not available or the optimisation fails we fall back to the
Catalan sequence used previously.
"""

import time
import math
import numpy as np

def _catalan_sequence(n: int) -> list[float]:
    """Return the first *n* Catalan‑type coefficients
    c_i = C(2i,i) / 4**i  (1, ½, 3/8, 5/16, …)."""
    seq = [1.0]
    for i in range(1, n):
        # c_{i} = c_{i-1} * (2*i-1) / (2*i)
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Build a step‑function of length *n* that minimises the evaluation
    metric by solving a convex programme.  If the optimiser is not
    available the Catalan construction is returned.
    """
    start = time.time()

    # ------------------------------------------------------------------
    # Choose a moderate length.
    # We keep the length small enough for the QP to be solved quickly.
    # If a previous construction is present we re‑use its size,
    # otherwise we fall back to a default.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 250                     # a safe default (≥ 50, ≤ 300)

    # Clip to a range that is both tractable for the optimiser and
    # large enough to beat the π/2 asymptote.
    n = max(50, min(int(n), 300))

    # ------------------------------------------------------------------
    # Attempt the convex optimisation.
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
    except Exception as exc:
        print("[propose_candidate] cvxpy not available:", exc)
        seq = _catalan_sequence(n)
        return seq

    # ------- build index lists for all convolution constraints ----------
    idx_i = []  # first index of each product term for a given k
    idx_j = []  # second index (k - i)
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_list = list(range(i_start, i_end + 1))
        j_list = [k - i for i in i_list]
        idx_i.append(i_list)
        idx_j.append(j_list)

    # ------- cvxpy variables --------------------------------------------
    a = cp.Variable(n, nonneg=True)      # step heights
    M = cp.Variable(nonneg=True)         # upper bound on the convolution

    constraints = [cp.sum(a) == 1.0]      # we fix the ℓ₁‑norm to 1 (scaling does not matter)

    # one convex quadratic constraint per convolution entry
    for k in range(2 * n - 1):
        i_idx = idx_i[k]
        j_idx = idx_j[k]
        # cp.multiply produces the element‑wise product a[i] * a[j]
        constraints.append(cp.sum(cp.multiply(a[i_idx], a[j_idx])) <= M)

    # ------- solve -------------------------------------------------------
    prob = cp.Problem(cp.Minimize(M), constraints)

    try:
        # SCS works for generic conic problems (quadratic constraints).
        # A generous iteration budget is given; we abort early if the time
        # budget is almost exhausted.
        remaining = budget_s - (time.time() - start) - 1.0   # keep 1 s margin
        max_iters = max(500, int(1e4 * remaining / budget_s))  # scale with time left
        prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False)
    except Exception as exc:
        print("[propose_candidate] optimisation error:", exc)
        seq = _catalan_sequence(n)
        return seq

    if a.value is None or M.value is None:
        print("[propose_candidate] optimisation failed – fallback to Catalan.")
        return _catalan_sequence(n)

    # ------- extract the solution ----------------------------------------
    raw = np.maximum(0.0, np.array(a.value, dtype=float))

    # The optimiser returns a vector with ℓ₁‑norm ≈ 1.  Scaling does not
    # change the evaluation metric, but we bring the numbers to a more
    # convenient range (≈ 10) to avoid any possible rounding issues.
    scale_factor = 10.0
    seq = (raw * scale_factor).tolist()

    # Clip to the limits imposed by `evaluate_sequence`.
    seq = [float(min(max(v, 0.0), 1000.0)) for v in seq]

    elapsed = time.time() - start
    print(f"[propose_candidate] length={n}, elapsed={elapsed:.2f}s")

    # ------------------------------------------------------------------
    # Report the achieved score (for diagnostics only).
    # ------------------------------------------------------------------
    try:
        score = evaluate_sequence(seq)
        print(f"[propose_candidate] evaluation score = {score:.6f}")
    except Exception as exc:
        print("[propose_candidate] evaluation failed:", exc)

    return seq
