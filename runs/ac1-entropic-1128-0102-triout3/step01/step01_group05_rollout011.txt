def simplex_projection(v, z=1):
    # project onto simplex {x>=0, sum x = z}
    # reference to https://arxiv.org/abs/1309.1541
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, z=1.0):
    # Based on algorithm from Duchi et al. 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑evaluation step‑function.

The algorithm works in three stages:

1. **Coarse grid search** –  
   Search a family a_i ∝ (i+1+offset)^{-p} (power‑law kernels) for several
   exponents `p` and offsets.  The best candidate is kept as a start point.

2. **Projected gradient descent on a smoothed max‑convolution** –  
   The true objective 2 n·max(conv(a,a))/‖a‖₁² is non‑smooth because of the
   ``max``.  We replace the max by a softmax with temperature `β` and perform
   gradient steps on the resulting smooth objective.  After each step the
   vector is projected onto the simplex `{x≥0, Σx=1}` to keep the ℓ¹‑norm
   fixed.  FFTs are used for both the self‑convolution and the correlation
   needed for the gradient, giving O(n log n) per iteration.

3. **Fine‑grained hill climbing** –  
   Starting from the best vector found in stage 2 we run a classic stochastic
   mass‑transfer hill‑climber (move a small amount of mass from one index to
   another) that only accepts improving moves (with a tiny probability of
   accepting a worse move to escape plateaus).  This phase exploits the full
   speed of the FFT‑based `fast_score` and typically performs many millions
   of iterations within the time budget.

The whole search respects the 1000 s budget, prints periodic progress
messages and finally returns the best sequence found as a plain Python
list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to ``evaluate_sequence`` without the safety
# checks).  Used internally; the final answer will be verified by the
# provided ``evaluate_sequence``.
def fast_score(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))/sum(a)²  –  n = len(a) (FFT based)."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(a, size)
    conv = np.fft.irfft(ft * ft, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of `v` onto the simplex {x≥0, Σx = z}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if z <= 0:
        raise ValueError("Target sum z must be positive")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be projected to zero
        return np.zeros_like(v)
    rho = ind[cond][-1]               # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def build_power_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """
    a_i = (i+1+offset)^{-p},  normalized to unit ℓ¹‑norm.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation value under the given `evaluate_sequence` function.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a sequence length.
    #    3000–4000 provides a good trade‑off between quality and speed.
    n = 3500

    # ------------------------------------------------------------------
    # 2. Coarse search over power‑law kernels.
    best_seq = None
    best_val = np.inf

    p_grid = np.linspace(0.30, 0.90, 13)        # 0.30 … 0.90  (step 0.05)
    offset_grid = [0.0, 0.2, 0.5, 1.0, 2.0]    # a few offsets near zero

    print("[search] scanning power‑law family …")
    for p in p_grid:
        for off in offset_grid:
            seq = build_power_sequence(n, p, off)
            val = fast_score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq
    print(f"[search] best after grid scan: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected gradient descent on a smooth approximation of the max.
    a = best_seq.copy()
    best_seq = a.copy()
    best_val = fast_score(a)

    # FFT size large enough for both the self‑convolution (length 2n‑1)
    # and the correlation (length 3n‑2).
    size_fft = 1 << ((3 * n - 2).bit_length())

    beta = 1.0            # start temperature for softmax
    beta_max = 300.0
    beta_mul = 1.015

    lr = 0.07             # initial learning rate
    lr_decay = 0.9995

    it = 0
    last_improve = 0

    print("[gd] starting projected gradient descent …")
    while time.time() < deadline:
        # ----- self‑convolution (used for both the objective and the softmax)
        ft_a = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(ft_a * ft_a, size_fft)[: 2 * n - 1]
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = it
            print(f"[gd] iter {it:6d}  new best {best_val:.6f}")

        # ----- smooth max (softmax) weights
        #   w_i ∝ exp(β·(conv_i - max_conv))  →   no overflow because ≤ 1
        w = np.exp(beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth objective:
        #   grad_j = 2 * Σ_i w_{i+j} * a_i   (linear correlation a ⋆ w)
        ft_w = np.fft.rfft(w, size_fft)
        # correlation a⋆w  = irfft( ft_a * conj(ft_w) )
        corr_full = np.fft.irfft(ft_a * np.conj(ft_w), size_fft)[: 3 * n - 2]

        # zero‑lag index in the full correlation = len(w) – 1 = 2n‑2
        start = 2 * n - 2
        grad = 2.0 * corr_full[start:start + n]

        # ----- gradient step + projection onto the simplex
        a_new = a - lr * grad
        a = simplex_projection(a_new, 1.0)

        # ----- schedule updates
        if it % 500 == 0 and it > 0:
            beta = min(beta * beta_mul, beta_max)
            lr = max(lr * lr_decay, 1e-6)

        if it - last_improve > 30000:          # stagnation → stop GD early
            break

        if it % 2000 == 0:
            elapsed = time.time() - (deadline - budget_s + 0.5)
            print(f"[gd] iter {it:6d}  cur {cur_val:.6f}  best {best_val:.6f}  "
                  f"β={beta:.1f}  lr={lr:.5e}")

        it += 1

    # ------------------------------------------------------------------
    # 4. Final hill‑climbing refinement (mass‑transfer moves).
    cur = best_seq.copy()
    cur_val = best_val
    step = 0.03                # relative amount of mass moved each trial

    hill_it = 0
    print("[hill] starting local hill‑climbing …")
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        delta = step * cur[i] * np.random.rand()
        delta = min(delta, cur[i])

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand /= cand.sum()               # keep ℓ¹‑norm = 1

        cand_val = fast_score(cand)

        if cand_val < cur_val:
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best_val, best_seq = cand_val, cand.copy()
                print(f"[hill] iter {hill_it:6d}  new best {best_val:.6f}")
        else:
            # tiny uphill acceptance to escape tiny plateaus
            if np.random.rand() < 0.001:
                cur, cur_val = cand, cand_val

        if hill_it % 5000 == 0:
            print(f"[hill] iter {hill_it:6d}  cur {cur_val:.6f}  best {best_val:.6f}")

        hill_it += 1

    print("[final] best evaluation found:", best_val)
    # Convert to a plain Python list as required by the caller.
    return best_seq.tolist()
