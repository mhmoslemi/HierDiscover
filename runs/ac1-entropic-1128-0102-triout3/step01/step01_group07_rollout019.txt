"""
Propose a step‑function sequence that minimizes

        2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Wide‐range parametric search** –  Several sequence lengths are tried.
   For each length we generate simple families (pure power‑law,
   mirrored power‑law, mirrored exponential) on a coarse grid of
   parameters.  The best candidate from this cheap sweep is kept.

2. **Smoothed sub‑gradient descent** –  Starting from the best candidate,
   we minimise the empirical maximum convolution entry.
   Instead of using only the single active index (as in the reference
   code) we use a soft‑max over the top‑K convolution entries, compute
   the corresponding sub‑gradient exactly and perform a projected gradient
   step onto the probability simplex.  No monotone‑sorting is applied,
   allowing non‑monotone shapes (U‑shaped distributions) that can achieve
   a lower ‑∞‑norm of the auto‑convolution.

3. **Random jitter & polishing** –  Periodic small random perturbations
   (followed by projection) help escape shallow plateaus.  The
   best‑ever sequence seen is returned.

All heavy work is done with NumPy; the total running time is kept well
below the 1000 s budget.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1          # zero‑based
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(seq: np.ndarray) -> float:
    """Score = 2 * n * max(conv(seq, seq)), assuming sum(seq)=1."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all produce non‑negative vectors that sum to 1)
# ----------------------------------------------------------------------


def _powerlaw_seq(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-beta}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta)
    seq /= seq.sum()
    return seq


def _double_tail_powerlaw_seq(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-beta} + (n-i)^{-beta}, i = 0 … n‑1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta) + (n + 1 - idx) ** (-beta)
    seq /= seq.sum()
    return seq


def _double_tail_exp_seq(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Smoothed sub‑gradient based optimisation
# ----------------------------------------------------------------------


def _softmax_weights(values: np.ndarray,
                     temperature: float,
                     top_k: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (indices, normalised weights) for a soft‑max over the
    ``top_k`` largest entries of ``values``.
    """
    if top_k >= values.size:
        idx = np.arange(values.size)
    else:
        # ``np.argpartition`` is O(n) – get the top‑k indices
        idx = np.argpartition(values, -top_k)[-top_k:]
    vk = values[idx]
    # soft‑max (shifted by the maximum for numerical stability)
    shift = vk.max()
    exp_vals = np.exp((vk - shift) / temperature)
    w = exp_vals / exp_vals.sum()
    return idx, w


def _gradient_from_top(seq: np.ndarray,
                       conv: np.ndarray,
                       top_k: int,
                       temperature: float) -> np.ndarray:
    """
    Compute a sub‑gradient of the smoothed max of the convolution.
    Only the ``top_k`` largest convolution entries participate,
    weighted by a soft‑max with ``temperature``.
    """
    n = seq.shape[0]
    idxs, w = _softmax_weights(conv, temperature, top_k)

    grad = np.zeros_like(seq)

    for k, wk in zip(idxs, w):
        # indices i that participate in conv_k
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i_range = np.arange(start, end + 1)
        # ∂conv_k/∂a_i = a_i + a_{k‑i}
        grad[i_range] += wk * (seq[i_range] + seq[k - i_range])
    return grad


def _refine_sequence(seq: np.ndarray,
                     max_iters: int,
                     deadline: float,
                     top_k: int = 8,
                     init_temp: float = 0.3,
                     init_eta: float = None) -> np.ndarray:
    """
    Projected (smoothed) sub‑gradient descent.
    Returns the best sequence seen during the run.
    """
    n = seq.shape[0]
    if init_eta is None:
        init_eta = 0.5 / n

    best_seq = seq.copy()
    best_score = _compute_score(best_seq)

    eta = init_eta
    temp = init_temp

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = np.convolve(seq, seq)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[refine] New global best {best_score:.6f} (n={n}, it={it})")

        # gradient step
        grad = _gradient_from_top(seq, conv, top_k, temp)
        seq = seq - eta * grad
        seq = _project_simplex(seq)

        # occasional jitter to break symmetry / escape plateaus
        if (it + 1) % 500 == 0:
            jitter = (np.random.rand(n) - 0.5) * eta * 0.2
            seq = _project_simplex(np.maximum(seq + jitter, 0.0))

        # schedule
        eta *= 0.9995
        temp = max(temp * 0.9995, 1e-4)

        it += 1

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Returns a list of non‑negative heights (the step‑function) that
    attempts to minimise the provided ``evaluate_sequence`` function.
    The routine is deterministic for a given ``seed``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # Initialise best known solution (if any)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        best_seq = np.array(height_sequence_1, dtype=np.float64)
        best_score = _compute_score(best_seq / best_seq.sum())
        # normalise to ℓ₁‑norm = 1 for the internal score computation
        best_seq = best_seq / best_seq.sum()
        print(f"[init] Existing sequence score {best_score:.6f}")
    else:
        best_seq = None
        best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Coarse grid over several lengths and families
    # ------------------------------------------------------------------
    lengths = [200, 400, 800, 1600, 2800]          # affordable sizes
    betas = np.linspace(0.0, 1.4, 29)              # step 0.05
    alphas = np.linspace(0.001, 0.02, 20)          # exponential rates

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- power‑law (single tail) -----
        for beta in betas:
            seq = _powerlaw_seq(n, beta)
            score = _compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[grid] New best {best_score:.6f} (n={n}, PL beta={beta:.3f})")

        # ----- mirrored power‑law (double tail) -----
        for beta in betas:
            seq = _double_tail_powerlaw_seq(n, beta)
            score = _compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[grid] New best {best_score:.6f} (n={n}, DPL beta={beta:.3f})")

        # ----- mirrored exponential -----
        for alpha in alphas:
            seq = _double_tail_exp_seq(n, alpha)
            score = _compute_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[grid] New best {best_score:.6f} (n={n}, Exp α={alpha:.5f})")

    # ------------------------------------------------------------------
    # 2) Refine the current best candidate with smoothed sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (always valid)
        n = 500
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score(best_seq)

    # We may wish to try a second refinement with a slightly larger length
    # (copy‑pad with zeros) – this sometimes gives a small extra gain.
    # Here we keep the length fixed for simplicity.
    refined = _refine_sequence(best_seq,
                               max_iters=20000,
                               deadline=deadline,
                               top_k=8,
                               init_temp=0.3)

    # Update global best if refinement succeeded
    refined_score = _compute_score(refined)
    if refined_score < best_score:
        best_score = refined_score
        best_seq = refined
        print(f"[final‑refine] Improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final sanity check via the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final reported score by evaluate_sequence: {final_score:.6f}")
    except Exception as e:
        # Should never happen – we fall back to the internal score
        print(f"Warning: evaluate_sequence raised {e}; using internal score.")
        final_score = best_score

    print(f"Returning sequence of length {len(final_seq)} with score {final_score:.6f}")
    return final_seq
