    2·n·max(conv(a,a)) / (sum(a)^2) .
The smoothing is a soft‑max (log‑sum‑exp) of the convolution entries,
controlled by a temperature τ that is gradually reduced.
Gradient of the smoothed objective is obtained analytically via a
single convolution, and the parameters are updated with the Adam optimizer.
After each update the vector is projected onto the probability simplex
(non‑negative, ℓ₁‑norm = 1).  Random jitter and occasional sub‑gradient
steps are added to escape shallow plateaus.

The routine tries several sequence lengths, several random restarts,
keeps the globally best sequence (according to the exact objective) and
returns it before the time budget expires.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution via FFT (real arrays)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real 1‑D arrays using FFT (real‑to‑real)."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << (n - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    fc = fa * fb
    c = np.fft.irfft(fc, size)[:n]
    return c


# ----------------------------------------------------------------------
# Objective helpers (exact, fast)
# ----------------------------------------------------------------------


def _exact_score(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a sequence with ℓ₁‑norm = 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i = exp(conv_i / τ) / Σ exp(conv_j / τ)
    in a numerically stable way.
    """
    # τ > 0
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / tau)
    w = exp_vals / exp_vals.sum()
    return w


def _softmax_gradient(seq: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of  τ·log Σ_i exp(conv_i/τ)  with respect to seq.
    Implements   g = 2 * (w *conv seq_rev)_{n-1 : n-1+n}
    where w are soft‑max weights.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)                     # length 2n‑1
    w = _softmax_weights(conv, tau)                # same length
    rev = seq[::-1]                                # reversed seq
    h = _conv_fft(w, rev)                          # length 3n‑2
    grad = 2.0 * h[n - 1 : n - 1 + n]               # slice n entries
    return grad


# ----------------------------------------------------------------------
# Sub‑gradient step (for the true max) – used as a final fine‑tuner
# ----------------------------------------------------------------------


def _subgradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """Projected sub‑gradient step for the exact max‑convolution term."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    k = int(np.argmax(conv))               # active index
    grad = np.zeros_like(seq)
    # derivative of conv_k w.r.t. seq_i is 2·seq_{k‑i} (zero if out‑of‑range)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1, dtype=int)
    grad[idx] = 2.0 * seq[k - idx]
    new_seq = seq - eta * grad
    return _project_simplex(new_seq)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights (a sequence) that attempts to
    minimise the provided `evaluate_sequence` function.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Keep the best sequence discovered so far
    best_seq = None
    best_score = float("inf")

    # If a previous construction is available, use it as a warm start
    if "height_sequence_1" in globals():
        cand = list(height_sequence_1)
        try:
            sc = evaluate_sequence(cand)
            if sc < best_score:
                best_score = sc
                best_seq = np.array(cand, dtype=float) / np.sum(cand)  # normalise to ℓ₁=1
                print(f"[init] Using previous best score {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter choices
    # ------------------------------------------------------------------
    lengths = [1800, 2400, 3000, 3600, 4200]          # a few increasing sizes
    restarts_per_len = 3
    max_iters = 2500            # per restart (may be cut early by the deadline)

    # Adam hyper‑parameters (shared across all runs)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08              # learning rate before decay
    tau_start = 0.4
    tau_min = 0.003
    tau_decay = 0.9985          # per iteration

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        for restart in range(restarts_per_len):
            if time.time() >= deadline:
                break

            # ----- initialise a random point on the simplex -----
            seq = np.random.rand(n)
            seq = _project_simplex(seq)   # ensures sum == 1
            # Adam state
            m = np.zeros_like(seq)
            v = np.zeros_like(seq)

            lr = base_lr
            tau = tau_start

            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                # ---- gradient of the smoothed objective ----
                grad = _softmax_gradient(seq, tau)

                # ---- Adam update ----
                t = it + 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                seq = seq - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---- project back onto the simplex ----
                seq = _project_simplex(seq)

                # ---- schedule updates ----
                lr *= 0.9997
                tau = max(tau * tau_decay, tau_min)

                # ---- occasional evaluation of the true objective ----
                if (it % 120) == 0:
                    cur_score = _exact_score(seq)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = seq.copy()
                        print(
                            f"[sgd] n={n} restart={restart} it={it} "
                            f"score={cur_score:.6f}"
                        )
                    # random jitter to shake out of plateaus
                    if np.random.rand() < 0.04:
                        jitter = (np.random.rand(n) - 0.5) * lr * 5.0
                        seq = _project_simplex(np.maximum(seq + jitter, 0.0))

                # ---- after half of the iterations, start a few exact
                # ---- sub‑gradient refinements to lower the true max ----
                if it == max_iters // 2:
                    # perform a short burst of exact sub‑gradient steps
                    eta = 0.12 / n
                    for sub_it in range(300):
                        if time.time() >= deadline:
                            break
                        seq = _subgradient_step(seq, eta)
                        # decay step size
                        eta *= 0.995
                        if (sub_it % 60) == 0:
                            cur_score = _exact_score(seq)
                            if cur_score < best_score:
                                best_score = cur_score
                                best_seq = seq.copy()
                                print(
                                    f"[sub] n={n} restart={restart} sub_it={sub_it} "
                                    f"score={cur_score:.6f}"
                                )
                    # resume main loop after sub‑gradient burst
            # end of iterations for this restart

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should not happen) return a uniform seq
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = _exact_score(best_seq)

    # Convert to plain Python list for the evaluator
    final_seq = best_seq.tolist()
    # Final sanity check using the provided evaluator (optional but cheap)
    try:
        final_eval = evaluate_sequence(final_seq)
        if final_eval < best_score:
            best_score = final_eval
            print(f"[final] verify score {final_eval:.6f}")
    except Exception:
        pass

    print(f"Final best score (exact) : {best_score:.6f}  length={len(final_seq)}")
    return final_seq
