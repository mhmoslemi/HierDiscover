def softmax_weights(conv, beta):
    maxc = conv.max()
    exps = np.exp(beta * (conv - maxc))
    sum_exps = exps.sum()
    return exps / sum_exps
```

def project_simplex(v, z=1.0):
    # https://arxiv.org/pdf/1101.6081.pdf
    # v: np.array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

                    best_seq = seq.copy()

                break

        # hill climbing refinement - give limited time

def _grad_smooth_max(a, beta):
    conv = np.convolve(a, a)
    # compute softmax
    max_c = conv.max()
    # shift for numerical stability
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()
    n = len(a)
    grad = np.empty_like(a)
    # compute gradient using sliding windows
    # w length = 2n-1
    for j in range(n):
        # w segment from j to j+n
        grad[j] = 2.0 * np.dot(a, w[j:j+n])
    return grad, conv.max()
```

def _smooth_max_descent(seq, deadline, max_iter=2000):
    a = seq.copy()
    n = len(a)
    # enforce normalization
    a = a * (_target_sum(n) / a.sum())
    best_a = a.copy()
    best_score = _evaluate(a)
    beta = 5.0
    lr = a.mean()*0.1
    for it in range(max_iter):
        if time.time() >= deadline:

        grad, cur_max = _grad_smooth_max(a, beta)
        # step
        a = a - lr * grad
        # enforce nonnegativity via projection
        a = _project_simplex(a, z=_target_sum(n))

        score = _evaluate(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # adapt: slowly increase beta and decrease lr
        beta = min(100.0, beta * 1.02)
        lr *= 0.995

"""
Improved search for a low‑score step function.

The evaluation is scale‑invariant, so we keep the L¹‑norm fixed to
√(2·n) (the same normalisation used by the reference code).  The search
consists of three stages:

1. **Parametric sweep** – generate analytically‑derived shapes
   a_i = (i+1+δ)^s – (i+δ)^s  (0.5 ≤ s ≤ 0.9, 0 ≤ δ ≤ 1) for several
   lengths n and keep the best candidate.

2. **Hill‑climbing refinement** – from the best candidate of the sweep we
   repeatedly transfer a small amount of mass between two random
   positions (preserving the L¹‑norm) and keep the move if the true
   evaluation improves.  This is a cheap stochastic optimiser that
   works well on the discrete problem.

3. **Smooth‑max gradient descent** – minimise a differentiable
   approximation of the max‑convolution
   f(a)= (1/β)·log Σ_k exp(β·(a∗a)_k).  The gradient can be written as a
   convolution, so each step costs only one extra convolution.  The
   step size and the β‑parameter are slowly adapted; after each step
   we project back onto the simplex (non‑negative with the required
   L¹‑norm).

The whole routine respects the 1000 s time budget, returns the best
sequence found and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Desired L¹‑norm for a sequence of length n (the normalisation used by the reference)."""
    return np.sqrt(2.0 * n)


def _analytic_power_sequence(n: int, s: float, delta: float = 0.0) -> np.ndarray:
    """
    Analytic family a_i = (i+1+δ)^s – (i+δ)^s .
    The sequence is scaled so that its L¹‑norm equals sqrt(2·n).
    """
    i = np.arange(n, dtype=float)
    base = np.power(i + 1.0 + delta, s) - np.power(i + delta, s)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _evaluate(seq: np.ndarray) -> float:
    """Wrap the provided evaluate_sequence (expects a Python list)."""
    return evaluate_sequence(seq.tolist())


def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = z}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform point on the simplex
        return np.full_like(v, z / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w


def _hill_climb(seq: np.ndarray,
                max_iter: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that transfers a small amount of mass
    between two randomly chosen positions.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    # initial transfer magnitude (a fraction of the mean entry)
    step_sigma = max(1e-9, seq.mean() * 0.1)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue

        # propose a signed transfer from i to j
        delta = np.random.randn() * step_sigma
        # keep the transfer feasible (no negative entries)
        delta = np.clip(delta, -seq[j], seq[i])

        old_i, old_j = seq[i], seq[j]
        seq[i] = old_i - delta
        seq[j] = old_j + delta

        # Renormalise to the fixed L¹ norm (tiny numerical drift may appear)
        seq *= _target_sum(n) / seq.sum()

        cand_score = _evaluate(seq)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = seq.copy()
        else:
            # reject – roll back
            seq[i] = old_i
            seq[j] = old_j
            seq *= _target_sum(n) / seq.sum()

        # decay the proposal scale slowly
        step_sigma *= 0.9995

    return best_seq, best_score


def _grad_smooth_max(a: np.ndarray, beta: float):
    """
    Gradient of the smooth‑max objective
        f(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)
    using the formula
        ∇f_j = 2 Σ_k a_k·w_{j+k},
    where w is the soft‑max over the convolution values.
    Returns the gradient and the current max‑convolution (for logging).
    """
    n = len(a)
    conv = np.convolve(a, a)                     # length 2n‑1
    # soft‑max weights, numerically stabilised
    max_c = conv.max()
    exp_vals = np.exp(beta * (conv - max_c))
    w = exp_vals / exp_vals.sum()                # length 2n‑1, Σ w = 1

    # gradient via a single convolution (correlation)
    w_rev = w[::-1]
    corr = np.convolve(a, w_rev)                 # length 3n‑2
    # grad_j = 2 * corr[ (2n‑2) - j ]
    idx_start = 2 * n - 2
    grad = 2.0 * corr[idx_start - np.arange(n)]

    return grad, conv.max()


def _smooth_max_descent(seq: np.ndarray,
                        deadline: float,
                        max_iter: int = 800) -> np.ndarray:
    """
    Gradient descent on the smooth‑max surrogate.
    The iterate stays on the simplex with the required L¹‑norm.
    """
    n = len(seq)
    a = seq.copy()
    a = a * (_target_sum(n) / a.sum())          # ensure exact normalisation

    best_a = a.copy()
    best_score = _evaluate(best_a)

    beta = 5.0                      # start with a modest smoothness
    step = max(1e-6, a.mean() * 0.08)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        grad, cur_max = _grad_smooth_max(a, beta)

        # tentative step
        a_new = a - step * grad
        a_new = _project_simplex(a_new, z=_target_sum(n))

        cand_score = _evaluate(a_new)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_new.copy()
            a = a_new
            step *= 1.02                     # increase step after success
        else:
            step *= 0.90                     # shrink step after failure

        # slowly tighten the smooth approximation
        beta = min(80.0, beta * 1.015)

    return best_a
