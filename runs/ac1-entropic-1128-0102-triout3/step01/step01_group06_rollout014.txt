"""
propose_candidate
-----------------

A targeted hill‑climbing optimiser for the “max‑self‑convolution / (sum²)”
objective used by ``evaluate_sequence``.  The idea is simple:

1.  Start from a known good sequence (the global ``height_sequence_1`` if
    present, otherwise a power‑law seed).  Normalise it to sum = 1.
2.  Repeatedly evaluate the full self‑convolution via a fast FFT.
    The index (or indices) where the convolution attains its maximum
    are the “bottlenecks’’ of the current solution.
3.  For those bottleneck positions compute, for each coefficient *a[i]*,
    its total contribution   a[i]·a[k‑i]  to all nearly‑maximal peaks.
    The coefficient with the largest contribution is a good *donor*,
    the one with the smallest contribution a good *receiver*.
4.  Move a modest amount of mass from the donor to the receiver,
    recompute the convolution and keep the move only if the maximal
    coefficient decreased.
5.  Occasionally (≈0.5 % of the iterations) restart from a fresh
    power‑law family and also perform random small perturbations to
    avoid being trapped in a local minimum.
6.  The loop runs until the allotted wall‑time is exhausted (default
    1000 s) and the best sequence seen so far is returned as a plain
    Python list.

The algorithm respects all constraints of ``evaluate_sequence``:
all entries stay non‑negative, never exceed 1000, and the sum never
drops below 0.01 (normalisation guarantees a sum of 1).  By focusing
the search on the actual contributors to the current worst convolution
coefficients it typically improves the bound much faster than a blind
random walk.
"""

import time
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper: fast maximal self‑convolution using a real FFT
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """Return (max_value, full_convolution) of a*a using an FFT of length ``fft_len``."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the current
    best bound 1.577182.  The function runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    finish_time = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise a feasible sequence (prefer the previously found one)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)   # type: ignore
        n = len(init_seq)
        if n == 0:
            raise ValueError
    except Exception:
        # fallback: a simple decreasing power‑law of length 2000
        n = 2000
        i = np.arange(n, dtype=np.float64)
        init_seq = (i + 1.0) ** (-0.5)

    # Clip/normalise – the evaluation routine itself also clips, but we stay
    # inside the safe region.
    init_seq = np.maximum(init_seq, 0.0)
    total = init_seq.sum()
    if total < 1e-12:                         # should never happen
        init_seq = np.ones(n, dtype=np.float64) / n
        total = 1.0
    a = init_seq / total                       # now sum(a) == 1

    # Pre‑computed FFT length for the whole run (does not change)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Keep track of the best solution seen so far
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_max, _ = _max_convolution(best_a, fft_len)
    best_score = 2.0 * n * best_max
    print(f"[init] length={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < finish_time:
        cur_max, conv = _max_convolution(a, fft_len)

        # record an improvement (very unlikely to be exactly equal)
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_score = 2.0 * n * cur_max
            print(f"[iter {iteration:05d}] new best score={best_score:.6f}")

        # --------------------------------------------------------------
        # 3a – identify the (near‑)maximal convolution indices
        # --------------------------------------------------------------
        # a tiny tolerance makes the method robust to flat tops
        tolerance = 1e-3
        thresh = cur_max * (1.0 - tolerance)
        peak_idxs = np.where(conv >= thresh)[0]

        # --------------------------------------------------------------
        # 3b – compute per‑coordinate contribution to those peaks
        # --------------------------------------------------------------
        # contribution[i] = sum_{k in peaks} a[i] * a[k-i]   (zero if out‑of‑range)
        contrib = np.zeros(n, dtype=np.float64)
        for k in peak_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contrib[lo : hi + 1] += a[idx] * a[k - idx]

        # --------------------------------------------------------------
        # 3c – pick donor (largest contribution) and receiver (smallest)
        # --------------------------------------------------------------
        src = int(np.argmax(contrib))
        dst = int(np.argmin(contrib))
        if src == dst:
            # fallback to a random pair if the contribution are equal
            src, dst = np.random.choice(n, size=2, replace=False)

        # --------------------------------------------------------------
        # 3d – propose a modest mass transfer from src -> dst
        # --------------------------------------------------------------
        # Transfer at most 10 % of the donor (or 5 % of the whole mass)
        max_transfer = min(0.10 * a[src], 0.05)
        if max_transfer <= 0.0:
            # nothing to move – try a tiny random move instead
            i1, i2 = np.random.choice(n, size=2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.02
            delta = np.clip(delta, -a[i1] + 1e-12, a[i2] - 1e-12)
            candidate = a.copy()
            candidate[i1] += delta
            candidate[i2] -= delta
        else:
            delta = np.random.uniform(0.0, max_transfer)
            candidate = a.copy()
            candidate[src] -= delta
            candidate[dst] += delta

        # --------------------------------------------------------------
        # 3e – accept the move only if the maximal convolution improves
        # --------------------------------------------------------------
        cand_max, _ = _max_convolution(candidate, fft_len)
        if cand_max < cur_max:
            a = candidate
            # (the new best will be recorded at the top of the loop)
        else:
            # Occasionally allow a neutral or slightly worse move to escape
            # a local trap – small simulated‑annealing flavour.
            if np.random.rand() < 0.001:
                a = candidate

        # --------------------------------------------------------------
        # 3f – occasional random restart from the parametric family
        # --------------------------------------------------------------
        if np.random.rand() < 0.005:
            alpha = np.random.uniform(0.35, 0.80)
            c = np.random.uniform(1.0, n / 2.0)
            i = np.arange(n, dtype=np.float64)
            restart = (i + c) ** (-alpha)
            restart /= restart.sum()
            a = restart
            # (no immediate improvement check – will be caught in the next loop)

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence witnessed during the run
    # ------------------------------------------------------------------
    # Convert to plain Python floats (the evaluation harness expects a list)
    return best_a.tolist()
