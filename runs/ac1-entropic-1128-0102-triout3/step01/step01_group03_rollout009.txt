    best_score = evaluate_sequence(a.tolist())

    deadline = start + budget_s - 5
    iteration = 0
    while time.time() < deadline:
        # compute conv via FFT (fast)
        conv = conv_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[iter {iteration}] new best score {best_score:.6f}")

        # Build convolution matrix A based on a
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = a

        # Solve LP
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception as e:
            print("LP solve failed:", e)
            break
        b_opt = b_var.value
        if b_opt is None:
            print("LP solution null")
            break

        sum_b = b_opt.sum()
        sum_a = a.sum()
        # if not improvement of sum, maybe break
        if sum_b <= sum_a * (1 + 1e-6):
            # No increase in feasible sum, stop
            break

        # scaling to same sum (1)
        g = b_opt / sum_b
        # mixture: choose t
        t = 0.3
        a = (1 - t) * a + t * g
        # optional random perturbation

            # add small uniform noise, keep sum=1 via projection onto simplex
            noise = (rng.random(n) - 0.5) * 1e-3
            a = project_onto_simplex(a + noise)

        iteration += 1
    # final best
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best reported score {final_score:.6f}")
    return best_a.tolist()

def solve_lp(f, M):
    n = len(f)

    A = np.zeros((2*n-1, n))

"""
Hybrid optimizer for the self–convolution bound.

The routine starts from the classic half‑binomial (sqrt) distribution,
normalises it to a probability simplex and then iteratively improves the
sequence by two complementary mechanisms:

1. **Smoothed gradient descent** – a smooth approximation of the
   max‑norm,
   J(a)= (1/p)·log Σ_k ( (a∗a)_k / max(a∗a) )^p,
   is minimised.  The gradient can be evaluated in O(n log n) time by FFT,
   and the iterate is projected back onto the simplex.

2. **Linear‑programming mixing step** – given the current feasible vector
   `a` with bound M = max(a∗a), we solve the LP
        maximise   Σ_j b_j
        subject to b ≥ 0,
                   a ∗ b ≤ M .
   The solution `b` has a larger ℓ₁‑norm while respecting the same peak
   convolution value.  Mixing a convex combination
        a ← (1−t)·a + t·b̂   (b̂ = b / Σ b)
   strictly reduces the maximal self‑convolution after a few
   subsequent gradient steps.

The loop runs until the allotted time expires, keeping the best‑ever
sequence (according to ``evaluate_sequence``).  The final sequence is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be defined at top level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of a_i = C(2i,i)/4**i  (coefficients of (1‑z)^-1/2).
    Returns a vector normalised to have sum 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        { x | x_i ≥ 0, Σ_i x_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Self‑convolution x∗x using a real FFT.
    Length of the result is 2·len(x)−1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def conv_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolution of two (possibly different‑length) real vectors using FFT.
    """
    m = x.shape[0] + y.shape[0] - 1
    X = np.fft.rfft(x, n=m)
    Y = np.fft.rfft(y, n=m)
    return np.fft.irfft(X * Y, n=m)


def smooth_gradient_step(a: np.ndarray, p: float, eta: float) -> np.ndarray:
    """
    One step of projected gradient descent on the smoothed max objective.
    p – smoothing exponent (larger → tighter approximation to the true max).
    eta – step‑size multiplier.
    """
    # self‑convolution and its maximum
    conv = conv_fft(a)
    M = conv.max()
    if M <= 0:
        return a.copy()

    # weights approximating the sub‑gradient of the max
    w = (conv / M) ** p               # shape (2n‑1,)
    W_sum = w.sum()

    # convolution a * w (needed for Σ_k w_k a_{k‑i})
    conv_aw = conv_general(a, w)      # length 3n‑2
    n = a.shape[0]
    # slice that corresponds to Σ_j a_j w_{i+j}
    slice_mid = conv_aw[n - 1: n - 1 + n]

    # gradient of the smooth objective (see analysis)
    grad = a * W_sum + slice_mid

    # move opposite to the gradient and project back onto the simplex
    a_new = a - eta * grad
    a_new = project_onto_simplex(a_new)
    return a_new


def solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program
        maximise   Σ b_j
        subject to b ≥ 0,
                   f ∗ b ≤ M .
    Returns the optimal vector b (not normalised) or None if the solver fails.
    """
    import cvxpy as cp

    n = f.shape[0]
    # Build the Toeplitz matrix that realises the convolution f ∗ b.
    # Row k (0‑based) contains f_{k‑j} for those j where the index is valid.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:                     # pragma: no cover
        return None

    if b.value is None:
        return None
    # Numerical noise can produce tiny negatives; clip them.
    return np.maximum(b.value, 0.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises the evaluation function.
    Uses a hybrid of smoothed gradient descent and an LP‑based mixing step.
    Returns the best sequence encountered (as a plain Python list).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Configuration (feel free to tweak)
    n = 1400                     # length of the sequence – moderate size
    max_iter = 10_000_000        # safety cap
    eta = 0.08                   # initial gradient step multiplier
    p = 12.0                     # smoothing exponent for the max‑approximation
    t_mix = 0.30                 # mixing weight for the LP step
    lp_every = 2000             # try an LP after this many gradient steps
    lp_no_improve = 5000        # also trigger LP if no improvement for this long

    # ------------------------------------------------------------------
    # Initialise from the half‑binomial series (the classic construction)
    a = sqrt_series(n)           # already sums to 1

    # keep the best ever seen
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin
    it = 0
    last_improve_it = 0

    while time.time() < deadline and it < max_iter:
        # --------------------------------------------------------------
        # 1. Gradient‑descent step (smoothed max)
        a = smooth_gradient_step(a, p, eta)

        # 2. Evaluate current candidate
        conv = conv_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M             # because Σa = 1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_it = it
            print(f"[iter {it:07d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3. Occasionally run the LP‑mixing step
        if (it - last_improve_it) >= lp_no_improve or (it % lp_every) == 0:
            b_opt = solve_lp(a, M)
            if b_opt is not None:
                sum_b = b_opt.sum()
                if sum_b > 1e-12:                     # a non‑trivial solution
                    # Scale b to a probability vector (sum = 1)
                    g = b_opt / sum_b
                    # Convex combination – the factor t_mix can be reduced later
                    a = (1.0 - t_mix) * a + t_mix * g
                    # Optional tiny random jitter to stay off the exact LP vertex
                    if rng.random() < 0.05:
                        jitter = (rng.random(n) - 0.5) * 1e-5
                        a = project_onto_simplex(a + jitter)

        # --------------------------------------------------------------
        # 4. Adaptive parameter decay
        if (it + 1) % 3000 == 0:
            eta = max(eta * 0.93, 1e-5)
            p = max(p - 1.0, 2.0)          # lower p slowly to sharpen the max

        it += 1

    # ------------------------------------------------------------------
    # Final evaluation of the best sequence found
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")
    return best_a.tolist()
