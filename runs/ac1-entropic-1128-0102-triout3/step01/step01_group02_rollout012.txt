"""
propose_candidate – construct a power‐law step function.

The functional to minimise is  

    E(a) = 2·n·max(a∗a) / (∑a_i)²

for a non‑negative vector a of length n.  For a sequence of the form  

    a_i = C·(i+1)^p ,      i = 0,…,n‑1

with the scaling constant C chosen so that the largest entry equals the
allowed ceiling 1000, the evaluation simplifies to  

    λ = max(a∗a)·n / (∑a)² = max((p+1)²·B(p+1,p+1), (p+1)²/n),

where B is the beta function.  The first term corresponds to the interior
of the convolution, the second to the endpoint term a_n².
For the usual problem size (n≈3‑4 k) the optimum lies at p≈5–6,
giving λ≈0.01 and therefore E≈0.02 – far better than the Catalan
construction (E≈1.57).

The function below tries a handful of exponents around the theoretical
optimum, evaluates each candidate (the cost of a single convolution is
tiny for n≤5000) and returns the best one found.
"""
import time
import math
import numpy as np

def _build_power_sequence(n: int, p: float) -> list[float]:
    """
    Build the sequence a_i = C·(i+1)^p with C chosen so that max(a)=1000.
    The result is clipped to the interval [0,1000] just in case of rounding.
    """
    # Scaling constant: the largest raw value is n**p, we want it to become 1000.
    scale = 1000.0 / (n ** p)
    seq = [scale * ((i + 1) ** p) for i in range(n)]
    # Enforce the hard limits imposed by evaluate_sequence.
    seq = [float(min(max(v, 0.0), 1000.0)) for v in seq]
    return seq


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Generate a near‑optimal step‑function by searching a few power‑law
    exponents.  The function respects the 1000‑second time budget (the
    search finishes in a fraction of a second) and returns the best
    sequence discovered.
    """
    start = time.time()

    # ------------------------------------------------------------------
    # Determine the length of the vector.
    # Prefer the length of a previously found construction if it exists,
    # otherwise fall back to a modest size that still yields a very good
    # bound while staying fast to evaluate.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 4000  # a safe default; any n≥2 works

    # Exponents to try.  Values around 5–6 are theoretically optimal.
    # A few extra points are added to be robust against rounding effects.
    exponents = [4.0, 5.0, 5.12, 5.2, 5.4, 5.6, 6.0, 7.0]

    best_score = math.inf
    best_seq = None

    for p in exponents:
        seq = _build_power_sequence(n, p)

        # Evaluate – any exception (e.g. overflow) is ignored.
        try:
            score = evaluate_sequence(seq)   # evaluate_sequence is provided by the framework
        except Exception:
            continue

        print(f"[propose_candidate] n={n}, p={p:.3f}, score={score:.8f}")

        if score < best_score:
            best_score = score
            best_seq = seq

    # If something went wrong and no sequence was accepted, fall back to the
    # Catalan construction used previously.
    if best_seq is None:
        print("[propose_candidate] fallback to Catalan sequence")
        try:
            from math import comb
        except Exception:
            pass
        # Simple Catalan generator (identical to the one in the previous code)
        def _catalan_sequence(m: int) -> list[float]:
            seq = [1.0]
            for i in range(1, m):
                seq.append(seq[-1] * (2 * i - 1) / (2 * i))

        print("[propose_candidate] warning: exceeded time budget")
    else:
        print(f"[propose_candidate] finished in {elapsed:.2f}s, best score={best_score:.8f}")

    return best_seq
