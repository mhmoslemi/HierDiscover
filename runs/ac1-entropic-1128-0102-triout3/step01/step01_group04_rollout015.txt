def _target_sum(n):
    return np.sqrt(2.0 * n)

def _fast_evaluate(seq):
    """Fast version of evaluate_sequence (without security checks)."""
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    max_b = np.max(np.convolve(seq, seq))
    return 2.0 * n * max_b / (sum_a * sum_a)

def _analytic_sqrt_sequence(n, delta=0.0):
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale

def _power_derivative_sequence(n, p, delta=0.0):
    i = np.arange(n, dtype=float)
    base = (i + 1.0 + delta) ** p - (i + delta) ** p
    # ensure no negative due to numerical errors (should be >=0)
    base = np.maximum(0.0, base)
    scale = _target_sum(n) / base.sum()
    return base * scale

def _geometric_sequence(n, r):
    i = np.arange(n, dtype=float)
    base = r ** i
    scale = _target_sum(n) / base.sum()
    return base * scale

def _rand_dirichlet_sequence(n, alpha):
    seq = np.random.dirichlet(np.full(n, alpha))
    seq = seq * _target_sum(n)
    return seq

    Returns the best sequence seen and its score.
    """

    # step size proportional to mean entry

            break

        cand_seq = seq.copy()

        # Choose a move type: single‑coordinate perturbation or a transfer between two sites.
        if np.random.rand() < 0.5:

            idx = np.random.randint(n)

            if new_val < 0.0:
                new_val = 0.0

            # renormalise (keeps the L1‑norm fixed)

        else:

                cand_seq[i] += amount

            best_seq = cand_seq.copy()
            seq = cand_seq  # continue from the improved point

"""
Search for a low‑score non‑negative step‑function.

The algorithm works in three stages inside the 1000 s budget:

1. **Random parametric sampling** – generate candidates from a handful
   of expressive families (sqrt‑derivative, power‑derivative,
   geometric decay, Dirichlet random vectors) at several lengths.
   The parameters (e.g. the shift δ, exponent p, decay r) are drawn
   from wide ranges.  Each raw candidate is evaluated and the best
   one is kept.

2. **Quick hill‑climbing refinements** – every promising candidate
   undergoes a short stochastic hill‑climb that either perturbs a single
   coordinate (followed by L¹‑normalisation) or transfers a small amount
   of mass between two indices.  Accepted moves strictly improve the
   objective; the step‑size slowly decays.

3. **Final intensive optimisation** – the overall best sequence is
   refined for the rest of the time budget with a longer hill‑climb.

All scores are computed with a fast replica of `evaluate_sequence`
(the objective is scale‑invariant, so we keep the L¹‑norm fixed only for
numerical stability).  The final sequence is re‑normalised to the
standard sum √(2 n) and clipped to the admissible range before being
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – pure functions, no closures
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Target L¹ norm used by the reference code: √(2 n)."""
    return np.sqrt(2.0 * n)


def _fast_evaluate(seq: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` for internal use.
    Returns 2·n·max(conv(seq,seq)) / (sum(seq)²).
    """
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    max_b = np.max(np.convolve(seq, seq))
    return 2.0 * n * max_b / (sum_a * sum_a)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """
    a_i = √(i+1+δ) – √(i+δ), then scaled to ∑a = √(2 n).
    """
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _power_derivative_sequence(n: int, p: float, delta: float = 0.0) -> np.ndarray:
    """
    Generalised derivative of xᵖ:
        a_i = (i+1+δ)ᵖ – (i+δ)ᵖ,
    scaled to the required L¹ norm.
    """
    i = np.arange(n, dtype=float)
    base = (i + 1.0 + delta) ** p - (i + delta) ** p
    base = np.maximum(base, 0.0)          # guard against tiny negative round‑off
    scale = _target_sum(n) / base.sum()
    return base * scale


def _geometric_sequence(n: int, r: float) -> np.ndarray:
    """
    Exponential decay a_i = rⁱ, 0 < r < 1, scaled to the target sum.
    """
    i = np.arange(n, dtype=float)
    base = r ** i
    scale = _target_sum(n) / base.sum()
    return base * scale


def _rand_dirichlet_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Random Dirichlet draw with concentration `alpha`, scaled to the target sum.
    """
    seq = np.random.dirichlet(np.full(n, alpha))
    return seq * _target_sum(n)


def _hill_climb(seq: np.ndarray, max_iter: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with two move types:
        – single‑coordinate Gaussian perturbation + renormalisation,
        – transfer of a small amount of mass between two indices.
    The step size (sigma) decays geometrically.
    Returns the best sequence seen and its score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _fast_evaluate(best_seq)

    # Initial proposal scale – a fraction of the mean entry.
    step_sigma = max(1e-8, seq.mean() * 0.2)

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        cand_seq = seq.copy()

        if np.random.rand() < 0.5:
            # ---- single‑coordinate perturbation ---------------------------------
            idx = np.random.randint(n)

        else:

            best_score = cand_score
