        neg = direction < 0

"""
Propose a step‑function that minimises the evaluation score.

Algorithm outline
-----------------
1. **Initial population** – start from the best known construction
   (`height_sequence_1` if it exists).  If it is unavailable a quick
   random search over the power‑law family a_i = (i + c)^‑α provides a
   decent warm‑start.
2. **Gradient‑based max‑convolution reduction** – repeatedly
   * compute the full linear convolution a*a,
   * locate the index `k*` where it attains its maximum,
   * obtain the exact gradient of the entry `conv[k*]` w.r.t. `a`
     (a closed‑form slice operation, no loops),
   * move `a` in the opposite direction while keeping the vector on the
     probability simplex (non‑negative and sum = 1).  Step size is
     adapted automatically.
3. **Exploration** – occasional large random perturbations, occasional
   restarts from the parametric family and an occasional simulated‑annealing‑style
   acceptance of a slightly worse move help to escape local minima.
4. **Termination** – the loop stops when the 1000 s budget is exhausted.
   The best sequence encountered is returned as a plain Python list.

All heavy work (convolution, gradient) is performed with FFT
(`numpy.fft.rfft`), giving O(n log n) per iteration.  The code respects
the constraints enforced by `evaluate_sequence` (non‑negative entries,
sum > 0.01, clipping at 1000 – the latter never occurs for our normalised

def linear_self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using a real FFT.
    Returns the full convolution array of length 2*len(a)-1.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def max_convolution_and_index(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Compute the maximum entry of a*a and the index where it is attained.
    """
    conv = linear_self_convolution(a, fft_len)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def gradient_of_conv_at_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the convolution entry conv[k] = (a*a)[k]
    with respect to a (the factor 2 is omitted – only direction matters).

    The result is a vector g such that
        d conv[k] / d a_i = 2 * g[i].

    The implementation uses pure NumPy slicing, O(n) time, no loops.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k < n:                     # left half (including centre)
        # g[0 : k+1] = a[k::-1]
        g[: k + 1] = a[k :: -1]
    else:                         # right half
        d = k - (n - 1)            # distance from the centre
        # slice from a[n-1] down to a[d] (inclusive)
        g[d:] = a[n - 1 : d - 1 : -1]

    # The true gradient is 2*g, but the factor 2 is irrelevant for a
    # direction vector.
    return g


def normalise_vector(v: np.ndarray) -> np.ndarray:
    """Project onto the simplex (non‑negative, sum = 1)."""
    s = v.sum()
    if s == 0.0:
        return v
    return v / s

            if np.random.rand() < 0.01:  # occasional big jump
                i1, i2 = np.random.choice(n, size=2, replace=False)
                delta = (np.random.rand() - 0.5) * 0.05   # up to ±5 % of donor
                if a[i1] + delta < 0.0:
                    delta = -a[i1] + 1e-12
                if a[i2] - delta < 0.0:
                    delta = a[i2] - 1e-12
                a[i1] += delta
                a[i2] -= delta
                a = normalise_vector(a)
            # If we have been stuck for a while, restart from a fresh parametric point
            if no_improve > 250:
                alpha = np.random.uniform(0.30, 0.80)
                c = np.random.uniform(1.0, n / 2.0)
                i = np.arange(n, dtype=np.float64)
                a = normalise_vector((i + c) ** (-alpha))
                no_improve = 0
                step_factor = 0.5
                print(f"[restart] α={alpha:.3f} c={c:.1f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # extremely unlikely – fallback to uniform distribution
        best_seq = np.ones(n) / n

    # Ensure compliance with the security checks in evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_seq = best_seq.tolist()
    print(f"[final] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq
