    best_seq = seq.copy()

    for it in range(max_iter):

        grad = np.zeros_like(seq)

"""
Optimisation of the step‑function bound
---------------------------------------

The score returned by ``evaluate_sequence`` (with the normalisation
∑a = √(2 n) used throughout) is exactly the maximal value of the
autocorrelation ``conv(a,a)``.  The goal is therefore to minimise that
maximum.

The optimiser works as follows:

1. **Diverse initial shapes** –  Two analytic families are used:

   * the discrete derivative of √(x+δ) (the “sqrt‑family”)
   * the discrete derivative of (x+δ)^γ for γ∈(0,1) (the “power‑family”)

   Random values for the parameters (δ,γ) give a wide variety of
   starting vectors.

2. **Spread large entries** –  Before any optimisation the vector can be
   permuted by a zig‑zag pattern that places the biggest entries alternately
   at the left and right ends.  This reduces the overlap of large values in
   the convolution and often lowers the maximum.

3. **Gradient‑descent on the max‑convolution** –  With the L¹‑norm fixed,
   the objective is ‖a∗a‖∞.  For the current maximal index *k* the
   gradient of the max‑convolution is
        g_i = 2·a_{k‑i}  (zero when the index is out of range).
   A simple projected gradient step (clipping negatives and renormalising)
   is performed repeatedly with a decaying step size.

4. **Mass‑transfer hill climbing** –  After the gradient phase a fast
   stochastic search is run: a random source index loses a small amount
   of mass which is added to a random target index.  The move is kept if
   it improves the score (or, with a tiny probability, if it does not),
   the step size being slowly reduced.

5. **Multiple restarts** –  The whole procedure is repeated with different
   lengths (800 … 2500), different shape parameters and optional
   re‑ordering.  The best sequence seen before the time budget expires is
   returned.  If a global ``height_sequence_1`` variable is present it is
   taken as an additional starting point.

The code is fully deterministic given the ``seed`` argument, uses only
NumPy (no external solvers) and respects the 1000 s budget.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Normalisation used in the reference:  sum = sqrt(2 n)."""
    return np.sqrt(2.0 * n)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """a_i = √(i+1+δ) – √(i+δ), scaled to the required L¹‑norm."""
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _analytic_power_sequence(n: int, gamma: float, delta: float = 0.0) -> np.ndarray:
    """
    a_i = (i+1+δ)^γ – (i+δ)^γ, an analogue of the sqrt‑family with a
    tunable exponent γ∈(0,1).  The result is scaled to the L¹‑norm.
    """
    i = np.arange(n, dtype=float)
    base = (i + 1.0 + delta) ** gamma - (i + delta) ** gamma
    # any numerical issue (e.g. gamma very small) is harmless – the scaling
    # restores a proper L¹‑norm.
    scale = _target_sum(n) / base.sum()
    return base * scale


def _evaluate(seq: np.ndarray) -> float:
    """Wrap the provided ``evaluate_sequence`` (expects a Python list)."""
    return evaluate_sequence(seq.tolist())


def _zigzag_permute(seq: np.ndarray) -> np.ndarray:
    """
    Spread large entries by placing them alternately at the left and right
    ends of the array (high‑low‑high‑low …).  This often lowers the peak of
    the autocorrelation.
    """
    sorted_vals = np.sort(seq)[::-1]               # descending
    n = len(seq)
    out = np.empty_like(seq)
    left, right = 0, n - 1
    toggle = True
    for v in sorted_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


# ----------------------------------------------------------------------
# Gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _gradient_descent(
    seq: np.ndarray,
    max_iter: int,
    deadline: float,
    init_step: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent that minimises the maximal convolution
    value.  The step size starts at ``init_step`` and decays whenever a
    move does not improve the score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    step = max(init_step, 1e-12)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current convolution and its maximal index
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        k = conv.argmax()

        # gradient of the maximal convolution term
        grad = np.zeros_like(seq)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * seq[j[mask]]

        # tentative update
        cand = seq - step * grad
        cand = np.maximum(cand, 0.0)                 # non‑negativity
        s = cand.sum()
        if s == 0.0:
            # cannot renormalise – shrink step and continue
            step *= 0.5
            continue
        cand *= _target_sum(n) / s                  # keep the L¹‑norm fixed

        cand_score = _evaluate(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand                               # keep the improvement
            # keep step size (or even grow a little)
            step *= 1.02
        else:
            # reject – shrink step
            step *= 0.95

    return best_seq, best_score


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------


def _mass_transfer_hill(
    seq: np.ndarray,
    max_iter: int,
    deadline: float,
    init_sigma: float,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a small amount of mass from one index to another.
    Moves that improve the score are always kept; worse moves are kept
    with a tiny probability that decays with a temperature schedule.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    sigma = max(init_sigma, 1e-12)
    temperature = best_score * 0.1  # initial temperature proportional to score

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # choose source (must have some mass) and distinct target
        src = np.random.randint(n)
        if seq[src] <= 0.0:
            continue
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        # propose a transfer amount
        delta = np.random.randn() * sigma
        # keep the source non‑negative (leave a little margin)
        delta = np.clip(delta, -0.9 * seq[src], 0.9 * seq[src])

        # apply move
        seq[src] -= delta
        seq[tgt] += delta

        cand_score = _evaluate(seq)

        # acceptance test
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # Metropolis rule
            prob = math.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()
            # gradually cool down
            temperature *= 0.9999
        else:
            # revert the move
            seq[tgt] -= delta
            seq[src] += delta
            # slightly warm the system again
            temperature *= 1.0001

        # slowly shrink proposal scale
        sigma *= 0.9997

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the evaluation
    function.  The routine runs for at most ``budget_s`` seconds and
    returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds margin for the outer loop exit handling
    deadline = start_time + max(0.0, budget_s - 5.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional: start from the globally supplied construction if any
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is injected by the surrounding framework
        init = np.array(height_sequence_1, dtype=float)
        if init.size > 0:
            n0 = init.size
            init = init * (_target_sum(n0) / init.sum())
            sc = _evaluate(init)
            best_seq = init.copy()
            best_score = sc
            print(f"[global] start from height_sequence_1 n={n0} score={sc:.6f}")
    except Exception:
        # missing or malformed – ignore
        pass

    # ------------------------------------------------------------------
    # Parameter ranges for the random restarts
    # ------------------------------------------------------------------
    possible_lengths = list(range(800, 2501, 200))   # 800,1000,…,2400
    shape_choices = ["sqrt", "power"]

    # ------------------------------------------------------------------
    # Outer restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = int(np.random.choice(possible_lengths))
        shape = np.random.choice(shape_choices)

        # ------------------- initialise a candidate --------------------
        if shape == "sqrt":
            delta = np.random.rand() * 1.0          # δ ∈ [0,1)
            seq = _analytic_sqrt_sequence(n, delta=delta)
        else:
            gamma = np.random.uniform(0.2, 0.8)      # γ ∈ (0,1)
            delta = np.random.rand() * 2.0          # δ ∈ [0,2)
            seq = _analytic_power_sequence(n, gamma=gamma, delta=delta)

        # occasional zig‑zag permutation to spread large entries
        if np.random.rand() < 0.3:
            seq = _zigzag_permute(seq)

        # evaluate the raw start
        cur_score = _evaluate(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[init] n={n:4d} shape={shape:5s} score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – gradient descent on the max‑convolution
        # --------------------------------------------------------------
        inner_deadline = time.time() + 0.45  # give ~0.45 s to this phase
        seq_gd, score_gd = _gradient_descent(
            seq.copy(),
            max_iter=2000,
            deadline=inner_deadline,
            init_step=seq.mean() * 0.15,
        )
        if score_gd < best_score:
            best_score = score_gd
            best_seq = seq_gd.copy()
            print(f"[gd ] n={n:4d} shape={shape:5s} score={score_gd:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – stochastic mass‑transfer hill climbing
        # --------------------------------------------------------------
        if time.time() < deadline:
            inner_deadline = time.time() + 0.45
            seq_hc, score_hc = _mass_transfer_hill(
                seq_gd.copy(),
                max_iter=5000,
                deadline=inner_deadline,
                init_sigma=seq_gd.mean() * 0.25,
            )
            if score_hc < best_score:
                best_score = score_hc
                best_seq = seq_hc.copy()
                print(f"[hc ] n={n:4d} shape={shape:5s} score={score_hc:.6f}")

        # ------------------------------------------------------------------
        # Early exit test – if we have already beaten the target we keep
        # searching only briefly to see if we can improve further.
        # ------------------------------------------------------------------
        if best_score < 1.5:
            # allow a few more restarts but with a very short budget
            # (still respecting the outer deadline)
            continue

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 10
        best_seq = np.ones(n) * (_target_sum(n) / n)

    # Return a plain list as required by the external evaluator
    return best_seq.tolist()
