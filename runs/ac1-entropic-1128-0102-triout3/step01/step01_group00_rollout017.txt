def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    best_score = inf; best_seq = None

    # define lengths list
    lengths = [800, 1200, 1600, 2000, 2500, 3000, 3500, 4000]
    # maybe shuffle
    random.shuffle(lengths)

    # loop through lengths multiple times
    while time.time() < deadline:
        for n in lengths:
            if time.time() >= deadline:
                break
            # Choose number of restarts = maybe 3
            for restart in range(3):
                if time.time() >= deadline:
                    break
                # random init simplex
                a = np.random.rand(n)
                a = a / a.sum()
                # set hyperparameters
                step0 = 0.5
                beta0 = 1.0
                beta_max = 80.0
                max_iter = 40000
                # compute FFT length for convolution
                fft_len = 1 << ((2*n - 1).bit_length())
                for it in range(1, max_iter+1):
                    if time.time() >= deadline:
                        break
                    # conv via FFT
                    Af = np.fft.rfft(a, n=fft_len)
                    conv = np.fft.irfft(Af * Af, n=fft_len)[:2*n-1]
                    max_val = conv.max()
                    cur_score = 2.0 * n * max_val
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[{time.time() - start:6.2f}s] n={n}, restart={restart}, iter={it}, score={cur_score:.6f}")
                    # update beta schedule
                    # maybe linear increase
                    beta = beta0 + (beta_max - beta0) * it / max_iter
                    # compute softmax
                    # subtract max for stability
                    shift = conv - max_val  # negative values
                    w = np.exp(beta * shift)
                    w_sum = w.sum()
                    if w_sum == 0:
                        # fallback to uniform w
                        w = np.full_like(w, 1.0 / w.size)
                    else:
                        w = w / w_sum
                    # gradient via correlation
                    # NB: correlation w (length 2n-1) with a (length n) result length n
                    grad = 2.0 * np.correlate(w, a, mode='valid')
                    # step size schedule diminishing
                    step = step0 / np.sqrt(it)
                    # update a
                    a = a - step * grad
                    # project onto simplex
                    a = _project_simplex(a)
                # end for it
    # after loops
    if best_seq is None:
        best_seq = np.full(10, 0.1) # sum=1
    return [float(x) for x in best_seq]

"""
Propose a candidate step‑function using a smoothed (soft‑max) sub‑gradient descent.

The evaluation is scale‑invariant, so we keep the vector on the simplex
(sum = 1, a_i ≥ 0).  At each iteration we:
  * compute the linear auto‑convolution a*a (FFT),
  * form a soft‑max weight w over the 2n‑1 convolution entries;
    when the temperature (β) is low the weights are almost uniform,
    and as β grows they concentrate on the largest entries,
    smoothly interpolating between a full gradient and the classic
    max‑convolution sub‑gradient,
  * the gradient of the weighted sum is 2·corr(w,a) (cross‑correlation),
  * take a diminishing step and project back onto the simplex.

β is increased from a small value to a large one, turning the method
into a refined max‑convolution descent.  After the soft‑max phase we
run a short pure max‑convolution sub‑gradient polishing.  Several
sequence lengths and random restarts are tried within the time budget.
The best sequence found (lowest score) is returned.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum(x) = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes
    `evaluate_sequence`.  The routine runs for at most `budget_s`
    seconds and prints progress messages.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to guarantee returning before the deadline
    deadline = start_time + budget_s - 1.0

    best_score = float('inf')
    best_seq = None

    # candidate lengths (larger n gives a lower theoretical bound)
    lengths = [1000, 1200, 1500, 1800, 2000, 2400, 3000, 3500]
    np.random.shuffle(lengths)

    # optional starting point – the best construction found by previous runs
    try:
        base_seq = list(height_sequence_1)   # type: ignore
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # Main optimisation loops
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # number of independent restarts for each length
        n_restarts = 3

        # pre‑compute FFT length for the a*a convolution (next power of two)
        fft_len = 1 << ((2 * n - 1).bit_length())

        for restart in range(n_restarts):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------------
            # initialise a random point on the simplex (or a perturbed known one)
            # ------------------------------------------------------------------
            if base_seq is not None and len(base_seq) <= n and restart == 0:
                a = np.array(base_seq, dtype=float)
                if a.size < n:
                    a = np.pad(a, (0, n - a.size), constant_values=0.0)
                # add a tiny random perturbation to avoid being stuck exactly at a
                a += 0.001 * np.random.rand(n)
            else:
                a = np.random.rand(n)

            # normalise to the simplex
            a = a / a.sum()
            a = _project_simplex(a)

            # ------------------- soft‑max gradient descent -------------------
            max_iter = 30000                     # cascade of iterations
            step0 = 0.20                         # initial step size

            beta_max = 50.0                      # final temperature (focuses on max)

                Af = np.fft.rfft(a, n=fft_len)

                max_val = conv.max()
                cur_score = 2.0 * n * max_val

                # keep best solution seen so far
                if cur_score < best_score:

                beta = beta0 + (beta_max - beta0) * it / max_iter

                step = step0 / math.sqrt(it)

                a = _project_simplex(a)

                    print(f"[{time.time() - start_time:6.2f}s] "

                          f"score={cur_score:.6f}")

                g = np.zeros_like(a)
                low = max(0, k_max - (n - 1))

                g[i_idx] = 2.0 * a[j_idx]

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    # Convert to plain Python list (the evaluator accepts any list of floats)
    return [float(x) for x in best_seq]
