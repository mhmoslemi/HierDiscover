"""
Propose a candidate step‑function by searching over power‑law decays
a_i = (i+1)^{-e}.  The functional to minimise is
    E(a) = 2·n·max_k (a∗a)[k] / (∑ a_i)².
For a fixed length n the quantity is invariant under a global scaling,
so we can vary only the shape.  Theory shows that sequences of the form
a_i ≈ i^{-e} with e≈0.7 give a much smaller maximal self‑convolution
than the Catalan construction (which corresponds to e=0.5).  The function
scans the exponent e in a coarse‑to‑fine manner, keeps the best sequence
found (according to ``evaluate_sequence``) and returns it.  The search
stops early if the allotted time budget is exhausted.
"""

import time
import numpy as np
import random

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function by optimising a single power‑law exponent.

    Parameters
    ----------
    seed : int, optional
        Random seed (unused, kept for signature compatibility).
    budget_s : int, optional
        Total time budget in seconds.  The routine will stop well before
        this limit and return the best sequence discovered.
    **kwargs : dict
        Ignored – kept for compatibility.

    Returns
    -------
    list[float]
        A list of non‑negative heights that gives a low evaluation score.
    """
    start = time.time()
    random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction exists we reuse its size,
    # otherwise we fall back to a moderately large length that still evaluates
    # quickly.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 4000

    # ------------------------------------------------------------------
    # Utility: build a power‑law sequence for a given exponent e (>0).
    # ------------------------------------------------------------------
    indices = np.arange(1, n + 1, dtype=float)  # i+1 values

    def make_sequence(e: float) -> list[float]:
        # a_i = (i+1)^{-e}
        seq = np.power(indices, -e)
        # Ensure we respect the clipping performed by evaluate_sequence
        # (all values already lie in [0,1] for the ranges we explore)
        return seq.tolist()

    # ------------------------------------------------------------------
    # Search over the exponent.
    # The problem is well‑behaved: the score is roughly unimodal in e.
    # We perform a coarse grid, then zoom in around the best point.
    # ------------------------------------------------------------------
    best_seq = None
