"""
Propose a step‑function using an accelerated sub‑gradient method.
The algorithm works on the probability simplex (sum = 1, a_i ≥ 0) and
minimises 2·n·max(conv(a,a)).  It improves upon the basic sub‑gradient
method by:
* using a *convex combination* of all constraints that are (almost) active
  (all convolution entries within a tiny tolerance of the maximum);
* adding a simple momentum term (exponential moving average of the
  sub‑gradients);
* employing a diminishing step size  step₀ / sqrt(t);
* optionally starting from the previously discovered best sequence
  (global ``height_sequence_1``) and continuing the optimisation for as
  long as the time budget allows.

The routine runs for at most ``budget_s`` seconds (default 1000 s) and
returns the best sequence found.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum(x) = 1}
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) convolution a * a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * n - 1]


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin so that the wrapper does not cut us off
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Initial sequence --------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the surrounding harness
        init = np.array([float(x) for x in height_sequence_1], dtype=np.float64)
        n = init.shape[0]
        if n == 0:
            raise ValueError
        a = init.copy()
        # ensure simplex feasibility
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            a = np.random.rand(n)
        a = a / a.sum()
        print(f"[{0:.2f}s] Starting from existing height_sequence_1 (n={n})")
    except Exception:
        # fallback: a random start of moderate size
        n = 1500
        a = np.random.rand(n)
        a = a / a.sum()
        print(f"[{0:.2f}s] Starting from random sequence (n={n})")

    # ------------------------------------------------------------------
    # Optimisation parameters -------------------------------------------
    # ------------------------------------------------------------------
    step0 = 0.8               # initial step magnitude
    beta = 0.9                # momentum coefficient
    rel_tol = 1e-8            # relative tolerance for “active” set
    abs_tol = 1e-12           # absolute tolerance for “active” set
    max_no_improve = 5000     # iterations after which a tiny perturbation is added

    # bookkeeping
    best_score = float('inf')
    best_seq = a.copy()
    v = np.zeros_like(a)      # momentum accumulator
    it = 0
    last_improve_it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop --------------------------------------------
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- convolution a * a (linear) --------------------------------
        conv = _conv_self(a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # ---- record improvement ----------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve_it = it
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.2f}s] iter={it:6d}, n={n:4d}, score={cur_score:.6f}")

        # ---- determine the (almost) active set -------------------------
        tol = max(rel_tol * max_val, abs_tol)
        active = np.where(conv >= max_val - tol)[0]
        n_active = active.size
        if n_active == 0:                # should never happen, but guard
            active = np.array([np.argmax(conv)], dtype=int)
            n_active = 1

        # ---- compute a convex combination of sub‑gradients -------------
        #    sub‑gradient for a single active index k is
        #    g_i = 2 * a_{k-i} (zero when out of range)
        g = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            i_idx = np.arange(low, high + 1, dtype=int)
            j_idx = k - i_idx
            g[i_idx] += a[j_idx]
        # average over the active constraints
        g = (2.0 / n_active) * g

        # ---- momentum update -------------------------------------------
        v = beta * v + (1.0 - beta) * g

        # ---- diminishing step size ---------------------------------------
        step = step0 / math.sqrt(it)

        # ---- gradient step + projection onto the simplex -----------------
        a = _project_simplex(a - step * v)

        # ---- occasional small random perturbation to escape plateaus --
        if it - last_improve_it > max_no_improve:
            # add a whisper of Gaussian noise, keep feasibility
            noise = np.random.randn(n) * 1e-3
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            last_improve_it = it   # reset the counter

    # ------------------------------------------------------------------
    # Return the best sequence found ------------------------------------
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    print(f"[{elapsed:6.2f}s] Finished. Best score = {best_score:.6f}")
    # ensure the output is a plain Python list of floats
    return [float(x) for x in best_seq]
