"""
Propose a step‑function sequence that minimises
evaluate_sequence(sequence).

The algorithm is a hybrid of:
* sub‑gradient descent on the true max‑convolution (using an exact
  sub‑gradient obtained from the active set of constraints),
* occasional random mass‑transfer moves (a simple simulated‑annealing
  style neighbourhood),
* periodic restarts from a Dirichlet draw to avoid being trapped in
  local minima.

All iterates are kept on the ℓ₁‑simplex {x ≥ 0, Σx = √(2n)} so that the
objective 2·n·max(conv)/(Σx)² reduces to the maximal entry of the
self‑convolution.  The routine runs until the time budget expires (or
the target bound 1.503 is reached) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full (2·len(a)‑1) result (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x ≥ 0, Σx = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max_subgrad(a: np.ndarray, mask: np.ndarray, nfft: int) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)[k]  at the current active set.
    mask is a binary array (same length as the convolution) with 1s at
    the indices where the convolution attains its current maximum.
    The gradient is 2 * (mask * a_rev) where * denotes convolution.
    Returned gradient has length len(a).
    """
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0:
        return np.zeros_like(a)

    w = w / w_sum                     # normalise to a probability vector
    a_rev = a[::-1]

    A_rev = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    # slice that corresponds to the gradient entries
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad

# Main search routine
# ----------------------------------------------------------------------

    """
    Hybrid sub‑gradient / simulated‑annealing search for a low‑max‑convolution
    step‑function.  Returns the best sequence found before the time budget
    expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # problem size, target ℓ₁‑norm and FFT length
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=np.float64)
    else:
        n = 2000

    nfft = _next_pow2(2 * n - 1)                 # convenient FFT length

    if a0 is not None:
        a = _project_to_simplex(a0, S)
    else:
        a = np.full(n, S / n, dtype=np.float64)  # uniform start

    conv = _convolve_self_fft(a, nfft)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

    best_a = a.copy()
    best_score = cur_score

    # hyper‑parameters
    # ------------------------------------------------------------------

    temperature = 0.12                            # initial Metropolis temperature
    temp_decay = 0.9995                           # geometric decay per iteration
    min_temp = 1e-6

                a_candidate = a.copy()

        elif r < p_subgrad + p_random:
            # -------- random mass‑transfer move -----------------------
            # pick a source proportional to its mass (focus on large entries)
            if a.sum() > 0:
                src_probs = a / a.sum()
                i = np.random.choice(n, p=src_probs)
            else:
                i = np.random.randint(n)
            # pick a destination uniformly (different from source)
            j = i
            while j == i:
                j = np.random.randint(n)

            # transfer a modest fraction of the source mass
            eps_transfer = 0.08
            delta = eps_transfer * a[i]
            if delta <= 0:
                delta = eps_transfer * S / n

            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

        else:
            # -------- occasional full restart ------------------------
            alpha = np.full(n, 0.4)                      # Dirichlet concentration
            a_candidate = np.random.dirichlet(alpha) * S
            print(f"[{iteration:06d}] Full restart performed.")
            # no need to project – Dirichlet already sums to S

        # ------------------------------------------------------------------
        # evaluate candidate (full objective)
        # ------------------------------------------------------------------
        cand_conv = _convolve_self_fft(a_candidate, nfft)
        cand_max = cand_conv.max()
        cand_score = 2.0 * n * cand_max / (a_candidate.sum() ** 2)

        # acceptance rule (greedy + Metropolis)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis probability
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_score = cand_score
            cur_max = cand_max
        else:
            # occasional jitter to get out of shallow plateaux
            if np.random.rand() < 0.01:
                jitter = np.random.randn(n) * (step_base * 0.5)
                a = _project_to_simplex(a + jitter, S)

        # temperature decay
        temperature = max(temperature * temp_decay, min_temp)

        # restart if stagnating for too long
        if no_improve >= max_no_improve:
            alpha = np.full(n, 0.3)
            a = np.random.dirichlet(alpha) * S
            no_improve = 0
            print(f"[{iteration:06d}] Stagnation restart (no improvement).")

        iteration += 1

    # ------------------------------------------------------------------
    # final verification and return
    # ------------------------------------------------------------------
    best_a = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return best_a.tolist()
