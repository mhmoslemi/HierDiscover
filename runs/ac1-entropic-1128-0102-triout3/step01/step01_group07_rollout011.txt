"""
Search for a non‑negative step‑function that minimises

        2 * n * max(a * a) / (sum(a)**2)

by a stochastic projected Adam optimiser.  The optimisation is performed
on the probability simplex (∑a_i = 1, a_i ≥ 0) and the vector is kept
monotonically decreasing (a_i ≥ a_{i+1}) after every projection – a shape
that is known to reduce the central convolution entry.  A smooth
approximation of the max‑operator (soft‑max) is used at the beginning,
then the temperature is slowly increased so that the algorithm converges
to the true objective.  Multiple random restarts with different lengths
and initialisations are run until the time budget expires.  The best
sequence found is returned.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1          # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Return soft‑max over the convolution entries with temperature 1/gamma."""
    # stable computation
    scaled = gamma * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    return exp_scaled / exp_scaled.sum()


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max objective.
    a – current sequence (length n)
    w – soft‑max weights (length 2n‑1) that sum to 1
    Returns gradient of size n.
    """
    n = a.shape[0]
    # correlation a ⋆ w  (full) gives length 3n‑2, slice the part we need
    corr = np.correlate(a, w, mode='full')
    start = 2 * n - 2
    grad = 2.0 * corr[start:start + n]
    return grad


def _score_from_seq(seq: np.ndarray) -> float:
    """
    Compute the exact evaluation score for a sequence that already satisfies
    ∑a_i = 1.  This is simply 2 * n * max(conv(seq, seq)).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


def _initial_powerlaw(n: int, beta: float) -> np.ndarray:
    """Monotonically decreasing power‑law (i+1)^{-beta}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-beta)
    return a / a.sum()


def _initial_random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex and sort it decreasingly."""
    a = np.random.rand(n)
    a = _project_simplex(a)
    a = np.sort(a)[::-1]
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights (a sequence) that attempts
    to minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # initialise the global best from any previously known construction
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    if 'height_sequence_1' in globals():
        try:
            cand = list(height_sequence_1)
            # normalise to ℓ1 = 1 (evaluate_sequence is scale‑invariant)
            cand = np.array(cand, dtype=np.float64)
            cand = cand / cand.sum()
            cand = np.sort(cand)[::-1]               # ensure monotone decreasing
            best_seq = cand.tolist()
            best_score = _score_from_seq(cand)
            print(f"[init] Using height_sequence_1 with score {best_score:.6f}")
        except Exception:
            best_seq = None
            best_score = float('inf')

    # ------------------------------------------------------------------
    # hyper‑parameters that will be reused
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 8000]          # a few representative sizes
    max_gamma = 30.0
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # ------------------------------------------------------------------
    # outer loop – keep launching new random restarts until time runs out
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1
        # pick a length  (larger lengths give more room for improvement)
        n = int(np.random.choice(lengths))
        # initial distribution – either power‑law or random simplex
        if np.random.rand() < 0.6:
            beta = np.random.uniform(0.0, 3.0)
            a = _initial_powerlaw(n, beta)
        else:
            a = _initial_random_simplex(n)

        # optimisation variables for Adam
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # learning‑rate schedule (large at start, decays slowly)
        init_lr = 0.1 / np.sqrt(n)          # scale with size
        lr = init_lr

        gamma = 0.5                         # start with a smooth max
        max_iters = 3500                    # per restart (adjustable)

        for it in range(1, max_iters + 1):
            # time check – abort inner loop if needed
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------------
            # evaluate current candidate (exact objective)
            # ------------------------------------------------------------------
            conv = np.convolve(a, a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[best] restart {restart}, n={n}, it={it}, "
                      f"score={best_score:.6f}")

            # ------------------------------------------------------------------
            # compute smooth max weights
            # ------------------------------------------------------------------
            w = _softmax_weights(conv, gamma)

            # ------------------------------------------------------------------
            # gradient of the smoothed objective
            # ------------------------------------------------------------------
            grad = _gradient_softmax(a, w)

            # ------------------------------------------------------------------
            # Adam update (pre‑projection)
            # ------------------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            # bias‑corrected moments
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step

            # ------------------------------------------------------------------
            # projection onto the simplex + enforce monotone decreasing shape
            # ------------------------------------------------------------------
            a = _project_simplex(a)
            a = np.sort(a)[::-1]                     # monotone decreasing

            # ------------------------------------------------------------------
            # occasional jitter to escape shallow plateaux
            # ------------------------------------------------------------------
            if (it % 400) == 0:
                jitter = np.random.randn(n) * lr * 0.05
                a += jitter
                a = np.maximum(a, 0.0)
                a = _project_simplex(a)
                a = np.sort(a)[::-1]

            # ------------------------------------------------------------------
            # schedule updates
            # ------------------------------------------------------------------
            lr *= 0.9995                               # slow decay
            gamma = min(max_gamma, gamma * 1.008)      # increase smoothness

        # end of a restart --------------------------------------------------
        # quick sub‑gradient polishing on the best candidate of this restart
        # (optional – cheap few hundred steps)
        if time.time() < deadline and best_seq is not None:
            a = best_seq.copy()
            eta = 0.5 / n
            for _ in range(300):
                conv = np.convolve(a, a)
                max_idx = int(np.argmax(conv))
                # build sub‑gradient for the active index
                grad = np.zeros_like(a)
                start = max(0, max_idx - (n - 1))
                end = min(max_idx, n - 1)
                idx = np.arange(start, end + 1, dtype=int)
                grad[idx] = 2.0 * a[max_idx - idx]
                a_new = a - eta * grad
                a_new = _project_simplex(a_new)
                a_new = np.sort(a_new)[::-1]
                new_score = _score_from_seq(a_new)
                cur_score = _score_from_seq(a)
                if new_score < cur_score:
                    a = a_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                eta *= 0.995

    # ----------------------------------------------------------------------
    # Fallback – if nothing was ever discovered use a modest uniform sequence
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 500
        a = np.full(n, 1.0 / n)
        best_seq = a
        best_score = _score_from_seq(a)

    # final check with the provided evaluate_sequence (just for safety)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final reported score by evaluate_sequence: {final_score:.6f}")
    except Exception:
        final_score = best_score
        print(f"Final reported score (internal): {best_score:.6f}")

    print(f"Returning sequence of length {len(best_seq)} with score {final_score:.6f}")
    return best_seq.tolist()
