def project_simplex(v):
    """Projection onto the probability simplex {x >=0, sum x = 1}."""
    # using algorithm from https://arxiv.org/abs/1309.1541
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0, return uniform
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.  The algorithm consists of three phases:

1. **Warm‑start** – a very fast random search over a parametric family
   a_i = (i + c)^(-α) (power‑law) to obtain a decent initial shape.
2. **Projected sub‑gradient descent** – at each iteration the index
   k* where the self‑convolution a*a attains its maximum is found.
   The sub‑gradient of max_k (a*a)_k with respect to a is
   g_j = 2·a_{k*‑j} (zero when the index is out of range).  A step
   opposite to this gradient is taken, the vector is clipped to
   non‑negative values and projected onto the probability simplex
   (∑a_i = 1).  The step size decays as 1/√t.
3. **Diversification** – occasional random restarts from the
   parametric family and occasional tiny “hill‑climbing” mass transfers
   keep the search from stagnating.

All heavy work (FFT‑based convolution, projection) is O(n log n) and
the loop runs until the time budget (≈ 1000 s) is exhausted.  The best
sequence found is returned as a plain Python list.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Compute the self‑convolution a*a via FFT and return (max value, argmax index).
    The convolution length is 2·len(a)‑1; only the relevant part is examined.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return the objective 2·n·max(a*a) for a given (already normalised) a."""
    n = len(a)
    max_conv, _ = _conv_max(a, fft_len)
    return 2.0 * n * max_conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is all zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _build_power_law(n: int, alpha: float, c: float) -> np.ndarray:
    """Return the normalised power‑law vector a_i = (i + c)^(-α)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + c) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s

                break
            cand = _build_power_law(n, float(alpha), float(c))
            sc = _fast_score(cand, fft_len)
            if sc < best_score:
                best_score, best_seq = sc, cand
                print(f"[grid] α={alpha:.3f} c={c} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent + diversification
    # ------------------------------------------------------------------
    step0 = 0.30               # initial step size
    it = 0

    while time.time() < finish_time:
        # ---- sub‑gradient step on the current best vector ----
        a = best_seq.copy()
        max_conv, k_star = _conv_max(a, fft_len)

        # sub‑gradient of max_k (a*a)_k : g_j = 2·a_{k*‑j}
        start_idx = max(0, k_star - (n - 1))
        end_idx = min(n - 1, k_star)
        idx_range = np.arange(start_idx, end_idx + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx_range] = 2.0 * a[k_star - idx_range]

        # descending step (step size decays like 1/√t)
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        sc_new = _fast_score(a_new, fft_len)
        if sc_new < best_score:
            best_score, best_seq = sc_new, a_new
            print(f"[pgd] it={it} improved → score={sc_new:.6f}")

        # ---- occasional tiny random mass transfer (hill‑climb) ----
        if np.random.rand() < 0.03:
            a_hc = best_seq.copy()
            i1, i2 = np.random.choice(n, size=2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.04   # up to ≈2 % of a[i1]
            # enforce non‑negativity while preserving total mass
            if a_hc[i1] + delta < 0.0:
                delta = -a_hc[i1] + 1e-12
            if a_hc[i2] - delta < 0.0:
                delta = a_hc[i2] - 1e-12
            a_hc[i1] += delta
            a_hc[i2] -= delta
            a_hc = np.maximum(a_hc, 0.0)
            a_hc = a_hc / a_hc.sum()
            sc_hc = _fast_score(a_hc, fft_len)
            if sc_hc < best_score:
                best_score, best_seq = sc_hc, a_hc
                print(f"[hill] it={it} improved → score={sc_hc:.6f}")

        # ---- random restart from the power‑law family (exploration) ----
        if np.random.rand() < 0.015:
            r_alpha = np.random.uniform(0.35, 0.75)
            r_c = np.random.uniform(1.0, float(n) / 2.0)
            cand = _build_power_law(n, r_alpha, r_c)
            sc_cand = _fast_score(cand, fft_len)
            if sc_cand < best_score:
                best_score, best_seq = sc_cand, cand
                print(f"[restart] it={it} α={r_alpha:.3f} c={r_c:.1f} → score={sc_cand:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    # Final sanity: ensure the list respects the clipping limits of
    # `evaluate_sequence` (0 ≤ x ≤ 1000).  Our values are tiny, so this is
    # automatically satisfied.
    return best_seq.tolist()
