def lp_improve(f, M, t=0.5):
    # Solve LP to maximize sum(g) such that f * g <= M
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Build A_ub matrix for constraints (2n-1 constraints)
    # Each row k: sum_i f_i g_{k-i} <= M
    # Represent as row vector A[k,i] = f_{k-i} if 0 <= k-i < n else 0
    # So basically A = toeplitz of f with zeros for shape.
    # Use basis: for i from 0 to n-1:
    #   For k from i to i+n-1: coefficient A[k,i] = f[i]
    # So each variable g_i influences outputs indices k = i .. i+n-1.

    # Build A as sparse matrix using scipy.sparse.lil
    rows = []
    cols = []
    data = []
    for i in range(n):
        start = i
        end = i + n  # exclusive
        # rows indices for this column i: start..end-1 (size n)
        # values f[i] repeated n times
        rows.extend(range(start, end))
        cols.extend([i]*n)
        data.extend([f[i]]*n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    b = np.full(2*n-1, M)

    # Solve using linprog high s
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback to original f
        return f

    g = res.x
    sum_g = g.sum()
    if sum_g < 1e-9:
        return f

    # scale g to sum = target Sum
    S = np.sqrt(2*n)
    g_scaled = g * (S / sum_g)

    # Interpolate: new f = (1 - t) * f + t * g_scaled
    new_f = (1 - t) * f + t * g_scaled

    # Renormalize to ensure sum = S (float rounding)
    new_f = new_f * (S / new_f.sum())

    return new_f

"""
Improved step‑function construction.

The algorithm starts from the analytic “sqrt‑derivative” shape
    a_i ∝ √(i+1+δ) – √(i+δ)
(which already beats the naïve constant sequence).  From there we run an
LP‑based “balancing” phase that is the exact counterpart of the method
described in the statement:

    * compute M = max(conv(a,a))
    * solve the linear programme
          maximise   Σ_j g_j
          subject to g_j ≥ 0  and   (a * g)[k] ≤ M   for all k
      (the constraints are linear because a is fixed)
    * rescale g to the required ℓ¹‑norm √(2n) and mix
          a ← (1‑t)·a + t·g                         (t≈0.6)
      then renormalise to keep the ℓ¹‑norm fixed.
      The mixing strictly reduces the maximal convolution value,
      so the procedure is iterated a few times.

After the LP phase a short stochastic hill‑climbing refinement (single‑
coordinate Gaussian moves) is performed to polish the result.

Multiple random restarts with different lengths n (≈ 500‑2000) and
different δ values are launched until the time budget expires.
The best sequence seen so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports – the code works without them (falls back to
# pure hill‑climbing) but they give a strong improvement.
# ----------------------------------------------------------------------
try:
    import scipy.sparse as _sp
    import scipy.optimize as _opt
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Desired ℓ¹‑norm: √(2·n)."""
    return np.sqrt(2.0 * n)


def _analytic_sqrt_sequence(n: int, delta: float = 0.0) -> np.ndarray:
    """
    Analytic starting shape:
        a_i = √(i+1+δ) – √(i+δ)
    scaled to ℓ¹‑norm √(2n).
    """
    i = np.arange(n, dtype=float)
    base = np.sqrt(i + 1.0 + delta) - np.sqrt(i + delta)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _evaluate(seq: np.ndarray) -> float:
    """Wrapper for the provided evaluate_sequence (expects a Python list)."""
    # evaluate_sequence is assumed to be present in the execution environment
    return evaluate_sequence(seq.tolist())


def _convolution_max(seq: np.ndarray) -> float:
    """Fast max‑convolution (used as a cheap proxy for the score)."""
    return np.max(np.convolve(seq, seq))


# ----------------------------------------------------------------------
# Linear‑programming step (the core improvement)
# ----------------------------------------------------------------------


def _lp_step(a: np.ndarray, t: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Perform one LP‑based balancing step.

    Parameters
    ----------
    a : np.ndarray
        Current non‑negative sequence (ℓ¹‑norm = √(2n)).
    t : float
        Mixing factor – the new sequence becomes (1‑t)·a + t·g.

    Returns
    -------
    new_a : np.ndarray
        Updated sequence (renormalised to the target ℓ¹‑norm).
    new_M : float
        max(conv(new_a,new_a)).
    """
    if not _SCIPY_AVAILABLE:          # pragma: no cover
        # SciPy not available → do nothing
        return a, np.max(np.convolve(a, a))

    n = a.shape[0]
    rows = 2 * n - 1                     # length of full convolution
    # Build the (rows × n) matrix that maps g → a * g
    # Each column i contributes a_i on rows i … i+n‑1
    A = _sp.lil_matrix((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = a[i]
    A = A.tocsr()

    # Constraint RHS : the current maximal convolution value
    M = np.max(np.convolve(a, a))
    b = np.full(rows, M, dtype=float)

    # Objective: maximise Σ g_j   → minimise –Σ g_j
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n

    # Solve the LP (highs is fast and handles sparse matrices)
    res = _opt.linprog(
        c, A_ub=A, b_ub=b, bounds=bounds,
        method='highs', options={'presolve': True}
    )

    if not res.success:                # pragma: no cover
        # In the unlikely event the LP fails, return the unchanged sequence
        return a, M

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:                   # pragma: no cover
        return a, M

    # Rescale g to the required ℓ¹‑norm √(2n)
    S = _target_sum(n)
    g = g * (S / sum_g)

    # Mix and renormalise
    new_a = (1.0 - t) * a + t * g
    new_a = new_a * (S / new_a.sum())

    # New maximal convolution (this is exactly the score after renorm)
    new_M = np.max(np.convolve(new_a, new_a))
    return new_a, new_M


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (used for polishing)
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray,
                max_iter: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Lightweight hill‑climbing: perturb a single coordinate by a Gaussian
    step, renormalise to keep the ℓ¹‑norm fixed and keep the move if the
    score improves.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    # initial step size – a modest fraction of the mean entry
    step_sigma = max(1e-8, seq.mean() * 0.2)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        idx = np.random.randint(n)
        delta = np.random.randn() * step_sigma
        old_val = seq[idx]
        new_val = old_val + delta
        if new_val < 0.0:
            new_val = 0.0
        seq[idx] = new_val

        # renormalise (target sum = √(2n))
        S = _target_sum(n)
        seq = seq * (S / seq.sum())

        cand_score = _evaluate(seq)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = seq.copy()
        else:
            # reject the move
            seq[idx] = old_val
            seq = seq * (S / seq.sum())

        # exponential decay of the step size
        step_sigma *= 0.9995

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.

    The routine performs many random restarts.  For each restart it
    creates an analytic sqrt‑derivative shape, runs a few LP‑balancing
    iterations, and finally applies a short hill‑climbing polish.
    The best sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer loop
    deadline = start_time + max(0.0, budget_s - 5.0)

    best_seq = None
    best_score = float("inf")

    # Reasonable lengths – a few hundred up to a few thousand items
    possible_lengths = list(range(500, 2100, 200))

    while time.time() < deadline:
        n = int(np.random.choice(possible_lengths))
        delta = np.random.rand() * 0.8               # δ ∈ [0,0.8)

        # ------------------------------------------------------------------
        # 1) analytic start
        # ------------------------------------------------------------------
        seq = _analytic_sqrt_sequence(n, delta=delta)
        cur_score = _evaluate(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[init ] n={n:4d} δ={delta:.3f} score={cur_score:.6f}")

        # ------------------------------------------------------------------
        # 2) a few LP balancing steps
        # ------------------------------------------------------------------
        max_lp_iters = 10
        t_mix = 0.6
        for lp_it in range(max_lp_iters):
            if time.time() >= deadline:
                break
            seq, new_score = _lp_step(seq, t=t_mix)
            if new_score < best_score:
                best_score = new_score
                best_seq = seq.copy()
                print(f"[LP   ] n={n:4d} it={lp_it+1:2d} score={new_score:.6f}")
            # stop early if the improvement stalls
            if abs(new_score - cur_score) < 1e-5:
                break
            cur_score = new_score

        # ------------------------------------------------------------------
        # 3) short hill‑climbing refinement (≈0.2 s per restart)
        # ------------------------------------------------------------------
        inner_deadline = time.time() + 0.2
        seq_opt, score_opt = _hill_climb(seq,
                                         max_iter=2000,
                                         deadline=inner_deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[hclim] n={n:4d} score={score_opt:.6f}")

        # optional early exit if we have already reached a very good bound
        if best_score < 1.505:
            # keep searching a little longer, but we have essentially hit the target
            pass

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1000
        best_seq = np.ones(n_fallback) * (_target_sum(n_fallback) / n_fallback)

    # Final reporting
    print(f"Best score achieved: {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
