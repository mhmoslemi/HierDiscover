def _smooth_descent(seq, deadline, n_iters_max, eta0, alpha0):
    seq = seq.copy()
    n = len(seq)
    best_seq = seq.copy()
    best_score = _compute_score(seq)
    eta = eta0
    alpha = alpha0
    for it in range(n_iters_max):
        if time.time() >= deadline:
            break
        # compute conv via FFT for speed
        conv = np.convolve(seq, seq)  # small n; or np.fft?
        max_conv = conv.max()
        # compute softmax weights
        c_shift = conv - max_conv
        exp_c = np.exp(alpha * c_shift)
        w = exp_c / exp_c.sum()
        # gradient via correlation
        corr = np.correlate(w, seq, mode='full')
        grad = 2 * corr[n-1:n-1+n]  # length n
        # gradient step
        new_seq = seq - eta * grad
        # project onto simplex
        new_seq = _project_simplex(new_seq)
        # compute new score
        new_score = _compute_score(new_seq)
        if new_score < best_score:

            best_seq = new_seq.copy()
        # acceptance
        if new_score <= _compute_score(seq):

    best_seq = seq.copy()
    best_score = _compute_score(seq)
    eta = eta0
    alpha = alpha0

        if time.time() >= deadline:
            break

"""
Propose a step‑function sequence with a very small
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm consists of three phases:

1.   **Linear‑programming (LP) shaping** –  A short sequence (≈256 elements)
     is repeatedly improved by solving a linear program that maximises the
     ℓ₁‑norm of a new vector `b` under the constraint `conv(a,b) ≤ max(conv(a,a))`.
     Mixing the current vector with the normalised LP solution pushes the
     distribution towards shapes with a flatter auto‑convolution.

2.   **Upscaling** –  The best short vector is interpolated to longer lengths
     (800, 1600, 3200).  The result is re‑normalised to sum 1.

3.   **Smoothed‑max gradient descent** –  For each length we minimise a
     smooth approximation of the maximal convolution entry:
        φ(a)= (1/α)·log Σ exp(α·(a*a)).
     The gradient is obtained as a correlation `corr(w,a)` where `w` are the
     soft‑max weights.  After each step we project back onto the probability
     simplex.  Step size η decays slowly, the smoothing parameter α grows
     (approaching the true max), and occasional random jitter helps to escape
     plateaus.

All operations respect the required non‑negativity and ℓ₁‑normalisation,
and the whole search stops well before the 1000 s budget.
"""

import time
import numpy as np
import scipy.sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x | x_i ≥ 0, sum x_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full length)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT (fast for large n)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << int(np.ceil(np.log2(m)))          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # Numerical noise may produce tiny negatives; clip for safety.
    conv = np.clip(conv, 0.0, None)
    return conv


# ----------------------------------------------------------------------
# Scoring (the objective is invariant under scaling – we fix sum = 1)
# ----------------------------------------------------------------------
def _compute_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (used for initial guesses)
# ----------------------------------------------------------------------
def _powerlaw_sequence(n: int, beta: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-beta)
    seq /= seq.sum()
    return seq


def _exponential_sequence(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Resizing / interpolation helper
# ----------------------------------------------------------------------
def _resize(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to length new_n."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = _project_simplex(new_seq)          # keep it a proper simplex point
    return new_seq


# ----------------------------------------------------------------------
# Linear‑programming improvement step (small n)
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray, t: float = 0.5) -> np.ndarray:
    """
    One LP iteration:
      maximise Σ b_j  subject to  b ≥ 0  and  conv(a,b) ≤ M,
    where M = max(conv(a,a)).  The solution b is normalised to sum 1
    and mixed with a using coefficient t.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    M = conv.max()

    # Build the Toeplitz‑style convolution matrix A (2n‑1 × n)
    # Row k contains a_{k‑j} for column j (0‑based).  Implemented via diagonals.
    diag_data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)                           # 0 … n‑1
    A = scipy.sparse.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')

    # Linear program: maximise Σ b  ⇔  minimise –Σ b
    c = -np.ones(n, dtype=np.float64)                # objective coefficients
    bounds = [(0.0, None)] * n                       # b_j ≥ 0
    b_ub = np.full(2 * n - 1, M)                     # RHS of A @ b ≤ M

    # Solve with HiGHS (fast interior‑point method; works with sparse A)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})

    if not res.success or res.x is None:
        # LP failed – simply return the original vector
        return a

    b_opt = np.maximum(res.x, 0.0)                  # protect against tiny negatives
    s = b_opt.sum()
    if s <= 0.0:
        return a
    g = b_opt / s                                    # normalise to sum 1

    # Mix and re‑project onto the simplex (keeps sum 1, non‑negativity)
    a_new = (1.0 - t) * a + t * g
    a_new = _project_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (large n)
# ----------------------------------------------------------------------
def _smooth_descent(init_seq: np.ndarray,
                    deadline: float,
                    max_iters: int = 30000,
                    eta0: float = 0.5,
                    alpha0: float = 10.0) -> (np.ndarray, float):
    """
    Minimises a smooth approximation of max(conv) using projected gradient
    descent.  The smoothing parameter alpha grows slowly (approaching the true
    max), while the step size eta decays.
    """
    seq = init_seq.copy()
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _compute_score(seq)

    eta = eta0 / n          # scale step size with length
    alpha = alpha0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(seq)
        max_c = conv.max()

        # ----- soft‑max weights (smooth approximation of the max) -----
        shifted = conv - max_c            # ≤ 0, improves numerical stability
        # prevent under‑flow for very large alpha
        exp_vals = np.exp(alpha * shifted)
        w = exp_vals / exp_vals.sum()    # shape (2n‑1,)

        # ----- gradient of the smoothed objective -----
        # grad_i = 2 * Σ_k w_k * a_{k-i}   (indices out of range give 0)
        corr = np.correlate(w, seq, mode='full')   # length 3n‑2
        grad = 2.0 * corr[n - 1:n - 1 + n]

        # ----- gradient step + projection onto the simplex -----
        new_seq = seq - eta * grad
        new_seq = _project_simplex(new_seq)

        # ----- acceptance test -----
        new_score = _compute_score(new_seq)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"[gd] it={it:,d}  new best {best_score:.6f}")

        # simple greedy acceptance; occasional stochastic move
        if new_score <= _compute_score(seq) or np.random.rand() < 0.005:
            seq = new_seq

        # ----- schedule updates -----
        eta *= 0.9997                     # slowly decay step size
        alpha = min(alpha * 1.001, 200.0) # slowly tighten the approximation

        # occasional jitter to escape flat plateaus
        if (it + 1) % 1000 == 0:
            jitter = (np.random.rand(n) - 0.5) * eta * 0.2
            seq = _project_simplex(np.maximum(seq + jitter, 0.0))

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that attempts to minimise the
    provided evaluate_sequence function.  The routine respects the 1000 s
    time budget and logs intermediate improvements.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # small safety margin

    # ------------------------------------------------------------------
    # Initialise from the best known construction if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        current = np.array(height_sequence_1, dtype=float)
        current = _project_simplex(current)          # ensure proper normalisation
    else:
        # a modest uniform start (length 500)
        current = np.full(500, 1.0 / 500)

    best_seq = current.copy()
    best_score = _compute_score(best_seq)
    print(f"[init] score {best_score:.6f}  length {len(best_seq)}")

    # ------------------------------------------------------------------
    # Phase 1 – LP shaping on a short vector (≈256 entries)
    # ------------------------------------------------------------------
    n_small = 256
    seq_small = _resize(best_seq, n_small)

    for lp_it in range(12):
        if time.time() >= deadline:
            break
        seq_small = _lp_improve(seq_small, t=0.5)
        score_small = _compute_score(seq_small)
        if score_small < best_score:
            best_score = score_small
            best_seq = seq_small.copy()
            print(f"[LP] iter {lp_it+1:2d}  new best {best_score:.6f}  n={n_small}")

    # ------------------------------------------------------------------
    # Phase 2 – Upscale and smooth‑max gradient descent
    # ------------------------------------------------------------------
    lengths = [800, 1600, 3200]          # progressively larger problems
    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise from the best short vector (interpolation)
        seq = _resize(best_seq, n)

        # add a tiny random perturbation to diversify the start
        jitter = (np.random.rand(n) - 0.5) * 1e-3
        seq = _project_simplex(seq + jitter)

        # run the smoothed‑max descent
        seq, score = _smooth_descent(seq,
                                     deadline,
                                     max_iters=40000,
                                     eta0=0.8,
                                     alpha0=12.0)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[gd] after length {n:4d}  new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing beat the initial construction
    # ------------------------------------------------------------------
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(500, 1.0 / 500)

    print(f"Final best score: {best_score:.6f}  length {len(best_seq)}")
    return best_seq.tolist()
