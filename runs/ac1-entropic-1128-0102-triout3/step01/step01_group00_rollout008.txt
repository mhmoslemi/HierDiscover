"""
Project‑gradient (smooth max) optimizer for the step‑function attack.

The evaluation metric is
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

The metric is scale‑invariant, so we keep the sequence on the probability
simplex (sum = 1, all entries ≥ 0).  The algorithm is a projected (sub‑)gradient
descent with a *smoothed* max objective:

    φβ(a) = (1/β)·log( Σk exp( β·convk(a) ) )
    ∇φβ(a) = 2 · ( w  *  a_rev )[(n‑1)…(2n‑2)]      (correlation)

where w_k = exp(β·(convk − max(conv))) and a_rev is a reversed copy of a.
For β → ∞ the weights concentrate at the maximal convolution index and the
gradient reduces to the simple sub‑gradient used in the reference solution.
We start with a small β (smooth landscape) and increase it gradually,
allowing the optimizer to escape shallow plateaus and later polish the
solution.

Each iteration:
    * compute a*a by FFT,
    * evaluate the current score,
    * build the weight vector w,
    * obtain the gradient by another FFT‑based correlation,
    * take a diminishing step (step0 / √t) and
    * project back onto the simplex (Duchi et al. 2008).

Multiple sequence lengths and random restarts are tried; the best sequence
found before the time budget (≈ 1000 s) is returned.
"""

import time
import math
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum(x) = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_auto_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear auto‑convolution a*a using FFT.
    Returns the first 2·len(a)‑1 entries.
    """
    Af = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Af * Af, n=fft_len)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_via_weights(a: np.ndarray,
                          conv: np.ndarray,
                          beta: float,
                          grad_fft_len: int) -> np.ndarray:
    """
    Compute the smoothed gradient of the max‑convolution objective.
    Uses weights w_k = exp(beta·(conv_k – max(conv))) and the
    correlation w ⋆ a (linear, not circular).
    """
    n = a.shape[0]
    max_conv = conv.max()
    # Numerical stability: shift by max, so the largest exponent is 1.
    w = np.exp(beta * (conv - max_conv))

    # Correlation w ⋆ a = (w * a_rev)  (linear convolution)
    a_rev = a[::-1]
    F_w = np.fft.rfft(w, n=grad_fft_len)
    F_a_rev = np.fft.rfft(a_rev, n=grad_fft_len)
    corr = np.fft.irfft(F_w * F_a_rev, n=grad_fft_len)
    # Slice the part that corresponds to indices 0 … n‑1
    grad = 2.0 * corr[(n - 1) : (n - 1 + n)]
    return grad


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that
    attains a low value of ``evaluate_sequence``.
    The algorithm runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    best_score = float('inf')
    best_seq = None

    # ------------------------------------------------------------------
    #   Candidate sequence lengths (larger n → lower theoretical bound)
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400, 3000]

    # If the previous run stored a sequence in the global variable
    # ``height_sequence_1`` we also try its length.
    if 'height_sequence_1' in globals():
        hs = globals()['height_sequence_1']
        if isinstance(hs, (list, tuple, np.ndarray)):
            n_hs = len(hs)
            if n_hs > 0:
                lengths.append(n_hs)

    np.random.shuffle(lengths)

    # ------------------------------------------------------------------
    #   Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    step0 = 0.20               # base step size (tuned empirically)
    beta_start = 0.5           # start temperature (smoother max)
    beta_max = 80.0            # final temperature (sharp max)
    max_iter_per_restart = 25000
    restarts_per_n = 3

    for n in lengths:
        if time.time() >= deadline:
            break

        # FFT lengths (next power of two for speed)
        conv_fft_len = 1 << ((2 * n - 1).bit_length())
        grad_fft_len = 1 << ((3 * n - 2).bit_length())

        # ------------------------------------------------------------------
        #   Multiple restarts for robustness
        # ------------------------------------------------------------------
        for restart in range(restarts_per_n):
            if time.time() >= deadline:
                break

            # --------- initialise a -------------------------------------------------
            if restart == 0 and 'height_sequence_1' in globals():
                # Warm‑start from the previous best (if length matches)
                hs = globals()['height_sequence_1']
                if len(hs) == n:
                    a = np.array(hs, dtype=float)
                else:
                    # truncate or pad with zeros to the desired size
                    a = np.zeros(n, dtype=float)
                    copy_len = min(len(hs), n)
                    a[:copy_len] = hs[:copy_len]
            elif restart == 0:
                # First restart: smooth sine shape (often a good basin)
                idx = np.arange(n)
                a = np.sin(np.pi * (idx + 0.5) / n)          # positive values
                a = a / a.sum()
            else:
                # Random starting point on the simplex
                a = np.random.rand(n)
                a = a / a.sum()

            # Simple momentum (optional, can be set to 0.0)
            momentum = 0.0
            a_prev = a.copy()

            # Annealing schedule for β (geometric increase)
            beta = beta_start
            if max_iter_per_restart > 1:
                beta_factor = (beta_max / beta_start) ** (1.0 / max_iter_per_restart)
            else:
                beta_factor = 1.0

            # ------------------------------------------------------------------
            #   Main optimisation loop
            # ------------------------------------------------------------------
            it = 0
            while time.time() < deadline and it < max_iter_per_restart:
                it += 1

                # ----- auto‑convolution a*a (linear) -----
                conv = _conv_auto_fft(a, conv_fft_len)

                # ----- current objective (sum(a) == 1) -----
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] "
                          f"New best: n={n}, restart={restart}, iter={it}, "
                          f"score={cur_score:.6f}")

                # ----- gradient of the smoothed max -----
                grad = _gradient_via_weights(a, conv, beta, grad_fft_len)

                # ----- step size (diminishing) -----
                step = step0 / math.sqrt(it)

                # ----- momentum update (if any) -----
                if momentum > 0.0:
                    a_tmp = a - step * grad
                    a = (1 - momentum) * a_tmp + momentum * a_prev
                else:
                    a_tmp = a - step * grad
                    a = a_tmp

                # ----- project back onto the probability simplex -----
                a = _project_simplex(a)

                # ----- store for momentum -----
                a_prev = a_tmp.copy()

                # ----- temperature schedule -----
                beta = min(beta_max, beta * beta_factor)

                # ----- occasional early exit if no improvement for long -----
                # (simple heuristic: stop after many cheap iterations)
                if it % 5000 == 0 and cur_score > best_score * 1.001:
                    # No substantial improvement in the last block
                    break

            # end of while‑loop for this restart
        # end of restarts for this n
    # end of lengths loop

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_default = 200
        best_seq = np.full(n_default, 1.0 / n_default)

    # Return a plain Python list (the evaluator accepts any sequence of floats)
    return [float(x) for x in best_seq]
