def convolve_fft(x, y):
    n = x.shape[0] + y.shape[0] - 1
    m = 1 << (n-1).bit_length()
    X = np.fft.rfft(x, n=m)
    Y = np.fft.rfft(y, n=m)
    conv = np.fft.irfft(X * Y, n=m)
    return conv[:n]

"""
Optimiser for the convolution‐based evaluation.

The code builds a probability vector `a` (sum = 1, a_i ≥ 0) and
minimises the quantity

        2·n·max_k (a * a)[k]

where `*` denotes discrete convolution.  The method is a
projected Adam optimiser that minimises a *smooth* approximation of
the max‑operator: the log‑sum‑exp (soft‑max) of the convolution
values.  The temperature (β) of the soft‑max is increased during the
run so that early iterations see a smooth landscape and later ones
focus on the true maximum.

All updates stay on the probability simplex by a cheap projection
(clip negatives → renormalise).  Progress is printed and the best
sequence found before the time limit is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑z)^{‑1/2}:
        a_i = C(2i,i) / 4**i,
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Simple (non‑Euclidean) projection onto the probability simplex:
        clip to non‑negative and renormalise so that sum = 1.
    This is cheap (O(n)) and sufficient for the optimisation.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute a * a (self‑convolution) using real FFT.
    Result length = 2*len(a)‑1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # power‑of‑two size for speed
    N = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:m]
    return conv


def conv_two_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolution of two (possibly different) real vectors using FFT.
    Length = len(x) + len(y) – 1.
    """
    m = x.shape[0] + y.shape[0] - 1
    N = 1 << (m - 1).bit_length()
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:m]
    return conv

def propose_candidate(seed: int = 1234,
                     budget_s: float = 950.0,
                     n: int = 2500) -> list[float]:
    """
    Search for a non‑negative sequence that minimises
        2·n·max(a*a) / (sum a)² .

    A soft‑max (log‑sum‑exp) surrogate for the max is optimised with
    Adam on the probability simplex.  The temperature β of the
    soft‑max grows during the run, turning the surrogate gradually
    into the true max.  The best sequence encountered before the time
    limit is returned.
    """
    rng = np.random.default_rng(seed)

    # ---- initialise ----------------------------------------------------
    a = sqrt_series(n)                     # half‑binomial seed, already sums to 1
    a += rng.random(n) * 1e-6              # tiny jitter to break exact symmetry
    a = project_onto_simplex(a)

    # Adam hyper‑parameters
    lr = 0.03                               # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1_pow = beta1
    b2_pow = beta2

    # soft‑max temperature schedule
    beta_temp = 1.0                         # start temperature
    beta_mul = 1.003                        # multiplicative increase per iteration
    beta_max = 15000.0

    # keep the best ever found
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    deadline = time.time() + budget_s - 5.0   # safety margin
    it = 0
    max_iter = 5_000_000

    while time.time() < deadline and it < max_iter:
        # ----- current objective ----------------------------------------
        conv = conv_self_fft(a)               # length 2n‑1
        max_val = conv.max()
        cur_score = 2.0 * n * max_val         # because sum(a)==1

        # record best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[iter {it:07d}] new best score = {best_score:.6f}")

        beta_temp = min(beta_max, beta_temp * beta_mul)
        shift = conv.max()                    # for numerical stability
        exp_vals = np.exp(beta_temp * (conv - shift))
        w = exp_vals / exp_vals.sum()         # size 2n‑1, sums to 1

        # ----- gradient of the smooth objective -------------------------
        # term 1: Σ_k w_k a_{k‑i}
        a_rev = a[::-1]
        grad1_full = conv_two_fft(w, a_rev)   # length 3n‑2
        grad1 = grad1_full[n-1 : n-1 + n]      # slice of length n

        pref = np.concatenate(([0.0], np.cumsum(w)))   # length 2n
