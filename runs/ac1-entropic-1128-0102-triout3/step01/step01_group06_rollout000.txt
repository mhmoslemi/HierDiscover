"""
Gradient‑descent + simplex projection search for low‑autocorrelation step
functions.

The algorithm works as follows:

1. Initialise a candidate sequence.
   * If the global variable ``height_sequence_1`` exists we start from it,
     otherwise we use a uniform distribution.
2. Normalise the candidate to have unit sum (the objective
   `evaluate_sequence` is scale‑invariant, so this is safe).
3. Repeatedly perform a *gradient‑descent step* on the index where the
   self‑convolution attains its maximum:
   * For the current maximal lag `k` the gradient of the maximal
     autocorrelation w.r.t. the coefficients is
     `g[i] = 2·a[k‑i]` (zero outside the valid range).
   * Move a small amount `eta·g` opposite to the gradient and project the
     result back onto the probability simplex (non‑negative, sum = 1).
4. If the gradient step does not improve the score we try a cheap random
   two‑index mass transfer (the hill‑climbing move used previously).
5. Occasionally a random restart from a power‑law family
   `a_i ∝ (i+c)^(-α)` is performed, and with low probability an
   exponentiated‑gradient step is tried.
6. The best sequence found before the time budget expires is returned.

All heavy work (FFT‑based convolution, gradient computation,
simplex projection) is performed with NumPy in compiled code,
allowing thousands of iterations within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _fft_max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Compute max value of the discrete self‑convolution a*a using FFT.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def _score_from_array(a: np.ndarray, fft_len: int) -> float:
    """
    Score used during the search.
    The sequence `a` is assumed to have unit sum, therefore the score is
    exactly the value returned by `evaluate_sequence` for the same `a`.
    """
    n = len(a)
    max_conv = _fft_max_convolution(a, fft_len)
    return 2.0 * n * max_conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i >= 0, sum_i x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform vector.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1          # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_step(a: np.ndarray, n: int, fft_len: int, eta: float):
    """
    One gradient‑descent step on the index of maximal autocorrelation.
    Returns the updated vector and the lag where the maximum was found.
    """
    # FFT‑based self‑convolution
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]

    k = int(np.argmax(conv))

    # Gradient g_i = 2 * a_{k-i} for indices that actually contribute.
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1) + 1          # exclusive
    idx = np.arange(i_start, i_end)
    partner = k - idx
    g[idx] = 2.0 * a[partner]

    # Descent step + projection onto simplex
    a_new = proj_simplex(a - eta * g)
    return a_new, k


def exponentiated_gradient_step(a: np.ndarray, n: int, fft_len: int, eta: float):
    """
    Multiplicative (exponentiated) gradient step on the maximal lag.
    Guarantees positivity, then re‑normalises to sum = 1.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    k = int(np.argmax(conv))

    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1) + 1
    idx = np.arange(i_start, i_end)
    partner = k - idx
    g[idx] = 2.0 * a[partner]

    upd = np.exp(-eta * g)
    a_new = a * upd
    a_new = a_new / a_new.sum()      # simple renormalisation (all >0)
    return a_new


def build_candidate(alpha: float, c: float, n: int) -> np.ndarray:
    """
    Power‑law family a_i = (i + c)^(-alpha), normalised to unit sum.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + c) ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a / s

                print(f"[restart] α={alpha:.3f} c={c:.1f} score={sc4:.6f}")

        # ---- Adaptive step‑size control when stagnating ----
        if time.time() - last_improve > 20.0:
            eta = min(eta * 2.0, 0.2)   # become more aggressive
            last_improve = time.time()

        it += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence found
    # ------------------------------------------------------------------
    return best_a.tolist()
