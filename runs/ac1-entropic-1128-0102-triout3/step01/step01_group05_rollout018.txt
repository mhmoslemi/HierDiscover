def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 4000 (maybe 5000 if time allowed)

    # Parametric search
    best_score = np.inf
    best_a = None
    p_vals = np.arange(0.4,0.61,0.01) # e.g., around 0.5
    offsets = np.arange(0.0,2.01,0.2)
    for p in p_vals:
        for off in offsets:
            idx = np.arange(1, n+1, dtype=np.float64) + off
            a = idx ** (-p)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a = a / a_sum
            # compute fast_score (or direct conv)
            val = fast_score(a)
            if val < best_score:
                best_score = val
                best_a = a.copy()
    print(f"[init] best val param search {best_score:.6f}")

    # Now hill climbing with incremental conv updates
    a = best_a
    conv = np.convolve(a,a)
    cur_max = conv.max()
    best_max = cur_max
    best_seq = a.copy()
    step = 0.5
    step_min = 1e-5
    temperature = 0.01 # scaling factor for acceptance
    iter = 0
    accept_window = 0
    accepted = 0
    while time.time() < deadline:
        # sample i weighted by a (high prob)
        i = np.random.choice(n, p=a)
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        # ensure not same
        if i == j:
            continue
        a_i = a[i]
        if a_i < 1e-15:
            continue
        delta = step * a_i * np.random.rand()
        if delta <= 0 or delta > a_i:
            continue

        # compute new conv using slice updates
        new_conv = conv.copy()
        # Using original a vector
        factor = 2.0 * delta
        new_conv[i:i+n] -= factor * a
        new_conv[j:j+n] += factor * a
        new_max = new_conv.max()
        diff = new_max - cur_max

        accept = False
        if diff < -1e-15: # improvement
            accept = True
        else:
            # random uphill
            prob = np.exp(-diff / (temperature * cur_max))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            accepted += 1

            # record best
            if cur_max < best_max:
                best_max = cur_max
                best_seq = a.copy()
                best_val = 2 * n * best_max
                print(f"[iter {iter}] new best {best_val:.6f}")

        # adapt step & temp
        iter += 1
        if iter % 10000 == 0:
            # decaying step
            step = max(step * 0.95, step_min)
            temperature = max(temperature * 0.95, 1e-6)

        if iter % 20000 == 0:
            # recompute conv from scratch for stability
            conv = np.convolve(a, a)
            cur_max = conv.max()
            # maybe recompute best if current best changed
        if iter % 5000 == 0:
            cur_val = 2 * n * cur_max
            best_val = 2 * n * best_max
            print(f"[iter {iter}] cur={cur_val:.6f}, best={best_val:.6f}, step={step:.6f}")

    final_val = 2 * n * best_max
    print(f"[final] best value {final_val:.6f}")
    return best_seq.tolist()

"""
propose_candidate()
------------------

Search for a non‑negative step‑function (list of heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv(a,a)` is the linear self‑convolution.
The algorithm consists of three stages:

1. **Parametric seed** –  a small grid search over the family  
       a_i ∝ (i+1+offset)^{-p}   (p≈0.5, offset≈0)   
   using a fast FFT based scorer.  The best seed is taken as the start
   point.

2. **Linear‑programming improvement** –  following the “balancing”
   technique from the reference: keep the current maximum `M`,
   maximise the ℓ¹‑norm of a vector `b` under the linear constraints
   `conv(a,b) ≤ M`.  If `sum(b) > sum(a)` we can scale `b` back to the
   original ℓ¹‑norm and mix it with `a`.  Each LP iteration strictly
   reduces the peak value `M`.  The LP is solved with CVXPY/ECOS;
   the step is repeated while it yields progress and time permits.

3. **Fine‑grained hill‑climbing** –  many cheap mass‑transfer moves are
   performed on the full‑size vector.  The convolution is updated
   incrementally (O(n) slice operations) so millions of iterations fit
   comfortably into the 1000 s budget.  Moves are accepted when they
   lower the current peak; a tiny simulated‑annealing probability lets
   the search escape plateaus.  The best sequence encountered is
   returned.

All numbers are kept non‑negative and the total ℓ¹‑norm is preserved,
so the final sequence can be fed directly to the provided
`evaluate_sequence` routine.

The implementation uses only NumPy and (optionally) CVXPY; it adapts
its runtime to the remaining time budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast scorer based on FFT (same formula as evaluate_sequence)
def fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT."""
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # length for a circular convolution that contains the linear one
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise the
    evaluation function.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # 1. Choose the length of the vector.
    #    2500‑3000 elements give a good trade‑off between quality and speed.
    n = 2800

    # --------------------------------------------------------------
    # 2. Light parametric sweep to obtain a good seed.
    #    a_i = (i+1+off)^{-p}   (p≈0.5).  Scan a modest grid.
    best_val = np.inf
    best_a = None

    p_vals = np.linspace(0.40, 0.60, 21)          # 0.40 … 0.60
    off_vals = np.linspace(0.0, 2.0, 21)          # 0 … 2

    print("[seed] scanning parametrised family …")
    for p in p_vals:
        for off in off_vals:
            idx = np.arange(1, n + 1, dtype=np.float64) + off
            a = idx ** (-p)
            a_sum = a.sum()
            if a_sum == 0.0:
                continue
            a = a / a_sum                         # ℓ¹‑norm = 1
            val = fast_score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    print(f"[seed] best after sweep: {best_val:.6f}")

    # If a global variable `height_sequence_1` exists, try it as a fallback.
    if best_a is None and 'height_sequence_1' in globals():
        cand = np.array(height_sequence_1, dtype=np.float64)
        cand = np.maximum(0.0, cand)
        cand /= cand.sum()
        best_a = cand
        best_val = fast_score(cand)
        print("[seed] using supplied height_sequence_1")

    # --------------------------------------------------------------
    # 3. Optional LP‑based improvement (the “balancing” step).
    #    Requires CVXPY; if unavailable we simply skip it.
    try:
        import cvxpy as cp

        a = best_a.copy()
        conv = np.convolve(a, a)
        M = conv.max()
        best_M = M
        lp_iter = 0

        # run at most 12 LP iterations or until time runs out
        while time.time() < deadline and lp_iter < 12:
            lp_iter += 1
            n = a.size

            # LP: maximise sum(b) subject to b>=0 and conv(a,b) <= M
            b = cp.Variable(n, nonneg=True)
            conv_ab = cp.conv(cp.Constant(a), b)          # length 2n‑1
            prob = cp.Problem(cp.Maximize(cp.sum(b)),
                              [conv_ab <= M])
            # ECOS is a lightweight interior‑point solver already present
            prob.solve(solver=cp.ECOS, warm_start=True,
                       abstol=1e-9, reltol=1e-8, feastol=1e-9,
                       verbose=False)

            if b.value is None:
                break                                        # LP failed

            b_val = np.maximum(0.0, np.array(b.value).flatten())
            sum_b = b_val.sum()
            if sum_b <= 1.0 + 1e-12:                         # no slack
                break

            # scale b back to ℓ¹‑norm = 1 and mix with current a
            scale = 1.0 / sum_b
            g = b_val * scale
            t = 0.2                                          # mixing rate
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(0.0, a_new)
            a_new /= a_new.sum()

            # recompute the peak of the self‑convolution
            conv_new = np.convolve(a_new, a_new)
            M_new = conv_new.max()
            val_new = 2.0 * n * M_new

            print(f"[lp {lp_iter:2d}] score={val_new:.6f}  (M={M_new:.6e})")
            if val_new < best_val - 1e-7:
                best_val = val_new
                best_M = M_new
                best_a = a_new.copy()
                a = a_new
                M = M_new
            else:
                # No improvement – stop the LP phase
                break
    except Exception as exc:        # CVXPY not available or LP failed
        print(f"[lp] skipped (reason: {exc})")
        a = best_a.copy()
        conv = np.convolve(a, a)
        M = conv.max()
        best_M = M

    # --------------------------------------------------------------
    # 4. Fine‑grained hill‑climbing with incremental convolution updates.
    #    Only O(n) work per move, enabling many millions of steps.
    cur_a = a.copy()
    conv = np.convolve(cur_a, cur_a)          # full linear convolution
    cur_M = conv.max()
    cur_val = 2.0 * n * cur_M

    best_seq = cur_a.copy()
    best_M = cur_M
    best_val = cur_val

    # parameters for the stochastic search
    step = 0.5                     # relative amount of mass transferred
    step_min = 1e-5
    temperature = 0.02            # for occasional uphill moves
    it = 0
    accepted = 0

    print("[hill] start hill‑climbing")
    while time.time() < deadline:
        # --- choose a move -------------------------------------------------
        # 30 % of the time we target the current peak, otherwise random
        if np.random.rand() < 0.3:
            # indices that contribute to the current maximum
            k_max = np.argmax(conv)
            contrib = np.zeros(n)
            for i in range(n):
                j = k_max - i
                if 0 <= j < n:
                    contrib[i] = cur_a[i] * cur_a[j]
            # source = index with largest contribution
            i = int(np.argmax(contrib))
            # pick a destination that contributes little to the peak
            low = np.where(contrib < np.median(contrib))[0]
            if low.size == 0:

                if j >= i:
                    j += 1
            else:

        else:

            j = np.random.randint(0, n - 1)
            if j >= i:
                j += 1

        if i == j:

        ai = cur_a[i]
        if ai <= 1e-15:
            continue

        delta = step * ai * np.random.rand()
        if delta <= 0.0 or delta > ai:
            continue

        # --- compute the new convolution using slice updates -------------
        # use the *old* vector cur_a for the linear update
        new_conv = conv.copy()
        factor = 2.0 * delta

        new_M = new_conv.max()

        accept = False
        if diff < -1e-15:                     # improvement

            prob = np.exp(-diff / (temperature * cur_M + 1e-12))
            if np.random.rand() < prob:

        if accept:

            cur_a[i] -= delta
            cur_a[j] += delta
            conv = new_conv
            cur_M = new_M
            cur_val = 2.0 * n * cur_M
            accepted += 1

            # record a new global best
            if cur_M < best_M - 1e-12:
                best_M = cur_M
                best_val = cur_val
                best_seq = cur_a.copy()
                print(f"[iter {it:7d}] NEW BEST {best_val:.6f}")

        # --- adapt parameters ---------------------------------------------
        it += 1
        if it % 20000 == 0:
            # slowly shrink the step size and temperature
            step = max(step * 0.94, step_min)
            temperature = max(temperature * 0.93, 1e-4)

        if it % 50000 == 0:
            # recompute convolution from scratch to counteract numerical drift
            conv = np.convolve(cur_a, cur_a)
            cur_M = conv.max()
            cur_val = 2.0 * n * cur_M

        if it % 10000 == 0:
            print(f"[iter {it:7d}] cur={cur_val:.6f}  best={best_val:.6f}  step={step:.5f}")

    # --------------------------------------------------------------
    # 5. Final clean‑up – enforce non‑negativity and unit ℓ¹‑norm
    best_seq = np.maximum(0.0, best_seq)
    if best_seq.sum() == 0.0:
        best_seq[0] = 1.0
    best_seq /= best_seq.sum()

    final_score = 2.0 * n * np.max(np.convolve(best_seq, best_seq))
    print(f"[final] best score = {final_score:.6f}  (iterations={it})")
    return best_seq.tolist()
