def propose_candidate(seed=42, budget_s=950.0, **kwargs):
    import time, numpy as np, scipy.optimize, scipy.sparse

    rng = np.random.default_rng(seed)

    # Choose length n, maybe 1500 (Half binomial length)
    n = 1500

    a = sqrt_series(n)   # sum =1
    best_a = a.copy()

    print(f"[init] n={n}, initial score={best_score:.6f}")

    start = time.time()
    deadline = start + budget_s - 10

    # parameters
    max_outer_iter = 30   # number of LP iteration cycles
    t = 0.5   # mixing ratio

    iter_count = 0
    while time.time() < deadline and iter_count < max_outer_iter:
        # compute convolution and max
        conv = conv_fft(a)
        M = conv.max()
        cur_score = 2.0 * n * M

        # build LP matrix
        # Build constraint matrix A_ub
        rows = np.empty(n * n, dtype=np.int64)
        cols = np.empty(n * n, dtype=np.int64)
        data = np.empty(n * n, dtype=np.float64)
        idx = 0
        for i in range(n):
            rows[idx:idx+n] = i + np.arange(n)
            cols[idx:idx+n] = np.arange(n)
            data[idx:idx+n] = a[i]        # f_i

        solve_time = time.time() - lp_start

        h = res.x
        sum_h = h.sum()

"""
propose_candidate
=================

A two‑stage optimizer for the convolution‑based bound.

Stage 1 – initialise with the classic half‑binomial (the “square‑root of
(1‑z)”) sequence.  It already yields the asymptotic constant π/2 ≈ 1.5708.

Stage 2 – repeatedly solve a small linear programme

        maximise   1·h                     (increase the total mass)
        subject to  f * h ≤ M               (point‑wise convolution bound)
                    h ≥ 0

where ``f`` is the current sequence, ``M = max(f*f)`` and ``*`` denotes the
discrete convolution.  The LP produces a promising direction ``h`` that can
carry a lot of probability mass without violating the current peak‐height
``M``.  After normalising ``h`` to unit L¹‑norm we mix it with the old
sequence (parameter ``mix``) and project back onto the probability simplex.
The mixing and a simple temperature schedule guarantee a monotone
decrease (or very rare uphill move) of the objective

        score = 2·n·max(f*f)      (= evaluate_sequence(f) for sum(f)=1)

The LP is built with a sparse matrix (O(n²) non‑zeros, n≈1500 gives ~2 Mi
entries ≈ 50 MiB) and solved with SciPy’s HiGHS interior‑point solver,
which can handle a few dozen iterations well within the 1000 s budget.

The best sequence found during the run is returned.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First *n* coefficients of a_i = C(2i,i)/4ⁱ  (the (1‑z)^‑½ series),
    normalised to L¹‑norm 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto {x | x≥0, sum(x)=1}.
    Duchi‑et‑al. (2008) implementation.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero after projection – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Fast convolution x * x using real FFT.
    Length of result is 2·len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    return np.fft.irfft(X * X, n=m)


def build_conv_constraint_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Return a sparse matrix A (shape (2n‑1, n)) such that
        (A @ h)[k]  ==  (f * h)[k]   for all k,
    i.e. each row encodes one convolution inequality.
    """
    n = f.shape[0]
    # repeat the index vectors
    i_idx = np.repeat(np.arange(n, dtype=np.int64), n)          # shape (n*n,)
    j_idx = np.tile(np.arange(n, dtype=np.int64), n)            # shape (n*n,)
    rows = i_idx + j_idx                                        # 0 … 2n‑2
    cols = j_idx
    data = np.repeat(f, n)                                      # f_i repeated n times
    return scipy.sparse.csr_matrix((data, (rows, cols)),
                                   shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a linear‑programming driven refinement of the half‑binomial seed.
    Returns the best sequence (list of floats) found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------ parameters -------------------------------
    n = 1500                     # length of the sequence (adjustable)
    max_outer_iter = 30          # LP+mix iterations (kept modest for safety)
    mix = 0.5                    # convex combination weight for the new direction
    temperature_start = 0.04     # for occasional uphill acceptance
    decay_factor = 0.92          # geometric decay per outer iteration

    # ------------------------ initial sequence -------------------------
    f = sqrt_series(n)           # already sums to 1 and is non‑negative
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------ main optimisation loop -------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin
    temperature = temperature_start

    for it in range(max_outer_iter):
        now = time.time()
        if now >= deadline:
            print("[time] budget exhausted, stopping early")
            break

        # current peak of the self‑convolution
        conv_f = conv_fft(f)
        M = conv_f.max()
        cur_score = 2.0 * n * M          # identical to evaluate_sequence for sum=1

        # ---------- build linear programme (maximise total mass) ----------
        A = build_conv_constraint_matrix(f)          # shape (2n‑1, n)
        b = np.full(2 * n - 1, M, dtype=np.float64) # right‑hand side vector
        c = -np.ones(n, dtype=np.float64)            # minimise –sum(h)  ⇒ maximise sum(h)

        # solve with HiGHS (fast interior‑point).  No explicit upper bounds.
        lp_res = scipy.optimize.linprog(c,
                                        A_ub=A,
                                        b_ub=b,
                                        bounds=[(0, None)] * n,
                                        method='highs',
                                        options={"presolve": True})

        if not lp_res.success:
            print(f"[iter {it}] LP failed: {lp_res.message}")
            # fall back to a tiny random perturbation and continue
            eps = 1e-6 * rng.random(n)
            f = project_onto_simplex(np.maximum(f - eps, 0.0))
            continue

        h = lp_res.x
        # Numerical noise can produce tiny negative entries – truncate safely.
        h = np.maximum(h, 0.0)

        sum_h = h.sum()
        if sum_h <= 0.0:
            print(f"[iter {it}] LP returned zero total mass – aborting.")
            break

        # normalise the LP solution to unit L¹‑norm
        h_norm = h / sum_h

        # ---- mixing ----------------------------------------------------
        # New candidate = (1‑mix)·old + mix·direction, then project onto simplex.
        cand = (1.0 - mix) * f + mix * h_norm
        cand = project_onto_simplex(cand)

        # evaluate the candidate (exact evaluation, protects against rounding)
        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_f = cand.copy()
            print(f"[iter {it:02d}] new best = {best_score:.6f}")

        # accept if it improves, otherwise accept with a Metropolis rule
        if cand_score <= cur_score:
            f = cand
            cur_score = cand_score
        else:
            delta = cand_score - cur_score
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                f = cand
                cur_score = cand_score
            # else keep the old f

        # geometrically decay mixing weight and temperature (optional)
        mix = max(0.05, mix * decay_factor)
        temperature = max(1e-5, temperature * decay_factor)

    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_f.tolist())
    print(f"[done] final reported score = {final_score:.6f} "
          f"(iterations used = {it + 1})")
    return best_f.tolist()
