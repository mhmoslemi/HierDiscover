def run_simulated_annealing(a, budget, seed):
    rng = np.random.default_rng(seed)
    n = len(a)
    best_a = a.copy()
    best_score = 2 * n * conv_fft(a).max()
    cur_a = a.copy()
    cur_score = best_score

    deadline = time.time() + budget
    it = 0
    temp = 0.05
    step_frac = 0.2
    decay_interval = 2000
    while time.time() < deadline:
        # propose move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        ai = cur_a[i]
        if ai <= 0:
            continue
        delta = rng.random() * min(ai, step_frac * ai)
        if delta == 0:
            continue
        trial = cur_a.copy()
        trial[i] -= delta
        trial[j] += delta
        trial = project_onto_simplex(trial)
        conv = conv_fft(trial)
        max_val = conv.max()
        trial_score = 2 * n * max_val
        if trial_score < cur_score or (temp > 0 and rng.random() < np.exp(-(trial_score - cur_score) / temp)):
            cur_a = trial
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_a = trial.copy()
                print(f"[SA iter {it}] new best score = {best_score:.6f}")
        # decay
        if (it + 1) % decay_interval == 0:
            temp = max(temp * 0.90, 1e-6)
            step_frac = max(step_frac * 0.97, 1e-4)
        it += 1
    return best_a, best_score

    # p in (0,1); a_i = (1-p) * p**i, truncated length n, normalized to sum=1

    a = (1 - p) * (p ** i)

"""
Simple stochastic‑annealing optimizer for the autocorrelation bound.

The routine builds a few diverse initial probability vectors
(half‑binomial, geometric, power‑law, Dirichlet and uniform),
then for each start runs a fast simulated‑annealing loop that repeatedly
moves a small amount of mass from one index to another while keeping the
vector on the probability simplex.  The objective is the evaluation
function (2·n·max(conv(a,a))) because the vectors are kept normalised
to sum = 1.  The best sequence encountered over all starts is returned.
All heavy work (FFT‑based convolution) is O(n log n) and fits comfortably
into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic building blocks --------------------------------------------------

def sqrt_series(n: int) -> np.ndarray:
    """
    Half‑binomial (central‑binomial) coefficients:
        a_i = C(2i,i) / 4**i ,  i=0..n-1,
    normalised to sum = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x_i ≥ 0,  Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Autoconvolution x * x using a real FFT.
    Length of the result is 2*len(x)-1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


# ----------------------------------------------------------------------
# Alternative initial distributions ---------------------------------------

def geometric_series(n: int, p: float = 0.99) -> np.ndarray:
    """Geometric distribution (1-p) p^i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - p) * (p ** i)
    a /= a.sum()
    return a


def powerlaw_series(n: int, alpha: float = 0.6) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-alpha}, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha)
    a /= a.sum()
    return a


def dirichlet_series(n: int, conc: float = 0.5, rng: np.random.Generator = None) -> np.ndarray:
    """Draw a random point from a Dirichlet distribution."""
    if rng is None:
        rng = np.random.default_rng()
    x = rng.gamma(conc, 1.0, size=n)
    a = x / x.sum()
    return a


def uniform_series(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Simulated annealing core -----------------------------------------------

def _anneal_one_start(a0: np.ndarray,
                      budget: float,
                      rng: np.random.Generator,
                      n: int) -> tuple[np.ndarray, float]:
    """
    Run a simple simulated‑annealing on a single starting vector.
    Returns (best_vector, best_score).
    """
    # ensure we are on the simplex
    a = project_onto_simplex(a0.copy())
    conv = conv_fft(a)
    cur_score = 2.0 * n * conv.max()       # sum(a) == 1, so denominator = 1
    best_a = a.copy()
    best_score = cur_score

    deadline = time.time() + budget
    it = 0
    temp = 0.08                     # initial temperature
    step_frac = 0.20                # up to 20 % of the source mass per move
    decay_every = 2500

    while time.time() < deadline:
        # propose a mass transfer i → j
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        src = a[i]
        if src <= 0.0:
            continue
        delta = rng.random() * min(src, step_frac * src)
        if delta == 0.0:
            continue

        trial = a.copy()
        trial[i] -= delta
        trial[j] += delta
        trial = project_onto_simplex(trial)

        conv = conv_fft(trial)
        trial_score = 2.0 * n * conv.max()

        # acceptance rule
        if trial_score < cur_score or rng.random() < np.exp(-(trial_score - cur_score) / temp):
            a = trial
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_a = trial.copy()
                print(f"[SA iter {it:08d}] new best score = {best_score:.6f}")

        # parameter decay
        if (it + 1) % decay_every == 0:
            temp = max(temp * 0.90, 1e-6)
            step_frac = max(step_frac * 0.97, 1e-4)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point --------------------------------------------------------
