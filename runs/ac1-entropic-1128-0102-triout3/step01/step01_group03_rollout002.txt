def solve_lp_max_sum(f, M):
    n = len(f)
    # Build Toeplitz matrix A
    from scipy.linalg import toeplitz
    # First column
    c = np.concatenate([f, np.zeros(n-1)])
    # First row
    r = np.concatenate([f[:1], np.zeros(n-1)])

    # Constraint: A @ g <= M (elementwise)
    # Build constraints for linprog: A_ub matrix = A, b_ub = np.full(2*n-1, M)
    # objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    # bounds: (0, None) or (0, M/min_a)
    min_f = f[np.nonzero(f)].min()
    ub = M / min_f if min_f > 0 else M / 1e-12
    bounds = [(0, ub) for _ in range(n)]
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        print("[lp] failed", res.message)
        return None
    g = res.x
    return g

"""
Propose a sequence of non‑negative heights that (hopefully) improves the
state‑of‑the‑art bound 1.5709 … → ≈1.50.

The algorithm has two phases.

Phase 1 – Linear‑programming “mass‑pumping”.
    * Start from the half‑binomial (sqrt) sequence (sum = 1).
    * Let M = max( a * a ).
    * Solve the linear program
          max  sum(g)
          s.t.  a * g ≤ M      (convolution is linear in g)
                g ≥ 0
      The Toeplitz matrix of a gives the linear constraints.
    * Normalise g to a probability vector and mix it with a
      (convex combination).  Several mixing ratios are tried; the
      best improvement (if any) is kept.
    * Repeat a few times – each iteration can increase the L¹‑mass
      without raising the peak, thereby lowering the evaluation
      2·n·max / (∑a)².

Phase 2 – Stochastic local search.
    * Starting from the best vector found in phase 1,
      repeatedly move a tiny amount of mass from one index to another.
    * Accept moves that improve the score, otherwise accept with a
      Metropolis probability that slowly cools (simulated annealing).
    * The convolution is recomputed with an FFT after each move.

Both phases respect the simplex constraint (non‑negative, sum = 1) by
projecting onto the probability simplex (Duchi et al. 2008).  The
routine stops when the allotted wall‑clock time (default 950 s) is
exhausted and returns the best sequence seen.
"""

def sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^(-1/2); normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.
    Implements Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Real‑FFT based convolution x * x.
    Returns an array of length 2·len(x)‑1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv

# Optional heavy imports (SciPy).  If SciPy is unavailable we fall back
# to pure‑gradient search.

try:
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz
    SCIPY_OK = True
except Exception:          # pragma: no cover
    SCIPY_OK = False


def _build_toeplitz_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n Toeplitz matrix that maps g → a * g.
    Each row k contains a_{k‑j} where j indexes the column.
    """
    n = a.shape[0]
    # first column = a followed by (n‑1) zeros
    col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    # first row = [a0, 0, 0, …]
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    return toeplitz(col, row)


def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Linear programme:
        maximise   sum(g)
        subject to a * g ≤ M   (element‑wise, a fixed)
                   g ≥ 0
    Returns the optimal g (may be all zeros if LP fails) or None.
    """
    if not SCIPY_OK:                     # pragma: no cover
        return None

    n = a.shape[0]
    A = _build_toeplitz_conv_matrix(a)   # shape (2n‑1, n)

    # Upper bound for each variable – a crude bound to help the solver
    # In any constraint the coefficient of g_j is at least min(a) (except
    # for very edge rows where the coefficient may be zero, but then that
    # row contains other positive coefficients).  Using this bound keeps
    # the LP bounded without hurting optimality.
    min_a = a[a > 0].min()
    ub = M / min_a if min_a > 0 else M * 1e6
    bounds = [(0.0, ub) for _ in range(n)]

    c = -np.ones(n)                      # minimise -sum(g)  ≡ maximise sum(g)

    # All rows share the same RHS = M
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # In rare cases the LP may be infeasible (numerical); ignore it.
        print("[LP] solver failed:", res.message)
        return None
    return res.x

def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises
        2 * n * max(conv(a,a)) / (sum(a)**2)

    The routine combines a small number of linear‑programming “mass‑pumping”
    steps with a long simulated‑annealing local‑search phase.
    The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) choose a length that balances quality and LP size.
    n = 800                     # 2n‑1 ≈ 1600 rows, comfortable for LP

    # Start from the classic half‑binomial (sqrt) distribution.
    a = sqrt_series(n)         # already normalised (∑a = 1)

    # If a previously discovered construction is present, keep the better.
    try:
        # height_sequence_1 is supplied by the evaluation harness (if any)
        seq0 = np.array(height_sequence_1, dtype=np.float64)
        # make sure it has the same length; truncate / pad with zeros
        if seq0.size != n:
            if seq0.size > n:
                seq0 = seq0[:n]
            else:
                seq0 = np.pad(seq0, (0, n - seq0.size))
        seq0 = np.maximum(seq0, 0.0)
        seq0 /= seq0.sum() if seq0.sum() > 0 else 1.0
        score0 = evaluate_sequence(seq0.tolist())
        score_a = evaluate_sequence(a.tolist())
        if score0 < score_a:
            a = seq0
            print("[init] used provided height_sequence_1, score =", score0)
    except Exception:          # pragma: no cover
        pass

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, start score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – linear‑programming mass‑pumping (a handful of iterations)
    # ------------------------------------------------------------------
    lp_iters = 0
    max_lp_iters = 12                               # enough to move the mass

    while time.time() < deadline and lp_iters < max_lp_iters:
        # current convolution peak
        conv_a = conv_fft(a)
        M = conv_a.max()

        # solve LP: maximise L1‑mass while keeping a * g ≤ M
        g = solve_lp_max_sum(a, M)
        if g is None:
            break

        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        g_norm = g / sum_g   # now a probability vector

        # try several mixing ratios and keep the best improvement
        improved = False
        for t in (0.6, 0.4, 0.2, 0.1):
            cand = (1.0 - t) * a + t * g_norm
            cand = project_onto_simplex(cand)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score - 1e-9:
                best_score = cand_score
                best_a = cand.copy()
                a = cand                       # accept as current iterate
                improved = True
                print(f"[LP iter {lp_iters:02d}] t={t:.2f} new best = {best_score:.6f}")
                break                     # accept first improvement

        if not improved:
            # No mixing ratio helped – stop the LP phase early.
            break

        lp_iters += 1

    # ------------------------------------------------------------------
    # Phase 2 – stochastic local search (projected random moves)
    # ------------------------------------------------------------------
    # Parameters for the annealing loop
    temp_start = 0.05
    temp = temp_start
    decay_rate = 0.9995           # slow exponential cooling
    step_factor = 0.12            # fraction of source mass to move
    max_iters = 10_000_000        # hard cap (time‑driven loop will stop earlier)
    it = 0

    # Pre‑compute FFT size to reuse in fast evaluations
    fft_len = 2 * n - 1

    # Helper: cheap evaluation using the known formula (no extra clipping)
    def fast_score(vec: np.ndarray) -> float:
        conv = np.fft.irfft(np.fft.rfft(vec, n=fft_len) *
                            np.fft.rfft(vec, n=fft_len), n=fft_len)
        return float(2.0 * n * conv.max())

    while time.time() < deadline and it < max_iters:
        # Randomly pick a source index with enough mass
        src = rng.integers(0, n)
        if a[src] < 1e-12:
            it += 1
            continue

        # Destination index different from source
        dst = rng.integers(0, n)
        while dst == src:
            dst = rng.integers(0, n)

        # Amount to shift – a fraction of the source entry
        delta = step_factor * a[src] * rng.random()
        if delta <= 0.0:
            it += 1
            continue

        # Propose new vector
        a_trial = a.copy()
        a_trial[src] -= delta
        a_trial[dst] += delta

        # Project back onto the simplex (ensures non‑negativity & sum = 1)
        a_trial = project_onto_simplex(a_trial)

        # Score (fast, using FFT – identical to evaluate_sequence up to clipping)
        trial_score = fast_score(a_trial)

        # Acceptance
        if trial_score < best_score - 1e-12:
            best_score = trial_score
            best_a = a_trial.copy()
            a = a_trial
            print(f"[SA iter {it:07d}] new best = {best_score:.6f}")
        else:
            # Metropolis rule
            delta_score = trial_score - best_score
            if rng.random() < np.exp(-delta_score / max(temp, 1e-12)):
                a = a_trial

        # Cool down
        temp *= decay_rate
        it += 1

    # ------------------------------------------------------------------
    # Final verification with the official evaluator (clipping etc.)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
