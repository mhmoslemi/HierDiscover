def project_onto_simplex(v):
    """Project v onto simplex (non-negative, sum=1)."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    # shift v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

Propose a step‑function (non‑negative heights) with a lower
self‑convolution bound than the reference implementation.

The algorithm works in three stages:

1. **Initialisation** – start from the best of a few cheap seeds
   (the supplied `height_sequence_1`, a uniform vector and a
   power‑law vector).  All vectors are normalised to sum = 1 (the
   objective is scale‑invariant).

2. **Smooth‑max gradient descent** – we minimise an
   approximation of   max_k (a*a)_k   obtained by a
   log‑sum‑exp (soft‑max) with a temperature λ.
   The gradient of the soft‑max w.r.t. a can be written as a
   cross‑correlation and is therefore computed efficiently with
   FFTs.  A simple line‑search (backtracking) together with a
   projection onto the probability simplex yields a feasible update.

3. **Polishing** – after the gradient phase we perform a few
   random mass‑transfer moves (as in the reference hill‑climber)
   to escape possible tiny plateaus.

The whole search respects the 1000 s budget, prints progress,
and finally returns the best sequence found as a plain Python list.

def _full_convolution_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the full (linear) convolution of two real vectors using a real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    L = len(x) + len(y) - 1
    return conv[:L]


def _self_convolution_max_fft(a: np.ndarray, fft_len: int) -> (float, np.ndarray):
    """
    Compute max_k (a*a)_k and also return the whole convolution vector.
    The convolution is performed via FFT for speed.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max()), conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.
    This implementation follows the O(n log n) algorithm based on sorting.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find the rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all elements are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        if w_sum == 0.0:   # should never happen, but be safe
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ----- gradient of the soft‑max (see analysis) -----
        a_rev = a[::-1]
        # cross‑correlation a ⋆ w_norm = conv(a_rev, w_norm) slice
        corr_full = _full_convolution_fft(a_rev, w_norm, fft_len_grad)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----- line‑search on step size (back‑tracking) -----
        # base step suggested by the inverse of the max absolute gradient entry
        base_step = 0.5 / (np.max(np.abs(grad)) + 1e-12)

        improved = False
        step = base_step
        for _ in range(8):               # up‑to 8 reductions of the step
            a_trial = a - step * grad
            # projection onto the simplex (maintains feasibility)
            a_trial = _project_onto_simplex(a_trial)

            # quick approximation of the new score
            max_b_new, _ = _self_convolution_max_fft(a_trial, fft_len_conv)
            approx_new = 2.0 * n * max_b_new

            if approx_new < best_approx - 1e-12:          # genuine improvement
                true_score = evaluate_sequence(a_trial.tolist())
                if true_score < best_score:
                    best_score = true_score
                    best_seq = a_trial.copy()
                    best_approx = approx_new
                    print(f"[iter {it:5d}] λ={lam:5.2f} step={step: .4e} → score={true_score:.6f}")
                a = a_trial
                improved = True
                break
            else:
                step *= 0.5               # shrink step and retry

        # ----- occasional random perturbation to escape plateaus -----
        if not improved and np.random.rand() < 0.07:
            i1, i2 = np.random.choice(n, size=2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.03   # up to ~3 % of donor
            # make sure we stay non‑negative
            if a[i1] + delta < 0.0:
                delta = -a[i1] + 1e-12
            if a[i2] - delta < 0.0:
                delta = a[i2] - 1e-12
            a[i1] += delta
            a[i2] -= delta
            a = _project_onto_simplex(a)

        # ----- temperature schedule -----
        lam = min(lam_max, lam + lam_inc)

        it += 1

        # early exit if we have already beaten the target comfortably
        if best_score < 1.55:
            break

    # --------------------------------------------------------------
    # 4. Final polishing – short random hill‑climb (like the reference)
    # --------------------------------------------------------------
    while time.time() < finish_time:
        i_src, i_dst = np.random.choice(n, size=2, replace=False)
        delta = (np.random.rand() - 0.5) * 0.02   # tiny shuffle
        if a[i_src] + delta < 0.0:
            delta = -a[i_src] + 1e-12
        if a[i_dst] - delta < 0.0:
            delta = a[i_dst] - 1e-12
        a[i_src] += delta
        a[i_dst] -= delta
        a = _project_onto_simplex(a)

        # evaluate only every few steps to save time
        if it % 20 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[polish] improved to {sc:.6f}")
        it += 1

    # --------------------------------------------------------------
    # 5. Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform (should never happen)
        best_seq = np.ones(n) / n
    return best_seq.tolist()
