"""
Propose a candidate step‑function using a smoothed sub‑gradient /
Adam optimizer on the probability simplex.

The evaluation score (scale‑invariant) is
    score = 2 * n * max(conv(a, a))                (sum(a)=1)

We minimise the smooth approximation
    f_μ(a) = μ·log(∑_k exp(conv_k/μ))
whose gradient is
    ∇f_μ(a)_i = 2·∑_k p_k·a_{k-i},  p_k = exp(conv_k/μ)/∑_j exp(conv_j/μ)

The gradient can be computed in O(n log n) by a single FFT‑based convolution.
We use Adam to obtain an adaptive step size and after every update project
the vector back onto the probability simplex (Euclidean projection).
The temperature μ is annealed (μ = max(μ_min, μ0/√t)).  Several
restart runs with different lengths are performed; the best sequence seen
throughout the allotted time is returned.

All numbers are forced to be non‑negative; the projector guarantees the
simplex constraints, so the returned list is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum(x) = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _auto_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear (non‑circular) auto‑convolution a*a of length 2·len(a)‑1."""
    Af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Af * Af, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_smoothed(a: np.ndarray, conv: np.ndarray, mu: float,
                       fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smoothed objective μ·log∑exp(conv/μ).
    Uses the identity
        g_i = 2·∑_k p_k·a_{k-i},
    where p_k = exp(conv_k/μ)/∑_j exp(conv_j/μ).
    The convolution is carried out by FFT for speed.
    """
    # softmax weights (stable)
    shift = conv - conv.max()
    exp_shift = np.exp(shift / mu)
    sum_exp = exp_shift.sum()
    if sum_exp == 0.0:               # under‑flow – use a delta at the max
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_shift / sum_exp

    # reversed a
    a_rev = a[::-1]

    # convolution a_rev * p (linear, length 3·n‑2)
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    P = np.fft.rfft(p, n=fft_len_grad)
    conv_gp = np.fft.irfft(A_rev * P, n=fft_len_grad)

    n = a.size
    start = n - 1                     # slice that gives ∑_j a_j·p_{i+j}
    g = 2.0 * conv_gp[start:start + n]
    return g

            a = _initial_sequence(n)

            # Adam state
            m = np.zeros(n)
            v = np.zeros(n)
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8

            # hyper‑parameters
            base_lr = 0.5          # base learning rate (scaled by 1/√t)
            mu0 = 0.02             # initial temperature
            mu_min = 5e-5

            it = 0
            max_iter = 300000       # generous hard cap

            while time.time() < deadline and it < max_iter:
                it += 1

                # ---------- convolution ----------
                conv = _auto_convolution(a, fft_len_conv)

                # current objective (sum(a)=1 ⇒ score = 2·n·max(conv))
                max_val = conv.max()
                cur_score = 2.0 * n * max_val
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time() - start_time:6.2f}s] "
                          f"Improved: n={n}, restart={restart}, iter={it}, "
                          f"score={cur_score:.6f}")

                # ---------- temperature schedule ----------
                mu = mu0 / math.sqrt(it)
                if mu < mu_min:
                    mu = mu_min

                # ---------- gradient (smoothed) ----------
                g = _gradient_smoothed(a, conv, mu, fft_len_grad)

                # ---------- Adam update ----------
                m = beta1 * m + (1.0 - beta1) * g
                v = beta2 * v + (1.0 - beta2) * (g * g)

                # bias‑corrected estimates
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)

                lr = base_lr / math.sqrt(it)
                a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---------- projection onto the simplex ----------
                a = _project_simplex(a)

            # end while (iteration loop)
        # end for (restart)
    # end for (length)

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(100, 1.0 / 100.0)

    # final normalisation (protect against numerical drift)
    best_seq = np.asarray(best_seq, dtype=float)
    s = best_seq.sum()
    if s == 0.0:
        best_seq = np.full(best_seq.shape, 1.0 / best_seq.size)
    else:
        best_seq = best_seq / s

    # return a plain Python list of floats
    return [float(x) for x in best_seq]
