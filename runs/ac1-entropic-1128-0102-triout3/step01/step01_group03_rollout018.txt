"""
Propose candidate for the convolution‑based evaluation.

The algorithm starts from the classic half‑binomial (a_i = C(2i,i)/4^i) sequence,
scaled so that its ℓ¹‑norm equals √(2 n).  For that scaling the evaluation
reduces to max (conv(a,a)).  We then minimise this maximum with a
projected “soft‑max” gradient descent:

* At each step the convolution is computed (FFT, O(n log n)).
* A smooth approximation of the max is built with
  w_k ∝ exp(τ·(conv_k−max(conv))) – a soft‑max over the convolution entries.
* The (sub)‑gradient of the smoothed objective is obtained analytically.
  For a weight vector w the gradient splits into two cheap parts:
      g_i = Σ_j a_j w_{i+j}  +  a_i·Σ_{k=i}^{i+n-1} w_k .
  The first term is a correlation (computed by a single FFT), the second by a
  cumulative sum.
* An additive step a ← a − η·g is taken, negative values are clipped,
  and the vector is renormalised to the target ℓ¹‑norm (projection onto the
  simplex with a fixed sum).  Occasionally a tiny random perturbation is added
  to avoid stagnation.
* A simple simulated‑annealing acceptance rule permits occasional uphill moves.
  Step size η, temperature and the soft‑max sharpness τ are decayed gradually.

The loop runs until the allotted time expires (default 950 s, leaving a safety
margin).  The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of a_i = C(2i,i)/4**i (half‑binomial series).
    Returned vector is normalised to sum 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a_i = a_{i-1} * (2*i-1) / (2*i)
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Compute the discrete autocorrelation x*x using a real FFT.
    Result length is 2*len(x)-1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolution of two real vectors using FFT.
    Length of the result is len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]

    n = 2500                                 # length of the sequence
    target_sum = np.sqrt(2.0 * n)            # after scaling, eval = max(conv)
    max_iter = 10_000_000                    # hard safety bound
    eta = 0.45                               # initial additive step size
    temperature = 0.12                       # SA temperature
    tau = 4.0                                # soft‑max sharpness
    decay_interval = 2000                    # interval for parameter decay

    # ------------------------------------------------------------------
    # initialise from the half‑binomial series (the known good construction)
    a = sqrt_series(n) * target_sum          # scale to the target ℓ¹‑norm
    conv = conv_fft(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_score = float(cur_max)              # because sum = √(2n) ⇒ score = max(conv)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation loop
    deadline = time.time() + budget_s - 5.0   # safety margin
    it = 0

    while time.time() < deadline and it < max_iter:
        # ----- current convolution & soft‑max weights -------------------
        conv = conv_fft(a)                    # length 2n‑1
        max_val = conv.max()
        # soft‑max over convolution entries (stable because we subtract max)
        w = np.exp(tau * (conv - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical safeguard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective ----------------------
        # part 2: a_i multiplied by sum of w over the interval [i, i+n‑1]
        cum = np.concatenate(([0.0], np.cumsum(w)))          # length 2n
        part2 = cum[n:] - cum[:n]                            # shape (n,)

        # part 1: correlation of w with a (implemented via convolution)
        a_rev = a[::-1]
        c = conv_fft_general(w, a_rev)                      # length 3n‑2
        part1 = c[n - 1: 2 * n - 1]                         # indices n‑1 … 2n‑2

        grad = part1 + a * part2

        # ----- additive update and projection onto the simplex ----------
        step = eta * grad
        a_trial = a - step
        a_trial = np.maximum(a_trial, 0.0)                   # keep non‑negative
        s = a_trial.sum()
        if s == 0.0:                                         # avoid division by zero
            a_trial = a.copy()
        else:
            a_trial *= (target_sum / s)                      # rescale to fixed sum

        # occasional random jitter to escape shallow plateaus
        if rng.random() < 0.008:
            jitter = (rng.random(n) - 0.5) * 2e-5
            a_trial += jitter
            a_trial = np.maximum(a_trial, 0.0)
            a_trial *= (target_sum / a_trial.sum())

        # ----- evaluate candidate ---------------------------------------
        score = evaluate_sequence(a_trial.tolist())          # safe call
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            print(f"[iter {it:08d}] new best score = {best_score:.6f}")

        # ----- acceptance (always take improvement) --------------------
        if score <= cur_max:
            a = a_trial
            cur_max = score
        else:
            # Metropolis acceptance for occasional uphill moves
            delta = score - cur_max
            if temperature > 0.0 and rng.random() < np.exp(-delta / temperature):
                a = a_trial
                cur_max = score
            # otherwise reject

        # ----- schedule decay -------------------------------------------
        if (it + 1) % decay_interval == 0:
            eta *= 0.93
            temperature *= 0.92
            tau = min(tau * 1.02, 200.0)           # slowly sharpen the soft‑max
            eta = max(eta, 1e-6)
            temperature = max(temperature, 1e-8)

        it += 1

    # ------------------------------------------------------------------
    # final verification (the evaluate_sequence call also protects against
    # any accidental out‑of‑bounds values)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, final reported score={final_score:.6f}")

    return best_a.tolist()
