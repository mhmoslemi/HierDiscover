"""
Propose a better step‑function using a *linear‑programming boost*.
-----------------------------------------------------------------

The objective of `evaluate_sequence` for a sequence whose L1–norm is
√(2·n) is exactly the maximal entry of its auto‑convolution:
    score = max (a * a) .
The classic half‑binomial (central‑binomial) construction gives a
constant convolution ≈ 1.571·(1/n) and a score ≈ π/2 ≈ 1.571.

The algorithm iteratively improves the shape:

1. Start from the half‑binomial vector and scale it to have sum
   √(2·n).  This guarantees that the score equals the convolution peak.

2. Given the current vector `f` and its peak `M = max(f*f)`,
   solve the linear program

        maximise   Σ b_i
        subject to  (f * b)_k ≤ M   for all k = 0,…,2n‑2
                    b_i ≥ 0

   (`f * b` is a linear operation in the variables `b_i`).  The LP
   yields a direction `b` that can carry *more* total mass than `f`
   without increasing the convolution bound `M`.

3. Rescale the LP solution to the required L1‑norm
   (`sqrt(2·n)`) → `g`.

4. Mix the old and the new vector:
        f_new = (1‑t)·f + t·g ,   t∈(0,1]
   The bound `max(f_new*f_new)` is guaranteed to drop for a small
   enough `t`; a fixed step `t = 0.4` works well in practice.

5. After the mixing step a few cheap projected‑gradient moves are
   performed to clean up any remaining peaks.

6. The best vector seen so far is kept.  The whole loop runs until
   the time budget (≈ 950 s) expires.

The LP has (2·n‑1) linear inequalities and `n` variables.
For `n≈1200` the dense coefficient matrix is modest
(~30 MiB) and each LP solve takes well under a second,
so dozens of iterations comfortably fit into the time limit.

def sqrt_series(n: int) -> np.ndarray:
    """Half‑binomial coefficients a_i = C(2i,i)/4**i, summed to 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(x: np.ndarray) -> np.ndarray:
    """Full auto‑convolution x*x using real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A for the constraints (a * b)_k ≤ M.
    A_{k,i} = a_{k‑i} for 0≤i<n, 0≤k‑i<n, otherwise 0.

    return toeplitz(first_col, first_row)   # dense, size (2n‑1, n)

            # gradient of the current peak (same pattern as the original code)
            grad = np.zeros_like(a)
            lo = max(0, max_idx - (n - 1))
            hi = min(max_idx, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            jdx = max_idx - idx
            grad[idx] += a[jdx]      # a_i * a_{k-i} part
            grad[idx] += a[idx]      # symmetric partner
            step = 0.07 * grad       # modest step size
            # ensure no negative entries
            step = np.minimum(step, a * 0.9)
            a = a - step
            # project back onto the simplex with the correct L1‑norm
            a = project_onto_simplex(a)
            a *= sum_target / a.sum()
        # -----------------------------------------------------------------
        # keep the best ever seen
        if M < best_score - 1e-9:
            best_score = M
            best_a = a.copy()
            print(f"[iter {it:03d}] NEW GLOBAL BEST M={best_score:.6f}")

        # occasional random jitter to avoid stagnation
        if rng.random() < 0.03:
            jitter = (rng.random(n) - 0.5) * 2e-2 * sum_target
            a = np.maximum(a + jitter, 0.0)
            a = project_onto_simplex(a)
            a *= sum_target / a.sum()
            # recompute M after jitter
            M = conv_self(a).max()

        it += 1

    # ------------------------------------------------------------------
    # final check with the provided evaluator (clipping etc.)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[done] iterations={it}, best reported score={final_score:.6f}")

    return best_a.tolist()
