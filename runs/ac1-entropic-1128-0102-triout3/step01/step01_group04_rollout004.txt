def lp_step(f):
    n = len(f)
    S = np.sqrt(2*n)
    A = toeplitz(np.concatenate([f, np.zeros(n-1)]), np.concatenate([f[:1], np.zeros(n-1)]))
    # shape: (2n-1, n)
    rows = A.shape[0]
    # Build A_ub: [A, -ones]
    A_ub = np.hstack([A, -np.ones((rows,1))])
    b_ub = np.zeros(rows)  # because constraint A*g - M <= 0
    # equality constraint sum(g) = S
    A_eq = np.zeros((1, n+1))
    A_eq[0,:n] = 1
    b_eq = np.array([S])
    bounds = [(0, None)]*n + [(0, None)]
    c = np.zeros(n+1)
    c[-1] = 1  # minimize M
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not result.success:
        return None, None
    x = result.x
    g = x[:n]
    M_lp = x[-1]
    return g, M_lp
```

def lp_refine(f, max_iters, deadline):
    n = len(f)
    S = np.sqrt(2*n)
    best_f = f.copy()
    best_M = np.max(np.convolve(best_f, best_f))
    for iter_idx in range(max_iters):
        if time.time() >= deadline:
            break
        # Build constraint matrix for current best_f
        A = toeplitz(np.concatenate([best_f, np.zeros(n-1)]),
                     np.concatenate([best_f[:1], np.zeros(n-1)]))
        rows = A.shape[0]
        # Append column for M variable
        A_ub = np.hstack([A, -np.ones((rows, 1))])
        b_ub = np.zeros(rows)

        # equality for sum
        A_eq = np.zeros((1, n+1))
        A_eq[0,:n] = 1
        b_eq = np.array([S])

        bounds = [(0, None)]*n + [(0, None)]

        c = np.zeros(n+1)
        c[-1] = 1

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x[:n]
        M_lp = res.x[-1]

        # compute auto-convolution max for g
        M_g = np.max(np.convolve(g, g))
        # improvement?
        if M_g < best_M - 1e-8:
            best_f = g
            best_M = M_g
            # optional: reduction of step?
            print(f"[lp] iter={iter_idx} n={n} M={best_M:.6f}")
        else:
            # try mixing
            mixed = False
            for t in [0.5, 0.3, 0.2, 0.1]:
                cand = (1-t)*best_f + t*g
                M_cand = np.max(np.convolve(cand, cand))
                if M_cand < best_M - 1e-8:
                    best_f = cand
                    best_M = M_cand
                    mixed = True
                    print(f"[lp-mix] iter={iter_idx} t={t} M={best_M:.6f}")
                    break
            if not mixed:
                # no improvement, stop.
                break
    return best_f, best_M

"""
Optimisation of a non‑negative step function for the bound

    2 * n * max(conv(a, a)) / (sum(a)**2)

The bound simplifies to the maximal self‑convolution value when the
ℓ1‑norm is fixed to √(2 n).  The algorithm works in three phases:

1.  Random analytic initialisations (discrete derivatives of x^α)
    are sampled for a few lengths n and the best one is kept.
2.  Starting from that best sequence an *LP‑driven Frank‑Wolfe*
    iteration is performed:  
    – Solve a linear programme that maximises Σ g_j under the
      constraints (f * g)[t] ≤ M (the current maximal convolution of f).  
    – Rescale g to the required ℓ1‑norm and move the current iterate
      f ← (1‑τ) f + τ g.  This strictly reduces the bound in practice.
    The step size τ is gradually decreased.
3.  A short stochastic hill‑climbing refinement (single‑coordinate
    Gaussian moves) finishes the search.

All steps respect the time budget (default 1000 s).  If SciPy is not
available the code falls back to the hill‑climbing only.  The final
sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Required ℓ1‑norm:  Σ a_i = √(2 n)."""
    return np.sqrt(2.0 * n)


def _analytic_power_sequence(n: int, alpha: float, delta: float = 0.0) -> np.ndarray:
    """
    Analytic family   a_i = (i+1+δ)^α – (i+δ)^α .
    The vector is scaled so that its ℓ1‑norm equals √(2 n).
    """
    i = np.arange(n, dtype=float)
    base = np.power(i + 1.0 + delta, alpha) - np.power(i + delta, alpha)
    scale = _target_sum(n) / base.sum()
    return base * scale


def _evaluate(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence."""
    return evaluate_sequence(seq.tolist())   # type: ignore[name-defined]


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return np.max(np.convolve(seq, seq))


# ----------------------------------------------------------------------
# Hill‑climbing (same as the reference implementation, but lightweight)
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray, max_iter: int, deadline: float) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing that respects the ℓ1‑norm."""
    n = seq.shape[0]
    best = seq.copy()
    best_score = _evaluate(best)
    step_sigma = max(1e-9, seq.mean() * 0.2)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        idx = np.random.randint(n)
        delta = np.random.randn() * step_sigma
        old = seq[idx]
        new = old + delta
        if new < 0.0:
            new = 0.0
        seq[idx] = new

        # renormalise to fixed ℓ1‑norm
        seq *= _target_sum(n) / seq.sum()

        cand_score = _evaluate(seq)
        if cand_score < best_score:
            best_score = cand_score
            best = seq.copy()
        else:
            seq[idx] = old
            seq *= _target_sum(n) / seq.sum()   # re‑normalise in case of drift

        step_sigma *= 0.9995                     # slow decay
    return best, best_score


# ----------------------------------------------------------------------
# LP‑based Frank‑Wolfe step (requires SciPy)
# ----------------------------------------------------------------------


try:
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz
    _SCIPY_AVAILABLE = True
except Exception:                 # pragma: no cover
    _SCIPY_AVAILABLE = False


def _lp_frank_wolfe_step(f: np.ndarray, tau: float, target_sum: float) -> tuple[np.ndarray, float]:
    """
    One Frank‑Wolfe iteration:

    1.  Solve   max Σ g_j   s.t. (f * g)[t] ≤ M  (M = max(conv(f,f))).
    2.  Rescale g to the required ℓ1‑norm.
    3.  Return the convex combination (1‑τ) f + τ g and its new max‑conv.
    """
    if not _SCIPY_AVAILABLE:               # pragma: no cover
        return f.copy(), _conv_max(f)

    n = f.shape[0]
    M = _conv_max(f)

    # ----- build Toeplitz matrix for the convolution constraints -----
    #   A[t, j] = f[t‑j]  (zero outside the range)
    col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=float)])
    A = toeplitz(col, row)                  # shape (2n‑1, n)

    b = np.full(A.shape[0], M, dtype=float)

    # ----- linear programme: maximise Σ g_j  (i.e. minimise –Σ g_j) -----
    c = -np.ones(n, dtype=float)            # minimise –sum(g)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:                     # fallback to unchanged f
        return f.copy(), M

    g_raw = res.x
    s = g_raw.sum()
    if s == 0.0:                             # unexpected, keep f
        return f.copy(), M

    # ----- rescale g to the prescribed ℓ1‑norm -----
    g = g_raw * (target_sum / s)

    # ----- convex combination -----
    f_new = (1.0 - tau) * f + tau * g
    M_new = _conv_max(f_new)
    return f_new, M_new


def _lp_refine(start_seq: np.ndarray, max_iter: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeated Frank‑Wolfe steps until no progress or time runs out.
    The step size τ is geometrically decayed.
    """
    if not _SCIPY_AVAILABLE:                 # pragma: no cover
        return start_seq.copy(), _conv_max(start_seq)

    n = start_seq.shape[0]
    target = _target_sum(n)

    f = start_seq.copy()
    best_M = _conv_max(f)
    tau = 0.4                                 # initial mixing weight

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        f_candidate, M_candidate = _lp_frank_wolfe_step(f, tau, target)

        if M_candidate < best_M - 1e-9:
            f = f_candidate
            best_M = M_candidate
            print(f"[lp] iter={it:3d} n={n:4d} τ={tau:.3f} M={best_M:.6f}")
            # keep the step size for the next iteration
        else:
            # no improvement → shrink τ
            tau *= 0.8
            if tau < 1e-4:
                break
    return f, best_M
