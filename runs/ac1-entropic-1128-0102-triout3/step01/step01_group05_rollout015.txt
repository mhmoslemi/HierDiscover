def generate_support(n, max_mult=2, target_m=None, seed):
    sum_counts = np.zeros(2*n-1, dtype=np.int8)
    support = []
    indices = list(range(n))

    for i in indices:
        # check self
        if sum_counts[2*i] + 1 > max_mult:
            continue
        ok = True
        for j in support:

            if sum_counts[s] + 1 > max_mult:
                ok = False
                break
        if not ok:
            continue
        # accept
        support.append(i)
        sum_counts[2*i] += 1
        for j in support[:-1]:  # previous ones (i added)
            s = i + j
            sum_counts[s] += 1
        # optionally break if target_m reached
        if target_m and len(support) >= target_m:
            break
    return support, sum_counts

"""
Search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The key idea is to use a *sparse* support.  If the set of indices
S = {i_0,…,i_{m‑1}} is chosen so that each possible sum
i+j (0 ≤ i,j < n) occurs only a few times, then the convolution
value at any position is the sum of at most a handful of products
a_i·a_j.  With a uniform weight w on the support (∑a = √(2n) is
enforced) the maximum convolution is roughly

    2·c·w²   with   w = √(2n)/m,   c = max number of pairs per sum.

Thus, by constructing a large support with a small bound c (e.g. c=2
or 3) we can drive the evaluation value far below the previous
record of ~1.58.  The program repeatedly builds candidate supports
with a greedy “add‑if‑still‑≤c” rule, keeps the best one found, and
finally refines the weights by a tiny stochastic hill‑climbing
procedure.  The whole search respects the 1000 s budget.
"""

import time
import numpy as np


def compute_max_conv(n: int, support: list[int], weights: np.ndarray) -> float:
    """
    Return the maximum value of the linear convolution a*a,
    where `a` is zero everywhere except on `support` with the
    given `weights`.  Uses O(m²) work with m = len(support).
    """
    size = 2 * n - 1
    conv = np.zeros(size, dtype=np.float64)
    max_val = 0.0
    m = len(support)

    for i in range(m):
        wi = float(weights[i])
        s_self = support[i] * 2
        conv[s_self] += wi * wi                     # self‑pair
        if conv[s_self] > max_val:
            max_val = conv[s_self]

        for j in range(i + 1, m):
            wj = float(weights[j])
            s = support[i] + support[j]
            val = 2.0 * wi * wj                      # ordered pair (i,j) + (j,i)
            conv[s] += val
            if conv[s] > max_val:
                max_val = conv[s]

    return max_val

        attempt += 1
        order = rng.permutation(n)               # random order of indices
        support = []
        # counts of how many unordered pairs already use a given sum
        sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

        for idx in order:
            s_self = idx * 2
            if sum_counts[s_self] + 1 > max_multiplicity:
                continue

            ok = True
            for j in support:
                s = idx + j
                if sum_counts[s] + 1 > max_multiplicity:
                    ok = False
                    break
            if not ok:
                continue

            # accept this index
            support.append(idx)
            sum_counts[s_self] += 1
            for j in support[:-1]:
                sum_counts[idx + j] += 1

        m = len(support)
        if m < 5:                     # degenerate candidate – ignore
            continue

        # uniform weights that satisfy the ℓ¹‑normalisation
        uniform_weight = target_sum / m
        w = np.full(m, uniform_weight, dtype=np.float64)

        max_conv = compute_max_conv(n, support, w)

        if max_conv < best_val:
            best_val = max_conv
            best_support = support.copy()
            best_weights = w.copy()
            print(f"[attempt {attempt:04d}] new best – size {m}, max_conv {max_conv:.6f}")

        # an early stop if we already reached the target region
        if best_val <= 1.01:
            break

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_support is None:
        seq = np.full(n, target_sum / n, dtype=np.float64)
        return seq.tolist()

    # ------------------------------------------------------------------
    # 2.  Weight refinement on the selected support – tiny stochastic
    #     hill‑climbing that preserves the total sum.
    # ------------------------------------------------------------------
    support = best_support
    m = len(support)
    weights = best_weights.copy()
    current_val = best_val
    step = 0.05
    hill_iter = 0

    while time.time() < deadline:
        i, j = rng.integers(0, m, size=2)
        if i == j:
            continue

        # move a random fraction of the mass from i to j
        delta = step * weights[i] * rng.random()
        if delta > weights[i]:
            delta = weights[i]

        new_weights = weights.copy()
        new_weights[i] -= delta
        new_weights[j] += delta

        new_val = compute_max_conv(n, support, new_weights)

        # accept improving moves (or a rare uphill move)
        if new_val < current_val or rng.random() < 0.001:
            weights = new_weights
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_weights = new_weights.copy()
                print(f"[hill {hill_iter:07d}] improved – max_conv {new_val:.6f}")

        # slowly shrink the step size
        if hill_iter % 5000 == 0 and hill_iter > 0:
            step = max(step * 0.9, 1e-6)

        hill_iter += 1

    # ------------------------------------------------------------------
    # 3.  Assemble the final full‑length sequence.
    # ------------------------------------------------------------------
    final_seq = np.zeros(n, dtype=np.float64)
    for idx, w in zip(best_support, best_weights):
        final_seq[idx] = w

    print(f"[final] best max_conv {best_val:.6f} on support size {len(best_support)}")
    return final_seq.tolist()
```
