"""
Hill‑Climbing Search for a Low‑Maximum Self‑Convolution Step Function.

The routine keeps the ℓ₁‑norm fixed at √(2·n) (the normalisation used by
`evaluate_sequence`).  In each iteration it
  1. computes the self‑convolution and locates the index where the maximum
     occurs,
  2. picks the coefficient that contributes most to that maximal entry,
  3. moves a small fraction of its mass to a coefficient that is currently
     very small,
  4. accepts the move if the true objective improves (or with a tiny
     uphill‑acceptance probability that slowly cools).

Occasional random jitter, a simple simulated‑annealing temperature schedule
and restarts from a random Dirichlet point help to escape plateaux.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``nfft``.
    Returns the first 2·len(a)‑1 real entries.
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x ≥ 0, Σ x = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray, n: int, nfft: int, target_sum: float) -> float:
    """
    Compute the exact objective 2·n·max(a*a) / (Σa)².
    ``a`` is assumed to be non‑negative.
    """
    conv = _convolve_self_fft(a, nfft)
    max_val = conv.max()
    # using the actual ℓ₁‑norm guards against tiny drift
    s = a.sum()
    return float(2.0 * n * max_val / (s * s))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------

    """

    # 1. initialise a feasible point (prefer the previously discovered one)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:
        n = 2401                     # fallback length used in the reference code

    S = np.sqrt(2.0 * n)            # required ℓ₁‑norm
    a = _project_to_simplex(a, S)   # ensure feasibility

    nfft = _next_pow2(2 * n - 1)

    min_step_factor = 0.001

            # uphill move with a tiny probability (simulated annealing)
            if temperature > 0:
                prob = np.exp(-(cand_score - cur_score) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_a = cand_a
            cur_score = cand_score
            no_improve_iters = 0
            # update global best
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand_a.copy()
                print(f"[{iteration:06d}] New best: {best_score:.6f} (max={cand_score:.6f})")
        else:
            no_improve_iters += 1

        # ----- (g) occasional random jitter to escape flat regions -----
        if np.random.rand() < 0.005:
            jitter_std = step_factor * S / n
            jitter = np.random.randn(n) * jitter_std
            jittered = cur_a + jitter
            jittered = np.maximum(jittered, 0.0)
            jittered = _project_to_simplex(jittered, S)
            jitter_score = _score(jittered, n, nfft, S)
            if jitter_score < cur_score:
                cur_a = jittered
                cur_score = jitter_score
                no_improve_iters = 0
                if jitter_score < best_score:
                    best_score = jitter_score
                    best_a = jittered.copy()
                    print(f"[{iteration:06d}] Jitter improved best to {best_score:.6f}")

        # ----- (h) schedule updates -----
        step_factor = max(min_step_factor, step_factor * decay_factor)
        temperature = max(temp_min, temperature * temp_decay)

        # ----- (i) restart if we have been idle for a long time -----
        if no_improve_iters >= max_no_improve:
            # random Dirichlet point on the simplex
            alpha = np.full(n, 0.3)
            cur_a = np.random.dirichlet(alpha) * S
            cur_score = _score(cur_a, n, nfft, S)
            no_improve_iters = 0
            print(f"[{iteration:06d}] Restarted from random simplex point (score {cur_score:.6f})")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. final sanity projection and verification
    # ------------------------------------------------------------------
    final_seq = _project_to_simplex(best_a, S)
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best final score = {final_score:.6f}")
    return final_seq.tolist()
