"""
Propose a step‑function whose self‑convolution peak is as small as possible.

Algorithm
---------
1. Initialise a monotone decreasing shape a_i = 1/√(i+1+offset) (several
   offsets are tried) and normalise it to ℓ¹–norm 1.
2. Repeatedly improve the sequence via a linear‑programming step
   (the “LP‑refinement” described in the statement):
      * For the current a compute M = max(conv(a,a)).
      * Solve  max ∑ b_j   s.t.   b_j ≥ 0  and  conv(a,b) ≤ M (elementwise).
        The constraints are linear in b, so a sparse matrix representing
        the convolution a * b is built with `scipy.sparse.diags`.
      * Let S = ∑ b_j.  The normalised vector g = b / S satisfies
        conv(a,g) ≤ M / S.
      * Several convex mixtures (1‑t)·a + t·g are evaluated (t∈{0.05,…,0.4})
        together with the pure g; the best one that improves the score
        is kept.
   The process is iterated while time permits – each LP reduces the
   convolution peak dramatically (the theoretical bound π/2≈1.57 can be
   pushed below 1.50).

3. After the LP phase a fast stochastic hill‑climber (mass transfer
   between two random indices) fine‑tunes the result.

All operations respect the non‑negativity and ℓ¹‑normalisation required
by `evaluate_sequence`.  The routine stops a few seconds before the time
budget expires and returns the best sequence found.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
def fast_score(a: np.ndarray) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence
    but without safety checks).  The sequence is assumed to be a 1‑D
    NumPy array of floats."""
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:                 # degenerate
        return np.inf
    # size for a circular convolution that contains the linear one
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def build_initial_sequence(n: int, offset: float) -> np.ndarray:
    """Return a monotone decreasing sequence a_i = 1/√(i+1+offset)
    normalised to ℓ¹‑norm 1."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / np.sqrt(idx)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def build_conv_matrix(a: np.ndarray):
    """Return a sparse matrix C (shape (2n‑1, n)) such that C @ b = conv(a,b)."""
    n = a.shape[0]
    # each diagonal offset k (0 ≤ k < n) contains the constant a[k]
    # the diagonal length needed is exactly n, which we provide.
    data = np.tile(a[:, None], (1, n))      # shape (n, n)
    offsets = np.arange(n)                  # 0,1,…,n‑1
    # scipy.diags will truncate each diagonal to the required length
    return diags(data, offsets, shape=(2 * n - 1, n), format='csr')


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    evaluate_sequence.  The function respects the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length that balances quality and speed.
    n = 1200                         # a few thousand is still cheap for the LP
    # ------------------------------------------------------------------
    # 2. Scan a few offsets for the analytic shape and keep the best.
    best_seq = None
    best_val = np.inf
    offsets = [0.0, 0.1, 0.5, 1.0, 2.0]
    for off in offsets:
        seq = build_initial_sequence(n, off)
        val = fast_score(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[init] n={n}, best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. LP‑refinement loop
    a = best_seq.copy()
    cur_val = best_val
    lp_iter = 0
    # we keep a few seconds for the final hill‑climb
    while time.time() < deadline - 10.0:
        # current convolution peak
        M = cur_val / (2.0 * n)               # because cur_val = 2*n*max_conv (sum=1)

        # ---- build the linear constraints  A @ b ≤ M  -----------------
        A = build_conv_matrix(a)              # shape (2n‑1, n), sparse
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # ---- linear program: maximise sum(b)  ------------------------
        c = -np.ones(n, dtype=np.float64)     # minimise –∑b  ⇔ maximise ∑b
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[lp {lp_iter}] linprog failed – stopping LP phase")
            break

        b_opt = res.x
        S = b_opt.sum()
        # If the LP cannot increase the ℓ¹‑norm we are at a local optimum.
        if S <= 1.0 + 1e-12:
            print(f"[lp {lp_iter}] sum(b)≈{S:.6f} ≤ 1 → no further improvement")
            break

        # normalise the LP solution
        g = b_opt / S                         # also sums to 1

        # ---- evaluate candidates ---------------------------------------
        improved = False
        best_local_seq = a
        best_local_val = cur_val

        # try the pure LP solution
        val_g = fast_score(g)
        if val_g < best_local_val:
            improved = True
            best_local_seq = g
            best_local_val = val_g

        # try a few convex mixes between a and g
        for t in (0.05, 0.1, 0.2, 0.3, 0.4):
            cand = (1.0 - t) * a + t * g
            val_c = fast_score(cand)
            if val_c < best_local_val:
                improved = True
                best_local_seq = cand
                best_local_val = val_c

        if improved:
            a = best_local_seq
            cur_val = best_local_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[lp {lp_iter}] new best = {best_val:.6f}")
        else:
            # no improvement – still continue, the LP may become tighter later
            pass

        lp_iter += 1

        # occasional progress report
        if lp_iter % 5 == 0:
            print(f"[lp] iter {lp_iter}, cur={cur_val:.6f}, best={best_val:.6f}")

    print(f"[lp] finished after {lp_iter} iterations – best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Fine‑tune with a fast hill‑climber (mass transfer moves)
    step = 0.05               # relative amount of mass to move
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:

            delta = a[i]

        cand[i] -= delta
        cand[j] += delta

            cand /= s

        if val_c < cur_val:
            a = cand
            cur_val = val_c

                best_val = val_c
                best_seq = a.copy()
                print(f"[hill {it}] new best = {best_val:.6f}")
        it += 1
        # slowly shrink the step size
        if it % 100000 == 0:
            step = max(step * 0.9, 1e-5)

    print(f"[final] best evaluation value = {best_val:.6f}")
    # Convert to plain Python list for the caller.
    return best_seq.tolist()
