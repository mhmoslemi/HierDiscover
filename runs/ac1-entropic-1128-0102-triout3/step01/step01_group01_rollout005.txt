"""
Adam‑based mirror descent on the simplex for minimizing the maximum
entry of the self‑convolution of a non‑negative step function.

The algorithm works with the logarithmic parametrisation
    a_i = exp(theta_i) ,   Σ a_i = √(2·n)
so that the ℓ₁‑constraint is satisfied by a simple renormalisation.
A soft‑max approximation of the max–convolution is used while the
temperature τ is annealed; in the final phase the exact sub‑gradient of
the true maximum is employed.  Adam adapts per‑coordinate step sizes,
while occasional random jitter and restarts avoid stagnation.
The best sequence encountered before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns the first (2·len(a)‑1) entries (real values).
    """
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, n=nfft)
    return conv[: 2 * a.size - 1]


def _grad_softmax(a: np.ndarray, conv: np.ndarray, tau: float,
                  nfft: int) -> np.ndarray:
    """
    Gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    Uses a numerically stable soft‑max.
    """
    # stable soft‑max weights
    c = conv.max()
    w = np.exp((conv - c) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:          # extremely unlikely, but guard against NaNs
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # convolution of w with reversed a
    A_rev = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    grad_full = np.fft.irfft(W * A_rev, n=nfft)[: 2 * a.size - 1]

    n = a.size
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def _grad_max(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a at the index k_max.
    The derivative of (a*a)[k] w.r.t. a_i equals 2·a_{k‑i} if the index is
    inside the support, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    if i_low <= i_high:
        idx = np.arange(i_low, i_high + 1)
        grad[idx] = 2.0 * a[k_max - idx]
    return grad


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σ x = target_sum}.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step function that minimises `evaluate_sequence`.
    Uses Adam on log‑variables with a soft‑max smoothing of the max
    convolution, followed by an exact max‑gradient phase.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # initialise sequence length and ℓ₁‑norm (the problem normalisation)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
    else:
        n = 2401                      # a reasonably large default
        init_seq = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64)

    S = np.sqrt(2.0 * n)               # required ℓ₁‑norm
    init_seq = init_seq * (S / init_seq.sum())

    # log‑variables (mirror descent on the simplex)
    eps = 1e-12
    log_theta = np.log(init_seq + eps)

    # bookkeeping of the best solution
    best_a = init_seq.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial score: {best_score:.6f}")

    # pre‑compute FFT length
    nfft = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters (operating on log‑variables)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.20                     # base learning rate in log‑space

    m = np.zeros_like(log_theta)
    v = np.zeros_like(log_theta)

    # soft‑max temperature schedule
    tau = 0.5
    tau_min = 0.01
    # we anticipate at most a few hundred thousand iterations
    total_iters_est = 300_000
    tau_decay = (tau_min / tau) ** (1.0 / total_iters_est)

    # stagnation handling
    no_improve = 0
    restart_limit = 8_000

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # reconstruct a from log‑variables and enforce Σa = S
        # ----------------------------------------------------------
        a = np.exp(log_theta)
        sum_a = a.sum()
        if sum_a == 0.0:                     # degenerated numerics → re‑initialise
            a = np.random.dirichlet(np.full(n, 0.5)).astype(np.float64) * S
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            continue
        a = a * (S / sum_a)

        # ----------------------------------------------------------
        # evaluate current objective (max entry of self‑convolution)
        # ----------------------------------------------------------
        conv = _convolve_self_fft(a, nfft)
        cur_max = conv.max()
        cur_score = cur_max                     # equal to evaluate_sequence after normalisation

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{iteration:08d}] New best: {best_score:.6f}")
        else:
            no_improve += 1

        # early stop if target has been reached
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration}")
            break

        # ----------------------------------------------------------
        # gradient of the (smoothed) max‑convolution
        # ----------------------------------------------------------
        if tau > tau_min * 2:
            grad_a = _grad_softmax(a, conv, tau, nfft)
        else:
            k_max = int(conv.argmax())
            grad_a = _grad_max(a, conv, k_max)

        # gradient w.r.t. log‑variables (chain rule)
        grad_log = grad_a * a

        # ----------------------------------------------------------
        # Adam update on log‑variables (descent)
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_theta = log_theta - step                     # descent

        # occasional random jitter to shake off plateaus
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * (eta * 0.05)
            log_theta = log_theta + jitter

        # temperature annealing
        tau = max(tau * tau_decay, tau_min)

        # restart if we have not improved for a long time
        if no_improve >= restart_limit:
            # perturb the current best a a little and re‑initialise log‑space
            a = best_a.copy()
            perturb = np.random.randn(n) * (eta * 0.2)
            a = a * np.exp(perturb)          # multiplicative perturbation
            a = a * (S / a.sum())
            log_theta = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{iteration:08d}] Restarted after stagnation.")

    # --------------------------------------------------------------
    # final sanity projection and verification
    # --------------------------------------------------------------
    final_seq = best_a * (S / best_a.sum())
    final_score = evaluate_sequence(final_seq.tolist())
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq.tolist()
