import time
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog

def sqrt_series(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2.0*i - 1.0) / (2.0*i)
    a /= a.sum()
    return a

def project_onto_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def conv_fft(x):
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimizes
    evaluate_sequence(...).  The algorithm alternates between a linear‑program
    improvement step (increase the ℓ₁ norm while keeping the worst convolution
    bound fixed) and a short projected‑gradient refinement.
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------------
    # configuration
    n = kwargs.get('n', 1500)                 # length of the sequence
    max_outer = kwargs.get('max_outer', 12)   # LP iterations
    gradient_steps = kwargs.get('gradient_steps', 150_000)
    # ---------------------------------------------------------------

    # initial sequence: half‑binomial (square‑root) series, already L1‑normalised
    a = sqrt_series(n)          # sum = 1
    conv = conv_fft(a)
    M = conv.max()
    cur_score = 2.0 * n * M
    best_a = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # ---------------------------------------------------------------
    # outer loop: linear‑programming mixing
    outer = 0
    while time.time() < deadline and outer < max_outer:
        # reuse current a and M
        # Build convolution matrix A where (A @ g)[k] = (a * g)[k]
        offsets = np.arange(n, dtype=np.int32)

        A = diags(data, offsets, shape=(2*n-1, n), format='csc')

        b = np.full(2*n-1, M, dtype=np.float64)

        # solve: max sum(g)  <=>  min -sum(g)
        c = -np.ones(n, dtype=np.float64)

            print(f"[outer {outer}] LP failed: {res.message}")
            break

        sum_g = g.sum()
        # If the LP did not increase the ℓ₁‑norm, we are (near) optimal.
        if sum_g <= 1.0 + 1e-12:
            print(f"[outer {outer}] No ℓ₁‑norm increase (sum_g={sum_g:.6e}), stop.")
            break

# Helper utilities (all top‑level, no closures)

def sqrt_series(n: int) -> np.ndarray:
    """
    Coefficients a_i = C(2i,i) / 4**i (the “half‑binomial” series) normalised
    to unit ℓ₁‑norm.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implements Duchi‑et‑al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray) -> np.ndarray:
    """
    Compute the discrete self‑convolution x∗x using a real FFT.
    Result length = 2·len(x)−1.
    """
    n = x.shape[0]
    m = 2 * n - 1
    X = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(X * X, n=m)
    return conv
