def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    
    # choose n based on budget, e.g., n = 4000
    n = 3000  # or maybe dynamic: adjust to budget
    # initial best from grid
    best_seq, best_val = None, np.inf
    offsets = np.linspace(0, 2, 9)  # 0,0.25,...,2
    exponents = np.linspace(0.4, 0.7, 7) #...
    for off in offsets:
        for p in exponents:
            idx = np.arange(1, n+1) + off
            a = idx**(-p)
            a /= a.sum()
            val = fast_score(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
    print(f"[grid] best value {best_val}")
    
    # gradient descent loop
    a = best_seq.copy()
    cur_val = best_val
    step = 0.5  # initial step size factor for movement magnitude
    it = 0
    while time.time() < deadline:
        # compute convolution
        size = 1 << ((2*n-1).bit_length())
        fft_a = np.fft.rfft(a, size)
        conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
        max_b = conv.max()
        cur_val = 2.0 * n * max_b
        # update best
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            print(f"[iter {it}] new best {best_val:.6f}")
        # gradient
        K = conv.argmax()
        # compute gradient
        # For each i in [0,n-1], g[i] = 2*a[K-i] if 0 <= K-i < n else 0
        # Use indexing: reversed = a[::-1]; maybe compute for slice.
        g = np.zeros_like(a)
        # compute valid range
        i_min = max(0, K-(n-1))
        i_max = min(n-1, K)
        # indices i in that range map to K-i
        i_range = np.arange(i_min, i_max+1)
        g[i_range] = 2.0 * a[K - i_range]
        # direction
        d = -g
        sum_d = d.sum()
        d -= sum_d / n  # adjust to zero sum
        # step size adapt
        # compute feasible step_max to keep positivity
        negative = d < 0
        if negative.any():
            max_step = np.min(-a[negative] / d[negative])
            # limit step
            step_allowed = min(step, max_step*0.999)
        else:
            step_allowed = step
        # line search: try step_allowed
        a_new = a + step_allowed * d
        # project onto simplex to enforce nonneg and sum 1
        a_new = project_to_simplex(a_new)
        val_new = fast_score(a_new)
        # accept if improvement
        if val_new < cur_val * (1 - 1e-8):
            a = a_new
            cur_val = val_new
            # maybe increase step a bit
            step = min(step * 1.05, 1.0)
        else:
            # reduce step
            step *= 0.5
        # occasional random perturbation
        if it % 2000 == 0:
            # small random shift
            rho = 0.01
            rand_move = np.random.rand(n)
            rand_move -= rand_move.mean()
            a = project_to_simplex(a + rho * rand_move)
        it += 1
    print("[final] best", best_val)
    return best_seq.tolist()

"""
Propose a low‑value step‑function by combining a parametric search,
exponential‑gradient refinement and a final hill‑climbing phase.

The algorithm proceeds in three stages (all respecting the time budget):

1. **Coarse parametric sweep** – search over the family
      a_i ∝ (i+offset)^{-p}
   for a moderate length `n`.  For each (p,offset) pair the sequence is
   normalised to ℓ¹‑norm 1 and evaluated quickly with FFT.  The best
   candidate from this sweep is used as the starting point.

2. **Multiplicative (exponentiated) gradient descent** – repeatedly
   identify the index `K` where the self‑convolution `b = a*a` attains its
   maximum, compute the sub‑gradient `g_i = 2·a_{K‑i}` (zero where out of
   range) and update
        a_i ← a_i·exp(−η·g_i)    (η adapts automatically)
   followed by renormalisation onto the probability simplex.  This move
   directly reduces the peak of the convolution.

3. **Stochastic hill‑climbing** – fine‑tune the best sequence by moving
   a small amount of mass between two random coordinates.  Only improving
   moves are kept (with a tiny uphill‑acceptance probability to escape
   plateaus).

All three stages keep track of the overall best sequence seen so far.
The routine runs until the prescribed time budget (default 1000 s) is
almost exhausted and then returns the best list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence but without the safety checks)
def fast_score(a: np.ndarray) -> float:
    """Return the evaluation value 2·n·max(conv(a,a))/(sum(a)**2)."""
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    size = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum x=1}."""
    if v.ndim != 1:
        raise ValueError("project_to_simplex works on 1‑D arrays")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find the largest rho satisfying u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def make_shape(n: int, p: float, offset: float) -> np.ndarray:
    """Return the normalised vector a_i ∝ (i+offset)^{-p} for i=1..n."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step sequence with a low evaluation value."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a working length.  The value 3000 works well for the
    #    1/√ kernel and is fast enough for many iterations.
    n = 3000

    # ------------------------------------------------------------------
    # 2. Coarse parametric sweep over (p, offset).
    best_seq = None
    best_val = np.inf

    # reasonable ranges discovered empirically
    p_vals = np.linspace(0.40, 0.65, 26)           # step ≈0.01
    off_vals = np.linspace(0.0, 3.0, 31)          # step ≈0.1

    for p in p_vals:
        for off in off_vals:
            seq = make_shape(n, p, off)
            val = fast_score(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
        # keep an eye on the clock
        if time.time() > deadline - 30:
            break

    print(f"[grid] best value after sweep: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Multiplicative (exponentiated) gradient descent.
    a = best_seq.copy()
    cur_val = best_val
    eta = 0.5                     # initial learning rate
    it = 0
    size_fft = 1 << ((2 * n - 1).bit_length())   # constant FFT length

    while time.time() < deadline - 5:
        # convolution and current objective
        fft_a = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fft_a * fft_a, size_fft)[: 2 * n - 1]
        max_b = conv.max()
        cur_val = 2.0 * n * max_b

        # keep global best
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[grad {it:06d}] new best {best_val:.6f}")

        K = int(conv.argmax())               # index of the peak

        # sub‑gradient of the peak w.r.t. a
        g = np.zeros_like(a)
        i_min = max(0, K - (n - 1))
        i_max = min(n - 1, K)
        i_range = np.arange(i_min, i_max + 1)
        g[i_range] = 2.0 * a[K - i_range]

        # multiplicative update: a_i ← a_i·exp(−η·g_i)
        factor = np.exp(-eta * g)
        a_new = a * factor
        s = a_new.sum()
        if s <= 0:
            # fall back to a smaller step size
            eta *= 0.5
            continue
        a_new /= s

        new_val = fast_score(a_new)

        # accept if we have improved the objective
        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            eta = min(eta * 1.02, 5.0)        # slowly increase step size
        else:
            eta *= 0.7                        # otherwise shrink it

        # occasional tiny random jitter to avoid stagnation
        if it % 5000 == 0 and it > 0:
            jitter = 0.001 * (np.random.rand(n) - 0.5)
            a = project_to_simplex(a + jitter)

        it += 1

    print(f"[grad] finished with best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final stochastic hill‑climbing refinement.
    a = best_seq.copy()
    cur_val = best_val
    step_frac = 0.05                     # relative mass transferred per move
    it_h = 0

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        delta = step_frac * a[i] * np.random.rand()
        if delta > a[i]:
            delta = a[i]

        new_seq = a.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # renormalise (numerical drift may appear)
        s = new_seq.sum()
        if s <= 0:
            continue
        new_seq /= s

        new_val = fast_score(new_seq)

        if new_val < cur_val:
            a = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()
                print(f"[hill {it_h:07d}] NEW BEST {best_val:.6f}")
        else:
            if np.random.rand() < 1e-3:   # rare uphill move
                a = new_seq
                cur_val = new_val

        # adapt step size slowly
        if it_h % 5000 == 0 and it_h > 0:
            step_frac = max(step_frac * 0.9, 1e-5)

        # progress report
        if it_h % 2000 == 0:
            print(f"[hill {it_h:07d}] best={best_val:.6f}, cur={cur_val:.6f}")

        it_h += 1

    print(f"[final] best value found: {best_val:.6f}")

    # Return a plain Python list as required by the harness
    return best_seq.tolist()
