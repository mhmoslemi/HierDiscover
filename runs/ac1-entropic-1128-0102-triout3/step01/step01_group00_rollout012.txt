"""
Two‑phase sub‑gradient optimizer for the step‑function scoring problem.

Phase 1 (exploration) : try a handful of sequence lengths with a small
number of sub‑gradient iterations (momentum + projection onto the simplex)
to discover a promising length and a decent starting point.

Phase 2 (deep optimisation) : continue the optimisation on the best
length for the remaining time budget.  The same momentum‑based
sub‑gradient method is used, but it is allowed to run for as many
iterations as the time limit permits.  The algorithm never exceeds the
given budget and prints intermediate improvements.

All sequences are kept on the probability simplex (non‑negative,
sum = 1), which makes the objective scale‑invariant.
"""

import time
import math
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum(x) = 1}.  Implements the algorithm of Duchi et al.
    (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
