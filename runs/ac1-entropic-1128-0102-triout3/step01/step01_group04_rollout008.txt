def _refine_sequence(seq, deadline):
    # Use several iterations of random pairwise moves guided by max index.
    n = seq.shape[0]
    target = _target_sum(n)
    # ensure normalized
    seq = seq * (target / seq.sum())
    best_seq = seq.copy()
    best_score = _fast_score(best_seq)
    step_factor = seq.mean() * 0.1
    while time.time() < deadline:
        # compute conv and max
        conv = np.fft.irfft(np.fft.rfft(seq, n=2*n-1) * np.fft.rfft(seq, n=2*n-1), n=2*n-1)
        max_val = conv.max()
        m = conv.argmax()
        # compute contributions
        # For i from 0..n-1, j=m-i
        # if valid, contribution = seq[i] * seq[j].
        # We'll choose i with prob proportional to contribution.
        # Build list of contributions
        idxs = np.arange(n)
        js = m - idxs
        mask = (js >=0) & (js < n)
        contrib = np.zeros(n)
        valid_i = idxs[mask]
        contrib[valid_i] = seq[valid_i] * seq[js[mask]]
        if contrib.sum() == 0:
            # fallback random
            i = np.random.randint(n)
        else:
            # sample i weighted by contrib (make probability)
            probs = contrib/ contrib.sum()
            i = np.random.choice(n, p=probs)
        # partner j = m - i
        j = m - i
        # pick a random destination k not equal to i,j
        k = np.random.randint(n)
        while k == i or k == j:
            k = np.random.randint(n)
        # Transfer small amount from i (or j) to k
        # Determine donor: choose i if seq[i] > seq[j] else j
        donor = i if seq[i] > seq[j] else j
        acceptor = k
        max_transfer = min(seq[donor], step_factor)
        if max_transfer <= 0:
            continue
        delta = np.random.rand() * max_transfer
        # apply move
        seq[donor] -= delta
        seq[acceptor] += delta
        # renormalize (should preserve sum but due to rounding)
        total = seq.sum()
        seq *= target / total
        # evaluate
        cand_score = _fast_score(seq)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = seq.copy()
        else:
            # revert move
            seq[donor] += delta
            seq[acceptor] -= delta
            # renorm again
            seq *= target / seq.sum()
        # adapt step_factor: maybe reduce slowly
        step_factor *= 0.999
    return best_seq, best_score

"""
Propose a low‑score step‑function.

The algorithm proceeds in three stages:

1. **Fast analytic sweep** – the family  
   a_i(p,δ) = (i+1+δ)^p – (i+δ)^p  (p∈(0,1), δ≥0) is generated for a
   range of lengths, powers and shifts.  The sequences are normalised to
   the required L¹‑norm `sqrt(2n)` and their score (the maximum of the
   self‑convolution) is obtained with a fast FFT based convolution.
   The best few candidates are retained.

2. **Refinement** – each retained candidate is improved by a stochastic
   pair‑wise mass‑exchange hill‑climber that is aware of the index where
   the convolution attains its maximum.  The move set consists of
   transferring a small amount of mass from one of the two indices that
   contribute most to the current peak to a random low‑mass index.
   The step‑size slowly decays.

3. **Finalisation** – the best sequence found before the time budget
   expires is returned as a plain Python list.  The final score is
   computed with the provided `evaluate_sequence` to guarantee that the
   external validator sees the exact same value.

All heavy work uses FFT (`np.fft`) which gives an *O(n log n)*
convolution and therefore fits comfortably into the 1000 s budget even
for lengths up to a few × 10³.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Required L¹ norm of a sequence of length ``n`` (the normalisation used by the reference code)."""
    return np.sqrt(2.0 * n)


def _analytic_family(n: int, power: float, delta: float) -> np.ndarray:
    """
    Analytic family a_i = (i+1+δ)^p – (i+δ)^p.

    Parameters
    ----------
    n : int
        Length of the sequence.
    power : float
        Exponent 0 < power < 1.
    delta : float
        Non‑negative shift.

    Returns
    -------
    np.ndarray
        Normalised non‑negative sequence of length ``n`` whose sum equals
        ``_target_sum(n)``.
    """
    i = np.arange(n, dtype=np.float64)
    seq = np.power(i + 1.0 + delta, power) - np.power(i + delta, power)
    # Numerical safety – the expression is non‑negative for the ranges we use.
    seq = np.maximum(seq, 0.0)
    scale = _target_sum(n) / seq.sum()
    return seq * scale


def _max_convolution_fft(seq: np.ndarray) -> float:
    """
    Return max_{k} (seq * seq)[k] using an FFT based convolution.

    The function assumes that ``seq`` is already normalised to the target
    L¹ norm; the returned value equals the score that ``evaluate_sequence``
    would output.
    """
    n = seq.shape[0]
    # Length of linear convolution = 2*n-1
    length = 2 * n - 1
    # Compute the discrete Fourier transform of the zero‑padded sequence
    fft_seq = np.fft.rfft(seq, n=length)
    # Convolution in time domain = product in frequency domain
    conv = np.fft.irfft(fft_seq * fft_seq, n=length)
    return float(conv.real.max())


def _fast_score(seq: np.ndarray) -> float:
    """Convenient wrapper that normalises ``seq`` and returns the fast score."""
    n = seq.shape[0]
    target = _target_sum(n)
    if seq.sum() == 0:
        return np.inf
    seq = seq * (target / seq.sum())
    return _max_convolution_fft(seq)


def _pairwise_hill_climb(start_seq: np.ndarray,
                         max_iter: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that transfers a small amount of mass between
    two indices.  The move set is biased towards the indices that
    contribute most to the current convolution peak.

    Parameters
    ----------
    start_seq : np.ndarray
        Normalised starting sequence (sum = sqrt(2n)).
    max_iter : int
        Upper bound on the number of iterations.
    deadline : float
        Absolute time (``time.time()``) after which the routine aborts.

    Returns
    -------
    tuple[np.ndarray, float]
        The best sequence encountered and its (fast) score.
    """
    seq = start_seq.copy()
    n = seq.shape[0]
    target = _target_sum(n)

    # current best
    best_seq = seq.copy()
    best_score = _fast_score(seq)

    # initial step size – a fraction of the mean entry
    step_sigma = max(1e-12, seq.mean() * 0.1)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # --- locate the current maximum of the convolution -----------------
        length = 2 * n - 1
        fft_seq = np.fft.rfft(seq, n=length)
        conv = np.fft.irfft(fft_seq * fft_seq, n=length)
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # --- pick a donor index – weighted by contribution to the peak -----
        i = np.arange(n, dtype=int)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(seq)
        contrib[mask] = seq[i[mask]] * seq[j[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            donor = np.random.choice(n, p=probs)
        else:
            donor = np.random.randint(n)

        # --- pick a random receiver (different from donor) -----------------
        receiver = np.random.randint(n)

Propose a low‑score step‑function.

The algorithm has three stages:

1️⃣ **Fast analytic sweep** – the family  
   a_i(p,δ) = (i+1+δ)^p – (i+δ)^p  (0<p<1, δ≥0) is evaluated for a
   range of lengths, powers and shifts.  Each candidate is normalised to
   the required L¹‑norm `sqrt(2·n)` and its score (the maximum value of the
   self‑convolution) is obtained with an FFT‑based convolution, which is
   O(n log n).  The best few candidates are kept.

2️⃣ **Refinement** – every kept candidate is improved by a stochastic
   pair‑wise mass‑exchange hill‑climber that is biased toward the indices
   that contribute most to the current convolution peak.  The move set
   consists of transferring a small amount of mass from a “donor” index
   (chosen proportionally to its contribution to the peak) to a random
   “receiver”.  The step‑size slowly decays.

3️⃣ **Finalisation** – the best sequence found before the time budget
   expires is returned as a plain Python list.  The final score is
   computed with the provided `evaluate_sequence` to guarantee compatibility
   with the external grader.

All heavy work uses FFT (`np.fft`) which easily fits the 1000 s budget
even for sequences of a few thousand entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Required L¹ norm of a sequence of length ``n`` (the normalisation used by the reference code)."""
    return np.sqrt(2.0 * n)


def _analytic_family(n: int, power: float, delta: float) -> np.ndarray:
    """
    Analytic family a_i = (i+1+δ)^p – (i+δ)^p.

    Parameters
    ----------
    n : int
        Length of the sequence.
    power : float
        Exponent 0 < power < 1.
    delta : float
        Non‑negative shift.

    Returns
    -------
    np.ndarray
        Normalised non‑negative sequence of length ``n`` whose sum equals
        ``_target_sum(n)``.
    """
    i = np.arange(n, dtype=np.float64)
    seq = np.power(i + 1.0 + delta, power) - np.power(i + delta, power)
    seq = np.maximum(seq, 0.0)                     # safety
    scale = _target_sum(n) / seq.sum()
    return seq * scale


def _max_convolution_fft(seq: np.ndarray) -> float:
    """
    Compute max_k (seq * seq)[k] using FFT.
    The sequence is assumed to already satisfy the normalisation
    sum(seq) = sqrt(2·n).  The returned value equals the score that
    ``evaluate_sequence`` would output.
    """
    n = seq.shape[0]
    length = 2 * n - 1
    fft_seq = np.fft.rfft(seq, n=length)
    conv = np.fft.irfft(fft_seq * fft_seq, n=length)
    return float(conv.real.max())


def _fast_score(seq: np.ndarray) -> float:
    """Normalise ``seq`` to the required L¹ norm and return its fast score."""
    n = seq.shape[0]
    target = _target_sum(n)
    if seq.sum() == 0:
        return np.inf
    seq = seq * (target / seq.sum())
    return _max_convolution_fft(seq)


def _pairwise_hill_climb(start_seq: np.ndarray,
                         max_iter: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a small amount of mass between a donor
    (chosen among the two indices that contribute most to the current
    convolution peak) and a random receiver.

    Parameters
    ----------
    start_seq : np.ndarray
        Normalised starting sequence (sum = sqrt(2·n)).
    max_iter : int
        Upper limit on the number of iterations.
    deadline : float
        Absolute ``time.time()`` after which the routine aborts.

    Returns
    -------
    tuple[np.ndarray, float]
        Best sequence encountered and its fast score.
    """
    seq = start_seq.copy()
    n = seq.shape[0]
    target = _target_sum(n)

    best_seq = seq.copy()
    best_score = _fast_score(seq)

    # step size – a modest fraction of the mean entry
    step_sigma = max(1e-12, seq.mean() * 0.1)

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # locate the current maximum of the convolution
        # --------------------------------------------------------------
        length = 2 * n - 1
        fft_seq = np.fft.rfft(seq, n=length)
        conv = np.fft.irfft(fft_seq * fft_seq, n=length)
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # pick a donor index – weighted by its contribution to the peak
        # --------------------------------------------------------------

        contrib[mask] = seq[i[mask]] * seq[j[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            donor = np.random.choice(n, p=probs)
        else:
