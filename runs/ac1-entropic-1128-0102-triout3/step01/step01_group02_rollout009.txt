        g = res.x

def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type numbers  c_i = C(2i,i)/4^i  as a NumPy array."""
    seq = np.empty(n, dtype=float)
    seq[0] = 1.0
    # recurrence  c_{i+1} = c_i * (2*i+1)/(2*i+2)
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def _beta_sequence(n: int, beta: float) -> np.ndarray:
    """
    Symmetric Beta‑like weights:  w_i = ((i+1)*(n-i))**beta .
    For beta<0 the distribution concentrates near the two ends.

    i = np.arange(n, dtype=float)
    w = ((i + 1) * (n - i)) ** beta
    return w

    # ---- symmetric Beta‑like family ----------------------------------- #
    for beta in np.linspace(-1.0, -0.05, 20):
        if time.time() - start > 0.45 * budget_s:
            break
        w = _beta_sequence(n, beta)
        seq = w / w.sum()
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            p = seq.copy()
            print(f"[beta] beta={beta:.3f} score={score:.6f}")

    # ------------------------------------------------------------------ #
    # 4. One Frank‑Wolfe / LP refinement (if SciPy is present) -------- #
    # ------------------------------------------------------------------ #
    if _SCIPY_AVAILABLE:
        # Build the convolution matrix A where (A @ g)_k = (p * g)[k]
        # A is a (2n‑1)×n Toeplitz matrix with constant diagonals = p_i.
        diag_vals = [np.full(n, p[i], dtype=float) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        A = _sp.diags(diag_vals, offsets, shape=(2 * n - 1, n), format='csr')

        # Current maximal convolution value
        cur_conv = np.convolve(p, p)
        M = cur_conv.max()

        # Linear programme: maximise sum(g)  ⇔  minimise -∑g
        c = -np.ones(n, dtype=float)                       # objective
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # Solve (may take a fraction of a second)
        try:
            res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                               method='highs', options={'presolve': True})
            if res.success:
                g = res.x
                # Normalise g and mix it into the current p
                g = g / g.sum()
                mix = 0.25                                     # small step size
                p = (1.0 - mix) * p + mix * g
                p = p / p.sum()
                score = evaluate_sequence(p.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = p.copy()
                    print(f"[LP] improved score={score:.6f}")
            else:
                print("[LP] linprog failed:", res.message)
        except Exception as exc:                              # pragma: no cover
            print("[LP] exception:", exc)

    # ------------------------------------------------------------------ #
    # 5. Stochastic hill‑climbing (mass transfer from early to late) ---- #
    # ------------------------------------------------------------------ #
    cur = p.copy()
    cur_score = best_score
    iteration = 0
    # Define regions for source (early) and target (late)
    early_max = n // 4
    late_min = 3 * n // 4

    while time.time() - start < 0.92 * budget_s:
        iteration += 1
        # pick a source index in the early quarter, a target in the late quarter
        i = rng.integers(0, early_max)
        j = rng.integers(late_min, n)
        if cur[i] == 0.0:
            continue
        # transfer a random fraction (up to 30 %) of the source mass
        delta = cur[i] * rng.random() * 0.3
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # evaluate – the array is tiny, so a full convolution is fine
        score = evaluate_sequence(new.tolist())
        # accept strictly improving moves; occasional uphill moves help escape plateaus
        if score < cur_score or rng.random() < 0.001:
            cur = new
            cur_score = score
            if score < best_score:
                best_score = score
                best_seq = new.copy()
                print(f"[hill] it={iteration} new best={score:.6f}")

    print(f"[final] iterations={iteration} best score={best_score:.6f}")
    # Clip to the limits enforced by evaluate_sequence (unnecessary here but safe)
    final_seq = [float(min(max(v, 0.0), 1000.0)) for v in best_seq.tolist()]
    return final_seq
