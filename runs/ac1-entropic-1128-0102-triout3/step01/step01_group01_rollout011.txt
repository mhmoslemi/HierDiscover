            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
            print(f"[{iteration}] new best {best_score:.6f}")

            no_improve += 1

        if best_score < 1.503:
            print("Target reached")
            break

        # Build constraint matrix A
        # Use sparse diagonal representation
        offsets = np.arange(n, dtype=int)
        A = sp.diags([f] * n, offsets, shape=(2*n-1, n), format='csr')
        # RHS vector
        b = np.full(2*n-1, cur_max, dtype=np.float64)

        # Solve LP: maximize sum(g)
        c = -np.ones(n, dtype=np.float64)  # minimize -sum(g)
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,

"""
Alternating linear‑programming + line‑search optimizer.

The objective of `evaluate_sequence` is
    score = 2·n·max(conv(a,a)) / (sum(a)^2) = max(conv(a,a))
when the ℓ₁‑norm is fixed to √(2n).  Hence we want to lower the
maximum entry of the linear convolution a * a.

The algorithm follows the scheme described in the statement:

* start from a feasible non‑negative simplex point (sum = √(2n)).
* fix the current maximum M = max(conv(f,f)).
* solve a linear program that maximises the total mass of a new
  vector g under the linear constraints   f * g ≤ M  (for every shift).
  This pushes mass into positions where f is small.
* rescale g to satisfy the ℓ₁‑norm constraint.
* perform a short line‑search in the convex combination (1–t)·f + t·g
  and keep the best point.
* repeat until the time budget expires, occasional random restarts
  avoid stagnation.

The LP has n variables (≈2400) and 2n‑1 constraints and is solved
with the HiGHS interior‑point method via ``scipy.optimize.linprog``.
All heavy work (FFT‑based convolution, simplex projection) is
implemented with NumPy/SciPy; the whole routine respects the 1000 s
budget and returns the best sequence found.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolve_self_fft(a: np.ndarray, nfft: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the first (2*len(a)-1) entries (real values).
    """
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * a.size - 1]
    return conv


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = target_sum}.
    Implements the classic O(n) algorithm of Duchi et al.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_toeplitz_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2n-1, n)) such that
         A @ g = conv(f, g)
    for linear convolution.  Row k corresponds to shift k.
    """
    n = f.shape[0]
    # Each diagonal i (offset i) contains the constant value f[i].
    # Offsets run from 0 to n-1.
    offsets = np.arange(n, dtype=int)
    # SciPy's ``diags`` interprets a list of scalars as constant diagonals.
    # ``list(f)`` provides one scalar per offset.
    A = sp.diags(list(f), offsets, shape=(2 * n - 1, n), format="csr")
    return A


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # 1.   Choose problem size and initialise the current point.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=np.float64)
    else:
        # a moderate size that still offers many degrees of freedom
        init = None

    if init is None:
        n = 2001                     # fallback if no global is present
        init = np.full(n, 0.0, dtype=np.float64)
    else:
        n = len(init)

    # target ℓ₁‑norm (the normalisation used in the original code)
    S = np.sqrt(2.0 * n)

    # start from a feasible point – project onto the simplex of sum S
    f = _project_to_simplex(init, S)

    # bookkeeping for the best solution ever seen
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"Initial score: {best_score:.6f}")

    # pre‑compute FFT length for convolution
    nfft = _next_pow2(2 * n - 1)

    # algorithmic parameters
    iteration = 0
    no_improve = 0
    max_no_improve = 500           # trigger a random restart after this many stalls
    line_search_ts = np.linspace(0.0, 1.0, 11)   # 0,0.1,…,1.0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 2.   Current convolution and its maximum (the true objective)
        # --------------------------------------------------------------
        conv_f = _convolve_self_fft(f, nfft)
        cur_max = conv_f.max()
        cur_score = cur_max                     # evaluate_sequence == cur_max when sum=S

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
            print(f"[{iteration:06d}] New best: {best_score:.6f}")
        else:
            no_improve += 1

        # early exit if we have already crossed the target bound
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration}")
            break

        # --------------------------------------------------------------
        # 3.   Build the linear constraints:  f * g ≤ cur_max  (for all shifts)
        # --------------------------------------------------------------
        A = _build_toeplitz_constraint_matrix(f)
        b = np.full(2 * n - 1, cur_max, dtype=np.float64)

        # --------------------------------------------------------------
        # 4.   Linear program: maximise 1ᵀ g   (i.e. maximise total mass)
        #      subject to A g ≤ b ,   g ≥ 0
        # --------------------------------------------------------------
        c = -np.ones(n, dtype=np.float64)   # linprog minimises cᵀx → maximise sum(g)
        bounds = [(0, None)] * n

        # solve with HiGHS; we hand a time limit to avoid huge runtimes
        remaining = max(0.0, deadline - time.time())
        try:
            res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                              method="highs",
                              options={"disp": False, "time_limit": remaining})
        except Exception as e:
            # fallback: jitter the current point and continue
            print(f"[{iteration}] LP failed ({e}); applying jitter.")
            jitter = np.random.randn(n) * (S / n) * 0.1
            f = _project_to_simplex(f + jitter, S)
            iteration += 1
            continue

        if not res.success:
            # LP infeasible or numerically unstable → jitter
            print(f"[{iteration}] LP not successful (status {res.status}); jitter.")
            jitter = np.random.randn(n) * (S / n) * 0.1
            f = _project_to_simplex(f + jitter, S)
            iteration += 1
            continue

        g0 = np.maximum(res.x, 0.0)        # clip tiny negatives possibly produced
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            # degenerate solution – random perturbation
            jitter = np.random.randn(n) * (S / n) * 0.1
            f = _project_to_simplex(f + jitter, S)
            iteration += 1
            continue

        # --------------------------------------------------------------
        # 5.   Scale g to satisfy the ℓ₁‑norm exactly (sum = S)
        # --------------------------------------------------------------
        g = g0 * (S / sum_g0)

        # --------------------------------------------------------------
        # 6.   Simple line‑search over convex combinations (1‑t)·f + t·g
        # --------------------------------------------------------------
        best_local_score = cur_score
        best_candidate = f
        best_t = 0.0

        for t in line_search_ts:
            if t == 0.0:
                continue
            cand = (1.0 - t) * f + t * g
            # (convex combination of two feasible points stays non‑negative)
            cand_conv = _convolve_self_fft(cand, nfft)
            cand_max = cand_conv.max()
            if cand_max < best_local_score:
                best_local_score = cand_max
                best_candidate = cand
                best_t = t

        if best_t > 0.0:
            f = best_candidate
        else:
            # no improvement – add a tiny random jitter to escape plateaus
            jitter = np.random.randn(n) * (S / n) * 0.05
            f = _project_to_simplex(f + jitter, S)

        # --------------------------------------------------------------
        # 7.   Random restart if we have been stuck for many iterations
        # --------------------------------------------------------------
        if no_improve >= max_no_improve:
            alpha = np.full(n, 0.3)          # Dirichlet concentration
            f = np.random.dirichlet(alpha) * S
            no_improve = 0
            print(f"[{iteration}] Restarted from a random simplex point.")

        iteration += 1

    # ------------------------------------------------------------------
    # 8.   Return the best sequence discovered
    # ------------------------------------------------------------------
    final_seq = best_f.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return final_seq
