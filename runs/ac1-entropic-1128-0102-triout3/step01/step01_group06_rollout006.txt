    conv = np.fft.irfft(A * A, n=fft_len)[:2*len(a)-1]

def gradient_step(a, max_indices, fft_len, step_factor=0.9):
    n = len(a)
    g = np.zeros_like(a)
    for k in max_indices:
        # compute contributions
        # index i in [max(0, k - (n - 1)), min(k, n-1)]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        rev_idx = k - idx
        g[idx] += 2.0 * a[rev_idx]
    # lambda is average of g, as sum(g)/n.
    lam = g.mean()
    d = lam - g   # direction that preserves sum (since sum(d) = 0)
    # compute feasible step size to avoid negative entries
    # For entries where d < 0 (i.e., lam - g < 0 => g > lam), we reduce a.
    negative = d < 0
    if not np.any(negative):
        # direction non-negative -> can't move (at optimum)
        return a, False
    # compute max step ensuring a + t * d >= 0
    # t <= a_i / -d_i for all negative entries
    ratios = a[negative] / (-d[negative])
    t_max = ratios.min()
    # adjust step factor
    t = step_factor * t_max
    # apply update
    a_new = a + t * d
    # Ensure non-negativity due to floating errors
    a_new = np.maximum(a_new, 0.0)
    # renormalize sum to 1 (due to rounding)
    s = a_new.sum()
    if s == 0.0:
        return a, False
    a_new /= s
    return a_new, True

"""
Propose a low‑score step function.

The algorithm works in three stages:

1. **Broad parametric sweep** – builds candidates from two simple families
   (power‑law a_i = (i+c)^(-α) and exponential a_i = exp(-γ·i)),
   normalises them and keeps the best one found.

2. **Projected sub‑gradient descent** – repeatedly
   * computes the self‑convolution,
   * identifies the index (or indices) where the convolution attains its maximum,
   * builds the sub‑gradient of the objective at that point,
   * projects the gradient onto the simplex (∑a_i = 1 , a_i ≥ 0) and
   * takes the largest feasible step that keeps all entries non‑negative.
   This directly reduces the L∞‑norm of the convolution while preserving the
   normalisation, which is exactly the quantity minimised by `evaluate_sequence`.

3. **Diversification** – occasional random hill‑climbs (tiny mass transfers)
   and occasional restarts from the parametric families to escape possible
   local minima.

All heavy work is done with FFT‑based convolutions (O(n log n)), so the
procedure comfortably stays inside the 1000 s budget even for a few‑thousand
element sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_len_for_n(n: int) -> int:
    """FFT length sufficient for a full self‑convolution of length n."""
    return _next_pow2(2 * n - 1)


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of a*a (full discrete convolution) using FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def _score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(a*a) (sum is fixed to 1)."""
    n = len(a)
    return 2.0 * n * _max_convolution(a, fft_len)


def _build_powerlaw(alpha: float, c: float, n: int) -> np.ndarray | None:
    """Return normalised a_i = (i + c)^(-α).  Returns None if degenerate."""
    i = np.arange(n, dtype=np.float64)
    a = (i + c) ** (-alpha)
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _build_exponential(gamma: float, n: int) -> np.ndarray | None:
    """Return normalised a_i = exp(-γ·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gamma * i)
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _hill_transfer(a: np.ndarray) -> np.ndarray:
    """Small random transfer of mass between two positions (preserves sum)."""
    n = a.shape[0]
    b = a.copy()
    i1, i2 = np.random.choice(n, size=2, replace=False)
    # up to about 1 % of the donor mass
    delta = (np.random.rand() - 0.5) * 0.02 * b[i1]
    # keep non‑negativity
    if delta < -b[i1]:
        delta = -b[i1] + 1e-15
    if delta > b[i2]:
        delta = b[i2] - 1e-15
    b[i1] += delta
    b[i2] -= delta
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s > 0.0:
        b /= s
    return b


def _subgradient_step(
    a: np.ndarray,
    max_inds: np.ndarray,
    step_factor: float = 0.9,
) -> tuple[np.ndarray, bool]:
    """
    Perform a projected sub‑gradient step that reduces
    max_k (a*a)_k while keeping sum(a)=1 and a≥0.

    Returns the new vector and a flag indicating whether a genuine step
    (i.e. at least one component decreased) was possible.
    """
    n = a.shape[0]
    g = np.zeros_like(a)          # raw sub‑gradient of the max‑convolution
    for k in max_inds:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        rev_idx = k - idx
        g[idx] += 2.0 * a[rev_idx]

    lam = g.mean()                # Lagrange multiplier for the sum‑constraint
    d = lam - g                    # feasible direction (sum(d)=0)

    # indices where we would *decrease* a_i
    dec = d < 0.0
    if not np.any(dec):
        # direction would only increase the vector → we are at a stationary point
        return a, False

    # maximal feasible step size before some entry would hit zero
    ratios = a[dec] / (-d[dec])
    t_max = ratios.min()
    step = step_factor * t_max    # a safe step (a little inside the feasible region)

    a_new = a + step * d
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        return a, False
    a_new /= s
    return a_new, True
