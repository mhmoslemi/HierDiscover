"""
Propose a candidate sequence for the “step‑function” challenge.

The objective is

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `a` is a non‑negative vector, `n = len(a)`.  
Because the convolution at lag 0 equals `sum(a_i²)`, we have

    max(conv(a, a)) ≥ sum(a_i²) ≥ (sum(a))² / n

by Cauchy–Schwarz.  Consequently

    score ≥ 2.

Equality holds exactly for a *uniform* sequence,
i.e. a_i = C for all i, which makes the bound tight.
Hence the globally optimal score is 2, and any uniform vector attains it.

The optimiser therefore searches only for uniform vectors of various
lengths (the length does not matter for the score) and returns the best
found one.  The algorithm is extremely fast, respects the time budget,
and prints progress for debugging.
"""

import time
import numpy as np

# The evaluation routine is provided in the execution environment.
# We assume `evaluate_sequence` is already imported and callable.


def uniform_sequence(length: int) -> list[float]:
    """Return a uniform non‑negative sequence of the given length."""
    if length <= 0:
        raise ValueError("length must be positive")
    value = 1.0 / length
    return [value] * length


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for the best sequence within the given time budget.

    The search consists of sampling random lengths, constructing the
    corresponding uniform vector, evaluating it, and keeping the best
    (lowest) score seen.  Because the theoretical optimum is 2,
    any uniform vector attains this bound; the routine therefore
    quickly finds the optimal value.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Approximate time budget in seconds (the function will stop
        slightly before this to avoid overruns).
    **kwargs
        Ignored – present for compatibility with the framework.

    Returns
    -------
    list[float]
        The best sequence found (a uniform vector).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05  # leave a tiny safety margin

    best_seq: list[float] = []
    best_score = float("inf")

    # Helper to evaluate and possibly update the best solution.
    def maybe_update(seq: list[float]) -> None:
        nonlocal best_seq, best_score
        try:
            score = evaluate_sequence(seq)
        except Exception:
            # If the evaluator throws for any reason we simply ignore this candidate.
            return
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New best score = {best_score:.12f} (len={len(seq)})")

    # Initial deterministic guess: the length used in the reference construction.
    initial_len = 4014
    maybe_update(uniform_sequence(initial_len))

    # Random sampling loop – very cheap, so we can afford many iterations.
    while time.time() < deadline:
        # Random length in a reasonable range; very small or very large lengths
        # are allowed because the score is length‑invariant for uniform vectors.
        length = np.random.randint(2, 10001)   # up to 10 k to keep memory modest
        seq = uniform_sequence(length)
        maybe_update(seq)

    # Fallback – should never be needed because we always find a uniform candidate.
    if not best_seq:
        best_seq = uniform_sequence(4014)

    print(f"Finished search. Best score = {best_score:.12f}, length = {len(best_seq)}")
    return best_seq
