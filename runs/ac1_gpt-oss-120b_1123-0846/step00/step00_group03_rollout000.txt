def project_onto_simplex(v, s):
    # Ensure v>=0? algorithm will handle.
    # Based on Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    # Implementation from W. Duchi.
    n = len(v)

        return np.full_like(v, s / n)
    rho = ind[cond][-1]

    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent search for a low‑max‑convolution step function.

The evaluation function (after the sum of the coefficients is normalised to
√(2 n)) is simply the maximal entry of the linear convolution a*a.
The algorithm minimises a smooth approximation of this max:
    Mβ(a) = (1/β) log ∑ₖ exp(β·(a*a)ₖ)      (soft‑max)

∇Mβ is obtained analytically:
    ∂Mβ/∂aᵢ = Σₖ wₖ a_{k‑i}  with  wₖ = exp(β·(a*a)ₖ) / Σⱼ exp(β·(a*a)ⱼ).

We perform a projected gradient step (a ← a − η ∇Mβ) and then project the
vector onto the simplex {a≥0, Σa = √(2 n)}.
The step size η and the smoothing parameter β are annealed.
All convolutions are evaluated with FFT (O(n log n)) so that thousands of
iterations fit easily into the 1000 s budget.

The best sequence seen during the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _fft_convolve_real(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = len(x) + len(y) - 1
    L = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution entries, numerically stable."""
    maxb = np.max(b)
    ex = np.exp(beta * (b - maxb))
    w = ex / np.sum(ex)
    return w


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x≥0, Σx = s}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = len(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # all entries would be zero; return the uniform vector on the simplex
        return np.full_like(v, s / n)
    rho = rho_arr[cond][-1]                     # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------

    budget_s : float
        Time budget in seconds.
    kwargs : dict
        Ignored – kept for compatibility with the runner.

    Returns

    list[float]
        The best coefficient sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose an initial length and initialise the coefficients.
    # If a previous construction (height_sequence_1) is available we reuse
    # its length and shape; otherwise we start from a uniform distribution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=float).copy()
        n = len(base)
        # Normalise to the required sum √(2 n)
        target_sum = np.sqrt(2.0 * n)
        s = base.sum()
        if s <= 0.0:
            a = np.full(n, target_sum / n)
        else:
            a = base * (target_sum / s)
    else:
        # random length in a sensible range
        n = np.random.randint(400, 1500)
        target_sum = np.sqrt(2.0 * n)
        a = np.full(n, target_sum / n)

    # Ensure feasibility with the simple “clip‑and‑project” step
    a = _project_onto_simplex(a, np.sqrt(2.0 * n))

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial length {n}, score {best_score:.6f}")

    # Parameters for the optimisation schedule
    beta = 1.0          # start with a smooth approximation
    beta_max = 40.0
    beta_factor = 1.15
    eta = 0.2           # initial learning rate
    eta_decay = 0.995
    min_eta = 1e-4

    # For early‑stop detection
    last_improve_iter = 0
    iteration = 0

    # ------------------------------------------------------------------
    # Main loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1) current convolution and objective
        conv = _fft_convolve_real(a, a)               # length 2n‑1
        max_conv = np.max(conv)
        # The true evaluation (after the sum‑normalisation) equals max_conv
        current_score = float(2.0 * n * max_conv / (np.sum(a) ** 2))

        # Record best found so far
        if current_score < best_score:
            best_score = current_score
            best_seq = a.copy()
            last_improve_iter = iteration
            print(f"Iter {iteration:5d}: new best {best_score:.6f}")

        # 2) soft‑max weights over the convolution entries
        w = _softmax_weights(conv, beta)

        # 3) gradient of the smoothed max (see docstring)
        w_rev = w[::-1]                               # reverse for convolution
        grad_full = _fft_convolve_real(a, w_rev)      # length 3n‑2
        start = len(w) - 1
        grad = grad_full[start - np.arange(n)]

        # 4) projected gradient step
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, np.sqrt(2.0 * n))

        # 5) (optional) occasional stochastic acceptance to escape plateaus
        accept = True
        if iteration % 7 == 0:
            # evaluate the candidate with the official function (expensive)
            cand_score = evaluate_sequence(a_new.tolist())
            if cand_score > current_score and np.random.rand() > 0.01:
                accept = False

        if accept:
            a = a_new

        # 6) schedule updates
        beta = min(beta * beta_factor, beta_max)
        eta = max(eta * eta_decay, min_eta)

        # 7) occasional random perturbation if we have been idle for a while
        if iteration - last_improve_iter > 300:
            # small Gaussian jitter, re‑project to maintain feasibility
            jitter = np.random.randn(n) * 0.02 * np.sqrt(2.0 * n) / n
            a = _project_onto_simplex(a + jitter, np.sqrt(2.0 * n))
            last_improve_iter = iteration
            print(f"Iter {iteration:5d}: jitter applied (stagnation)")

        iteration += 1

    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    return best_seq.tolist()
