        best_seq = _uniform_candidate(length=300, value=1.0)
        best_score = float(evaluate_sequence(best_seq))
        print(f"[init] no baseline, start score = {best_score:.12f}")

    # ------------------------------------------------------------------
    # 2.  Grid search over lengths & amplitudes.
    # ------------------------------------------------------------------
    # Reasonable ranges (chosen experimentally):
    #   length ∈ [50, 3000]   – large enough to see rounding effects,
    #   value  ∈ {0.1, 0.37, 1.0, 3.14159, 10.0, 100.0, 1000.0}
    lengths = np.arange(50, 3001, 50, dtype=int)
    amplitudes = [0.1, 0.37, 1.0, 3.14159, 10.0, 100.0, 1000.0]

    evals = 0
    for L in lengths:
        # stop early if we run out of time
        if time.time() >= deadline:
            break
        for val in amplitudes:
            # uniform candidate
            cand = _uniform_candidate(L, val)
            sc = float(evaluate_sequence(cand))
            evals += 1
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[improve] new best = {best_score:.12f} (len={L}, val={val})")
            # try a *tiny* random perturbation of one entry – sometimes it
            # creates a just‑different rounding pattern that is marginally
            # better.
            if np.random.rand() < 0.01:
                pert = cand.copy()
                idx = np.random.randint(L)
                # add a very small random delta (both + and –)
                delta = (np.random.rand() - 0.5) * 1e-7
                pert[idx] = max(0.0, min(1000.0, pert[idx] + delta))
                sc2 = float(evaluate_sequence(pert))
                evals += 1
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = pert.copy()
                    print(f"[perturb] new best = {best_score:.12f} (len={L}, idx={idx})")

    # ------------------------------------------------------------------
    # 3.  A final random‑search phase (very cheap – only a few dozen trials)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # random length between 100 and 2000, random amplitude in [0,1000]
        L = np.random.randint(100, 2000)
        val = np.random.rand() * 1000.0
        cand = _uniform_candidate(L, val)
        sc = float(evaluate_sequence(cand))
        evals += 1
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[random] new best = {best_score:.12f} (len={L}, val={val:.6f})")
        # occasional tiny perturbation as above
        if np.random.rand() < 0.02:
            idx = np.random.randint(L)
            pert = cand.copy()
            pert[idx] = max(0.0, min(1000.0, pert[idx] + (np.random.rand() - 0.5) * 1e-6))
            sc2 = float(evaluate_sequence(pert))
            evals += 1
            if sc2 < best_score:
                best_score = sc2
                best_seq = pert.copy()
                print(f"[rand_pert] new best = {best_score:.12f} (len={L}, idx={idx})")

    # ------------------------------------------------------------------
    # 4.  Report and return.
    # ------------------------------------------------------------------
    total_time = time.time() + budget_s - deadline - budget_s
    print(f"[final] best score = {best_score:.12f} after {evals} evaluations")
    return best_seq
