    # -------------------------------------------------------------
    try:
        score = evaluate_sequence(seq)   # evaluate_sequence is provided in the environment
        print(f"[propose_candidate] Generated triangular sequence of length {n} – score = {score:.6f}")
    except Exception as exc:   # pragma: no cover
        # In the unlikely event evaluate_sequence raises (e.g., due to a bug),
        # we still return the sequence; the surrounding system will handle the
        # infinite score.
        print("[propose_candidate] Evaluation failed:", exc)

    # -------------------------------------------------------------
    # Return the constructed sequence.
    # -------------------------------------------------------------
    return seq
