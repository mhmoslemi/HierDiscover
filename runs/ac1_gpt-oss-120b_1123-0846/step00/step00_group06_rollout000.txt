"""
Propose a candidate step‑function sequence.

The algorithm searches many random initializations (different lengths
between 200 and 2000).  For each length a non‑negative vector is drawn,
scaled to the required L1‑norm √(2 n) and then refined by a simple
projected gradient descent that tries to make the self‑convolution as
flat as possible (minimising the L2 distance to the constant average).
The projection keeps the vector on the simplex (non‑negative, fixed sum).
Whenever a better sequence (lower evaluate_sequence value) is found it
is stored as the current global best.  The search stops when the
allocated time budget expires (default 1000 s) and returns the best
sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def projection_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {w >= 0, sum(w) = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the largest rho where u_i > cssv_i / i
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def scale_to_target_sum(a: np.ndarray) -> np.ndarray:
    """Scale non‑negative vector a so that sum(a)=√(2·len(a))."""
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    total = a.sum()
    if total == 0.0:
        # uniform distribution as a fallback
        return np.full(n, target / n)
    return a * (target / total)


def evaluate_max_conv(a: np.ndarray) -> float:
    """Return the evaluation function value for a (peak of a*a)."""
    # evaluate_sequence already implements the same logic, but calling it
    # directly is slower because of the safety checks.  We compute the
    # peak directly which is equivalent after the scaling.
    b = np.convolve(a, a)
    max_b = b.max()
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


def gradient_flatten_conv(a: np.ndarray) -> np.ndarray:
    """
    Gradient (w.r.t. a) of the L2 loss
        ½‖a*a – avg‖²  , where avg = sum(a)²/(2n‑1).

    The expression can be written compactly using a convolution:
        g = 2·conv(err, a_rev)[n‑1 : n‑1+n]
    where err = a*a – avg (vector of length 2n‑1) and a_rev is a reversed.
    """
    n = a.shape[0]
    # target average of convolution entries (when sum = √(2n))
    target_sum = math.sqrt(2.0 * n)
    avg = (target_sum ** 2) / (2.0 * n - 1.0)

    conv_aa = np.convolve(a, a)                # length 2n‑1
    err = conv_aa - avg                        # same length
    a_rev = a[::-1]
    # full convolution err * a_rev : length 3n‑2
    full = np.convolve(err, a_rev)
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def improve_sequence(a_init: np.ndarray, deadline: float,
                     max_iters: int = 3000) -> tuple[list[float], float]:
    """
    Projected gradient descent that flattens the self‑convolution.
    Returns the best sequence (as a Python list) and its evaluation score.
    """
    a = a_init.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_score = evaluate_max_conv(best_a)

    # initial step scaling factor (relative to the simplex size)
    step_scale = 0.1 * target_sum

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # gradient of flat‑convolution objective
        g = gradient_flatten_conv(a)

        norm_g = np.linalg.norm(g)
        if norm_g < 1e-12:
            break  # stuck

        # step size chosen so that the update magnitude is roughly step_scale
        step = step_scale / norm_g
        a_new = a - step * g

        # keep feasibility (non‑negative + fixed sum)
        a_new = projection_onto_simplex(a_new, target_sum)

        # evaluate the true objective (peak of convolution)
        cur_score = evaluate_max_conv(a_new)

        # keep the best ever seen in this run
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # shrink the step scale slowly when we are making progress
            step_scale *= 0.99
        else:
            # if no improvement, decay the step scale more aggressively
            step_scale *= 0.95

        a = a_new

    return best_a.tolist(), float(best_score)


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function stops ~1 s before this).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small margin for the final return
    deadline = start_time + float(budget_s) - 1.0

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Use the known construction (if available) as a starting point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            seq0 = np.asarray([float(x) for x in height_sequence_1], dtype=float)
            seq0 = scale_to_target_sum(seq0)               # enforce the required sum
            score0 = evaluate_max_conv(seq0)
            best_seq = seq0.tolist()
            best_score = score0
            print(f"[init] height_sequence_1 → score {best_score:.6f} (n={len(seq0)})")
        except Exception as e:
            print("[init] failed to use height_sequence_1:", e)

    # ------------------------------------------------------------------
    # 2) Always keep a uniform baseline (n≈1500 gives the classical bound 2)
    # ------------------------------------------------------------------
    n_uniform = 1805  # matching the length mentioned in the description
    uniform = np.full(n_uniform, math.sqrt(2.0 * n_uniform) / n_uniform)
    uniform_score = evaluate_max_conv(uniform)
    if uniform_score < best_score:
        best_seq = uniform.tolist()
        best_score = uniform_score
        print(f"[init] uniform baseline → score {best_score:.6f} (n={n_uniform})")

    # ------------------------------------------------------------------
    # 3) Main stochastic search
    # ------------------------------------------------------------------
    # We will try many random lengths and initialise each with an exponential
    # distribution (which yields a wide variety of shapes).  Each candidate is
    # refined by the gradient‑based routine above.
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        # Random length – larger lengths give a larger feasible set,
        # but we keep them within a comfortable range.
        n = np.random.randint(200, 2000)

        # Random start: exponential + simplex scaling
        init = np.random.exponential(scale=1.0, size=n).astype(float)
        init = scale_to_target_sum(init)

        # Optimise this start
        seq_opt, score_opt = improve_sequence(init, deadline)

        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt
            print(f"[update] new global best {best_score:.6f} (n={n}, attempt={attempts})")

        # Occasionally try a pure uniform start for the same length –
        # this helps to explore the “flat” region of the landscape.
        if np.random.rand() < 0.07:
            uniform_start = np.full(n, math.sqrt(2.0 * n) / n)
            seq_opt2, score_opt2 = improve_sequence(uniform_start, deadline)
            if score_opt2 < best_score:
                best_score = score_opt2
                best_seq = seq_opt2
                print(f"[update] uniform‑start best {best_score:.6f} (n={n}, attempt={attempts})")

        # Simple early‑exit if we already beat the target bound.
        if best_score <= 1.503:
            print("[goal] target bound reached, stopping early.")
            break

    # ------------------------------------------------------------------
    # 4) Final report
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but guard against it
        raise RuntimeError("No sequence produced during the search.")

    print(f"[final] best score {best_score:.6f} after {attempts} attempts; returning length {len(best_seq)}")
    return best_seq
