def propose_candidate(seed=0, budget_s=1000):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function uses a simple projected‐gradient
    descent starting from a uniform sequence (which is provably optimal
    with value 2).  The optimizer keeps the sum of the coefficients fixed,
    projects onto the simplex after each update and records the best
    sequence seen before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    except Exception:

    # start from the uniform distribution (provably optimal)
    a = np.full(base_len, 1.0 / base_len, dtype=float)

    # keep the current best
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = np.inf

    # parameters for projected gradient descent
    eta = 0.5          # initial step size
    decay = 0.995      # multiplicative decay per iteration

    while time.time() < deadline:
        # full linear convolution
        conv = np.convolve(a, a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]

        # gradient of max(conv) w.r.t. a (sub‑gradient at the active lag)
        g = np.zeros_like(a)
        # populate g_i = 2 * a_{max_idx-i} when the index is valid
        for i in range(len(a)):
            j = max_idx - i
            if 0 <= j < len(a):
                g[i] = 2.0 * a[j]

        # take a gradient step and project back onto the probability simplex
        a_new = a - eta * g
        # enforce non‑negativity
        a_new = np.maximum(a_new, 0.0)

        # project onto the simplex (sum = 1, all components >= 0)
        # using the algorithm of Wang & Carreira‑Perpiñán (2013)
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(u)) + 1
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond][-1]
            theta = cssv[cond][-1] / rho
            a = np.maximum(a_new - theta, 0.0)
        else:
            a = np.zeros_like(a_new)

        # occasionally inject a random perturbation to escape flat regions
        if np.random.rand() < 0.01:
            idx = np.random.randint(len(a))
            a[idx] = max(0.0, a[idx] + np.random.randn() * 0.01)
            a = a / a.sum()

        # evaluate
        try:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                # logging
                print(f"New best score {best_score:.6f} at {time.time():.2f}")
        except Exception:
            pass

        # decay the step size
        eta *= decay

    # Return the best sequence we have found (as a plain list of floats)
    return best_seq.tolist()

"""
Simple projected‑gradient optimiser for the step‑function problem.

The objective is    F(a) = 2 * n * max(conv(a, a)) / (sum(a))**2 .
For non‑negative a the theoretical minimum of F is 2, attained by the
uniform distribution.  The optimiser starts from the uniform vector,
performs a projected gradient descent on the sub‑gradient of the
max‑convolution entry and keeps the best sequence seen before the time
budget expires.  The projection enforces non‑negativity and keeps the
sum of the coefficients equal to 1 (the value of the objective is
scale‑invariant, so the actual magnitude of the sum is irrelevant).

Even though the uniform distribution is provably optimal, the algorithm
still carries out a search – it may discover a sequence that attains
the same value with floating‑point quirks or simply confirms optimality.
"""
import time
import numpy as np

def _project_onto_simplex(v):
    """
    Projects a vector v onto the probability simplex {x | x >= 0, sum(x)=1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v == 0):
        # avoid division by zero – return a uniform vector
        return np.full_like(v, 1.0 / v.size)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1  # running sum minus 1
    ind = np.arange(1, len(v) + 1)

    # find the rho where u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically degenerate case – fall back to uniform
        return np.full_like(v, 1.0 / v.size)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
