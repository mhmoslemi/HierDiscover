"""
Greedy peak‑flattening algorithm with simulated annealing.

The evaluation function is scale‑invariant, therefore we keep the sum of the
sequence fixed to √(2·n) (the normalisation for which the evaluation equals the
maximum value of the auto‑convolution).  Starting from a uniform sequence or
the previously known construction we repeatedly

1. compute the full self‑convolution using an FFT,
2. locate the index k where the convolution attains its maximum,
3. find the entry a_i that contributes most to this peak
   (largest product a_i·a_{k‑i}) and a low‑contributing entry a_j,
4. transfer a small amount of mass from i to j,
5. renormalise the sequence to the prescribed sum,
6. accept the move if the evaluation improves; otherwise accept it with a
   probability that decays with an annealing temperature.

The temperature is reduced geometrically, allowing large exploratory steps
early on and fine‑grained adjustments later.  The best sequence seen is
returned after the allotted time budget (default 1000 s).  The algorithm
operates in O(n log n) per iteration thanks to the FFT, so thousands of
iterations are feasible for n of a few thousand.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level as required.
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the self‑convolution a*a using a real FFT of length ``fft_len``.
    The result is trimmed to the length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    return conv_full[: 2 * a.size - 1]


def _partner_array(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    For a given peak index k_max return an array p where
    p[i] = a[k_max‑i] if the index is valid, otherwise 0.
    """
    n = a.size
    p = np.zeros_like(a)
    start = max(0, k_max - (n - 1))
    end = min(n, k_max + 1)
    idx = np.arange(start, end, dtype=int)
    p[idx] = a[k_max - idx]
    return p


def _select_move(a: np.ndarray, k_max: int) -> tuple | None:
    """
    Choose a pair of indices (i_high, i_low) and a transfer amount delta.
    i_high is the index that contributes most to the current peak,
    i_low contributes the least.  The function returns (i_high, i_low, delta)
    or ``None`` if a move cannot be performed.
    """
    n = a.size
    partner = _partner_array(a, k_max)
    contrib = a * partner

    i_high = int(np.argmax(contrib))
    i_low = int(np.argmin(contrib))

    if i_high == i_low:
        # pick a different low index at random
        i_low = (i_high + np.random.randint(1, n)) % n

    if a[i_high] <= 0.0:
        return None

    # Transfer at most half of the high entry; the exact amount is random.
    max_delta = 0.5 * a[i_high]
    delta = max_delta * np.random.rand()
    if delta < 1e-12:
        return None
    return i_high, i_low, delta

        Random seed for reproducibility.
