"""
Propose a candidate step‑function (a list of non‑negative heights) that
minimises the given evaluation function.

The evaluation is scale‑invariant:
    evaluate(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

For a probability vector (sum = 1) this reduces to 2·n·max(conv).
If we take a decreasing power‑law shape
        a_i ∝ (n‑i)^α ,   α > 0
the central convolution term (shift n‑1) dominates and can be made
arbitrarily small by increasing α.  The asymptotic value of the
evaluation for large *n* is

        2·(α+1)²·B(α+1,α+1) ,

where B is the beta function.  Even α = 2 already yields a score ≈0.6,
α = 4 gives ≈0.08, and α = 6 yields ≈0.01 – well below the target
1.503.

The routine performs a quick random search over the exponent α and
over the length *n* (within a reasonable range) while staying inside
the 1000 s budget.  The best sequence found is returned.
"""

import time
import numpy as np
from math import gamma

# evaluate_sequence is assumed to be already imported in the execution environment.


def beta(a: float, b: float) -> float:
    """Beta function via Gamma, safe for the range we use."""
    return gamma(a) * gamma(b) / gamma(a + b)


def generate_powerlaw_sequence(n: int, alpha: float) -> list[float]:
    """
    Create a decreasing power‑law sequence a_i ∝ (n‑i)^α
    and normalise it to have sum = 1 (any scale works because the
    objective is scale‑invariant).
    """
    # indices 0 … n‑1  →  (n‑i) runs from n down to 1
    i = np.arange(n)
    seq = (n - i) ** alpha
    seq_sum = np.sum(seq)
    seq = seq / seq_sum
    # Ensure plain Python list of floats (evaluate_sequence expects a list)
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a good decreasing power‑law sequence.
    The search is deliberately simple: random samples of (n, α) are
    generated until the time budget is almost exhausted; the best
    sequence (lowest evaluation score) is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep a small safety margin

    # Default search ranges – can be tweaked via kwargs
    n_min = int(kwargs.get("n_min", 300))
    n_max = int(kwargs.get("n_max", 3000))
    alpha_min = float(kwargs.get("alpha_min", 0.5))
    alpha_max = float(kwargs.get("alpha_max", 10.0))

    best_score = float("inf")
    best_seq = None

    # A quick deterministic fallback (α=1) – guaranteed to be feasible
    fallback_seq = generate_powerlaw_sequence(1805, 1.0)
    fallback_score = evaluate_sequence(fallback_seq)

    # Main random search loop
    while time.time() < deadline:
        # random length and exponent
        n = np.random.randint(n_min, n_max + 1)
        alpha = np.random.uniform(alpha_min, alpha_max)

        seq = generate_powerlaw_sequence(n, alpha)

        # the evaluation routine already guards against NaNs/infs etc.
        try:
            score = evaluate_sequence(seq)
        except Exception:
            # Any unexpected problem – skip this candidate
            continue

        if score < best_score:
            best_score = score
            best_seq = seq
            # Log progress
            print(
                f"New best: n={n}, α={alpha:.4f} → score={score:.6f}"
            )
            # If we already beat the target by a comfortable margin we can stop early
            if best_score < 0.5:
                break

    # If the search never produced a valid sequence (unlikely) fall back
    if best_seq is None:
        best_seq = fallback_seq
        best_score = fallback_score
        print(
            f"Fallback used (n=1805, α=1). Score={best_score:.6f}"
        )
    else:
        print(f"Best score achieved: {best_score:.6f}")

    return best_seq
