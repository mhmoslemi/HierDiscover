    """Projects a vector v onto the probability simplex (sum=1, nonnegative)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

"""
Subgradient based optimizer for the max‑convolution bound.

The objective to minimise is
    f(a) = 2·n·max(conv(a,a)) / (sum(a)^2)
which is invariant under scaling of `a`.  Hence we restrict the search to the
probability simplex (non‑negative entries, sum = 1).  For a fixed vector `a`
the element attaining the maximal entry of the convolution, `k*`, gives the
sub‑gradient

    ∂f/∂a_i = 2·a_{k*−i}     (zero if the index is out of range).

A simple exponentiated‑gradient step using this sub‑gradient reduces the
convolution peak.  After each step we renormalise to stay on the simplex.
Random jitter is added occasionally to escape plateaux and the whole search
runs until the supplied time budget expires.  The best sequence encountered
is returned.

The code is completely self‑contained, uses only NumPy and the supplied
`evaluate_sequence` function and respects all platform constraints.
"""

import time
import numpy as np

def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Return a non‑negative vector with the same direction as *v* and unit sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _exponential_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Exponential‑gradient update and renormalisation."""
    # a > 0 already, g >= 0.
    # We use exp(-eta * g) to decrease components that contribute to the max.
    factor = np.exp(-eta * g)
    a_new = a * factor
    return _simplex_normalise(a_new)


def _initial_reverse_hann(n: int) -> np.ndarray:
    """Generate a symmetric “reverse‑Hann” profile – large at the ends,
    zero in the centre – and normalise it to a probability vector."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    dist = np.abs(i - centre)
    # 0 at centre, 1 at the two ends
    a = 0.5 * (1.0 - np.cos(np.pi * dist / centre))
    a = _simplex_normalise(a)
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑height sequence with a low value of `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops a few seconds before the
        deadline to guarantee a return.
    kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence found within the allocated time.
    """
    np.random.seed(seed)
    # We leave a small safety margin (≈5 s) for the final return.
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose the length of the vector.
    #    A larger `n` gives a tighter theoretical lower bound but also makes
    #    each iteration slower.  Values around 2000–3500 work well on the
    #    reference hardware.
    # ------------------------------------------------------------------
    n = np.random.randint(1800, 2600)

    # ------------------------------------------------------------------
    # 2. Initialise the probability vector.
    #    We either start from the previously discovered SOTA sequence (if
    #    available) or from a reverse‑Hann profile.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and np.random.rand() < 0.5:
        a = np.array(height_sequence_1, dtype=float)
        a = _simplex_normalise(a)
        if a.size != n:
            # truncate or pad with zeros to obtain the chosen length
            if a.size > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - a.size), constant_values=0.0)
    else:
        a = _initial_reverse_hann(n)

    # ------------------------------------------------------------------
    # 3. Keep track of the best sequence seen so far.
    # ------------------------------------------------------------------
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float("inf")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop – exponentiated‑gradient sub‑gradient descent.
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        # Compute convolution and locate the maximal entry
        conv = np.convolve(a, a)
        M = conv.max()
        k_star = int(conv.argmax())

        # Build the sub‑gradient: g_i = 2 * a_{k_star - i} (zero if out of range)
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        # indices i = low … high contribute
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = 2.0 * a[k_star - idx]

        # Step size schedule – decays like 1/√t for robustness
        eta = 0.8 / np.sqrt(it + 1)

        # Exponential‑gradient update followed by renormalisation
        a = _exponential_update(a, g, eta)

        # Every few hundred iterations inject a tiny random jitter to avoid
        # stagnation.
        if it % 250 == 0 and it > 0:
            jitter = np.random.randn(n) * 0.005
            a = a + jitter
            a = _simplex_normalise(a)

        # Evaluate the candidate
        try:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Iter {it:05d}] New best: {best_score:.6f} (n={n})")
        except Exception:
            # The evaluation routine returns np.inf on malformed data;
            # we simply ignore such candidates.
            pass

        it += 1

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    return best_seq.tolist()

"""
Subgradient optimizer for the max‑convolution bound.

The goal is to minimise

    f(a) = 2·n·max(conv(a, a)) / (sum(a)²)

which is invariant under scaling of `a`.  Consequently we restrict the search
to the probability simplex (non‑negative entries, sum = 1).  For a fixed
vector `a` let `k*` be an index where the convolution attains its maximum.
A sub‑gradient of `max(conv(a,a))` is then

    ∂/∂a_i  max(conv) = 2·a_{k*‑i}   (zero when the index is out of range).

An exponentiated‑gradient step using this sub‑gradient reduces the peak of the
convolution while the simplex normalisation keeps the iterate feasible.
Occasional random jitter helps to escape plateaux.  The method runs until the
provided time budget expires and returns the best sequence seen.

Only NumPy and the supplied `evaluate_sequence` function are used.
"""

import time
import numpy as np

def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to a uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _exponential_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """Perform an exponentiated‑gradient update and renormalise to the simplex."""
    # decreasing the components that contribute to the current maximum
    a = a * np.exp(-eta * g)
    return _simplex_normalise(a)


def _initial_reverse_hann(n: int) -> np.ndarray:
    """Create a symmetric “reverse‑Hann” profile (large at the ends, zero centre)."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    dist = np.abs(i - centre)
    # a = 0 at the centre, a = 1 at both ends
    a = 0.5 * (1.0 - np.cos(np.pi * dist / centre))
    return _simplex_normalise(a)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑height sequence with a low value of `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.  The function stops a few seconds before the
        deadline to guarantee a return.
    kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered within the allotted time.
    """
    np.random.seed(seed)
    # Leave a small safety margin so we can return before the timeout.
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose a problem size.
    #    A length between 1800 and 2600 gives a good trade‑off between
    #    theoretical lower bound and per‑iteration cost.
    # ------------------------------------------------------------------
    n = np.random.randint(1800, 2600)

    # ------------------------------------------------------------------
    # 2. Initialise the probability vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and np.random.rand() < 0.5:
        a = np.array(height_sequence_1, dtype=float)
        a = _simplex_normalise(a)
        if a.size != n:
            if a.size > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - a.size), constant_values=0.0)
    else:
        a = _initial_reverse_hann(n)

    # ------------------------------------------------------------------
    # 3. Keep the best sequence seen so far.
    # ------------------------------------------------------------------
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float("inf")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop – exponentiated‑gradient sub‑gradient descent.
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        # Convolution and the index of the maximal entry.
        conv = np.convolve(a, a)
        M = conv.max()
        k_star = int(conv.argmax())

        # Sub‑gradient: g_i = 2 * a_{k_star - i} (zero when out of range).
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = 2.0 * a[k_star - idx]

        eta = 0.8 / np.sqrt(it + 1)

        a = _exponential_update(a, g, eta)

    # 5. Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    return best_seq.tolist()
