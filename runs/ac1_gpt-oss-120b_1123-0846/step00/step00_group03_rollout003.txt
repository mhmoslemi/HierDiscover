def project_onto_simplex(v):
    # ensures v >=0 and sum(v)=1
    # use algorithm by Duchi et al.
    n = len(v)
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a better height sequence for the evaluation function.

Algorithm Overview
------------------
1.  Initialise with the provided `height_sequence_1` (if available) and a few
    handcrafted window based sequences (Hann, Hamming, Blackman, Kaiser,
    exponential decay, cosine‑squared, triangular).  All sequences are
    normalised to unit sum because the objective is homogeneous
    (scaling does not change the score).

2.  Keep the best sequence found so far (lowest `evaluate_sequence` score).

3.  Refine any promising sequence by *smooth‑max* projected gradient descent:
    -   Use a smooth approximation of `max(conv(a,a))` via a soft‑max
        (log‑sum‑exp) with a temperature `eps`.
    -   The gradient of `max` is a weighted sum of the gradients of each
        convolution entry; for a fixed weight vector `w` the gradient can be
        computed efficiently as   `grad = 2 * convolve(w, a, mode='valid')`.
    -   Perform a simple step `a ← a – η·grad` with a diminishing step‑size,
        then project back onto the probability simplex (non‑negative, sum = 1).

4.  Multiple random restarts and a short simulated‑annealing style perturbation
    are performed while staying within the overall time budget (default 950 s,
    leaving a short safety margin).

5.  The best sequence encountered during the whole run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def project_simplex(v):
    """
    Projects a vector onto the probability simplex {x >= 0, sum(x)=1}.
    Implementation of the algorithm from Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    if v.ndim != 1:
        raise ValueError("Simplex projection only supports 1‑D vectors.")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def smooth_max_and_grad(a, eps):
    """
    Returns:
        - soft max value (unused here, but handy for diagnostics)
        - gradient of the soft max w.r.t. a
        - the raw convolution vector
    """
    conv = np.convolve(a, a)
    maxc = conv.max()
    # soft‑max weights (stable)
    exp_vals = np.exp((conv - maxc) / eps)
    w = exp_vals / exp_vals.sum()
    # gradient: 2 * (w * a) convolution, ‘valid’ gives exactly length=len(a)
    grad = 2.0 * np.convolve(w, a, mode='valid')
    # smooth max (not needed for optimisation)
    smooth_val = maxc + eps * np.log(exp_vals.sum())
    return smooth_val, grad, conv


def evaluate(seq):
    """Convenient wrapper – guarantees list type for the evaluator."""
    return evaluate_sequence(seq.tolist() if isinstance(seq, np.ndarray) else list(seq))


# ----------------------------------------------------------------------
# Window / parametric families
# ----------------------------------------------------------------------
def hann_window(n):
    i = np.arange(n)
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def hamming_window(n):
    i = np.arange(n)
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))


def blackman_window(n):
    i = np.arange(n)
    a0, a1, a2 = 0.42, 0.5, 0.08
    return (a0
            - a1 * np.cos(2.0 * np.pi * i / (n - 1))
            + a2 * np.cos(4.0 * np.pi * i / (n - 1)))


def kaiser_window(n, beta):
    return np.kaiser(n, beta)


def exponential_decay(n, rate):
    i = np.arange(n)
    return np.exp(-rate * i)


def cosine_squared(n):
    i = np.arange(n)
    return np.sin(np.pi * i / (n - 1)) ** 2


def triangular_window(n):
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    return 1.0 - np.abs(i - mid) / mid


def random_dirichlet(n, concentration=1.0):
    # concentration < 1 gives sparser vectors
    return np.random.gamma(concentration, 1.0, size=n)


# ----------------------------------------------------------------------
# Core refinement routine (smooth‑max projected gradient)
# ----------------------------------------------------------------------
def refine_sequence(a, deadline, best_score_ref, best_seq_ref):
    """
    Runs projected gradient descent on the smooth maximum of the convolution.
    Updates the external best score / sequence references if a better point
    is discovered.
    """
    a = np.asarray(a, dtype=float)
    a = a / a.sum()
    it = 0
    while time.time() < deadline:
        # temperature adapts to the current spread of the convolution
        conv = np.convolve(a, a)
        spread = conv.max() - conv.min()
        eps = max(1e-8, 0.02 * (spread + 1e-12))

        _, grad, _ = smooth_max_and_grad(a, eps)
        # diminishing step size
        eta = 0.5 / np.sqrt(it + 1)
        a = a - eta * grad
        a = project_simplex(a)

        if it % 50 == 0:
            cur_score = evaluate(a)
            if cur_score < best_score_ref[0]:
                best_score_ref[0] = cur_score
                best_seq_ref[0] = a.tolist()
                print(f"[Refine] iter={it} n={len(a)} improved score={cur_score:.6f}")
        it += 1
        if it > 4000:        # safety stop for very long runs
            break
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=950, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function respects a wall‑clock budget
    (default 950 seconds) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Initialise best sequence with any existing construction
    # ------------------------------------------------------------------
    best_sequence = None
    best_score = float('inf')
    try:
        seq0 = list(height_sequence_1)          # global variable may exist
        sc0 = evaluate(seq0)
        best_sequence = seq0
        best_score = sc0
        print(f"[Init] height_sequence_1 score = {sc0:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) Explore parametric families (different lengths & params)
    # ------------------------------------------------------------------
    families = [
        (hann_window,      [{}]),
        (hamming_window,   [{}]),
        (blackman_window,  [{}]),
        (kaiser_window,    [{'beta': 0}, {'beta': 3}, {'beta': 5}, {'beta': 8}, {'beta': 12}]),
        (exponential_decay,[{'rate': 0.001}, {'rate': 0.005}, {'rate': 0.02}, {'rate': 0.05}]),
        (cosine_squared,   [{}]),
        (triangular_window,[{}]),
        (random_dirichlet, [{'concentration': 0.2},
                            {'concentration': 0.5},
                            {'concentration': 1.0},
                            {'concentration': 2.0}])
    ]

    lengths = [64, 128, 256, 512, 1024]

    # Keep references for live updates inside refine
    best_score_ref = [best_score]
    best_seq_ref = [best_sequence]

    for n in lengths:
        if time.time() >= deadline:
            break
        for func, param_list in families:
            if time.time() >= deadline:
                break
            for params in param_list:
                # generate
                try:
                    seq = func(n, **params) if params else func(n)
                except Exception:
                    continue
                seq = np.asarray(seq, dtype=float)
                if np.all(seq == 0):
                    continue
                seq = seq / seq.sum()          # scale to unit sum (homogeneous)
                score = evaluate(seq)
                if score < best_score_ref[0]:
                    best_score_ref[0] = score
                    best_seq_ref[0] = seq.tolist()
                    print(f"[Family] {func.__name__} n={n} params={params} score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) Refine the current best (and a few random restarts) using gradient descent
    # ------------------------------------------------------------------
    # First refinement from the best found so far
    if best_seq_ref[0] is not None:
        print("[Refine] Starting refinement from best sequence.")
        refine_sequence(best_seq_ref[0], deadline, best_score_ref, best_seq_ref)

    # Additional random restarts (same length as best or a new length)
    n_rest = 4
    for _ in range(n_rest):
        if time.time() >= deadline:
            break
        # Pick a length (either same as best or random)
        n = len(best_seq_ref[0]) if best_seq_ref[0] is not None else np.random.choice(lengths)
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()
        print(f"[Random] Restart with random vector length={n}")
        refine_sequence(a0, deadline, best_score_ref, best_seq_ref)

    # ------------------------------------------------------------------
    # 4) Return the best sequence discovered
    # ------------------------------------------------------------------
    final_seq = best_seq_ref[0] if best_seq_ref[0] is not None else [1.0]
    final_score = best_score_ref[0]
    print(f"[Result] Best score = {final_score:.6f} with length = {len(final_seq)}")
    return final_seq
