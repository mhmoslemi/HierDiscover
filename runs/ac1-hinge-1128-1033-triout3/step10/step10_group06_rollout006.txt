    n = a.shape[0]
    out_len = 2*n-1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv

def convolution_with(a, b):  # a length m, b length n
    m = a.shape[0]; n = b.shape[0]
    out_len = m + n - 1

    af = np.fft.rfft(a, size)
    bf = np.fft.rfft(b, size)
    conv = np.fft.irfft(af * bf, size)[:out_len]
    return conv

Algorithm
---------

1. **Rich initialisation** – we enumerate a large collection of
   parametric families (exponential, power‑law, beta, arcsine,
   sine‑power, piecewise‑constant) plus many random Dirichlet draws.
   The best seed (lowest exact objective) is kept.

2. **Soft‑max gradient descent** – the objective for a probability
   vector `a` (∑a=1) is 2·n·max(a*a).  We replace the hard max by a
   smooth soft‑max with a temperature parameter λ, compute the gradient
   analytically (via a convolution) and perform an Adam‑style update.
   λ is ramped from a small value to a large one, so the optimisation
   first moves in a smooth landscape and then concentrates on the true
   max.  After each update we project back onto the simplex.

3. **Stagnation handling** – if no improvement is seen for many
   iterations we inject a small Gaussian perturbation and reset Adam’s
   moments, helping the search escape plateaus.

4. **Final polish** – a tiny convex combination with the uniform
   distribution (α∈{0.001,…,0.020}) is tried; the best of these is
   returned.

All heavy work respects the 1000 s wall‑clock budget; the function
`propose_candidate` is the entry point required by the evaluation
harness.

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution of a (len m) with b (len n)."""
    m, n = a.shape[0], b.shape[0]
    out_len = m + n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    bf = np.fft.rfft(b, size)
    conv = np.fft.irfft(af * bf, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a probability vector (∑a=1)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)

# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Phase 1 – soft‑max Adam gradient descent on the convolution max
# ----------------------------------------------------------------------
def _softmax_gradient_descent(a0: np.ndarray,
                              n: int,
                              start: float,
                              deadline: float,
                              best_a: np.ndarray,
                              best_val: float,
                              target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of 2·n·max(a*a) using a soft‑max surrogate.
    """
    a = a0.copy()
    a_rev = a[::-1]                     # reused in each iteration
    rng = np.random.default_rng(42)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature schedule
    lam_start, lam_end = 5.0, 5000.0

    # Learning‑rate schedule (initial value will be scaled by Adam)
    base_lr = 0.8

    # Stagnation detection
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 15000

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        lam = lam_start + (lam_end - lam_start) * progress
        lr = base_lr * (1.0 - progress)   # mild linear cooling

        # ----- exact objective & current max index -----
        c = _full_convolution(a)
        max_c = c.max()
        cur_val = 2.0 * n * max_c

        # store global best
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[GD {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- soft‑max weights (focus on the peak) -----
        diff = c - max_c               # ≤ 0, zero at the peak(s)
        # protect against under‑flow for very negative exponents
        w = np.exp(lam * diff, dtype=np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
            w[ np.argmax(c) ] = 1.0
        else:
            w /= w_sum

        # ----- gradient of soft‑max surrogate -----
        # g_i = 2 * Σ_k w_k * a_{k-i}
        conv = _conv_two(w, a_rev)   # length = 3n-2
        g = 2.0 * conv[n - 1: 2 * n - 1]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        # compute candidate objective (exact, not surrogate)
        cand_val = _objective_value(a_candidate)

            a = a_candidate
            a_rev = a[::-1]   # update the reversed copy

            if np.random.rand() < 1e-4:
                a = a_candidate
                a_rev = a[::-1]

            a = np.maximum(a, 0.0)
            a = a / a.sum()
            a_rev = a[::-1]

            v.fill(0.0)
            no_improve = 0
            print(f"[GD {it:7d}] perturbation after stagnation")

        it += 1

    return best_a, best_val

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform distribution, try a few alphas."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):

        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val
