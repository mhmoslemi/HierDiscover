"""
Propose a step‑function that minimizes   2·n·max(conv(a,a)) / (sum(a))²
using a fast projected‑gradient descent on a smooth approximation of the
maximum of the self‑convolution.  The procedure:

* initialise a symmetric probability vector a (Σa=1) from a sensible parametric family
  (arcsine shape, plus a few random Dirichlet draws);
* repeatedly
    – compute the exact self‑convolution b = a*a (FFT, O(n log n));
    – build a soft‑max weight w that concentrates on the current peak of b;
    – compute the gradient of the soft‑max objective:
          ∇f(a) = (w ⋆ a) + a   (⋆ = cross‑correlation);
    – take a small step against the gradient, project back onto the simplex,
      re‑symmetrise the vector (a_i = a_{n‑1‑i}) and renormalise;
    – keep the best vector seen so far.
* after the time budget, perform a tiny convex blend with the uniform distribution
  (a “polish”) and return the best sequence.

All operations stay in the real domain and use NumPy’s FFT (rfft/irfft);
the algorithm respects the 1000 s budget, works on a few‑thousand‑element
vectors and typically drives the objective below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1.0) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray, size_self: int) -> np.ndarray:
    """Exact linear self‑convolution of a using a pre‑allocated FFT size."""
    fa = np.fft.rfft(a, size_self)
    conv = np.fft.irfft(fa * fa, size_self)[: 2 * len(a) - 1]
    return conv


def _cross_correlation_fft(w: np.ndarray, a: np.ndarray,
                           size_corr: int) -> np.ndarray:
    """
    Compute the full cross‑correlation (w ⋆ a) for vectors of lengths
    2n‑1 and n.  Returns the first n entries of the correlation,
    which equal (w ⋆ a)[j] for j = 0 … n‑1.
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    fa_rev = np.fft.rfft(a_rev, size_corr)
    corr_full = np.fft.irfft(fw * fa_rev, size_corr)
    # length of corr_full = len(w)+len(a)-1 = 3n‑2
    # The needed values are corr_full[:n] reversed.
    return corr_full[: len(a)][::-1]

# Main optimisation routine
# ----------------------------------------------------------------------

                     budget_s: int = 1000,
                     **kwargs) -> list[float]:

    Optimise a step‑function within the given time‑budget.
    Returns a list of non‑negative floats (Σ ≈ 1) ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # problem size – a few thousand entries is a sweet spot
    n = 9000

    # ------------------------------------------------------------------

    i = np.arange(n, dtype=float)

    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()

    # try a few random dirichlet draws and keep the best one
    best_a = a0.copy()
    best_val = 2.0 * n * _full_convolution_fft(best_a, _next_pow2(2 * n - 1)).max()
    rng = np.random.default_rng(seed + 12345)
    for _ in range(5):
        cand = rng.dirichlet(np.ones(n))
        cand = (cand + cand[::-1]) / 2.0          # enforce symmetry
        cand /= cand.sum()
        val = 2.0 * n * _full_convolution_fft(cand, _next_pow2(2 * n - 1)).max()
        if val < best_val:

    a = best_a.copy()                            # current iterate
