def full_convolution_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n-1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv

def _smooth_max_gradient_descent(a0, n, start, deadline, best_val):
    a = a0.copy()
    best_a = a.copy()
    best_local_val = best_val

    beta = 5.0  # starting smoothness
    beta_multiplier = 1.1  # increase each iteration (or periodically)
    max_beta = 2e6

    # step size
    base_lr = 0.5  # initial learning rate
    # we store iteration count
    it = 0
    no_improve_iters = 0
    NO_IMPROVE_LIMIT = 20000

    while time.time() < deadline:
        # Compute convolution
        c = _full_convolution(a)
        max_c = c.max()
        cur_val = 2.0 * n * max_c

        if cur_val < best_local_val - 1e-12:
            best_local_val = cur_val
            best_a = a.copy()
            print(f"[GD {it:7d}] new best = {best_local_val:.9f}")
            no_improve_iters = 0
        else:
            no_improve_iters += 1

        # adapt beta
        beta = min(beta * beta_multiplier, max_beta)

        # Compute softmax weights
        # subtract max to avoid overflow; use double
        diff = c - max_c
        # exponent maybe extremely tiny; we compute exp(beta * diff)
        # diff <= 0, so exp <= 1.
        # Might underflow if beta*diff << -700; but diff minimal approx -max_c ~ -8e-5; beta up to 2e6 => exponent ~ -160, okay.
        exp_vals = np.exp(beta * diff)
        w = exp_vals / exp_vals.sum()

        # gradient
        grad = _gradient_from_weights(a, w)

        # learning rate schedule
        lr = base_lr / (1.0 + beta * 0.00001)  # just some scaling

        # update using gradient descent
        a_new = a - lr * grad
        # project onto simplex
        a_new = _project_simplex(a_new)

        a = a_new
        it += 1

        # restart or perturb if stuck
        if no_improve_iters > NO_IMPROVE_LIMIT:
            # Add small Gaussian noise and reproject
            a = best_a.copy()
            noise = np.random.normal(scale=1e-3, size=n)
            a = np.maximum(a + noise, 0)
            a = _project_simplex(a)
            no_improve_iters = 0
            print(f"[GD {it:7d}] perturb after stagnation")

    return best_a, best_local_val

"""
Gradient‑descent optimizer for the step‑function problem.

The algorithm:
  1. Perform a cheap parametric sweep (exponential, power‑law, beta,
     arcsine, piecewise‑constant, Dirichlet) to obtain a strong initial
     seed.
  2. Minimise a smoothed version of the objective
        f_β(a) = (1/β)·log Σ_k exp(β·c_k) ,   c = a*a .
     The gradient of f_β is computed analytically via FFT‑based
     correlation; after each gradient step the vector is projected onto
     the probability simplex (non‑negative, Σ a = 1).
     β is increased during the run, turning the smooth objective into
     the true max‑convolution.  Simple back‑off, random perturbations
     and a short restart policy avoid stagnation.
  3. A final “uniform‑mix” polish blends a tiny fraction of the uniform
     distribution and is guaranteed not to increase the exact objective.
  4. The resulting list of non‑negative floats is returned.
"""

import time
import math
import numpy as np

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using real FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of vectors x and y (both real)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv

def _objective_value(a: np.ndarray) -> float:
    """Exact objective (Σ a = 1)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max objective.

    g = 2 * corr_valid(w, a) = 2 * (a_rev * w)[n‑1 : n‑1+n]
    where a_rev is a reversed copy of a.
    """
    a_rev = a[::-1]
    conv_rev = _full_convolution_two(a_rev, w)   # length = 3n‑2
    n = a.shape[0]
    # slice the “valid’’ part (correlation)
    grad = 2.0 * conv_rev[n-1 : n-1 + n]
    return grad

def _initial_candidate(n: int):

    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)            # 0 … n‑1
    i1 = i + 1.0                              # 1 … n

    # 1. Exponential tails
    for r in np.linspace(0.995, 0.999999, 20):
        _try(r ** i)                         # decreasing
        _try(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.1, 3.0, 30):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Inverse‑sqrt (special case of power‑law)
    _try(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))

    # 4. Beta densities
    beta_vals = np.linspace(0.1, 1.5, 15)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 5. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # 6. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 12):
        _try(base_sin ** p)

    # 7. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.005, 0.1, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 8. Random Dirichlet draws
    rng = np.random.default_rng(123456)
    for _ in range(80):
        _try(rng.dirichlet(np.ones(n)))

    # 9. Interpolation of a previously found solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    # invariant under reversal.
    best_a = (best_a + best_a[::-1]) / 2.0
    best_a = _project_simplex(best_a)
    best_val = _objective_value(best_a)

    return best_a, best_val

def _smooth_max_gradient_descent(a0: np.ndarray,
                                 start: float,
                                 deadline: float,
                                 best_val_init: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed version of the max‑convolution.

    Parameters
    ----------
    a0 : ndarray
        Starting vector (already normalised, Σ = 1).
    start, deadline : float
        Wall‑clock limits.
    best_val_init : float
        Current best known objective (for early exit).

    Returns
    -------
    best_a, best_val
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = best_val_init

    # smoothing parameter – start small, increase slowly
    beta = 5.0
    beta_mul = 1.08
    max_beta = 2_000_000.0

    # learning‑rate scale – we will shrink it when beta grows
    base_lr = 0.8

    it = 0
    no_improve = 0
    NO_IMPROVE_LIMIT = 25000

    while time.time() < deadline:
        # ----- exact evaluation (needed for progress) -----
        c = _full_convolution(a)
        max_c = c.max()
        cur_val = 2.0 * n * max_c

        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = a.copy()
            print(f"[GD {it:7d}] new best = {best_val:.9f}")
            no_improve = 0
        else:
            no_improve += 1

        # optional early exit if we already beat the target by a comfortable margin
        if best_val <= 1.503:
            break

        # ----- increase smoothing -----
        beta = min(beta * beta_mul, max_beta)

        # ----- soft‑max weights for the current convolution -----
        # exp(β·(c‑max_c)) is ≤ 1, very stable.
        exp_vals = np.exp(beta * (c - max_c))
        w = exp_vals / exp_vals.sum()          # w sums to 1, symmetric automatically

        # ----- gradient of the smoothed objective -----
        grad = _gradient_from_weights(a, w)

        # ----- step size (inversely proportional to β) -----
        lr = base_lr / (1.0 + beta * 1e-5)

        # ----- gradient step + projection onto simplex -----
        a_next = a - lr * grad
        a_next = _project_simplex(a_next)

        a = a_next
        it += 1

        # ----- occasional random perturbation to escape stagnation -----
        if no_improve >= NO_IMPROVE_LIMIT:
            # restart from the current best with a tiny Gaussian jitter
            a = best_a.copy()
            noise = np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[GD {it:7d}] perturb after {NO_IMPROVE_LIMIT} iters")

    return best_a, best_val

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with the uniform vector; tiny α values only."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)       # keep exact normalisation
        val = _objective_value(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    return best_a, best_val

"""
Gradient‑descent optimizer for the step‑function problem.

The algorithm proceeds in three stages:

1. **Broad parametric sweep** – a collection of exponential, power‑law,
   beta, arcsine, sine‑power and piecewise‑constant families (plus a few
   random Dirichlet draws) is sampled.  The best candidate (Σ ≈ 1) is kept.

2. **Smooth‑max projected gradient descent** – we minimise a differentiable
   surrogate of the true objective
       f_β(a) = (1/β)·log Σ_k exp(β·c_k) ,   c = a*a .
   For β→∞ the surrogate converges to the original max‑convolution.
   The gradient of f_β is computed analytically via FFT‑based correlation,
   after each step the vector is projected onto the probability simplex.
   β is increased gradually, the learning rate is shrunk accordingly and
   a jitter‑restart policy prevents stagnation.

3. **Uniform‑mix polish** – a tiny convex combination with the uniform
   distribution (α∈{0.001,…,0.020}) is tried; this can only lower the exact
   objective and serves as a cheap final tweak.

The routine respects the supplied time budget (default 1000 s) and returns
a list of non‑negative floats (Σ≈1) suitable for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two (possibly different‑length) vectors
    using a real FFT.
    """
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective assuming Σ a = 1: 2·n·max(a*a)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute   g = 2·corr_valid(w, a)   where
    corr_valid(w, a)_i = Σ_j w_{i+j}·a_j  (i = 0…n‑1).
    Implemented via FFT convolution of the reversed a with w.
    """
    a_rev = a[::-1]
    conv_rev = _full_convolution_two(a_rev, w)   # length = 3·n‑2
    n = a.shape[0]
    grad = 2.0 * conv_rev[n-1 : n-1 + n]
    return grad


# ----------------------------------------------------------------------
# Phase 0 – broad parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Search a rich collection of parametric families for a low‑objective seed."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)            # 0 … n‑1
    i1 = i + 1.0                              # 1 … n

    # 1. Exponential tails (left‑hand and symmetric)
    for r in np.linspace(0.995, 0.999999, 20):
        _try(r ** i)                         # decreasing
        _try(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

    # 2. Power‑law tails (left‑hand and symmetric)
    for p in np.linspace(0.1, 3.0, 30):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Inverse‑sqrt (special case of power‑law)
    _try(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))

    # 4. Beta densities – finer grid around the U‑shape (a,b < 1)
    beta_vals = np.linspace(0.1, 1.5, 15)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 5. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # 6. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 12):
        _try(base_sin ** p)

    # 7. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.005, 0.1, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 8. Random Dirichlet draws for diversity
    rng = np.random.default_rng(123456)
    for _ in range(80):
        _try(rng.dirichlet(np.ones(n)))

    # 9. Interpolation of a previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    # Enforce exact symmetry (optional, never hurts)
    best_a = (best_a + best_a[::-1]) / 2.0
    best_a = _project_simplex(best_a)
    best_val = _objective_value(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max projected gradient descent
# ----------------------------------------------------------------------
def _smooth_max_gradient_descent(a0: np.ndarray,
                                 start: float,
                                 deadline: float,
                                 best_val_init: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the max‑convolution using projected
    gradient descent.  The smoothing parameter β is increased during the run.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = best_val_init

    beta = 5.0                 # initial smoothness
    beta_mul = 1.08
    max_beta = 2_000_000.0

    base_lr = 0.8              # raw learning‑rate factor
    it = 0
    no_improve = 0
    NO_IMPROVE_LIMIT = 25000

    while time.time() < deadline:
        # exact objective (needed for monitoring)
        c = _full_convolution(a)
        max_c = c.max()
        cur_val = 2.0 * n * max_c

        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = a.copy()
            print(f"[GD {it:7d}] new best = {best_val:.9f}")
            no_improve = 0
        else:
            no_improve += 1

        if best_val <= 1.503:
            break

        # increase smoothness
        beta = min(beta * beta_mul, max_beta)

        # soft‑max weights for the current convolution
        exp_vals = np.exp(beta * (c - max_c))   # max(c) gives exponent 0 → stable
        w = exp_vals / exp_vals.sum()          # Σ w = 1, w is symmetric

        # gradient of the smooth objective
        grad = _gradient_from_weights(a, w)

        # learning‑rate adaption (smaller steps for larger β)
        lr = base_lr / (1.0 + beta * 1e-5)

        # gradient step + projection onto the simplex
        a = a - lr * grad
        a = _project_simplex(a)

        it += 1

        # occasional jitter to escape plateaus
        if no_improve >= NO_IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[GD {it:7d}] perturb after {NO_IMPROVE_LIMIT} iters")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – tiny uniform mixing polish (same as the previous code)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a tiny share of the uniform distribution."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)
        val = _objective_value(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ≈1) ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # problem size – a few thousand entries gives a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # Phase 0 – broad parametric sweep
    # ------------------------------------------------------------------
    print("[Init] parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max projected gradient descent
    # ------------------------------------------------------------------
    best_a, best_val = _smooth_max_gradient_descent(best_a,
                                                    start=start_time,
                                                    deadline=deadline,
                                                    best_val_init=best_val)
    print(f"[GD] best = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – final tiny uniform mixing polish
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided wrapper (optional logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.9f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
