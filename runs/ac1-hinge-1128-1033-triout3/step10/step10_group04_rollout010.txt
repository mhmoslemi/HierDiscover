"""
propose_candidate – fast heuristic for the max‑convolution bound.

The goal is to find a non‑negative vector a (Σ a = 1) that minimises
    2·n·max_k (a * a)[k] .
The algorithm works in three stages:

1.  **Rich initialisation** – a sweep over many parametric families
    (exponential, power‑law, beta, sinusoidal, piecewise‑constant,
    logistic) as well as random Dirichlet draws with various
    concentration parameters.  The best vector from this pool is kept.

2.  **Targeted mass‑transfer refinement** – starting from the current
    best vector we repeatedly relocate probability mass from indices
    that contribute most to the current maximal convolution entry to
    indices that contribute the least.  The amount transferred is
    chosen by an *exact* line‑search: the largest δ that does not raise
    any convolution entry above the new maximum is computed analytically.
    This is a greedy active‑set step that reduces the true objective
    in each successful move.

3.  **Uniform mixing** – a very small convex combination with the
    uniform distribution (α∈{0.001,…,0.010}) often smooths residual spikes.

All heavy work (convolutions, line‑search) uses FFT‑based full
convolution for speed.  The whole optimiser is bounded by the supplied
time budget (default 1000 s).  Intermediate progress (objective value)
is printed for monitoring.  The returned list can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective for a vector already normalised to sum = 1
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Build a diverse pool of candidates and keep the best one
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. exponential tails (decreasing & symmetric)
    for r in np.linspace(0.995, 0.99999, 14):
        _try(r ** i)                         # monotonically decreasing
        _try(r ** i + r ** (n - 1 - i))      # symmetric “U”

    # 2. power‑law tails
    for p in np.linspace(0.2, 6.0, 18):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. beta‑distribution shapes (including arcsine)
    for a_param in np.linspace(0.2, 6.0, 9):
        for b_param in np.linspace(0.2, 6.0, 9):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. sin‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _try(base_sin ** p)

    # 5. piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.005, 0.12, 12):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 6. logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.003, 0.04, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 7. random Dirichlet draws – more concentrated alphas help sparsity
    rng = np.random.default_rng(12345)
    for alpha in (0.1, 0.2, 0.5, 1.0):
        for _ in range(30):
            a = rng.dirichlet(np.full(n, alpha))
            _try(a)

    # 8. warm‑start from a previously found construction, if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted active‑set refinement with exact line‑search
# ----------------------------------------------------------------------
def _refine_line_search(a0: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfers from a “heavy” index to a “light” one.
    The transferred amount δ is the largest value that keeps every
    convolution entry ≤ the new maximum after the move.
    """
    a = a0.copy()
    best_local_a = best_a.copy()
    best_local_val = best_val
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000   # after this we shake the vector

    while time.time() < deadline:
        # 1. current convolution and objective
        b = _full_convolution(a)
        M = b.max()
        cur_val = 2.0 * n * M

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[RefineLS {it:7d}] new best = {best_local_val:.6f}")
        else:
            no_improve += 1

        # 2. active set (indices whose value is practically maximal)
        eps = max(M * 5e-8, 1e-12)
        active = np.where(b >= M - eps)[0]

        # 3. contribution weights w_i = Σ_{k∈active} a_{k-i}
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += a[k - idx]

        # 4. pick source (large w) and target (small w); add a dash of randomness
        top_k = max(1, n // 20)                # top 5 %
        bottom_k = max(1, n // 20)             # bottom 5 %
        sorted_idx = np.argsort(w)
        src_candidates = sorted_idx[-top_k:]
        tgt_candidates = sorted_idx[:bottom_k]
        src = int(np.random.choice(src_candidates))
        tgt = int(np.random.choice(tgt_candidates))
        if src == tgt or a[src] < 1e-15:
            # fallback to deterministic extreme choice
            src = int(np.argmax(w))
            tgt = int(np.argmin(w))
            if src == tgt:
                # pick any different pair
                src = int(np.argmax(w))
                tgt = (src + n // 2) % n

        # 5. pre‑compute the contribution of src/tgt to each convolution entry
        src_contrib = np.zeros(2 * n - 1)
        src_contrib[src:src + n] = a
        tgt_contrib = np.zeros(2 * n - 1)
        tgt_contrib[tgt:tgt + n] = a

        w_src = w[src]
        w_tgt = w[tgt]

        # 6. compute the maximal admissible shift δ
        R = w_src - w_tgt - src_contrib + tgt_contrib          # change coefficient per entry
        pos_mask = R > 1e-15                                   # only positive coefficients restrict δ
        if np.any(pos_mask):
            slack = M - b[pos_mask]                            # how much headroom we have
            delta_limits = slack / (2.0 * R[pos_mask])
            delta_max = delta_limits.min()
        else:
            delta_max = a[src]                                 # no restriction besides source mass

        delta_max = min(delta_max, a[src])
        if delta_max <= 1e-15:
            # nothing can be moved – perform a small random perturbation to escape stagnation
            if no_improve > IMPROVE_LIMIT:
                noise = np.random.normal(scale=0.005, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                no_improve = 0
                continue
            else:
                # give up on this pair for now
                it += 1
                continue

        # 7. apply move (use a safety factor)
        delta = 0.999 * delta_max
        a[src] -= delta
        a[tgt] += delta

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Tiny uniform mixing – final polish
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_a, best_val = cand, val
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function within the given time budget.
    Returns a list whose sum is 1 and that (hopefully) beats the current
    record on `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # Problem size – a few thousand entries give a good balance between
    # representational power and computational speed.
    n = 9000

    # ------------------------------------------------------------------
    # 0️⃣ Initial parametric sweep
    # ------------------------------------------------------------------
    print("[Init] searching parametric families")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Aggressive active‑set refinement with exact line‑search
    # ------------------------------------------------------------------
    phase_refine_deadline = deadline
    best_a, best_val = _refine_line_search(
        best_a, n,
        start=start_time,
        deadline=phase_refine_deadline,
        best_a=best_a,
        best_val=best_val,
    )
    print(f"[Refine] best after line‑search = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Final tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Report final score using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
