"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence.  The method builds on three ideas:

1️⃣  Initialise from a saved solution (if present) and from a cheap
    exploration of the Beta‑family.
2️⃣  Optimise the smooth approximation of the objective
    f(a)=2·n·max(conv(a,a)) with an Adam‑style projected gradient
    descent.  The “max” is replaced by a soft‑max with a temperature
    γ that grows from 5 to ≈50 during the run, yielding a well‑behaved
    gradient that gradually concentrates on the active lag.
3️⃣  Keep the iterate on the probability simplex (Σa=1, a≥0) by an
    Euclidean projection, and retain the best exact solution found.

The algorithm is fully vectorised and uses FFT‑based convolutions,
allowing many thousands of iterations within the 1000 s budget.
The final sequence is handed to evaluate_sequence for the official
score.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------


def _full_convolution(x: np.ndarray, y: np.ndarray = None) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    If y is None, convolves x with itself.
    """
    if y is None:
        y = x
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective value 2·n·max(conv(a,a))  (the sum of a is assumed to be 1).
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    Σx = 1,  x ≥ 0.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}(n‑i+1)^{q‑1},   i = 1,…,n .
    If the parameters produce an all‑zero vector, None is returned.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a

# Main entry point
# ------------------------------------------------------------
