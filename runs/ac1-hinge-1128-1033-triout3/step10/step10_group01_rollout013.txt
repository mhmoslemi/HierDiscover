def _adam_optimize(a0, n, deadline, start_time):
    # a0 is initial a (np.ndarray)
    # ensure no zeros
    eps = 1e-12
    a0 = np.maximum(a0, eps)
    a0 /= a0.sum()
    u = np.log(a0)   # log for initialization
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step_size = 0.1
    best_a = a0.copy()
    best_val = _objective(a0)
    no_improve = 0
    max_no_improve = 5000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute a
        a = np.exp(u)
        sum_a = a.sum()
        if sum_a == 0:
            # Reset to uniform
            a = np.full(n, 1.0/n)
            sum_a = 1.0
        a /= sum_a
        # convolution
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:6d}] new best = {best_val:.6f}")
            if best_val <= 1.503:
                break
        else:
            no_improve += 1

        # smoothing parameter schedule
        total_time = deadline - start_time
        elapsed = now - start_time
        alpha_initial = 5.0
        alpha_target = 200.0
        alpha = alpha_initial + (alpha_target - alpha_initial) * min(1.0, elapsed/total_time)

        # compute w
        diff = conv - max_conv  # <= 0
        w = np.exp(alpha * diff)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # gradient wrt a
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)
        # slice
        grad = 2.0 * conv_rev[n-1:n-1+n]  # shape (n,)

        # chain rule for u
        g_bar = np.dot(a, grad)  # scalar
        du = a * (grad - g_bar)

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * du
        v = beta2 * v + (1 - beta2) * (du * du)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        u = u - step_size * m_hat / (np.sqrt(v_hat) + eps_adam)

        # restart if stuck
        if no_improve >= max_no_improve:
            # add noise to u
            noise = np.random.normal(scale=0.5, size=n)
            u = np.log(best_a + eps) + noise
            no_improve = 0
            print(f"[Adam {iteration}] restart due to stagnation")
    return best_a, best_val

```python
import time
import math
import numpy as np

# Convolution utilities
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1) via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors x and y via FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _adam_optimize(a0, n: int, deadline: float, global_start: float):
    """
    Adam‑style exponentiated‑gradient optimizer on the simplex.
    a0 can be None (random start) or a numpy array of length n.
    Returns (best_a, best_val).
    """
    eps = 1e-12
    if a0 is None:
        a0 = np.random.dirichlet(np.ones(n))
    else:
        a0 = np.maximum(a0, eps)
        a0 = a0 / a0.sum()
    # work in log‑space to stay on the simplex automatically
    u = np.log(a0 + eps)

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    step_size = 0.12
    adam_eps = 1e-8

    best_a = a0.copy()
    best_val = _objective(best_a)
    no_improve = 0
    max_no_improve = 5000
    it = 0

    # smoothing schedule
    alpha_initial = 5.0
    alpha_target = 200.0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # produce current a from log‑variables
        a = np.exp(u)
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, 1.0 / n)
            sum_a = 1.0
        a = a / sum_a

        # full self‑convolution and objective
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {it:6d}] new best = {best_val:.6f}")
            if best_val <= 1.503:
                break
        else:
            no_improve += 1

        # schedule for the smooth max temperature
        elapsed = now - global_start
        total = deadline - global_start
        alpha = alpha_initial + (alpha_target - alpha_initial) * min(1.0, elapsed / total)

        # soft‑max weights on the convolution entries
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback to pure sub‑gradient (single max)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # gradient of the smoothed objective w.r.t. a
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)                # length 3·n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]        # slice of length n

        # back‑prop through the soft‑max (log‑space) map
        g_bar = np.dot(a, grad)
        du = a * (grad - g_bar)

        # Adam update on u
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * du
        v = beta2 * v + (1.0 - beta2) * (du * du)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        u = u - step_size * m_hat / (np.sqrt(v_hat) + adam_eps)

        # stagnation handling
        if no_improve >= max_no_improve:
            # perturb around the current best
            noise = np.random.normal(scale=0.5, size=n)
            u = np.log(best_a + eps) + noise
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            no_improve = 0
            print(f"[Adam {it}] restart after stagnation")
    return best_a, best_val

def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Intensive simulated‑annealing hill‑climbing (same as the reference).
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # Update convolution efficiently
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``x`` with itself.
    Length = 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``x`` and ``y`` using FFT.
    """
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient optimiser on the simplex
# ----------------------------------------------------------------------
def _adam_optimize(a0, n: int, deadline: float, global_start: float):
    """
    Optimises a probability vector of length ``n``.
    ``a0`` may be ``None`` (random Dirichlet start) or a numpy array.
    Returns the best (a, value) found before ``deadline``.
    """
    eps = 1e-12
    if a0 is None:
        a0 = np.random.dirichlet(np.ones(n))
    else:
        a0 = np.maximum(a0, eps)
        a0 = a0 / a0.sum()

    # log‑parametrisation guarantees positivity and makes projection trivial
    u = np.log(a0 + eps)

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    step_size = 0.12          # learning‑rate for Adam on the log‑variables
    adam_eps = 1e-8

    # smooth‑max schedule
    alpha_initial = 5.0
    alpha_target = 200.0

    best_a = a0.copy()
    best_val = _objective(best_a)
    no_improve = 0
    max_no_improve = 5_000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- build current a from log‑variables -----
        a = np.exp(u)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
            s = 1.0
        a = a / s

        # ----- evaluate current objective -----
        conv = _full_convolution(a)                # a * a
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {it:6d}] new best = {best_val:.6f}")
            if best_val <= 1.503:
                break
        else:
            no_improve += 1

        # ----- smooth‑max weight (soft‑max over the convolution entries) -----
        elapsed = now - global_start
        total = deadline - global_start
        alpha = alpha_initial + (alpha_target - alpha_initial) * min(1.0, elapsed / total)

        w = np.exp(alpha * (conv - max_conv))      # ≤ 1, sum ≤ len(conv)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fall back to the pure sub‑gradient (single max entry)
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # ----- gradient of the smoothed objective w.r.t. a -----
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)            # length = 3·n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]    # slice of length n

        # ----- chain rule: gradient w.r.t. log‑variables (u) -----
        g_bar = np.dot(a, grad)
        du = a * (grad - g_bar)

        # ----- Adam update on u -----
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * du
        v = beta2 * v + (1.0 - beta2) * (du * du)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        u = u - step_size * m_hat / (np.sqrt(v_hat) + adam_eps)

            u = np.log(best_a + eps) + noise
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            no_improve = 0

def _hillclimb_anneal(a0: np.ndarray,

        T = temp_start * (temp_end / temp_start) ** prog

        mask = (idx >= 0) & (idx < n)

        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights[i_src] = 0.0

                    accept = True

            a[j_tgt] += delta

            cur_val = new_val
