def _full_convolve(x, y):
    n = len(x)
    m = len(y)
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv
```

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Initialise a candidate of length ``n`` (default 8000):
   * interpolate a previous solution if ``height_sequence_1`` exists,
   * a few random Beta‑shaped vectors,
   * a uniform vector.
2. Pick the best initial candidate (according to the exact objective
   ``2·n·max(conv(a,a))``) and improve it with a projected Adam optimiser.
   * At each iteration we compute the full autocorrelation by FFT.
   * A smooth approximation of the max is used:
       w_i ∝ exp(α·(b_i−max(b)))   (α is increased from 5 to 80).
     The gradient of the smoothed objective is
       ∇f = 2·corr(a,w)  (cross‑correlation),
     efficiently obtained by another FFT.
   * Adam‑style moments are maintained, then a step is taken and the
     result is projected onto the probability simplex (Σa=1, a≥0).
   * Small random jitter is occasionally added to escape plateaus.
3. The best sequence found before the time budget expires is finally
   evaluated with ``evaluate_sequence`` and returned.

The whole search respects the 1000 s budget (a 0.5 s safety margin is kept
for the final call to ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a vector with itself.
    Length of result: 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two vectors (any lengths).
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a normalised vector (Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    Σx = 1,  x ≥ 0.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},   i = 1,…,n .
    Returns ``None`` if the shape degenerates to zero.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _interpolate_to_length(old: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``old`` to a new length ``new_len``."""
    if old.shape[0] == new_len:
        return old.copy()
    old_x = np.linspace(0.0, 1.0, old.shape[0])
    new_x = np.linspace(0.0, 1.0, new_len)
    new = np.interp(new_x, old_x, old)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        return np.full(new_len, 1.0 / new_len)
    return new / s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a short safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # problem size – a compromise between expressive power and speed
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Gather initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # a) previous solution, if it exists
    prev = globals().get('height_sequence_1')
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=float)
            cand = _interpolate_to_length(prev_arr, n)
            candidates.append(cand)
            print("[init] interpolated previous solution")
        except Exception:
            pass

    # b) uniform vector (always a feasible start)
    candidates.append(np.full(n, 1.0 / n))

    # c) a handful of random Beta shapes
    log_min, log_max = np.log(1e-3), np.log(20.0)
    for _ in range(8):
        logp = rng.uniform(log_min, log_max)
        logq = rng.uniform(log_min, log_max)
        p, q = np.exp(logp), np.exp(logq)
        cand = _beta_shape(p, q, n)
        if cand is not None:
            candidates.append(cand)

    # ------------------------------------------------------------------
    # 1️⃣  Pick the best of the initial candidates (exact objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for idx, cand in enumerate(candidates):
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[init] candidate {idx} best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Refine with projected Adam (smooth max approximation)
    # ------------------------------------------------------------------
    # Pre‑computed FFT size (covers the larger convolution a * w_rev)
    size = 1 << ((3 * n - 2) - 1).bit_length()

    a = best_seq.copy()
    m = np.zeros_like(a)          # first Adam moment
    v_arr = np.zeros_like(a)      # second Adam moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # hyper‑parameter schedules
    lr_start, lr_end = 0.3, 1e-4
    alpha_start, alpha_end = 5.0, 80.0
    jitter_prob = 0.001
    jitter_scale = 0.001

    global_best_seq = a.copy()
    global_best_val = best_val

    while True:
        now = time.time()
        if now >= deadline:
            break

        # schedule progress (0…1)
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (1.0 - progress) + lr_end * progress
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        # ----  compute convolution a*a  (full length 2n‑1)  ----
        A = np.fft.rfft(a, size)
        conv_self_full = np.fft.irfft(A * A, size)[:2 * n - 1]
        b = conv_self_full
        max_b = b.max()

        # ----  build smooth max weights  ----
        exp_term = alpha * (b - max_b)                # ≤ 0
        w = np.exp(exp_term)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ----  gradient of the smoothed objective  ----
        w_rev = w[::-1]
        W = np.fft.rfft(w_rev, size)
        conv_aw_full = np.fft.irfft(A * W, size)[:3 * n - 2]
        # slice the part that corresponds to indices 0 … n‑1
        grad_slice = conv_aw_full[n - 1:n - 1 + n][::-1]
        grad = 2.0 * grad_slice

        # ----  Adam update + projection ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_arr = beta2 * v_arr + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_arr / (1.0 - beta2 ** t)
        step = lr / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat

        # occasional jitter to escape flat plateaus
        if rng.random() < jitter_prob:
            a_candidate += rng.normal(scale=jitter_scale, size=n)

        # projection onto simplex (ensures Σa = 1, a ≥ 0)
        a = _project_simplex(a_candidate)

        # ----  evaluate new point (exact objective) ----
        A_new = np.fft.rfft(a, size)
        conv_self_new = np.fft.irfft(A_new * A_new, size)[:2 * n - 1]
        cur_val = float(2.0 * n * conv_self_new.max())

        if cur_val < global_best_val:
            global_best_val = cur_val
            global_best_seq = a.copy()
            print(f"[adam] new best = {global_best_val:.6f} (iter {t})")
            # stop early if we already beat the target
            if global_best_val <= 1.503:
                break

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = np.inf

    return global_best_seq.tolist()
