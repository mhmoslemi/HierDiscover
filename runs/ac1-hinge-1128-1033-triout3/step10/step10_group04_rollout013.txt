# ----------------------------------------------------------------------
# Basic convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector already normalised (sum = 1)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex {x ≥ 0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣  Adam optimisation on a log‑sum‑exp surrogate of max(conv)

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   start: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth approximation of the true max‑convolution."""
    # initialise logits (soft‑max parametrisation)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.004
    eps_start, eps_end = 1e-1, 1e-13

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 100_000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-13:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:07d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule (linear in wall‑clock progress)
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        # guard against overflow
        shifted = shifted - np.max(shifted)          # extra stability
        weights = np.exp(shifted)
        p = weights / weights.sum()                 # soft‑max over convolution positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a 2·n·∑ p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        # convolution a * p_rev (full length n+(2n‑1)‑1 = 3n‑2)
        out_len = n + (2 * n - 1) - 1
        size = 1 << (out_len - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv_full = np.fft.irfft(fa * fp, size)[:out_len]
        # the central n entries correspond to the gradient components
        grad_pre = conv_full[n - 1: 2 * n - 1][::-1]
        grad_a = 4.0 * n * grad_pre

        # back‑prop through the soft‑max (logits → probabilities)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.009:
            w += np.random.normal(scale=0.009, size=n)

        # restart if stuck for a long time
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:07d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.22
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         start: float,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Generic SA that moves mass between coordinates."""
    a = a0.copy()
    cur_val = best_val
    best_local_a = a.copy()
    best_local_val = cur_val

    # temperature schedule (geometric)
    T_start = max(0.02 * best_val, 1e-4)
    T_end = 1e-12

    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T_start * (T_end / T_start) ** progress

        # source index – probability proportional to current mass
        if np.random.rand() < 0.7:
            probs = a / a.sum()
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.random.randint(n))

        if a[i] < 1e-16:
            continue

        # target index – uniform choice (but not equal to source)
        j = int(np.random.randint(n - 1))
        if j >= i:
            j += 1

        # step size slowly shrinks
        max_frac = 0.35 * (1 - progress) + 0.001 * progress
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        new_val = _objective_value(a_candidate)

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_val - cur_val) / (T + 1e-30))

            a = a_candidate
            cur_val = new_val

            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Refine {it:07d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            break

        # active set – entries essentially at the maximum
        eps = max_b * 4e-7
        active = np.where(b >= max_b - eps)[0]
        if active.size == 0:
            break

        # weight w_i = Σ_{k∈active} a_{k‑i}
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or a[src] <= MIN_STEP:
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        delta = a[src] * step_factor
        if delta < MIN_STEP:
            step_factor *= 0.5
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        new_val = _objective_value(a_candidate)

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-13:
                best_local_val, best_local_a = new_val, a.copy()
        else:
            step_factor *= 0.71

        step_factor = max(step_factor * decay, MIN_STEP)
        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# 6️⃣  Small polishing: convex mixture with a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Mix the current vector with a small uniform component."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point – the function invoked by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the supplied time budget.
    Returns a list of floats (normalised to sum = 1) that can be fed
    directly to `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # Problem size – 9000 gives a good trade‑off between speed and bound.
    n = 9000

    # ------------------------------------------------------------------
    # 0️⃣  Initial parametric sweep (≈5 % of budget)
    # ------------------------------------------------------------------
    print("[Init] starting extended parametric sweep")
    best_a, best_val = _initial_candidate_extended(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    TARGET = 1.503  # goal that was set in the original brief

    # ------------------------------------------------------------------
    # 1️⃣  Adam on smoothed surrogate (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * (deadline - start_time)
    best_a, best_val = _adam_optimize(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase A] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Simulated annealing (≈20 % of budget)
    # ------------------------------------------------------------------
    phase_sa_deadline = start_time + 0.55 * (deadline - start_time)
    best_a, best_val = _simulated_annealing(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_sa_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase SA] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Targeted hill‑climbing (≈10 % of budget)
    # ------------------------------------------------------------------
    phase_hill_deadline = start_time + 0.65 * (deadline - start_time)
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_sa_deadline,
        deadline=phase_hill_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    best_val = _objective_value(best_a)
    print(f"[Phase Hill] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected sub‑gradient descent (≈15 % of budget)
    # ------------------------------------------------------------------
    phase_sgd_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_hill_deadline,
        deadline=phase_sgd_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase SGD] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Exact active‑set refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _refine_exact_phase(
        best_a, n,
        start=phase_sgd_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase Refine] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Report final score via the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
