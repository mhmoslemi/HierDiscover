def _expgrad_step(a, conv, alpha, eta):
    cmax = conv.max()
    # compute softmax weights
    w = np.exp(alpha * (conv - cmax))
    w /= w.sum()
    # compute gradient via correlation
    w_rev = w[::-1]
    size = 1 << ((len(a) + len(w_rev) - 1 - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fw = np.fft.rfft(w_rev, size)
    corr = np.fft.irfft(Fa * Fw, size)[:len(a) + len(w_rev) - 1]
    grad = 2.0 * corr[(2*len(a)-2) - np.arange(len(a))]
    # scaled gradient to relative step
    # optionally clip gradient values
    # compute exponent factor
    # shift to avoid overflow due to large eta * grad
    # compute scaled factor = exp(-eta * (grad - grad.max())) 
    # after scaling, a_new = a * factor; normalize.
    max_g = grad.max()
    factor = np.exp(-eta * (grad - max_g))
    a_new = a * factor
    s = a_new.sum()
    if s == 0:
        a_new = np.full_like(a, 1.0 / len(a))
    else:
        a_new /= s
    return a_new

def _smooth_step(a, conv, alpha, base_eta):
    cmax = conv.max()
    # compute softmax weights
    w = np.exp(alpha * (conv - cmax))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0/conv.size)
    else:
        w = w / w_sum
    # gradient via correlation
    w_rev = w[::-1]
    size = 1 << ((a.size + w_rev.size - 1 - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fw = np.fft.rfft(w_rev, size)
    corr = np.fft.irfft(Fa * Fw, size)[:a.size + w_rev.size - 1]
    grad = 2.0 * corr[(2*a.size - 2) - np.arange(a.size)]
    # scale step
    max_abs = np.max(np.abs(grad))
    if max_abs == 0:
        return a
    step = base_eta / max_abs
    a_new = a - step * grad
    a_new = _project_simplex(a_new)
    return a_new
```

Exponential‑gradient + soft‑max optimiser for the max‑convolution objective.

The routine works in three logical stages inside the allotted time budget:

1️⃣ **Initialisation** – a collection of seeds (the previous best,
   a grid of Beta‑shaped families, a few Dirichlet draws and the uniform
   vector) are interpolated / normalised and the best among them is
   recorded.

2️⃣ **Smooth optimisation** – for the first part of the run we minimise a
   smooth surrogate of the “max” using a soft‑max (log‑sum‑exp) over the
   full self‑convolution.  The gradient of this surrogate is obtained by
   a single FFT‑based correlation, after which an additive step with a
   Euclidean projection onto the probability simplex is taken.  The
   temperature α of the soft‑max grows from a small value to a very large
   one, gradually turning the smooth objective into the true max.

3️⃣ **Polishing** – once α is large the surrogate behaves like the true
   max.  We switch to a mirror‑descent update that only uses the M
   largest convolution entries (the same idea as the “beta‑family sweep”
   code you provided).  Stagnation is detected and broken by a brief
   Gaussian perturbation around the current best solution.

All iterations stay on the simplex (∑a = 1, a_i ≥ 0) and the best
sequence observed during the run is finally fed to `evaluate_sequence`.
The implementation respects the 1000 s time limit and prints occasional
progress reports.

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – fast for large n)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _correlation_with_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * sum_k w_k * a_{k-i}
    (i.e. 2 × cross‑correlation of a with w).
    a has length n, w has length 2n‑1.
    Returns a vector of length n.
    """
    n = a.shape[0]
    # reverse w for the correlation via convolution
    w_rev = w[::-1]
    size = 1 << ((n + w_rev.shape[0] - 1 - 1).bit_length())
    Fa = np.fft.rfft(a, size)
    Fw = np.fft.rfft(w_rev, size)
    corr = np.fft.irfft(Fa * Fw, size)[: n + w_rev.shape[0] - 1]
    offset = 2 * n - 2                      # = len(w_rev)-1
    # grad_i = 2 * corr[offset - i]
    grad = 2.0 * corr[offset - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for initial seeding
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Return a normalised Beta‑shaped sequence of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


# ----------------------------------------------------------------------
# Objective used during the search (sum(a) == 1)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth optimisation step (soft‑max surrogate of the max)
# ----------------------------------------------------------------------
def _smooth_step(a: np.ndarray,
                 conv: np.ndarray,
                 alpha: float,
                 base_eta: float) -> np.ndarray:
    """
    One iteration of the smooth optimisation.

    *conv* is the full self‑convolution of *a*.
    *alpha* is the inverse temperature of the soft‑max (larger ⇒ sharper).
    *base_eta* is a base step‑size; the actual step is scaled by the
    maximal absolute gradient component.
    """
    # soft‑max weights on convolution entries
    cmax = conv.max()
    # subtraction of cmax avoids overflow, all entries ≤ 0
    w = np.exp(alpha * (conv - cmax))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # gradient of the smooth surrogate (cross‑correlation)
    grad = _correlation_with_weights(a, w)

    # scale the step so that it does not explode on large gradients
    max_abs = np.max(np.abs(grad))
    if max_abs == 0.0:
        return a
    step = base_eta / max_abs
    a_new = a - step * grad
    a_new = _project_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Mirror‑descent step that uses only the M largest convolution entries
# ----------------------------------------------------------------------
def _mirror_update(a: np.ndarray,
                   conv: np.ndarray,
                   top_idx: np.ndarray,
                   weights: np.ndarray,
                   eta: float) -> np.ndarray:
    """
    Entropy‑mirror descent update restricted to a subset of convolution
    entries.  *top_idx* are the selected indices, *weights* sum to one.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    for k, w in zip(top_idx, weights):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        if low > high:
            continue
        j = np.arange(low, high + 1)
        a_shift = a[k - j]               # a_{k-j}
        g[j] += 2.0 * w * a_shift

    # multiplicative (entropy) update
    a_new = a * np.exp(-eta * g)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new
