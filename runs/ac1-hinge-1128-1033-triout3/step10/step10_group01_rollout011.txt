"""
Improved step‑function optimizer.

The code builds a non‑negative sequence whose value under
``evaluate_sequence`` (i.e. 2·n·max(a∗a) / sum(a)²) is as small as
possible.  The approach is:

1.  Generate a diverse set of starting points (uniform, arcsine‑like,
    random Dirichlet draws and a few random Beta‑family shapes).
2.  For each start run a *soft‑max* smoothed gradient descent.
    The hard “max’’ in the objective is replaced by a smooth
    soft‑max with a temperature that is increased from α≈0.5 to
    α≈50 during the run.  The gradient of the smoothed objective is
    obtained analytically (a convolution) and an Adam optimiser
    together with Euclidean projection onto the probability simplex
    keeps the iterates feasible.
3.  The best vector found across all starts is handed to the original
    simulated‑annealing hill‑climber (the “intensive hill‑climbing’’
    from the reference solution) for fine local optimisation.
4.  The final sequence is returned (the scaling is irrelevant for the
    evaluation function).

All heavy work respects the 1000 s budget.  Intermediate progress
(and the current best objective) is printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Full linear convolution (via FFT) – length 2·n‑1
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*x (full length 2·len(x)‑1)."""
    n = x.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a vector with sum = 1 (scale‑invariant)
# ----------------------------------------------------------------------
def _objective_norm(a: np.ndarray) -> float:
    """2·n·max(a∗a) assuming Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient of the soft‑max smoothed objective
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, alpha: float):
    """
    Compute the gradient of f(a)=∑_k w_k·(a∗a)_k where
    w_k ∝ exp(α·(a∗a)_k) (soft‑max).  Returns (grad, conv).
    """
    n = a.shape[0]

    # full convolution a*a
    conv = _full_convolution_fft(a)

    # soft‑max weights (stable numerics)
    maxc = conv.max()
    shifted = conv - maxc
    exp_alpha = np.exp(alpha * shifted)
    w = exp_alpha / exp_alpha.sum()        # ∑ w = 1, all w ≥ 0

    # gradient = 2·∑_k w_k·a_{k‑j}
    #   = 2·(w_rev * a)_{2n‑2‑j}
    w_rev = w[::-1]                       # reverse the weight vector
    size = 1 << ((3 * n - 2).bit_length())   # size for convolution w_rev * a
    fw = np.fft.rfft(w_rev, size)
    fa = np.fft.rfft(a, size)
    conv_rev_full = np.fft.irfft(fw * fa, size)[: 3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_rev_full[idx]
    return grad, conv


# ----------------------------------------------------------------------
# Smoothed gradient descent (Adam + simplex projection)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max smoothed objective.
    Alpha is increased from 0.5 to 50 as the run progresses.
    Returns the best vector seen and its exact (hard‑max) objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_norm(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    it = 0
    no_improve = 0

    # schedule limits for α
    alpha_start = 0.5
    alpha_end = 50.0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress fraction w.r.t the TOTAL allocated time for this run
        prog = (now - start_time) / (deadline - start_time + 1e-12)
        alpha = alpha_start * (alpha_end / alpha_start) ** prog

        grad, conv = _softmax_gradient(a, alpha)

        # Adam update (bias‑corrected)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)

        cur_val = _objective_norm(a_new)

        # keep the best exact value seen
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a_new.copy()
            no_improve = 0
            print(f"[SmoothOpt {it:6d}] new best = {best_val:.6f}")
            # tiny target – stop early if we are already very good
            if best_val <= 1.48:
                a = a_new
                break
        else:
            no_improve += 1

        # occasional random jitter to escape stagnation
        if no_improve > 5000:
            a_new = np.maximum(a_new + np.random.normal(scale=0.01, size=n), 0.0)
            a_new = a_new / a_new.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[SmoothOpt {it:6d}] jitter to escape plateau")

        a = a_new

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.48) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time + 1e-12)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – more likely if it contributes strongly
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – prefers indices that are currently light
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update convolution efficiently
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s
    time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the final evaluate

    # prefer a single moderate size – 8000 worked well in the reference
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise diverse seeds
    # ------------------------------------------------------------------
    seeds = []

    # 0.1 uniform
    seeds.append(np.ones(n) / n)

    # 0.2 arcsine‑like (Beta(0.5,0.5))
    i = np.arange(1, n + 1, dtype=np.float64)
    arcsine = (i * (n - i + 1)) ** (-0.5)
    arcsine = arcsine / arcsine.sum()
    seeds.append(arcsine)

    # 0.3 a few random Beta‑family shapes
    for _ in range(4):
        p = 10 ** np.random.uniform(-1.0, 1.0)   # 0.1 … 10
        q = 10 ** np.random.uniform(-1.0, 1.0)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a = a / a.sum()
        seeds.append(a)

    # 0.4 Dirichlet draws with different concentration levels
    for conc in [0.3, 1.0, 5.0]:
        a = np.random.dirichlet(np.full(n, conc))
        seeds.append(a)

    # 0.5 interpolate previous best if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a_interp = np.interp(new_x, old_x, old)
        a_interp = np.maximum(a_interp, 0.0)
        a_interp = a_interp / a_interp.sum()
        seeds.append(a_interp)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Smoothed gradient descent on all seeds (time‑boxed)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = np.inf

    # allocate roughly half of the budget to the smoothed optimisation
    smooth_deadline = start_time + (deadline - start_time) * 0.45

    for s_idx, a0 in enumerate(seeds):
        if time.time() >= smooth_deadline:
            break
        # ensure a proper probability vector
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()

        # give each seed up to a few seconds (or whatever remains)
        per_seed_deadline = time.time() + min(6.0,
                                             smooth_deadline - time.time())
        a_opt, val_opt = _smooth_optimize(a0, n, per_seed_deadline, start_time)

        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
            print(f"[seed {s_idx}] new global best = {global_best_val:.6f}")

        # early exit if the target is already beaten
        if global_best_val <= 1.48:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Final intensive hill‑climbing on the best sequence found
    # ------------------------------------------------------------------
    if global_best_a is not None and time.time() < deadline:
        remaining = deadline - time.time()
        hc_deadline = time.time() + remaining
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            n,
            deadline=hc_deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.48,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation (just for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
