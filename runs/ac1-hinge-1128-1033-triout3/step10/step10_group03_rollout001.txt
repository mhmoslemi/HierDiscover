Propose a non‑negative step‑function sequence that (approximately) minimizes
``evaluate_sequence``.  The algorithm consists of three phases:

1️⃣  **Initialisation** – build a diverse pool of candidates from
    * the previous solution (if available),
    * the Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1},
    * a “U‑shaped’’ family a_i ∝ (i+1)^{-p} + (n-i)^{-p},
    * a handful of Dirichlet draws.

   f(a)=2·n·max(conv(a,a)) with Σa=1) becomes the starting point.

2️⃣  **Projected sub‑gradient descent** – repeatedly

    * average their sub‑gradients (∂max/∂a_i = 2·a_{k‑i} if the index is valid),

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective (assuming Σa = 1):
        f(a) = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        Σx = 1,  x_i ≥ 0
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho such that u[rho] > theta
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Beta‑family shape a_i ∝ i^{p‑1} (n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _u_shape(p: float, n: int) -> np.ndarray:
    """
    “U‑shaped’’ family:
        a_i ∝ (i+1)^{-p} + (n-i)^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    a = ( (i + 1.0) ** (-p) + (n - i) ** (-p) )
    a_sum = a.sum()
    if a_sum == 0.0:   # extremely unlikely
        a = np.full_like(a, 1.0 / n)
    else:
        a /= a_sum
    return a


def _evaluate_obj_with_conv(a: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (objective, full convolution) for a (assumed normalized)."""
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    obj = float(2.0 * n * max_b)
    return obj, b
