def _corr_gradient(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    # w length = 2n-1; a length n
    # returns 2 * correlation(w, a) valid (length n)
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((m + n - 1).bit_length())
    # compute FFT of w and a
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    # correlation: inverse FFT of fw * conj(fa)
    corr_full = np.fft.irfft(fw * np.conj(fa), size)
    # take valid part (slice n-1 to n-1+n)
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad
```

"""
Propose candidate step‑function heights.

The algorithm consists of three phases:

1️⃣  Initialise a probability vector (non‑negative, sum = 1) from a
    previously saved solution if available, otherwise from a small
    pool of simple parametric families (beta‑shapes, uniform, arcsine,
    a few Dirichlet draws).  The best initial vector is kept.

2️⃣  Optimise the exact objective  O(a)=2·n·max(conv(a,a))  using a
    smoothed‑max surrogate:
        M_τ(a)=τ·log∑_k exp(conv_k/τ) .
    Its gradient is  ∇M_τ = 2·corr(w,a)  where w is the soft‑max
    over the convolution entries.  We compute the gradient by FFT‑based
    correlation and perform Adam updates on the simplex (projected
    Euclidean projection).  The temperature τ is annealed linearly
    from 2·10⁻¹ to 1·10⁻⁴, while the learning rate decays from 1.0 to
    1·10⁻³.

3️⃣  A short “polishing’’ stage runs a classic projected sub‑gradient
    descent (the same update used in the reference code) to fine‑tune
    the solution.

The best vector seen during the whole run is finally evaluated with
``evaluate_sequence`` and returned.  All operations respect the 1000 s
time budget and keep every entry non‑negative.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a))  (assumes Σa = 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},   i = 1,…,n .
    If the parameters produce a zero vector None is returned.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _grad_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute 2·corr(w, a) where `w` has length 2·n‑1 and `a` length n.
    The correlation is the “valid’’ part:   grad_i = 2·Σ_j w_{i+j}·a_j .
    Implemented via FFT for speed.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reverse for convolution
    size = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[: len(w) + n - 1]
    start = n - 1                       # start of the “valid’’ slice
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Reserve a small safety margin for the final evaluation call.
    deadline = start_time + budget_s - 0.5

    # Problem size – a compromise between expressive power and speed.
    # Feel free to increase this if the budget permits.
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Build a pool of initial candidates and pick the best.
    # ------------------------------------------------------------------
    candidates = []

    # 0a – use a previously saved solution when available.
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        candidates.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # 0b – a few classic analytic shapes.
    candidates.append(np.full(n, 1.0 / n))                 # uniform
    candidates.append(_beta_shape(0.5, 0.5, n))            # arcsine (Beta(½,½))

    # 0c – a dense grid over the Beta family.
    p_grid = np.concatenate(
        [
            np.geomspace(0.001, 0.1, 8, endpoint=False),
            np.geomspace(0.1, 5.0, 12, endpoint=False),
            np.geomspace(5.0, 20.0, 5),
        ]
    )
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is not None:
                candidates.append(cand)

    # 0d – a handful of random Dirichlet draws for extra diversity.
    rng = np.random.default_rng()
    for _ in range(6):
        candidates.append(rng.dirichlet(np.ones(n)))

    # Pick the best among the pooled candidates.
    best_a = None
    best_val = np.inf
    for idx, cand in enumerate(candidates):
        if cand is None:
            continue
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[init] candidate {idx} best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Smooth‑max Adam optimisation (the heavy lifting).
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_seq = a.copy()
    best_score = best_val

    # Adam hyper‑parameters.
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # --- evaluate current objective ---
        b = _full_convolution(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            print(f"[adam] new best = {best_score:.6f} (iter {iteration})")
            # target reached? (the benchmark is 1.503)
            if best_score <= 1.503:
                break

        # --- annealing schedules ---
        progress = (now - start_time) / (deadline - start_time)
        # temperature for the smooth‑max (log‑sum‑exp)
        tau = 0.2 * (1.0 - progress) + 1e-4 * progress
        # learning‑rate schedule (log‑linear decay)
        lr = 1.0 * (1.0 - progress) + 1e-3 * progress

        # --- soft‑max weights over convolution entries ---
        # shift by max_b to avoid overflow; all exponents ≤ 0.
        w = np.exp((b - max_b) / tau)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # --- gradient of the smooth‑max ---
        grad = _grad_corr(w, a)          # shape (n,)

        # --- Adam update (projected onto the simplex) ---
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a_candidate)

        # occasional tiny jitter to escape flat plateaus
        if np.random.rand() < 0.001:
            a += np.random.normal(scale=5e-4, size=n)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # 2️⃣  Optional polishing with classic projected sub‑gradient descent.
    # ------------------------------------------------------------------
    # (run only if time permits)
    if time.time() < deadline - 0.5:
        # reuse same step‑size schedule but without smoothing
        step_start, step_end = 0.3, 1e-5
        mu = 0.9                     # momentum factor
        v_mom = np.zeros_like(a)

        while True:
            now = time.time()
            if now >= deadline - 0.5:
                break
            # current objective
            b = _full_convolution(a)
            max_b = b.max()
            cur_score = 2.0 * n * max_b
            if cur_score < best_score:
                best_score, best_seq = cur_score, a.copy()
                print(f"[subgrad] new best = {best_score:.6f}")

            # sub‑gradient of the max entry
            k_max = int(np.argmax(b))
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[idx[mask]]

            # step‑size (linear decay)
            progress = (now - start_time) / (deadline - start_time)
            step = step_start * (1.0 - progress) + step_end * progress

            # momentum + projection
            v_mom = mu * v_mom - step * grad
            a_candidate = a + v_mom
            a = _project_simplex(a_candidate)

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
