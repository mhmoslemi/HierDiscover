"""
Improved step‑function optimizer.

The algorithm consists of three stages:

1️⃣  **Initialisation** – reuse the previously best sequence (if any) and
    optionally interpolate it to the chosen length.  Otherwise start from a
    random Dirichlet draw.

2️⃣  **Smoothed exponential‑gradient descent** – we minimise a smooth
    surrogate of the objective

        Fβ(a) = (1/β)·log(∑_k exp(β·(a*a)_k))

    which converges to the true max when β → ∞.  For a fixed β the gradient
    of Fβ is

        ∇Fβ(a)_i = 2·∑_k w_k·a_{k-i},   w_k ∝ exp(β·(a*a)_k),

    i.e. a correlation of the probability vector w with a.  The gradient is
    computed by FFT‑based correlation, and the update is an *exponentiated
    gradient* (mirror descent on the simplex):
        a ← a·exp(−η·∇Fβ)  and renormalise.
    β is increased in a continuation schedule (0.5 → 160) so that the
    optimisation first moves globally and then refines the true max.

3️⃣  **Fine‑grained hill‑climbing** – the same simulated‑annealing mass‑transfer
    scheme used in the reference solution is applied to the best sequence
    obtained from the smoothed descent.

The whole procedure respects the 1000 s budget and returns a list of floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent (continuation on β)
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    beta_schedule,
                    deadline: float,
                    start_time: float,
                    target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth surrogate of max(conv) using exponentiated
    gradient updates on the simplex.  Returns the best a found and its
    true objective value.
    """
    a = a0.copy()
    cur_val = _objective(a)
    best_a, best_val = a.copy(), cur_val

    # FFT size sufficient for both the convolution (2n‑1) and the correlation
    # (n + (2n‑1) - 1 = 3n‑2).
    size = 1 << ((3 * n - 2).bit_length())

    for beta in beta_schedule:
        # base step size – larger for small β, tiny for large β
        base_eta = 0.5 / (beta + 1.0)
        eta = base_eta

        max_iter = 500
        for it in range(max_iter):
            now = time.time()
            if now >= deadline:
                return best_a, best_val

            # ----- compute gradient at current a ---------------------------------
            A = np.fft.fft(a, size)                     # FFT of a (zero‑padded)
            conv_full = np.fft.ifft(A * A).real[:2 * n - 1]  # a*a
            bmax = conv_full.max()
            # smoothed weights w_k ∝ exp(β·(b_k-bmax))
            w = np.exp(beta * (conv_full - bmax))
            w /= w.sum()

            # correlation a ⋆ w  →  gradient = 2·(a ⋆ w)
            W = np.fft.fft(w, size)
            corr_full = np.fft.ifft(np.conj(A) * W).real
            grad = 2.0 * corr_full[:n]   # gradient of Fβ

            # ----- line‑search with exponentiated update -------------------------
            step_eta = eta
            improved = False
            for _ in range(15):                 # back‑track attempts
                # exponentiated‑gradient step (mirrored descent on simplex)
                a_candidate = a * np.exp(-step_eta * grad)
                s = a_candidate.sum()
                if s > 0.0:
                    a_candidate /= s
                else:
                    # numerical disaster – re‑initialise
                    a_candidate = np.random.dirichlet(np.ones(n))

                # evaluate true objective (no smoothing)
                A_c = np.fft.fft(a_candidate, size)
                conv_c = np.fft.ifft(A_c * A_c).real[:2 * n - 1]
                val_c = 2.0 * n * conv_c.max()

                if val_c < cur_val - 1e-15:
                    a = a_candidate
                    cur_val = val_c
                    improved = True
                    if val_c < best_val - 1e-12:
                        best_val, best_a = val_c, a.copy()
                        print(f"[Smooth β={beta:.2f} it={it}] new best = {best_val:.6f}")
                        if best_val <= target_val:
                            return best_a, best_val
                    # increase step size slightly after success
                    step_eta *= 1.1
                    break
                else:
                    step_eta *= 0.5          # shrink step

            if not improved:
                # no improvement → shrink base step size, eventually give up
                eta *= 0.5
                if eta < 1e-8:
                    break

        # end of iterations for this β
    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive hill‑climbing (same as reference, minor tweaks)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly transfers a small amount of mass from
    indices that dominate the current maximal convolution entry to indices
    that contribute little.  A temperature schedule allows occasional
    uphill moves.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        # update convolution entries affected by moving delta from i_src to j_tgt
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        elif T > 0:
            prob = math.exp(-(new_val - cur_val) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s
    time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – 8000 is a good trade‑off for the time budget.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from the previous best if available.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if prev.ndim != 1:
            raise ValueError
        if prev.shape[0] != n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, prev)
        else:
            a0 = prev.copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0.0:
            a0 /= a0.sum()
        else:
            a0 = np.random.dirichlet(np.ones(n))
        print("[init] interpolated previous solution")
    except Exception:
        # Fallback random start
        a0 = np.random.dirichlet(np.ones(n))
        print("[init] random Dirichlet start")

    # ------------------------------------------------------------------
    # Global best trackers
    # ------------------------------------------------------------------
    global_best_a = a0.copy()
    global_best_val = _objective(global_best_a)

    # ------------------------------------------------------------------
    # Stage 1 – smoothed exponential‑gradient descent (continuation on β)
    # ------------------------------------------------------------------
    beta_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]

    # allow a few independent restarts if time permits
    restarts = 3
    for r in range(restarts):
        if time.time() >= deadline - 5.0:
            break
        if r == 0:
            start_a = global_best_a.copy()
        else:
            start_a = np.random.dirichlet(np.ones(n))
        a_opt, val_opt = _smooth_descent(start_a,
                                         n,
                                         beta_sched,
                                         deadline,
                                         start_time,
                                         target_val=1.503)
        if val_opt < global_best_val:
            global_best_a, global_best_val = a_opt.copy(), val_opt
            print(f"[restart {r}] new global best after smoothing = {global_best_val:.6f}")

        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # Stage 2 – intensive hill‑climbing (same code as the reference)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        global_best_a, global_best_val = _hillclimb_anneal(global_best_a,
                                                            n,
                                                            deadline,
                                                            start_time,
                                                            global_best_a,
                                                            global_best_val,
                                                            target_val=1.503)
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation (the competition harness will also call it)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
