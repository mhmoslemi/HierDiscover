    provided ``evaluate_sequence`` which normalises by the sum.

2️⃣  Deterministic “mass‑transfer’’ descent.  At every iteration the
    convolution peak `k_max` is identified.  The contribution of each
    index `i` to this peak is `c_i = a_i·a_{k_max‑i}` (if the partner
    index exists).  Mass is moved from the index with the largest
    contribution to an index that does **not** affect the same peak
    (or, if that is impossible, to the index whose partner contribution
    is smallest).  The step size is chosen so that the peak value would
    drop at most to the second‑largest convolution value; the move is
    applied only if the peak really decreases.  The convolution is
    updated incrementally (O(n) per step) and the vector is re‑normalised
    occasionally to avoid numerical drift.

3️⃣  A tiny convex combination with the uniform distribution to polish
    the final answer.

All helpers are top‑level functions; the entry point required by the
evaluation harness is ``propose_candidate``.  The routine respects the
time budget (default 1000 s) and returns a list of non‑negative floats
that minimises the objective.
"""

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector with Σ a = 1 (i.e. 2·n·max(b))."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int, rng: np.random.Generator):
    """Search a collection of parametric families for a low‑objective seed."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)               # 0 … n‑1
    i1 = i + 1.0                                 # 1 … n

    # 1. Exponential tails (decreasing and symmetric U‑shape)
    for r in np.linspace(0.995, 0.99999, 30):
        _try(r ** i)                             # left‑hand
        _try(r ** i + r ** (n - 1 - i))          # symmetric

    # 2. Power‑law tails (left‑hand and symmetric)
    for p in np.linspace(0.1, 4.0, 40):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Inverse‑sqrt (special case)
    _try(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))

    # 4. Beta densities – denser grid around the U‑shape (a,b < 1)
    beta_vals = np.linspace(0.1, 1.5, 18)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a_vec = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a_vec)

    # 5. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # 6. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 14):
        _try(base_sin ** p)

    # 7. Piecewise‑constant “high‑at‑edges’’ shapes
    for k_frac in np.linspace(0.004, 0.08, 14):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.04, 0.35, 14):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a_vec = np.full(n, low)
            a_vec[:k] = high_frac
            a_vec[-k:] = high_frac
            _try(a_vec)

    # 8. Random Dirichlet draws for diversity
    for _ in range(200):
        _try(rng.dirichlet(np.ones(n)))

    # 9. Interpolation of a previously found solution, if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – deterministic greedy mass‑transfer descent
# ----------------------------------------------------------------------
def _greedy_descent(a0: np.ndarray, n: int,
                    start: float, deadline: float,
                    best_a: np.ndarray, best_val: float,
                    target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Greedy optimisation of the exact max‑convolution.
    Repeatedly moves probability mass from the index that contributes
    most to the current peak towards an index that does not affect that
    peak (or has the smallest counterpart).  The step size is chosen
    so that the peak would drop at most to the second‑largest value.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = best_a.copy()
    best_local_val = best_val

    no_improve = 0
    it = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1. locate the current maximal convolution bin
        # ------------------------------------------------------------------
        k_max = int(np.argmax(b))
        M = b[k_max]

        # second‑largest value (ignoring the current maximum)
        if k_max == 0:
            M2 = b[1:].max()
        elif k_max == len(b) - 1:
            M2 = b[:-1].max()
        else:
            M2 = max(b[:k_max].max(), b[k_max + 1:].max())

        # ------------------------------------------------------------------
        # 2. determine source index (largest contribution to the peak)
        # ------------------------------------------------------------------
        start_idx = max(0, k_max - (n - 1))
        end_idx = min(n - 1, k_max)
        idx = np.arange(start_idx, end_idx + 1, dtype=int)
        partner = k_max - idx                     # counterpart indices
        contrib = a[idx] * a[partner]             # a_i * a_{k‑i}
        src_local = int(np.argmax(contrib))
        src = idx[src_local]
        partner_src = partner[src_local]          # index that pairs with src

        # ------------------------------------------------------------------
        # 3. choose a target index that does not increase this peak
        # ------------------------------------------------------------------
        # indices that lie outside the range that contributes to k_max
        outs = []
        if start_idx > 0:
            outs.append(np.arange(0, start_idx, dtype=int))
        if end_idx < n - 1:
            outs.append(np.arange(end_idx + 1, n, dtype=int))
        if outs:
            tgt_candidates = np.concatenate(outs)
            tgt = int(tgt_candidates[np.argmin(a[tgt_candidates])])
            # counterpart of the target is outside the valid range ⇒ zero
            partner_tgt = None
            diff = a[partner_src]                  # reduction factor
        else:
            # every index participates in the peak → pick the minimal counterpart
            min_partner = float('inf')
            tgt = src
            for ii in idx:
                if ii == src:
                    continue
                pk = k_max - ii
                if a[pk] < min_partner:
                    min_partner = a[pk]
                    tgt = ii
            partner_tgt = k_max - tgt
            diff = a[partner_src] - a[partner_tgt]

        if diff <= 0.0:
            # nothing useful can be done with this peak; abort early
            no_improve += 1
            if no_improve > IMPROVE_LIMIT:
                # random perturbation to escape a possible deadlock
                a = best_local_a.copy()
                a += np.random.normal(scale=0.005, size=n)
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                b = _full_convolution(a)
                cur_val = 2.0 * n * b.max()
                no_improve = 0
                print(f"[GD {it}] perturb after dead‑lock")
            it += 1
            continue

        # ------------------------------------------------------------------
        # 4. compute a step size that would (ideally) bring the peak down to M2
        # ------------------------------------------------------------------
        progress = (time.time() - start) / (deadline - start)
        step_frac = 0.5 * (1.0 - progress) + 0.001 * progress   # decays from 0.5 to 0.001
        delta_needed = (M - M2) / (2.0 * diff) if (M - M2) > 0 else 0.0
        delta = min(a[src] * step_frac, delta_needed * 0.9)        # a small safety factor

        if delta <= 1e-15:
            # move is too tiny – count as no improvement
            no_improve += 1
            it += 1
            continue

        # ------------------------------------------------------------------
        # 5. incremental update of the convolution (O(n))
        # ------------------------------------------------------------------
        b_new = b.copy()
        # removal from source
        b_new[src:src + n] -= 2.0 * delta * a
        # addition to target
        b_new[tgt:tgt + n] += 2.0 * delta * a
        # diagonal corrections
        b_new[2 * src] += delta * delta
        b_new[2 * tgt] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_M = b_new.max()

        # ------------------------------------------------------------------
        # 6. accept only if the peak really decreased
        # ------------------------------------------------------------------
        if new_M < M - 1e-15:
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = 2.0 * n * new_M
            no_improve = 0
            if cur_val < best_local_val - 1e-15:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[GD {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # 7. occasional re‑normalisation to curb numerical drift
        # ------------------------------------------------------------------
        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        # ------------------------------------------------------------------
        # 8. restart after long stagnation
        # ------------------------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            a = best_local_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[GD {it:7d}] perturb after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small final polishing – blend with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.0005, 0.001, 0.002, 0.004, 0.008):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ ≈ 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # a few thousand elements give a good trade‑off between speed and quality
    n = 8192

    # ------------------------------------------------------------------
    # Phase 0 – broad parametric sweep (cheap but informative)
    # ------------------------------------------------------------------
    print("[Init] parametric sweep")
    best_a, best_val = _initial_candidate(n, rng)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic greedy descent (exact max‑convolution)
    # ------------------------------------------------------------------
    best_a, best_val = _greedy_descent(
        best_a, n,
        start=start_time,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
    )
    print(f"[GD] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – tiny uniform mixing polish
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided wrapper (just for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
