```
def compute_gradient_smooth(a, gamma):
    n = len(a)
    size = 1 << ((2*n-2).bit_length())
    # compute conv
    fx = np.fft.rfft(a, size)
    conv = np.fft.irfft(fx * fx, size)[:2*n-1]
    # compute w = softmax(gamma * conv)
    # To avoid overflow, subtract max
    maxc = conv.max()
    exp_term = np.exp(gamma * (conv - maxc))
    w = exp_term / exp_term.sum()
    # compute term2 = sum_i a_i * w_{i+j}
    # compute w_rev
    w_rev = w[::-1]  # reversed
    # convolution a * w_rev
    fx_a = np.fft.rfft(a, size)
    fx_wr = np.fft.rfft(w_rev, size)
    conv2 = np.fft.irfft(fx_a * fx_wr, size)[: (len(a)+len(w_rev)-1)]
    # conv2 length = n + (2n-1) -1 = 3n-2? Actually len(w_rev) = 2n-1, so conv2 length = 3n-2
    # term2_j = conv2[2n-2 - j]
    term2 = conv2[ (2*n-2) - np.arange(n) ]
    grad = a + term2
    return grad, conv, w
```

"""
Propose a step‑function that minimises the evaluation score.

The algorithm starts from several simple seeds (uniform, interpolated
previous solution, Dirichlet draws, a few Beta‑family shapes) and keeps the
best one.  It then performs a *smooth‑max* projected gradient descent on the
simplex: the objective 2·n·max(a∗a) is approximated by
Mγ(a)=2·n·(1/γ)·log∑exp(γ·(a∗a)).
The gradient of Mγ can be written in closed form and is evaluated by two
FFT‑based convolutions per iteration (full linear convolution of a with
itself and convolution of a with the reversed soft‑max weights).  A simple
back‑tracking line‑search adapts the step size, while the iterate is
projected onto the probability simplex after each step.  The smoothing
parameter γ is increased gradually (starting from 1 and growing to 64) so
that the optimisation first explores a smooth landscape and then focuses on
the true max.

If progress stalls, a random perturbation or a full restart from a random
Dirichlet vector is performed.  After the gradient phase, a short
hill‑climbing phase further reduces the exact maximum by moving a small
amount of mass from indices that contribute most to the current peak to
indices that contribute little.

The whole routine respects the 1000 s wall‑time limit, returning the best
sequence found (as a list of non‑negative floats).  All helper functions are
top‑level and avoid closures.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise exactly to 1
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / len(w))
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Gradient of the smoothed max objective
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, gamma: float, fft_size: int):
    """
    Return (grad_pre, conv, w) where
        grad_pre = a + term2  (the sub‑gradient of the *unsmoothed* max multiplied by 1/(2n))
        conv    = a * a (full linear convolution)
        w       = soft‑max weights: w_k = exp(γ·conv_k)/∑exp(γ·conv)
    The true gradient of the objective 2·n·max(a∗a) is
        grad = 2·n * grad_pre .
    """
    n = a.shape[0]

    # FFT of a (re‑used)
    a_fft = np.fft.rfft(a, fft_size)

    # full auto‑convolution a∗a
    conv = np.fft.irfft(a_fft * a_fft, fft_size)[: 2 * n - 1]

    # soft‑max over the convolution entries
    maxc = conv.max()
    # subtract max for numerical stability
    exp_vals = np.exp(gamma * (conv - maxc))
    w = exp_vals / exp_vals.sum()

    # term2_j = Σ_i a_i · w_{i+j}
    # compute via convolution of a with reversed w
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, fft_size)
    conv_rev = np.fft.irfft(a_fft * w_rev_fft, fft_size)[: (n + (2 * n - 1) - 1)]
    # indices  (2n‑2‑j) give the required shift
    term2 = conv_rev[(2 * n - 2) - np.arange(n)]

    grad_pre = a + term2
    return grad_pre, conv, w


# ----------------------------------------------------------------------
# Simple hill‑climbing that shuffles a small amount of mass
# ----------------------------------------------------------------------
def _hillclimb(a: np.ndarray, deadline: float, start_time: float):
    """
    Local search that repeatedly moves a tiny fraction of probability mass
    from indices that dominate the current maximal convolution entry to
    indices that contribute little.  The routine stops when time runs out
    or when the exact objective falls below the target 1.503.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)
    cur_a = a.copy()
    cur_val = best_val
    cur_conv = _full_convolution(cur_a)

    # configuration
    target = 1.503
    eps = 1e-12
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        # locate the peak of the convolution
        k_max = int(np.argmax(cur_conv))

        # contributions of each index to the peak entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(cur_a)
        contrib[mask] = cur_a[mask] * cur_a[idx[mask]]
        total = contrib.sum()

        # choose a source index (biased by contribution)
        if total > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total)
        else:
            src = int(np.argmax(contrib))

        if cur_a[src] <= eps:
            # nothing to move – stochastic restart
            src = np.random.randint(n)

        # pick a target index (prefer those with small contribution)
        tgt_weights = np.ones_like(cur_a)
        tgt_weights[src] = 0.0
        # down‑weight indices that already have large a_j
        tgt_weights = 1.0 / (tgt_weights + eps)
        tgt_weights[src] = 0.0
        tgt = np.random.choice(n, p=tgt_weights / tgt_weights.sum())

        # amount to transfer (fraction of source mass)
        delta = cur_a[src] * np.random.rand() * 0.02  # ≤2 % of src
        if delta < eps:
            continue

        # tentative move
        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-15:
            # accept
            cur_a = a_new
            cur_val = new_val
            cur_conv = _full_convolution(cur_a)
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, cur_a.copy()
                print(f"[Hill {iteration:6d}] new best = {best_val:.6f}")
                if best_val <= target:
                    break
        else:
            no_improve += 1
            if no_improve > IMPROVE_LIMIT:
                # perturb and reset counters
                cur_a = best_a.copy()
                noise = np.random.normal(scale=0.01, size=n)
                cur_a = np.maximum(cur_a + noise, 0.0)
                cur_a /= cur_a.sum()
                cur_val = _objective(cur_a)
                cur_conv = _full_convolution(cur_a)
                no_improve = 0
                print(f"[Hill {iteration:6d}] stagnation – perturb")
        iteration += 1
    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to minimise
    ``evaluate_sequence``.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  20000 offers many degrees of freedom yet
    # stays comfortably within the time budget.
    # ------------------------------------------------------------------
    n = 20000
    fft_size = 1 << ((3 * n - 2 - 1).bit_length())   # for the two convolutions

    # ------------------------------------------------------------------
    # Generate a pool of initial seeds and pick the best one.
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # warm‑start from the previous solution (if it exists)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a_interp = np.interp(new_x, old_x, old)
        a_interp = np.maximum(a_interp, 0.0)
        a_interp /= a_interp.sum()
        init_candidates.append(a_interp)
        print("[Init] interpolated previous solution")
    except Exception:
        pass

    # a few random Dirichlet draws (high diversity)
    for _ in range(4):
        init_candidates.append(np.random.dirichlet(np.ones(n)))

    # a handful of Beta‑family shapes (U‑/bell‑shaped)
    i = np.arange(1, n + 1, dtype=np.float64)
    for p, q in [(0.5, 0.5), (0.5, 1.5), (1.5, 0.5), (0.8, 0.8), (2.0, 2.0)]:
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a /= a.sum()
        init_candidates.append(a)

    best_a = None
    best_val = np.inf
    for a in init_candidates:
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    print(f"[Init] best initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Smooth‑max projected gradient descent
    # ------------------------------------------------------------------
    a_curr = best_a.copy()
    cur_val = best_val
    gamma = 1.0                      # start smoothing
    gamma_max = 64.0                 # final smoothing (close to true max)
    gamma_factor = 1.02              # increase per iteration
    step = 1.0                       # initial step size (to be adapted)
    min_step = 1e-8
    max_step = 5.0
    no_improve = 0
    IMPROVE_LIMIT = 300_000
    iter_cnt = 0

    while time.time() < deadline:
        # compute gradient and convolution at current point
        grad_pre, conv, _ = _grad_smooth_max(a_curr, gamma, fft_size)
        # true gradient of the original (unsmoothed) objective
        grad = (2.0 * n) * grad_pre

        # --- back‑tracking line search ---
        step_try = step
        while True:
            a_trial = a_curr - step_try * grad
            a_trial = np.maximum(a_trial, 0.0)
            a_trial_sum = a_trial.sum()
            if a_trial_sum == 0.0:
                # avoid division by zero – random restart
                a_trial = np.random.dirichlet(np.ones(n))
                a_trial_sum = 1.0
            a_trial /= a_trial_sum
            trial_val = _objective(a_trial)

            if trial_val < cur_val - 1e-15:
                # accept
                a_curr = a_trial
                cur_val = trial_val
                # enlarge step a little
                step = min(step_try * 1.07, max_step)
                no_improve = 0
                if trial_val < best_val - 1e-12:
                    best_val, best_a = trial_val, a_curr.copy()
                    print(f"[PGD {iter_cnt:6d}] new best = {best_val:.6f}")
                    if best_val <= 1.503:
                        break
                break
            else:
                # reject – shrink step
                step_try *= 0.5
                if step_try < min_step:
                    # give up on this direction, perform a small random perturbation
                    noise = np.random.normal(scale=0.01, size=n)
                    a_curr = np.maximum(a_curr + noise, 0.0)
                    a_curr /= a_curr.sum()
                    cur_val = _objective(a_curr)
                    step = 1.0
                    break

        # increase smoothing parameter slowly
        if gamma < gamma_max:
            gamma = min(gamma * gamma_factor, gamma_max)

        # occasional random perturbation to escape flat regions
        if iter_cnt % 5000 == 0 and iter_cnt > 0:
            noise = np.random.normal(scale=0.005, size=n)
            a_curr = np.maximum(a_curr + noise, 0.0)
            a_curr /= a_curr.sum()
            cur_val = _objective(a_curr)
            step = max(step, 0.1)
            print(f"[PGD {iter_cnt:6d}] added noise – val = {cur_val:.6f}")

        # stagnation handling
        if cur_val >= best_val * (1.0 - 1e-7):
            no_improve += 1
            if no_improve > IMPROVE_LIMIT:
                # restart from a fresh random Dirichlet
                a_curr = np.random.dirichlet(np.ones(n))
                cur_val = _objective(a_curr)
                step = 1.0
                no_improve = 0
                print(f"[PGD {iter_cnt:6d}] stagnation – random restart")
        else:
            no_improve = 0

        iter_cnt += 1
        if best_val <= 1.503:
            break

    print(f"[PGD] finished after {iter_cnt} iterations – best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climbing phase (uses the remaining time budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        print("[Hill] starting final hill‑climb")
        best_a, best_val = _hillclimb(best_a, deadline, start_time)
        print(f"[Hill] finished – best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final score check (for debugging – the evaluation function is
    # provided by the surrounding system)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
