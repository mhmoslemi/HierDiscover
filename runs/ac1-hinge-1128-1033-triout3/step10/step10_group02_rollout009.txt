def _adam_optimize2(a0, n, deadline, start_time, best_a, best_val):
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.9, 0.999
    lr_start, lr_end = 0.5, 0.005
    eps_start, eps_end = 0.5, 1e-12
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 2000
    while time.time() < deadline:
        # decode
        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam2] New best: {best_val:.6f}")
        else:
            idle += 1
        # schedule
        prog = (time.time() - start_time)/(deadline - start_time)
        lr = lr_start * (lr_end/lr_start) ** prog
        eps = eps_start * (eps_end/eps_start) ** prog
        # compute gradient
        _, grad = _smooth_max_and_grad(a, eps)
        # chain rule
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)
        # Adam update
        t += 1
        m = beta1*m + (1-beta1)*grad_w
        v = beta2*v + (1-beta2)*(grad_w*grad_w)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # occasional jitter
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.05, size=n)
        if idle > MAX_IDLE:
            print("[Adam2] restart due to idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0); v.fill(0.0); t=0; idle=0
    return best_a, best_val

"""
Improved step‑function optimiser
--------------------------------

The programme searches for a non‑negative vector `a` (length `n`) with
∑a = 1 that minimises

        2·n·max(conv(a,a))

which is exactly the value returned by ``evaluate_sequence``.
The optimiser proceeds in three stages:

1. **Diverse initialisation** – a large collection of Beta‑shaped
   candidates (including a few uniform / Dirichlet seeds) is examined
   and the best one is kept.

2. **Fine‑grained Adam optimisation** – we minimise a smooth
   approximation of the L∞‑norm of the convolution.
   The smoothing parameter ε is annealed from 0.5 to 1e‑12 and the
   learning‑rate from 0.5 down to 0.005.  The optimisation runs on the
   log‑softmax parameters of ``a`` and restarts automatically when it
   stalls.

3. **Exact sub‑gradient polishing** – a projected sub‑gradient method
   on the true objective (max‑convolution) is performed until the time
   budget expires.

The best vector found is finally evaluated with the secure
``evaluate_sequence`` function and returned as a plain list of floats.
All code is written without closures or lambdas and uses only NumPy and
the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights (log‑sum‑exp trick)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------

    The smoothing ε is annealed from 0.5 to 1e‑12 and the learning
    rate from 0.5 down to 0.005.
    """

    lr_start, lr_end = 0.5, 0.005

    v = np.zeros_like(w)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.05, size=n)

        # restart if stalled
        if idle > MAX_IDLE:
            print("[Adam]  restart after idle")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a)=2·n·max(conv(a,a))
    using the analytic sub‑gradient.  The step size decays linearly.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # sub‑gradient: sum of contributions from all maximal diagonals
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # tiny random jitter every few hundred iterations
        if it % 500 == 0 and np.random.rand() < 0.03:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate (used for a cheap initial sweep)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # roughly 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # problem size (larger n gives a slightly richer search space)
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – many cheap candidates, keep the best one
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # 0.1  Try to reuse a previously discovered solution (if present)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[Init] Interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # 0.2  Scan a few Beta shapes (grid + random)
    for _ in range(120):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] Beta candidate improved, value={best_val:.6f}")

    # 0.3  A few Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val
                print(f"[Init] Dirichlet(alpha={alpha}) improved, value={best_val:.6f}")

    # fallback if nothing succeeded (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print(f"[Init] fallback uniform, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Aggressive Adam optimisation (≈40 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.40 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – More focussed sub‑gradient polishing (≈55 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.95 * budget_s
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        max_iters=500_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Final fine‑tune (remaining seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _adam_optimize(
            best_a, n,
            deadline=deadline,
            start_time=phase_b_deadline,
            best_a=best_a,
            best_val=best_val,
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure a plain list of floats is returned
    return best_a.tolist()
