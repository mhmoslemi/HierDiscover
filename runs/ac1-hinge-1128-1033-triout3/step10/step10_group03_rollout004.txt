"""
Optimise a non‑negative step function by a two‑phase approach:

1️⃣  Initialise the solution by scanning the 2‑parameter Beta family
    (including a few Dirichlet draws).  The best candidate from this
    cheap search is used as a start point.

2️⃣  Refine the start point with a *smoothed* gradient descent.
    The max‑convolution term  max(a⋆a) is replaced by a
    log‑sum‑exp smoothing
        φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i) .
    Its gradient is the weighted cross‑correlation
        ∇φ_β(a)_j = 2·Σ_i w_i·a_{i−j},
    where w_i are the normalised soft‑max weights.
    We perform many projected gradient steps on the simplex
    (Σa=1, a≥0) while slowly increasing β so that the smoothed
    problem approaches the original max‑convolution objective.
    The step size decays as 1/√t and a tiny random jitter is added
    occasionally to escape flat plateaus.

The best sequence found during the search is finally evaluated with
``evaluate_sequence`` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters produce an all‑zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _gradient_smooth(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the smoothed objective
        φ_β(a) = (1/β)·log Σ_i exp(β·(a⋆a)_i)

    Returns (grad, max_conv) where grad is the Euclidean gradient
    (same shape as a) and max_conv is max(a⋆a) (useful for the exact
    objective value).
    """
    n = a.shape[0]

    # ----- 1. full convolution a⋆a (length 2n‑1) -----
    out_len = 2 * n - 1
    size_conv = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[:out_len]        # (a⋆a)

    max_conv = conv.max()

    # ----- 2. soft‑max weights w_i -----
    # shift by max to avoid overflow
    exp_arg = beta * (conv - max_conv)            # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    w_norm = w / w_sum                            # Σ_i w_i = 1

    # ----- 3. gradient = 2·(a ⋆_cross w_norm) -----
    # Cross‑correlation (a ⋆ w_norm)[j] = Σ_k a_k·w_norm_{k+j}
    rev_w = w_norm[::-1]                          # reverse for convolution
    out_len_corr = n + out_len - 1                # = 3n‑2
    size_corr = 1 << (out_len_corr - 1).bit_length()
    Fa = np.fft.rfft(a, size_corr)
    Fw = np.fft.rfft(rev_w, size_corr)
    corr_full = np.fft.irfft(Fa * Fw, size_corr)[:out_len_corr]

    offset = out_len - 1                          # position of lag 0 in conv_full
    idx = offset - np.arange(n)                   # indices for j = 0…n‑1
    grad = 2.0 * corr_full[idx]

    return grad, max_conv


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered before
    the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # problem size – a compromise between expressivity and speed
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previous solution when possible
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = None
        else:
            a0 /= a0.sum()
            print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse search in the Beta family (grid + random)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    # dense deterministic grid
    p_grid = np.concatenate(
        [
            np.geomspace(0.001, 0.1, 8, endpoint=False),
            np.geomspace(0.1, 5.0, 12, endpoint=False),
            np.geomspace(5.0, 20.0, 5)
        ]
    )
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is None:
                continue
            val = _objective(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[grid] new best = {best_val:.6f} (p={p:.4g}, q={q:.4g})")

    # log‑uniform random sampling in (p,q)-space
    rng = np.random.default_rng()
    random_trials = 8000
    log_min, log_max = np.log(1e-3), np.log(20.0)
    for _ in range(random_trials):
        p = np.exp(rng.uniform(log_min, log_max))
        q = np.exp(rng.uniform(log_min, log_max))
        cand = _beta_shape(p, q, n)
        if cand is None:
            continue
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand] new best = {best_val:.6f} (p={p:.4g}, q={q:.4g})")
        if time.time() >= deadline - 30.0:   # keep ~30 s for refinement
            break

    # a few Dirichlet draws for extra diversity
    for _ in range(6):
        cand = rng.dirichlet(np.ones(n))
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[dir] new best = {best_val:.6f}")

    # fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] uniform start, value=", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed gradient descent on the exact objective
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_seq = a.copy()
    best_score = best_val

    # schedule parameters
    step0 = 0.25                      # initial step size (will decay as 1/√t)
    beta = 0.5                        # start of smoothing
    beta_max = 200.0                  # final β – essentially the true max
    beta_mul = 1.001                  # multiplicative increase per iteration
    jitter_prob = 0.001
    jitter_scale = 0.001

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # step size decays like 1/√t
        step = step0 / np.sqrt(iteration + 1.0)

        # gradient of the smoothed objective and current max‑convolution
        grad, cur_max = _gradient_smooth(a, beta)
        cur_val = 2.0 * n * cur_max          # exact objective for this a

        if cur_val < best_score:
            best_score, best_seq = cur_val, a.copy()
            print(f"[smooth] new best = {best_score:.6f} (iter {iteration}, β={beta:.2f})")
            # early exit if we have already crossed the target
            if best_score <= 1.503:
                break

        # projected gradient step
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        # occasional tiny random jitter to shake off flat regions
        if np.random.rand() < jitter_prob:
            a_candidate += np.random.normal(scale=jitter_scale, size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate

        # increase smoothing parameter slowly
        beta = min(beta_max, beta * beta_mul)

        iteration += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
