"""
A light‑weight multi‑phase optimiser for the step‑function “attack”.

The code performs the following stages within the 1000 s budget:

0️⃣  Initialise from any previously found solution, a few random Beta
    families, and a handful of Dirichlet draws.

1️⃣  Smooth‑max Adam optimisation – the loss 2·n·max(conv(a,a)) is
    approximated by a log‑sum‑exp (soft‑max) and the parameters are
    updated with Adam on the simplex (via a log‑softmax parametrisation).
    No restarts are performed, the learning‑rate and smoothing ε are
    annealed continuously.

2️⃣  Simulated‑annealing hill‑climber – tiny amounts of mass are moved
    from “high‑contribution’’ indices to “low‑contribution’’ ones.
    The schedule for the maximal transfer fraction and the temperature
    are much more aggressive than in the reference implementation.

3️⃣  Exact sub‑gradient polishing – a projected sub‑gradient step on the
    true objective 2·n·max(conv(a,a)) with a linearly decaying step size.
    Momentum is not used here; the step schedule starts larger and ends
    near 1e‑6.

4️⃣  Optional blending with the uniform distribution to smooth residual
    spikes.

The best sequence encountered (as a plain list of floats) is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # soft‑max weights

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n

    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (no idle restarts)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex for the loss L(a)=2·n·smooth_max(a).
    The smoothing ε and the learning‑rate are annealed continuously.
    """
    w = np.log(a0 + 1e-12)          # log‑softmax parameters

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.001
    eps_start, eps_end = 0.5, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector (softmax)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[Adam] new best = {best_val:.6f}")

        # schedule (linear in progress)
        prog = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog

        # smooth‑max and gradient
        _, grad = _smooth_max_and_grad(a, eps)   # already includes 2·n

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional tiny jitter to avoid pathological plateaus
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.01, size=n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (more aggressive schedule)
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Fast local search moving a small chunk of mass from a highly‑contributing
    index to a low‑contributing one.  The transfer fraction starts at 0.9
    and the temperature decays exponentially.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.9, 0.001
    temp_start, temp_end = 0.05, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 50000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # choose source proportional to its contribution
        if total > 0.0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target biased towards low contribution indices (inverse weighting)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback: pick any other index uniformly
            j = int(np.random.rand() * (n - 1))
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling – perturb and restart
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] stagnation – perturb (it={iteration})")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a) = 2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        # analytic sub‑gradient (sum of shifted copies of a)
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        prog = (now - start_time) / (deadline - start_time)
        step = 0.5 * (1.0 - prog) + 1e-6 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape flat plateaus
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray, n: int, cur_best_val: float):
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta family candidate
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ln(1e‑4) .. ln(10)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final call

    # ------------------------------------------------------------------
    # problem size (larger n gives more degrees of freedom)
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise (reuse old solution, random Beta, Dirichlet)
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf

    old = globals().get("height_sequence_1", None)   # type: ignore
    if old is not None:
        try:
            old = np.asarray(old, dtype=float)
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            val = _objective(a0)
            best_a, best_val = a0.copy(), val
            print(f"[Init] interpolated previous solution, value={best_val:.6f}")
        except Exception:
            pass

    # a few random Beta draws
    for _ in range(100):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] random Beta improved value={best_val:.6f}")

    # several Dirichlet draws with varied concentration
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(5):
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val
                print(f"[Init] Dirichlet(alpha={alpha}) improved value={best_val:.6f}")

    # fallback uniform if everything failed
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print(f"[Init] fallback uniform, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – split the remaining time into three phases
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break
        remaining = deadline - now

        # --------------------------------------------------------------
        # Phase 1 – smooth‑max Adam (≈30 % of remaining time)
        # --------------------------------------------------------------
        phase_deadline = now + 0.30 * remaining
        best_a, best_val = _adam_optimize(
            best_a, n, phase_deadline, now, best_a, best_val
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

        # --------------------------------------------------------------
        # Phase 2 – aggressive hill‑climbing (≈35 % of remaining time)
        # --------------------------------------------------------------
        now = time.time()
        if now >= deadline:
            break
        remaining = deadline - now
        phase_deadline = now + 0.35 * remaining
        best_a, best_val = _hillclimb(
            best_a, n, phase_deadline, now, best_a, best_val
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

        # --------------------------------------------------------------
        # Phase 3 – exact sub‑gradient polishing (remaining time)
        # --------------------------------------------------------------
        now = time.time()
        if now >= deadline:
            break
        best_a, best_val = _subgradient_polish(
            best_a, n, deadline, now, best_a, best_val, max_iters=500_000
        )
        best_a, best_val = _blend_with_uniform(best_a, n, best_val)

        # --------------------------------------------------------------
        # Small stochastic perturbation to potentially escape a basin
        # --------------------------------------------------------------
        if (deadline - time.time()) > 10.0:
            noise = rng.normal(scale=0.01, size=n)
            cand = np.maximum(best_a + noise, 0.0)
            cand /= cand.sum()
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_a, best_val = cand, cand_val
                print(f"[Perturb] improved value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
