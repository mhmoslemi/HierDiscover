def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum x = 1}."""
    # Based on algorithm by Wang & Carreira-Perpinan
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def gradient_logsumexp(a, b, tau):
    # b = full convolution a*a
    # compute w = softmax(b/tau)
    maxb = np.max(b)
    expb = np.exp((b - maxb) / tau)
    w = expb / expb.sum()
    # compute gradient g = 2 * correlation(w, a)
    # Use reversed w and convolution:
    w_rev = w[::-1]
    # conv_len = len(a) + len(w_rev) - 1 = n + (2n-1) - 1 = 3n - 2
    # Perform convolution via FFT
    size = 1 << ((len(a) + len(w_rev) - 1 - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:(len(a) + len(w_rev) - 1)]
    # g_i = 2 * conv[n-1 : 2*n-1][::-1]
    start = len(a) - 1
    end = start + len(a)
    g = 2.0 * conv[start:end][::-1]
    return g
```

def simplex_projection(v):
    # v: numpy array
    # reference: https://doi.org/10.1109/ICMLA.2014.71
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    # v is 1D numpy array.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):

        return np.full(n, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)

        step = 0.5
        for _ in range(200):
            if time.time() >= deadline:
                break
            # compute conv
            b = _full_convolution_fft(a)
            maxb = b.max()
            # compute softmax weights
            # shift for numeric stability
            shifted = (b - maxb) / tau
            w = np.exp(shifted)
            w_sum = w.sum()
            w /= w_sum
            # compute gradient: 2 * correlation(w, a)
            # using FFT: convolution of a and reversed w
            w_rev = w[::-1]
            size = 1 << ((len(a) + len(w_rev) - 1 - 1).bit_length())
            fa = np.fft.rfft(a, size)
            fw = np.fft.rfft(w_rev, size)
            conv = np.fft.irfft(fa * fw, size)[:len(a) + len(w_rev) - 1]
            start = n - 1
            grad = 2.0 * conv[start:start+n][::-1]   # length n
            # gradient descent step
            a_new = a - step * grad
            # project onto simplex
            a_new = _project_simplex(a_new)
            # evaluate
            val = _objective_value(a_new)
            # accept if improvement, else maybe reduce step
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
                # optionally increase step a little
                step = min(step * 1.05, 1.0)
            else:
                step *= 0.7
            a = a_new
        # reduce temperature
        tau *= 0.7
    return best_a, best_val

"""
Propose a non‑negative step‑function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

under the 1000 s budget.

Algorithm
---------
1. **Problem size** – use n = 15000 (2·n‑1 ≤ 32768, so a single FFT of
   size 32768 suffices for all convolutions).

2. **Initialisation** – reuse an existing `height_sequence_1` if present,
   otherwise pick the best vector from a cheap parametric sweep
   (exponential, power‑law, cosine‑window, Dirichlet draws).

3. **Phase 1 – Exponentiated‑gradient descent** – mirror‑descent on the
   probability simplex using the exact sub‑gradient of the maximal
   convolution entry (as in the reference implementation).

4. **Phase 2 – Smoothed gradient descent** – minimise a
   log‑sum‑exp surrogate of the max‑convolution:
       fτ(a)=τ·log∑ exp((a∗a)/τ).
   The gradient is a correlation of the softmax weights with the current
   vector and is obtained by a single FFT‑based convolution.
   τ is decreased geometrically (0.5 → 1e‑5) while a projected gradient
   step with a simple Euclidean projection onto the simplex is applied.

5. **Phase 3 – Random local hill‑climbing** – repeatedly move a tiny
   fraction of mass from a random source index to a random target index
   and keep the move if the true objective improves.

6. **Phase 4 – Tiny uniform mixing** – try convex combinations with a
   small uniform component (α∈{0.001,…,0.020}) and keep the best.

7. **Final evaluation** – return the best vector found (as a plain
   ``list[float]``).  The harness will call ``evaluate_sequence`` on it.

All heavy work (convolutions, gradients) is performed with FFTs of
fixed length 32768, making each iteration cheap enough to explore many
candidate vectors within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolution (a*a)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    L = 2 * n - 1
    size = 1 << ((L - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a probability vector (sum == 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simplex mixing with a tiny uniform component
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for several α and keep the best."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)   # keep sum = 1
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Parameter‑family generator (used when no previous solution is present)
# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, rng: np.random.Generator):
    """Yield vectors from a few simple families."""
    i = np.arange(n, dtype=np.float64)

    # 1. Exponential U‑shape: a_i ∝ r**i + r**(n‑1‑i)
    for r in np.linspace(0.55, 0.9999, 30):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 2. Power‑law (edge‑peaked) family
    for p in np.linspace(-4.0, 4.0, 30):
        idx = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
        base = idx * (n + 1 - idx) + 1e-12
        a = np.power(base, -p)
        yield a

    # 3. Cosine‑window family
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    for beta in np.linspace(0.5, 4.0, 20):
        a = np.power(w, beta)
        yield a

    # 4. Random Dirichlet draws
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        yield a


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror) sub‑gradient descent
# ----------------------------------------------------------------------
def _eg_subgradient_descent(
    a0: np.ndarray,
    n: int,
    start: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Mirror descent on the simplex using the true max‑convolution sub‑gradient."""
    eps = 1e-12
    log_w = np.log(a0 + eps)                 # initialise log‑weights
    eta = 0.5                                 # initial step size
    decay = 0.99995                           # geometric decay
    best_a = a0.copy()
    best_val = _objective_value(best_a)

    no_improve = 0
    IMPROVE_LIMIT = 30000
    it = 0

    while time.time() < deadline:
        # compute current probability vector
        max_log = np.max(log_w)
        w = np.exp(log_w - max_log)          # stabilised exponentiation
        a = w / w.sum()

        # evaluate objective
        b = _full_convolution_fft(a)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[EG  {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of the maximal entry
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n, dtype=int)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # multiplicative update on the simplex
        log_w -= eta * grad
        eta *= decay

        # occasional jitter
        if rng.random() < 0.003:
            log_w += rng.normal(scale=0.005, size=n)

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            log_w = np.log(best_a + eps)
            log_w += rng.normal(scale=0.02, size=n)
            eta = 0.5
            no_improve = 0
            print(f"[EG  {it:7d}] perturb after stagnation")

        it += 1
        if time.time() > deadline - 0.2:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on τ·log‑sum‑exp(conv) with decreasing τ."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_value(best_a)

    tau = 0.5                     # start with a relatively smooth surrogate
    while time.time() < deadline and tau > 1e-5:
        step = 0.5                # initial step size for this temperature
        # Inner loop – a few gradient steps at the current τ
        for _ in range(200):
            if time.time() >= deadline:
                break

            b = _full_convolution_fft(a)
            maxb = b.max()
            # soft‑max weights for the surrogate
            shifted = (b - maxb) / tau
            w = np.exp(shifted)
            w /= w.sum()

            # gradient = 2 * correlation(w, a)
            w_rev = w[::-1]
            size = 1 << ((n + len(w_rev) - 2).bit_length())
            fa = np.fft.rfft(a, size)
            fw = np.fft.rfft(w_rev, size)
            conv = np.fft.irfft(fa * fw, size)[: n + len(w_rev) - 1]
            start = n - 1
            grad = 2.0 * conv[start:start + n][::-1]

            # descent step with Euclidean projection onto the simplex
            a_new = a - step * grad
            a_new = _project_simplex(a_new)

            val = _objective_value(a_new)
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
                step = min(step * 1.05, 1.0)   # slight step‑size increase on success
            else:
                step *= 0.70                  # otherwise shrink

            a = a_new

        tau *= 0.70                 # cool down – focus more on the true max
    return best_a, best_val


# ----------------------------------------------------------------------
# Random local hill climbing (tiny mass transfers)
# ----------------------------------------------------------------------
def _random_hill_climb(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Move a tiny fraction of mass between random positions and keep improvements."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_value(best_a)

    while time.time() < deadline:
        i = rng.integers(n)
        if a[i] < 1e-8:
            continue
        # choose a different target index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # transfer a small fraction of the source mass
        frac = rng.uniform(1e-4, 1e-2)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective_value(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            a = a_new
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list of floats (the heights) that approximately minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the final external evaluation
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – we can afford n = 15000 (FFT size stays 32768)
    # --------------------------------------------------------------
    n = 15000

    # --------------------------------------------------------------
    # Initialise a – reuse a previous construction if available,
    # otherwise fall back to a cheap parametric sweep.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.clip(a0, 0.0, None)
        a0 /= a0.sum()
        print("[init] Re‑used height_sequence_1 (interpolated).")
    except Exception:
        best_val = np.inf
        best_a = None
        for cand in _generate_parametric_candidates(n, rng):
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s <= 0.0:
                continue
            cand = cand / s
            val = _objective_value(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
        if best_a is None:
            best_a = np.full(n, 1.0 / n)
            best_val = _objective_value(best_a)
        a0 = best_a
        print(f"[init] Parametric sweep gave initial objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Exponentiated‑gradient descent (mirrored simplex)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * (deadline - start_time)
    a_eg, val_eg = _eg_subgradient_descent(a0, n, start_time, phase1_deadline, rng)
    print(f"[phase1] EG best objective = {val_eg:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Smoothed log‑sum‑exp gradient descent
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.80 * (deadline - start_time)
    a_smooth, val_smooth = _smooth_gradient_descent(a_eg, n, phase2_deadline, rng)
    print(f"[phase2] Smooth descent best objective = {val_smooth:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Local random hill climbing
    # --------------------------------------------------------------
    a_local, val_local = _random_hill_climb(a_smooth, n, deadline, rng)
    print(f"[phase3] Hill‑climb best objective = {val_local:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Tiny uniform mixing (often gives a final tweak)
    # --------------------------------------------------------------
    a_mix, val_mix = _mix_with_uniform(a_local, n)
    print(f"[mix] Uniform mixing best objective = {val_mix:.6f}")

    # --------------------------------------------------------------
    # Final external evaluation (the harness will also call it)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_mix.tolist())
        print(f"[final] evaluate_sequence returned {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_mix.tolist()
