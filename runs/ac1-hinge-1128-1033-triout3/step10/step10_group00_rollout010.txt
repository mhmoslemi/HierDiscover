"""
Non‑negative step‑function optimiser (advanced).

The routine starts from a good seed (previous solution, a Beta‑family
candidate or a Dirichlet draw) and then performs a continuous
entropy‑mirror descent on the probability simplex.  At each iteration
the objective  2·n·max(conv(a,a))  is approximated by a smooth
log‑sum‑exp of the convolution entries; the resulting weighted gradient
is computed via an FFT‑based correlation (exact and O(n log n)).  A
time‑dependent stepsize, temperature and soft‑max “τ’’ schedule drive the
algorithm: early iterations move aggressively with a smooth objective,
later iterations concentrate on the true maxima.  If progress stalls a
small Gaussian perturbation of the current best solution is applied.
The best sequence found before the time budget expires is finally
checked with `evaluate_sequence` and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two (used for FFT padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility returning the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Full linear convolution a*a (length 2·n‑1) via FFT
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Correlation w ⋆ a (shifted sum Σ_i w_i a_{i-j})  – length 3·n‑2
# Returned array is of length 3·n‑2, the first 3·n‑2 values of the
# linear correlation.
# ----------------------------------------------------------------------
def _correlation_w_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute correlation(w, a) = Σ_i w_i a_{i+j} for all integer shifts j.
    Implemented via FFT: correlation = conv(w, reverse(a)).
    The returned array length is (len(w) + len(a) - 1).
    """
    size = _next_pow2(len(w) + len(a) - 1)
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a[::-1], size)   # reverse a
    corr = np.fft.irfft(fw * fa_rev, size)[:len(w) + len(a) - 1]
    return corr


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised vector (sum == 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a)) for a probability vector (sum(a)=1).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Entropy (multiplicative) mirror‑descent update
# ----------------------------------------------------------------------
def _entropy_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Perform a single EG step a ← a·exp(-η·grad) and renormalise."""
    a_new = a * np.exp(-eta * grad)
    s = a_new.sum()
    if s <= 0.0:                         # safety fallback
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s
    return a_new

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------

                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  Uses a fast FFT‑based mirror descent
    with a log‑sum‑exp smoothing of the max‑convolution.
    """

    start_time = time.time()
    # reserve ~0.4 s for the final evaluation call
    deadline = start_time + budget_s - 0.4

    # --------------------------------------------------------------
    # Problem size – 8000 works well with the time budget
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Initialise a (use previous solution if available)
    # --------------------------------------------------------------
    best_a = None

        old_len = old.shape[0]

        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        cur = _objective(a0)
        best_a, best_val = a0, cur
        print(f"[init] interpolated previous solution – val={best_val:.6f}")
    except Exception:

    # --------------------------------------------------------------

    # --------------------------------------------------------------

        conv_cand = np.fft.irfft(fa_cand * fa_cand, size_conv)[:2 * n - 1]
        cand_val = float(2.0 * n * np.max(conv_cand))

        # ----- Metropolis acceptance -----
        accept = False
        if cand_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(cand_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_candidate
            cur_val = cand_val
            # (best update already handled at loop top)

        if stagnation_cnt > IMPROVE_LIMIT:

            else:
                a /= a_sum

            # recompute val for logging
