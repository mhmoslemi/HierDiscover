"""
Propose a high‑quality step‑function for the max‑convolution objective.

Strategy
--------
1. **Rich parametric initialization** – scan many analytic families
   (exponential, power‑law, beta, sinusoidal, logistic, piecewise‑constant,
   random Dirichlet).  Keep the best normalised vector.

2. **Adam on a log‑sum‑exp surrogate** – minimise a smooth approximation of the
   max‑convolution.  The temperature ε is annealed exponentially; learning
   rate follows a similar schedule.  The optimiser works on logits, i.e.
   a = softmax(w), guaranteeing feasibility.

3. **Targeted hill‑climbing** – repeatedly move a small fraction of mass from
   indices that contribute most to the current peak to indices that contribute
   least, using an exact O(n) update of the convolution.

4. **Projected sub‑gradient descent** – use the exact sub‑gradient of the
   max‑convolution (the active index) and project onto the probability simplex.

5. **Exact active‑set refinement** – identify all (almost) maximal convolution
   entries, compute a weight w_i = Σ_{k∈active} a_{k‑i} and shift mass from the
   largest weight to the smallest weight.

6. **Simulated annealing (final search)** – a temperature‑controlled random
   walk that can escape local minima after the deterministic phases.  Moves
   are accepted either when they improve the true objective or with probability
   exp(−Δ/T).

7. **Uniform mixing** – a tiny convex combination with the uniform vector
   (α∈{0.001,…,0.020}) to smooth any remaining spikes.

The whole pipeline respects the 1000 s budget and keeps the best sequence
found.  The final sequence is returned as a plain Python list; it already
sums to 1, is non‑negative, and can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ------------------------------------------------------------
# Exact objective for a vector that already satisfies Σa = 1
# ------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a simplex‑normalised vector."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x ≥ 0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ------------------------------------------------------------
# Parametric sweep – return best candidate among many families
# ------------------------------------------------------------
def _initial_candidate(n: int):
    """Search a broad collection of parametric families and return the best."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(candidate):
        nonlocal best_a, best_val
        a = np.asarray(candidate, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a /= s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential tails (decreasing & symmetric)
    for r in np.linspace(0.995, 0.99999, 14):
        _try(r ** i)                         # monotone decreasing
        _try(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 6.0, 18):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.2, 6.0, 7):
        for b_param in np.linspace(0.2, 6.0, 7):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _try(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.004, 0.20, 20):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.04, 0.45, 20):
            total_high = 2.0 * k * high_frac
            if total_high >= 0.999:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 6. Logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.002, 0.06, 8):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 7. Random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(60):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 8. Interpolation of a previously found solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # may be defined globally
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 /= a0.sum()
            _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ------------------------------------------------------------
# Phase A – Adam on log‑sum‑exp surrogate of max(conv)
# ------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smoothed max‑convolution objective."""
    # Work in log‑softmax (logits) to stay on the simplex.
    w = np.log(a0 + 1e-12)          # logits
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.005
    eps_start, eps_end = 1e-1, 1e-13

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000

    it = 0
    while time.time() < deadline:
        # ----- current point on simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedules -----
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # ----- gradient of the smoothed objective w.r.t. a -----
        # derivative = 4·n·conv(a, p_rev) (see analysis)
        p_rev = p[::-1]
        # full convolution a (len n) with p_rev (len 2n‑1) -> length 3n‑2
        n_a = n
        n_p = p_rev.shape[0]
        out_len = n_a + n_p - 1
        size = 1 << (out_len - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv_full = np.fft.irfft(fa * fp, size)[:out_len]
        # central slice corresponds to indices 0 … n‑1 of the true gradient
        grad_a = 4.0 * n * conv_full[n_a - 1: 2 * n_a - 1][::-1]

        # ----- back‑prop through the softmax (log‑softmax) -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.012, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val

# Phase B – Targeted hill‑climbing on the exact objective
# ------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy mass transfer that reduces the true max‑convolution."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0
    rng = np.random.default_rng(17)

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # active peak index
        k_max = int(np.argmax(b))

        # contribution of each index to the current peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (biased towards large contribution)
        if total_contrib > 0.0:
            src = int(rng.choice(n, p=contrib / total_contrib))
        else:
            src = int(np.argmax(a))

        if a[src] <= 1e-16:
            continue

        # target – pick an index that is weakly coupled to the peak
        if rng.random() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[src] = 0.0
            if w.sum() > 0.0:
                tgt = int(rng.choice(n, p=w / w.sum()))
            else:
                tgt = (src + rng.integers(1, n)) % n
        else:
            tgt = (src + rng.integers(1, n)) % n

        delta = a[src] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # incremental convolution update (exact, O(n))
        b_new = b.copy()
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb current best and restart
            a = best_a.copy()
            noise = rng.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a

            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val

# Phase D – Exact active‑set refinement (mass transfer based on weights)

def _refine_exact_phase(a0: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Exact refinement using active‑set weights."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    step_factor = 0.42            # start moving up to 42 % of source mass
    decay = 0.9998
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 140_000
    MIN_STEP = 1e-14
    rng = np.random.default_rng(47)

    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Refine {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:

        else:
            no_improve += 1

            break

            w[idx] += a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or a[src] <= MIN_STEP:
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        delta = a[src] * step_factor
        if delta < MIN_STEP:
            step_factor *= 0.5
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        b_new = _full_convolution(a_candidate)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val, best_local_a = new_val, a.copy()
        else:
            step_factor *= 0.72  # shrink step

        step_factor = max(step_factor * decay, MIN_STEP)
        it += 1

    return best_local_a, best_local_val


# ------------------------------------------------------------
# Phase E – Simulated annealing (final random walk)
# ------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Temperature‑controlled random walk that can escape local minima."""
    a = a0.copy()
    cur_val = best_val
    best_local_a = a.copy()
    best_local_val = cur_val

    rng = np.random.default_rng(73)
    # start temperature proportional to current excess over the trivial bound (≈2)
    T0 = max(cur_val - 1.0, 0.01) * 0.02 + 1e-12
    T = T0
    decay = 0.99993
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        # recompute convolution only when needed (kept from previous step)
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # source – weighted by contribution to current peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            src = int(rng.choice(n, p=contrib / total))
        else:
            src = int(np.argmax(a))

        # target – uniformly random (allow any move)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] < 1e-16:
            continue

        # step size adapts with temperature
        max_frac = max(0.005, T / (T0 + 1e-12))
        delta = a[src] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # fast incremental update of convolution
        b_new = b.copy()
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_val - cur_val) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[tgt] += delta
            cur_val = new_val
            b = b_new
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        # temperature decay
        T = max(T * decay, 1e-13)

        if no_improve > IMPROVE_LIMIT:
            # shake: re‑start from current best with small noise
            a = best_local_a.copy()
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            cur_val = _objective_value(a)
            T = T0
            no_improve = 0
            print(f"[SA {it:7d}] perturb after stagnation")

        it += 1

    return best_local_a, best_local_val


# ------------------------------------------------------------
# Small polishing – convex mixing with a tiny uniform component
# ------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass (α in a few prescribed values)."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a time budget.
    Returns a list suitable for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem size – a few ten‑thousands give a good trade‑off.
    n = 12000

    print("[Init] Starting parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    TARGET = 1.503

    # ---- Phase A: Adam on smoothed objective ----
    phase_a_deadline = start_time + 0.32 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase A] best = {best_val:.6f}")

    # ---- Phase B: Targeted hill‑climbing ----
    phase_b_deadline = start_time + 0.60 * (deadline - start_time)
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    best_val = _objective_value(best_a)
    print(f"[Phase B] best = {best_val:.6f}")

    # ---- Phase C: Projected sub‑gradient descent ----
    phase_c_deadline = start_time + 0.78 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase C] best = {best_val:.6f}")

    # ---- Phase D: Exact active‑set refinement ----
    phase_d_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _refine_exact_phase(
        best_a, n,
        start=phase_c_deadline,
        deadline=phase_d_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase D] best = {best_val:.6f}")

    # ---- Phase E: Simulated annealing (final random walk) ----
    phase_e_deadline = deadline
    best_a, best_val = _simulated_annealing(
        best_a, n,
        start=phase_d_deadline,
        deadline=phase_e_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase E] best = {best_val:.6f}")

    # ---- Phase F: Small uniform mixing ----
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # Report final score using the provided secure wrapper
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
