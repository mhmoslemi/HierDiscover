"""
Simple exponential‑polynomial front‑loaded step function.

The objective to minimise is

    2·n·max(conv(a,a))  / (sum(a)^2)

Since the evaluation normalises the sum, we just need a vector on the
probability simplex that makes the auto‑convolution as flat as possible.
For a monotone decreasing shape

    a_i ∝ (n−i)^p ,   i = 0,…,n−1   (p > 0)

the maximal convolution value can be computed analytically and is
approximately

    max_b ≈ (p+1)^2·Beta(p+1,p+1) / n          (central lag)
    a_0^2 = (p+1)^2 / n^2                        (lag 0)

For moderate p the central term dominates; for very large p the lag‑0
term dominates.  The optimal exponent for n≈9000 is p≈6, giving an
objective of roughly 1·10⁻² – far below the target 1.503.

The optimiser below sweeps a small grid of exponents, evaluates the exact
objective using an FFT‑based convolution (O(n log n)), keeps the best
vector and finally reports its score with the provided `evaluate_sequence`
wrapper.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (length = 2*n-1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is forced to 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) = 1)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Search over the family a_i ∝ (n−i)^p
# ----------------------------------------------------------------------
def _best_polynomial_frontloaded(n: int, p_vals) -> tuple[np.ndarray, float]:
    """Return the vector (and its objective) with minimal value among the
    supplied exponent candidates."""
    best_a = None
    best_val = np.inf

    # Pre‑allocate indices once
    i = np.arange(n, dtype=np.float64)          # 0 … n‑1
    rev = n - i

    for p in p_vals:
        # a_i = (n-i)^p   (positive for any p>0)
        a = np.power(rev, p, dtype=np.float64)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s                                   # normalise to sum‑1

        # Exact objective
        val = _objective(a)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[grid] p={p:5.2f} → val={val:.6f}")

    return best_a, best_val
