def eg_multi_active(a0, n, deadline, rng):
    eps = 1e-12
    log_w = np.log(a0 + eps)
    eta = 0.5
    decay = 0.99996
    best_val = objective(a0)
    best_a = a0.copy()
    no_improve = 0
    IMPROVE_LIMIT = 10000
    it = 0
    while time.time() < deadline:
        max_log = np.max(log_w)
        w = np.exp(log_w - max_log)
        a = w / w.sum()
        b = full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[EG {it:6d}] new best={best_val:.6f}")
        else:
            no_improve += 1
        max_b = b.max()
        active = np.where(b >= max_b * (1 - 1e-9))[0]  # relative tolerance
        grad = np.zeros_like(a)
        for k in active:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        grad /= len(active)  # average

"""
Optimise a non‑negative step‑function for the objective

    2·n·max(conv(a,a)) / (sum(a)^2)

by a three‑phase approach:

1.  Fast random search in a handful of expressive parametric families
    (Beta, exponential, Gaussian, power‑law, cosine‑window, Dirichlet).
    The best candidate becomes the initial point.

2.  Mirror‑descent on the probability simplex (exponentiated‑gradient).
    The sub‑gradient is taken as the *average* of all convolution
    entries that are (almost) maximal – this better respects the whole
    active set than the single‑max variant used previously.
    Simple heuristics (step‑size decay, occasional jitter, stagnation
    perturbations) are employed and the loop runs until the time budget
    expires.

3.  A tiny convex mix with the uniform distribution often improves the
    final bound.

All heavy work (convolutions) is performed by FFT, which makes each
iteration cheap enough to afford many thousands of steps within the
1000 s budget.  The routine respects the security checks of
``evaluate_sequence`` – the returned list sums to 1, contains only
non‑negative floats and does not rely on any external resources.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution a*a using FFT.
    The result has length 2*len(a)-1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """
    Objective for a probability vector (sum(a)=1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple uniform mixing -------------------------------------------------
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for a few tiny α and keep the best."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families ---------------------------------------------------
# ----------------------------------------------------------------------
def _sample_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    # a_i ∝ x_i^{α-1} (1-x_i)^{β-1}
    α = 10**rng.uniform(-1.0, 1.0)  # 0.1 … 10
    β = 10**rng.uniform(-1.0, 1.0)
    x = np.linspace(0.0, 1.0, n)
    a = np.power(x, α - 1.0) * np.power(1.0 - x, β - 1.0)
    return a


def _sample_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    # a_i ∝ r**i   (monotonically decreasing)
    r = rng.uniform(0.85, 0.9999)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(i * np.log(r))
    return a


def _sample_mixed_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    # a_i ∝ r**i + c·r**(n‑1‑i)
    r = rng.uniform(0.85, 0.9999)
    c = rng.uniform(0.1, 2.0)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(i * np.log(r)) + c * np.exp((n - 1 - i) * np.log(r))
    return a


def _sample_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    # a_i ∝ exp(- (i-μ)^2 / (2σ^2) )
    mu = rng.uniform(0.0, n - 1)
    sigma = rng.uniform(0.01 * n, 0.5 * n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a


def _sample_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    # a_i ∝ (i+1)^{-p}
    p = rng.uniform(-3.0, 3.0)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    return a


def _sample_cosine_window(n: int, rng: np.random.Generator) -> np.ndarray:
    # Hann window raised to a power β
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    beta = rng.uniform(0.5, 4.0)
    a = np.power(w, beta)
    return a


def _sample_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (multi‑active sub‑gradient) ----------
# ----------------------------------------------------------------------
def _eg_multi_active(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the average gradient of all
    (almost) maximal convolution entries.  The routine stops when the
    time budget runs out and returns the best vector seen.
    """
    eps = 1e-12
    log_w = np.log(a0 + eps)                    # initialise log‑weights
    eta = 0.5                                    # initial step size
    decay = 0.99996                              # geometric decay
    best_a = a0.copy()
    best_val = _objective_value(best_a)

    no_improve = 0
    IMPROVE_LIMIT = 15000
    it = 0

    while time.time() < deadline:
        # ---- current probability vector ----
        max_log = np.max(log_w)
        w = np.exp(log_w - max_log)            # stabilised exponentiation
        a = w / w.sum()

        # ---- evaluate objective ----
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[EG  {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---- active set of near‑max convolution entries ----
        max_b = b.max()
        # tolerance chosen very tight to keep the true active set
        active = np.where(b >= max_b * (1.0 - 1e-9))[0]

        grad = np.zeros_like(a)
        for k in active:
            idx = k - np.arange(n, dtype=int)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        # average over the active constraints
        grad /= max(1, len(active))

        # ---- multiplicative update (mirror descent) ----
        log_w -= eta * grad
        eta *= decay

        # ---- occasional random jitter to escape plateaus ----
        if rng.random() < 0.005:
            log_w += rng.normal(scale=0.003, size=n)

        # ---- stagnation handling ----
        if no_improve > IMPROVE_LIMIT:
            log_w = np.log(best_a + eps)
            log_w += rng.normal(scale=0.02, size=n)
            eta = 0.5
            no_improve = 0
            print(f"[EG  {it:7d}] perturb after stagnation")

        it += 1
        # safety margin for final evaluate_sequence call
        if time.time() > deadline - 0.2:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm mixes a cheap random parametric sweep with an
    exponentiated‑gradient refinement and a final uniform mixing.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size: 15 000 gives a nice trade‑off between expressiveness
    # and per‑iteration cost (FFT size 32768 → <1 ms per convolution).
    # ------------------------------------------------------------------
    n = 15000

    # ------------------------------------------------------------------
    # 1️⃣  Initialise – reuse previous construction if available,
    # otherwise run a quick random parametric sweep.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)          # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] Re‑used height_sequence_1")
    except Exception:
        # ---- random parametric sweep (≈20 % of the total budget) ----
        param_deadline = start_time + 0.20 * (deadline - start_time)
        best_a = None
        best_val = np.inf

        families = [
            _sample_beta,
            _sample_exponential,
            _sample_mixed_exponential,
            _sample_gaussian,
            _sample_powerlaw,
            _sample_cosine_window,
            _sample_dirichlet,
        ]

        while time.time() < param_deadline:
            sam = rng.choice(families)
            cand = sam(n, rng)
            # clip, normalise, and discard pathological draws
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s <= 0.0:
                continue
            cand /= s
            val = _objective_value(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()

        # fallback to uniform if, for any reason, nothing was generated
        if best_a is None:
            best_a = np.full(n, 1.0 / n)
            best_val = _objective_value(best_a)
        a0 = best_a
        print(f"[init] Parametric sweep best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Phase‑1 – Exponentiated‑gradient descent with multi‑active
    #          gradients (runs until the remaining budget is exhausted).
    # ------------------------------------------------------------------
    phase1_deadline = deadline
    best_a, best_val = _eg_multi_active(
        a0, n,
        deadline=phase1_deadline,
        rng=rng,
    )

    # ------------------------------------------------------------------
    # 3️⃣  Phase‑2 – Tiny uniform mixing (often gives a small final drop).
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final reporting (the evaluation harness will call evaluate_sequence
    # again, but we also log it here for debugging).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
