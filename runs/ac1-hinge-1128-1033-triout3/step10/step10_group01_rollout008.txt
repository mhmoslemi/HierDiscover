def _adam_softmax_optimize(a0, n, opt_deadline, start_time):
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a) # using exact max
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    step = lr
    iteration = 0
    alpha = 1.0
    alpha_max = 30.0
    while True:
        now = time.time()
        if now >= opt_deadline:
            break
        # update alpha gradually based on progress fraction
        prog = (now - start_time) / (opt_deadline - start_time)
        alpha = 1.0 + (alpha_max - 1.0) * prog

        # compute b via FFT
        size_b = 1 << ((2*n-2).bit_length())
        fx = np.fft.rfft(a, size_b)
        b = np.fft.irfft(fx*fx, size_b)[:2*n-1]
        maxb = b.max()
        val = 2.0 * n * maxb
        if val < best_val - 1e-12:
            best_val, best_a = val, a.copy()
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")

        # softmax weights computation
        b_centered = b - maxb
        exp_term = np.exp(alpha * b_centered)
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp

        # gradient
        w_rev = w[::-1]
        # full conv a * w_rev
        size_c = 1 << ((3*n-2 - 1).bit_length())
        fx = np.fft.rfft(a, size_c)
        fw = np.fft.rfft(w_rev, size_c)
        conv_rev = np.fft.irfft(fx * fw, size_c)[:3*n-2]
        segment = conv_rev[n-1:2*n-1][::-1]  # length n
        grad = 4.0 * n * segment

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step_dir = m_hat / (np.sqrt(v_hat) + eps)

        # use learning rate schedule: lr = lr_start * (lr_end/lr_start) ** prog
        lr_start = 0.1
        lr_end = 0.001
        lr_cur = lr_start * (lr_end / lr_start) ** prog
        a_new = a - lr_cur * step_dir
        # ensure non-negative
        a_new = np.maximum(a_new, 0.0)
        # project onto simplex
        a_sum = a_new.sum()
        if a_sum == 0.0:
            # random restart
            a_new = np.random.dirichlet(np.ones(n))
        else:
            a_new = a_new / a_sum

        a = a_new

        # optional early exit if best_val <= target
        if best_val <= 1.503:
            break
    return best_a, best_val
```

"""
Hybrid optimizer for the step‑function challenge.

The algorithm works in three successive stages for a given length `n`.

1️⃣  **Warm‑start** – if a previous solution exists (`height_sequence_1`),
    it is linearly interpolated to the new size; otherwise a few
    Dirichlet draws are tried and the best is kept.

2️⃣  **Soft‑max Adam descent** – we minimise a smooth approximation of
    the objective
        F(a)=2·n·max_k (a∗a)_k
    by replacing the max with a soft‑max (log‑sum‑exp) using a temperature
    parameter `α`.  The gradient of the smoothed objective is obtained
    analytically via a single FFT‑based correlation, and the step is
    taken with the Adam optimiser.  After each step the vector is projected
    onto the probability simplex (non‑negative, sum = 1).

3️⃣  **Refinement** – a short projected‑gradient descent (sub‑gradient of
    the active entry) followed by the intensive simulated‑annealing hill‑climb
    from the reference implementation finish the search.

The routine allocates the 1000 s budget across a few problem sizes
(4000, 8000, 12000).  The best sequence found over all sizes is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution_self(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two (real) vectors via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution_self(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple initialisation (interpolation, Dirichlet seeds)
# ----------------------------------------------------------------------
def _initial_sequence(n: int) -> np.ndarray:
    """Obtain a first feasible sequence of length n."""
    # try to reuse a previously found solution
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        print(f"[init] interpolated previous solution to n={n}")
        return a
    except Exception:

    # fallback: a few Dirichlet draws, keep the best
    best_a = None
    best_val = np.inf

    print(f"[init] Dirichlet start for n={n}, best value={best_val:.6f}")
    return best_a


# ----------------------------------------------------------------------
# Adam optimiser on a soft‑max surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_softmax_optimize(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(a∗a) via a smooth log‑sum‑exp surrogate.
    Uses Adam with Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedules
    alpha_start, alpha_end = 1.0, 30.0          # smoothness schedule
    lr_start, lr_end = 0.12, 0.002            # learning‑rate schedule

    t = 0
    no_improve = 0
    stagnation_limit = 20000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress fraction (local to this optimiser)
        prog = (now - start_time) / (deadline - start_time)
        alpha = alpha_start + (alpha_end - alpha_start) * prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # ----- compute convolution and true objective -----
        b = _full_convolution_self(a)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {t:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- soft‑max weights -----
        # shift by max to avoid overflow
        shifted = b - maxb
        exp_term = np.exp(alpha * shifted)
        w = exp_term / exp_term.sum()

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]                         # reverse for correlation
        conv_rev = _full_convolution_fft(a, w_rev)   # length 3n‑2
        segment = conv_rev[n - 1: 2 * n - 1][::-1]    # length n, contains Σ a[t] w[t+j]
        grad = 4.0 * n * segment                  # ∇ (2·n·softmax) = 4·n·segment

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:
            a_candidate = np.random.dirichlet(np.ones(n))
        else:
            a_candidate /= s

        a = a_candidate

        # ----- occasional perturbation on stagnation -----
        if no_improve > stagnation_limit:
            a = best_a.copy()
            a += np.random.normal(scale=0.02, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0
            print(f"[Adam {t:7d}] stagnation – random perturb")

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (as in the reference code)
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  target_val: float = 1.503) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a, best_val = a.copy(), _objective(a)

    step = 0.02
    min_step = 1e-12
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    # pre‑compute FFT size for self‑convolution
    size_conv = 1 << ((2 * n - 2).bit_length())

    while True:
        now = time.time()
        if now >= deadline:
            break

        # convolution and active index
        fx = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(fx * fx, size_conv)[:2 * n - 1]
        kmax = int(conv.argmax())
        maxb = conv.max()
        cur_val = 2.0 * n * maxb

        # sub‑gradient of the active entry
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # tentative step
        a_candidate = a - step * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:
            a_candidate = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_candidate /= s

        # evaluate
        fx_c = np.fft.rfft(a_candidate, size_conv)
        conv_c = np.fft.irfft(fx_c * fx_c, size_conv)[:2 * n - 1]
        new_val = 2.0 * n * conv_c.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[PGD {int(time.time()-start_time):7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.5
            no_improve += 1
            if step < min_step:
                break

        if no_improve > IMPROVE_LIMIT:
            # random perturbation
            a = best_a.copy()
            a += np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            step = 0.02
            no_improve = 0
            print(f"[PGD {int(time.time()-start_time):7d}] stagnation – perturb")

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    a = a0.copy()
    b = _full_convolution_self(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (where mass is taken from)
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index (where mass is moved to)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update the convolution incrementally
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.025, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution_self(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # spare for final call

    # Problem sizes to explore – a multi‑scale schedule
    sizes = [4000, 8000, 12000]

    global_best_a = None
    global_best_val = np.inf
    global_best_n = None

    prev_a = None
    prev_n = None

    for n in sizes:
        if time.time() >= deadline:

            a0 = np.maximum(a0, 0.0)

            print(f"[init] interpolated solution from n={prev_n} to n={n}")
        else:
            a0 = _initial_sequence(n)

        # ---- Adam soft‑max optimisation ---------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate up to half of the remaining time to the Adam phase
        adam_deadline = time.time() + min(30.0, remaining * 0.5)
        a_adam, val_adam = _adam_softmax_optimize(
            a0, n,
            deadline=adam_deadline,
            start_time=start_time,
            target_val=1.503,
        )
        print(f"[Adam] n={n} best after Adam = {val_adam:.6f}")

        # ---- short PGD refinement ---------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        pgd_deadline = time.time() + min(15.0, remaining * 0.3)
        a_pgd, val_pgd = _pgd_optimize(
            a_adam, n,
            deadline=pgd_deadline,
            start_time=start_time,
            target_val=1.503,
        )
        print(f"[PGD] n={n} best after PGD = {val_pgd:.6f}")

        # pick the better of the two
        if val_pgd < val_adam:
            local_best_a, local_best_val = a_pgd, val_pgd
        else:
            local_best_a, local_best_val = a_adam, val_adam

        if local_best_val < global_best_val:
            global_best_val, global_best_a, global_best_n = (
                local_best_val,
                local_best_a.copy(),
                n,
            )
            print(f"[global] new best from n={n}: {global_best_val:.6f}")

        # early exit if we already beat the target
        if global_best_val <= 1.503:
            break

        prev_a, prev_n = a_pgd, n  # carry forward the refined solution

    # ---- final intensive hill‑climbing -------------------------------
    remaining = deadline - time.time()
    if global_best_a is not None and remaining > 1.0:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            global_best_n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ---- final secure evaluation ------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
