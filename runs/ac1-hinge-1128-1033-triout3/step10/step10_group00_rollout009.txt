def _cross_corr(w, a):
    # w length L=2n-1, a length n
    n = a.shape[0]
    a_rev = a[::-1]
    L = len(w)
    out_len = L + n - 1  # = 3n-2

    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:out_len]
    # extract indices n-1 to n-1+n-1 inclusive (i.e., n entries)
    start = n - 1
    g = conv[start:start+n]  # length n
    return g

def _next_pow2(x: int) -> int:

    """Full convolution of x with itself (real FFT, length 2·len(x)-1)."""

    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv
