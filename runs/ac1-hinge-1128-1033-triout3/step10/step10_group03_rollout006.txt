Propose a non‑negative step function that minimises
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2).

The algorithm works in three phases:

1️⃣  **Exploratory shape search** – generate many candidate vectors from
    flexible parametric families (Beta, exponential, power‑law) and from
    random Dirichlet draws.  All candidates are normalised to Σa=1 and
    evaluated with a fast FFT‑based convolution.  The best candidate is
    kept as a starting point.

2️⃣  **Refinement with weighted sub‑gradient descent** – the exact
    objective is 2·n·max_b where max_b = max(conv(a,a)).
    At each iteration we
        • compute the full autocorrelation via FFT,
        • collect the indices whose value is within a tiny tolerance of the
          maximum,
        • assign softmax weights (temperature λ grows over time) to these
          indices and build a weighted sub‑gradient
            ∂/∂a_i  Σ w_k·b_k = 2·Σ w_k·a_{k‑i},
        • take a projected step on the probability simplex,
        • enforce a monotone‑decreasing shape by isotonic regression
          (piecewise constant, then renormalise),
        • occasionally add a tiny random jitter to escape plateaus.

   The step‑size decays linearly from 0.8 to 1e‑6.  The loop runs until
   the time budget is exhausted or the target value (≤ 1.503) is reached.

3️⃣  **Final secure evaluation** – the best vector found is passed to the
    provided ``evaluate_sequence`` function and returned.

All helper functions are defined at top level and avoid closures.

import time

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx = 1, x ≥ 0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Beta‑shaped vector a_i ∝ i^{p‑1} (n‑i+1)^{q‑1}, i=1..n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(r: float, n: int) -> np.ndarray:
    """Exponential shape a_i ∝ r^{i‑1}, 0 < r < 1."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _powerlaw_shape(alpha: float, beta: float, n: int) -> np.ndarray:
    """Power‑law shape a_i ∝ (i + α)^{‑β}, i = 1..n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i + alpha) ** (-beta)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _isotonic_decreasing(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the cone of non‑increasing vectors.
    The result is then normalised to sum to 1 (simplex).
    """
    n = len(v)
    # PAVA for decreasing sequence (store block sums and counts)
    sums = v.tolist()
    counts = [1] * n
    i = 0
    while i < len(sums) - 1:
        avg_i = sums[i] / counts[i]
        avg_next = sums[i + 1] / counts[i + 1]
        if avg_i < avg_next - 1e-12:  # violation of non‑increasing
            # merge blocks i and i+1
            sums[i] += sums[i + 1]
            counts[i] += counts[i + 1]
            del sums[i + 1]
            del counts[i + 1]
            i = max(i - 1, 0)
        else:
            i += 1
    # expand merged averages back to a full‑length vector
    out = np.empty(n, dtype=np.float64)
    idx = 0
    for s, c in zip(sums, counts):
        avg = s / c
        out[idx: idx + c] = avg
        idx += c
    # Clip tiny negatives (numeric noise) and renormalise to the simplex
    out = np.maximum(out, 0.0)
    total = out.sum()
    if total > 0.0:
        out /= total
    else:
        out = np.full_like(out, 1.0 / n)
    return out

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # Problem size – a compromise between expressivity and speed.
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previous solution (if available)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        # Interpolate old vector onto the current resolution
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
            val = _objective(a0)
            best_a, best_val = a0.copy(), val
            print(f"[init] interpolated previous solution, val={val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad family search (Beta, exponential, power‑law, Dirichlet)
    # ------------------------------------------------------------------
    # fallback uniform if nothing better has been found yet
    uniform = np.full(n, 1.0 / n)
    uniform_val = _objective(uniform)
    if uniform_val < best_val:
        best_a, best_val = uniform.copy(), uniform_val
        print(f"[init] uniform start, val={uniform_val:.6f}")

    # ----- Beta family (grid) -----
    p_grid = np.geomspace(0.001, 20.0, 30)
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is None:
                continue
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val
                print(f"[beta_grid] new best={val:.6f} (p={p:.4g}, q={q:.4g})")
        if time.time() >= deadline - 80:
            break

    # ----- Exponential family (grid) -----
    r_grid = np.linspace(0.5, 0.999, 30)
    for r in r_grid:
        cand = _exp_shape(r, n)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val
            print(f"[exp_grid] new best={val:.6f} (r={r:.4g})")
        if time.time() >= deadline - 70:
            break

    # ----- Power‑law random draws -----
    rng = np.random.default_rng(seed + 12345)
    for _ in range(2000):
        alpha = rng.uniform(0.0, 10.0)
        beta = rng.uniform(0.5, 5.0)
        cand = _powerlaw_shape(alpha, beta, n)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val
            print(f"[powlaw_rand] new best={val:.6f} (α={alpha:.3f}, β={beta:.3f})")
        if time.time() >= deadline - 60:
            break

    # ----- Dirichlet samples (extra diversity) -----
    for _ in range(12):
        cand = rng.dirichlet(np.ones(n))
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val
            print(f"[dirichlet] new best={val:.6f}")
        if time.time() >= deadline - 50:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Refine with weighted sub‑gradient descent
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_seq = a.copy()
    best_score = best_val

    # step‑size schedule (linear decay)
    step_start = 0.8
    step_end = 1e-6

    # active‑set tolerance (relative)
    eps_factor = 5e-4

    # λ‑schedule for soft‑max weighting of the active constraints
    lambda_start = 50.0
    lambda_end = 5000.0

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- compute full autocorrelation ----
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        # ---- keep track of the best seen ----
        if cur_val < best_score:
            best_score, best_seq = cur_val, a.copy()
            print(f"[subgrad] iter {iteration} new best={cur_val:.6f}")
            if best_score <= 1.503:
                break

        # ---- active set (indices close to the maximum) ----
        eps = max(eps_factor * max_b, 1e-12)
        active_idx = np.where(b >= max_b - eps)[0]

        # ---- progress‑dependent parameters ----
        progress = (now - start_time) / (deadline - start_time)
        lam = lambda_start * (1.0 - progress) + lambda_end * progress
        step = step_start * (1.0 - progress) + step_end * progress

        # ---- soft‑max weights on the active indices ----
        # (all values ≤ 0, so exponentials are ≤ 1)
        exp_vals = np.exp(lam * (b[active_idx] - max_b))
        weight_sum = exp_vals.sum()
        if weight_sum == 0.0:
            weights = np.full_like(active_idx, 1.0 / len(active_idx))
        else:
            weights = exp_vals / weight_sum

        # ---- weighted sub‑gradient ----
        grad = np.zeros_like(a)
        for k, w in zip(active_idx, weights):
            # indices i for which a_{k‑i} is defined
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            i_range = np.arange(start_i, end_i + 1, dtype=int)
            a_sub = a[k - i_range]
            grad[i_range] += 2.0 * w * a_sub

        # ---- gradient step + projection onto simplex ----
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        # ---- enforce monotone‑decreasing shape (isotonic regression) ----
        a_candidate = _isotonic_decreasing(a_candidate)

        # ---- occasional jitter to escape flat regions ----
        if rng.random() < 0.001:
            a_candidate += rng.normal(scale=0.001, size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)
            a_candidate = _isotonic_decreasing(a_candidate)

        a = a_candidate
        iteration += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final evaluation (secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
