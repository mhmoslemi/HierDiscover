import time, math, numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _conv_full_fft(x, y):
    n = len(x) + len(y) - 1
    size = _next_pow2(n)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv

def _full_convolution_fft(a):
    return _conv_full_fft(a, a)

def _objective_value(a, n):
    b = _full_convolution_fft(a)
    return 2.0 * n * b.max()

def _project_simplex(v):
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _initial_candidate(n):
    # As before but may keep simpler families for time.
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a, n)
    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    i = np.arange(n, dtype=float)
    i1 = i + 1.0
    # 1. Exponential tails
    for r in np.linspace(0.995, 0.999999, 30):
        _try(r ** i)
        _try(r ** i + r ** (n - 1 - i))
    # 2. Power-law
    for p in np.linspace(0.1, 5.0, 40):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))
    # 3. Beta densities
    beta_vals = np.linspace(0.1, 2.0, 15)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a_vec = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a_vec)
    # 4. Arcsine
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)
    # 5. Sine-power
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.2, 8.0, 20):
        _try(base_sin ** p)
    # 6. Random Dirichlet draws
    rng = np.random.default_rng(123456)
    for _ in range(120):
        _try(rng.dirichlet(np.ones(n)))
    # ignore interpolation of old solution
    return best_a, best_val

def _subgradient(w, a, n):
    a_rev = a[::-1]
    conv = _conv_full_fft(w, a_rev)   # length 3n-2
    start = n - 1
    g = conv[start:start + n]
    return 4.0 * n * g

def _mix_with_uniform(a, n):
    best_a = a.copy()
    best_val = _objective_value(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)  # should already sum to 1 but just in case
        val = _objective_value(cand, n)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    return best_a, best_val

"""
Projected sub‑gradient descent for the step‑function bound.
1. A broad parametric sweep yields a strong initial shape.
2. Iteratively minimise the max‑convolution
   (the quantity 2·n·max(conv) = evaluate_sequence output for Σa=1)
   by moving opposite to a sub‑gradient of the max‑convolution.
   The step size is adjusted automatically and the iterate is projected
   onto the probability simplex after each move.
3. A tiny uniform mixing finishes the polishing stage.
The whole routine respects the required `propose_candidate` signature and
runs within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two for FFT padding
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Convolution helpers (FFT based – exact)
# ----------------------------------------------------------------------
def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of x with y using FFT.
    Length = len(x) + len(y) - 1.
    """
    n = len(x) + len(y) - 1
    size = _next_pow2(n)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_full_fft(a, a)


# ----------------------------------------------------------------------
# Objective (sum is assumed to be 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray, n: int) -> float:
    """Exact objective for a normalised vector (Σ a = 1)."""
    b = _full_convolution_fft(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – broad parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Search a rich collection of parametric families for a low‑objective seed."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a, n)

    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)            # 0 … n‑1
    i1 = i + 1.0                              # 1 … n

    # 1. Exponential tails (decreasing and symmetric)
    for r in np.linspace(0.995, 0.999999, 30):
        _try(r ** i)                         # decreasing
        _try(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.1, 5.0, 40):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta densities (a,b in [0.1, 2])
    beta_vals = np.linspace(0.1, 2.0, 15)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a_vec = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a_vec)

    # 4. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # 5. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.2, 8.0, 20):
        _try(base_sin ** p)

    # 6. Random Dirichlet draws for diversity
    rng = np.random.default_rng(123456)
    for _ in range(120):
        _try(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (scaled by 4·n)
# ----------------------------------------------------------------------
def _subgradient(w: np.ndarray, a: np.ndarray, n: int) -> np.ndarray:
    """
    Compute 4·n·∑_k w_k·a_{k‑i} for all i.
    w is the active‑set probability vector (length 2·n‑1),
    a is the current step‑function (length n).
    """
    a_rev = a[::-1]
    conv = _conv_full_fft(w, a_rev)          # length = 3·n‑2
    start = n - 1
    g = conv[start:start + n]                # size n
    return 4.0 * n * g


# ----------------------------------------------------------------------
# Tiny uniform mixing polish
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int):
    """Convex combination with the uniform distribution; keeps Σa=1."""
    best_a = a.copy()
    best_val = _objective_value(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        # projection is not strictly necessary – the combination already sums to 1,
        # but we keep it for safety against round‑off.
        cand = _project_simplex(cand)
        val = _objective_value(cand, n)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the 1000 s budget.
    The method consists of:
      * a parametric sweep (Phase 0) to obtain a good seed,
      * projected sub‑gradient descent on the max‑convolution (Phase 1),
      * a final uniform‑mix polishing step (Phase 2).
    Returns a list of non‑negative floats (Σ ≈ 1) ready for
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # --------------------------------------------------------------
    # Phase 0 – broad sweep for a strong seed
    # --------------------------------------------------------------
    n = 9000
    print("[Phase 0] parametric sweep")
    a, cur_val = _initial_candidate(n)
    best_a = a.copy()
    best_val = cur_val
    print(f"[Phase 0] best after sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent
    # --------------------------------------------------------------
    print("[Phase 1] start sub‑gradient descent")
    step = 0.08               # initial step length
    min_step = 1e-6
    no_improve = 0
    iteration = 0
    target_val = 1.503

    while time.time() < deadline:
        # current objective
        b = _full_convolution_fft(a)
        max_b = b.max()
        cur_obj = 2.0 * n * max_b

        # keep the best ever seen
        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[Iter {iteration:07d}] new best = {best_val:.6f}")
            no_improve = 0
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # active set for the max‑convolution
        tol = max_b * 1e-12 + 1e-15
        active = (b >= max_b - tol).astype(float)
        act_sum = active.sum()
        if act_sum == 0:
            active = (b == max_b).astype(float)
            act_sum = active.sum()
        w = active / act_sum

        # sub‑gradient (scaled)
        g = _subgradient(w, a, n)

        # try several step‑size candidates; accept only improvements
        improved = False
        for factor in (step, step * 0.5, step * 0.25):
            a_candidate = a - factor * g
            a_candidate = _project_simplex(a_candidate)
            cand_obj = _objective_value(a_candidate, n)
            if cand_obj < cur_obj:
                a = a_candidate
                step = factor * 1.1         # slight increase after success
                improved = True
                break
        if not improved:
            step = max(step * 0.5, min_step)
            # occasional random perturbation when stuck for long
            if no_improve > 20000:
                noise = np.random.normal(scale=0.005, size=n)
                a = a + noise
                a = np.maximum(a, 0.0)
                a = _project_simplex(a)
                no_improve = 0

        iteration += 1

    print(f"[Phase 1] best after descent = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – tiny uniform mixing polish
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Phase 2] best after mixing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final log using the provided wrapper (just for information)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
