"""
step‑function optimizer (≈ 1000 s)

1.  Random search over two‑parameter Beta‑shaped densities
    a_i ∝ (i+1)^{p-1}·(n-i)^{q-1}  (n≈1500 during the search).
    The sum of the vector is normalised to 1, the objective
    is 2·n·max(conv(a,a)) – exactly the value returned by
    ``evaluate_sequence`` (the scaling by the L1‑norm cancels).

2.  The best density is interpolated to the target length
    (8000 entries) and normalised again.

3.  A cheap projected sub‑gradient polish (exact max‑convolution)
    runs on the full length until the time budget is almost spent.

The whole routine respects the 1000 s limit and returns the
best sequence found (plain list of floats, all non‑negative)."""

import time
import numpy as np
from scipy.special import beta as beta_func  # Beta function B(x,y)


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x | x ≥ 0, sum x = 1}."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polish (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on
        f(a) = 2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        # indices achieving the maximum (usually a small set)
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # tiny random jitter to escape plateaus
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑shaped decreasing density (p>0, q>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _power_shape(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  Returns a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # --------------------------------------------------------------
    # 0️⃣  Parameter search on a modest size (≈1500)
    # --------------------------------------------------------------
    n_search = 1500
    best_a_search = None
    best_val_search = np.inf

    # always evaluate uniform baseline
    uniform = np.full(n_search, 1.0 / n_search)
    best_val_search = _objective(uniform)
    best_a_search = uniform.copy()
    print(f"[Init] uniform baseline = {best_val_search:.6f}")

    # time budget for this phase (≈40 % of total)
    phase1_deadline = start_time + 0.40 * budget_s

    while time.time() < phase1_deadline:
        # Randomly decide which family to sample
        if rng.random() < 0.55:                     # beta family
            logp = rng.uniform(-5.0, 2.0)            # p ∈ [e⁻⁵, e²] ≈ [0.0067, 7.4]
            logq = rng.uniform(-5.0, 2.0)
            p = np.exp(logp)
            q = np.exp(logq)
            cand = _beta_shape(n_search, p, q)
        else:                                       # simple power‑law
            logp = rng.uniform(-5.0, 2.0)
            p = np.exp(logp)
            cand = _power_shape(n_search, p)

        val = _objective(cand)
        if val < best_val_search:
            best_val_search = val
            best_a_search = cand.copy()
            print(f"[Search] new best = {best_val_search:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Upsample to the target length (8000) preserving shape
    # --------------------------------------------------------------
    target_n = 8000
    x_old = np.linspace(0.0, 1.0, n_search)
    x_new = np.linspace(0.0, 1.0, target_n)
    best_a = np.interp(x_new, x_old, best_a_search)
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()
    best_val = _objective(best_a)
    print(f"[Upsample] objective after interpolation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Short sub‑gradient polishing on the full size
    # --------------------------------------------------------------
    # Use whatever time is left, but cap iterations to keep within budget
    polishing_deadline = deadline
    best_a, best_val = _subgradient_polish(
        best_a,
        target_n,
        deadline=polishing_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=120_000,          # enough iterations for a decent refinement
    )

    # --------------------------------------------------------------
    # 3️⃣  Final secure evaluation (optional but useful for logging)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain list of floats (non‑negative, sum > 0)
    return best_a.tolist()
