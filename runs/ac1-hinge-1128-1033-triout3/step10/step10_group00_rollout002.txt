Improved step‑function optimiser.

The routine combines:
* a dense parametric sweep of a few analytic families (Beta, sine, Hann)
  to obtain a high‑quality start;
* a stochastic mirror‑descent on the simplex that focuses on the
  largest convolution entries.  The gradient weight is a soft‑max
  over the top entries with a temperature that rises during the run,
  allowing the algorithm to first act on many entries and later hone
  in on the true maximum.
* a symmetry projection (a_i = a_{n‑1‑i}) and a late‑stage random
  perturbation to escape stagnation.

The method stays within the 1000 s budget, uses only NumPy and
produces a sequence that improves the previous best bound.

# ----------------------------------------------------------------------
# Convolution helper (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised sequence (sum == 1)
# ----------------------------------------------------------------------
def _objective_normed(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for sum(a)=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex – fallback safeguard
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Return a normalised Beta‑shaped sequence of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


def _sine_window(n: int) -> np.ndarray:
    """Sine window a_i = sin(pi*(i+0.5)/n), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _hann_window(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Mirror‑descent update using a soft‑max over the M largest conv entries
# ----------------------------------------------------------------------
def _mirror_update(a: np.ndarray,
                   conv: np.ndarray,
                   top_idx: np.ndarray,
                   top_vals: np.ndarray,
                   eta: float,
                   beta: float) -> np.ndarray:
    """
    Perform one entropy‑mirror‑descent step.
    a      – current probability vector (sum==1)
    conv   – full self‑convolution of a
    top_idx/top_vals – indices and values of the selected convolution entries
    eta    – stepsize
    beta   – inverse temperature for the soft‑max weighting of the top entries
    """
    n = a.shape[0]

    # soft‑max weighting – the larger beta, the more the update focuses on the
    # absolute maximum.  For numerical stability we subtract the max argument.
    weights = np.exp(beta * top_vals - np.max(beta * top_vals))
    weights_sum = weights.sum()
    if weights_sum == 0.0:
        weights = np.full_like(weights, 1.0 / len(weights))
    else:
        weights /= weights_sum

    g = np.zeros_like(a)

    for k, w in zip(top_idx, weights):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        if low > high:
            continue
        j = np.arange(low, high + 1)
        a_shift = a[k - j]                     # a_{k‑j}
        g[j] += 2.0 * w * a_shift

    # Entropy (multiplicative) mirror step
    a_new = a * np.exp(-eta * g)

    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s

    # Enforce symmetry a_i = a_{n‑1‑i}
    a_new = (a_new + a_new[::-1]) * 0.5
    a_new /= a_new.sum()
    return a_new

    return conv


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised sequence (sum == 1)
# ----------------------------------------------------------------------
def _objective_normed(a: np.ndarray) -> float:
    """Compute 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (fallback)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families for quick seeding
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑shaped density on {1,…,n}, normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


def _sine_window(n: int) -> np.ndarray:
    """Sine window a_i = sin(π·(i+½)/n), normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _hann_window(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Mirror‑descent step using a soft‑max over the M largest convolution entries
# ----------------------------------------------------------------------
def _mirror_update(a: np.ndarray,
                   conv: np.ndarray,
                   top_idx: np.ndarray,
                   top_vals: np.ndarray,
                   eta: float,
                   beta: float) -> np.ndarray:
    """
    Entropy mirror‑descent update.
    a          – current probability vector (Σa = 1)
    conv       – full self‑convolution of a
    top_idx    – indices of the chosen convolution entries
    top_vals   – their current values
    eta        – stepsize
    beta       – inverse temperature for the soft‑max weighting
    """
    n = a.shape[0]

    # Soft‑max weights over the selected entries
    weights = np.exp(beta * top_vals - np.max(beta * top_vals))
    w_sum = weights.sum()
    if w_sum == 0.0:
        weights = np.full_like(weights, 1.0 / len(weights))
    else:
        weights /= w_sum

    g = np.zeros_like(a)

    for k, w in zip(top_idx, weights):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        if low > high:
            continue
        j = np.arange(low, high + 1)
        a_shift = a[k - j]                # a_{k‑j}
        g[j] += 2.0 * w * a_shift

    # Multiplicative (entropy) mirror step
    a_new = a * np.exp(-eta * g)

    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s

    # Enforce symmetry a_i = a_{n‑1‑i}
    a_new = (a_new + a_new[::-1]) * 0.5
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  The routine respects the 1000 s time
    budget and returns the best sequence it discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 12 000 points give a good trade‑off
    # --------------------------------------------------------------
    n = 12000

    # --------------------------------------------------------------
    # Initialise best solution (use previous run if available)
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf

    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        a0 = (a0 + a0[::-1]) * 0.5
        a0 /= a0.sum()
        best_a = a0
        best_val = _objective_normed(a0)
        print("[init] interpolated previous solution – val={:.6f}".format(best_val))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Sweep analytic families (Beta, sine, Hann)
    # --------------------------------------------------------------
    p_vals = np.concatenate([
        np.linspace(0.1, 1.0, 12),
        np.geomspace(1.0, 20.0, 18)[1:]
    ])
    q_vals = p_vals.copy()

    for p in p_vals:
        for q in q_vals:
            a = _beta_family(n, p, q)
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
            val = _objective_normed(a)
            if val < best_val:
                best_a, best_val = a, val
                print("[beta] improved – p={:.3f}, q={:.3f}, val={:.6f}".format(p, q, val))

    a = _sine_window(n)
    val = _objective_normed(a)
    if val < best_val:
        best_a, best_val = a, val
        print("[sine] improved – val={:.6f}".format(val))

    a = _hann_window(n)
    val = _objective_normed(a)
    if val < best_val:
        best_a, best_val = a, val
        print("[hann] improved – val={:.6f}".format(val))

    # --------------------------------------------------------------
    # 2️⃣  Diverse random Dirichlet seeds
    # --------------------------------------------------------------
    for _ in range(15):
        a = np.random.dirichlet(np.ones(n))
        a = (a + a[::-1]) * 0.5
        a /= a.sum()
        val = _objective_normed(a)
        if val < best_val:
            best_a, best_val = a, val
            print("[dir] improved – val={:.6f}".format(val))

    # --------------------------------------------------------------
    # Fallback – uniform (should never be needed)
    # --------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective_normed(best_a)
        print("[fallback] using uniform – val={:.6f}".format(best_val))

    # --------------------------------------------------------------
    # 3️⃣  Mirror‑descent / simulated‑annealing phase (remaining time)
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    stagnation_cnt = 0
    IMPROVE_LIMIT = 4000          # after this many non‑improvements we perturb
    eta_start, eta_end = 1.5, 0.001
    beta_start, beta_end = 0.3, 6.0
    T_start, T_end = 5e-2, 1e-8
    M_top_max, M_top_min = 60, 8

    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(a)
        cur_val = float(2.0 * n * conv.max())

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            stagnation_cnt = 0
            print("[iter] new best = {:.6f} at t={:.1f}s".format(
                best_val, now - start_time))
        else:
            stagnation_cnt += 1

        prog = (now - start_time) / (deadline - start_time)

        # linearly shrink the number of top entries considered
        M_top = int(M_top_max * (1.0 - prog) + M_top_min * prog)
        M_top = max(M_top_min, min(M_top, M_top_max))

        top_idx = np.argpartition(-conv, M_top - 1)[:M_top]
        top_vals = conv[top_idx]

        eta = eta_start * (1.0 - prog) + eta_end * prog
        beta = beta_start * (1.0 - prog) + beta_end * prog
        T = T_start * (T_end / T_start) ** prog

        a_candidate = _mirror_update(a, conv, top_idx, top_vals, eta, beta)

        conv_cand = _full_convolution(a_candidate)
        cand_val = float(2.0 * n * conv_cand.max())

        accept = False
        if cand_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(cand_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_candidate
            cur_val = cand_val

        # ------------------------------------------------------------------
        # Stagnation handling – Gaussian perturbation around global best
        # ------------------------------------------------------------------
        if stagnation_cnt > IMPROVE_LIMIT:
            std = 0.04 * (1.0 - prog) + 0.005 * prog
            noise = np.random.normal(scale=std, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
            stagnation_cnt = 0
            tmp_val = float(2.0 * n * _full_convolution(a).max())
            print("[reset] perturbation – val={:.6f} at t={:.1f}s".format(
                tmp_val, now - start_time))

    # --------------------------------------------------------------
    # 4️⃣  Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
