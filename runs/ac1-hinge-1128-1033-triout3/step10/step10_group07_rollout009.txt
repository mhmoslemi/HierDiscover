"""
A gradient‐based optimizer for the max‑self‑convolution problem.

The code builds a probability vector `a` (non‑negative, sum = 1) of length `n`
and minimizes the quantity

    2·n·max(conv(a,a))

where `conv(a,a)` is the full linear convolution of `a` with itself.
Because the objective involves a non‑smooth max, we minimise a smooth
approximation `∑_k (conv_k)^p` with a continuation on the exponent `p`.
The gradient of this surrogate can be written in closed form using a
correlation (implemented via FFT).  An Adam optimiser on the log‑weights
(soft‑max parametrisation) drives the search, with occasional jitter and
re‑initialisation on stagnation.  After the main phase a tiny uniform
mixing is tried, and the best sequence found is returned.

The routine respects the 1000 s budget, works on a single CPU and only
uses NumPy.  It can also warm‑start from a previously discovered
`height_sequence_1` if that variable is defined in the global namespace.
"""

import time
import numpy as np


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    max_log = np.max(logits)
    exp_shift = np.exp(logits - max_log)
    return exp_shift / exp_shift.sum()


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the full linear convolution a*a using an FFT of length `size`.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


def _grad_from_weight(a: np.ndarray, weight: np.ndarray,
                     size_corr: int) -> np.ndarray:
    """
    Given a probability vector `a` and a weight vector `weight`
    (same length as conv(a,a)), compute
        grad_a = 2 * Σ_k weight_k * a_{k‑i}
    for every i, i.e. 2·correlation(a, weight).
    Implemented via FFT of length `size_corr`.
    """
    # correlation a ⋆ weight = conv(a, weight[::-1])
    w_rev = weight[::-1]
    fa = np.fft.rfft(a, size_corr)
    fw = np.fft.rfft(w_rev, size_corr)
    corr_full = np.fft.irfft(fa * fw, size_corr)[: 3 * a.shape[0] - 2]
    n = a.shape[0]
    # extract the needed slice: grad_i = 2 * corr_full[2n‑2‑i]
    indices = 2 * n - 2 - np.arange(n, dtype=int)
    grad = 2.0 * corr_full[indices]
    return grad


def _objective_val(a: np.ndarray, conv_cache: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)).
    `conv_cache` must be the full convolution a*a.
    """
    n = a.shape[0]
    return float(2.0 * n * np.max(conv_cache))


def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Try a tiny uniform mixture and keep the best result."""
    n = a.shape[0]
    best_a, best_val = a.copy(), _objective_val(a, _full_conv_fft(a, _next_pow2(2 * n - 1)))
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        conv = _full_conv_fft(cand, _next_pow2(2 * n - 1))
        val = _objective_val(cand, conv)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing
    `evaluate_sequence`.  Returns the best sequence found within the
    allotted time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final call to evaluate_sequence()
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more expressive power but costs
    # more per iteration.  20000 is a good compromise for a 1000 s budget.
    # ------------------------------------------------------------------
    n = 20000

    # ------------------------------------------------------------------
    # Warm‑start: reuse a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] Re‑used height_sequence_1 (interpolated).")
    except Exception:
        # random Dirichlet start – the soft‑max will enforce positivity.
        a0 = rng.dirichlet(np.ones(n))
        print("[init] Random Dirichlet start.")

    # Initialise the log‑weights for the soft‑max parametrisation.
    eps = 1e-15
    logit = np.log(a0 + eps)

    # Adam settings
    lr = 0.2                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    # FFT sizes (constant throughout the run)
    size_conv = _next_pow2(2 * n - 1)          # for a*a
    size_corr = _next_pow2(3 * n - 2)          # for correlation a ⋆ weight

    # Stagnation handling
    best_a = a0.copy()
    conv = _full_conv_fft(best_a, size_conv)
    best_val = _objective_val(best_a, conv)
    no_improve = 0
    IMPROV_LIMIT = 25000
    t = 1  # iteration counter

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Current probability vector from logits
        # --------------------------------------------------------------
        a = _softmax(logit)

        # --------------------------------------------------------------
        # Full self‑convolution (needed for both objective and weight)
        # --------------------------------------------------------------
        conv = _full_conv_fft(a, size_conv)          # length 2n‑1
        cur_val = _objective_val(a, conv)

        # Keep the best seen vector
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[iter {t:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Continuation on the smoothing exponent p
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        p = 2.0 + 10.0 * min(elapsed / total, 1.0)   # varies from 2 → 12

        # --------------------------------------------------------------
        # Build the weight vector for the smoothed L^p objective
        # --------------------------------------------------------------
        safe_conv = np.maximum(conv, 1e-30)          # avoid 0**neg
        weight = p * safe_conv ** (p - 1.0)          # length 2n‑1

        # --------------------------------------------------------------
        # Gradient w.r.t. the probability vector `a`
        # --------------------------------------------------------------
        grad_a = _grad_from_weight(a, weight, size_corr)

        # --------------------------------------------------------------
        # Chain rule to obtain gradient w.r.t. the logits (soft‑max)
        # --------------------------------------------------------------
        avg = np.dot(a, grad_a)                     # scalar ⟨a,∇a⟩
        grad_log = a * (grad_a - avg)

        # --------------------------------------------------------------
        # Adam update on logits
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Simple learning‑rate schedule – linear decay towards the end
        lr_adj = lr * (1.0 - elapsed / total) * 0.5 + 0.01

        logit -= lr_adj * m_hat / (np.sqrt(v_hat) + 1e-8)

        # --------------------------------------------------------------
        # Occasional jitter to escape shallow plateaus
        # --------------------------------------------------------------
        if rng.random() < 0.005:
            logit += rng.normal(scale=0.01, size=n)

        # --------------------------------------------------------------
        # Re‑initialise on long stagnation
        # --------------------------------------------------------------
        if no_improve > IMPROV_LIMIT:
            logit = np.log(best_a + eps) + rng.normal(scale=0.02, size=n)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[iter {t:7d}] re‑initialised after stagnation")

        t += 1

        # early break if we are about to exceed the deadline
        if time.time() > deadline - 0.2:
            break

    # ------------------------------------------------------------------
    # Phase 2 – tiny uniform mixing (often yields a small extra gain)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a)

    # ------------------------------------------------------------------
    # Final normalisation (evaluate_sequence does not re‑scale)
    # ------------------------------------------------------------------
    best_a /= best_a.sum()

    # ------------------------------------------------------------------
    # Report the final evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
