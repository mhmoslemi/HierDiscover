"""
Propose a step‑function (non‑negative sequence) with a low
self‑convolution peak.  The optimizer combines:

* a cheap parametric sweep (U‑shaped power‑law families),
* many short random restarts on a reduced problem size,
* a simple projected sub‑gradient descent on the exact max‑convolution,
* a final tiny uniform mixing.

All heavy work is done with FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full self‑convolution
# ----------------------------------------------------------------------
def _full_conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Return the full linear convolution a * a using FFT.
    Length = 2*len(a) - 1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (the same quantity that evaluate_sequence returns)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)**2)  (n = len(a))."""
    n = a.shape[0]
    max_b = _full_conv_fft(a).max()
    s = a.sum()
    # protection against division by zero (never happens in our usage)
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, sum x = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent on the exact max‑convolution.
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          max_iter: int,
                          step0: float,
                          decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Starting from a0 (already on the simplex), perform a few
    sub‑gradient steps on the exact max‑convolution objective.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    step = step0

    n = a.shape[0]

    for _ in range(max_iter):
        # compute full convolution and the index of the maximal entry
        conv = _full_conv_fft(a)
        k = int(np.argmax(conv))

        # gradient of the maximal entry w.r.t. a:
        #   ∂/∂a_i  (a * a)_k = 2·a_{k-i}   (if the index is valid)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # projected gradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        val = _objective(a_new)

        if val < best_val:
            best_val = val
            best_a = a_new.copy()

        a = a_new
        step *= decay

    return best_a, best_val

def _family_powerlaw(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (x_i·(1−x_i))^{‑p}."""
    eps = 1e-12

    a = (x * (1.0 - x) + eps) ** (-p)
    a = a / a.sum()
    return a


def _family_exponential(n: int, r: float) -> np.ndarray:
    """Symmetric geometric shape a_i ∝ r**i + r**(n‑1‑i)."""
    i = np.arange(n)
    a = r ** i + r ** (n - 1 - i)
    a = a / a.sum()
    return a

    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = a / a.sum()
    return a

    for p in np.linspace(0.10, 0.55, 10):
        cand = _family_powerlaw(Nbig, p)
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[pow  p={p:.3f}] new best = {best_val:.6f}")

    for r in np.linspace(0.95, 0.9999, 8):
        cand = _family_exponential(Nbig, r)
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[exp  r={r:.4f}] new best = {best_val:.6f}")

    cand = _family_sine(Nbig)
    val = _objective(cand)
    if val < best_val:
        best_val, best_a = val, cand.copy()
        print(f"[sine] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Short sub‑gradient polishing of the current best vector
    # ------------------------------------------------------------------
    # give it a few thousand iterations – cheap because FFT is used
    best_a, best_val = _subgradient_optimize(
        best_a,
        max_iter=5000,
        step0=0.5 / Nbig,
        decay=0.9998,
    )
    print(f"[sgd] after polishing best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Many random restarts on a smaller problem, up‑sample and keep the best.
    # ------------------------------------------------------------------
    # small working size – cheap to optimise many times
    Nsmall = 1800
    uniform_small = np.full(Nsmall, 1.0 / Nsmall)

    restart = 0
    while time.time() < deadline:
        # ----- generate a random start (Dirichlet) --------------------
        a = rng.dirichlet(np.ones(Nsmall))

        # ----- a few sub‑gradient steps on the small problem ----------
        a_opt, _ = _subgradient_optimize(
            a,
            max_iter=1200,
            step0=0.6 / Nsmall,
            decay=0.999,
        )

        # ----- upscale to the final resolution -----------------------
        xs = np.linspace(0.0, 1.0, Nsmall)
        xf = np.linspace(0.0, 1.0, Nbig)
        a_big = np.interp(xf, xs, a_opt)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # ----- quick objective evaluation (FFT) ----------------------
        val_big = _objective(a_big)

        if val_big < best_val:
            best_val, best_a = val_big, a_big.copy()
            print(f"[restart {restart:03d}] new best = {best_val:.6f}")

        # ----- tiny uniform mixing (often gives a marginal win) -----
        for alpha in (0.001, 0.003, 0.007):
            cand = (1.0 - alpha) * a_big + alpha * uniform_big
            cand /= cand.sum()
            val = _objective(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[mix {alpha:.3f}] improved best = {best_val:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # 4. Final uniform mixing (tiny) – safe post‑processing
    # ------------------------------------------------------------------
    for alpha in (0.001, 0.002, 0.004, 0.008):
        cand = (1 - alpha) * best_a + alpha * uniform_big
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()

    # ------------------------------------------------------------------
    # 5. Report to the harness (optional, never hurts)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
