def _full_convolution(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv
```

import time
import numpy as np

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _objective(a):
    b = _full_convolution(a)
    n = a.shape[0]
    return float(2.0 * n * b.max())

def _beta_candidate(n, alpha):
    # symmetric beta(alpha,alpha)
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    p = np.clip(x * (1.0 - x), eps, None)
    a = np.exp((alpha - 1.0) * np.log(p))
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

def _exp_candidate(n, r):
    i = np.arange(n)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

def _ends_power_candidate(n, p):
    i = np.arange(n)
    c = (i + 1) * (n - i)
    a = np.power(c, -p)
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a

def _random_dirichlet_candidate(n, rng):
    a = rng.dirichlet(np.ones(n))
    return a

def _local_refine(a, n, iters, rng):
    # simple hill climbing: move mass from i_max to j_min
    a = a.copy()
    best_val = _objective(a)
    best_a = a.copy()
    for _ in range(iters):
        b = _full_convolution(a)
        k = int(np.argmax(b))
        # contributions to entry k
        contrib = np.zeros(n)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib[mask] = a[mask] * a[idx[mask]]
        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break
        # choose a target with small a and not equal i
        # simple heuristic: choose index with smallest a
        j = int(np.argmin(a + (np.arange(n) == i) * 1e9))
        if i == j:
            break
        delta = a[i] * 0.05 * rng.random()  # move up to 5% of source mass
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
    return best_a, best_val

"""
Search for a step‑function that minimises `evaluate_sequence`.

The algorithm explores several parametric families that produce a dense
U‑shaped distribution (the only kind known to reduce the maximal
self‑convolution).  For each family a grid of parameters is tried,
the objective is evaluated with a fast FFT‑based convolution and the
best candidate is kept.  A cheap hill‑climber refines the best vector,
and finally a tiny amount of uniform mass is mixed in – this often yields a
small extra gain.

All arithmetic is performed with the sequence normalised to sum = 1,
so the objective returned by the helper `_objective` matches the value
computed by the supplied `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    b = _full_convolution(a)
    return float(2.0 * a.shape[0] * b.max())


# ----------------------------------------------------------------------
# Generators for parametric families
# ----------------------------------------------------------------------
def _beta_candidate(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(alpha,alpha) density on [0,1] (alpha>0)."""
    x = (np.arange(n) + 0.5) / n                # centre of each bin
    eps = 1e-12
    p = np.clip(x * (1.0 - x), eps, None)       # avoid log(0)
    # density ∝ p^{alpha‑1}
    a = np.exp((alpha - 1.0) * np.log(p))
    a /= a.sum()
    return a


def _exp_candidate(n: int, r: float) -> np.ndarray:
    """U‑shaped vector a_i ∝ r^i + r^{n‑1‑i} (0<r<1)."""
    i = np.arange(n)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    a /= a.sum()
    return a


def _ends_power_candidate(n: int, p: float) -> np.ndarray:
    """U‑shape a_i ∝ ((i+1)*(n‑i))^{-p}  (p>0)."""
    i = np.arange(n)
    c = (i + 1) * (n - i)                       # product of distances to the two ends
    a = np.power(c, -p)
    a /= a.sum()
    return a


def _random_dirichlet_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (simple baseline)."""
    return rng.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Very cheap local improvement: move a small amount of mass
# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, n: int,
                      max_iters: int,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Move a fraction of mass from a source index to a target index if it improves."""
    best_a = a.copy()
    best_val = _objective(best_a)
    a = best_a.copy()

    for _ in range(max_iters):
        # pick a source index proportional to its current contribution
        # to the maximal convolution entry
        b = _full_convolution(a)
        k = int(np.argmax(b))

        # contributions of each i to entry k
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break

        # pick a target with tiny value (heuristic)
        j = int(np.argmin(a + (np.arange(n) == i) * 1e9))

        if i == j:
            break

        # move up to a few percent of the source mass
        delta = a[i] * 0.03 * rng.random()
        if delta <= 0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            a = cand
        # else discard the move

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a step‑function under the 1000 s time budget.
    Returns a list of non‑negative floats that (approximately) minimise
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin for the wrapper call

    # problem size – a compromise between expressive power and speed
    n = 12000

    # ------------------------------------------------------------------
    # 1️⃣  Initial sweep over simple parametric families
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)              # uniform is a guaranteed baseline
    best_val = _objective(best_a)
    print(f"[init] uniform value = {best_val:.6f}")

    # ---- Symmetric Beta family (α ∈ (0.1, 1.0]) ----
    for alpha in np.linspace(0.10, 1.00, 19):
        a = _beta_candidate(n, alpha)
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[beta] α={alpha:5.3f} → {val:.6f}")

    # ---- Exponential U‑shape (r ∈ (0.55, 0.98)) ----
    for r in np.linspace(0.55, 0.98, 22):
        a = _exp_candidate(n, r)
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[exp ] r={r:5.3f} → {val:.6f}")

    # ---- Power‑of‑distance to ends (p ∈ (0.1, 1.0)) ----
    for p in np.linspace(0.10, 1.00, 10):
        a = _ends_power_candidate(n, p)
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[pow ] p={p:5.3f} → {val:.6f}")

    # ---- Random Dirichlet draws – cheap stochastic baseline ----
    for _ in range(30):
        a = _random_dirichlet_candidate(n, rng)
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[rand] → {val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Finetune the best candidate with a tiny hill‑climber.
    # ------------------------------------------------------------------
    # Reserve a small slice of the remaining budget for a quick local
    # optimisation; the loop is deliberately cheap (few hundred moves).
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_iters = 500
        best_a, best_val = _local_hill_climb(best_a, n, max_iters, rng)
        print(f"[ref ] after hill‑climb value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final mixing with a tiny uniform component (often beneficial)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[mix ] α={alpha:5.3f} → {val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final evaluation with the provided wrapper (the harness will
    #     also call it, but we print the value for debugging).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
