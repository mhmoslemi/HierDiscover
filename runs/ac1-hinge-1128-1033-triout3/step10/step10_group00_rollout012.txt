"""
Non‑negative step‑function optimiser (smooth max & exponentiated mirror descent).

The algorithm works directly with the scale‑invariant formulation.
For a probability vector a ( Σa = 1 ) the objective is
    F(a) = 2·n·max_k (a * a)[k] .
We minimise a smooth approximation of the max:
    f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k]) .
Its gradient can be written in closed form and is obtained by one
additional convolution.  An exponentiated‑gradient (entropy mirror
descent) step keeps the iterate on the simplex.
The schedule starts with a very smooth loss (small λ) and a large
stepsize, then gradually focuses on the true max (λ→∞) while the
steps shrink.  Random Gaussian perturbations are injected after a
long period without improvement to escape stagnation.
The routine respects the 1000 s budget, returns the best sequence
found and finally evaluates it with the supplied `evaluate_sequence`
function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Convolution helpers (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray):
    """Full linear convolution of x with itself (length 2·len(x)-1)
    using a real FFT. Returns the convolution, the FFT of x and the FFT length."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv, fx, size


def _conv_with_fft(a: np.ndarray, fa: np.ndarray, size: int, b: np.ndarray):
    """Full linear convolution a * b (using the already‑computed FFT of a)."""
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:a.shape[0] + b.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# 2. Smooth gradient of the log‑sum‑exp of the self‑convolution
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, fa: np.ndarray, size: int,
                     conv_self: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute ∇_a  (1/λ)·log Σ_k exp(λ·(a*a)[k]).
    The result is a vector of length n (same as a).
    """
    n = a.shape[0]

    # soft‑max weights w_k  (stable formulation)
    max_c = conv_self.max()
    exp_term = np.exp(lam * (conv_self - max_c))
    w = exp_term / exp_term.sum()                # length 2·n‑1, sum to 1

    # gradient = 2 * Σ_k w_k * shift(a, k)
    # This is a convolution of a with the reversed weight vector.
    w_rev = w[::-1]
    conv_rev = _conv_with_fft(a, fa, size, w_rev)   # length n + (2n‑1)‑1 = 3n‑2

    # Extract the needed n entries: conv_rev[2n‑2], conv_rev[2n‑3], … , conv_rev[n‑1]
    start = 2 * n - 2
    grad = 2.0 * conv_rev[start:start - n:-1]       # slice with step -1 gives n values
    return grad


# ----------------------------------------------------------------------
# 3. Exponentiated (entropy) mirror‑descent update
# ----------------------------------------------------------------------
def _exp_mirror_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    a ← a * exp(-η·grad)   followed by renormalisation onto the simplex.
    Works in log‑space for numerical stability.
    """
    # log‑space update
    log_a = np.log(a + 1e-30) - eta * grad
    log_a -= log_a.max()                 # prevent overflow
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# 4. Simple utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a already normalised (Σa=1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a)[0]
    return float(2.0 * n * conv.max())


def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑shaped seed (i**(p‑1)*(n‑i+1)**(q‑1) normalised)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


def _interpolate(old: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of an old solution to a new length n."""
    old_len = old.shape[0]
    if old_len == n:
        return old.copy()
    new_x = np.linspace(0.0, 1.0, n)
    old_x = np.linspace(0.0, 1.0, old_len)
    a = np.interp(new_x, old_x, old)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 5. Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that attempts to
    minimise `evaluate_sequence`.  The optimisation runs for up to
    `budget_s` seconds and prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small margin for the final call

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise best candidate (try to reuse the historic solution)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        best_a = _interpolate(old, n)
        best_val = _objective(best_a)
        print("[init] loaded previous solution – val = {:.6f}".format(best_val))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 0️⃣b  A cheap sweep over the Beta family as extra seeds
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback initialisation: a modest Beta shape
        best_a = _beta_family(n, p=0.4, q=3.0)

    # make sure we have a valid simplex point
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()
    best_val = _objective(best_a)

    # also keep a random Dirichlet seed for diversity
    rand_a = np.random.dirichlet(np.ones(n))
    rand_val = _objective(rand_a)
    if rand_val < best_val:
        best_a, best_val = rand_a, rand_val
        print("[init] random Dirichlet improved – val = {:.6f}".format(best_val))

    # ------------------------------------------------------------------
    # 1️⃣  Main optimisation loop – smooth log‑sum‑exp + exponentiated GD
    # ------------------------------------------------------------------
    a = best_a.copy()
    stagnation = 0
    IMPROVE_LIMIT = 4000          # after this many non‑improvements we perturb

    # schedule parameters
    eta_start, eta_end = 2.0, 0.001
    lam_start, lam_end = 0.2, 150.0

    while time.time() < deadline:

        # --------------------------------------------------------------
        prog = (time.time() - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - prog) + eta_end * prog
        lam = lam_start * (lam_end / lam_start) ** prog

        # --------------------------------------------------------------

        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:               # pragma: no cover

        final_score = None
