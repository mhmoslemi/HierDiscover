from scipy.optimize import linprog
from scipy import sparse

def solve_lp(f, M):
    n = len(f)
    rows = 2*n - 1
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for j in [i_start..i_end], the coefficient is f[i = k-j]
        for j in range(i_start, i_end+1):
            i = k - j
            # f_i coefficient for g_j
            data.append(f[i])
            row_ind.append(k)
            col_ind.append(j)
    A_ub = sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    b_ub = np.full(rows, M)
    c = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

    from scipy.optimize import linprog
    from scipy.linalg import toeplitz

    c = -np.ones(n)  # minimize -sum(g)

"""
Fast linear‑programming descent for the max‑convolution problem.

The algorithm proceeds in three stages.

1️⃣  **Initial sweep** – generate a diverse family of candidate
    vectors (exponential, power‑law, beta, sinusoidal, piece‑wise
    constant, Dirichlet draws) and keep the best one after normalising
    it to Σa = √(2 n).  With this normalisation the evaluation function
    is exactly the maximal entry of the self‑convolution,
    i.e.  evaluate_sequence(a) = max(a∗a).

2️⃣  **LP descent** – given the current vector `f` with
    `M = max(f∗f)`, solve the linear programme

        maximise   Σ g_j
        subject to  f ∗ g ≤ M   (2 n‑1 linear inequalities)
                    g ≥ 0

    The optimum `g` puts as much mass as possible into positions that
    are *least* constrained by `f`.  After scaling `g` to have the same
    ℓ₁‑norm as `f` we take a convex combination
    `f←(1‑t) f + t g`.  The largest `t∈(0,1]` for which the new
    maximal convolution does not exceed `M` is found by a tiny line
    search; this guarantees a monotone decrease of the objective.
    The step is repeated until the time budget is exhausted (or no
    further improvement is possible).

3️⃣  **Polishing** – a few cheap hill‑climbing moves transfer a small
    fraction of mass from indices that contribute most to the current
    peak toward indices that contribute least.  This often squeezes
    the bound a little further.

All operations stay on the probability simplex (after the
normalisation to √(2 n) the simplex becomes a sphere of radius 1 in the
evaluation metric).  The program never exceeds the 1000 s limit and
returns the best sequence it found.
"""

# Convolution helpers (direct O(n²) is fine for n≈1000)

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    return np.convolve(a, a)


def _objective_value(a: np.ndarray) -> float:
    """Exact objective used by the evaluation harness."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (used only as a
# safety net – we keep the sum exactly √(2 n) elsewhere)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x ≥ 0, Σx = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------

    """

    The returned vector sums to 1 (later we will renormalise it to √(2 n)).
    """
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

        _try(i1 ** (-p) + (n - i) ** (-p))

    for a_param in np.linspace(0.2, 6.0, 7):
        for b_param in np.linspace(0.2, 6.0, 7):

        if k == 0 or 2 * k >= n:

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear programme: maximise total mass under convolution bound
# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float):
    """
    Solve
        maximise   Σ g_j
        subject to  f ∗ g ≤ M  (2n‑1 inequalities)
                    g ≥ 0
    Returns the optimal g (may be None if the solver fails).
    """
    n = f.shape[0]
    # Build the Toeplitz matrix that implements the linear convolution f∗g
    # First column: f followed by zeros of length n‑1
    col = np.concatenate([f, np.zeros(n - 1)])
    # First row: f[0] followed by zeros (size n)
    row = np.concatenate([f[:1], np.zeros(n - 1)])
    A = toeplitz(col, row)                     # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linprog minimises; to maximise Σg we minimise –Σg
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})
    if res.success:
        # Clip tiny negatives caused by numeric noise
        g = np.maximum(res.x, 0.0)
        return g
    else:
        return None


# ----------------------------------------------------------------------
# Normalise a vector to a given L1‑norm (keeps non‑negativity)
# ----------------------------------------------------------------------
def _normalize(a: np.ndarray, target_sum: float) -> np.ndarray:
    s = a.sum()
    if s == 0:
        # avoid division by zero – fall back to uniform
        return np.full_like(a, target_sum / a.shape[0])
    return a * (target_sum / s)


# ----------------------------------------------------------------------
# Simple hill‑climbing step (used for a final polish)
# ----------------------------------------------------------------------
def _hillclimb(a: np.ndarray, target_sum: float, time_deadline: float):
    """
    Greedy transfer of a small fraction of mass from indices that
    contribute most to the current peak towards those that contribute
    least.  Stops when time runs out or no improvement is observed.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_M = _full_convolution(a).max()

    while time.time() < time_deadline:
        conv = _full_convolution(best_a)
        M = conv.max()
        # early exit if we cannot improve further
        if M <= best_M - 1e-12:
            best_M = M
        else:
            # no progress – break
            break

        # active set (indices whose convolution is almost maximal)
        eps = M * 5e-7
        active = np.where(conv >= M - eps)[0]
        if active.size == 0:
            break

        # compute the weight w_i = Σ_{k∈active} a_{k‑i}
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += best_a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or best_a[src] < 1e-12:
            break

        delta = best_a[src] * 0.07   # move 7 % of the source mass
        cand = best_a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _normalize(cand, target_sum)

        new_M = _full_convolution(cand).max()
        if new_M < best_M - 1e-12:
            best_a = cand
            best_M = new_M
            print(f"[Hill] improved max = {best_M:.6f}")
        else:
            # no improvement – stop
            break

    return best_a

    try:

    except Exception as exc:  # pragma: no cover
