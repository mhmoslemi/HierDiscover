"""
Propose a low‑value step‑function using a multi‑start Nesterov‑projected
gradient descent with simple line‑search.

The algorithm works in three phases (all fitting inside the 1000 s budget)

1️⃣  **Initial candidates** – generate a diverse pool of Beta‑shaped vectors,
    a few Dirichlet draws and (if available) the previously best solution.
    All candidates are normalised to sum = 1.

2️⃣  **Refinement** – keep the *k* best candidates (default 5) and run a
    projected Nesterov gradient descent on each of them.
    In every iteration we
      • compute the full autocorrelation via FFT,
      • locate the lag `k_max` where the maximum `max_b` occurs,
      • form the sub‑gradient `grad_j = 2·a_{k_max‑j}` (zero outside the support),
      • perform a short back‑tracking line‑search on the step size,
      • apply Nesterov momentum,
      • project back onto the probability simplex.
    The step‑size starts at 0.5 and decays geometrically; momentum
    coefficient is 0.9 (decreasing slowly towards 0).

3️⃣  **Finalisation** – the best sequence seen during the whole run is
    evaluated with the supplied ``evaluate_sequence`` routine and returned.

The code is pure NumPy, contains no hidden closures, and prints occasional
progress information.  It respects the time‑budget by reserving a small
margin for the final call to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of ``x`` with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # size = next power of two ≥ out_len for an efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already satisfies ``sum(a)=1``:
    f(a) = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    Σx = 1,  x ≥ 0.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped probability vector:
        a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},   i = 1,…,n .
    If the parameters give an all‑zero vector (underflow), ``None`` is returned.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    # Guard against overflow/underflow
    with np.errstate(divide='ignore', invalid='ignore'):
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    if np.any(np.isnan(a)) or np.any(np.isinf(a)):
        return None
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv(a,a))`` w.r.t. ``a``.
    For the index `k_max` where the maximum is attained,
    the gradient component j is 2·a_{k_max‑j} for feasible indices.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    # Indices i that contribute to entry k_max: i ∈ [max(0, k_max-(n-1)), min(n-1, k_max)]
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    idx_i = np.arange(i_min, i_max + 1, dtype=int)
    idx_j = k_max - idx_i
    # grad[j] accumulates 2 * a[i] for each pair (i, j)
    grad[idx_j] = 2.0 * a[idx_i]
    return grad


def _line_search(a: np.ndarray, grad: np.ndarray, cur_val: float,
                 step_init: float, min_step: float,
                 max_iter: int = 10) -> tuple[np.ndarray, float, float]:
    """
    Simple back‑tracking line‑search on the step size.
    Returns the new point, its objective value and the step finally used.
    """
    step = step_init
    for _ in range(max_iter):
        cand = a - step * grad
        cand = _project_simplex(cand)
        val = _objective(cand)
        if val < cur_val:
            return cand, val, step
        step *= 0.5
        if step < min_step:
            break
    # No improvement – stay at the current point
    return a, cur_val, 0.0

        print("[fallback] uniform start, val={:.6f}".format(best_val))

    # Keep only the top‑k most promising candidates for intensive refinement
    k_keep = 5
    # sort by objective value (ascending)
    sorted_idx = np.argsort([_objective(c) for c in candidates])[:k_keep]
    keepers = [candidates[i].copy() for i in sorted_idx]
    print("[select] retained {} candidates for refinement".format(len(keepers)))

    # ------------------------------------------------------------------
    # 1️⃣  Refine each keeper with projected Nesterov SGD + line‑search
    # ------------------------------------------------------------------
    # Parameters for the inner optimisation
    init_step = 0.5          # initial step size (will be decayed)
    min_step = 1e-6
    momentum_beta = 0.9
    decay_factor = 0.998     # geometric decay per iteration

    # Global best across all refinements
    global_best_val = best_val
    global_best_seq = best_seq.copy()

    for cand_id, a in enumerate(keepers):
        vel = np.zeros_like(a)           # momentum buffer
        step = init_step
        iteration = 0
        while True:
            now = time.time()
            if now >= deadline - 30.0:   # reserve ~30 s for final evaluation
                break

            # compute current objective and max‑lag
            conv_full = _full_convolution(a)
            max_b = conv_full.max()
            cur_val = 2.0 * n * max_b

            # record best found so far
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print("[refine {}] new global best {:.6f} (iter {})".format(
                    cand_id, global_best_val, iteration))

                # optional early stop if we beat a target
                if global_best_val <= 1.503:
                    print("[refine {}] target reached, exiting early".format(cand_id))
                    deadline = time.time() + 0.2   # break outer loops soon
            # compute sub‑gradient at the maximal lag
            k_max = int(np.argmax(conv_full))
            grad = _subgradient_max_conv(a, k_max)

            # Nesterov‑style momentum update
            # y = a + momentum_beta * vel  (look‑ahead point)
            y = a + momentum_beta * vel
            # gradient evaluated at y
            conv_y = _full_convolution(y)
            k_max_y = int(np.argmax(conv_y))
            grad_y = _subgradient_max_conv(y, k_max_y)

            # line‑search from the look‑ahead point
            cand, cand_val, used_step = _line_search(
                y, grad_y, cur_val, step, min_step)

            # Update momentum and position
            new_vel = momentum_beta * vel - used_step * grad_y
            a = cand
            vel = new_vel

            # decay step size
            step *= decay_factor
            iteration += 1

    # ------------------------------------------------------------------
    # 2️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
        print("[final] evaluate_sequence returned {:.6f}".format(final_score))
    except Exception as exc:                 # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = np.inf

    # Return the best sequence we have (even if final_score is inf)
    return global_best_seq.tolist()
