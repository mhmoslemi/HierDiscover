"""
Hybrid optimizer for the step‑function problem.

1.  A broad parametric sweep (exponential, power‑law, beta, arcsine,
    Gaussian, triangular, random Dirichlet with small concentration) is
    used to obtain a strong seed.
2.  A projected sub‑gradient descent on the exact objective
    (`2·n·max(a*a)`) is performed.  The sub‑gradient is computed from the
    set of convolution indices attaining the current maximum and the
    iterate is projected onto the probability simplex after each step.
    Simple step‑size decay and occasional random perturbations allow the
    search to escape shallow plateaus.
3.  A tiny convex combination with the uniform distribution finishes the
    polish.

The routine respects the required `propose_candidate` signature and
returns a list of non‑negative floats (Σ≈1) ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Exact linear convolution a*a using FFT (full length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    # Numerical noise may generate tiny negatives; clip for safety.
    conv = np.maximum(conv, 0.0)
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector with Σ a = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(v.shape[0]) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidate(n: int):
    """Explore a rich collection of parametric families and random draws."""
    rng = np.random.default_rng(123456)
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                           # 1 … n

    # 1. Exponential tails (decreasing / symmetric)
