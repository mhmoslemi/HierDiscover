def deterministic_refine_improved(a0, n, t_start, t_deadline, best_val):
    a = a0.copy()
    b = full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_val_local = cur_val
    best_a_local = a.copy()
    while time.time() < t_deadline:
        k_max = int(np.argmax(b))
        # Compute contributions to k_max
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        i_range = np.arange(start_i, end_i + 1)
        contrib = a[i_range] * a[k_max - i_range]
        i = int(i_range[np.argmax(contrib)])
        if a[i] <= 0:
            break
        # Find j candidate with minimal partner value
        partner = k_max - np.arange(n)
        partner_vals = np.where((partner >= 0) & (partner < n), a[partner], 0.0)
        # Exclude i
        partner_vals[i] = np.inf  # don't pick same index
        # Select K candidates with smallest partner_vals
        K = 5
        cand_idxs = np.argpartition(partner_vals, K)[:K]
        # compute second max
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        # compute a_i_partner
        a_i_partner = a[k_max - i] if 0 <= k_max - i < n else 0.0
        best_delta = 0.0
        best_j = None
        for j in cand_idxs:
            if j == i:
                continue
            a_j_partner = a[k_max - j] if 0 <= k_max - j < n else 0.0
            diff = a_i_partner - a_j_partner
            if diff <= 1e-12:
                continue
            delta_opt = (b[k_max] - second_max) / (2.0 * diff + 1e-12)
            delta = min(delta_opt * 0.99, a[i])
            if delta > best_delta:
                best_delta = delta
                best_j = j
        if best_j is None or best_delta <= 1e-12:
            break
        # apply incremental update using best_j, best_delta
        j = best_j
        delta = best_delta
        # update b using incremental as earlier
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = b.shape[0]
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta
        a[i] -= delta
        a[j] += delta
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val_local - 1e-14:
            best_val_local, best_a_local = cur_val, a.copy()
            if best_val_local <= 1.503:
                break
    return best_a_local, best_val_local

def simulated_annealing_improved(a0, n, start, deadline, best_a, best_val, rng):
    a = a0.copy()
    b = full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = a.copy()
    best_local_val = cur_val
    T0 = cur_val * 0.02   # initial temperature 2% relative
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0
    while time.time() < deadline:
        # compute progress fraction
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)
        max_frac = 0.3 * (1.0 - prog) + 0.001 * prog

        k_max = int(np.argmax(b))

        # source selection as before
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        i_range = np.arange(start_i, end_i+1)
        contrib = a[i_range] * a[k_max - i_range]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(i_range, p=probs_src)
        else:
            i = int(np.argmax(a))

        # target selection based on minimal partner
        partner = k_max - np.arange(n)
        partner_vals = np.where((partner >=0) & (partner < n), a[partner], 0.0)
        # mask out i for target
        partner_vals[i] = np.inf
        # we also can choose randomly with small probability
        if rng.random() < 0.1:
            # random target different from i
            j = rng.integers(0, n-1)
            while j == i:
                j = rng.integers(0, n-1)
        else:
            # choose candidate with smallest partner values
            j = int(np.argmin(partner_vals))

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # incremental update
        b_old = b.copy()
        a_old_i = a_i
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0 and rng.random() < np.exp(-(new_val - cur_val) / temperature):
                accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA2 {it:7d}] new best = {best_local_val:.6f}")
        else:
            b = b_old
            a[i] = a_old_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb: add small noise and renormalize
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA2 {it:7d}] perturb after stagnation")
        it += 1

    return best_local_a, best_local_val

The algorithm works entirely within the allowed 1000 s budget:

1. **Initialisation** – reuse the previously found best vector (if any) and
   re‑scale it to a probability simplex (sum = 1).  If unavailable a tiny
   handcrafted family is used.

2. **Adam with a smooth max surrogate** – the objective `max(conv(a,a))`
   is approximated by a log‑sum‑exp; ADAM optimisation on the soft‑max
   (log‑parameter) representation is performed while the smoothing
   parameter is annealed to near‑zero.

3. **Deterministic greedy refinement** – repeatedly moves a small amount
   of mass from the index that contributes most to the current maximal
   convolution entry to a target whose *partner* in that entry is the
   smallest.  The analytically optimal transfer size (to bring the peak
   down to the second‑largest value) is taken, guaranteeing a strict
   reduction whenever possible.

4. **Simulated‑annealing hill‑climbing** – a stochastic counterpart of the
   greedy step that explores a larger neighbourhood.  The temperature
   and move size are linearly cooled; moves are accepted according to the
   Metropolis rule.

5. **Uniform mixing (post‑processing)** – a tiny convex combination with
   the uniform distribution is tried; it often smooths remaining spikes.

All vectors are kept non‑negative and normalised to sum = 1, so the
evaluation function `evaluate_sequence` returns `2·n·max(conv)` exactly.
The code is deliberately written with no closures, only top‑level
functions, and respects the 1‑CPU restriction.

The routine returns a plain Python list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple objective utilities
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric: a_i = a_{n-1-i}."""
    a = (a + a[::-1]) * 0.5
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase A – Adam on a log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimizer on a smooth approximation of max(conv)."""
    n = a0.shape[0]
    # log‑softmax parametrisation (logits w)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.004
    eps_start, eps_end = 0.2, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 40000

    it = 0
    while time.time() < deadline:
        # current point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:6d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # fraction of elapsed time in this phase
        prog = (time.time() - start) / (deadline - start)
        # schedules
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   d/da 2·n·Σ p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 3·n‑2
        # needed slice corresponds to indices n‑1 … 2n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.006, size=n)

        # restart if stagnation persists
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:6d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic greedy refinement (analytical delta)
# ----------------------------------------------------------------------
def _deterministic_refine(a0: np.ndarray, start: float, deadline: float,
                          best_val: float, target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move mass from the worst contributor to the index whose partner is minimal."""
    n = a0.shape[0]
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_val = cur_val
    best_local_a = a.copy()

    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # ------- contributions to the maximal entry ----------
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        i_range = np.arange(start_i, end_i + 1)
        contrib = a[i_range] * a[k_max - i_range]
        i = int(i_range[np.argmax(contrib)])

        if a[i] <= 0:
            break

        # ------- partner values for every possible target j ----
        partner_idx = k_max - np.arange(n)
        partner_vals = np.where(
            (partner_idx >= 0) & (partner_idx < n), a[partner_idx], 0.0
        )
        partner_vals[i] = np.inf                     # do not target the source

        # pick a few candidates with smallest partner values
        K = 7
        cand_js = np.argpartition(partner_vals, K)[:K]

        # second largest convolution entry (excluding the current max)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()

        a_i_partner = a[k_max - i] if 0 <= k_max - i < n else 0.0

        best_delta = 0.0
        best_j = None

        for j in cand_js:
            if j == i:
                continue
            a_j_partner = a[k_max - j] if 0 <= k_max - j < n else 0.0
            diff = a_i_partner - a_j_partner
            if diff <= 1e-12:
                continue
            delta_opt = (b[k_max] - second_max) / (2.0 * diff + 1e-15)
            delta = min(delta_opt * 0.99, a[i])
            if delta > best_delta:
                best_delta = delta
                best_j = j

        if best_j is None or best_delta <= 1e-12:
            break

        j = best_j
        delta = best_delta

        # ---- incremental updates of convolution ------------
        # note: a is still the *old* vector while we use it for the update
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = b.shape[0]
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            print(f"[Refine {it:6d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break

        it += 1
        if it % 1500 == 0:
            # occasional symmetrisation to keep the shape balanced
            a = _symmetrize(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase C – Simulated‑annealing hill‑climbing (partner‑based moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         rng: np.random.Generator,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """SA that moves mass from the dominant contributor towards an index whose partner is small."""
    n = a0.shape[0]
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = a.copy()
    best_local_val = cur_val

    # initial temperature = 2 % of starting value
    T0 = cur_val * 0.02
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)               # linear cooling
        max_frac = 0.30 * (1.0 - prog) + 0.001 * prog

        k_max = int(np.argmax(b))

        # ----- source i -------------------------------------------------
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        i_range = np.arange(start_i, end_i + 1)
        contrib = a[i_range] * a[k_max - i_range]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = int(rng.choice(i_range, p=probs_src))
        else:
            i = int(np.argmax(a))

        # ----- target j (partner‑min) -----------------------------------
        partner_idx = k_max - np.arange(n)
        partner_vals = np.where(
            (partner_idx >= 0) & (partner_idx < n), a[partner_idx], 0.0
        )
        # occasional randomisation
        if rng.random() < 0.07:
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            # smallest partner value (excluding the source)
            partner_vals[i] = np.inf
            j = int(np.argmin(partner_vals))

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----- incremental update ---------------------------------------
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = b.shape[0]
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0 and rng.random() < np.exp(-(new_val - cur_val) / temperature):
                accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA {it:6d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            # revert
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb current state lightly and continue
            print(f"[SA {it:6d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.012, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (often yields a marginal improvement)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a little uniform mass."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a step‑function (non‑negative vector) under the 1000 s budget.
    Returns a list of floats that (approximately) minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.45

    # problem size – 9000 is a good trade‑off between expressiveness & speed
    n = 9000

    # ------------------------------------------------------------------
    # Initialise – reuse previous best if present, otherwise use a handcrafted start
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the surrounding environment
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 /= a0.sum()
        else:
            raise ValueError
    except Exception:
        # fallback: a simple arcsine‑like shape (heavy at the ends) + normalisation
        x = (np.arange(n) + 0.5) / n
        a0 = 1.0 / np.sqrt(np.clip(x * (1 - x), 1e-12, None))
        a0 /= a0.sum()

    # enforce symmetry from the start (helps the optimisation)
    a0 = _symmetrize(a0)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam (≈25 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.25 * (deadline - start_time)
    best_a, best_val = _adam_phase(best_a, start_time, phase_a_deadline,
                                   best_a, best_val, target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase B – Deterministic greedy refinement (≈25 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.50 * (deadline - start_time)
    best_a, best_val = _deterministic_refine(best_a, phase_a_deadline,
                                             phase_b_deadline,
                                             best_val, target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase C – Simulated‑annealing hill climbing (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _simulated_annealing(best_a, phase_b_deadline,
                                            phase_c_deadline,
                                            best_a, best_val, rng,
                                            target_val=TARGET)

    # ------------------------------------------------------------------
    # Final tiny uniform mixing (the usual post‑processing)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
