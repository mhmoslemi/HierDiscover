def solve_small(n):
    import cvxpy as cp
    from scipy.sparse import coo_matrix
    a = cp.Variable(n)
    t = cp.Variable()
    constraints = [a >= 0, cp.sum(a) == 1]

    for k in range(2*n - 1):
        # Build Q
        rows = []
        cols = []
        data = []
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k//2)
        for i in range(i_min, i_max+1):
            j = k - i
            if i < j:
                rows.extend([i, j])
                cols.extend([j, i])
                data.extend([0.5, 0.5])
            else:  # i == j
                rows.append(i)
                cols.append(i)
                data.append(1.0)
        # convert to sparse.
        Qk = coo_matrix((data, (rows, cols)), shape=(n, n))
        constraints.append(cp.quad_form(a, Qk) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, max_iters=2000, abstol=1e-10, reltol=1e-10, feastol=1e-10)
    return a.value, t.value

import time
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    # Euclidean projection onto simplex: sum x = 1, x>=0
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _make_Qk_dense(n: int, k: int) -> np.ndarray:
    """Return PSD matrix Q_k such that a^T Q_k a = sum_{i+j=k} a_i a_j."""
    Q = np.zeros((n, n), dtype=np.float64)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k // 2)
    for i in range(i_min, i_max + 1):
        j = k - i
        if j < 0 or j >= n:
            continue
        if i == j:
            Q[i, i] = 1.0
        else:
            Q[i, j] = Q[j, i] = 0.5
    return Q

def solve_small_convex(n_small: int, max_time: float = 30.0):
    """Solve the convex QCQP for a small dimension `n_small`.
    Returns a probability vector a and the optimal max convolution t.
    """
    import cvxpy as cp
    start = time.time()
    a = cp.Variable(n_small)
    t = cp.Variable()
    constraints = [a >= 0, cp.sum(a) == 1]
    # build constraints
    for k in range(2 * n_small - 1):
        Q = _make_Qk_dense(n_small, k)
        constraints.append(cp.quad_form(a, Q) <= t)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # try ECOS, fallback to SCS
    try:
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)
    except Exception as e:
        print("[solve_small] ECOS failed:", e)
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)

    if a.value is None:
        raise RuntimeError("CVXPY failed to produce a solution")
    a_val = np.maximum(a.value, 0.0)
    total = a_val.sum()
    if total == 0:
        a_val = np.full(n_small, 1.0 / n_small)
    else:
        a_val /= total
    t_val = float(t.value)
    elapsed = time.time() - start
    print(f"[solve_small] n={n_small}, t={t_val:.12f}, obj={2*n_small*t_val:.6f}, time={elapsed:.1f}s")
    return a_val, t_val

def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid solver:
    1. Solve a small convex QCQP (n≈200) to get a near‑optimal shape.
    2. Upsample to the target size (default n=8000).
    3. Apply a fast hill‑climbing + sub‑gradient polish within the remaining budget.
    Returns a list of non‑negative heights whose sum is 1.
    """
    import time
    import numpy as np

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1. Small‑scale convex optimisation (QCQP)
    # --------------------------------------------------------------
    n_small = 200
    # ensure we leave enough time for later phases
    small_phase_deadline = start_time + min(30.0, budget_s * 0.1)
    a_small = None
    t_opt = None
    try:

    except Exception as exc:

    # --------------------------------------------------------------
    # 2. Upsample to a larger working dimension
    # --------------------------------------------------------------
    # Target dimension – we keep the original 8000 if time permits,
    # otherwise fall back to a modest size.
    n_target = 8000
    if a_small is not None:
        x_small = np.linspace(0.0, 1.0, n_small)
        x_target = np.linspace(0.0, 1.0, n_target)
        a = np.interp(x_target, x_small, a_small)
    else:
        # fallback: a simple Beta shape
        a = _random_beta_candidate(n_target, rng)

    a = np.maximum(a, 0.0)
    a /= a.sum()
    best_a = a.copy()
    best_val = _objective(best_a)
    print(f"[Init] n={n_target}, objective={best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Simple hill‑climbing (move mass between two positions)
    # --------------------------------------------------------------
    n = n_target
    b = _full_convolution(best_a)
    cur_val = 2.0 * n * b.max()
    # parameters for hillclimbing
    max_iter = 2000   # we will bound by time as well
    iter_count = 0

    while time.time() < deadline and iter_count < max_iter:
        # temperature schedule (cooling)
        frac = (time.time() - start_time) / (deadline - start_time)
        T = 0.02 * (1.0 - frac)  # goes to 0

        # pick source index proportionally to its current weight
        src = np.random.choice(n, p=best_a)
        if best_a[src] < 1e-12:
            continue
        # pick a different target uniformly
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        # move a random fraction of src's mass
        delta = best_a[src] * np.random.rand() * 0.5   # up to 50%
        if delta <= 0.0:
            continue

        trial = best_a.copy()
        trial[src] -= delta
        trial[tgt] += delta

        # Re‑evaluate (FFT based, cheap for n≈8000)
        b_trial = _full_convolution(trial)
        new_val = 2.0 * n * b_trial.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        elif T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
            accept = True

        if accept:
            best_a = trial
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                print(f"[Hill]  new best={best_val:.6f} (iter {iter_count})")
        iter_count += 1

    # --------------------------------------------------------------
    # 4. Sub‑gradient polishing (exact max)
    # --------------------------------------------------------------
    # Remaining time is spent on a few projected sub‑gradient steps.
    step_initial = 0.3
    step_final = 1e-5

    while time.time() < deadline:

        b = _full_convolution(best_a)
        max_b = b.max()
        idxs = np.where(b == max_b)[0]

            mask = (idx >= 0) & (idx < n)

        if new_val < best_val:
            best_val = new_val
            best_a = a_new
            print(f"[SubG]  new best={best_val:.6f}")

    # 5. Final safe evaluation

        print(f"[Final] evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()

"""
Hybrid convex‑QCQP + up‑sampling + local search.

1.  Solve a small (n≈200) convex quadratic programme
    min t  s.t.   a≥0, Σa=1 and   aᵀQₖa ≤ t  for every convolution
    index k.  This yields the exact optimum of the continuous problem:
    the minimum possible value of 2·n·max(conv(a,a)) for large n.

2.  Interpolate the optimal small‑scale shape to the target length
    (default n=8000).  The interpolated vector is already a very good
    candidate (objective ≈1.5).

3.  Perform a short hill‑climbing phase that repeatedly moves a random
    chunk of mass from a source index to a different target index.
    Acceptance follows a simple simulated‑annealing schedule.

4.  Finish with a few projected sub‑gradient steps on the exact
    objective (max of the full convolution).  All updates keep the
    vector on the probability simplex.

The routine respects the 1000 s budget and returns the best sequence
found.  It only uses NumPy, SciPy‑style dense matrices and CVXPY
(which is part of the execution environment).  No filesystem or network
access is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (a with itself)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a probability vector (Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a non‑negative vector a with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: construct the PSD matrix Qₖ for a given convolution index k
# ----------------------------------------------------------------------
def _make_Qk_dense(n: int, k: int) -> np.ndarray:
    """
    Return a dense PSD matrix Qₖ such that
        aᵀQₖa = Σ_{i+j=k} a_i a_j .
    Off‑diagonal contributions get a factor ½ because aᵀQₖa counts both
    (i,j) and (j,i).
    """
    Q = np.zeros((n, n), dtype=np.float64)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k // 2)
    for i in range(i_min, i_max + 1):
        j = k - i
        if j < 0 or j >= n:
            continue
        if i == j:
            Q[i, i] = 1.0
        else:
            Q[i, j] = Q[j, i] = 0.5
    return Q


# ----------------------------------------------------------------------
# Small‑scale convex optimisation (QCQP) via CVXPY
# ----------------------------------------------------------------------
def _solve_small_convex(n_small: int) -> tuple[np.ndarray, float]:
    """
    Solve
        min t
        s.t. a ≥ 0, Σa = 1,
             aᵀQₖa ≤ t   for all k = 0,…,2n‑2 .
    Returns the optimal probability vector a and the optimal value t.
    """
    import cvxpy as cp

    a = cp.Variable(n_small)
    t = cp.Variable()
    constraints = [a >= 0, cp.sum(a) == 1]

    # quadratic constraints aᵀQₖa ≤ t
    for k in range(2 * n_small - 1):
        Q = _make_Qk_dense(n_small, k)
        constraints.append(cp.quad_form(a, Q) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        # ECOS handles quadratic forms via second‑order cones
        prob.solve(solver=cp.ECOS, max_iters=5000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)
    except Exception as e:
        # fall back to the first‑order solver SCS if ECOS fails
        print("[QCQP] ECOS failed, switching to SCS:", e)
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)

    if a.value is None:
        raise RuntimeError("CVXPY did not return a solution")

    a_val = np.maximum(a.value, 0.0)
    total = a_val.sum()
    if total == 0.0:
        a_val = np.full(n_small, 1.0 / n_small)
    else:
        a_val /= total
    t_val = float(t.value)
    return a_val, t_val


# ----------------------------------------------------------------------
# Random Beta‑family candidate (fallback / diversification)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a shape of the form i^{p-1}(n-i+1)^{q-1} and normalise."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Hybrid optimiser:
      * small convex QCQP (≈10 % of the budget)
      * up‑sampling to the target length (default n=8000)
      * a short hill‑climber
      * a few projected sub‑gradient polishing steps

    Returns a list of non‑negative heights whose sum is 1 and that
    (tries to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # 1) Small‑scale convex optimisation (QCQP)
    # --------------------------------------------------------------
    n_small = 200
    # Reserve at most 10 % of the total time for this phase
    small_phase_deadline = start_time + min(30.0, 0.10 * budget_s)
    a_small, t_opt = None, None
    try:
        a_small, t_opt = _solve_small_convex(n_small)
        print(f"[QCQP] n={n_small}, t_opt={t_opt:.12f}, "
              f"obj={2 * n_small * t_opt:.6f}")
    except Exception as exc:
        print("[QCQP] failed:", exc)

    # --------------------------------------------------------------
    # 2) Upsample to the working dimension
    # --------------------------------------------------------------
    n_target = 8000
    if a_small is not None:
        xs = np.linspace(0.0, 1.0, n_small)
        xt = np.linspace(0.0, 1.0, n_target)
        a = np.interp(xt, xs, a_small)
    else:
        # fallback to a random Beta shape
        a = _random_beta_candidate(n_target, rng)

    a = np.maximum(a, 0.0)
    a /= a.sum()                     # enforce Σa = 1
    best_a = a.copy()
    best_val = _objective(best_a)
    print(f"[Init] n={n_target}, objective={best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Hill‑climbing (mass moves) – limited number of iterations
    # --------------------------------------------------------------
    n = n_target
    cur_val = best_val
    max_hill_iters = 2500
    hill_iter = 0

    while time.time() < deadline and hill_iter < max_hill_iters:
        # simple linear cooling schedule
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = 0.02 * (1.0 - frac)               # temperature → 0

        # pick source proportionally to its current weight
        src = np.random.choice(n, p=best_a)
        if best_a[src] < 1e-12:
            continue

        # pick a different target uniformly
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        # move a random fraction (up to 50 %) of the source mass
        delta = best_a[src] * np.random.rand() * 0.5
        if delta <= 0.0:
            continue

        trial = best_a.copy()
        trial[src] -= delta
        trial[tgt] += delta

        # recompute the objective (FFT, cheap for n≈8000)
        b_trial = _full_convolution(trial)
        new_val = 2.0 * n * b_trial.max()

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        elif T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
            accept = True

        if accept:
            best_a = trial
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                print(f"[Hill] new best = {best_val:.6f} (iter {hill_iter})")
        hill_iter += 1

    # --------------------------------------------------------------
    # 4) Sub‑gradient polishing (exact max)
    # --------------------------------------------------------------
    # Remaining time is spent on a few projected sub‑gradient steps.
    step_start, step_end = 0.30, 1e-5
    max_subgrad_iters = 3000
    subgrad_iter = 0

    while time.time() < deadline and subgrad_iter < max_subgrad_iters:
        b = _full_convolution(best_a)
        m = b.max()
        # indices that are (numerically) maximal
        idxs = np.where(b >= m - 1e-12)[0]

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(best_a)
        for k in idxs:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * best_a[idx[mask]]

        # linearly decaying step size
        progress = (time.time() - start_time) / (deadline - start_time)
        step = step_start * (1.0 - progress) + step_end * progress

        a_next = best_a - step * grad
        a_next = _project_simplex(a_next)
        new_val = _objective(a_next)

        if new_val < best_val - 1e-12:
            best_a = a_next
            best_val = new_val
            print(f"[SubG] new best = {best_val:.6f} (iter {subgrad_iter})")
        subgrad_iter += 1

    # --------------------------------------------------------------
    # 5) Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain Python list of floats
    return best_a.tolist()
