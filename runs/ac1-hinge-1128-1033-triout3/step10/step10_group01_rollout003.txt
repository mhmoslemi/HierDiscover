def full_convolution_general(x: np.ndarray, y: np.ndarray):
    n1 = x.shape[0]; n2 = y.shape[0]; out_len = n1 + n2 - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8; lr=0.1
    alpha_start=1.0; alpha_end=200.0
    for t in range(1, max_iters+1):

            if best_val <= 1.503:

"""
Step‑function optimizer (≈2026).

The algorithm searches for a non‑negative sequence `a`
(minimising 2·n·max(a∗a) / (∑a)², see ``evaluate_sequence``) by
three complementary stages:

1️⃣  **Diverse seeding** – a handful of Dirichlet draws with
    different concentration parameters and an optional interpolation
    of the previously best solution (if the global variable
    ``height_sequence_1`` exists).  The best seed is kept.

2️⃣  **Smooth‑gradient Adam** – we replace the non‑smooth max
    by a log‑sum‑exp (soft‑max) with a temperature‐parameter `α`.
    For a given `α` the objective is smooth, its gradient can be
    written as a convolution and is evaluated with FFTs.
    An Adam optimiser (projected onto the probability simplex)
    is run while increasing `α` from 1 to 200, thus gradually
    morphing the smooth problem into the original one.

3️⃣  **Mass‑transfer hill‑climbing** – a refined local search
    (the “anneal” routine from the reference code) finishes the
    optimisation, exploiting the current maximal convolution entry
    to move a small amount of mass from a “source’’ index to a
    “target’’ index.  The schedule of the transferred fraction
    and of the Metropolis temperature is adaptive.

All phases respect the 1000 s time budget; the best vector found
is finally evaluated with ``evaluate_sequence`` and returned as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1. Convolution utilities (FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution_self(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (possibly different) 1‑D arrays."""
    n1, n2 = x.shape[0], y.shape[0]
    out_len = n1 + n2 - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# 2. Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto {x≥0, Σx = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 3. Soft‑max (log‑sum‑exp) helpers for the smooth objective
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> tuple[np.ndarray, float]:
    """
    Given the raw convolution ``conv`` (length 2·n‑1) and a temperature
    ``alpha`` returns the soft‑max weights (sum = 1) and the smooth
    approximation to ``max(conv)``:
        f_smooth = logsumexp(alpha·conv) / alpha.
    """
    maxc = conv.max()
    # Shift by the maximum to avoid overflow
    scaled = alpha * (conv - maxc)
    exps = np.exp(scaled)
    sumexp = exps.sum()
    w = exps / sumexp
    f_smooth = maxc + np.log(sumexp) / alpha
    return w, f_smooth


def _smooth_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. ``a``.
    ``w`` are the soft‑max weights (length 2·n‑1) associated with the
    current convolution.  The formula reduces to a convolution of
    ``a`` with the reversed weight vector.
    """
    n = a.shape[0]
    w_rev = w[::-1]                              # length 2·n‑1
    conv = _full_convolution(a, w_rev)          # length 3·n‑2
    # Extract the slice that corresponds to the gradient
    # (indices n‑1 … 2·n‑2, reversed order)
    grad_part = conv[n - 1 : 2 * n - 1][::-1]
    return 2.0 * grad_part


# ----------------------------------------------------------------------
# 4. Adam optimiser on the smooth surrogate (projected onto simplex)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray | None,
                   best_val: float,
                   max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate.
    The temperature ``α`` is increased exponentially from 1 to 200.
    Each iterate is projected back onto the probability simplex.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12                     # learning rate (tuned empirically)

    alpha_start, alpha_end = 1.0, 200.0

    for t in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break

        # temperature schedule (geometric)
        alpha = alpha_start * (alpha_end / alpha_start) ** ((t - 1) / max_iters)

        # 1) current self‑convolution (needed for the weights)
        conv = _full_convolution_self(a)

        # 2) soft‑max weights and smooth value (for diagnostics)
        w, _ = _softmax_weights(conv, alpha)

        # 3) gradient of the smooth surrogate
        grad = _smooth_gradient(a, w)

        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_simplex(a_new)

        a = a_new

        # evaluate the *true* (non‑smooth) objective occasionally
        if t % 200 == 0 or t == 1:
            conv_self = _full_convolution_self(a)
            cur_val = float(2.0 * n * conv_self.max())
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[Adam {t:6d}] new best = {best_val:.6f}")
                # early exit if we already beat the target
                if best_val <= 1.503:
                    break

    # Return the best we have seen (could be the very last iterate)
    return best_a if best_a is not None else a, best_val


# ----------------------------------------------------------------------
# 5. Mass‑transfer annealing hill‑climbing (slightly tuned)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Intensive local search that repeatedly transfers a small amount
    of mass from a “source’’ index (large contribution to the current
    maximal convolution entry) to a “target’’ index (small contribution).
    The transferred fraction and the Metropolis temperature decay
    smoothly with the elapsed wall‑time.
    """
    a = a0.copy()
    b = _full_convolution_self(a)                     # current convolution
    cur_val = float(2.0 * n * b.max())
    eps = 1e-14

    # schedules (geometric decay)
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)  # 0 → 1
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # identify current maximal convolution index
        k_max = int(np.argmax(b))

        # indices that actually contribute to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        # contribution of each coordinate to the maximal entry
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose a source index (biased towards large contributors)
        if total_contrib > 0 and np.random.rand() < 0.92:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # choose a target index (prefer indices that *do not* contribute much)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback: random distinct index
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # amount of mass to move
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # ----- fast update of the convolution (no FFT) -----
        # we only need the affected window of length n around each moved mass.
        b_new = b.copy()
        # contribution of moved mass with *all* entries of a
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        # self‑interaction terms (quadratic in delta)
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max)

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # occasional perturbation to escape stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution_self(a)
            cur_val = float(2.0 * n * b.max())
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – random perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 6. Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence`` under the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # 6.1  Choose problem size (a trade‑off between quality and speed)
    # ------------------------------------------------------------------
    n = 8000     # the size that gave the previous best result; can be
                 # changed to 12000 if time allows

    # ------------------------------------------------------------------
    # 6.2  Warm‑start from a previously found construction (if any)
    # ------------------------------------------------------------------
    a0 = None
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old_seq.ndim == 1 and old_seq.size > 0:
            old_len = old_seq.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old_seq)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 6.3  Diverse random seeding (Dirichlet with several concentrations)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    concentrations = [0.3, 0.5, 0.7, 1.0, 2.0, 5.0]
    seeds_per_conc = 8

    for conc in concentrations:
        for _ in range(seeds_per_conc):
            a = np.random.dirichlet(np.full(n, conc))
            conv = _full_convolution_self(a)
            val = float(2.0 * n * conv.max())
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[seed] conc={conc:.2f} → best={best_val:.6f}")

    # keep interpolated start if it is better
    if a0 is not None:
        conv0 = _full_convolution_self(a0)
        val0 = float(2.0 * n * conv0.max())
        if val0 < best_val:
            best_val, best_a = val0, a0.copy()
            print(f"[init] interpolated start beats seeds: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.4  Smooth‑gradient Adam optimisation (time‑boxed)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and best_a is not None:
        adam_deadline = time.time() + min(120.0, remaining * 0.6)
        best_a, best_val = _adam_optimize(best_a, n,
                                          deadline=adam_deadline,
                                          start_time=start_time,
                                          best_a=best_a,
                                          best_val=best_val,
                                          max_iters=80_000)
        print(f"[Adam] best after refinement = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.5  Final hill‑climbing (uses whatever time is left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5 and best_a is not None:
        best_a, best_val = _hillclimb_anneal(best_a, n,
                                             deadline=deadline,
                                             start_time=start_time,
                                             best_a=best_a,
                                             best_val=best_val,
                                             target_val=1.503)
        print(f"[Hill] final best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.6  Final secure evaluation (for the competition system)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
