def _smooth_gradient(a: np.ndarray, mu: float) -> np.ndarray:
    """Compute gradient of smooth approximation of max(b) where b = conv(a,a)."""
    n = a.shape[0]
    b = _full_convolution(a)  # length 2n-1
    # compute softmax weights
    # to improve numerical stability, subtract max before exponent.
    max_b = b.max()
    # shift to avoid overflow: use b_shift = b - max_b
    exp_terms = np.exp((b - max_b) / mu)  # use stable exponent
    w = exp_terms / exp_terms.sum()
    rev_w = w[::-1]
    # convolution a * rev_w
    L = len(b)
    out_len = len(a) + len(rev_w) - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(rev_w, size)
    conv2 = np.fft.irfft(fa * fw, size)[:out_len]
    # gradient entries: 2 * conv2[L-1 - i] for i in 0..n-1
    grad = 2.0 * conv2[L - 1 - np.arange(n)]
    return grad

    ∇f_μ = 2·(a ⋆ w) where w are the soft‑max weights.

    with a linearly decaying step‑size and an exponentially decaying
    smoothing parameter μ.  Occasional jitter helps to escape plateaus.
3.  Optionally restart the smooth descent from a few random starts if time

def _full_convolution(x: np.ndarray) -> np.ndarray:

    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv(a,a)) (assumes Σa = 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:

    Return a normalised Beta‑shaped density:
        a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},   i = 1,…,n .
    If the parameters lead to the zero vector ``None`` is returned.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0 or np.isnan(total):
        return None
    a /= total
    return a


def _smooth_gradient(a: np.ndarray, mu: float) -> np.ndarray:
    """
    Gradient of the smooth approximation μ·log Σ_k exp(b_k/μ) where
    b = conv(a,a).  Returns the vector ∇f_μ (same size as ``a``).
    """
    n = a.shape[0]
    # convolution b = a * a
    b = _full_convolution(a)          # length L = 2n-1
    L = b.shape[0]

    # soft‑max weights (shifted for stability)
    max_b = b.max()
    shifted = b - max_b
    # avoid division by zero
    if mu <= 0.0:
        mu = 1e-12
    w = np.exp(shifted / mu)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / L)
    else:
        w /= w_sum

    # reverse w to obtain the cross‑correlation kernel
    rev_w = w[::-1]

    # convolution a * rev_w  (full length n+L-1 = 3n-2)
    out_len = n + L - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(rev_w, size)
    conv2 = np.fft.irfft(fa * fw, size)[:out_len]

    # gradient_i = 2 * conv2[L-1-i]    (i = 0,…,n-1)
    idx = L - 1 - np.arange(n)
    grad = 2.0 * conv2[idx]
    return grad
