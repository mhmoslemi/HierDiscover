"""
Propose a high‐quality step function.

The algorithm works in three phases:

1️⃣  Initialise from the historic solution if available and search a
    rich family of analytic shapes (Beta, exponential, Dirichlet).
2️⃣  Run a *continuation* projected gradient descent.
    We minimise a smooth approximation of the objective

        f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    whose gradient can be computed by one extra convolution.
    α starts small (≈0.5) and is increased slowly up to ≈500,
    turning the smoothed problem into the original max‑norm.
    After each gradient step we project onto the probability simplex
    (Σa = 1, a_i ≥ 0).  Momentum and occasional jitter are added
    to escape plateaus.
3️⃣  The best sequence seen during the budget is returned and will be
    evaluated with the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution (length len(x)+len(y)-1) using FFT.
    Both inputs are real 1‑D arrays.
    """
    out_len = x.shape[0] + y.shape[0] - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence when Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a, a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    Σx = 1, x ≥ 0.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Beta‑shaped vector a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} (i=1…n)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(rate: float, n: int) -> np.ndarray:
    """Exponential decay shape a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Softmax with temperature 1/α (stable version)."""
    # prevent overflow/underflow
    b_max = b.max()
    exp_part = np.exp(alpha * (b - b_max))
    w = exp_part / exp_part.sum()
    return w


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – the original solution used 8000, we keep the same.
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Load a historic solution if it exists (helps with continuity)
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]

        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad search in low‑dimensional families
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val

    # ----- Beta family (grid + random log‑uniform) -----

            best_val, best_a = val, cand.copy()

        if time.time() >= deadline - 30.0:

    # exponential shapes (different decay rates)

    for _ in range(6):

        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[dirichlet] new best={best_val:.6f}")

    # fallback – uniform distribution (guaranteed feasible)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] uniform start, value=", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Continuation projected gradient descent (smooth → max)
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_seq = a.copy()
    best_score = best_val

    # step‑size parameters
    step_start = 0.3
    step_end = 1e-5
    jitter_prob = 0.001
    jitter_scale = 0.001

    # continuation parameters for the smoothing factor α
    alpha = 0.5
    alpha_max = 500.0
    alpha_inc = 1.02               # geometric increase each iteration

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # --- exact current objective (used for bookkeeping) ---
        b_full = _full_convolution(a, a)
        cur_val = 2.0 * n * b_full.max()

        if cur_val < best_score:
            best_score, best_seq = cur_val, a.copy()
            print(f"[sgd] new best={best_score:.6f} (iter {iteration})")
            # early exit if we beat the target comfortably
            if best_score <= 1.503:

        # --- smooth gradient based on softmax of the convolution ---
        w = _softmax_weights(b_full, alpha)          # shape (2n-1,)

        # gradient = 2 * (w * a_rev)  (correlation)
        a_rev = a[::-1]
        grad_full = _full_convolution(w, a_rev)       # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]        # slice length n

        # --- step size (linear decay) ---
        progress = (now - start_time) / (deadline - start_time)
        step = step_start * (1.0 - progress) + step_end * progress

        # projected gradient step
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        # occasional jitter to escape flat areas
        if np.random.rand() < jitter_prob:
            a_candidate += np.random.normal(scale=jitter_scale, size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate

        # increase smoothing parameter α slowly (continuation)
        alpha = min(alpha * alpha_inc, alpha_max)

        iteration += 1

    # 3️⃣  Final secure evaluation (outside the loop, for reporting)
    # ------------------------------------------------------------------

        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
