"""
Optimisation of a non‑negative step function via an active‑set
linear‑programming refinement of the classic alternating LP scheme.
We start from a good Beta‑shaped seed, then repeatedly:
 – compute the current autocorrelation (full convolution);
 – select the K largest lags (active set);
 – solve a linear programme that maximises the ℓ₁‑norm of a
   candidate vector g under the constraint (f⋆g)_∞ ≤ M;
 – take a small convex combination f←(1‑t)f+t·g and project onto the
   probability simplex;
 – keep the best vector seen.
The loop runs until the time budget expires (≈1000 s) or the target
value 1.503 is reached.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _objective(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence when Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σx = 1, x ≥ 0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Return a Beta‑shaped probability vector (i^{p‑1}(n‑i+1)^{q‑1})₁ₙ."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a

def _build_active_constraints(f: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """Construct A_ub for the constraints (f⋆g)[k] ≤ M, k∈ks."""
    n = f.shape[0]
    K = ks.shape[0]
    A = np.zeros((K, n), dtype=np.float64)
    for row_idx, k in enumerate(ks):
        # indices j where 0 ≤ k‑j < n  →  j ∈ [max(0,k‑n+1), min(k,n‑1)]
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        if j_start <= j_end:
            # the slice f[k‑j] for j=j_start…j_end
            j = np.arange(j_start, j_end + 1)
            A[row_idx, j_start:j_end + 1] = f[k - j]
    return A

        print("[init] interpolated previous solution")
    except Exception:
        pass

    best_val = np.inf

        best_a, best_val = a.copy(), val
        print(f"[init] initial value = {best_val:.6f}")

        cand = _beta_shape(p, q, n)
        if cand is None:
            continue

        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand] new best = {best_val:.6f} (p={p:.4g}, q={q:.4g})")
        if time.time() >= deadline - 30.0:
            break

    for _ in range(8):

    t_end = 0.005

    jitter_scale = 0.001

        if now >= deadline:

            if best_score <= 1.503:
                print("[lp] target reached, exiting")
                break

        K = min(active_K, conv.size)
        # np.argpartition gives unsorted; use -K for largest
        top_idx = np.argpartition(conv, -K)[-K:]

        A = _build_active_constraints(a, top_idx)
        b = np.full(K, cur_M, dtype=np.float64)

        # linear programme: maximise 1ᵀg  ⇔ minimise −1ᵀg
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

                conv_new = _full_convolution(a_candidate)
                if conv_new.max() <= cur_M * (1.0 + 1e-12):
                    a = a_candidate
                else:
                    # fallback: keep the old a, maybe reduce t next round
                    pass
        else:
            # LP failed – fall back to a tiny random jitter
            pass

        # occasional random jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            a = a + np.random.normal(scale=jitter_scale, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

        iteration += 1

    # 3️⃣  Final evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()

Optimisation of a non‑negative step function using an active‑set
linear‑programming refinement of the classic alternating‑LP scheme.

* Initialise a good β‑shaped seed (grid + random log‑uniform + Dirichlet).
* Repeatedly:
    – compute the full self‑convolution (FFT);
    – take the K largest lags as an *active set*;
    – solve   max Σ g_j   s.t.   (f⋆g)[k] ≤ M  (k in active set), g ≥ 0
      (a tiny LP with ≤30 constraints);
    – normalise the LP solution to a probability vector and mix it with the
      current iterate f by a small convex combination f←(1‑t)f+t·g;
    – project back onto the probability simplex.
* Keep the best vector seen.  The loop runs until the 1000 s budget expires
  or the target value 1.503 is reached.
The final sequence is returned as a plain list of floats.
"""

from scipy.optimize import linprog

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value returned by ``evaluate_sequence`` when Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
