def _gradient(a, w):
    # a: length n
    # w: length 2n-1 normalized weights
    # compute grad_i = 2 * sum_j a_j * w_{i+j}
    # compute with FFT:
    n = a.shape[0]
    w_rev = w[::-1]
    size = 1 << ((n + len(w_rev) - 2).bit_length())
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv_full = np.fft.irfft(fa * fw, size)[:3*n-2]  # length 3n-2
    segment = conv_full[n-1:2*n-1]  # length n
    grad = 2.0 * segment[::-1]
    return grad
```

"""
Non‑negative step‑function optimiser (soft‑max mirror descent).

The routine builds on the idea that, after fixing the L1 norm
∑a_i = 1, the evaluation score is proportional to the maximal
entry of the self‑convolution b = a * a.  Directly minimising the
max‑norm is hard because the function is non‑smooth, so we replace
the max by a smooth log‑sum‑exp approximation.
For a given temperature β the smooth objective is

    J_β(a) = (1/β)·log Σ_k exp(β·b_k),

and its Euclidean gradient can be written very compactly as

    ∇J_β(a) = 2·(a ★ w),

where w_k = exp(β·b_k)/Σ exp(β·b) and ★ denotes the cross‑correlation
(a * reversed(w)).  Using an entropy mirror step (multiplicative update)
keeps the iterates non‑negative and automatically enforces the simplex
constraint after normalisation.

The algorithm proceeds as follows

1. Initialise a from the best known construction (global variable
   ``height_sequence_1`` if present) or from a random Dirichlet seed.
2. Repeatedly
   * compute the full self‑convolution,
   * form the soft‑max weights w with a temperature β that is slowly
     increased (the approximation becomes tighter),
   * compute the gradient via a single FFT‑based cross‑correlation,
   * perform an exponentiated‑gradient (entropy mirror) update with a
     stepsize η that is annealed,
   * accept the candidate either if the true max‑norm improves or
     probabilistically (Metropolis) to escape shallow minima.
3. Occasionally inject a tiny Gaussian perturbation when progress stalls.
4. Keep the best vector seen so far and finally evaluate it with the
   provided ``evaluate_sequence`` function.

All heavy work is done with real FFTs, giving O(n log n) per iteration
(easily fits into the 1000 s budget for n = 8000).  The routine returns
the list of floats that yielded the lowest score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (length = len(x)+len(y)-1)
# ----------------------------------------------------------------------
def _full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) approximation of max(b)
# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇J_β(a) = 2·(a ★ w) where w is a probability vector of length 2n‑1.
    The cross‑correlation is performed via FFT.
    """
    n = a.shape[0]
    # reverse w for cross‑correlation
    w_rev = w[::-1]
    # length of the full convolution (a * w_rev) is 3n‑2
    full_len = 3 * n - 2
    size = 1 << (full_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv_full = np.fft.irfft(fa * fw, size)[:full_len]

    # segment conv_full[n‑1 : 2n‑1] (size n) corresponds to
    #   Σ_j a_j w_{i+j} for i = n‑1 … 0.
    segment = conv_full[n - 1: 2 * n - 1]
    grad = 2.0 * segment[::-1]            # reorder to i = 0 … n‑1
    return grad

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,

    Produce a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  Uses a soft‑max mirror‑descent scheme
    with annealed temperature and stepsize.
    """
    np.random.seed(seed)
    start_time = time.time()

    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8 000 works well for the time budget
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:

        # --------------------------------------------------------------
