def propose_candidate(seed=42,budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 12000   # chosen size

    # initialize a using Beta(0.5,0.5) 
    a = _beta_family(n, 0.5, 0.5)
    a = a / a.sum()

    # store best
    best_a = a.copy()
    best_val = _objective(best_a)
    # log

    # Adam parameters
    z = np.log(a + 1e-16)   # logit representation
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    iteration = 0
    stagnation = 0

    while time.time() < deadline:
        iteration+=1

        # compute convolution
        conv = _full_convolution(a)
        max_b = conv.max()
        obj = 2.0 * n * max_b

        # update best
        if obj < best_val - 1e-12:
            best_val = obj
            best_a = a.copy()
            stagnation = 0
            print(f"[iter {iteration}] new best = {best_val:.6f}")
        else:
            stagnation += 1

        # gradient computation using top L indices
        L = 40
        top_idx = np.argpartition(-conv, L-1)[:L]
        # weights proportional to conv values (softmax perhaps)
        top_vals = conv[top_idx]
        w = top_vals / top_vals.sum()   # shape (L,)

        # compute subgradient s
        s = np.zeros_like(a)
        for k, wk in zip(top_idx, w):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            # i range
            i = np.arange(low, high+1)
            s[i] += 2.0 * wk * a[k - i]

        # compute mu = a dot s
        mu = np.dot(a, s)

        # gradient in z
        g_z = a * (s - mu)

        # update Adam
        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * (g_z ** 2)

        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        # schedule step size alpha
        prog = (time.time() - start) / (deadline - start)
        alpha_start = 0.5
        alpha_end = 0.001

        # update z
        z = z - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # to avoid overflow
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)

# ----------------------------------------------------------------------
# Convolution helper (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised sequence (sum == 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for sum(a)=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Beta‑family generator (used for cheap initialisation)
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Return a normalised Beta‑shaped sequence of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a
