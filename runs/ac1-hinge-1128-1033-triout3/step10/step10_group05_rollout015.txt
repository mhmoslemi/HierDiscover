def _subgradient_descent(a, n, size, rng, deadline, start_time, best_a, best_val):
    cur = a.copy()
    cur_val = objective(cur)
    eta = 0.5 / n
    decay = 0.9995
    it = 0
    no_imp = 0
    while time.time() < deadline:
        # compute k_max
        b = _full_convolution(cur)
        max_val = b.max()
        k = np.argmax(b)
        grad = np.zeros_like(cur)
        # contributions
        idx = k - np.arange(n)
        mask = (idx >=0) & (idx < n)
        grad[mask] = 2*cur[idx[mask]]
        # step
        v = cur - eta * grad
        cur = _project_simplex(v)
        cur_val = 2 * n * max_val  # compute after projection? Actually may change max, compute anew
        # compute new val
        cur_val = objective(cur)
        if cur_val < best_val:
            best_val, best_a = cur_val, cur.copy()
            no_imp = 0
        else:
            no_imp += 1
        eta *= decay
        if no_imp > 200000:
            # perturb
            cur = best_a.copy()
            cur += rng.normal(scale=0.001, size=n)
            cur = np.maximum(cur, 0)
            cur /= cur.sum()
            eta = 0.5 / n
            no_imp = 0
        it += 1
    return best_a, best_val

def _mix_uniform(a, n):
    uniform = np.full(n, 1/n)
    best = a
    best_val = objective(a)
    for alpha in [0.001, 0.005, 0.01, 0.02, 0.05]:
        cand = (1-alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = objective(cand)
        if val < best_val:
            best_val, best = val, cand
    return best, best_val

"""
A fast SA + sub‑gradient optimizer for the min‑max self‑convolution problem.

The algorithm works on a modest size (default n=2000) so that many
iterations can be performed within the 1000 s budget.

1. Several hand‑crafted initial vectors (uniform, sine, arcsine,
   exponential) are generated.
2. A simulated‑annealing loop repeatedly moves a small amount of mass
   from a high‑mass index to a random target while accepting worse
   moves with a slowly decreasing temperature.  The current best vector
   over all restarts is kept.
3. A short projected sub‑gradient polishing phase reduces any remaining
   peak.
4. Finally a tiny mixture with the uniform distribution is tried –
   this often fixes a lingering maximal entry.

The code uses only NumPy, works on a single CPU and returns a plain
list of floats ready for ``evaluate_sequence``.
"""

from __future__ import annotations
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a*a (length = 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


def _objective(a: np.ndarray, n: int, size: int) -> float:
    """
    Exact objective:   2·n·max(conv(a,a)),   assuming sum(a)=1.
    """
    conv = _full_convolution(a, size)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of initial candidates (all summed to 1)
# ----------------------------------------------------------------------
def _make_initial_candidates(n: int, rng: np.random.Generator):
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # half‑wave sine (zero at the ends, peak in the middle)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    cand.append(a / a.sum())

    # arcsine‑like (heavy at the ends)
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    cand.append(a / a.sum())

    # symmetric exponential / “U‑shaped”
    for r in (0.9999, 0.9995, 0.998, 0.996):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        cand.append(a / a.sum())

    # a few random Dirichlet draws
    for _ in range(5):
        a = rng.dirichlet(np.ones(n))
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Simulated‑annealing optimiser (fast, many restarts)
# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    size: int,
    rng: np.random.Generator,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    SA that moves a small chunk of mass from a source index (chosen
    proportionally to its current weight) to a random target index.
    """
    a = a0.copy()
    cur_val = _objective(a, n, size)

    # keep the best ever seen globally
    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()

    # start temperature ~= 2‑5 % of the initial value
    T0 = cur_val * 0.04
    it = 0
    no_imp = 0
    IMP_LIMIT = 80_000

    while time.time() < deadline:
        # linear cooling schedule
        prog = (time.time() - (deadline - (deadline - time.time()))) / (deadline - start_time)
        temperature = max(T0 * (1.0 - prog), 1e-12)
        # source weight fraction, shrinks to ~0.001 towards the end
        max_frac = 0.12 * (1.0 - prog) + 0.001 * prog

        # ---- choose source index (biased towards heavy entries) ----
        i = rng.choice(n, p=a)          # a already sums to 1

        # ---- choose target index uniformly ----
        j = rng.integers(n)
        if i == j:
            continue

        # ----- propose a transfer -----
        delta = a[i] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val_new = _objective(a_new, n, size)

        # ----- Metropolis acceptance -----
        accepted = False
        if val_new < cur_val:
            accepted = True
        else:
            # exponential acceptance, may reject when temperature is tiny
            if rng.random() < np.exp(-(val_new - cur_val) / temperature):
                accepted = True

        if accepted:
            a = a_new
            cur_val = val_new
            no_imp = 0
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            no_imp += 1

        # occasional large perturbation if stuck for many steps
        if no_imp > IMP_LIMIT:
            # shake around the best seen so far
            a = best_a.copy()
            a += rng.normal(scale=0.005, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            cur_val = _objective(a, n, size)
            no_imp = 0

        # simple progress output
        if it % 10_000 == 0:
            print(f"[SA  {it:7d}] cur={cur_val:.6f} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    size: int,
    rng: np.random.Generator,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient step on the true max‑convolution objective."""
    a = a0.copy()
    cur_val = _objective(a, n, size)

    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()

    eta0 = 0.5 / n          # comparable to average entry size
    eta = eta0
    decay = 0.9997
    it = 0
    no_imp = 0
    IMP_LIMIT = 120_000

    while time.time() < deadline:
        conv = _full_convolution(a, size)
        cur_val = float(2.0 * n * conv.max())
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # index of the maximal convolution entry
        k_max = int(np.argmax(conv))

        # sub‑gradient of max entry: 2·a_{k_max-i} (if index valid)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # projected gradient step
        v = a - eta * grad
        a = _project_simplex(v)
        eta *= decay

        if no_imp > IMP_LIMIT:
            # shake & reset step size
            a = best_a.copy()
            a += rng.normal(scale=0.003, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            eta = eta0
            no_imp = 0

        if it % 5_000 == 0:
            print(f"[SGD  {it:7d}] cur={cur_val:.6f} best={best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (often yields a small extra drop)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int, size: int) -> tuple[np.ndarray, float]:
    """Convex combination with a little uniform mass."""
    best_a, best_val = a, _objective(a, n, size)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n, size)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a step‑function giving a low value of
    ``evaluate_sequence``.  The routine stays well within the 1000 s limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final ``evaluate_sequence`` call
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 2000
    # FFT size – next power of two ≥ 2·n‑1
    conv_size = 1 << ((2 * n - 1) - 1).bit_length()

    # ------------------------------------------------------------------
    # Initialise a pool of start points
    # ------------------------------------------------------------------
    init_pool = _make_initial_candidates(n, rng)

    # start from the best among the hand‑crafted pool
    best_a = init_pool[0]
    best_val = _objective(best_a, n, conv_size)
    for a in init_pool[1:]:
        val = _objective(a, n, conv_size)
        if val < best_val:
            best_a, best_val = a, val

    print(f"[init] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Repeated SA runs (each run gets a slice of the remaining budget)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:          # keep ~30 s for polishing
        # allocate a short chunk (≈10 s) to a fresh SA run
        chunk_deadline = min(deadline - 30.0, time.time() + 10.0)

        # pick a random start from the pool (or jitter the current best)
        if rng.random() < 0.3:
            a0 = best_a.copy()
            a0 += rng.normal(scale=0.004, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = init_pool[rng.integers(len(init_pool))].copy()

        best_a, best_val = _simulated_annealing(
            a0,
            n,
            conv_size,
            rng,
            chunk_deadline,
            best_a,
            best_val,
        )
        print(f"[chunk] best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Polishing via projected sub‑gradient descent (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(
            best_a,
            n,
            conv_size,
            rng,
            deadline,
            best_a,
            best_val,
        )
        print(f"[polish] after SGD = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Tiny uniform mixing (may give a marginal improvement)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n, conv_size)
    print(f"[final mix] value = {best_val:.6f}")

    # final call to the provided evaluator (for the harness)
    try:
        score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
