def _full_convolution(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _initial_candidate(n, rng):
    best_a = np.full(n, 1.0/n)
    best_val = _objective_value(best_a)

    def try_candidate(x):
        nonlocal best_a, best_val
        a = np.asarray(x, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # uniform
    try_candidate(np.ones(n))

    # exponential families
    for r in (0.9999, 0.9995, 0.998, 0.996, 0.995, 0.99, 0.97):
        i = np.arange(n, dtype=float)
        a = r**i + r**(n-1-i)
        try_candidate(a)

    # half-wave sine
    a = np.sin(np.pi * (np.arange(1, n+1) / (n+1)))
    try_candidate(a)

    # arcsine-like
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(np.clip(x * (1 - x), eps, None))
    try_candidate(a)

    # power law
    for p in (0.5, 0.75, 1.0, 1.25, 1.5, 1.8):
        i = np.arange(n, dtype=np.float64) + 1
        a = (i**(-p) + (n - i + 1)**(-p))
        try_candidate(a)

    # Gaussian
    mid = (n - 1) / 2.0
    for frac in (0.12, 0.15, 0.18, 0.22, 0.25):
        sigma = n * frac
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((i - mid) / sigma)**2)
        try_candidate(a)

    # Beta-shaped
    for alpha, beta in ((0.5,0.5),(0.8,0.8),(1.2,1.2),(1.5,1.5)):
        i = np.arange(1, n+1)
        a = (i / n) ** (alpha - 1) * (1 - i / n) ** (beta - 1)
        try_candidate(a)

    # cosine squared
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - mid) / (n - 1)) ** 2
    try_candidate(a)

    # random Dirichlet draws
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)

    return best_a, best_val

def _adam_phase(a0, n, start, deadline, best_a, best_val, target_val=1.503):
   w = np.log(a0 + 1e-12)
   beta1, beta2 = 0.9, 0.999
   lr_start, lr_end = 0.3, 0.01
   eps_start, eps_end = 0.5, 1e-12
   m = np.zeros_like(w)
   v = np.zeros_like(w)
   t = 0
   no_improve = 0
   IMPROVE_LIMIT = 20000
   it = 0
   while time.time() < deadline:
       w_shift = w - np.max(w)
       a = np.exp(w_shift)
       a /= a.sum()
       cur_val = _objective_value(a)
       if cur_val < best_val - 1e-14:
           best_val, best_a = cur_val, a.copy()
           no_improve = 0
           print(f"[Adam {it:7d}] new best = {best_val:.6f}")
           if best_val <= target_val:
               break
       else:
           no_improve += 1
       # schedule
       prog = (time.time() - start) / (deadline - start)
       eps = eps_start * (eps_end / eps_start) ** prog
       lr = lr_start * (lr_end / lr_start) ** prog
       # compute smooth max using log-sum-exp
       b_full = _full_convolution(a)
       max_b = b_full.max()
       shifted = (b_full - max_b) / eps
       weights = np.exp(shifted)
       p = weights / weights.sum()
       # gradient of smooth objective w.r.t a: 4 n conv(a, p_rev)
       p_rev = p[::-1]
       conv_ab = _conv_fft(a, p_rev)  # length n + n -1? But using _conv_fft yields length 2n-1
       # segment corresponds to indices n-1 : 2n-1 (since conv_ab length =2n-1)
       segment = conv_ab[n-1: 2*n-1]
       grad_a = 4.0 * n * segment[::-1]
       # chain rule through softmax: grad_w = a * (grad_a - np.dot(grad_a, a))
       dot = np.dot(grad_a, a)
       grad_w = a * (grad_a - dot)
       # Adam update
       t += 1
       m = beta1 * m + (1.0 - beta1) * grad_w
       v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
       m_hat = m / (1.0 - beta1 ** t)
       v_hat = v / (1.0 - beta2 ** t)
       w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
       # occasional jitter
       if np.random.rand() < 0.005:
           w += np.random.normal(scale=0.01, size=n)
       if no_improve > IMPROVE_LIMIT:
           print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
           w = np.random.randn(n) * 0.2
           m.fill(0.0)
           v.fill(0.0)
           t = 0
           no_improve = 0
       it += 1
   return best_a, best_val

def _simulated_annealing_phase(a0, n, start, deadline, best_a, best_val, rng, target_val=1.503):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = a.copy()
    best_local_val = cur_val
    T_factor = 0.07  # initial temperature factor
    T0 = cur_val * T_factor
    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 300000
    no_improve = 0
    it = 0
    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        # Determine source index with high contribution to current maximum entry
        k_max = int(np.argmax(b))
        # Compute contributions of each i to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0 and rng.random() < 0.85:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))
        # Choose target j - prefer indices with low contribution to the max entry
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)
        if i == j:
            continue
        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue
        # incremental update
        b_old = b.copy()
        # update convolution for all positions impacted by moving mass delta from i to j
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        ii = i + i
        jj = j + j
        ij = i + j
        if jj < L:
            b[jj] += delta * delta
        if ii < L:
            b[ii] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta
        a[i] -= delta
        a[j] += delta
        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True
        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            # revert
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1
        if no_improve > IMPROVE_LIMIT:
            # shake
            print(f"[SA {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.02, size=n)  # larger noise
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
        it += 1
    # ensure best_local_a normalized
    best_local_a = best_local_a / best_local_a.sum()
    return best_local_a, best_local_val

def _subgradient_phase(a0, n, start, deadline, best_a, best_val, target_val=1.503):

    eta = eta0

    it = 0
    no_improve = 0

        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:

            no_improve += 1

        grad = np.zeros_like(a)

            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
        it += 1
    return best_local_a, best_local_val

def _deterministic_refine(a0, n, start, deadline, best_val):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_val = cur_val
    best_local_a = a.copy()
    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]
        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break
        # choose j that minimizes contribution a[k_max - j], tie break by smallest a
        # compute complement values for index j within bounds
        comp_vals = np.full(n, np.inf)
        mask_j = (k_max - np.arange(n) >= 0) & (k_max - np.arange(n) < n)
        comp_vals[mask_j] = a[k_max - np.arange(n)[mask_j]]
        # also consider small a values
        # pick index with minimal comp_vals, ensure not i
        sorted_candidates = np.argsort(comp_vals)
        for cand in sorted_candidates:
            if cand != i:
                j = int(cand)
                break
        a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
        if a_k_i <= a_k_j + 1e-14:
            break
        # second max conv entry
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max == -np.inf:
            second_max = 0.0
        denom = 2.0 * (a_k_i - a_k_j) + 1e-12
        delta_needed = (b[k_max] - second_max) / denom
        delta = min(delta_needed * 0.9, a[i])
        if delta <= 1e-13:
            break
        # update convolution incrementally
        b_old = b.copy()

        b[i:i + n] -= 2.0 * delta * a
        L = len(b)

        jj = j + j
        ij = i + j

            b[jj] += delta * delta
        if ii < L:
            b[ii] += delta * delta
        if ij < L:

"""
Propose a step‑function (non‑negative vector) minimizing
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm proceeds in stages:
  1. Generate many hand‑crafted and random candidates and keep the best.
  2. Adam on a smooth (log‑sum‑exp) surrogate of the max‑convolution.
  3. Aggressive simulated‑annealing hill‑climbing with large moves.
  4. Projected sub‑gradient descent for polishing.
  5. Deterministic greedy refinement that moves mass analytically.
  6. Fine mixing with a uniform distribution (grid search over α).
  7. Optional tiny random perturbations + a short refinement.

All phases respect the global 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _candidate_eval(vec: np.ndarray, n: int):
    """Normalise a candidate vector and return (normalised, objective)."""
    a = np.asarray(vec, dtype=np.float64)
    if a.shape != (n,):
        return None, None
    if np.any(a < 0):
        return None, None
    s = a.sum()
    if s <= 0:
        return None, None
    a = a / s
    val = _objective_value(a)
    return a, val


# ----------------------------------------------------------------------
# Phase 0 – generate many hand‑crafted families + random Dirichlet draws
# ----------------------------------------------------------------------
def _initial_candidate(n: int, rng: np.random.Generator):
    """Return a promising starting vector (already normalised)."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # uniform
    a, val = _candidate_eval(np.ones(n), n)
    if val is not None and val < best_val:
        best_val, best_a = val, a

    # exponential families (U‑shaped)
    for r in (0.9999, 0.9995, 0.998, 0.996, 0.995, 0.99, 0.97):
        i = np.arange(n, dtype=float)
        a_raw = np.power(r, i) + np.power(r, n - 1 - i)
        a, val = _candidate_eval(a_raw, n)
        if val is not None and val < best_val:
            best_val, best_a = val, a

    # half‑wave sine
    a_raw = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a, val = _candidate_eval(a_raw, n)
    if val is not None and val < best_val:
        best_val, best_a = val, a

    # arcsine‑like (heavy at ends)
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n
    a_raw = 1.0 / np.sqrt(np.clip(x * (1 - x), eps, None))
    a, val = _candidate_eval(a_raw, n)
    if val is not None and val < best_val:
        best_val, best_a = val, a

    # power‑law families
    for p in (0.5, 0.75, 1.0, 1.25, 1.5, 1.8):
        i = np.arange(1, n + 1, dtype=np.float64)
        a_raw = i ** (-p) + (n - i + 1) ** (-p)
        a, val = _candidate_eval(a_raw, n)
        if val is not None and val < best_val:
            best_val, best_a = val, a

    # Gaussian bells
    mid = (n - 1) / 2.0
    for frac in (0.12, 0.15, 0.18, 0.22, 0.25):
        sigma = n * frac
        i = np.arange(n, dtype=np.float64)
        a_raw = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        a, val = _candidate_eval(a_raw, n)
        if val is not None and val < best_val:
            best_val, best_a = val, a

    # Beta‑shaped (U‑ish and bell‑ish)
    for alpha, beta in ((0.5, 0.5), (0.8, 0.8), (1.2, 1.2), (1.5, 1.5)):
        i = np.arange(1, n + 1, dtype=np.float64)
        a_raw = (i / n) ** (alpha - 1) * (1 - i / n) ** (beta - 1)
        a, val = _candidate_eval(a_raw, n)
        if val is not None and val < best_val:
            best_val, best_a = val, a

    # cosine‑squared (peak in the centre)
    i = np.arange(n, dtype=np.float64)
    a_raw = np.cos(np.pi * (i - mid) / (n - 1)) ** 2
    a, val = _candidate_eval(a_raw, n)
    if val is not None and val < best_val:
        best_val, best_a = val, a

    # random Dirichlet draws
    for _ in range(20):
        a_raw = rng.dirichlet(np.ones(n))
        a, val = _candidate_eval(a_raw, n)
        if val is not None and val < best_val:
            best_val, best_a = val, a

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp approximation of max(conv)."""
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.01          # learning‑rate schedule
    eps_start, eps_end = 0.5, 1e-12       # smoothing schedule

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 20000
    it = 0

    while time.time() < deadline:
        # simplex point
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                     # soft‑max over positions

        # gradient of the smoothed objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                  # length = 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Aggressive SA that moves sizable mass from harmful to harmless indices."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    T0 = cur_val * 0.07               # initial temperature (7 % of start)
    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)               # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        k_max = int(np.argmax(b))

        # contributions of each i to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose a source index (mostly weighted by contribution)
        if total_contrib > 0 and rng.random() < 0.85:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # choose a target index that is light for the maximal entry
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        ii = i + i
        jj = j + j
        ij = i + j
        if jj < L:
            b[jj] += delta * delta
        if ii < L:
            b[ii] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            # revert
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            print(f"[SA {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    # normalise before returning (tiny numerical drift may appear)
    best_local_a = best_local_a / best_local_a.sum()
    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase C – Exact projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n          # initial step size ≈ half of average entry
    eta = eta0
    decay = 0.9998
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            print(f"[SGD {it:7d}] perturb after stagnation")
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase D – Deterministic greedy refinement
# ----------------------------------------------------------------------
def _deterministic_refine(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Move mass from the worst contributor to the most favourable index."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_val = cur_val
    best_local_a = a.copy()

    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contribution of each i to the maximal entry
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]
        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break

        # pick target j with the smallest complement contribution
        comp = np.full(n, np.inf)
        mask_j = (k_max - np.arange(n) >= 0) & (k_max - np.arange(n) < n)
        comp[mask_j] = a[k_max - np.arange(n)[mask_j]]
        # tie‑break by smaller a[j]
        order = np.lexsort((a, comp))
        for cand in order:
            if cand != i:
                j = int(cand)
                break

        a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
        if a_k_i <= a_k_j + 1e-14:
            break

        # second‑largest entry (excluding the current maximum)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max == -np.inf:
            second_max = 0.0

        denom = 2.0 * (a_k_i - a_k_j) + 1e-12
        delta_needed = (b[k_max] - second_max) / denom
        delta = min(delta_needed * 0.9, a[i])
        if delta <= 1e-13:
            break

        # incremental convolution update
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        ii = i + i
        jj = j + j
        ij = i + j
        if jj < L:
            b[jj] += delta * delta
        if ii < L:
            b[ii] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        if new_val < cur_val - 1e-14:
            cur_val = new_val
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                if best_local_val <= 1.503:
                    break
        else:
            # revert – move was not beneficial
            b = b_old
            a[i] += delta
            a[j] -= delta
            break

        it += 1
        if it % 500 == 0:
            print(f"[Refine {it:7d}] best = {best_local_val:.6f}")

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: fine mixing with a uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Search a fine grid of α∈[0,0.2] for the best convex combo with uniform."""
    best_a = a
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)

    for alpha in np.linspace(0.001, 0.200, 200):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Final tiny random perturbations + short refinement
# ----------------------------------------------------------------------
def _final_perturbation(a: np.ndarray, n: int,
                       rng: np.random.Generator,
                       time_limit: float) -> tuple[np.ndarray, float]:
    """Apply very small random tweaks and run a short deterministic refinement."""
    best_a = a
    best_val = _objective_value(a)
    deadline = time.time() + time_limit

    while time.time() < deadline:
        noise = rng.normal(scale=0.002, size=n)
        cand = np.maximum(best_a + noise, 0.0)
        cand /= cand.sum()
        # short deterministic refinement (max 0.5 s)
        sub_deadline = time.time() + 0.5
        cand_ref, cand_val = _deterministic_refine(cand, n,
                                                   start=time.time(),
                                                   deadline=sub_deadline,
                                                   best_val=best_val)
        if cand_val < best_val - 1e-12:
            best_val, best_a = cand_val, cand_ref
            print(f"[Perturb] improved to {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a step‑function under the given time budget and returns a list
    of non‑negative floats (the heights of the step function).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 9 000 offers a good trade‑off
    n = 9000

    # ------------------------------------------------------------------
    # Initialisation – many handcrafted families + random draws
    # ------------------------------------------------------------------
    a0, best_val = _initial_candidate(n, rng)
    best_a = a0.copy()
    print(f"[Init] initial best value = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈25 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.25 * (deadline - start_time)
    best_a, best_val = _adam_phase(best_a, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (≈45 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _simulated_annealing_phase(best_a, n,
                                                  start=phase_a_deadline,
                                                  deadline=phase_b_deadline,
                                                  best_a=best_a,
                                                  best_val=best_val,
                                                  rng=rng,
                                                  target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase C – Exact projected sub‑gradient descent (≈15 % of budget)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(best_a, n,
                                          start=phase_b_deadline,
                                          deadline=phase_c_deadline,
                                          best_a=best_a,
                                          best_val=best_val,
                                          target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase D – Deterministic greedy refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _deterministic_refine(best_a, n,
                                             start=phase_c_deadline,
                                             deadline=deadline,
                                             best_val=best_val)

    # ------------------------------------------------------------------
    # Post‑processing: fine mixing with the uniform distribution
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Mix] after uniform mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional tiny random perturbations (if any time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_a, best_val = _final_perturbation(best_a, n, rng,
                                               time_limit=remaining - 5.0)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
