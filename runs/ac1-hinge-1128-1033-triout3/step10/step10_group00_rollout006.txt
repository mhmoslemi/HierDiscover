def _smooth_gradient(a, w):
   # a: array length n; w length L=2n-1.
   n = a.shape[0]
   # compute convolution of a and w_rev using FFT.
   L = w.shape[0] # = 2n-1
   out_len = len(a) + len(w) - 1  # n + (2n-1) -1 = 3n-2
   size = 1 << (out_len - 1).bit_length()
   rfft_a = np.fft.rfft(a, size)
   rfft_w_rev = np.fft.rfft(w[::-1], size)
   conv = np.fft.irfft(rfft_a * rfft_w_rev, size)[:out_len]
   # indices: g_i = 2 * conv[ (2n-2) - i ] for i=0..n-1.
   offset = 2 * n - 2
   g = 2.0 * conv[offset - np.arange(n)]
   return g

"""
Non‑negative step‑function optimiser – smooth‑max mirror descent.

The algorithm works on the simplex (Σ a_i = 1, a_i ≥ 0) and seeks to
minimise 2·n·max(conv(a,a)).  It proceeds as follows:

1. Initialise a from a previous solution (if available) or from a
   short parametric sweep of the Beta family.
2. Repeatedly:
   * compute the full self‑convolution,
   * form a soft‑max weighting of all convolution entries with an
     increasing temperature parameter β (starting very smooth and
     ending sharp – approximating the true max),
   * obtain the gradient of the smooth surrogate,
   * perform an exponentiated‑gradient (mirror) update with a
     decreasing stepsize η,
   * project back onto the probability simplex.
3. Accept occasional uphill moves with a simulated‑annealing temperature
   schedule and apply Gaussian perturbations after long periods of
   stagnation.
4. Return the best sequence seen before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two (for FFT sizing)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Full linear convolution via FFT
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, size: int | None = None) -> np.ndarray:
    """
    Return the full linear convolution of x with itself.
    Length is 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    if size is None:
        size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max weighted convolution (soft‑max of the full autocorrelation)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights w_k ∝ exp(beta·conv_k) that sum to one.
    The maximum entry of conv is subtracted for numerical stability.
    """
    max_c = conv.max()
    # shift to avoid overflow, clip to avoid underflow
    arg = beta * (conv - max_c)
    arg = np.clip(arg, -50, None)      # exp(-50) ≈ 1.9e-22
    w = np.exp(arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        # fall back to uniform weights (should never happen)
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate:
#   g_i = 2·∑_l a_l · w_{i+l}
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max objective.
    Both a (length n) and w (length 2n‑1) are real vectors.
    Returns a vector of length n.
    """
    n = a.shape[0]
    # Convolution a * reverse(w)
    w_rev = w[::-1]
    out_len = len(a) + len(w_rev) - 1          # = 3·n‑2
    # FFT of both vectors (size already a power of two ≥ out_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]

    # g[i] = 2 * conv[(2n‑2)‑i] for i = 0..n‑1
    offset = 2 * n - 2
    idx = offset - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx_i = 1, x_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex (Euclidean norm)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generate a Beta‑family shape (used for cheap initial sweep)
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Return a normalised Beta‑shaped density of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a
