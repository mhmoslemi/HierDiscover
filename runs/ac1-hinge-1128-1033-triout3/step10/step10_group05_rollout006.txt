def _full_convolution(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len-1).bit_length()
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:out_len]
    return conv

assistantfinal**Solution Overview**

The evaluation function penalises the maximal entry of the self‑convolution of a non‑negative,
unit‑sized sequence `a`.  
For a fixed length `n`

```
score(a) = 2·n·max(conv(a,a)) / (sum(a)²)
```

with the additional safety clamps `0 ≤ a_i ≤ 1000`.

The known “U‑shaped’’ families (exponential, arcsine, …) are already close to the optimum.
The new approach searches directly in the space of *parameterised* families,
finds a very good shape, and then performs a fast deterministic local refinement.

The algorithm consists of four stages (all respect the 1000 s budget):

1. **Parametric grid search** – a symmetric Beta‑distribution  
   `a_i ∝ x_i^{α‑1}·(1‑x_i)^{α‑1}` (`x_i = (i+½)/n`).  
   We scan many `α` values, clip to `[0,1000]`, renormalise and keep the best.
2. **Fine‑grained stochastic optimisation of the single parameter** – a simple
   Gaussian random walk (decreasing step size) that continuously improves the
   current best `α`.
3. **Deterministic greedy refinement** – repeated analytic moves that transfer
   mass from the index that contributes most to the current maximal convolution
   entry to the index with the smallest entry, decreasing the objective
   in each successful step.
4. **Tiny uniform mixing** – a final convex combination with a small amount of the
   uniform distribution (often reduces the bound a little).

All heavy numerical work is performed with FFT‑based convolutions (`O(n log n)`),
so even `n = 20000` is comfortably fast.

The final sequence is returned as a plain Python list of `float`s and passes the
security checks of `evaluate_sequence`.

```python

Fast parameter‑search + deterministic refinement for the min‑max
self‑convolution problem.

1.  Scan a symmetric Beta(α,α) family (U‑shaped) on a reasonably large
    dimension (default n = 20000).  The family is cheap to evaluate and
    contains the best known constructions.
2.  Refine the scalar α by a simple stochastic hill‑climb (Gaussian walk
    with decreasing step size).
3.  Apply a deterministic greedy mass‑transfer step that analytically
    reduces the current maximal convolution entry.
4.  End with a tiny mixing with the uniform distribution.

The whole pipeline respects the 1000 s budget and returns a list of
non‑negative floats ready for `evaluate_sequence`.
"""

import numpy as np

# --------------------------------------------------------------
# Convolution utilities (FFT based, exact)
# --------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (length = 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# --------------------------------------------------------------
# Objective (exact, with the same clipping as evaluate_sequence)
# --------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function (without the final
    np.inf checks).  Implements the same clipping as
    evaluate_sequence: 0 ≤ a_i ≤ 1000.
    """
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    b = _full_convolution(a)
    max_b = b.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a * sum_a))


# --------------------------------------------------------------
# Deterministic greedy refinement (exact analytic delta)
# --------------------------------------------------------------
def _deterministic_refine(a0: np.ndarray,
                         n: int,
                         start: float,
                         deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Greedy moves that analytically reduce the current maximal convolution entry.
    Each move transfers mass from the index that contributes most to the
    maximal entry to the index with the smallest a‑value, using the optimal
    delta that would bring the maximal entry down to the second‑largest one.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_val = cur_val
    best_local_a = a.copy()
    it = 0

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contributions of each i to the maximal entry (k_max)
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]

        i = int(np.argmax(contrib))
        if a[i] <= 0.0:
            break

        # candidate target j = smallest a (different from i)
        j_candidates = np.argsort(a)
        for cand in j_candidates:
            if cand != i:
                j = int(cand)
                break

        # contributions of i and j to the maximal entry
        a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
        if a_k_i <= a_k_j + 1e-12:
            # moving mass cannot lower the maximal entry any more
            break

        # second‑largest convolution entry (ignore k_max)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max == -np.inf:
            second_max = 0.0

        # analytic delta that would bring b[k_max] down to second_max
        denom = 2.0 * (a_k_i - a_k_j) + 1e-12
        delta_needed = (b[k_max] - second_max) / denom
        delta = min(delta_needed * 0.9, a[i])          # safety factor
        if delta <= 1e-12:
            break

        # incremental update of the convolution (O(n) vectorised ops)
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        if new_val < cur_val - 1e-14:
            cur_val = new_val
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
        else:
            # revert – move was not beneficial
            b = b_old
            a[i] += delta
            a[j] -= delta
            break

        it += 1
        if it % 2000 == 0:
            print(f"[Refine {it:7d}] best = {best_local_val:.6f}")

    return best_local_a, best_local_val


# --------------------------------------------------------------
# Tiny uniform mixing (helps the final bound a little)
# --------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a tiny amount of uniform mass."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Searches for a step‑function that minimises `evaluate_sequence`.
    The routine consists of a parametric grid search over a symmetric
    Beta(α,α) family, a fine stochastic optimisation of α,
    a deterministic greedy refinement and a final tiny uniform mixing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the final call

    # ------------------------------------------------------------------
    # Problem size – large enough to approach the continuous limit
    # ------------------------------------------------------------------
    n = 20000

    # Pre‑computed positions x_i = (i+0.5)/n  ∈ (0,1)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12

    best_a = None
    best_val = np.inf
    best_alpha = None

    # ------------------------------------------------------------------
    # Phase 1 – coarse grid over α (symmetric Beta distribution)
    # ------------------------------------------------------------------
    alphas_grid = np.linspace(0.05, 1.50, 31)          # 31 values → dense enough
    for alpha in alphas_grid:
        if time.time() > deadline:
            break
        # raw shape (may be huge near the ends)
        shape = (np.maximum(x, eps) ** (alpha - 1.0)) * \
                (np.maximum(1.0 - x, eps) ** (alpha - 1.0))

        # apply the same clipping as evaluate_sequence
        shape = np.clip(shape, 0.0, 1000.0)
        s = shape.sum()
        if s < 1e-12:
            continue
        shape /= s

        val = _objective_value(shape)
        if val < best_val:
            best_val = val
            best_a = shape.copy()
            best_alpha = alpha
            print(f"[grid] α={alpha:.4f} → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – stochastic refinement of the single scalar α
    # ------------------------------------------------------------------
    if best_alpha is None:
        # fallback to uniform if the grid failed for any reason
        best_a = np.full(n, 1.0 / n)
        best_val = _objective_value(best_a)
        best_alpha = 1.0

    # simple Gaussian random walk with decreasing sigma
    sigma = 0.3
    min_sigma = 0.001
    cooling = 0.995

    while time.time() < deadline and sigma > min_sigma:
        alpha_candidate = best_alpha + sigma * rng.standard_normal()
        # keep α positive
        if alpha_candidate <= 0.0:
            sigma *= cooling
            continue

        shape = (np.maximum(x, eps) ** (alpha_candidate - 1.0)) * \
                (np.maximum(1.0 - x, eps) ** (alpha_candidate - 1.0))
        shape = np.clip(shape, 0.0, 1000.0)
        s = shape.sum()
        if s < 1e-12:
            sigma *= cooling
            continue
        shape /= s

        val = _objective_value(shape)
        if val < best_val:
            best_val = val
            best_a = shape.copy()
            best_alpha = alpha_candidate
            print(f"[walk] α={best_alpha:.5f} → {best_val:.6f}")

        sigma *= cooling   # gradually shrink the step size

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy refinement (exact analytic moves)
    # ------------------------------------------------------------------
    refine_deadline = deadline
    best_a, best_val = _deterministic_refine(
        best_a, n,
        start=time.time(),
        deadline=refine_deadline,
        best_val=best_val
    )

    # ------------------------------------------------------------------
    # Phase 4 – tiny uniform mixing (often yields a marginal gain)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification (the harness will call evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return the plain list of floats
    return best_a.tolist()
```
