"""
Smoothed‑gradient descent + deterministic annealing hill‑climb.
The algorithm starts from the previously best step‑function (if available)
or from a few cheap parametric seeds (beta‑family, sinusoid, random Dirichlet).
It then performs a continuation over a temperature‑like parameter β,
optimising a smooth approximation of the max‑convolution objective.
The result is finally refined by a simulated‑annealing mass‑transfer
hill‑climber similar to the reference implementation.
All work respects the 1000 s budget.
"""

import time
import math
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a with Σa=1."""
    n = a.shape[0]
    c = _full_convolution(a)
    return float(2.0 * n * c.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _smooth_gd_optimize(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Continuation gradient descent on a smooth approximation of the max‑convolution.
    The temperature parameter β grows from 0.5 to 1000.
    After each successful step the step size is increased, otherwise it is halved.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_val = _objective(a)
    best_a, best_val = a.copy(), cur_val

    # FFT size needed for the two convolutions a * w and a_rev * w
    size_full = 1 << ((3 * n - 2 - 1).bit_length())

    # schedule of smoothing parameters (β)
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    # initial step size; we will adapt it online
    step = 1.0
    min_step = 1e-12
    no_improve = 0
    IMPROVE_LIMIT = 100_000

    for beta in betas:
        # break if time is exhausted
        if time.time() >= deadline:
            break

        # Main inner loop for a given beta
        while True:
            now = time.time()
            if now >= deadline:
                break

            # ----- convolution a*a (needed for weights) -----
            conv_aa = _full_convolution(a)          # length 2n‑1
            max_c = conv_aa.max()

            # ----- smooth weights approximating the max  -----
            # use log‑sum‑exp trick for stability
            w = np.exp(beta * (conv_aa - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv_aa, 1.0 / conv_aa.size)
            else:
                w /= w_sum

            # ----- gradient of the smoothed objective -----
            # we need two convolutions: a * w  (first n entries)
            # and a_rev * w (entries n‑1 … 2n‑2)
            A = np.fft.rfft(a, size_full)
            W = np.fft.rfft(w, size_full)

            full1 = np.fft.irfft(A * W, size_full)[:3 * n - 2]
            grad1 = full1[:n]

            a_rev = a[::-1]
            A_rev = np.fft.rfft(a_rev, size_full)
            full2 = np.fft.irfft(A_rev * W, size_full)[:3 * n - 2]
            grad2 = full2[(n - 1):(2 * n - 1)]

            grad = 2.0 * (grad1 + grad2)

            # ----- trial step -----
            a_cand = a - step * grad
            a_cand = np.maximum(a_cand, 0.0)
            s = a_cand.sum()
            if s == 0.0:
                a_cand = np.random.dirichlet(np.ones(n))
                s = 1.0
            a_cand /= s

            new_val = _objective(a_cand)

            # ----- acceptance -----
            if new_val < cur_val - 1e-15:
                a = a_cand
                cur_val = new_val
                step = min(step * 1.07, 2.0)
                no_improve = 0
                if new_val < best_val - 1e-12:
                    best_val, best_a = new_val, a.copy()
                    print(f"[SmoothGD β={beta:.1f}] new best = {best_val:.6f}")
                    if best_val <= target_val:
                        return best_a, best_val
            else:
                step *= 0.5
                if step < min_step:
                    break
                no_improve += 1

            # ----- occasional random perturbation to avoid stagnation -----
            if no_improve > IMPROVE_LIMIT:
                noise = np.random.normal(scale=0.01, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                step = 1.0
                no_improve = 0
                print(f"[SmoothGD β={beta:.1f}] stagnation – random perturb")
                # recompute cur_val after perturb
                cur_val = _objective(a)
                if cur_val < best_val:
                    best_val, best_a = cur_val, a.copy()
                    print(f"[SmoothGD β={beta:.1f}] new best after perturb = {best_val:.6f}")

            # optional early break if improvement is tiny (optional)
            # we continue until time runs out or step collapses

    return best_a, best_val

def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Same annealed hill‑climbing as the reference implementation but with
    slightly more aggressive parameters.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # Update convolution efficiently
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to minimise
    evaluate_sequence.  The routine respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    # ----------------------------------------------------------------------
    # Determine problem size and initial solution
    # ----------------------------------------------------------------------
    try:
        old_seq = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old_seq.shape[0]
        old = np.maximum(old_seq, 0.0)
        old = old / old.sum()
        print(f"[init] using previous solution of length {n}")
    except Exception:
        n = 8000
        old = None
        print(f"[init] no previous solution, using n={n}")

    seeds = []

    if old is not None:
        seeds.append(old)

    # uniform seed
    seeds.append(np.full(n, 1.0 / n))

    # a few random Dirichlet seeds
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # simple beta‑family sweep (coarse grid)
    i = np.arange(1, n + 1, dtype=float)
    p_vals = np.linspace(0.5, 3.0, 7)   # 7 values for speed
    q_vals = p_vals
    best_beta = None
    best_beta_val = np.inf
    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a = a / s
            val = _objective(a)
            if val < best_beta_val:
                best_beta_val = val
                best_beta = a.copy()
    if best_beta is not None:
        seeds.append(best_beta)

    # sinusoidal seed
    sin_a = np.sin(np.pi * i / (n + 1))
    sin_a = np.maximum(sin_a, 0.0)
    sin_a /= sin_a.sum()
    seeds.append(sin_a)

    # pick the best of the seeds
    a0 = None
    best_val = np.inf
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            a0 = s.copy()
    if a0 is None:
        # fallback – random Dirichlet
        a0 = np.random.dirichlet(np.ones(n))

    print(f"[init] best seed objective = {best_val:.6f}")

    # ----------------------------------------------------------------------
    # 1️⃣  Smoothed gradient descent (continuation on β)
    # ----------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        # allocate 60 % of the remaining budget to the smooth GD phase
        gd_deadline = time.time() + remaining * 0.6
        best_a, best_val = _smooth_gd_optimize(a0, gd_deadline, start_time)
        print(f"[SmoothGD] best after continuation = {best_val:.6f}")

        # early exit if target already reached
        if best_val <= 1.503:
            final_seq = best_a.tolist()
            try:
                final_score = evaluate_sequence(final_seq)
                print(f"[final] score = {final_score:.6f}")
            except Exception as exc:               # pragma: no cover
                print("[final] evaluate_sequence raised:", exc)
            return final_seq

    # ----------------------------------------------------------------------
    # 2️⃣  Intensive hill‑climbing (annealed mass‑transfer)
    # ----------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        best_a, best_val = _hillclimb_anneal(
            best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=best_a,
            best_val=best_val,
            target_val=1.503,
        )
        print(f"[HillClimb] final best = {best_val:.6f}")

    # ----------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()

"""
Smoothed‑gradient descent + deterministic annealing hill‑climb.

* Start from the previously discovered sequence (if available) or from a few
  cheap parametric seeds (uniform, Dirichlet, a simple Beta family and a
  sinusoid) – the best of them is taken as the initial point.

* Perform a *continuation* gradient descent on a smooth approximation of the
  max‑convolution objective:
      f_β(a) = 2·n· ( 1/β )·log Σ_k exp(β·(a∗a)_k) .
  For small β the landscape is smooth, for large β the function approaches the
  true objective.  The step size is adapted online; a stagnation detector adds
  a small random perturbation to escape flat regions.

* After the continuation phase, a simulated‑annealing mass‑transfer hill
  climber (the same core as the reference code) is run on the remaining
  time budget to polish the solution.

All work respects the 1000 s budget; the final list of floats is returned.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed (soft‑max) gradient descent with continuation on β
# ----------------------------------------------------------------------
def _smooth_gd_optimize(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Continuation gradient descent on a smooth approximation of the max‑convolution.
    β grows from 0.5 up to 1000.  The step size is increased after a successful
    step and halved otherwise.  A large number of consecutive non‑improvements
    triggers a small random perturbation.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_val = _objective(a)
    best_a, best_val = a.copy(), cur_val

    # FFT size for the convolutions a * w and a_rev * w (length 3n‑2)
    size_full = 1 << ((3 * n - 2 - 1).bit_length())

    # schedule of smoothing parameters β
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    step = 1.0                # initial step size
    min_step = 1e-12
    no_improve = 0
    IMPROVE_LIMIT = 100_000

    for beta in betas:
        if time.time() >= deadline:
            break

        while True:
            now = time.time()
            if now >= deadline:
                break

            # ----- (a∗a) and soft‑max weights -----
            conv_aa = _full_convolution(a)          # length 2n‑1
            max_c = conv_aa.max()
            w = np.exp(beta * (conv_aa - max_c))    # stable soft‑max
            w_sum = w.sum()
            if w_sum == 0.0:                        # should not happen, but guard
                w = np.full_like(conv_aa, 1.0 / conv_aa.size)
            else:
                w /= w_sum

            # ----- gradient of the smoothed objective -----
            A = np.fft.rfft(a, size_full)
            W = np.fft.rfft(w, size_full)

            full1 = np.fft.irfft(A * W, size_full)[:3 * n - 2]
            grad1 = full1[:n]

            a_rev = a[::-1]
            A_rev = np.fft.rfft(a_rev, size_full)
            full2 = np.fft.irfft(A_rev * W, size_full)[:3 * n - 2]
            grad2 = full2[(n - 1):(2 * n - 1)]

            grad = 2.0 * (grad1 + grad2)

            # ----- trial step & projection -----
            a_cand = a - step * grad
            a_cand = np.maximum(a_cand, 0.0)
            s = a_cand.sum()
            if s == 0.0:
                a_cand = np.random.dirichlet(np.ones(n))
                s = 1.0
            a_cand /= s

            new_val = _objective(a_cand)

            # ----- acceptance test -----
            if new_val < cur_val - 1e-15:
                a = a_cand
                cur_val = new_val
                step = min(step * 1.07, 2.0)
                no_improve = 0
                if new_val < best_val - 1e-12:
                    best_val, best_a = new_val, a.copy()
                    print(f"[SmoothGD β={beta:.1f}] new best = {best_val:.6f}")
                    if best_val <= target_val:
                        return best_a, best_val
            else:
                step *= 0.5
                if step < min_step:
                    break
                no_improve += 1

            # ----- occasional perturbation to escape stagnation -----
            if no_improve > IMPROVE_LIMIT:
                noise = np.random.normal(scale=0.01, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                step = 1.0
                no_improve = 0
                cur_val = _objective(a)
                print(f"[SmoothGD β={beta:.1f}] stagnation – random perturb")
                if cur_val < best_val:
                    best_val, best_a = cur_val, a.copy()
                    print(f"[SmoothGD β={beta:.1f}] new best after perturb = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive hill‑climbing (same core as reference)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated‑annealing hill‑climber.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # efficiently update convolution after the mass transfer
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val
