"""
Step‑function optimizer for the self‑convolution max‑norm.

The algorithm is deliberately different from the previous multi‑phase
optimizer:

1. **Parameterised families.**  We generate a large number of symmetric
   candidate vectors that are cheap to evaluate:
   * Power‑law families   a_i ∝ (i+δ)^{-p} + (n‑1‑i+δ)^{-p}
   * Exponential families a_i ∝ r^i + r^{n‑1‑i}
   * Simple mixtures with the uniform distribution.

   For each (p,δ) or (r) we normalise the vector to sum 1 and compute
   ``obj = 2·n·max(conv(a,a))`` (the exact value returned by
   ``evaluate_sequence`` when the sum is 1).  The best vector seen so far
   is kept.

2. **Local polishing.**  Starting from the best vector we run a short
   projected sub‑gradient descent on the true objective (the max‑convex
   piecewise‑quadratic function).  The sub‑gradient of the maximal
   convolution entry is known analytically, and after each step we project
   back onto the probability simplex.

3. **Final tiny uniform mix** – a convex combination with a few
   percent of the uniform vector sometimes lowers the peak a little
   further.

All work respects the 1000 s budget; the routine returns a plain Python
list of non‑negative floats that (approximately) minimises
``evaluate_sequence``.  The implementation uses only NumPy and no
external solvers.
"""

import time
import numpy as np

# ---------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ---------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length = 2·len(a)-1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ---------------------------------------------------------------
# Simple simplex projection (Euclidean)
# ---------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------
# Exact objective for a probability vector (sum = 1)
# ---------------------------------------------------------------
def _objective_prob(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ---------------------------------------------------------------
# Family generators
# ---------------------------------------------------------------
def _make_powerlaw(n: int, p: float, delta: float) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ (i+δ)^{-p} + (n‑1‑i+δ)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p) + (n - 1 - i + delta) ** (-p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


def _make_exponential(n: int, r: float) -> np.ndarray:
    """Symmetric exponential shape a_i ∝ r**i + r**(n‑1‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s


# ---------------------------------------------------------------
# Polishing: projected sub‑gradient descent on the exact max‑conv objective
# ---------------------------------------------------------------
def _polish_subgradient(a0: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_val: float) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient descent with decreasing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_obj = best_val

    # step‑size schedule (starts with about half the average entry)
    eta = 0.5 / n
    decay = 0.995
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        cur_obj = 2.0 * n * b.max()
        if cur_obj < best_obj - 1e-12:
            best_obj, best_a = cur_obj, a.copy()
            # early exit if we already beat the target
            if best_obj < 1.503:
                break

        # index where the convolution attains its maximum
        k_max = int(np.argmax(b))

        # sub‑gradient of that entry:
        #   ∂/∂a_i  (a*a)_kmax = 2·a_{kmax‑i}  (if the index is valid)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # gradient step + projection onto the simplex
        a = _project_simplex(a - eta * grad)

        eta *= decay
        it += 1

    return best_a, best_obj


# ---------------------------------------------------------------
# Main entry point required by the evaluation harness
# ---------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low self‑convolution peak.
    Uses a dense grid over a few parametric families, followed by a
    brief projected sub‑gradient polishing phase.
    Returns a list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # -----------------------------------------------------------
    # Choose a problem size (a balance between speed and resolution)
    # -----------------------------------------------------------
    n = 5000   # a few thousand is fast enough for many evaluations

    # -----------------------------------------------------------
    # Initialise with the uniform distribution
    # -----------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_prob(best_a, n)

    # -----------------------------------------------------------
    # 1. Grid search over power‑law families
    # -----------------------------------------------------------
    # p ∈ [0.05, 2.0] (inclusive), 40 steps
    # delta ∈ {0, 0.001·n, 0.005·n, 0.01·n, 0.02·n, 0.05·n}
    p_vals = np.linspace(0.05, 2.0, 40)
    delta_fracs = [0.0, 0.001, 0.005, 0.01, 0.02, 0.05]

    for p in p_vals:
        for df in delta_fracs:
            if time.time() >= deadline:
                break
            delta = max(df * n, 1e-8)   # avoid exact zero when df==0
            a = _make_powerlaw(n, p, delta)
            val = _objective_prob(a, n)
            if val < best_val:
                best_val, best_a = val, a.copy()
                print(f"[grid PL] p={p:.3f} δ={df:.3f} → {best_val:.6f}")

    # -----------------------------------------------------------
    # 2. Grid search over exponential families
    # -----------------------------------------------------------
    r_vals = np.linspace(0.95, 0.9999, 30)   # r close to 1 yields U‑shape
    for r in r_vals:
        if time.time() >= deadline:
            break
        a = _make_exponential(n, r)
        val = _objective_prob(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[grid EXP] r={r:.5f} → {best_val:.6f}")

    # -----------------------------------------------------------
    # 3. Mix the best shape with a small amount of uniform mass
    # -----------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020, 0.050):
        if time.time() >= deadline:
            break
        cand = (1.0 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        val = _objective_prob(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[mix] α={alpha:.3f} → {best_val:.6f}")

    # -----------------------------------------------------------
    # 4. (Optional) Random Dirichlet perturbations – cheap exploration
    # -----------------------------------------------------------
    # Only a few draws, because they are unlikely to beat the structured families,
    # but they occasionally stumble upon a good corner of the space.
    for _ in range(12):
        if time.time() >= deadline:
            break
        a = rng.dirichlet(np.ones(n))
        a /= a.sum()
        val = _objective_prob(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[rand] new best = {best_val:.6f}")

    # -----------------------------------------------------------
    # 5. Local polishing with projected sub‑gradient descent
    # -----------------------------------------------------------
    # Give about the remaining half of the time budget to polishing.
    time_left = deadline - time.time()
    if time_left > 0.0:
        polish_deadline = time.time() + time_left * 0.9   # keep a tiny margin
        best_a, best_val = _polish_subgradient(best_a, n,
                                               start=start_time,
                                               deadline=polish_deadline,
                                               best_val=best_val)

    # -----------------------------------------------------------
    # 6. Final tiny uniform mixing (often gives a marginal improvement)
    # -----------------------------------------------------------
    for alpha in (0.001, 0.003, 0.005, 0.010):
        cand = (1.0 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        val = _objective_prob(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[final mix] α={alpha:.3f} → {best_val:.6f}")

    # -----------------------------------------------------------
    # 7. Final evaluation via the provided harness (purely for logging)
    # -----------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return as plain Python list
    return best_a.tolist()
