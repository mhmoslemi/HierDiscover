def propose_candidate(...):
    import time, numpy as np

    # define helper functions as above
    # define parametric shapeu, shape_exponential, shape_mix
    # define evaluate_fast (our _objective)
    # main search

    start = time.time()
    deadline = start + budget_s - 0.5

    n_small = 2000
    n_large = 8000

    # Phase 0: parametric search
    best_seq_small = None
    best_val = np.inf

    # Uniform baseline
    seq = np.full(n_small, 1.0 / n_small)
    val = _objective(seq)
    best_seq_small, best_val = seq, val
    print(f"[Init] uniform small n, val={best_val}")

    # Edge shape search
    for p in np.linspace(0.0, 2.5, 51):
        a = ((np.arange(1, n_small+1) * (n_small - np.arange(0, n_small))) ** -p)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum
        val = _objective(a)
        if val < best_val:
            best_seq_small, best_val = a, val
            print(f"[Init] edge p={p:.3f}, val={val:.6f}")

    # Exponential shape search
    for lam in np.linspace(0.0, 10.0, 51):
        a = np.exp(-lam*np.arange(n_small)) + np.exp(-lam*np.arange(n_small-1, -1, -1))
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_seq_small, best_val = a, val
            print(f"[Init] exp lam={lam:.3f}, val={val}")

    # Mixture with uniform search: lambda weight w.
    for p in [0.5, 1.0, 1.5, 2.0]:
        edge = ((np.arange(1, n_small+1) * (n_small - np.arange(0, n_small))) ** -p)
        edge = edge / edge.sum()
        for w in np.linspace(0.0, 1.0, 11):
            a = w*edge + (1-w)*np.full(n_small, 1.0/n_small)
            a = a / a.sum()
            val = _objective(a)
            if val < best_val:
                best_seq_small, best_val = a, val
                print(f"[Init] mix p={p}, w={w:.2f}, val={val}")

    # Random Beta draws (like original) but fewer.
    rng = np.random.default_rng(0)
    for _ in range(200):
        p = rng.uniform(0.2, 5.0) # choose moderate
        q = rng.uniform(0.2, 5.0)
        i = np.arange(1, n_small+1, dtype=np.float64)
        a = (i**(p-1))*((n_small-i+1)**(q-1))
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_seq_small, best_val = a, val

    # Now small n may have best maybe ~1.5.
    # Phase 1: Adam optimization on small n until some time budget.
    phase_a_deadline = start + 0.5 * budget_s  # first half for small optimization
    if time.time() < phase_a_deadline:

            best_seq_small, n_small,
            deadline=phase_a_deadline,
            start_time=start,
            best_a=best_seq_small,
            best_val=best_val,
        )

    # Phase 2: Hillclimbing on small n (optional)
    phase_b_deadline = start + 0.75 * budget_s
    if time.time() < phase_b_deadline:
        best_seq_small, best_val = _hillclimb(
            best_seq_small, n_small,
            deadline=phase_b_deadline,
            start_time=phase_a_deadline,
            best_a=best_seq_small,
            best_val=best_val,

    # Phase 3: Upscale to large n

"""
Optimisation for the step‑function attack.

This implementation first searches a low‑dimensional parametric
family (inverse‑product “U‑shape”, exponential, and mixtures with the
uniform distribution) on a modest size (n≈2000) – this is very fast
and quickly finds a good starting point.  The candidate is then
refined with a smooth‑max Adam optimiser and a fast simulated‑annealing
hill‑climber on the small problem.  The resulting vector is finally
interpolated to the target size (n=8000) and polished with a short
hill‑climbing pass.  All operations respect the 1000 s budget and the
best sequence found is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (includes factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # the part that contributes to the gradient is the slice
    # of length n centred around index n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   best_a: np.ndarray, best_val: float):
    """Adam optimiser on the simplex for the loss  L(a)=2·n·smooth_max."""
    w = np.log(a0 + 1e-12)          # log‑softmax parameters

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-6

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 5000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # scheduling
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.02, size=n)

        # restart if stalled
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Fast hill‑climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray, n: int,
               deadline: float, start_time: float,
               best_a: np.ndarray, best_val: float):
    """
    Fast local search that repeatedly moves a small chunk of mass from a
    “high‑contribution’’ index to a “low‑contribution’’ index.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.7, 0.001
    temp_start, temp_end = 0.02, 1e-8

    no_improve = 0
    IMPROVE_LIMIT = 60000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # choose source index proportionally to its contribution
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target: bias towards indices with low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback excluding i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] stagnation – perturb (it={iteration})")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _shape_edge(p: float, n: int) -> np.ndarray:
    """U‑shape a_i ∝ ((i+1)*(n-i))^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 .. n
    prod = i * (n - i + 1)                           # (i)*(n-i+1) = (i+1)*(n-i) with shift
    # avoid zero; add tiny epsilon
    a = np.power(prod, -p, where=prod > 0, out=np.ones_like(prod))
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum


def _shape_exponential(lam: float, n: int) -> np.ndarray:
    """U‑shape a_i ∝ exp(-lam*i) + exp(-lam*(n-1-i))."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum


def _mix_with_uniform(base: np.ndarray, w: float) -> np.ndarray:
    """Convex mix w·base + (1‑w)·uniform, then renormalise."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n)
    a = w * base + (1.0 - w) * uniform
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final call

    # ------------------------------------------------------------------
    # Choose problem sizes
    # ------------------------------------------------------------------
    n_small = 2000
    n_large = 8000

    # ------------------------------------------------------------------
    # Phase 0 – exhaustive search in parametric families (small n)
    # ------------------------------------------------------------------
    best_seq_small = np.full(n_small, 1.0 / n_small)
    best_val = _objective(best_seq_small)
    print(f"[Init] uniform (n={n_small})  value={best_val:.6f}")

    # Edge shape sweep
    for p in np.linspace(0.0, 2.5, 51):
        cand = _shape_edge(p, n_small)
        val = _objective(cand)
        if val < best_val:
            best_seq_small, best_val = cand, val
            print(f"[Init] edge p={p:.3f}          value={val:.6f}")

    # Exponential shape sweep
    for lam in np.linspace(0.0, 5.0, 51):
        cand = _shape_exponential(lam, n_small)
        val = _objective(cand)
        if val < best_val:
            best_seq_small, best_val = cand, val
            print(f"[Init] exp  λ={lam:.3f}        value={val:.6f}")

    # Mixture of edge shape with uniform
    edge_base = _shape_edge(1.0, n_small)   # a representative edge shape
    for w in np.linspace(0.0, 1.0, 11):
        cand = _mix_with_uniform(edge_base, w)
        val = _objective(cand)
        if val < best_val:
            best_seq_small, best_val = cand, val
            print(f"[Init] mix  w={w:.2f}           value={val:.6f}")

    # Random Beta draws (symmetric)
    for _ in range(200):
        p = rng.uniform(0.1, 5.0)
        q = rng.uniform(0.1, 5.0)
        i = np.arange(1, n_small + 1, dtype=np.float64)
        a = (i ** (p - 1.0)) * ((n_small - i + 1) ** (q - 1.0))
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_seq_small, best_val = a, val
            print(f"[Init] random Beta p={p:.2f} q={q:.2f} value={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (small n)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    if time.time() < phase_a_deadline:
        best_seq_small, best_val = _adam_optimize(
            best_seq_small, n_small,
            deadline=phase_a_deadline,
            start_time=start_time,
            best_a=best_seq_small,
            best_val=best_val,
        )

    # ------------------------------------------------------------------
    # Phase 2 – Hill‑climbing (small n)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * budget_s
    if time.time() < phase_b_deadline:
        best_seq_small, best_val = _hillclimb(
            best_seq_small, n_small,
            deadline=phase_b_deadline,
            start_time=phase_a_deadline,
            best_a=best_seq_small,
            best_val=best_val,
        )

    # ------------------------------------------------------------------
    # Phase 3 – Upscale to the target length
    # ------------------------------------------------------------------
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    seq_large = np.interp(x_large, x_small, best_seq_small)
    seq_large = np.maximum(seq_large, 0.0)
    seq_large /= seq_large.sum()
    best_val_large = _objective(seq_large)
    best_seq_large = seq_large
    print(f"[Upscale] value on n={n_large}  {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final polishing on the large problem (short hill‑climb)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq_large, best_val_large = _hillclimb(
            best_seq_large, n_large,
            deadline=deadline,
            start_time=phase_b_deadline,
            best_a=best_seq_large,
            best_val=best_val_large,
        )

    # ------------------------------------------------------------------
    # Final evaluation through the provided security wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq_large.tolist())
        print(f"[Final] evaluate_sequence => {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Return plain list
    return best_seq_large.tolist()

"""
Optimisation for the step‑function attack.

The algorithm works in four stages:

1. **Parametric sweep (small problem, n≈2000).**  
   Fast exhaustive search over a few low‑dimensional families:
   – inverse‑product “U‑shape’’ a_i ∝ ((i+1)*(n‑i))^{-p}
   – symmetric exponential a_i ∝ exp(-λi)+exp(-λ(n‑1-i))
   – mixtures of the above with a uniform background
   – a handful of random symmetric Beta distributions.
   The best candidate is kept.

2. **Refinement on the small problem.**  
   A smooth‑max Adam optimiser followed by a fast simulated‑annealing
   hill‑climber further reduces the objective while staying on the
   probability simplex.

3. **Upscaling.**  
   The refined small‑n vector is linearly interpolated to the target
   length (n=8000) and renormalised.

4. **Final polishing (large problem).**  
   A short hill‑climbing pass on the full size vector finishes the
   optimisation.  The resulting sequence is returned; its quality is
   finally reported through the provided ``evaluate_sequence`` wrapper.

All phases respect the 1000 s total budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # gradient already contains 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   best_a: np.ndarray, best_val: float):
    """Adam optimiser on the simplex for the loss L(a)=2·n·smooth_max."""
    w = np.log(a0 + 1e-12)          # log‑softmax parameters

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-6

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 5000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # contains 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.02, size=n)

        # restart if stalled
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Fast hill‑climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray, n: int,
               deadline: float, start_time: float,
               best_a: np.ndarray, best_val: float):
    """
    Fast local search that repeatedly moves a tiny chunk of mass from a
    “high‑contribution’’ index to a “low‑contribution’’ index.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.7, 0.001
    temp_start, temp_end = 0.02, 1e-8

    no_improve = 0
    IMPROVE_LIMIT = 60000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # current most influential convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index (high contribution)
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index (low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # O(n) update of the convolution
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] stagnation – perturb (it={iteration})")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _shape_edge(p: float, n: int) -> np.ndarray:
    """U‑shape a_i ∝ ((i+1)*(n-i))^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)           # 1 … n
    prod = i * (n - i + 1)                               # (i)*(n-i+1) = (i+1)*(n-i)
    a = prod ** (-p)                                     # safe – prod ≥ 1
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _shape_exponential(lam: float, n: int) -> np.ndarray:
    """U‑shape a_i ∝ exp(-lam*i) + exp(-lam*(n-1-i))."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _mix_with_uniform(base: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·base + (1‑w)·uniform, renormalised."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n)
    a = w * base + (1.0 - w) * uniform
    a = a / a.sum()
    return a
