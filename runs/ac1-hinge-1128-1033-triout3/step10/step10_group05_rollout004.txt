"""
Optimization of a non‑negative step function to minimise

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in three stages:

1. **Warm‑start** – use a known solution (global ``height_sequence_1`` if
   available) or a hand‑crafted arcsine shape, normalised to a simplex.

2. **Analytic greedy refinement** – repeatedly identify the diagonal of the
   self‑convolution with the largest entry, then move a carefully chosen
   amount of mass from the index that contributes most to that diagonal to a
   low‑mass index.  The exact amount (`delta`) is derived analytically so that
   the maximal convolution entry is driven down towards the second‑largest
   entry.  When no profitable pair can be found we inject a tiny random
   perturbation and continue.  This loop runs for most of the allotted time.

3. **Uniform mixing** – a final convex combination with a small uniform
   component (often beneficial for the “max‑convolution” landscape).

The whole procedure respects the 1000 s budget and returns a plain list of
floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²).  Assumes a non‑negative."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max() / (a.sum() ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial candidate generation (hand‑crafted families + fallback)
# ----------------------------------------------------------------------
def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a simplex‑normalised start vector (the best among a few families)."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(vec):
        nonlocal best_a, best_val
        a = np.asarray(vec, dtype=np.float64)
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a /= s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # arcsine‑like (heavy at the ends)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = 1.0 / np.sqrt(np.clip(x * (1.0 - x), 1e-12, None))
    _try(a)

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1)))
    _try(a)

    # symmetric exponential (U‑shaped)
    for r in (0.9999, 0.9995, 0.998, 0.996, 0.995):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # a few Dirichlet draws
    for _ in range(8):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a


# ----------------------------------------------------------------------
# Greedy analytic refinement – main workhorse
# ----------------------------------------------------------------------
def _greedy_refine(a: np.ndarray,
                   deadline: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Repeatedly transfer mass from the index that contributes most to the
    current worst self‑convolution entry to a low‑mass index.
    The exact transferred amount is computed analytically.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # full convolution (len = 2n‑1)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(123456)

    # stagnation handling
    no_improve = 0
    IMPROVE_LIMIT = 250_000

    while time.time() < deadline:
        # keep track of best
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Refine] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # locate the worst diagonal
        k_max = int(np.argmax(b))
        max_b = b[k_max]

        # second–largest entry (excluding the worst)
        tmp = b.copy()
        tmp[k_max] = -np.inf
        second_max = tmp.max()

        # contribution of every i to the worst diagonal
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        # candidate sources – top contributors
        top_src = np.argsort(-contrib)[:15]   # 15 strongest contributors
        # candidate targets – smallest masses (avoid the source itself)
        low_tgt = np.argsort(a)[:15]

        moved = False
        for i in top_src:
            if a[i] <= 1e-12:
                continue
            # product partner of i on the worst diagonal
            idx_i = k_max - i
            a_k_i = a[idx_i] if 0 <= idx_i < n else 0.0
            if a_k_i <= 0:
                continue
            for j in low_tgt:
                if i == j:
                    continue
                idx_j = k_max - j
                a_k_j = a[idx_j] if 0 <= idx_j < n else 0.0
                # we can only improve if the partner of i is larger than that of j
                if a_k_i <= a_k_j + 1e-12:
                    continue

                # analytic delta that would bring the worst entry down to the
                # second‑largest entry (with a safety factor)
                denom = 2.0 * (a_k_i - a_k_j) + 1e-12
                delta_needed = (max_b - second_max) / denom
                # safety factor 0.8 keeps us from overshooting
                delta = min(delta_needed * 0.8, a[i])
                if delta <= 1e-12:
                    continue

                # ---- incremental convolution update ----
                # before we modify `a` we use the *old* values for the update.
                b[j:j + n] += 2.0 * delta * a
                b[i:i + n] -= 2.0 * delta * a
                L = b.shape[0]
                two_j = 2 * j
                two_i = 2 * i
                ij = i + j
                if two_j < L:
                    b[two_j] += delta * delta
                if two_i < L:
                    b[two_i] += delta * delta
                if ij < L:
                    b[ij] -= 2.0 * delta * delta

                # move the mass
                a[i] -= delta
                a[j] += delta

                moved = True
                break
            if moved:
                break

        if not moved:
            # No profitable deterministic move – inject a small random perturbation
            noise = rng.normal(scale=0.008, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                a[:] = 1.0 / n
            else:
                a /= a_sum
            b = _full_convolution(a)

        cur_val = 2.0 * n * b.max()

        if no_improve > IMPROVE_LIMIT:
            # shake the current solution a bit and reset the counter
            print("[Refine] stagnation – applying global perturbation")
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a_sum = a.sum()
            a = a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (occasionally improves the final bound)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combine the candidate with a little uniform mass."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.002, 0.004, 0.008, 0.016):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative floats that (approximately) minimise
    the `evaluate_sequence` objective.  The routine respects the 1000 s
    time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the final call

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressiveness and speed.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # 0. Warm‑start – reuse the previously discovered best vector if present.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    except Exception:
        a0 = _initial_candidate(n, rng)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # 1. Greedy analytic refinement – consumes the bulk of the budget.
    # ------------------------------------------------------------------
    phase_deadline = deadline   # use the remaining time
    best_a, best_val = _greedy_refine(best_a, phase_deadline, target_val=TARGET)

    # ------------------------------------------------------------------
    # 2. Final tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # 3. Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()

This implementation starts from the best known construction (or a strong arcsine‑shaped guess), then performs a fast analytic greedy mass‑transfer loop that directly attacks the maximal self‑convolution entry.  The loop runs for essentially the whole time budget, applying occasional random perturbations to escape plateaus.  Finally a minute amount of uniform mass is mixed in to polish the result.  The algorithm respects all problem constraints and stays within the 1000 s limit.
