Simple Simulated Annealing + Sub‑gradient polish for minimizing
evaluate_sequence(sequence). The algorithm builds an initial candidate from
several parametric families (interpolated old solution, Beta, Dirichlet,

import time

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta family candidate
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # log‑scale parameters (roughly 1e‑4 … 10)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a, b, cur_val = a_new, b_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a_new.copy()
                print(f"[SA] iter {it:<7d} new best = {best_val:.6f}")

        # occasional jitter to avoid plateaus
        if it % 25000 == 0 and it > 0:
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            print(f"[SA] iter {it:<7d} jitter, cur_val={cur_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase B – Projected sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        kmaxes = np.where(b == max_b)[0]

        # sub‑gradient of the exact objective (2·n·max)
        grad = np.zeros_like(a)
        for k in kmaxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step‑size schedule – linear decay from 0.30 to 1e‑5
        prog = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a_new.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # tiny random perturbation helps to escape flat regions
        if rng.random() < 0.002:
            a = np.maximum(a + rng.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Final evaluation (the only place where the secured function is called)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure the output is a plain Python list
    return best_a.tolist()
