    a = a0.copy()
    best_a = a.copy()

    eps = 1e-8

    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0
    while time.time() < deadline:
        t_iter += 1
        # schedule lambda: start low to high
        progress = (time.time() - start_time) / (deadline - start_time)
        lam = 0.5 + 39.5 * progress  # from 0.5 to 40

"""
Improved search for the step‑function minimising the evaluation function.

The algorithm combines three ideas:

1️⃣  **Parametric warm‑start** – a quick sweep of the two‑parameter
    Beta family (a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}) and a few Dirichlet draws
    produces a few promising starting points.  If a previous solution is
    available in ``height_sequence_1`` it is interpolated as well.

2️⃣  **Smooth max optimisation** – instead of optimising the exact
    non‑differentiable max (a∗a) we minimise a smooth approximation

          f_λ(a)= (1/λ)·log(∑_k e^{λ·(a∗a)_k})

    whose gradient can be written as a convolution and is evaluated with
    FFTs.  Adam (adaptive‑momentum) is used together with a continuation
    on λ (λ grows from 0.5 to ≈40) and a Euclidean projection onto the
    probability simplex.

3️⃣  **Final local refinement** – the intensive annealing hill‑climber
    from the reference implementation is run on the remaining time.

All phases respect the 1000 s budget; the best sequence found is
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family sweep (dense but cheap)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search the beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max optimiser (Adam + continuation on λ)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float) -> tuple[np.ndarray, float]:
    """
    Optimises 2·n·max(conv(a,a)) using a smoothed max objective.
    Adam updates are projected onto the simplex after each step.
    λ grows from 0.5 to ~40 as the optimisation proceeds.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # ensure Σa = 1

    best_a = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.07                     # base learning rate
    t = 0

    # FFT sizes (pre‑computed for the whole run)
    size1 = 1 << ((2 * n - 2).bit_length())       # for a*a
    size2 = 1 << ((3 * n - 3).bit_length())       # for a_rev * w

    while True:
        now = time.time()
        if now >= deadline:
            break
        t += 1

        # ---------- λ continuation ----------
        prog = (now - start_time) / (deadline - start_time)
        lam = 0.5 + 39.5 * prog          # goes from 0.5 → 40

        # ---------- a*a ----------
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[:2 * n - 1]
        max_c = conv.max()

        # ---------- smooth weights ----------
        # w_k ∝ exp(λ·(conv_k – max_c))
        exps = np.exp(lam * (conv - max_c))
        w = exps / exps.sum()            # length 2·n‑1, sum to 1

        # ---------- gradient ----------
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, size2)
        fw = np.fft.rfft(w, size2)
        conv_aw = np.fft.irfft(fa_rev * fw, size2)[:3 * n - 2]
        # slice i+n‑1 … i+n‑1+n‑1 gives n entries
        grad = 2.0 * conv_aw[n - 1 : n - 1 + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha * (m_hat / (np.sqrt(v_hat) + eps))

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s
        a = a_new

        # ---------- keep the best ----------
        cur_val = 2.0 * n * max_c          # objective of the *old* a
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SmoothOpt {t:7d}] new best = {best_val:.6f} λ={lam:.2f}")
            if best_val <= 1.503:
                break

        # optional stagnation reset (every 800 steps without improvement)
        if t % 800 == 0:
            if cur_val >= best_val * 0.9999:
                # small random perturbation to escape plateaus
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                m.fill(0.0)
                v.fill(0.0)
                print(f"[SmoothOpt {t:7d}] stagnation – random perturb")

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

    start_time = time.time()
    # reserve a little margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # Problem size – a few thousand entries give a good trade‑off.
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣ Gather diverse initial seeds
    # ------------------------------------------------------------------
    seeds = []

    # a) previous solution, if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        seeds.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) best from a dense Beta‑family sweep
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_a is not None:
            seeds.append(beta_a)
            print(f"[init] beta‑family seed (value={beta_val:.6f})")
    except Exception as exc:
        print("[init] beta family failed:", exc)

    # c) a few random Dirichlet draws
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 1️⃣ Smooth‑max optimisation (share the budget roughly equally)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_val = np.inf

    # 70 % of the remaining time is given to the smooth optimiser
    time_for_smooth = (deadline - time.time()) * 0.70
    if time_for_smooth <= 0:
        time_for_smooth = 0.0
    per_seed = time_for_smooth / max(1, len(seeds))

    for i, a_seed in enumerate(seeds):
        if time.time() >= deadline:
            break
        local_deadline = time.time() + per_seed
        a_opt, val_opt = _smooth_optimize(a_seed, n, local_deadline, start_time)
        print(f"[seed {i}] smooth_opt best = {val_opt:.6f}")
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2️⃣ Final intensive hill‑climbing on the remaining time slice
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final secure evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
