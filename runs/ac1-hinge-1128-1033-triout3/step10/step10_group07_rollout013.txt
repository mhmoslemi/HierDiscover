def correlation(w, a):

   # returns linear correlation array length m+n-1, where corr[i] = sum_{j} w[j] * a[i + j] (a zero-padded)
   # Compute via FFT

   corr_full = np.fft.irfft(W * np.conj(A), L)[:m + n - 1]
   return corr_full

"""
Propose a step‑function that reduces the objective
    2·n·max(conv(a,a))

Algorithm (smooth “soft‑max” sub‑gradient + Adam)
-------------------------------------------------
* Represent the step heights a as a softmax of free logits z,
  guaranteeing a_i≥0 and Σa_i=1.
* Use a smoothed version of the max‑convolution entry:
      sm_γ(b) = (1/γ)·log Σ exp(γ·b_k)   (γ gradually grows)
  Its gradient w.r.t. b is the softmax of γ·b.
* For a given a compute
      b = a * a                              (FFT‑based full convolution)
      w = softmax(γ·b)                       (weights for the smooth max)
  The exact gradient of 2·n·sm_γ(b) w.r.t. a is
      ∂/∂a_i = 4·n· Σ_k w_k·a_{k‑i}          (a linear correlation)
  which can be obtained by a single FFT convolution of w with the
  reversed a.
* Apply Adam on the logits z using the chain‑rule
      ∂obj/∂z_i = a_i·(∂obj/∂a_i – Σ_j a_j·∂obj/∂a_j).
* γ is increased from 1.0 to ≈30 throughout the run, turning the
  smooth objective into the true max‑objective.
* The best (lowest exact) sequence seen is kept.  After the main
  optimisation a tiny uniform mixing (α∈[0,0.02]) is tried to
  polish the result.
* The routine respects the 1000 s time budget, works on a single
  CPU and uses only NumPy.
"""

import time
import numpy as np

def _softmax_from_logits(logits: np.ndarray) -> np.ndarray:
    """Stable softmax on a 1‑D array."""
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution a*a using one FFT size.
    Returns the first (2·len(a)-1) entries.
    """
    Fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Fa * Fa, n=size)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _conv_w_a_rev_fft(w: np.ndarray, a_rev: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution w * a_rev (both 1‑D) using the given FFT size.
    Returns the first (len(w) + len(a_rev) - 1) entries.
    """
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A, n=size)
    return conv[: w.shape[0] + a_rev.shape[0] - 1]


def _objective_exact(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(conv(a,a)).  Assumes Σa=1."""
    n = a.shape[0]
    # a*a via FFT (size just needs to cover 2n‑1)
    size = 1 << ((2 * n - 1) - 1).bit_length()
    b = _full_convolution_fft(a, size)
    return float(2.0 * n * b.max())


def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for a few tiny α and keep the best."""
    best_a, best_val = a, _objective_exact(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_exact(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


def _generate_initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a candidate using a few simple parametric families.
    The best of the sampled families (under the exact objective) is returned.
    """
    best_a = None
    best_val = np.inf

    # 1. Exponential (U‑shaped) family
    i = np.arange(n, dtype=np.float64)
    for r in np.linspace(0.55, 0.9999, 25):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        val = _objective_exact(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # 2. Power‑law edge‑peaked family
    for p in np.linspace(-4.0, 4.0, 25):
        i1 = np.arange(1, n + 1, dtype=np.float64)
        base = i1 * (n + 1 - i1) + 1e-12
        a = np.power(base, -p)
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        val = _objective_exact(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # 3. Cosine window (Hann) with exponent
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    for beta in np.linspace(0.5, 4.0, 20):
        a = np.power(w, beta)
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        val = _objective_exact(a)
        if val < best_val:

    # 4. A few random Dirichlet draws
    for _ in range(15):

        val = _objective_exact(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # Fallback – uniform
    if best_a is None:

    return best_a

            last_improve_iter = 0
            print("[iter] new best = {:.6f} (γ≈{:.2f})".format(best_val,
                                                             gamma_start + (gamma_end - gamma_start) *
                                                             ((time.time() - start_time) /
                                                              (deadline - start_time))))
        else:
            last_improve_iter += 1

        # ----- smooth‑max weight vector w -----
        # γ grows linearly with elapsed time
        progress = (time.time() - start_time) / (deadline - start_time)
        gamma = gamma_start + (gamma_end - gamma_start) * progress
        scaled = gamma * conv_aa
        scaled_max = np.max(scaled)
        w_exp = np.exp(scaled - scaled_max)
        w = w_exp / w_exp.sum()               # w has length 2n‑1, non‑negative, Σw = 1

        # ----- gradient w.r.t. a (4·n·correlation) -----
        a_rev = a[::-1]
        # convolution w * a_rev (full) via FFT
        W = np.fft.rfft(w, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        conv_rev = np.fft.irfft(W * A_rev, n=size_fft)[: 3 * n - 2]
        # slice the middle block of length n (indices n‑1 … 2n‑2)
        grad_a = 4.0 * n * conv_rev[n - 1: n - 1 + n]

        # ----- chain‑rule to obtain gradient w.r.t. logits -----
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # ----- Adam update on logits -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits * grad_logits)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional jitter to escape plateaus -----
        if last_improve_iter > IMPROVE_LIMIT:
            jitter = rng.normal(scale=0.015, size=n)
            logits += jitter
            last_improve_iter = 0
            print("[iter] jitter injected after {} stagnant iterations".format(IMPROVE_LIMIT))

        # ----- time‑guard (leave a little leeway) -----
        if time.time() > deadline - 0.2:
            break

    # ------------------------------------------------------------------
    # Tiny uniform mixing (often improves the final bound)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will call evaluate_sequence as well)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] best evaluation reported = {:.6f}".format(final_score))
    except Exception as exc:    # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
