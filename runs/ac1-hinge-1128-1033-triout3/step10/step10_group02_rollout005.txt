"""
Propose a step‑function (non‑negative vector) that minimises
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum a)²

The algorithm improves on the previous implementation by:

*  A much larger random‑Beta initial search (thousands of candidates).
*  A corrected O(n) hill‑climber (the quadratic term for the i+j entry
   was previously under‑subtracted, preventing many beneficial moves).
*  Slightly re‑balanced time budgets (more time for hill‑climbing and
   sub‑gradient polishing) while still respecting the 1000 s limit.
*  Minor hyper‑parameter tweaks (longer idle before Adam restarts,
   tighter smooth‑max ε schedule) and a final uniform blending step.

All helper functions are top‑level, there are no closures or lambdas,
and the only external call is to the already‑provided ``evaluate_sequence``.

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and gradient (log‑sum‑exp) – includes the factor 2·n
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient
    of 2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights (log‑sum‑exp trick)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for the loss L(a)=2·n·smooth_max(a).
    The smoothness ε is annealed from 0.5 → 1e‑8.
    """
    w = np.log(a0 + 1e-12)          # log‑softmax parameters

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.25, 0.001     # a bit slower start
    eps_start, eps_end = 0.5, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 8000   # allow longer stagnation before restart

    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule
        prog = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # grad already contains 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.02, size=n)

        # restart if stalled
        if idle > MAX_IDLE:
            print("[Adam] restart after idle")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (exact O(n) updates)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Fast local search that repeatedly moves a small chunk of mass from a
    “high‑contribution’’ index to a “low‑contribution’’ index.
    The convolution update now correctly subtracts 2·δ² from the (i+j)
    entry (a bug in the original code prevented many improvements).
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-14

    # schedules
    max_frac_start, max_frac_end = 0.6, 0.001
    temp_start, temp_end = 0.02, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 50000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # choose source index proportionally to its contribution
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target: bias towards indices with low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback excluding i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # --- exact O(n) convolution update (corrected) ---
        b_new = b.copy()
        # linear part (2·δ·a) for both source and target
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic self‑terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        # cross term (i+j) gets -2·δ² (fixed!)
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] stagnation – perturb (it={iteration})")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a) = 2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    The step size decays linearly.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear schedule: start ~0.3, finish ~1e‑5
        prog = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape flat plateaus
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for several t."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Random Beta‑family candidate generator
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    # log‑uniform exponents covering a wide range
    logp = rng.uniform(-9.2, 2.3)   # ≈ 1e‑4 … 10
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print(f"[Init] fallback uniform, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Hill‑climbing (≈45 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.75 * budget_s
    best_a, best_val = _hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=deadline,
        start_time=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        max_iters=250_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation (using the provided evaluator)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # Ensure we return a plain Python list of floats
    return best_a.tolist()
