    a = a0.copy()
    b = full_convolution(a)

    best_local_a = best_a.copy()
    best_local_val = best_val

    T_start = (cur_val - target_val) * 0.25 if cur_val > target_val else 0.001
    T_end = 1e-9
    # step size schedule
    max_frac_start = 0.5
    max_frac_end = 0.0005

    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T_start * (T_end / T_start) ** progress
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # compute contributions to current peak
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(rng.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        rev_idx = k_max - np.arange(n)

"""
propose_candidate.py

A lightweight optimiser for the “max‑convolution” bound.

Algorithm
---------
1. **Broad parametric sweep** – evaluate many families (exponential,
   power‑law, beta, sine‑power, piecewise‑constant, logistic, Gaussian,
   raised‑cosine, random Dirichlet).  The best normalised vector is kept.

2. **Simulated‑annealing refinement** – starting from the best sweep
   candidate we repeatedly move a small amount of mass from a source
   index that contributes most to the current maximal convolution entry
   to a target index that contributes least.  The move is accepted if it
   strictly lowers the true objective, otherwise it may be accepted
   with a Metropolis probability that slowly cools to zero.  This
   exploration escapes shallow local minima and typically pushes the
   bound well below the sweep result.

3. **Uniform mixing** – a final convex combination with a tiny share of
   the uniform distribution smooths residual spikes.

The whole pipeline respects the 1000‑second wall‑clock budget and
returns a non‑negative list whose sum is 1 (the objective is scale‑invariant).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a) for a vector with unit sum."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (unused but handy)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x ≥ 0, Σ x = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        _try(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

def _simulated_annealing(a0: np.ndarray, n: int,

    a = a0.copy()

    best_local_a = best_a.copy()
    best_local_val = best_val

    T_start = max((cur_val - target_val) * 0.25, 1e-4)
    T_end = 1e-9

    max_frac_start, max_frac_end = 0.45, 0.001

    # stagnation handling
    no_improve = 0
    IMPROVE_LIMIT = 160_000
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T_start * (T_end / T_start) ** progress
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # --- locate the peak of the convolution -------------------------
        k_max = int(np.argmax(b))

        # contributions of each index to the current peak
        idx = k_max - np.arange(n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(rng.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]

        if w_sum > 0.0:
            probs_j = w / w_sum
            j = int(rng.choice(n, p=probs_j))

        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic self‑terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0.0 and rng.random() < np.exp((cur_val - new_val) / T):
            accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        # occasional stochastic restart if stuck for a long time
        if no_improve > IMPROVE_LIMIT:
            # perturb the current best and continue
            a = best_local_a.copy()
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: convex combination with a tiny share of the uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int):
    """Blend the current vector with a tiny amount of uniform mass."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point – called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a non‑negative step‑function under the given time budget.
    Returns a list of floats (Σ = 1) that (hopefully) beats the current record.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – a few thousand entries give a good trade‑off
    n = 9000

    # ------------------------------------------------------------------
    # 1️⃣ Initial parametric sweep
    # ------------------------------------------------------------------
    print("[Init] starting parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Simulated‑annealing refinement (uses the remaining budget)
    # ------------------------------------------------------------------
    best_a, best_val = _simulated_annealing(
        best_a, n,
        start=start_time,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
    )
    print(f"[SA] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Report final score using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # ensure a valid simplex point (scaling does not affect the objective)
    best_a = best_a / best_a.sum()
    return best_a.tolist()
