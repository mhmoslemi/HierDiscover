    best_f = f.copy()
    best_val = val

        M = _full_convolution(f).max()
        # Build conv matrix
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for k in range(2*n-1):

            i_end = min(k, n-1)

"""
Propose a step‐function that minimizes the evaluation metric.

Algorithm
---------
1. **Initial seed** – a broad parametric sweep (exponential, power‑law,
   beta, arcsine, sine‑power, piecewise‑constant, random Dirichlet) generates
   a good probability vector `f` (∑ f = 1).

2. **Alternating linear‑programming refinement** – keep the current maximal
   convolution value `M = max(f∗f)`.  Solve the linear program

        max   Σ_j g_j
        s.t.  (f ∗ g)_k ≤ M   for all k
              g_j ≥ 0

   (the convolution with a fixed `f` is linear in `g`).  The optimal `g`
   has a larger total mass than `f`; after normalising `g` to sum = 1 we
   take a convex combination `f←(1‑t)f+t·g` with a time‑dependent mixing
   factor `t`.  This strictly reduces the maximal convolution and is cheap
   (one LP per iteration, dense matrix size (2n‑1)×n).

3. **Local polish** – a simple stochastic “mass‑transfer’’ step
   (move a tiny amount of probability from an index that contributes heavily
   to the current peak to an index that contributes little) is applied a
   few thousand times to squeeze out any remaining slack.

4. **Uniform blend** – finally a tiny mixture with the uniform distribution
   is tested; for some problems it gives a marginal improvement.

All vectors are kept on the probability simplex (non‑negative, sum = 1);
the evaluation metric of the competition is then `2·n·max(f∗f)`.  The
procedure runs for the allotted time (≈ 1000 s) and returns the best
vector found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    # Numerical noise may make a few entries slightly negative
    np.maximum(conv, 0, out=conv)
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector with Σ a = 1 (i.e. 2·n·max(a∗a))."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1} (standard O(n) algorithm)."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – broad parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Search a rich collection of parametric families for a low‑objective seed."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                           # 1 … n

    # 1. Exponential tails (left‑hand and symmetric)
    for r in np.linspace(0.995, 0.999999, 20):
        _try(r ** i)                         # decreasing
        _try(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

    # 2. Power‑law tails (left‑hand and symmetric)
    for p in np.linspace(0.1, 3.0, 30):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Inverse‑sqrt (special case of power‑law)
    _try(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))

    # 4. Beta densities – finer grid around the U‑shape (a,b < 1)
    beta_vals = np.linspace(0.1, 1.5, 15)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 5. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # 6. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 12):
        _try(base_sin ** p)

    # 7. Piecewise‑constant “high‑at‑edges’’ shapes
    for k_frac in np.linspace(0.005, 0.1, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 8. Random Dirichlet draws for diversity
    rng = np.random.default_rng(123456)
    for _ in range(80):
        _try(rng.dirichlet(np.ones(n)))

    # 9. Interpolate a previously known solution, if present
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper: build the convolution matrix A (rows = 2n‑1, cols = n)
# such that (A @ g)[k] = (f ∗ g)[k].
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return dense matrix A with A[k, j] = f[k‑j] (zero outside range)."""
    n = f.shape[0]
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return toeplitz(c, r)


# ----------------------------------------------------------------------
# Linear‑programming refinement stage
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, deadline: float) -> np.ndarray:
    """
    Repeatedly solve the LP described in the algorithmic overview,
    mixing the solution back into `f`.  Stops when time runs out.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_val = _objective_value(best_f)

    it = 0
    while time.time() < deadline:
        M = _full_convolution(f).max()
        # Build A once per iteration
        A = _conv_matrix(f)                      # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear program: maximize sum(g)  →  minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            print(f"[LP {it:04d}] linprog failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g                     # normalise to the simplex

        # mixing weight – start relatively aggressive, then decay
        progress = (time.time() - (deadline - 1000)) / (1000.0)   # approximate total progress
        t = max(0.05, 0.20 * (1.0 - progress))   # between 0.20 and 0.05

        candidate = (1.0 - t) * f + t * g
        candidate = _project_simplex(candidate)

        cand_val = _objective_value(candidate)
        if cand_val < best_val - 1e-14:
            best_val = cand_val
            best_f = candidate.copy()
            f = candidate
            print(f"[LP {it:04d}] improved: {best_val:.6f} (t={t:.3f})")
        else:
            # try a smaller mixing factor before giving up this iteration
            improved = False
            for factor in (0.5, 0.25, 0.1):
                t2 = t * factor
                candidate = (1.0 - t2) * f + t2 * g
                candidate = _project_simplex(candidate)
                cand_val = _objective_value(candidate)
                if cand_val < best_val - 1e-14:
                    best_val = cand_val
                    best_f = candidate.copy()
                    f = candidate
                    improved = True
                    print(f"[LP {it:04d}] improved with reduced t={t2:.3f}: {best_val:.6f}")
                    break
            if not improved:
                # optional jitter to escape a stall
                if it % 15 == 0:
                    noise = np.random.normal(scale=0.01, size=n)
                    f = np.maximum(f + noise, 0.0)
                    f = _project_simplex(f)
        it += 1
    return best_f


# ----------------------------------------------------------------------
# Simple stochastic local polishing (mass‑transfer)
# ----------------------------------------------------------------------
def _local_refine(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Perform many cheap random mass‑transfer moves, accepting only
    improvements (no simulated‑annealing temperature).  This is a fast
    final polish after the LP stage.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = _objective_value(best)
    b = _full_convolution(a)
    cur_val = best_val

    it = 0
    while time.time() < deadline:
        # index where the current convolution attains its maximum
        k_max = int(np.argmax(b))

        # source index i – proportional to its contribution to the peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] < 1e-12:
            it += 1
            continue

        # target index j – favour positions that hardly affect the peak
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        probs_j = w / w.sum()
        j = int(np.random.choice(n, p=probs_j))

        if i == j:
            it += 1
            continue

        # move a tiny fraction of mass
        delta = a[i] * np.random.rand() * 0.02   # 2 % of source
        if delta <= 0:
            it += 1
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_simplex(a_new)          # keep on simplex

        new_val = _objective_value(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            b = _full_convolution(a)
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = a.copy()
                print(f"[Polish {it:06d}] new best = {best_val:.6f}")

        # periodic re‑normalisation to curb numerical drift
        if it % 2000 == 0:
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = _objective_value(a)

        it += 1
    return best


# ----------------------------------------------------------------------
# Small final polishing – blend with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a tiny share of the uniform distribution."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

# ----------------------------------------------------------------------

    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # problem size – a few thousand entries is a good trade‑off
    n = 2000

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    print("[Init] parametric sweep")

    print(f"[Init] best after sweep = {val:.6f}")
