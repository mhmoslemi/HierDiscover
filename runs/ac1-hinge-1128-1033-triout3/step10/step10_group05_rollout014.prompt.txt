# Convolution utilities

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)-1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _deterministic_refine(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Greedy moves that analytically reduce the current maximal convolution entry.
    Each move transfers mass from the index that contributes most to the
    maximal entry to an index with the smallest contribution.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_val = cur_val
    best_local_a = a.copy()

    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contribution of each i to the maximal entry
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]

        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break  # nothing left to move

        # pick target j with smallest a (heuristic)
        j_candidates = np.argsort(a)
        for cand in j_candidates:
            if cand != i:
                j = int(cand)
                break

        # contributions of i and j to the maximal entry
        a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
        if a_k_i <= a_k_j + 1e-12:
            # cannot reduce the maximal entry with this pair
            break

        # second‑largest convolution entry (excluding k_max)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max == -np.inf:
            second_max = 0.0

        # analytic delta that would bring b[k_max] down to second_max
        denom = 2.0 * (a_k_i - a_k_j) + 1e-12
        delta_needed = (b[k_max] - second_max) / denom
        delta = min(delta_needed * 0.9, a[i])          # safety factor 0.9
        if delta <= 1e-12:
            break

        # incremental convolution update (store a copy for possible revert)
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        if new_val < cur_val - 1e-14:
            cur_val = new_val
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                if best_local_val <= 1.503:
                    break
        else:
            # revert – move was not beneficial
            b = b_old
            a[i] += delta
            a[j] -= delta
            break

        it += 1
        if it % 2000 == 0:
            print(f"[Refine {it:7d}] best = {best_local_val:.6f}")

    return best_local_a, best_local_val

# Small post‑processing: mix a tiny amount of uniform mass

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a little uniform mass (often helps)."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):

        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
