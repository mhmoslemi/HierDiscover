    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idx_list:

The algorithm:
1.  Initialise a simplex vector from a small grid of Beta‑shaped

    Dirichlet draws, keeping the best.
2.  Repeatedly apply an *entropy mirror‑descent* step that moves the
    probability vector in the direction of the sub‑gradient of the
    worst (largest) convolution entry.  The step size and a
    Metropolis temperature are annealed.
3.  If no improvement is seen for a while, perturb the current best
    with a small Gaussian noise and continue.
4.  After the time budget the best vector is returned and evaluated

The sub‑gradient of the maximum convolution entry k is
    g_i = a_{k‑i}           (i = k‑i)
    g_i = 2·a_{k‑i}         (i ≠ k‑i)

# Helper: full linear convolution using FFT (length 2·n‑1)

    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

# ----------------------------------------------------------------------

    b = _full_convolution(a)
    return float(2.0 * n * b.max())

# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Beta‑family generator (normalised)
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑shaped probability vector of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of the maximum convolution entry (or entries)
# ----------------------------------------------------------------------
def _max_conv_subgradient(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max_k (a*a)[k] for the supplied indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        i = np.arange(low, high + 1, dtype=np.int64)
        j = k - i
        # mask where i == j (diagonal term)
        diag = i == j
        # diagonal contribution: derivative = a_i
        g[i[diag]] += a[i[diag]]
        # off‑diagonal contribution: derivative = 2·a_j
        g[i[~diag]] += 2.0 * a[j[~diag]]
    return g

    else:

        T = T_start * (T_end / T_start) ** prog

        # exact sub‑gradient of the worst entry
        grad = _max_conv_subgradient(a, max_idxs)

        # propose mirror‑descent update
        a_cand = _mirror_step(a, grad, eta)

        # evaluate candidate
        conv_cand = _full_convolution(a_cand)
        cand_max_val = conv_cand.max()
        cand_val = float(2.0 * n * cand_max_val)

        # Metropolis acceptance (allows occasional uphill moves)
        accept = False
        if cand_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(cand_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_cand
            cur_val = cand_val

        # --------------------------------------------------------------
        # Stagnation handling – Gaussian perturbation around the best
        # --------------------------------------------------------------
        if stagnation > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            stagnation = 0
            conv_tmp = _full_convolution(a)
            print("[reset] perturbation – tmp val={:.6f}".format(
                float(2.0 * n * conv_tmp.max())))

        iteration += 1
        # safeguard against accidental endless loops
        if iteration > 200_000:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation and return the best sequence found
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
