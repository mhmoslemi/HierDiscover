```
def _project_simplex(v):
    n = v.shape[0]
    # standard algorithm from Duchi et al 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Gradient‑descent + soft‑max refinement for the step‑function problem.

1. Perform a very broad parametric sweep (exponential, power‑law, beta,
   arcsine, sine‑power, piecewise‑constant, Dirichlet draws) to obtain a
   good seed (Σ a_i = 1).  The sweep is deliberately generous – many random
   families are tried – because the later optimisation is sensitive to the
   starting shape.

2. Repeatedly reduce the L∞‑norm of the self‑convolution using a smooth
   approximation of the max (soft‑max).  At each iteration we:
   * compute the exact convolution a∗a,
   * build weights w_k = exp(λ·(b_k – max(b))) that concentrate on the
     current peak and slowly sharpen (λ grows from 5 to ≈200),
   * obtain a gradient g = 2·(w ∗ a) centred on the original length
     (the analytic sub‑gradient of Σ w_k b_k),
   * take a projected‑gradient step (step size decays linearly with time)
     and project back onto the probability simplex.
   Tiny random perturbations are injected after long periods without
   improvement to escape plateaus.

3. A very short simulated‑annealing phase is kept as a “polish” – it can
   still find improvements that the gradient step misses.

4. Finally we blend the best candidate with a tiny amount of the uniform
   distribution; this inexpensive step often lowers the peak a little
   further.

All calculations use exact FFT‑based convolutions, are fully vectorised
and respect the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities – exact, FFT based
# ----------------------------------------------------------------------
def full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of x with y using real FFT.
    Length = len(x) + len(y) - 1.
    """
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[:n]
    return conv


def autocorrelation(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return full_convolution(a, a)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto { w ≥ 0, Σ w = 1 } (Duchi et al., 2008)."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u > cssv / (np.arange(len(v)) + 1))[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – broad parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """
    Search a rich collection of parametric families for a low‑objective seed.
    Returns a pair (best_a, best_val) with Σ a = 1 and best_val = 2·n·max(b).
    """
    best_a = np.full(n, 1.0 / n)
    best_val = 2.0 * n * autocorrelation(best_a).max()

    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = 2.0 * n * autocorrelation(a).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                           # 1 … n

    # 1. Exponential tails (left‑hand and symmetric)
    for r in np.linspace(0.995, 0.999999, 20):
        _try(r ** i)                       # decreasing
        _try(r ** i + r ** (n - 1 - i))    # symmetric U‑shape

    # 2. Power‑law tails (left‑hand and symmetric)
    for p in np.linspace(0.1, 3.0, 30):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Inverse‑sqrt (special case of power‑law)
    _try(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))

    # 4. Beta densities – finer grid
    beta_vals = np.linspace(0.05, 2.0, 25)          # finer than before
    for a_param in beta_vals:
        for b_param in beta_vals:
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 5. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # 6. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 15):
        _try(base_sin ** p)

    # 7. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.002, 0.08, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.03, 0.45, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 8. Random Dirichlet draws for diversity (more draws)
    rng = np.random.default_rng(123456)
    for _ in range(200):
        _try(rng.dirichlet(np.ones(n)))

    # 9. Interpolation from a previously found solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – soft‑max gradient descent on the peak of a∗a
# ----------------------------------------------------------------------
def _gradient_descent(a0: np.ndarray, n: int,
                      start: float, deadline: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smoothed version of the max‑norm.
    Uses a soft‑max weight w_k = exp(λ·(b_k - max_b)) that slowly sharpens.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    b = autocorrelation(a)
    best_val = 2.0 * n * b.max()
    lam = 5.0                     # initial soft‑max temperature
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 30000        # after this many idle steps re‑perturb

    while time.time() < deadline:
        it += 1
        # --- compute convolution and objective -------------------------
        b = autocorrelation(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        # record best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[GD {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # --- increase λ slowly (sharpen the soft‑max) -----------------
        lam = min(200.0, lam * 1.008)   # exponential growth, capped

        # --- soft‑max weights                                          
        # subtract max_b to avoid overflow; all exponents ≤ 0
        w = np.exp(lam * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to one‑hot at the current peak
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # --- gradient of Σ w_k b_k (≈ smoothed max) --------------------
        conv_wa = full_convolution(w, a)               # length = 3n‑2
        start_idx = n - 1
        g = 2.0 * conv_wa[start_idx:start_idx + n]     # size n

        # --- step‑size schedule (linear decay) ------------------------
        progress = (time.time() - start) / (deadline - start)
        step_scale = 0.6 * (1.0 - progress) + 0.02 * progress   # from 0.6 → 0.02
        grad_norm = np.linalg.norm(g)
        if grad_norm == 0.0:
            # add a tiny random nudge
            a += np.random.normal(scale=1e-4, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            continue
        # step is larger when norm is small to keep absolute change ≈ step_scale
        step = step_scale / grad_norm

        # --- gradient descent step + projection -------------------------
        a_new = a - step * g
        a = _project_simplex(a_new)

        # --- occasional random perturbation to escape stagnation -------
        if no_improve >= IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)   # modest noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = autocorrelation(a)   # refresh
            no_improve = 0
            print(f"[GD {it:7d}] perturbation after stagnation")

    return best_a, best_val

# Small final polishing – blend with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with the uniform distribution; cheap post‑process."""
    best_a = a.copy()
    best_val = 2.0 * n * autocorrelation(a).max()
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)  # ensure exact simplex
        val = 2.0 * n * autocorrelation(cand).max()
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Mix] alpha={alpha:.3f} improved to {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Optional simulated‑annealing phase (kept as a cheap polish)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Very light SA – identical to the reference implementation but with a
    tighter iteration budget.  It is called only after gradient descent.
    """
    a = a0.copy()
    b = autocorrelation(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = best_a.copy()
    best_local_val = best_val

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T0 = 0.02 * cur_val
        T = max(T0 * (1.0 - progress), 1e-12)

        k_max = int(np.argmax(b))

        # source index – weighted by contribution to the current peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # target index – favour low contribution to the peak
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0.0:
            probs_j = w / w.sum()
            j = int(np.random.choice(n, p=probs_j))
        else:
            j = (i + np.random.randint(1, n)) % n

        if j == i:
            continue

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (exact)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        elif np.random.rand() < math.exp(-(new_val - cur_val) / T):
            accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_local_val - 1e-12:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = autocorrelation(a)
            cur_val = 2.0 * n * b.max()

        if no_improve > IMPROVE_LIMIT:
            a = best_local_a.copy()
            a += np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = autocorrelation(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val

    n = 9000

    # ------------------------------------------------------------------

    # Phase 2 – brief simulated‑annealing polishing (optional)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_a, best_val = _simulated_annealing(
            best_a, n,
            start=start_time,
            deadline=deadline,
            best_a=best_a,
            best_val=best_val,
            target_val=1.503,
        )
        print(f"[SA] best after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – tiny uniform mixing polish
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Logging via the provided wrapper (not part of the optimisation)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
