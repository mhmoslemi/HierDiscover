        a0 /= a0.sum()
        val = _objective(a0)
        best_seq = a0.copy()
        best_score = val
        print("[init] loaded previous solution:", best_score)
    except Exception:
        pass

    # Phase 1: random shapes
    rng = np.random.default_rng()
    # allocate 20% of time for searching random shapes
    shape_deadline = start + budget_s * 0.2
    while time.time() < shape_deadline and time.time() < deadline - 30.0:
        cand = _random_shape(n, rng)
        if cand is None:
            continue
        val = _objective(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[rand shape] new best {best_score:.6f}")

    # Phase 2: multi‑restart gradient descent
    # number of restarts based on remaining time
    remaining = deadline - time.time()
    # allocate at least 5 restarts, each ~remaining/5 seconds
    restart_budget = remaining / max(5, int(remaining // 30))  # approximate per restart
    # but cap restarts to e.g., 10
    max_restarts = 10
    restarts = 0
    while time.time() < deadline and restarts < max_restarts:
        restarts += 1
        # random init
        init_type = rng.choice(['beta','dirichlet','uniform'])
        if init_type == 'beta':
            p = np.exp(rng.uniform(np.log(1e-3), np.log(10)))
            q = np.exp(rng.uniform(np.log(1e-3), np.log(10)))
            a = _beta_shape(p,q,n)
        elif init_type == 'dirichlet':
            alpha = np.exp(rng.uniform(np.log(0.01), np.log(10)))
            a = rng.dirichlet(np.full(n, alpha))
        else:
            a = np.full(n, 1.0/n)

        if a is None:
            a = np.full(n, 1.0/n)
        a = a.astype(np.float64)

        # perform projected (or exponentiated) subgradient descent for this restart
        # schedule
        t0 = time.time()
        max_iter = 200000  # upper bound; we'll break by time
        step_start = 0.8
        step_end = 1e-5
        beta_mom = 0.0
        a_prev = a.copy()
        # use Nesterov momentum on the variable a directly (additive)
        for it in range(max_iter):
            now = time.time()
            if now >= deadline:
                break
            # momentum predictor
            beta = it / (it + 3.0)  # classic Nesterov weight
            y = a + beta * (a - a_prev)
            # compute conv of y
            b = _full_convolution(y)
            max_b = b.max()
            cur_val = 2.0 * n * max_b
            if cur_val < best_score:
                best_score = cur_val
                best_seq = y.copy()
                print(f"[restart {restarts}] iter {it} new best {best_score:.6f}")
                if best_score <= 1.503:
                    break

            # gradient of max entry (use the first max index)
            k = int(np.argmax(b))
            # gradient
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(y)
            grad[mask] = 2.0 * y[idx[mask]]

            # step size
            progress = (now - start) / (deadline - start)
            eta = step_start * (1 - progress) + step_end * progress

            # projected gradient step
            a_new = y - eta * grad
            a_new = np.maximum(a_new, 0.0)
            sum_a = a_new.sum()
            if sum_a > 0:
                a_new /= sum_a
            else:
                a_new = np.full_like(a_new, 1.0 / n)

            # occasional jitter
            if rng.random() < 0.001:
                a_new = a_new + rng.normal(scale=0.001, size=n)
                a_new = np.clip(a_new, 0, None)
                a_new /= a_new.sum()

            a_prev = a
            a = a_new

        # after each restart, optionally break if best_score already low
        if best_score <= 1.503:
            break

    # final evaluation
    final_seq = best_seq.tolist() if best_seq is not None else np.full(n, 1.0/n).tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print("[final] evaluate_sequence =", final_score)
    except Exception as e:
        print("[final] error:", e)

"""
Propose a step‐function (non‑negative vector) that minimises
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))**2

Algorithm (≈1000 s budget)

    For each family we keep the best (small‑n) parameters.
    Afterwards the best parameters are up‑scaled to the target length
    n≈8000 and evaluated – this gives a strong starting point.

2️⃣  **Multi‑restart Nesterov‑accelerated projected sub‑gradient descent**
    on the exact objective for the full length.
    * At each iteration we compute the full linear convolution via FFT.
    * The index k where the maximum occurs yields a sub‑gradient
      g_i = 2·a_{k‑i} (zero where the shift is out of range).
    * Nesterov extrapolation y = a + β·(a‑a_prev) with β = it/(it+3).
    * Projected step: a_new = max(0, y‑η·g); renormalise to Σa=1.
    * Step size η decays linearly from 0.8 to 1e‑5, jitter is added

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1) via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv(a,a)) – assumes Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Shape factories (parameterised families)
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Beta‑type shape a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    return a / total


def _power_shape(p: float, n: int) -> np.ndarray | None:
    """Power‑law shape a_i ∝ i^{p‑1} (monotone, includes uniform as p=1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (p - 1.0)
    total = a.sum()
    if total <= 0.0:
        return None
    return a / total


def _sine_shape(exp: float, n: int) -> np.ndarray | None:
    """Sine‑power shape a_i ∝ sin(π·(i+½)/n)^{exp} (symmetric, bell‑like)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** exp
    total = a.sum()
    if total <= 0.0:
        return None
    return a / total


def _sinc_shape(width: float, n: int) -> np.ndarray | None:
    """Scaled sinc shape a_i ∝ |sinc(width·(i‑mid)/n)|.
    ``width`` controls the spread – larger → sharper centre."""
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.abs(np.sinc(width * i / n))
    total = a.sum()
    if total <= 0.0:
        return None
    return a / total

# Main entry point required by the evaluation framework

def propose_candidate(seed: int = 42,
