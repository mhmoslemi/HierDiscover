"""
Propose a step‑function that (hopefully) beats the current record
for the “max‑convolution” evaluation problem.

Algorithm
---------
1. **Rich initial sweep** – evaluate many parametric families
   (exponential, power, beta, sine‑power, logistic, piecewise‑constant)
   together with a large batch of Dirichlet draws with various
   concentration parameters.  Keep the best normalised candidate.

2. **Adam on a log‑sum‑exp surrogate** – minimise a smooth
   approximation of the max‑convolution.  Temperature (ε) is annealed
   from 1.0 to 1e‑13 while the learning‑rate decays from 0.5 to 0.001.
   The gradient is obtained analytically via FFT‑based convolution.

3. **Simulated‑annealing hill‑climb** – repeatedly move a small
   fraction of mass from indices that contribute most to the current
   peak towards indices that contribute least.  Moves are accepted
   either if they improve the true objective or with a Metropolis
   probability that shrinks over time.

4. **Projected sub‑gradient descent** – a classic sub‑gradient method
   on the exact max‑convolution, projected onto the probability simplex.

5. **Exact active‑set refinement** – compute the set of near‑maximal
   convolution entries, form the weight vector w_i=Σ_{k∈active}a_{k−i},
   and shift mass from the index with largest w_i to the one with
   smallest w_i.  The step size is adaptively decreased.

6. **Uniform mixing** – a final tiny convex combination with the uniform
   distribution often smooths remaining spikes.

All phases run for a fraction of the 1000 s budget, stopping early if
the target 1.503 is reached.  The sequence is always kept normalised
to sum = 1, guaranteeing that the evaluation function is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective – exact value (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Build a diverse pool of candidate shapes
# ----------------------------------------------------------------------
def _initial_candidate(n: int, seed: int = 0) -> tuple[np.ndarray, float]:
    """Return the best vector among many parametrised families."""
    rng = np.random.default_rng(seed)
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _maybe_update(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential families
    for r in np.linspace(0.995, 0.99999, 14):
        _maybe_update(r ** i)                         # decreasing
        _maybe_update(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 6.0, 18):
        _maybe_update(i1 ** (-p))
        _maybe_update(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.2, 6.0, 7):
        for b_param in np.linspace(0.2, 6.0, 7):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _maybe_update(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _maybe_update(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.005, 0.15, 12):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _maybe_update(a)

    # 6. Logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.003, 0.04, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _maybe_update(a)
        _maybe_update(np.flip(a))

    # 7. Random Dirichlet draws with varied concentration
    alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for _ in range(500):
        alpha = float(rng.choice(alphas))
        a = rng.dirichlet(alpha * np.ones(n))
        _maybe_update(a)

    # 8. If a previous solution exists, interpolate it
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam optimiser on a smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smooth approximation of max(conv)."""
    n = a0.shape[0]
    # work in logits to enforce positivity
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.001
    eps_start, eps_end = 1.0, 1e-13

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80_000
    it = 0

    while time.time() < deadline:
        # Map logits to simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted - np.max(shifted))          # improve stability
        p = weights / weights.sum()                         # soft‑max over positions

        # ----- gradient wrt a -----
        p_rev = p[::-1]
        # convolution a * p_rev (full length 3n‑2)
        n_a = n
        n_p = p_rev.shape[0]
        out_len = n_a + n_p - 1
        size = 1 << (out_len - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv = np.fft.irfft(fa * fp, size)[:out_len]
        # central segment that corresponds to ∂/∂a_i
        grad_a = 4.0 * n * conv[n_a - 1: 2 * n_a - 1][::-1]

        # back‑prop through logits (softmax)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.02, size=n)

        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _annealing_hillclimb(a_start: np.ndarray, start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Hill‑climb with a Metropolis acceptance rule."""
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T0 = best_val * 0.2                     # initial temperature
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 300_000

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T0 * (0.01 ** progress)        # exponential cooling

        # identify the active (max) convolution index
        k_max = int(np.argmax(b))

        # contribution of each index to the active entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – proportional to its contribution
        if total_contrib > 0:
            i = int(np.random.choice(n, p=contrib / total_contrib))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # target index – prefers indices that *do not* help the max entry
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0:
            j = int(np.random.choice(n, p=w / w.sum()))
        else:
            j = (i + np.random.randint(1, n)) % n

        # propose a modest transfer
        delta = a[i] * np.random.rand() * 0.08          # up to 8 % of source mass
        if delta <= 0.0:
            continue

        # ----- incremental update of convolution -----
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.6 / a.shape[0]
    eta = eta0
    decay = 0.9994
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * a.shape[0] * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient (active index)
        idx = k_max - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # decay step size
        eta = max(eta * decay, 1e-12)

        if no_improve > IMPROVE_LIMIT:
            # shake and restart
            noise = np.random.normal(scale=0.012, size=a.shape[0])
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase D – Exact active‑set refinement
# ----------------------------------------------------------------------
def _refine_exact_phase(a0: np.ndarray, start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Active‑set refinement moving mass from heavy to light weighted indices."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    step_factor = 0.5            # initial fraction of source mass to move
    decay = 0.9997
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 180_000
    MIN_STEP = 1e-12

    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * a.shape[0] * max_b

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Refine {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # give up if no progress for a long time
            break

        # ---- active set (indices within a tiny tolerance of the peak) ----
        eps = max_b * 5e-7
        active = np.where(b >= max_b - eps)[0]
        if active.size == 0:
            break

        # weight w_i = Σ_{k∈active} a_{k‑i}
        n = a.shape[0]
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))

        if src == tgt or a[src] <= MIN_STEP:
            # fall back to a random move
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        delta = a[src] * step_factor
        if delta < MIN_STEP:
            step_factor *= 0.5
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        b_new = _full_convolution(a_candidate)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val, best_local_a = new_val, a.copy()
        else:
            step_factor *= 0.7       # shrink move size on rejection

        step_factor = max(step_factor * decay, MIN_STEP)
        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass – useful for a final tiny tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(a.shape[0], 1.0 / a.shape[0])
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point – the function invoked by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the supplied time budget.
    Returns a list of floats (Σ = 1) that (hopefully) beats the current record.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 9000

    # --------------------------------------------------------------
    # 0️⃣ Initial parametric sweep + random Dirichlet sampling
    # --------------------------------------------------------------
    print("[Init] starting parametric sweep")
    best_a, best_val = _initial_candidate(n, seed=seed)
    print(f"[Init] best after sweep = {best_val:.6f}")

    TARGET = 1.503   # ambitious target

    # --------------------------------------------------------------
    # 1️⃣ Adam on a smooth surrogate (≈20 % of budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        best_a,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase A] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Simulated‑annealing hill‑climb (≈30 % of budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.50 * (deadline - start_time)
    best_a, best_val = _annealing_hillclimb(
        best_a,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase B] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Projected sub‑gradient descent (≈20 % of budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(
        best_a,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase C] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Exact active‑set refinement (remaining time)
    # --------------------------------------------------------------
    best_a, best_val = _refine_exact_phase(
        best_a,
        start=phase_c_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase D] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣ Final polishing – tiny uniform mixing
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6️⃣ Report final score using the provided secure wrapper
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
