"""
propose_candidate – minimise 2·n·max(conv(a,a)) with sum(a)=1.

The algorithm proceeds in three stages:

1. **Initial seeding** – Re‑use a previously found sequence if available,
   otherwise generate a fleet of parametric candidates (exponential,
   power‑law, cosine‑window, arcsine‑shaped and Dirichlet draws) and pick the
   best one according to the exact objective.

2. **Soft‑max gradient descent on the simplex** –  
   The true objective is  max_k (a∗a)_k.  We replace the max by a smooth
   approximation φ_λ(a)= (1/λ)·log Σ_k exp(λ·b_k) (b = a∗a) and perform
   exponentiated‑gradient descent on the probability simplex.
   The gradient of φ_λ is
       ∇φ_λ(i) = 2· Σ_{j=0}^{n‑1} a_j·p_{i+j},
   where p is the soft‑max distribution over the convolution entries.
   This can be evaluated efficiently via a single FFT‑based convolution.
   The temperature λ is increased slowly so the method first follows a
   smooth landscape and later concentrates on the true maximum.

3. **Tiny uniform mixing** –  A final pass mixes the best vector with a
   tiny amount of the uniform distribution; this occasionally reduces the
   peak value further.

All operations respect the 1000 s time budget, use only NumPy and run on a
single CPU.  The returned list sums to 1, contains only non‑negative
values and can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a*a using FFT (real-valued)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _grad_softmax_convolution(a: np.ndarray, lam: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the gradient of the soft‑max smoothed max‑convolution
    together with the raw convolution vector b = a∗a.

    Returns (grad, b) where
        grad_i = 2·Σ_j a_j·p_{i+j},
    p = softmax(λ·b) and b = full_convolution(a).
    """
    n = a.shape[0]

    # 1) convolution b = a*a
    b = _full_convolution(a)
    max_b = b.max()

    # 2) soft‑max weights over the 2n‑1 entries
    if lam > 0.0:
        # shift by max to avoid overflow
        exp_vals = np.exp(lam * (b - max_b))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0 or not np.isfinite(sum_exp):
            # fallback to pure max‑indicator
            p = np.zeros_like(b)
            p[np.argmax(b)] = 1.0
        else:
            p = exp_vals / sum_exp
    else:
        # pure max (λ = 0)
        p = np.zeros_like(b)
        p[np.argmax(b)] = 1.0

    # 3) gradient = 2 * conv(a, p_rev)[n‑1:2n‑1][::-1]
    p_rev = p[::-1]
    conv_len = len(a) + len(p_rev) - 1          # = 3n‑2
    size = 1 << (conv_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[:conv_len]

    grad = 2.0 * conv[n - 1 : 2 * n - 1][::-1]   # shape (n,)
    return grad, b


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple mixing with uniform (tiny α values)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for several α and keep the best."""
    best_a = a
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Families of parametric candidates (used when no previous solution)
# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, rng: np.random.Generator):
    """Yield a variety of candidate vectors built from simple families."""
    # 1. Exponential (U‑shaped) family
    i = np.arange(n, dtype=np.float64)
    for r in np.linspace(0.55, 0.9999, 30):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 2. Power‑law (edge‑peaked) family
    idx = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    for p in np.linspace(-4.0, 4.0, 30):
        base = idx * (n + 1 - idx) + 1e-12
        a = np.power(base, -p)
        yield a

    # 3. Cosine‑window family
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))   # Hann window
    for beta in np.linspace(0.5, 4.0, 20):
        a = np.power(w, beta)
        yield a

    # 4. Arcsine‑shaped (U‑shape, singular at the edges)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    yield a

    # 5. Random Dirichlet draws
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        yield a

    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final call

    # ------------------------------------------------------------------
    # Problem size – a few‑ten‑thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 12000  # feel free to adjust; 2·n‑1 ≈ 24000 FFT size ≈ 2ⁿⁱ⁶, cheap.

    # ------------------------------------------------------------------
    # Initialise a – reuse a known good construction if possible,
    # otherwise pick the best parametric candidate.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
        print("[init] Re‑used height_sequence_1")
    except Exception:
        best_a = None
        best_val = np.inf
        for cand in _generate_parametric_candidates(n, rng):
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s <= 0.0:
                continue
            cand = cand / s
            val = _objective_value(cand)
            if val < best_val:
                best_val = val
                best_a = cand.copy()
        if best_a is None:
            best_a = np.full(n, 1.0 / n)
            best_val = _objective_value(best_a)
        a0 = best_a
        print(f"[init] Parametric sweep best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Soft‑max (smooth max) exponentiated‑gradient descent
    # ------------------------------------------------------------------
    eps = 1e-12
    log_w = np.log(a0 + eps)                      # work in log‑space
    eta = 0.5                                      # initial learning rate
    decay = 0.99997                                # geometric decay
    lam = 1.0                                      # start temperature
    lam_inc = 1.02                                 # increase factor per iteration
    lam_max = 200.0                                # cap for numerical safety

    best_a = a0.copy()
    best_val = _objective_value(best_a)

    no_improve = 0
    IMPROVE_LIMIT = 60000
    it = 0

    while time.time() < deadline:
        # --- reconstruct probability vector ---
        max_log = np.max(log_w)
        w = np.exp(log_w - max_log)   # stabilise
        a = w / w.sum()

        # --- gradient and convolution ---
        grad, b = _grad_softmax_convolution(a, lam)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[{it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # --- exponentiated‑gradient update on log‑weights ---
        log_w -= eta * grad
        eta *= decay

        # --- temperature schedule (smooth → true max) ---
        if lam < lam_max:
            lam = min(lam * lam_inc, lam_max)

        # --- occasional jitter to escape plateaus ---
        if rng.random() < 0.003:
            log_w += rng.normal(scale=0.005, size=n)

        # --- stagnation handling ---
        if no_improve > IMPROVE_LIMIT:
            # re‑centre around the current best and perturb
            log_w = np.log(best_a + eps)
            log_w += rng.normal(scale=0.02, size=n)
            eta = 0.5
            no_improve = 0
            print(f"[{it:7d}] perturb after stagnation")

        it += 1

        # safety break
        if time.time() > deadline - 0.2:
            break

    # ------------------------------------------------------------------
    # Phase 2 – Tiny uniform mixing (often yields a final tiny boost)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final validation via the provided evaluate_sequence() call.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
