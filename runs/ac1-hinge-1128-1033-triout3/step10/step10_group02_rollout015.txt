def _sa_hillclimb(a0, n, deadline):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0*n*b.max()
    best_local_a = a.copy()
    best_local_val = cur_val

    # start temperature
    T_start = 0.1
    T_end = 1e-6

    while time.time() < deadline:
        progress = (time.time() - start_time) / total_hill_time
        T = T_start * (T_end/T_start)**progress

        # Choose source i - heavy contributors
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = np.argmax(a)

        # target j - low contributions
        # Use inverse of contrib but ensure not i.
        low_mask = np.copy(mask)
        low_mask[i] = False
        inv = np.zeros_like(a)
        inv[low_mask] = 1.0 / (contrib[low_mask] + 1e-12)
        sum_inv = inv.sum()
        if sum_inv > 0:
            j = rng.choice(n, p=inv / sum_inv)
        else:
            # choose random other index
            j = rng.choice([idx for idx in range(n) if idx != i])

        # choose delta as fraction of a[i]
        max_frac = 0.1  # can be d
        delta = a[i] * rng.random() * max_frac
        if delta <= 0:
            continue

        # incremental update same as earlier _hillclimb
        # compute new b efficiently using slices
        b_new = b.copy()
        b_new[j:j+n] += 2.0*delta*a
        b_new[i:i+n] -= 2.0*delta*a
        b_new[2*j] += delta*delta
        b_new[2*i] += delta*delta
        b_new[i+j] -= delta*delta

        new_max = b_new.max()
        new_val = 2.0*n*new_max

        accept = False
        if new_val < cur_val:
            accept=True
        else:
            if T>0 and np.exp(-(new_val-cur_val)/T) > rng.random():
                accept=True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            # store best
            if cur_val < best_local_val:
                best_local_val = cur_val
                best_local_a = a.copy()
    return best_local_a, best_local_val
```

"""
Step‑function optimizer for the “evaluate_sequence” attack.

The algorithm works in three stages within the 1000 s budget:

1️⃣  A broad random‑initialisation phase that samples a few
    thousand simple parametric families (uniform, exponential,
    power‑law, Beta) and keeps the best candidate.

2️⃣  A smooth‑max Adam optimisation that minimises a
    log‑sum‑exp approximation of max(conv(a,a)).  The Adam run
    occupies roughly a quarter of the total time and uses a
    decreasing smoothing parameter ε.

3️⃣  A simulated‑annealing hill‑climber that moves tiny chunks of
    mass from indices that contribute most to the current maximal
    convolution entry towards indices that contribute the least.
    The temperature and maximal move size are annealed and the
    best solution from this phase is fed to a short sub‑gradient
    polishing pass.

Finally the best sequence is evaluated with ``evaluate_sequence`` and
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ------------------------------------------------------------
# Exact objective (assumes sum(a)==1)
# ------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes 2·n)
# ------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient
    of 2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # already includes 2·n factor
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ------------------------------------------------------------
# Projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Adam optimisation of the smooth‑max (no variance penalty)
# ------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex for the loss 2·n·smooth_max."""
    # log‑softmax parameters (always stay on the simplex)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.003          # a slightly larger start LR
    eps_start, eps_end = 1e-1, 1e-7        # start smoothing fairly high

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    while time.time() < deadline:
        # decode logits
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[Adam] new best = {best_val:.6f}")

        # schedule learning‑rate & ε
        prog = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog

        # gradient of smooth max
        _, grad = _smooth_max_and_grad(a, eps)   # already contains 2·n factor

        # chain rule through soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid exact stagnation
        if np.random.rand() < 0.02:
            w += np.random.normal(scale=0.02, size=n)

    return best_a, best_val


# ------------------------------------------------------------
# Simulated‑annealing hill‑climber (fast O(n) updates)
# ------------------------------------------------------------
def _sa_hillclimb(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_a: np.ndarray,
                  best_val: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search that repeatedly moves a small
    amount of mass from a “high‑contribution’’ index to a
    “low‑contribution’’ index.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules (more aggressive at the start)
    max_frac_start, max_frac_end = 0.5, 0.001
    temp_start, temp_end = 0.05, 1e-7

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index (probability proportional to contribution)
        if total > 0.0:
            src_probs = contrib / total
            i = rng.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – bias towards low contribution
        tgt_weights = np.zeros_like(a)
        # avoid division by zero and ignore the source itself
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            # fallback: any index except i
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        # amount to move
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (vectorised, O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val

            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[SA]   new best = {best_val:.6f}")

    return best_a, best_val


# ------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact objective."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (more aggressive start)
        prog = (now - start_time) / (deadline - start_time)
        step = 0.35 * (1.0 - prog) + 5e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # tiny jitter to escape flat regions
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ------------------------------------------------------------
# Candidate generators
# ------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return a normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _power_candidate(n: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-exponent}, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exponent)
    return _project_simplex(a)


def _exponential_candidate(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return _project_simplex(a)

        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # uniform fallback in case nothing else succeeded
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print(f"[Init] fallback uniform, value={best_val:.6f}")

    # a quick sweep over a few families
    # ------------------------------------------------------------
    # Random Beta draws
    # ------------------------------------------------------------
    for _ in range(200):
        cand = _random_beta_candidate(n, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] random Beta improved value={best_val:.6f}")

    # ------------------------------------------------------------
    # Power‑law candidates (i^{-p})
    # ------------------------------------------------------------
    for p in np.linspace(0.0, 5.0, 26)[1:]:   # skip p=0 (uniform)
        cand = _power_candidate(n, p)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] power‑law p={p:.2f} improved value={best_val:.6f}")

    # ------------------------------------------------------------
    # Exponential candidates (exp(-λ i))
    # ------------------------------------------------------------
    for lam in np.linspace(0.0, 5.0, 26)[1:]:
        cand = _exponential_candidate(n, lam)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Init] exponential λ={lam:.2f} improved value={best_val:.6f}")

    # ------------------------------------------------------------
    # Exhaustive Beta grid (log‑space)
    # ------------------------------------------------------------
    logp_vals = np.linspace(-9.2, 2.3, 30)
    logq_vals = np.linspace(-9.2, 2.3, 30)
    i_arr = np.arange(1, n + 1, dtype=np.float64)
    for lp in logp_vals:
        p = np.exp(lp)
        for lq in logq_vals:
            q = np.exp(lq)
            a = (i_arr ** (p - 1.0)) * ((n - i_arr + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_a, best_val = a, val
                print(f"[Init] Beta p={p:.4f} q={q:.4f} improved value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈25 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.25 * budget_s
    best_a, best_val = _adam_optimize(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climb (≈45 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * budget_s
    best_a, best_val = _sa_hillclimb(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        rng=rng,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=deadline,
        start_time=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        max_iters=300_000,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure the output is a plain list of floats
    return best_a.tolist()
