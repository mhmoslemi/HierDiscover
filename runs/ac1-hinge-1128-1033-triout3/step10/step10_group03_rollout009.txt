"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence().  The algorithm works in three phases:

1️⃣  Broad search over several expressive families (Beta, exponential,
    Gaussian, power‑sine, triangular).  Candidates are normalised to
    Σa=1 and the best few are kept.

2️⃣  Refine the current best candidate with a mirror‑descent style
    optimiser on the exact objective
        f(a)=2·n·max(conv(a,a))
    – the sub‑gradient of the max‑convolution is built from the
      top‑L most active lags,
    – a momentum term (β≈0.9) stabilises the descent,
    – multiplicative (exponentiated) updates keep the iterates in the
      interior of the simplex,
    – occasional Gaussian jitter helps to escape flat plateaus.

3️⃣  The finally best sequence is evaluated with the provided
    ``evaluate_sequence`` and returned.

The whole search respects the supplied time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of 2·n·max(conv(a,a)) assuming Σa = 1.
    Used only during the exploration phase for cheap comparison.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex Σx=1, x≥0.
    (fallback for additive updates or jitter handling.)
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape families for the initial wide exploration
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """a_i ∝ i^{p‑1}(n‑i+1)^{q‑1},   i = 1,…,n ."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(c: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·|i-mid|)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-c * np.abs(i - mid))
    a /= a.sum()
    return a


def _gaussian_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the middle."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _power_sine_shape(p: float, n: int) -> np.ndarray:
    """a_i ∝ sin(π·i/(n+1))^p ."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1.0)) ** p
    a /= a.sum()
    return a


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(i, n + 1 - i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution (active‑set version)
# ----------------------------------------------------------------------


def _subgradient_max(a: np.ndarray,
                     b: np.ndarray,
                     max_active: int = 5,
                     rel_tol: float = 1e-6) -> np.ndarray:
    """
    Return a sub‑gradient of max(b) w.r.t. a.
    The active set consists of the `max_active` largest entries of b,
    optionally filtered by a relative tolerance.
    """
    n = a.shape[0]
    max_val = b.max()
    # pick the largest indices (may be fewer than max_active if ties)
    if max_active >= b.size:
        active = np.arange(b.size)
    else:
        # argpartition is O(n)
        active = np.argpartition(b, -max_active)[-max_active:]
        # keep only those sufficiently close to the global maximum
        mask = b[active] >= max_val * (1.0 - rel_tol)
        if not np.any(mask):
            # fall back to the absolute maximiser
            active = np.array([int(np.argmax(b))])
        else:
            active = active[mask]

    grad = np.zeros_like(a)
    for k in active:
        # valid index range where a_{k-j} exists
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] += a[k - idx]
    grad *= 2.0
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    returns the best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed.
    # ------------------------------------------------------------------
    n = 8000

    best_seq: np.ndarray | None = None
    best_val = np.inf

    def _maybe_update(candidate: np.ndarray):
        """Update global best if candidate improves the exact objective."""
        nonlocal best_seq, best_val
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[search] new best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously saved solution (if any)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        # interpolate to the current resolution
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Wide exploration over several parametric families
    # ------------------------------------------------------------------
    # (a) dense Beta grid – already proven to be useful
    p_grid = np.concatenate(
        [
            np.geomspace(0.001, 0.1, 8, endpoint=False),
            np.geomspace(0.1, 5.0, 12, endpoint=False),
            np.geomspace(5.0, 20.0, 5),
        ]
    )
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is not None:
                _maybe_update(cand)

    # (b) massive random log‑uniform Beta draws
    rng = np.random.default_rng()
    log_min, log_max = np.log(1e-3), np.log(20.0)
    random_beta_trials = 12000
    for _ in range(random_beta_trials):
        p = np.exp(rng.uniform(log_min, log_max))
        q = np.exp(rng.uniform(log_min, log_max))
        cand = _beta_shape(p, q, n)
        if cand is not None:
            _maybe_update(cand)

        if time.time() >= deadline - 200.0:      # reserve ~200 s for refinement
            break

    # (c) exponential decay family
    for _ in range(2000):
        c = rng.uniform(0.0, 5.0)
        cand = _exp_shape(c, n)
        _maybe_update(cand)

    # (d) Gaussian family
    for _ in range(2000):
        sigma = rng.uniform(0.5, n / 5.0)
        cand = _gaussian_shape(sigma, n)
        _maybe_update(cand)

    # (e) power‑sine family
    for _ in range(2000):
        p = rng.uniform(0.1, 8.0)
        cand = _power_sine_shape(p, n)
        _maybe_update(cand)

    # (f) simple triangular window
    cand = _triangular_shape(n)
    _maybe_update(cand)

    # (g) sparse Dirichlet‑like samples (few non‑zero entries)
    for _ in range(800):
        # random support size between 10 and 1500
        support = rng.integers(10, 1500)
        idx = rng.choice(n, size=support, replace=False)
        vals = rng.random(support)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx] = vals
        cand /= cand.sum()
        _maybe_update(cand)

    # fallback – uniform if nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)
        print("[fallback] uniform start, value=", best_val)

    # ------------------------------------------------------------------
    # 2️⃣  Refine the current best using a mirror‑descent optimiser
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)          # momentum accumulator
    beta_mom = 0.9                # momentum coefficient
    it = 0
    no_improve = 0
    max_no_improve = 8000        # after this many stagnant iterations we jitter

    while time.time() < deadline:
        # ----- evaluate current state -----
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
            print(f"[optim] iter {it} new best {best_val:.6f}")
        else:
            no_improve += 1

        # ----- compute a sub‑gradient from the most active lags -----
        grad = _subgradient_max(a, conv, max_active=5, rel_tol=1e-6)

        # ----- momentum (exponential moving average) -----
        m = beta_mom * m + (1.0 - beta_mom) * grad

        # ----- step‑size schedule (linear decay) -----
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total
        step = 0.5 * (1.0 - progress) + 1e-5 * progress   # 0.5 → 1e‑5

        # ----- multiplicative (exponentiated) update -----
        log_factor = -step * m
        # keep exponentials in a numerically safe range
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a = a * np.exp(log_factor)
        # renormalise to stay on the simplex
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            # – fallback to a uniform perturbation if something went wrong
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # ----- occasional jitter to escape flat plateaus -----
        if np.random.rand() < 0.001 or no_improve >= max_no_improve:
            jitter = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            no_improve = 0
            # reset momentum to avoid dragging from the stale direction
            m.fill(0.0)

        it += 1

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:           # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
