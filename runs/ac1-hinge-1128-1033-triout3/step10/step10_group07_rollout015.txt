Algorithm (smooth‑max Adam on the simplex)
------------------------------------------------

def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow_two(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray, size_corr: int) -> np.ndarray:
    """
    Valid cross‑correlation (w ⋆ a) using FFT.

    Returns an array of length len(a) (=n) where
        (w ⋆ a)[i] = Σ_j w[j]·a[j+i]   (indices out of range give 0).
    """
    # reverse a for convolution → correlation
    a_rev = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    fa = np.fft.rfft(a_rev, size_corr)
    conv = np.fft.irfft(fw * fa, size_corr)
    n = a.shape[0]
    start = n - 1                     # first index of the “valid’’ part
    end = start + n
    return conv[start:end]


def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector (Σa=1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try tiny convex combinations with the uniform distribution."""
    best_a = a
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _simplex_projection(cand)   # re‑normalise
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


def _generate_parametric_candidates(n: int, rng: np.random.Generator):
    """
    Yield a few hand‑crafted families (exponential, power‑law, cosine window,
    Dirichlet draws) – used only if no previous solution is available.
    """
    # 1. Exponential (U‑shaped) family
    for r in np.linspace(0.55, 0.9999, 30):

        yield a

    # 4. Random Dirichlet draws
    for _ in range(30):

        yield a

    stagnation_limit = 3000
    best_a = a.copy()
    best_val = _objective_value(a)
    it = 0

    # size for the correlation FFT (needs to accommodate len(w)+len(a)-1)
    size_corr = _next_pow_two((2 * n - 1) + n - 1)

    while time.time() < deadline:
        # ---- convolution and exact objective ---------------------------------
        conv = _full_convolution_fft(a)
        max_b = conv.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            stagnation_counter = 0
            print(f"[it {it:7d}] new best = {best_val:.6f}")
        else:
            stagnation_counter += 1

        # ---- soft‑max weights ------------------------------------------------
        # w_k ∝ exp(β·(conv_k – max_b))   (all entries ≤ 0)
        w = np.exp(beta_s * (conv - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth objective (via correlation) ------------
        grad = 2.0 * _correlation_fft(w, a, size_corr)

        # ---- Adam update ----------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # ---- projection onto the simplex (non‑negative, Σ=1) ----------------
        a = _simplex_projection(a_new)

        # ---- enforce symmetry (helps to keep the convolution balanced) -----
        a = 0.5 * (a + a[::-1])

        # ---- occasional jitter to escape possible plateaus -----------------
        if stagnation_counter > stagnation_limit:
            jitter = rng.normal(scale=1e-3, size=n)
            a = _simplex_projection(a + jitter)
            stagnation_counter = 0
            print(f"[it {it:7d}] jitter injected after stagnation")

        # ---- gradually sharpen the soft‑max ---------------------------------
        beta_s = min(beta_s_max, beta_s * beta_inc)

        # safety break – leave a tiny margin for the final evaluation
        if time.time() > deadline - 0.3:
            break

    # --------------------------------------------------------------
    # Tiny uniform mixing (often yields a small extra gain)
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # --------------------------------------------------------------
    # Final log (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
