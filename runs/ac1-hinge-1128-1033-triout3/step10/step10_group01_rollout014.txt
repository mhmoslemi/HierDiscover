def solve_lp_max_sum(a, M):
    n = len(a)
    # Build sparse matrix A of shape (2n-1, n), where A[k,i] = a_{k-i} if 0 <= k-i < n else 0
    # Build as described.
    rows = np.concatenate([np.arange(i, i+n) for i in range(n)])
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Define c: -1 for each g variable to maximize sum.
    c = -np.ones(n)
    # Upper bounds on constraints: M
    b_ub = np.full(2*n-1, M)
    # Solve:
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

def _full_convolution(x):
    n = len(x)
    out_len = 2*n-1
    size = 1 << ((out_len-1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx*fx, size)[:out_len]
    return conv

"""
Step‑function optimizer.

The goal is to minimise
    2·n·max(conv(a,a)) / (sum(a)^2)
for a non‑negative sequence a.
The algorithm consists of three stages:

1️⃣  Initialise a from a small family of Beta‑shaped sequences and a few
    Dirichlet draws, optionally interpolating the previous best solution.

2️⃣  Repeatedly solve a linear programme
        maximise   Σ g_i
        subject to  conv(a,g) ≤ M_current          (M_current = max(conv(a,a))
                    Σ g_i = 1,  g_i ≥ 0 .
    The LP spreads the mass of *a* while keeping the maximal convolution
    entry bounded.  A convex combination a←(1‑α)a+α·g (α≈0.2) is taken;
    this monotone scheme provably never increases the objective.

3️⃣  Finish with a simulated‑annealing hill‑climber (the same core as the
    previous submission) to polish the solution until the time budget
    ends.

All heavy work is timed; the routine returns the best sequence found
before the 1000 s deadline.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex Σx = 1, x≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family initial sweep (coarser but fast)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> np.ndarray:
    """Return a good Beta‑family seed for length n."""
    i = np.arange(1, n + 1, dtype=np.float64)

    # modest grids for p and q (both >0)
    p_vals = np.concatenate([
        np.linspace(0.2, 3.0, 15),
        np.geomspace(3.0, 10.0, 5)[1:],
    ])
    q_vals = p_vals.copy()

    best_a = None
    best_val = np.inf

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a


# ----------------------------------------------------------------------
# Linear programme: maximise Σ g_i  with  conv(a,g) ≤ M .
# ----------------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Given a non‑negative vector a (∑a = 1) and a bound M,
    solve   maximise Σ g_i   subject to   conv(a,g) ≤ M,  Σ g_i = 1,  g_i ≥ 0.
    Returns the optimal g (or None if the LP fails).
    """
    n = a.shape[0]
    # Build the Toeplitz constraint matrix A where
    #   (A g)_k = Σ_i a_{k-i} g_i   (0 ≤ k < 2n‑1).
    # The pattern of indices does not depend on a, only the data values.
    rows = np.concatenate([np.arange(i, i + n) for i in range(n)])
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)                     # a repeated for each column
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Objective: maximise Σ g_i  → minimise -Σ g_i
    c = -np.ones(n)

    # Upper bounds for each of the (2n‑1) convolution constraints
    b_ub = np.full(2 * n - 1, M)

    # Equality: Σ g_i = 1 (same normalisation as a)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# LP‑based refinement (monotone improvement)
# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP redistribution step while time permits.
    Returns the best sequence encountered.
    """
    a = a0.copy()
    # ensure a is a probability vector (sum = 1)
    a = _project_simplex(a)

    best_a = a.copy()
    best_val = _objective(a)

    start = time.time()
    iteration = 0
    alpha = 0.2                     # mixing factor (can be tuned)

    while True:
        now = time.time()
        if now - start > time_budget:
            break

        # current maximal convolution entry
        conv_a = _full_convolution(a)
        M = conv_a.max()

        # solve LP
        g = _solve_lp_max_sum(a, M)
        if g is None:
            # LP failed – give up further LP steps
            break

        # Scale g to have the same total mass as a (here ∑a = 1)
        # The LP already enforces Σg = 1, so scaling is unnecessary.
        # However we keep the scaling step for safety if we ever change the normalisation.
        # g_scaled = g * (a.sum() / g.sum())
        g_scaled = g  # Σg = 1 already

        # Convex combination
        a_new = (1.0 - alpha) * a + alpha * g_scaled
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        val_new = _objective(a_new)

        if val_new < best_val - 1e-12:
            best_val, best_a = val_new, a_new.copy()
            print(f"[LP   {iteration:5d}] new best = {best_val:.6f}")

        # Accept only if the objective strictly improves; otherwise keep the old a.
        if val_new < _objective(a) - 1e-15:
            a = a_new
        else:
            # No improvement – shrink α to become more conservative.
            alpha = max(alpha * 0.7, 0.02)

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (unchanged core, adapted for any n)
# ----------------------------------------------------------------------
def _hillclimb_anneal(
    a0: np.ndarray,
    deadline: float,
    start_time: float,
    best_val: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.  Uses a geometric temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = _objective(a)

    eps = 1e-12
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick source index – weighted by contribution if possible
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # pick a target index – inversely weighted by contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # Incremental update of the full convolution (O(n) work)
        b_new = b.copy()
        b_new[j_tgt : j_tgt + n] += 2.0 * delta * a
        b_new[i_src : i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_val = _objective(a)  # placeholder (updated below)

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max / (a.sum() ** 2))

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb a little to escape stagnation
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = _objective(a)
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val
