"""
Non‑negative step‑function optimiser – gradient‑flattening + mirror steps.

The algorithm consists of three stages:
1️⃣  A dense sweep of the Beta(p,q) family (including extreme p,q) and a few
    random Dirichlet seeds – cheap way to get a decent initial shape.
2️⃣  Gradient descent on the *flattening* objective  Σ_k (a*a)_k².
    This objective pushes the convolution towards a (near) uniform distribution,
    which directly reduces the max‑convolution term that the final score depends on.
    The gradient is obtained analytically via a second FFT‑convolution.
3️⃣  A final mirror‑descent phase that directly attacks the current largest
    convolution entries (sub‑gradient of the max‑norm).  Stagnation is broken by
    Gaussian perturbations around the best point seen so far.

All iterates stay on the probability simplex (Σ a_i = 1, a_i ≥ 0).
The routine respects the 1000 s budget and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based linear convolution helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (length 2·len(a)‑1) using FFT.
    Result is real‑valued (float64).
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y via FFT."""
    m, n = len(x), len(y)
    out_len = m + n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective used inside the optimiser (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – identical to evaluate_sequence up to the scaling invariance."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Beta‑family generator (used for the cheap sweep)
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Return a normalised Beta‑shaped probability vector of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


# ----------------------------------------------------------------------
# Gradient of Σ (a*a)_k²  (flattens the convolution)
# ----------------------------------------------------------------------
def _grad_flatten(a: np.ndarray) -> np.ndarray:
    """
    Compute   g = ∂/∂a Σ_k (a*a)_k² .
    Closed form: g = 4 * ( (a*a)  ★  rev(a) )_slice ,
    where ★ denotes linear convolution and rev(a) is a reversed.
    """
    n = a.shape[0]
    b = _full_convolution(a)                    # length 2n‑1
    rev_a = a[::-1]                             # length n
    # convolution of b with rev_a  → length 3n‑2
    conv = _fft_convolve(b, rev_a)
    # extract the needed slice: conv[n‑1], conv[n‑2], …, conv[0]
    g = 4.0 * conv[n - 1::-1][:n]
    return g


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑norm (used in the final mirror‑descent phase)
# ----------------------------------------------------------------------
def _subgrad_max(a: np.ndarray, conv: np.ndarray,
                 top_idx: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)_k   using only the M currently largest entries.
    The formula is  g_i = Σ_{k∈top} 2·w_k·a_{k‑i}  (when the index is valid).
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    for k, w in zip(top_idx, weights):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        if low > high:
            continue
        j = np.arange(low, high + 1)
        g[j] += 2.0 * w * a[k - j]

    return g

        if stagnation_cnt > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.030, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            stagnation_cnt = 0
            tmp_val = float(2.0 * n * _full_convolution(a).max())
            print("[reset] perturbation after stagnation – val={:.6f}".format(tmp_val))

    # ------------------------------------------------------------------
    # 2️⃣  Final secure evaluation (uses the provided evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
