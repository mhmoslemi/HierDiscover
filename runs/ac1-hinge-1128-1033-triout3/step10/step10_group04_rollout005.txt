"""
Propose a low‑max‑convolution step function.

Algorithm
---------
1. **Rich parametric sweep** – generate many candidate shapes
   (exponential, power‑law, beta, sinusoidal, logistic, piecewise‑constant,
   random Dirichlet and an interpolation of the previous best solution).  
   Keep the best normalised vector.

2. **Adam on a smooth surrogate** – minimise a log‑sum‑exp approximation of
   the max‑convolution using Adam on the soft‑max parametrisation.
   This quickly reaches a good basin.

3. **Simulated annealing (SA)** – starting from the Adam result we run a
   long SA loop.  In each iteration we transfer a random fraction of mass
   from a source index to a target index, update the full convolution
   incrementally (O(n) work), and accept the move either if it improves
   the true objective or with a Metropolis probability depending on a
   temperature that slowly cools.  This allows the search to escape local
   minima.

4. **Exact active‑set refinement** – once the time budget is almost spent
   we shrink the remaining peak by repeatedly moving mass from the index
   with the largest “weight’’ (sum of contributions to all active maximal
   convolution entries) to the index with the smallest weight.

5. **Uniform mixing** – a tiny convex combination with the uniform vector
   can smooth isolated spikes.

All phases respect the non‑negativity and normalisation constraints,
and the best sequence found before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generate a good starting point by scanning a rich collection of families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector among a large set of parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential tails (decreasing & symmetric)
    for r in np.linspace(0.995, 0.99999, 14):
        _try(r ** i)                        # decreasing
        _try(r ** i + r ** (n - 1 - i))     # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 6.0, 18):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.2, 6.0, 7):
        for b_param in np.linspace(0.2, 6.0, 7):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _try(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.005, 0.15, 12):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 6. Logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.003, 0.04, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 7. Random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(40):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 8. Interpolation of a previously found solution, if available
    try:
        # ``height_sequence_1`` may be defined in the surrounding environment.
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val

# ----------------------------------------------------------------------

    beta1, beta2 = 0.90, 0.999

    eps_start, eps_end = 1e-1, 1e-13

    no_improve = 0
    IMPROVE_LIMIT = 80000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        n_a = a.shape[0]
        n_p = p_rev.shape[0]
        out_len = n_a + n_p - 1
        size = 1 << (out_len - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv = np.fft.irfft(fa * fp, size)[:out_len]
        grad_a = 4.0 * n * conv[n_a - 1: 2 * n_a - 1][::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated annealing on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simulated annealing with incremental convolution updates.
    Moves mass from a source index to a target index.
    """
    # initialise
    cur_a = a0.copy()
    cur_b = _full_convolution(cur_a)
    cur_val = 2.0 * n * cur_b.max()

    best_local_a = cur_a.copy()
    best_local_val = cur_val

    # temperature schedule
    T0 = max(1e-4, cur_val * 0.02)   # start with a modest temperature
    T_min = 1e-9
    it = 0
    # fraction of source mass that may be moved (linearly cooled)
    max_frac_start, max_frac_end = 0.5, 0.001

    while time.time() < deadline:
        # linear cooling of the allowed fraction
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # temperature (exponential cooling)
        T = max(T_min, T0 * np.exp(-5.0 * progress))

        # select source proportionally to its weight
        i = int(np.random.choice(n, p=cur_a))  # cur_a sums to 1
        # pick a different target uniformly
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        if cur_a[i] <= 0.0:
            continue

        delta = cur_a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of the convolution (exact)
        b_new = cur_b.copy()
        b_new[j:j + n] += 2.0 * delta * cur_a
        b_new[i:i + n] -= 2.0 * delta * cur_a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit move
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_b = b_new
            cur_val = new_val

            if cur_val < best_local_val - 1e-12:
                best_local_val, best_local_a = cur_val, cur_a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")

                if best_local_val <= target_val:
                    break
        it += 1

    # keep the globally best found
    if best_local_val < best_val:
        best_a, best_val = best_local_a, best_local_val

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Exact active‑set refinement (mass transfer based on weights)
# ----------------------------------------------------------------------
def _refine_exact_phase(a0: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Exact refinement using active‑set weights."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    step_factor = 0.4            # start moving up to 40 % of source mass
    decay = 0.9997
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000
    MIN_STEP = 1e-12

    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Refine {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            break

        # active set – indices whose value is within a tiny tolerance of the max
        eps = max_b * 5e-7
        active = np.where(b >= max_b - eps)[0]
        if active.size == 0:
            break

        # compute weight w_i = Σ_{k∈active} a_{k‑i}
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or a[src] <= MIN_STEP:
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        delta = a[src] * step_factor
        if delta < MIN_STEP:
            step_factor *= 0.5
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        b_new = _full_convolution(a_candidate)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val, best_local_a = new_val, a.copy()
        else:
            step_factor *= 0.7

        step_factor = max(step_factor * decay, MIN_STEP)
        it += 1

    if best_local_val < best_val:
        best_a, best_val = best_local_a, best_local_val
    return best_a, best_val


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point – the function invoked by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the supplied time budget.
    Returns a list of floats (Σ ≈ 1) that (hopefully) beats the current record.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem size – a few thousand entries give a good trade‑off.
    n = 9000

    # ------------------------------------------------------------------
    # 1️⃣ Initial parametric sweep
    # ------------------------------------------------------------------
    print("[Init] starting parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    TARGET = 1.503  # target bound we would love to reach

    # ------------------------------------------------------------------
    # 2️⃣ Phase A – Adam on smoothed objective
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase A] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Phase B – Simulated annealing (exact objective)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _simulated_annealing_phase(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase B] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Phase C – Exact active‑set refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _refine_exact_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase C] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Report final score using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
