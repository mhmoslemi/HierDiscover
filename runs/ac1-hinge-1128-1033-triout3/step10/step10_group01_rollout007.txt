        hill_deadline = deadline

def _smooth_adam_optimize(a0, n, deadline, start_time):
    a = a0.copy()
    # convert to y via log with epsilon
    eps=1e-12
    y = np.log(np.maximum(a, eps))
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05
    epsadam = 1e-8

    size_a = 1 << ((2*n - 2).bit_length())
    size_corr = 1 << ((3*n - 3).bit_length())  # 3n-2 ~

    best_a = a.copy()
    # compute objective for a
    conv_a = _full_convolution(a)
    best_val = float(2.0 * n * conv_a.max())

    no_improve = 0
    improve_limit = 20000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute a via softmax
        maxy = y.max()
        expy = np.exp(y - maxy)
        a = expy / expy.sum()

        # compute convolution of a with itself
        fx = np.fft.rfft(a, size_a)
        conv_a = np.fft.irfft(fx * fx, size_a)[:2*n-1]
        max_conv = conv_a.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:6d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # schedule for alpha
        prog = (now - start_time) / (deadline - start_time)
        alpha = 0.5 + 30.0 * prog   # from 0.5 to 30.5
        # compute p
        scaled = alpha * conv_a
        max_scaled = scaled.max()
        p = np.exp(scaled - max_scaled)
        p = p / p.sum()

        # compute correlation grad
        rev_a = a[::-1]
        # compute conv of p and rev_a
        # using generic full convolution
        fx_p = np.fft.rfft(p, size_corr)
        fx_rev = np.fft.rfft(rev_a, size_corr)
        conv_corr_full = np.fft.irfft(fx_p * fx_rev, size_corr)[: (len(p)+len(rev_a)-1)]

        # extract segment for gradient
        start = n - 1
        grad_corr = conv_corr_full[start:start + n]

        grad_a = 4.0 * n * grad_corr

        # project gradient onto simplex tangent (softmax)
        inner = np.dot(a, grad_a)
        grad_y = a * (grad_a - inner)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        iteration += 1

        # restart if stagnated
        if no_improve > improve_limit:
            # jitter around best
            y = np.log(np.maximum(best_a, eps))
            noise = np.random.normal(scale=0.1, size=n)
            y += noise
            m.fill(0); v.fill(0)
            no_improve = 0

"""
Hybrid optimizer for the step‑function challenge.

* Phase 1 – Smooth max Adam:*  The objective 2·n·max(a∗a) is approximated by
  a soft‑max (log‑sum‑exp) with a temperature parameter α that grows from
  ≈0.5 up to ≈30 over the allotted time.  The variables are
  parametrised by a soft‑max (y → a) so the simplex constraint Σa=1 is
  satisfied automatically.  Gradients are obtained analytically and an
  Adam update is applied.  Restarts with Gaussian jitter avoid stagnation.

* Phase 2 – Mass‑transfer hill‑climbing:*  The best sequence from Phase 1 is
  refined by a specialised annealing scheme that repeatedly moves a small
  amount of mass from indices that dominate the current maximal convolution
  entry to indices that contribute little.  This step is identical to the
  “intensive hill‑climbing” used previously but works on the new starting
  point.

The routine respects the 1000 s budget, prints intermediate best values,
and finally returns the best list of heights found.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors (length len(x)+len(y)-1)."""
    out_len = len(x) + len(y) - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Initial guess – either interpolate old solution or use a random Dirichlet
# ----------------------------------------------------------------------
def _initial_guess(n: int) -> np.ndarray:
    """Construct an initial probability vector of length n."""
    old = globals().get("height_sequence_1")
    if old is not None:
        try:
            old_arr = np.asarray(old, dtype=float)
            old_len = old_arr.shape[0]
            if old_len > 1:
                x_old = np.linspace(0.0, 1.0, old_len)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, old_arr)
                a = np.maximum(a, 0.0)
                if a.sum() > 0:
                    a /= a.sum()
                    print(f"[init] interpolated previous solution to n={n}")
                    return a
        except Exception:
            pass
    # fallback: random Dirichlet
    a = np.random.dirichlet(np.ones(n))
    print(f"[init] random Dirichlet start for n={n}")
    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser on the simplex
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed version of the max‑convolution objective.
    The simplex constraint Σa=1 is enforced via a soft‑max parametrisation.
    """
    eps = 1e-12
    # initialise y such that softmax(y) = a0
    y = np.log(np.maximum(a0, eps))

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    epsadam = 1e-8

    # pre‑compute FFT sizes
    size_a = 1 << ((2 * n - 2).bit_length())
    size_corr = 1 << ((3 * n - 3).bit_length())   # >= 3n‑2

    # bookkeeping
    best_a = None
    best_val = np.inf
    no_improve = 0
    IMPROVE_LIMIT = 20000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # -------- compute current a and objective ----------
        maxy = np.max(y)
        expy = np.exp(y - maxy)                     # stable soft‑max
        a = expy / expy.sum()

        # convolution a*a (full)
        fx = np.fft.rfft(a, size_a)
        conv_a = np.fft.irfft(fx * fx, size_a)[: 2 * n - 1]
        max_conv = conv_a.max()
        cur_val = 2.0 * n * max_conv

        # keep the best ever seen
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:6d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ------------- smooth‐max weights ---------------
        # α grows linearly from 0.5 to 30.0 over the whole run
        prog = (now - start_time) / (deadline - start_time)
        alpha = 0.5 + 30.0 * prog
        scaled = alpha * conv_a
        max_scaled = scaled.max()
        p = np.exp(scaled - max_scaled)            # avoid overflow
        p_sum = p.sum()
        if p_sum == 0:
            p = np.full_like(p, 1.0 / p.size)
        else:
            p = p / p_sum

        # ----------- gradient w.r.t. a (smooth) ----------
        rev_a = a[::-1]
        conv_corr_full = _full_convolution_two(p, rev_a)   # length 3n‑2
        start = n - 1
        grad_corr = conv_corr_full[start : start + n]      # length n
        grad_a = 4.0 * n * grad_corr                       # ∂f/∂a

        # project gradient onto simplex tangent (soft‑max Jacobian)
        inner = np.dot(a, grad_a)
        grad_y = a * (grad_a - inner)

        # --------------- Adam update --------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # ----------- stagnation handling ---------------
        if no_improve > IMPROVE_LIMIT:
            # jitter around the current best solution
            if best_a is not None:
                y = np.log(np.maximum(best_a, eps))
                y += np.random.normal(scale=0.1, size=n)
            else:
                y = np.random.normal(scale=0.1, size=n)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Adam {iteration:6d}] stagnation – restart")

        # optional early stop if we have already passed the target
        if best_val <= 1.503:
            break

    # safeguard
    if best_a is None:
        best_a = np.maximum(a, 0.0)
        best_a /= best_a.sum()
        best_val = _objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (adapted from the previous code)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local mass‑transfer annealing that reduces max(a∗a).  Starts from a0,
    keeps track of the global best (best_a, best_val) and stops when the
    target bound is reached or the time budget expires.
    """
    a = a0.copy()
    b = _full_convolution(a)                # full autocorrelation
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        # contributions to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick source index (prefer large contributors)
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # pick target index (prefers small contributors)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental update of the autocorrelation (exact)
        b_new = b.copy()
        b_new[j_tgt : j_tgt + n] += 2.0 * delta * a
        b_new[i_src : i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb around the current best to escape traps
            a = best_a.copy()
            a += np.random.normal(scale=0.025, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val

# Main entry point – orchestrates the two phases
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to minimise
    ``evaluate_sequence``.  The routine respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose a problem size – 8000 gives a good trade‑off between quality
    # and speed.  If enough time remains we also try 12000 as a second pass.
    # --------------------------------------------------------------
    sizes = [8000, 12000]

    global_best_a = None
    global_best_val = np.inf
    global_best_n = None

    for n in sizes:
        now = time.time()
        if now >= deadline:
            break

        # ----------- phase 1 – smooth Adam optimizer ------------
        a0 = _initial_guess(n)

        # allocate at most half of the remaining budget to the Adam phase
        remaining = deadline - now
        phase1_deadline = time.time() + min(remaining * 0.5, 180.0)
        a_opt, val_opt = _smooth_adam_optimize(a0, n, phase1_deadline, start_time)

        if val_opt < global_best_val:
            global_best_val, global_best_a, global_best_n = val_opt, a_opt.copy(), n
            print(f"[size {n}] after Adam best = {global_best_val:.6f}")

        # if we already hit the target we can stop early
        if global_best_val <= 1.503:
            break

        # --------------- phase 2 – hill‑climbing ----------------
        remaining = deadline - time.time()
        if remaining > 2.0:
            a_hill, val_hill = _hillclimb_anneal(
                global_best_a,
                global_best_n,
                deadline=deadline,
                start_time=start_time,
                best_a=global_best_a,
                best_val=global_best_val,
                target_val=1.503,
            )
            if val_hill < global_best_val:
                global_best_val, global_best_a, global_best_n = val_hill, a_hill.copy(), n
                print(f"[size {n}] after hillclimb best = {global_best_val:.6f}")

        if global_best_val <= 1.503:
            break

    # --------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # --------------------------------------------------------------
    if global_best_a is None:
        global_best_a = np.random.dirichlet(np.ones(4000))
        global_best_n = len(global_best_a)

    # final secure evaluation (for logging only)
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
