    # For smoother gradient compute conv to get w:
    step = 0.1
    epsilon = 0.1  # fractional of current max? We'll compute as max_b * 0.1 maybe.
    iteration = 0
    while time.time() < deadline - 0.1:
        # compute conv using FFT
        fa = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(fa * fa, size_conv)[:2*n -1]
        max_b = conv.max()
        cur_val = 2.0 * n * max_b
        # compute epsilon schedule: start larger, gradually shrink
        eps = max_b * max(0.01, 0.5 ** (iteration / 2000))  # simple exponential decay
        # compute weights
        # shift conv by -max_b to improve numeric stability
        exp_weights = np.exp((conv - max_b) / eps)  # max term becomes 1
        w = exp_weights / exp_weights.sum()
        # gradient compute using FFT cross-correlation
        # w_rev:
        w_rev = w[::-1]
        # compute convolution of a and w_rev
        size = 1 << ((n + (2*n-1) - 1).bit_length())  # n + L - 1
        fa = np.fft.rfft(a, size)
        fw = np.fft.rfft(w_rev, size)
        conv_full = np.fft.irfft(fa * fw, size)
        L = 2*n - 1
        grad = 2.0 * conv_full[L - 1 - np.arange(n)]
        # descent step
        step_size = step
        # ensure step not too large relative to eps? We'll adjust via line search
        for _ in range(20):
            a_candidate = a - step_size * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            s = a_candidate.sum()
            if s == 0:
                a_candidate = np.random.dirichlet(np.ones(n))
                s = 1.0
            a_candidate /= s
            # new conv
            fa_new = np.fft.rfft(a_candidate, size_conv)
            conv_new = np.fft.irfft(fa_new * fa_new, size_conv)[:2*n -1]
            new_val = 2.0 * n * conv_new.max()
            if new_val < cur_val - 1e-15:
                a = a_candidate
                cur_val = new_val
                if new_val < best_val:
                    best_val, best_a = new_val, a.copy()
                    print(f"[SGD {iteration:6d}] new best {best_val:.6f}")
                break
            else:
                step_size *= 0.5
        iteration += 1
        # optional increase step if many successes?
    return best_a, best_val

"""
Propose a step‑function sequence with a low value of
evaluate_sequence(sequence).

Algorithm
---------
1. **Initialisation** – try three simple families:
   * a best Beta‑family shape (a_i ∝ i^{p‑1} (n‑i+1)^{q‑1}),
   * a few Dirichlet random draws,
   * an interpolation of the previous best solution (if any).
   Keep the best of these as the start point.

2. **Smoothed gradient descent** – minimise the smooth
   approximation of the max‑convolution
       fε(a) = ε·log Σ_k exp( (a*a)_k / ε )
   using Euclidean projection onto the probability simplex.
   The gradient is obtained by a cross‑correlation of the current
   weight vector w_k = softmax((a*a)_k / ε) with the sequence a.
   ε is decreased exponentially during the run, turning the
   smoothed problem into the original non‑smooth one.
   This stage quickly reduces the peak autocorrelation.

3. **Intensive hill‑climbing** – a specialised annealing move
   that transfers a small amount of mass from indices that
   contribute most to the current maximal convolution entry to
   indices that contribute little.  The move set is identical to
   the previous solution but now starts from a much better
   point.

The whole search respects the 1000 s budget.  The best sequence
found is returned as a plain list of floats; the return value is
compatible with the provided ``evaluate_sequence`` routine.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple parametric families for a quick warm start
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_gd_optimize(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth approximation of the max‑convolution using
    projected gradient descent.  The smoothing parameter ε starts
    relatively large and is multiplied by a constant <1 each iteration.
    """
    a = a0.copy()
    best_a, best_val = a.copy(), _objective(a)

    L = 2 * n - 1                     # length of a*a
    size_conv = 1 << ((L - 1).bit_length())          # ≥ 2n‑1
    size_grad = 1 << ((3 * n - 2 - 1).bit_length())  # ≥ n+L‑1

    step = 0.1          # initial step size
    min_step = 1e-12
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- convolution and current objective -----
        fa = np.fft.rfft(a, size_conv)
        conv = np.fft.irfft(fa * fa, size_conv)[:L]
        max_b = conv.max()
        cur_val = 2.0 * n * max_b

        # ----- smoothing parameter ε (exponential decay) -----
        eps = max_b * max(0.05, 0.5 ** (iteration / 5000))

        # ----- soft‑max weights -----
        # stable: subtract max before exponentiation
        weights = np.exp((conv - max_b) / eps)
        w_sum = weights.sum()
        if w_sum == 0.0:
            w = np.full_like(weights, 1.0 / len(weights))
        else:
            w = weights / w_sum

        # ----- gradient (cross‑correlation a ⋆ w) -----
        w_rev = w[::-1]                # reversed weight vector
        fa_grad = np.fft.rfft(a, size_grad)
        fw = np.fft.rfft(w_rev, size_grad)
        conv_full = np.fft.irfft(fa_grad * fw, size_grad)
        # grad_j = 2 * conv_full[L-1-j]
        grad = 2.0 * conv_full[L - 1 - np.arange(n)]

        # ----- back‑tracking line search -----
        cand_step = step
        success = False
        for _ in range(20):
            a_candidate = a - cand_step * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            s = a_candidate.sum()
            if s == 0.0:
                a_candidate = np.random.dirichlet(np.ones(n))
                s = 1.0
            a_candidate /= s

            # evaluate candidate
            fa_c = np.fft.rfft(a_candidate, size_conv)
            conv_c = np.fft.irfft(fa_c * fa_c, size_conv)[:L]
            new_val = 2.0 * n * conv_c.max()

            if new_val < cur_val - 1e-15:
                a = a_candidate
                cur_val = new_val
                success = True
                no_improve = 0
                if new_val < best_val - 1e-12:
                    best_val, best_a = new_val, a.copy()
                    print(f"[SGD {iteration:7d}] new best = {best_val:.6f}")
                # gently increase the step after a success
                step = min(step * 1.07, 0.5)
                break
            else:
                cand_step *= 0.5

        if not success:
            step *= 0.5
            no_improve += 1

        # ----- stagnation handling -----
        if step < min_step or no_improve > IMPROVE_LIMIT:
            # random perturbation around the best known point
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            step = 0.1
            no_improve = 0
            print(f"[SGD {iteration:7d}] stagnation – random perturb")

        iteration += 1

        # early exit if we have already reached the target
        if best_val <= 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (same core as before)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        # indices that contribute to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source of mass (prefer a large contributor)
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target indices – favour positions that currently contribute little
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental update of the full convolution (exact)
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
