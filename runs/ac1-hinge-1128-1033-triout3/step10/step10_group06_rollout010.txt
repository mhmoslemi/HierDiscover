"""
Hybrid greedy mass‑transfer optimiser for the step‑function problem.

The algorithm proceeds as follows:
1. A broad parametric sweep searches a diverse set of analytic families and
   Dirichlet draws, keeping the best vector (Σ ≈ 1).
2. The main optimisation loop repeatedly moves a small amount of mass from
   an index that contributes heavily to the current maximal convolution bin
   to an index that contributes little.  The amount of mass is chosen so
   that the peak of the convolution is reduced to (or close to) the second‑largest
   peak – this yields deterministic improvement without the need for a
   Metropolis acceptance rule.
3. The vector is projected onto the probability simplex after each move,
   and the convolution is updated incrementally (O(n) per move).
4. If progress stalls, a short random perturbation is applied.
5. Finally a tiny convex combination with the uniform distribution is
   performed as a polish.
The routine respects the required `propose_candidate` signature and
stops before the supplied time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector with Σ a = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – broad parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Search many analytic families and random Dirichlet draws for a good seed."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)            # 0 … n‑1
    i1 = i + 1.0                              # 1 … n

    # 1. Exponential tails (left‑hand and symmetric)
    for r in np.linspace(0.995, 0.999999, 20):
        _try(r ** i)                         # decreasing
        _try(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

    # 2. Power‑law tails (left‑hand and symmetric)
    for p in np.linspace(0.1, 3.0, 30):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Inverse‑sqrt (special case of power‑law)
    _try(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))

    # 4. Beta densities – finer grid around the U‑shape (a,b < 1)
    beta_vals = np.linspace(0.1, 1.5, 15)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 5. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # 6. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 12):
        _try(base_sin ** p)

    # 7. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.005, 0.1, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:

        for high_frac in np.linspace(0.05, 0.4, 12):

            if total_high >= 1.0:

            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 8. Random Dirichlet draws for diversity
    rng = np.random.default_rng(123456)
    for _ in range(80):

    # 9. Interpolation of a previous solution, if available
    try:

        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

# ----------------------------------------------------------------------
# Small final polishing – blend with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

# Core optimisation – greedy mass transfer
# ----------------------------------------------------------------------
def _greedy_optimisation(a: np.ndarray,
                         deadline: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Repeatedly moves a small amount of probability mass from a source
    index that heavily contributes to the current convolution peak to a
    target index that contributes little.  The step size is chosen so that
    the current peak is driven down to (or just below) the second‑largest
    peak.  The procedure stops when the time budget expires or the target
    value is reached.
    """
    n = a.shape[0]
    # initial full convolution and objective
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    # counters for stagnation handling
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        # locate the current maximal bin and the second maximal value
        k_max = int(np.argmax(b))
        max_b = b[k_max]
        # second largest – use partition to avoid full sort
        second_max = np.partition(b, -2)[-2]

        # contributions of each index to the maximal bin
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index i – weighted by contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

            # nothing to take from this index
            continue

        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n, dtype=float)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]

            j = int(np.random.choice(n, p=probs_j))
        else:
            j = (i + np.random.randint(1, n)) % n

        if i == j:

"""
Hybrid greedy mass‑transfer optimiser for the step‑function problem.

The algorithm works in three stages:

1. **Broad parametric sweep** – a large collection of analytic families
   (exponential, power‑law, beta, arcsine, sine‑power, piecewise‑constant)
   and random Dirichlet draws are normalised to sum = 1 and the best
   candidate (lowest exact objective) is kept.

2. **Greedy mass‑transfer loop** – the current maximal convolution bin
   `k_max` is identified.  A source index `i` that contributes heavily to
   that peak is chosen proportionally to its contribution
   `a[i]·a[k_max‑i]`.  A target index `j` that contributes little to the
   same peak is chosen with probability proportional to the inverse
   contribution.  The amount of mass `δ` moved from `i` to `j` is chosen
   so that the peak is driven down to (or just below) the second‑largest
   peak.  The update of the full convolution is performed incrementally
   in O(n) time.  The vector is projected onto the probability simplex,
   and the best solution is remembered.  If no improvement is seen for a
   long stretch, a small random perturbation is applied to escape a
   plateau.

3. **Uniform polishing** – a tiny convex combination with the uniform
   distribution (α ∈ {0.001,…,0.020}) is tried; the best result is kept.

The routine respects the required `propose_candidate` signature and
stops before the supplied time budget expires.  All numbers are
non‑negative and the final list is returned directly for
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector with Σ a = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – broad parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Search many analytic families and random Dirichlet draws for a good seed."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)            # 0 … n‑1
    i1 = i + 1.0                              # 1 … n

    # 1. Exponential tails (left‑hand and symmetric)
    for r in np.linspace(0.995, 0.999999, 20):
        _try(r ** i)                         # decreasing
        _try(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

    # 2. Power‑law tails (left‑hand and symmetric)
    for p in np.linspace(0.1, 3.0, 30):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Inverse‑sqrt (special case of power‑law)
    _try(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))

    # 4. Beta densities – finer grid around the U‑shape (a,b < 1)
    beta_vals = np.linspace(0.1, 1.5, 15)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 5. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # 6. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 12):
        _try(base_sin ** p)

    # 7. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.005, 0.1, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 8. Random Dirichlet draws for diversity
    rng = np.random.default_rng(123456)
    for _ in range(80):
        _try(rng.dirichlet(np.ones(n)))

    # 9. Interpolation of a previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Small final polishing – blend with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Core optimisation – greedy mass transfer
# ----------------------------------------------------------------------
def _greedy_optimisation(a0: np.ndarray,
                         start: float,
                         deadline: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from a source index that
    heavily contributes to the current maximal convolution bin to a
    target index that contributes little.  The transferred amount is
    chosen so that the peak is driven down towards the second‑largest
    peak.  The procedure stops when the time budget is exhausted or the
    target value is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    # parameters for the schedule
    max_frac_start, max_frac_end = 0.6, 0.001
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Identify the current peak and the second‑largest value
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))
        max_b = b[k_max]
        # second largest (use partition to avoid full sort)
        second_max = np.partition(b, -2)[-2]

        # --------------------------------------------------------------
        # Choose a source index i (heavy contribution to the peak)
        # --------------------------------------------------------------
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            # nothing usable at this index
            continue

        # --------------------------------------------------------------
        # Choose a target index j (low contribution to the same peak)
        # --------------------------------------------------------------
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n, dtype=float)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0                     # avoid moving mass back immediately
        if w.sum() > 0.0:
            probs_j = w / w.sum()
            j = int(np.random.choice(n, p=probs_j))
        else:
            j = (i + np.random.randint(1, n)) % n

        if i == j:
            continue

        # --------------------------------------------------------------
        # Compute how much mass we can move without raising the peak again
        # --------------------------------------------------------------
        cont_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        cont_j = a[k_max - j] if 0 <= k_max - j < n else 0.0

        if cont_i <= cont_j + 1e-15:
            # moving mass here cannot reduce the peak
            continue

        # linear reduction of the peak (ignoring the tiny quadratic term)
        denom = 2.0 * (cont_i - cont_j)
        delta_needed = (max_b - second_max) / denom if denom > 0 else 0.0

        # schedule a fraction of the source mass
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # admissible delta is the smallest of:
        #   * what is needed to bring the peak down,
        #   * the allowed fraction of the source mass,
        #   * the source mass itself
        delta_cap = a[i] * max_frac
        delta = min(delta_needed * 0.9, delta_cap)   # 0.9 gives a safety margin
        if delta <= 0.0:
            # fall back to a modest random step
            delta = a[i] * max_frac * np.random.rand()
            if delta <= 0.0:
                continue

        # --------------------------------------------------------------
        # Incremental update of the full convolution
        # --------------------------------------------------------------
        b_new = b.copy()
        # removal from i
        b_new[i:i + n] -= 2.0 * delta * a
        # addition to j
        b_new[j:j + n] += 2.0 * delta * a
        # self‑terms
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        if idx2i < len(b_new):
            b_new[idx2i] += delta * delta
        if idx2j < len(b_new):
            b_new[idx2j] += delta * delta
        if idxij < len(b_new):
            b_new[idxij] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # --------------------------------------------------------------
        # Accept only improving moves (deterministic greedy descent)
        # --------------------------------------------------------------
        if new_val < cur_val - 1e-15:
            # apply the move
            a[i] -= delta
            a[j] += delta
            # numerical safety – re‑normalise occasionally
            if it % 5000 == 0:
                a = np.maximum(a, 0.0)
                a /= a.sum()
                b = _full_convolution(a)
                cur_val = 2.0 * n * b.max()
                continue
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val - 1e-15:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Greedy {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Stagnation handling
        # --------------------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            # a short random perturbation to escape a flat region
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Greedy {it:7d}] perturbed after stagnation")

        # --------------------------------------------------------------
        # Periodic re‑normalisation to curb round‑off accumulation
        # --------------------------------------------------------------
        if it % 10000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        it += 1

    return best_a, best_val
