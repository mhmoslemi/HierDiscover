"""
Smooth exponentiated‑gradient descent for the max‑convolution objective,
followed by a hard EG refinement and a tiny uniform mixing step.

The algorithm works on the probability simplex (∑a_i = 1) and uses a
soft‑max approximation of the max‑convolution to obtain a smoother
gradient.  The smoothing parameter α is slowly annealed, the learning
rate η decays geometrically, and occasional jitter helps escaping
shallow plateaus.  After the smooth phase a classic EG update on the
actual max‑entry is performed for fine‑tuning.  Finally a convex blend
with the uniform distribution is tried.  The best sequence seen during
the whole run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a probability vector (∑a=1)."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for several tiny α and keep the best."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = cand / cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


def _generate_parametric_candidates(n: int, rng: np.random.Generator):
    """Yield a few hand‑crafted families of candidate vectors."""
    i = np.arange(n, dtype=np.float64)

    # 1. Double‑exponential (U‑shaped)
    for r in np.linspace(0.55, 0.9999, 30):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 2. Power‑law edge‑peaked
    for p in np.linspace(-4.0, 4.0, 30):
        base = (i + 1) * (n - i) + 1e-12
        a = np.power(base, -p)
        yield a

    # 3. Cosine‑window with exponent
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))     # Hann window
    for beta in np.linspace(0.5, 4.0, 20):
        a = np.power(w, beta)
        yield a

    # 4. Arcsine‑like (1/√x(1‑x)) shape
    eps = 1e-12
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x) + eps)
    yield a

    # 5. Random Dirichlet draws (a few samples)
    for _ in range(30):
        yield rng.dirichlet(np.ones(n))

    # ------------------------------------------------------------------

        if s > 0.0:
            a0 = a0 / s
        else:
            raise ValueError
        print("[init] Re‑used height_sequence_1")
    except Exception:
        best_val = np.inf
        best_a = None
        for cand in _generate_parametric_candidates(n, rng):
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s <= 0.0:

            eta = 0.5
            alpha = 2.0
            no_improve = 0
            print(f"[smooth EG {it:7d}] re‑initialised after stagnation")

    # ------------------------------------------------------------------

    no_improve = 0
