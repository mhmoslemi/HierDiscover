def _gradient_lp(a, p):
    n = len(a)
    conv = _full_convolution(a)
    # compute sum-p
    conv_p = conv ** p
    sum_p = conv_p.sum()
    if sum_p == 0:
        # fallback uniform gradient zero?
        return np.zeros_like(a)
    # compute conv_pow (conv^(p-1))
    conv_pow = conv ** (p-1)
    # grad = 2 * (conv_pow ⋆ a) / (sum_p ** ((p-1)/p))
    # compute correlation via FFT
    rev = conv_pow[::-1]
    size = 1 << ((len(rev) + len(a) - 1 - 1).bit_length())
    fft_rev = np.fft.rfft(rev, size)
    fft_a = np.fft.rfft(a, size)

    L = len(conv)
    # extract segment:
    segment = corr_full[L-1 : L-1 + n]
    grad = 2.0 * segment / (sum_p ** ((p-1)/p))
    return grad

"""
Non‑negative step‑function optimiser (gradient‑descent on an Lp‑smooth proxy).

The algorithm
* loads the previous best construction (if available) and interpolates it to a
  new length `n`;
* starts from a Beta‑shaped seed (p=q=0.5 -> mass at the ends);
* iteratively minimises the smooth objective  
        f_p(a) = ( Σ_k (a*a)_k^p )^{1/p}                (p≫1)
  whose gradient can be computed by a single FFT‑based correlation.
  The exponential‑mirror (entropy) update preserves non‑negativity and
  guarantees Σa_i=1.
* a linear schedule raises `p` from 2 to 20 (more emphasis on the peak) and
  simultaneously lowers the stepsize and a Metropolis temperature to allow
  occasional uphill moves.
* after a long streak without improvement a small Gaussian perturbation is
  added to escape stagnation.
* the best vector found during the run is finally evaluated with
  `evaluate_sequence` and returned.

The whole routine respects the 1000 s budget and works on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (a with itself)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, size: int) -> np.ndarray:
    """Return the linear convolution x*x of length 2·len(x)-1 using FFT."""
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[: 2 * x.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Beta‑family generator (used for the very first seed)
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Generate a normalised Beta‑shaped vector of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


# ----------------------------------------------------------------------
# Gradient of the Lp‑smooth proxy (p>1)
# ----------------------------------------------------------------------
def _gradient_lp(a: np.ndarray,
                 conv: np.ndarray,
                 p: float,
                 fft_size: int,
                 fft_a: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a ( Σ conv^p )^{1/p}.
    `conv` must be the full convolution a*a (length 2·n‑1).
    `fft_a` is the rfft of a (pre‑computed for the given fft_size).
    """
    n = a.shape[0]
    L = conv.shape[0]                     # = 2*n-1

    # denominator = ( Σ conv^p )^{(p‑1)/p}
    sum_pow = np.sum(conv ** p)
    # avoid division by zero – in practice sum_pow > 0
    denom = sum_pow ** ((p - 1.0) / p) if sum_pow > 0 else 1.0

    # conv_pow = conv^{p‑1}
    conv_pow = conv ** (p - 1.0)

    # Correlation w ⋆ a where w = conv_pow
    # w ⋆ a equals the convolution of w[::-1] with a.
    rev = conv_pow[::-1]
    fft_rev = np.fft.rfft(rev, fft_size)

    # full correlation (length L + n - 1)
    corr_full = np.fft.irfft(fft_rev * fft_a, fft_size)[: L + n - 1]

    # extract segment that corresponds to shifts 0 … n‑1
    # start index = L‑1 (centre of the correlation)
    start = L - 1
    segment = corr_full[start : start + n]

    # gradient = 2 * segment / denom
    grad = 2.0 * segment / denom
    return grad


# ----------------------------------------------------------------------
# Exponential‑mirror (entropy) update, stays on the simplex
# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform a multiplicative mirror‑descent step:
        a_i ← a_i * exp(-η * grad_i)   and renormalise to sum=1.
    """
    delta = -eta * grad
    # avoid overflow/underflow inside the exponential
    delta = np.clip(delta, -30.0, 30.0)
    a_new = a * np.exp(delta)
    s = a_new.sum()
    if s == 0.0:
        # fallback to uniform if everything collapsed
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s
    time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a short margin for the final secure call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two helps the FFT implementation.
    # --------------------------------------------------------------
    n = 16384                     # ≈ 1.6 · 10⁴  (fits comfortably in time)

    # --------------------------------------------------------------
    # Initialise from the previous best solution if it exists.
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf

    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        best_a = a0
    except Exception:
        # fall back to a simple Beta seed (mass near the ends)
        a0 = _beta_family(n, p=0.5, q=0.5)
        best_a = a0

    # --------------------------------------------------------------
    # Pre‑compute FFT size (needs to accommodate a*a and w⋆a)
    # --------------------------------------------------------------
    fft_size = 1 << ((3 * n - 2 - 1).bit_length())   # ≥ 3n‑2

    # initialise the current iterate
    a = best_a.copy()
    # first full convolution and its FFT

    cur_val = float(2.0 * n * conv.max())

    best_val = cur_val
    best_a = a.copy()

    # --------------------------------------------------------------
    # Schedules
    # --------------------------------------------------------------
    eta_start, eta_end = 1.0, 0.001
    T_start, T_end = 5e-2, 1e-7
    p_start, p_end = 2.0, 20.0

    stagnation = 0
    IMPROVE_LIMIT = 3000      # after this many non‑improvements we perturb

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress fraction 0 … 1
        prog = (now - start_time) / (deadline - start_time)

        # parameters for the current iteration
        p_cur = p_start + (p_end - p_start) * prog
        eta = eta_start * (1.0 - prog) + eta_end * prog
        T = T_start * (T_end / T_start) ** prog

        # ---- gradient of the Lp‑smooth proxy --------------------
        grad = _gradient_lp(a, conv, p_cur, fft_size, fft_a)

        # ---- propose a mirror‑descent update --------------------
        a_candidate = _exp_update(a, grad, eta)

        # ---- evaluate candidate ---------------------------------
        fft_a_cand = np.fft.rfft(a_candidate, fft_size)
        conv_cand = _full_convolution(a_candidate, fft_size)
        cand_val = float(2.0 * n * conv_cand.max())

        # ---- Metropolis acceptance ------------------------------
        accept = False
        if cand_val < cur_val - 1e-15:
            accept = True
        else:
            # allow occasional uphill steps
            if T > 0.0:
                prob = math.exp(-(cand_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_candidate
            fft_a = fft_a_cand
            conv = conv_cand
            cur_val = cand_val
            stagnation = 0
        else:
            stagnation += 1

        # ---- keep the best ever seen ----------------------------
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print("[iter] new best = {:.6f} at t={:.1f}s".format(
                best_val, now - start_time))

        # ---- stagnation handling --------------------------------
        if stagnation > IMPROVE_LIMIT:
            # Gaussian perturbation around the best solution found so far
            noise = np.random.normal(scale=0.04, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            fft_a = np.fft.rfft(a, fft_size)
            conv = _full_convolution(a, fft_size)
            cur_val = float(2.0 * n * conv.max())
            stagnation = 0
            print("[reset] perturbation after stagnation – val={:.6f}".format(
                cur_val))

        # occasional cheap reporting (every ~5 s)
        if int((now - start_time) * 10) % 50 == 0:
            pass   # silent – remove if you want more output

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
