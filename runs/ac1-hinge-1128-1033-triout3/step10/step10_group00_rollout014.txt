"""
Non‑negative step‑function optimiser (adam‑style).

The routine builds an initial candidate from a (possibly) saved solution,
scans a dense beta‑family of U‑shaped distributions, adds a few Dirichlet
seeds and then runs a projected Adam optimiser that directly minimises
a smooth approximation of the infinity‑norm of the self‑convolution.
The optimiser works on the simplex (Σa=1, a_i≥0), uses a moving
temperature to occasionally accept uphill moves and performs a Gaussian
re‑initialisation after long periods without improvement.

The method differs from the previous mirror‑descent approach by:
  * using Adam momentum,
  * weighting the sub‑gradient by the relative size of the largest
    convolution entries (soft‑max–like),
  * a more aggressive schedule for the learning rate,
  * a richer beta‑parameter grid (including very small exponents).

The objective that is minimised is 2·n·max(conv(a,a)) which is exactly the
value returned by ``evaluate_sequence`` for a normalised sequence (the
scale of the sequence does not matter).  The function respects the 1000 s
budget and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)-1) via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑shaped distribution on {1,…,n}, normalised to sum 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


def _gradient_top(a: np.ndarray,
                  conv: np.ndarray,
                  top_idx: np.ndarray,
                  top_weights: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the weighted max‑conv term using only the selected
    top entries (weights must sum to 1).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k, w in zip(top_idx, top_weights):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1)
        # contribution from a_{k-i}
        g[idx] += 2.0 * w * a[k - idx]
    return g

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  The routine respects the 1000 s time
    budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a tiny margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs
    # --------------------------------------------------------------
    n = 8192

    # --------------------------------------------------------------
    # 0️⃣  Load a previous solution if present
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        best_a, best_val = a0, _objective(a0)
        print("[init] interpolated previous solution – val={:.6f}".format(best_val))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Dense sweep of the Beta‑family (including very sharp U‑shapes)
    # --------------------------------------------------------------
    # Very fine grid: small exponents give strong U‑shapes, larger ones
    # give bell‑shapes.  The total number of evaluations stays well below
    # a few thousand.
    p_vals = np.concatenate([
        np.linspace(0.01, 0.2, 20),            # extreme U‑shapes
        np.linspace(0.2, 2.0, 19),             # moderate shapes
        np.geomspace(2.0, 10.0, 9)[1:]         # tail of the grid
    ])
    q_vals = p_vals.copy()

    for p in p_vals:
        for q in q_vals:
            a = _beta_family(n, p, q)
            val = _objective(a)
            if val < best_val:
                best_a, best_val = a, val
                print("[beta] improved – p={:.3f}, q={:.3f}, val={:.6f}".format(p, q, val))
        # early‑exit if time is almost up
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2️⃣  A few random Dirichlet seeds for diversity
    # --------------------------------------------------------------
    for _ in range(10):
        a = np.random.dirichlet(np.ones(n))
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val
            print("[dir] improved – val={:.6f}".format(val))

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] uniform – val={:.6f}".format(best_val))

    # --------------------------------------------------------------
    # 3️⃣  Adam optimiser on the (soft‑max weighted) max‑convolution term
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    best_global_a = best_a.copy()
    best_global_val = best_val

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # schedules
    eta_start, eta_end = 0.25, 0.0008           # learning‑rate
    T_start, T_end = 2e-2, 1e-8                # temperature for Metropolis moves
    M_top = 40                                 # number of convolution entries used
    stagnation = 0
    IMPROVE_LIMIT = 3000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- current objective and best‑update -----
        conv = _full_convolution(a)
        cur_max = conv.max()
        cur_val = float(2.0 * n * cur_max)

        if cur_val < best_global_val - 1e-12:
            best_global_val = cur_val
            best_global_a = a.copy()
            stagnation = 0
            print("[iter] new best = {:.6f} at t={:.1f}s".format(best_global_val,
                                                                now - start_time))
        else:
            stagnation += 1

        # ----- top‑M entries of the convolution -----
        top_idx = np.argpartition(-conv, M_top - 1)[:M_top]
        top_vals = conv[top_idx]
        if top_vals.sum() == 0.0:
            # degenerate case – fall back to uniform weights
            top_weights = np.full(M_top, 1.0 / M_top)
        else:
            top_weights = top_vals / top_vals.sum()

        # ----- gradient (weighted sub‑gradient of the max‑conv) -----
        g = _gradient_top(a, conv, top_idx, top_weights)

        # ----- Adam step -----
        t_iter += 1
        beta1, beta2 = 0.9, 0.999
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        prog = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - prog) + eta_end * prog
        step = eta * m_hat / (np.sqrt(v_hat) + 1e-12)

        a_candidate = a - step
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        # ----- evaluate candidate (full convolution) -----
        conv_cand = _full_convolution(a_candidate)
        cand_max = conv_cand.max()
        cand_val = float(2.0 * n * cand_max)

        # ----- Metropolis acceptance -----
        accept = False
        if cand_val < cur_val - 1e-15:
            accept = True
        else:
            T = T_start * (T_end / T_start) ** prog
            prob = math.exp(-(cand_val - cur_val) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            cur_val = cand_val
            # (global best already handled at loop top)

        # ----- stagnation handling – Gaussian perturbation -----
        if stagnation > IMPROVE_LIMIT:
            # blend the global best with a fresh Dirichlet sample
            noise = np.random.normal(scale=0.03, size=n)
            a = best_global_a + noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0
            print("[reset] perturbation after stagnation – val={:.6f}".format(
                float(2.0 * n * _full_convolution(a).max())))

    # --------------------------------------------------------------
    # 4️⃣  Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_global_a.tolist())
        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_global_a.tolist()
