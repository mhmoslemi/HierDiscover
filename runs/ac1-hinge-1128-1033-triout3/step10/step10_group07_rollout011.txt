def _softmax_eg_descent(a0, n, deadline, rng):
    log_w = np.log(a0 + eps)
    eta = 0.5
    decay = 0.99995
    best_a = a0.copy()
    best_val = _objective_value(best_a)
    no_improve = 0
    IMPROVE_LIMIT = 20000
    it = 0
    alpha = 1.0
    alpha_factor = 1.001
    max_alpha = 50000.0

    size_fft = 1 << ((3 * n - 2) - 1).bit_length()

    while time.time() < deadline:
        # compute a from log_w
        max_log = np.max(log_w)
        w = np.exp(log_w - max_log)
        a = w / w.sum()

        # convolution a*a
        fa = np.fft.rfft(a, size_fft)
        conv_full = np.fft.irfft(fa * fa, size_fft)[:2 * n - 1]

        # compute objective value
        cur_val = 2 * n * conv_full.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            if it % 1000 ==0:
                print(f"[EG  {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # compute softmax weights
        # comfortable scaling: use alpha * conv_full
        # currently alpha may be small; computing alpha * conv_full.
        scaled = alpha * conv_full
        # subtract max for stability
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w_weights = exp_scaled / np.sum(exp_scaled)   # length 2n-1

        # compute gradient: 2 * conv(w_weights, a_rev)
        a_rev = a[::-1]
        fw = np.fft.rfft(w_weights, size_fft)
        fa_rev = np.fft.rfft(a_rev, size_fft)
        conv2_full = np.fft.irfft(fw * fa_rev, size_fft)[:3 * n - 2]
        grad = 2.0 * conv2_full[n-1:n-1+n]

        # multiplicative EG update
        log_w -= eta * grad
        eta *= decay

        # occasional jitter
        if rng.random() < 0.005:
            log_w += rng.normal(scale=0.003, size=n)

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            log_w = np.log(best_a + eps)
            log_w += rng.normal(scale=0.02, size=n)
            eta = 0.5
            no_improve = 0
            print(f"[EG {it:7d}] perturb after stagnation")
        # increase alpha slowly
        alpha = min(max_alpha, alpha * alpha_factor)

        it += 1

        # break if near deadline
        if time.time() > deadline - 0.2:
            break

    return best_a, best_val

"""
Optimisation of a non‑negative step function via a smoothed
exponentiated‑gradient (EG) method.

The exact objective is
    f(a) = 2·n·max(conv(a, a)) / (∑a)² .
All candidate vectors are normalised to ∑a = 1, so the objective
reduces to 2·n·max(conv(a, a)).  The function to minimise is therefore
the ℓ∞‑norm of the self‑convolution.

The algorithm proceeds as follows
-------------------------------------------------------------
1.  Initialise a probability vector `a` (sum = 1).  If a previous
    construction `height_sequence_1` exists it is interpolated;
    otherwise a small parametric sweep (exponential, power‑law,
    Hann, Kaiser, Chebyshev, Dirichlet draws) provides a decent seed.

2.  Smoothed EG descent
    •  At each iteration compute the full convolution `b = a*a`.
    •  Replace the nondifferentiable max by a soft‑max:
           w_k ∝ exp(α·b_k)      (α is a temperature that grows).
    •  The gradient of the soft‑max surrogate is
           g = 2·conv(w, a_rev)   (a_rev = a reversed).
    •  Perform a multiplicative update on the log‑weights:
           log_w ← log_w – η·g .
       After the update the probability vector is recovered by
       exponentiation and normalisation.
    •  η decays geometrically, α increases slowly, occasional random
       jitter and a stagnation‑re‑initialisation avoid local traps.

3.  Tiny mixing with the uniform distribution (α∈{0.001,…,0.02})
    often reduces the exact objective further.

The whole optimisation is run until the allotted 1000 s budget is
nearly exhausted (a 0.5 s safety margin is kept for the final call to
`evaluate_sequence`).  The best vector seen is returned as a plain
list of Python floats.
"""

import time
import numpy as np

# SciPy is optional – we use Chebyshev windows if available.
try:
    from scipy.signal import chebwin
except Exception:  # pragma: no cover
    chebwin = None


# ----------------------------------------------------------------------
# FFT‑based full convolution (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """2·n·max(conv(a, a)) for a probability vector."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Mix with a tiny amount of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for a few α and keep the best."""
    n = a.shape[0]
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for α in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - α) * a + α * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families for a quick initialisation
# ----------------------------------------------------------------------
def _generate_parametric_candidates(
    n: int, rng: np.random.Generator
) -> list[np.ndarray]:
    """Yield a handful of hand‑crafted candidates of length n."""
    cand_list = []

    # 1. Exponential (U‑shaped) family
    i = np.arange(n, dtype=np.float64)
    for r in np.linspace(0.55, 0.9999, 30):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        cand_list.append(a)

    # 2. Power‑law (edge‑peaked) family
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(-4.0, 4.0, 30):
        base = idx * (n + 1 - idx) + 1e-12
        a = np.power(base, -p)
        cand_list.append(a)

    # 3. Cosine (Hann) window raised to β
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    for β in np.linspace(0.5, 4.0, 20):
        a = np.power(w, β)
        cand_list.append(a)

    # 4. Kaiser windows – β controls the taper
    for β in np.linspace(0.0, 12.0, 12):
        a = np.kaiser(n, β)
        cand_list.append(a)

    # 5. Chebyshev (Dolph‑Chebyshev) windows, if SciPy is present
    if chebwin is not None:
        for att in np.linspace(50, 150, 10):  # attenuation in dB
            a = chebwin(n, at=att, sym=True)
            cand_list.append(a)

    # 6. Random Dirichlet draws – cheap stochastic exploration
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        cand_list.append(a)

    # Normalise all candidates and clip negatives (just in case)
    normalised = []
    for a in cand_list:
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            normalised.append(a / s)

    return normalised


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _softmax_eg_descent(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative updates driven by the gradient of a soft‑max surrogate
    of the max‑convolution entry.  Returns the best vector seen.
    """
    eps = 1e-12
    n = a0.shape[0]

    # log‑weights for the simplex (a = softmax(log_w))
    log_w = np.log(a0 + eps)

    # hyper‑parameters
    eta = 0.5                # initial step size
    decay = 0.99994           # geometric decay per iteration
    α = 1.0                  # initial soft‑max temperature
    α_factor = 1.001         # slow growth
    α_max = 50000.0

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    no_improve = 0
    IMPROVE_LIMIT = 25000
    it = 0

    # Pre‑compute the FFT size that can hold the largest convolution
    size_fft = 1 << ((3 * n - 2) - 1).bit_length()

    while time.time() < deadline:
        # ----- rebuild probability vector -----
        max_log = np.max(log_w)
        w = np.exp(log_w - max_log)
        a = w / w.sum()

        # ----- self‑convolution, objective, soft‑max weights -----
        conv_aa = _full_convolution_fft(a, a)          # length 2·n‑1
        cur_val = 2.0 * n * conv_aa.max()
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            if it % 500 == 0:
                print(f"[SM‑EG {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ----- soft‑max over the convolution -----
        scaled = α * conv_aa
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w_weights = exp_scaled / np.sum(exp_scaled)     # length 2·n‑1

        # ----- gradient of the soft‑max surrogate -----
        #   g = 2·conv(w_weights, a_rev)  (a_rev = a reversed)
        conv_wa_rev = _full_convolution_fft(w_weights, a[::-1])
        grad = 2.0 * conv_wa_rev[n - 1 : n - 1 + n]     # slice length n

        # ----- EG multiplicative update -----
        log_w -= eta * grad
        eta *= decay

        # ----- jitter to escape shallow plateaus -----
        if rng.random() < 0.004:
            log_w += rng.normal(scale=0.003, size=n)

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            log_w = np.log(best_a + eps)
            log_w += rng.normal(scale=0.02, size=n)
            eta = 0.5
            no_improve = 0
            print(f"[SM‑EG {it:7d}] perturb after stagnation")

        # ----- schedule temperature growth -----
        α = min(α_max, α * α_factor)

        it += 1

        # safety break close to the deadline
        if time.time() > deadline - 0.2:
            break

    return best_a, best_val
