        active = np.where(b >= max_b - eps)[0]

        # compute weight w_i
        w = np.zeros(n)
        for k in active:
            i0 = max(0, k - (n - 1))
            i1 = min(k, n - 1)
            idxs = np.arange(i0, i1 + 1)
            w[idxs] += a[k - idxs]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt:
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))
        if a[src] <= 1e-16:

        while delta > 1e-15:

"""
Optimiser for the max‑convolution bound.

The objective for a non‑negative vector `a` with Σa = 1 is
    f(a) = 2·n·max_k (a * a)_k .
We seek a vector with a small maximum convolution entry.

Algorithm
---------
1. **Parametric sweep** – evaluate many analytic families (exponential,
   power‑law, Beta, sinusoidal, logistic, piecewise‑constant, random Dirichlet)
   and keep the best normalised vector.

2. **Adam on a smooth surrogate** – minimise a log‑sum‑exp approximation of
   the max‑convolution on the probability simplex.  Adam with an exponential
   learning‑rate schedule is used; the surrogate temperature ε is annealed
   from 1e‑1 to 1e‑13.

3. **Refined hill‑climbing** – repeatedly transfer mass from indices that
   contribute most to the current peak (the “active set”) to indices that
   contribute least.  The transfer size is chosen by a simple back‑tracking
   line‑search that guarantees a strict reduction of the true objective.

4. **Final uniform mixing** – a tiny convex combination with the uniform
   distribution smoothes residual spikes.

The whole pipeline respects the 1000 s budget, prints intermediate scores
and finally returns the best sequence found (as a plain list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial parametric sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector among a large set of parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential tails
    for r in np.linspace(0.995, 0.99999, 20):
        _try(r ** i)                        # decreasing
        _try(r ** i + r ** (n - 1 - i))     # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 8.0, 24):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.1, 8.0, 13):
        for b_param in np.linspace(0.1, 8.0, 13):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 8.0, 16):
        _try(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.01, 0.25, 15):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.35, 15):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 6. Logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.002, 0.05, 8):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 7. Random Dirichlet draws with varied concentration
    rng = np.random.default_rng(12345)
    for conc in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(30):
            a = rng.dirichlet(np.full(n, conc))
            _try(a)

    # 8. Interpolation of a previously found solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smoothed log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp (smooth max) surrogate."""
    # work in log‑space (soft‑max) to stay on the simplex
    w = np.log(a0 + 1e-12)          # logits
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.004
    eps_start, eps_end = 1e-1, 1e-13

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        size = 1 << ((2 * n - 1) - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv = np.fft.irfft(fa * fp, size)[:2 * n - 1]
        grad_a = 4.0 * n * conv[n - 1:2 * n - 1][::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape flat regions
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.02, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Refined hill‑climbing with exact line search
# ----------------------------------------------------------------------
def _refined_hillclimb(a_start: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Greedy mass transfer that reduces the exact peak using a back‑tracking line search."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = a.copy()
    best_local_val = cur_val

    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    while time.time() < deadline:
        max_b = b.max()
        # active set: entries within a tiny relative tolerance of the max
        eps = max_b * 5e-7
        active = np.where(b >= max_b - eps)[0]

        # compute weight w_i = Σ_{k∈active} a_{k-i}
        w = np.zeros(n)
        for k in active:
            i0 = max(0, k - (n - 1))
            i1 = min(k, n - 1)
            idxs = np.arange(i0, i1 + 1)
            w[idxs] += a[k - idxs]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or a[src] <= 0.0:
            # fallback to a simple alternative
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        # try to move a fraction of the source mass – back‑track until improvement
        delta = a[src] * 0.5
        improved = False
        while delta > 1e-15:
            # incremental update of the full convolution (exact)
            new_b = b.copy()
            new_b[tgt:tgt + n] += 2.0 * delta * a
            new_b[src:src + n] -= 2.0 * delta * a
            new_b[2 * tgt] += delta * delta
            new_b[2 * src] += delta * delta
            new_b[src + tgt] -= 2.0 * delta * delta
            new_val = 2.0 * n * new_b.max()
            if new_val < cur_val - 1e-15:
                # accept this move
                a[src] -= delta
                a[tgt] += delta
                b = new_b
                cur_val = new_val
                improved = True
                break
            else:
                delta *= 0.5

        if improved:
            no_improve = 0
            if cur_val < best_local_val - 1e-12:
                best_local_val, best_local_a = cur_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1
            if no_improve > IMPROVE_LIMIT:
                # perturb current best and restart
                a = best_local_a.copy()
                noise = np.random.normal(scale=0.015, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                b = _full_convolution(a)
                cur_val = 2.0 * n * b.max()
                no_improve = 0
                print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    # keep the globally best vector
    if best_local_val < best_val:
        best_val, best_a = best_local_val, best_local_a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point – the function invoked by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the supplied time budget.
    Returns a list of floats (Σ = 1) that (hopefully) beats the current record.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – 9 000 entries have proved a good trade‑off
    n = 9000

    print("[Init] parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase A] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Refined hill‑climbing with exact line search (≈50 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _refined_hillclimb(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase B] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – Final uniform mixing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Report final score using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
