def full_convolution(a):
    n = a.shape[0]
    m = 2*n - 1
    size = 1 << ((m-1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    return conv

def conv_fft(x, y):
    n = x.shape[0]
    m = y.shape[0]
    m_out = n + m - 1
    size = 1 << ((m_out-1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:m_out]
    return conv

Now project_simplex:

def project_simplex(v):
    # Implementation returns projection onto simplex sum(v)=1, v>=0
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # Determine rho
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def beta_pdf(alpha, n):
    import math
    # compute log probability for stability
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # Avoid x exactly 0 or 1; they are >0 and <1.
    logB = math.lgamma(alpha) + math.lgamma(alpha) - math.lgamma(2*alpha)
    # Compute logpdf
    log_pdf = (alpha - 1) * (np.log(x) + np.log(1 - x)) - logB
    # Convert to pdf
    pdf = np.exp(log_pdf)
    return pdf

def adam_phase(a0, n, start, deadline, best_a, best_val, target_val=1.503):
    # Implementation
    import numpy as np
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.9, 0.999
    lr_start, lr_end = 0.2, 0.005
    eps_start, eps_end = 0.1, 1e-8  # start with higher epsilon
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 100000
    while time.time() < deadline:
        # compute a from w
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        a = symmetrize(a)
        a = project_simplex(a)  # ensure sum 1 and nonneg
        # evaluate
        cur_val = objective_val(a)
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule progress
        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * ((eps_end / eps_start) ** prog)
        lr = lr_start * ((lr_end / lr_start) ** prog)

        # compute full convolution and smooth max
        b = full_convolution(a)
        max_b = b.max()
        # compute softmax weights p
        shifted = (b - max_b) / eps
        # Clip extreme values to avoid overflow
        # Using np.exp on large positive values can overflow; shifted <=0, so exponent <=1.
        # Actually if eps small, (b-max_b)/eps is <=0 always.
        weights = np.exp(shifted)
        p = weights / weights.sum()
        # compute gradient
        p_rev = p[::-1]
        conv_ab = conv_fft(a, p_rev)
        segment = conv_ab[n-1: 2*n-1]
        grad_a = 4.0 * n * segment[::-1]  # shape n

        # chain rule to grad_w
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w * grad_w)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        if no_improve > IMPROVE_LIMIT:
            # restart
            print("[Adam] restart after stagnation")
            w = np.random.randn(n) * 0.2
            m.fill(0)
            v.fill(0)
            t = 0
            no_improve = 0

    return best_a, best_val

def subgradient_phase(a0, n, start, deadline, best_a, best_val, target_val=1.503):
    a = a0.copy()
    best_local_a = best_a.copy()
    best_local_val = best_val
    eta0 = 1.0 / n  # initial step size
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 50000
    while time.time() < deadline:
        b = full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val = cur_val
            best_local_a = a.copy()
            no_improve = 0
            print(f"[Subgrad] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))
        # gradient of max entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step with projection onto simplex
        v = a - eta * grad
        a = project_simplex(v)
        a = symmetrize(a)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # perturb
            print("[Subgrad] perturb after stagnation")
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            a = symmetrize(a)
            eta = eta0
            no_improve = 0

        it += 1

    return best_local_a, best_local_val

def deterministic_refine(a0, n, start, deadline, best_val):
    a = a0.copy()
    b = full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_val = cur_val
    best_local_a = a.copy()
    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        # compute contributions of each index to this max entry
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]
        # source idx with max contrib
        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break
        # target idx with smallest a (excluding i)
        # we can pick j = argmin of a (excluding i)
        j = int(np.argmin(a + (np.arange(n) == i) * 1e9))
        # compute a_{k-i}, a_{k-j}
        a_ki = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_kj = a[k_max - j] if 0 <= k_max - j < n else 0.0
        if a_ki <= a_kj + 1e-12:
            # cannot reduce max entry with this pair
            # try next smallest maybe
            # find next candidate j among sorted indices
            sorted_indices = np.argsort(a)
            found = False
            for cand in sorted_indices:
                if cand == i:
                    continue
                a_kj = a[k_max - cand] if 0 <= k_max - cand < n else 0.0
                if a_ki > a_kj + 1e-12:
                    j = int(cand)
                    found = True
                    break
            if not found:
                break

        # evaluate second max
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        # analytic delta to bring b[k_max] down to second_max
        denom = 2.0 * (a_ki - a_kj) + 1e-12
        delta_needed = (b[k_max] - second_max) / denom
        if delta_needed <= 0:
            # maybe small epsilon
            delta_needed = 0
        delta = min(delta_needed * 0.95, a[i])  # safety factor
        if delta <= 1e-12:
            break

        # incremental update to convolution
        b_old = b.copy()
        # Adding delta at position j adds 2*delta*a to entries b[j:j+n]
        b[j:j+n] += 2.0 * delta * a
        # Removing from i subtracts 2*delta*a
        b[i:i+n] -= 2.0 * delta * a
        # adjust diagonal terms
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        # adjust a
        a[i] -= delta
        a[j] += delta
        a = symmetrize(a)  # maintain symmetry
        # reproject onto simplex
        a = project_simplex(a)

        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val = cur_val
            best_local_a = a.copy()
            if best_local_val <= best_val:
                best_val = best_local_val
                print(f"[Refine] new best = {best_local_val:.6f}")
                # could break early if target reached
                if best_local_val <= target_val: # need target_val variable accessible? We'll use global TARGET.
                    pass
        else:
            # revert if not beneficial
            b = b_old
            a[i] += delta
            a[j] -= delta
            # no further moves break?
            # continue loop
        it += 1
        if it % 2000 == 0:
            print(f"[Refine] iter {it}, best = {best_local_val:.6f}")

    return best_local_a, best_local_val

def mix_with_uniform(a, n):
    best_a = a.copy()
    best_val = objective_val(best_a)
    uniform = np.full(n, 1.0 / n)
    for alpha in [0.001,0.002,0.005,0.01,0.02]:
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = objective_val(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    return best_a, best_val

"""Optimise a non‑negative step function for the max‑self‑convolution objective.

The algorithm follows three ideas:
1.  Initialise with a family of symmetric Beta‑distributions (a low‑dimensional
    parametric search) and keep the best candidate.
2.  Refine the candidate with Adam on a smooth (log‑sum‑exp) approximation of
    the max‑convolution, maintaining the simplex constraints via the soft‑max
    parametrisation.
3.  Polish the result with a projected sub‑gradient method and a greedy
    deterministic “mass‑transfer” refinement.
Finally a tiny uniform mixing is tried.  All steps respect the time budget
(`budget_s` seconds)."""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    out_len = x.shape[0] + y.shape[0] - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def objective_val(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i}."""
    return (a + a[::-1]) / 2.0


# ----------------------------------------------------------------------
# Parametric initialisation – symmetric Beta densities
# ----------------------------------------------------------------------
def beta_pdf(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) density evaluated on n mid‑points on [0,1]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # log‑Beta function for stability
    logB = math.lgamma(alpha) + math.lgamma(alpha) - math.lgamma(2.0 * alpha)
    log_pdf = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x)) - logB
    pdf = np.exp(log_pdf)
    return pdf


def beta_grid_search(n: int, alphas: np.ndarray,
                     start: float, deadline: float) -> tuple[np.ndarray, float]:
    """Search the one‑parameter Beta family; return best (a, value)."""
    best_a = None
    best_val = np.inf
    for alpha in alphas:
        if time.time() >= deadline:
            break
        a = beta_pdf(alpha, n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        a = symmetrize(a)
        val = objective_val(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth surrogate of max(conv)
# ----------------------------------------------------------------------
def adam_phase(a0: np.ndarray, n: int,
               start: float, deadline: float,
               best_a: np.ndarray, best_val: float,
               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on log‑sum‑exp surrogate of max(conv)."""
    w = np.log(a0 + 1e-12)               # logits → soft‑max
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.004      # aggressive start, gentle finish
    eps_start, eps_end = 0.10, 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80000

    while time.time() < deadline:
        # ---- map logits to the simplex ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        a = symmetrize(a)
        a = project_simplex(a)

        cur_val = objective_val(a)
        if cur_val < best_val - 1e-14:
            best_val, best_a, no_improve = cur_val, a.copy(), 0
            print(f"[Adam] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- schedules (fraction of total phase) ----
        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * ((eps_end / eps_start) ** prog)
        lr = lr_start * ((lr_end / lr_start) ** prog)

        # ---- smooth‑max via log‑sum‑exp ----
        b = full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps          # ≤ 0
        weights = np.exp(shifted)
        p = weights / weights.sum()          # soft‑max over convolution positions

        # ---- gradient of the smooth objective w.r.t. a ----
        p_rev = p[::-1]
        conv_ab = conv_fft(a, p_rev)         # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]   # the part that aligns with a
        grad_a = 4.0 * n * segment[::-1]

        # ---- chain rule through soft‑max ----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to shake out of plateaux
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.009, size=n)

        if no_improve > IMPROVE_LIMIT:
            print("[Adam] restart after stagnation")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def subgradient_phase(a0: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    best_local_a = best_a.copy()
    best_local_val = best_val
    eta0 = 1.0 / n            # initial step size
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 60000

    while time.time() < deadline:
        b = full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Subgrad] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))
        # sub‑gradient of the active (max) entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # gradient step + Euclidean projection onto simplex
        a = project_simplex(a - eta * grad)
        a = symmetrize(a)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            print("[Subgrad] perturb after stagnation")
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            a = symmetrize(a)
            eta = eta0
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase C – Deterministic greedy refinement
# ----------------------------------------------------------------------
def deterministic_refine(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         current_best: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Greedy transfers that analytically lower the current maximal convolution."""
    a = a0.copy()
    b = full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_val = cur_val
    best_a = a.copy()
    it = 0

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contributions of each index to the maximal entry
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]

        i = int(np.argmax(contrib))
        if a[i] <= 0.0:
            break

        # choose a target with the smallest current mass (different from i)
        j = int(np.argmin(a + (np.arange(n) == i) * 1e9))
        # verify that moving i→j can actually reduce the peak
        a_ki = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_kj = a[k_max - j] if 0 <= k_max - j < n else 0.0
        if a_ki <= a_kj + 1e-12:
            # try a slightly larger target if the first fails
            sorted_idx = np.argsort(a)
            moved = False
            for cand in sorted_idx:
                if cand == i:
                    continue
                a_kj = a[k_max - cand] if 0 <= k_max - cand < n else 0.0
                if a_ki > a_kj + 1e-12:
                    j = int(cand)
                    moved = True
                    break
            if not moved:
                break

        # second‑largest convolution entry (excluding the current max)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max == -np.inf:
            second_max = 0.0

        # analytic delta that would drop the peak to second_max
        denom = 2.0 * (a_ki - a_kj) + 1e-12
        delta_needed = (b[k_max] - second_max) / denom
        if delta_needed <= 0.0:
            break
        delta = min(delta_needed * 0.93, a[i])   # safety factor

        # ---- incremental update of the convolution ----
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta
        a = project_simplex(a)          # ensure feasibility
        a = symmetrize(a)

        new_val = 2.0 * n * b.max()
        if new_val < best_val - 1e-14:
            best_val, best_a = new_val, a.copy()
            print(f"[Refine] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            # revert – move was not beneficial
            b = b_old
            a[i] += delta
            a[j] -= delta
            # break to avoid endless futile attempts
            break

        it += 1
        if it % 2500 == 0:
            print(f"[Refine] iter {it:5d} – best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Small post‑processing: mix a tiny amount of uniform mass
# ----------------------------------------------------------------------
def mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a little uniform mass (often helps)."""
    best_a, best_val = a, objective_val(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.002, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = objective_val(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val
