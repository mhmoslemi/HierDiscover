def _mirror_descent_opt(a0, n, deadline):
    a = a0.copy()
    best_a, best_val = a.copy(), _objective(a)
    step0 = 0.5
    it = 0
    while time.time() < deadline:
        # conv
        size = 1 << ((2*n -2).bit_length())
        fx = np.fft.rfft(a, size)
        conv = np.fft.irfft(fx * fx, size)[:2*n-1]
        kmax = int(conv.argmax())
        # gradient
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]
        # step size schedule
        eta = step0 / np.sqrt(it+1)
        # update with exp
        a = a * np.exp(-eta * grad)
        # enforce positivity small epsilon
        eps = 1e-12
        a = np.maximum(a, eps)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a.copy(), val
            print(f"[Mirror {it}] new best {best_val:.6f}")
            # optionally break if val < target.
        it += 1
        # maybe early exit if iter count exceeds something.
        if it > 200000:
            break
    return best_a, best_val

"""
Improved step‑function optimizer.

The search proceeds in three stages:

1️⃣  **Parametric warm‑start** – a symmetric Beta family
    a_i ∝ (i·(n‑i+1))^{p‑1} is optimized by a cheap
    golden‑section line search on the exponent *p*.
    The best shape, a few Dirichlet draws and an
    interpolation of the previous best solution
    (if the global ``height_sequence_1`` variable exists)
    are collected as starting points.

2️⃣  **Mirror‑descent (exponentiated‑gradient) on the simplex** –  
    starting from the best warm‑start we perform many
    sub‑gradient steps of the objective  f(a)=2·n·max(a∗a).
    The sub‑gradient for the active convolution index k∗ is
    g_j = 2·a_{k∗‑j}.  The update uses a multiplicative
    rule a_j ← a_j·exp(‑η·g_j) and a Euclidean re‑normalisation,
    i.e. an entropic‑mirror descent appropriate for the
    probability simplex.  The step size η decays like 1/√t.
    This phase quickly pushes the maximal autocorrelation down.

3️⃣  **Intensive hill‑climbing** – a refined version of the
    simulated‑annealing mass‑transfer routine from the reference
    implementation.  It repeatedly moves a random fraction of
    mass from a “source’’ index that heavily contributes to the
    current maximal convolution entry to a “target’’ index that
    contributes little.  The schedule for the transferred
    fraction and the temperature are annealed linearly.
    Stagnation triggers a small Gaussian perturbation.

The whole pipeline respects the 1000 s budget; intermediate
objective values are printed for monitoring.  The final
candidate is returned as a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())   # next power of two
    fx = np.fft.rfft(a, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Symmetric Beta family a_i ∝ (i·(n‑i+1))^{p‑1}
# ----------------------------------------------------------------------
def _beta_shape(p: float, q: float, n: int) -> np.ndarray:
    """Return a normalized symmetric Beta shape for given exponents."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        # fallback to uniform if parameters are pathological
        a = np.ones_like(a) / n
    else:
        a /= s
    return a


def _beta_sym_optimize(n: int,
                       p_low: float = 0.2,
                       p_high: float = 4.0,
                       tol: float = 1e-3) -> tuple[np.ndarray, float, float]:
    """
    Golden‑section search for the exponent p that minimises the objective
    of the symmetric Beta family a_i ∝ (i·(n‑i+1))^{p‑1}.
    Returns the best shape, its value and the optimal p.
    """
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    invphi = 1.0 / phi
    a = p_low
    b = p_high
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)

    a_vec = _beta_shape(c, c, n)
    b_vec = _beta_shape(d, d, n)
    fc = _objective(a_vec)
    fd = _objective(b_vec)

    while (b - a) > tol:
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - invphi * (b - a)
            a_vec = _beta_shape(c, c, n)
            fc = _objective(a_vec)
        else:
            a, fc = c, fd
            c = d
            d = a + invphi * (b - a)
            b_vec = _beta_shape(d, d, n)
            fd = _objective(b_vec)

    p_opt = 0.5 * (a + b)
    best = _beta_shape(p_opt, p_opt, n)
    best_val = _objective(best)
    return best, best_val, p_opt


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the simplex
# ----------------------------------------------------------------------
def _mirror_descent_opt(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iter: int = 200_000,
                        init_eta: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent using the sub‑gradient of the maximal
    autocorrelation entry.  The step size η decays as η₀/√t.
    """
    a = a0.copy()
    size = 1 << ((2 * n - 1).bit_length())
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        # convolution and active index
        fx = np.fft.rfft(a, size)
        conv = np.fft.irfft(fx * fx, size)[:2 * n - 1]
        kmax = int(conv.argmax())

        # sub‑gradient for the active entry
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step size schedule
        eta = init_eta / math.sqrt(it + 1.0)

        # exponentiated update and re‑normalisation
        a = a * np.exp(-eta * grad)
        eps = 1e-15
        a = np.maximum(a, eps)
        a /= a.sum()

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Mirror {it:7d}] new best = {best_val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing (mass‑transfer annealing) – refined version
# ----------------------------------------------------------------------
def _hillclimb_opt(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local stochastic search that repeatedly moves a fraction of mass
    from indices heavily contributing to the current maximal
    convolution entry to indices contributing little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # annealing schedules
    max_frac_start, max_frac_end = 0.9, 0.001
    temp_start, temp_end = 0.05, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while time.time() < deadline:
        prog = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))
        # Above expression is a bit awkward – we just use linear progress:
        prog = min(1.0, max(0.0, (deadline - time.time()) / (deadline - time.time() + 1e-12)))
        # actually we don't need prog; we control schedules directly with iteration
        prog = iteration / max(1.0, (deadline - (deadline - time.time())))

        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each i to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source index (large contribution)
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= 1e-15:
            iteration += 1
            continue

        # choose a target index that contributes little
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + 1e-15)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fall back to a random different index
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= 1e-15:
            iteration += 1
            continue

        # Update convolution efficiently (same formulae as the reference code)
        b_new = b.copy()
        # Adding delta to a[j_tgt] contributes 2*delta*a to relevant positions
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        # Removing delta from a[i_src] subtracts 2*delta*a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        # Self‑terms that were double‑counted
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb to escape stagnation
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val

        print("[phase] starting mirror descent …")
        a_md, val_md = _mirror_descent_opt(global_best_a, n, md_deadline)
        if val_md < global_best_val:
            global_best_a, global_best_val = a_md, val_md
            print(f"[phase] after mirror descent best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Intensive hill‑climbing with annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0 and global_best_a is not None:
        print("[phase] starting hill‑climbing …")
        global_best_a, global_best_val = _hillclimb_opt(
            global_best_a,
            n,
            deadline,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
