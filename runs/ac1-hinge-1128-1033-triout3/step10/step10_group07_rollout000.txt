**Projection onto simplex**:

Implementation:

```
def project_onto_simplex(v):
    # ensures non-negative and sum=1
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step function that (approximately) minimises
the evaluate_sequence objective:
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in several stages:
1. Generate a pool of diverse initial candidates (previous solution,
   exponential / power‑law families, Dirichlet draws, etc.) and pick the
   best one (normalized to sum = 1).
2. Refine the candidate with a simple projected‑gradient descent on the
   probability simplex.  The gradient is the sub‑gradient of the maximal
   convolution entry; we use a line‑search over a few step sizes and
   project back onto the simplex after each step.
3. If progress stalls we add a small random jitter or re‑initialise near
   the current best.
4. After the gradient phase we perform a cheap random‑swap local search
   (move a small amount of mass between two coordinates) to escape
   shallow minima.
5. Finally we try a tiny convex combination with the uniform vector,
   which sometimes lowers the bound further.

All heavy work uses NumPy’s FFT (via rfft) for O(n log n) convolution.
The whole routine respects the 1000 s budget and prints intermediate
objective values for monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)−1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical cleanup
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Parameter‑family generator for the initial pool
# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, rng: np.random.Generator):
    """Yield a variety of candidate vectors built from simple parametric families."""
    i = np.arange(n, dtype=np.float64)

    # 1. Exponential (U‑shaped) family: a_i ∝ r**i + r**(n‑1‑i)
    for r in np.linspace(0.2, 0.99, 30):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 2. Power‑law (edge‑peaked) family: a_i ∝ ((i+1)*(n‑i))**(-p)
    for p in np.linspace(-8.0, 8.0, 36):
        base = (i + 1) * (n - i) + 1e-12
        a = np.power(base, -p)
        yield a

    # 3. Symmetric reciprocal powers: a_i ∝ (i+1)**-p + (n-i)**-p
    for p in np.linspace(0.2, 6.0, 24):
        a = np.power(i + 1, -p) + np.power(n - i, -p)
        yield a

    # 4. Cosine‑window family with exponent β
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))   # Hann window
    for beta in np.linspace(0.5, 5.0, 25):
        a = np.power(w, beta)
        yield a

    # 5. Random Dirichlet draws – a few quick samples
    for _ in range(50):
        a = rng.dirichlet(np.ones(n))
        yield a


# ----------------------------------------------------------------------
# Gradient of the max‑convolution entry (sub‑gradient)
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute the sub‑gradient of 2·n·max(conv(a,a)) w.r.t. a
    (ignoring the constant 2·n) and return the gradient and the current
    objective value.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    k_max = int(np.argmax(conv))
    # gradient_i = 2 * a_{k_max - i} if the shifted index lies in [0, n‑1]
    idx = k_max - np.arange(n, dtype=int)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[idx[mask]]
    # The full objective (including the factor 2·n) is 2·n·max_val
    obj = 2.0 * n * max_val
    return grad, obj


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex with simple line‑search
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray, n: int, deadline: float, rng: np.random.Generator):
    """
    Refine a0 using projected gradient descent (sub‑gradient of the max‑conv).
    Returns the best vector seen and its objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_obj = _objective_value(a)

    # Initial step‑size (can be large because we project after the step)
    step_size = 0.8
    no_improve = 0
    iter_cnt = 0
    IMPROVE_LIMIT = 12000   # after this many stagnating iters we jitter

    while time.time() < deadline:
        # gradient of the current max‑convolution entry
        grad, cur_obj = _max_conv_gradient(a)

        # Try a few step sizes and keep the best resulting point
        candidates = []
        for scale in (step_size, step_size * 0.5, step_size * 0.25):
            a_candidate = _project_onto_simplex(a - scale * grad)
            val = _objective_value(a_candidate)
            candidates.append((val, a_candidate, scale))
        # Choose the best candidate
        candidates.sort(key=lambda x: x[0])
        new_val, new_a, used_scale = candidates[0]

        # Update best solution if improved
        if new_val < best_obj - 1e-14:
            best_obj, best_a = new_val, new_a.copy()
            no_improve = 0
            # modestly increase step size after a win
            step_size = min(step_size * 1.05, 2.0)
            print(f"[PGD {iter_cnt:6d}] new best = {best_obj:.6f}  step = {used_scale:.4f}")
        else:
            no_improve += 1
            # shrink step size when not improving
            step_size = max(step_size * 0.92, 1e-4)

        a = new_a

        # Occasional random jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(scale=0.001, size=n)
            a = _project_onto_simplex(a + jitter)
            print(f"[PGD {iter_cnt:6d}] jitter applied")

        # Restart from the best if we have been stagnant for a while
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += rng.normal(scale=0.01, size=n)   # moderate perturbation
            a = _project_onto_simplex(a)
            step_size = 0.8
            no_improve = 0
            print(f"[PGD {iter_cnt:6d}] restart after stagnation")

        iter_cnt += 1

        # Stop a little early to leave room for the final pass
        if time.time() > deadline - 0.3:
            break

    return best_a, best_obj


# ----------------------------------------------------------------------
# Tiny uniform mixing (often beneficial at the very end)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for several α and keep the best."""
    n = a.shape[0]
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_onto_simplex(cand)   # keep sum = 1
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Local random swap (mass transfer) search
# ----------------------------------------------------------------------
def _local_swap_search(a: np.ndarray, n: int, deadline: float, rng: np.random.Generator):
    """
    Simple stochastic local search: move a small fraction of mass from a
    randomly chosen index i to another index j (preserving the sum).  Keep
    any improvement.
    """
    best_a = a.copy()
    best_val = _objective_value(best_a)
    iter_cnt = 0
    # Number of attempts is limited by remaining time
    while time.time() < deadline:
        i = rng.integers(n)
        if best_a[i] <= 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        # Transfer at most 5 % of the source weight
        delta = best_a[i] * rng.uniform(0.0, 0.05)
        if delta <= 0:
            continue
        a_candidate = best_a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # No need to re‑project – sum is unchanged and all entries stay non‑negative
        val = _objective_value(a_candidate)
        if val < best_val:
            best_val, best_a = val, a_candidate
            print(f"[Swap {iter_cnt:6d}] new best = {best_val:.6f}")
        iter_cnt += 1
        if time.time() > deadline - 0.2:
            break
    return best_a, best_val
