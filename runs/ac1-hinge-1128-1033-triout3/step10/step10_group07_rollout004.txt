"""
Propose a non‑negative step function (probability vector) that lowers
evaluate_sequence = 2·n·max(conv(a,a)) / (sum a)^2.
The algorithm mixes three ideas:

1. **Warm start** – reuse a previously found sequence if available,
   otherwise pick the best among several simple parametric families
   (exponential, power‑law, cosine‑window, Dirichlet draws).

2. **Soft‑max mirror descent** – replace the nonsmooth max by a smooth
   log‑sum‑exp (soft‑max).  For a given smoothing parameter λ we minimise
   Fλ(a)=log∑_k exp(λ·(a*a)_k) on the probability simplex using an
   exponentiated‑gradient update.
   The gradient can be written as a weighted correlation
   ∇Fλ(a) = 2·corr(w,a) / ∑w where w_k = exp(λ·((a*a)_k−max)).
   Both the self‑convolution and the correlation are evaluated with a
   single FFT of size 2^⌈log2(3n)⌉, so one iteration is cheap
   (≈0.5‑2 ms on a typical machine).

   We start with a small λ (very smooth) and increase it stepwise,
   allowing the iterates to first explore a wide region and then
   focus on the true max‑objective.

3. **Local stochastic refinement** – after the deterministic phase we
   perform cheap random “mass‑transfer’’ moves (move ε mass from one
   index to another) and keep any improvement.  This helps to escape
   tiny plateaus that the gradient step may miss.

Finally a tiny mixing with the uniform distribution is tried, as it
often reduces the peak a little further.

The whole routine respects the 1000 s budget, works on a single CPU
and uses only NumPy.  It returns a plain ``list[float]`` that the
evaluation harness will pass to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Compute the full linear convolution a*a using a real FFT of length
    ``size_fft`` (must be ≥ 2·len(a)-1).  Returns an array of length
    2·len(a)-1.
    """
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * a.shape[0] - 1]
    return conv


def _correlation_fft(w: np.ndarray, a_rev: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Compute correlation (w ⋆ a)[i] = Σ_j w[i+j]·a[j] for i=0…len(a)-1.
    It is obtained as the central slice of the convolution w*a_rev.
    ``size_fft`` must be ≥ len(w)+len(a_rev)-1.
    """
    fw = np.fft.rfft(w, size_fft)
    fa_rev = np.fft.rfft(a_rev, size_fft)
    corr_full = np.fft.irfft(fw * fa_rev, size_fft)[: len(w) + len(a_rev) - 1]
    n = a_rev.shape[0]
    # slice indices n-1 … n-1+n-1  (length n)
    return corr_full[n - 1 : n - 1 + n]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a probability vector.
    (The denominator (sum a)^2 is 1 because we keep a normalized.)
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initial candidate generation
# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, rng: np.random.Generator):
    """Yield a handful of hand‑crafted candidates (non‑negative, not normalised)."""
    i = np.arange(n, dtype=np.float64)

    # 1. Exponential‑U shape
    for r in np.linspace(0.55, 0.9999, 30):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 2. Power‑law edge‑peaked
    for p in np.linspace(-4.0, 4.0, 30):
        base = (i + 1) * (n - i) + 1e-12
        a = np.power(base, -p)
        yield a

    # 3. Cosine‑window (Hann) with exponent β
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    for beta in np.linspace(0.5, 4.0, 20):
        a = np.power(w, beta)
        yield a

    # 4. Random Dirichlet draws
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        yield a


# ----------------------------------------------------------------------
# Soft‑max exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _softmax_eg_descent(
    a0: np.ndarray,
    size_fft: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform an exponentiated‑gradient descent on the smooth approximation
    of max(conv).  Returns the best vector seen and its objective value.
    """
    eps = 1e-15
    n = a0.shape[0]
    # Initialise log‑weights
    log_w = np.log(a0 + eps)

    # Schedule of smoothing parameters λ (larger → tighter approximation)
    lambda_schedule = [
        1e1,
        5e1,
        1e2,
        5e2,
        1e3,
        5e3,
        1e4,
        5e4,
        1e5,
    ]

    best_a = a0.copy()
    best_val = _objective(best_a)

    # Global counters
    total_iters = 0

    for lam in lambda_schedule:
        if time.time() >= deadline:
            break

        # Step‑size (decays slowly inside each λ‑phase)
        eta = 0.5
        decay = 0.99997

        # Stagnation handling
        no_improve = 0
        IMPROVE_LIMIT = 5000

        while time.time() < deadline:
            # ----------- reconstruct probability vector -------------
            max_log = np.max(log_w)
            w = np.exp(log_w - max_log)  # stabilised
            a = w / w.sum()

            # ----------- convolution & soft‑max weights -------------
            conv = _full_convolution_fft(a, size_fft)  # length 2n-1
            max_b = conv.max()
            # soft‑max weights (max_b gets weight 1)
            wsoft = np.exp(lam * (conv - max_b))

            Z = wsoft.sum()
            # ----------- gradient via weighted correlation ----------
            a_rev = a[::-1]
            grad_vec = _correlation_fft(wsoft, a_rev, size_fft)  # length n
            grad = (2.0 * grad_vec) / Z

            # ----------- exponentiated‑gradient update -------------
            log_w -= eta * grad
            eta *= decay

            # occasional jitter to escape flat regions
            if rng.random() < 0.002:
                log_w += rng.normal(scale=0.008, size=n)

            # ----------- evaluate current objective ---------------
            cur_val = 2.0 * n * max_b  # identical to _objective(a)
            if cur_val < best_val - 1e-15:
                best_val = cur_val
                best_a = a.copy()
                no_improve = 0
                print(f"[EG λ={lam:g} it={total_iters:6d}] new best = {best_val:.6f}")
            else:
                no_improve += 1

            # stagnation handling – re‑centre around the best so far
            if no_improve > IMPROVE_LIMIT:
                log_w = np.log(best_a + eps)
                log_w += rng.normal(scale=0.02, size=n)
                eta = 0.5
                no_improve = 0
                print(f"[EG λ={lam:g} it={total_iters:6d}] perturb after stagnation")

            total_iters += 1

            # safety break close to deadline
            if time.time() > deadline - 0.15:
                break

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (helps to shave off a few %)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α values."""
    n = a.shape[0]
    best_a, best_val = a, _objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Random local refinement (mass‑transfer moves)
# ----------------------------------------------------------------------
def _local_random_refinement(a: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Perform cheap stochastic moves: move a small amount of mass from a
    random index to another.  Keep any improvement.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(best_a)

    # Pre‑allocate a buffer for the vector to avoid repeated allocations
    cand = best_a.copy()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random fraction of the source mass (up to 5 % of its value)
        eps = rng.random() * 0.05 * best_a[i]
        if eps <= 0.0:
            continue
        cand[i] = best_a[i] - eps
        cand[j] = best_a[j] + eps
        # maintain non‑negativity (should already hold)
        # compute objective – use the fast FFT routine
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            # keep the improvement for the next move
            #print(f"[local] improved to {best_val:.6f}")
        else:
            # revert changes
            cand[i] = best_a[i]
            cand[j] = best_a[j]

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimises a non‑negative step function under the given time budget.
    Returns a list of floats (probability vector) ready for evaluate_sequence.
    """
    # ----- reproducibility -------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ----- problem size ----------------------------------------------------
    # 9000 gives a good trade‑off between expressiveness and FFT cost.
    n = 9000
    size_fft = _next_pow2(3 * n - 2)  # enough for correlation (w * a_rev)

    # ----- initialise -------------------------------------------------------
    try:
        # reuse a previously discovered good construction if present
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        a0 = np.interp(x_new, x_old, old)
        a0 = np.clip(a0, 0.0, None)
        a0 /= a0.sum()
        print("[init] Re‑used height_sequence_1")
    except Exception:
        # parametric sweep to obtain a decent starter
        best_candidate = None
        best_val = np.inf
        for cand in _generate_parametric_candidates(n, rng):
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s <= 0.0:
                continue
            cand /= s
            val = _objective(cand)
            if val < best_val:
                best_val, best_candidate = val, cand.copy()
        if best_candidate is None:
            best_candidate = np.full(n, 1.0 / n)
            best_val = _objective(best_candidate)
        a0 = best_candidate
        print(f"[init] Parametric sweep gave initial value = {best_val:.6f}")

    # ----- Phase 1: soft‑max exponentiated‑gradient descent -----------------
    phase1_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _softmax_eg_descent(
        a0, size_fft=size_fft, deadline=phase1_deadline, rng=rng
    )
    print(f"[phase1] best value after EG = {best_val:.6f}")

    # ----- Phase 2: random local refinement --------------------------------
    phase2_deadline = start_time + 0.95 * (deadline - start_time)
    best_a, best_val = _local_random_refinement(best_a, phase2_deadline, rng)
    print(f"[phase2] best value after local search = {best_val:.6f}")

    # ----- Phase 3: tiny uniform mixing ------------------------------------
    best_a, best_val = _mix_with_uniform(best_a)
    print(f"[mix] best value after uniform mixing = {best_val:.6f}")

    # ----- Final evaluation (the harness will also call evaluate_sequence) -
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
