import cvxpy as cp

def solve_lp_max_sum(f):
    n = len(f)
    M = np.max(_full_convolution(f))
    # build sparse matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        low = max(0, k-(n-1))
        high = min(n-1, k)
        # i from low to high:
        i_vals = np.arange(low, high+1)
        b_idx = k - i_vals
        rows.extend([k]*len(i_vals))
        cols.extend(b_idx.tolist())
        data.extend(f[i_vals].tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    if b_var.value is None:
        return None
    return np.maximum(b_var.value, 0.0)

Improved non‑negative step‑function optimiser.
Algorithm:
1. Initialise with a set of parametric families (Beta, exponential,
   power‑law) and a few random Dirichlet seeds;
2. Keep the best candidate (minimal 2·n·max(conv)).
3. Refine the best candidate by an entropy‑mirror‑descent on the
   simplex, but the sub‑gradient is built from the *M* largest
   convolution entries (rather than a fixed small set) and uses the
   full derivative a_j + a_{k−j}.
4. Occasionally accept uphill moves (Metropolis) and after a long
   period without improvement inject Gaussian noise.
The schedule for the stepsize η and temperature T is aggressive,
starting large and decaying to near‑zero, which lets the optimiser
escape shallow plateaus and then finely polish the solution.
"""

import time, math, numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    a /= total
    return a

def _exp_family(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i = r**i (0 < r < 1)."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    a /= total
    return a

def _power_law_family(n: int, alpha: float) -> np.ndarray:
    """Power‑law decay a_i ∝ (i+1)^{-alpha}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    a /= total
    return a

def _mirror_update(a: np.ndarray,
                   conv: np.ndarray,
                   top_idx: np.ndarray,
                   top_vals: np.ndarray,
                   eta: float) -> np.ndarray:
    """
    Entropy mirror‑descent step built from the *M* largest
    convolution entries.  For each selected index k the sub‑gradient
    of (a∗a)_k is a_j + a_{k−j}.
    """
    n = a.shape[0]
    # Normalise weights – they already sum to 1
    weights = top_vals / top_vals.sum()
    g = np.zeros_like(a)

    for k, w in zip(top_idx, weights):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        if low > high:
            continue
        j = np.arange(low, high + 1, dtype=np.int64)
        # a_j term
        g[j] += w * a[j]
        # a_{k-j} term
        g[j] += w * a[k - j]

    # multiplicative update (entropy mirror)
    a_new = a * np.exp(-eta * g)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]

            cur_val = cand_val

        # Stagnation handling – Gaussian perturbation around the best.
        # ------------------------------------------------------------------

            a = np.maximum(best_a + noise, 0.0)

                float(2.0 * n * _full_convolution(a).max())))

    # 3️⃣  Final secure evaluation.

        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    """Full linear convolution of ``x`` with itself via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

    n = a.shape[0]

# ----------------------------------------------------------------------

                   top_vals: np.ndarray,

    """

    (a∗a)_k is a_j + a_{k‑j}.  The contributions are weighted by the
    normalised values of the chosen entries.

    # Normalise the weights – they already sum to one but we make it explicit
    weights = top_vals / top_vals.sum()

    for k, w in zip(top_idx, weights):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        if low > high:

        j = np.arange(low, high + 1, dtype=np.int64)
        # derivative part a_j

        # derivative part a_{k‑j}

    s = a_new.sum()
