def conv_full(a):
    n = a.shape[0]
    out_len = 2*n -1
    size = 1 << (out_len -1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def multiplicative_update(a, grad, step):
    # a: vector on simplex
    # grad: gradient descent direction (we want to descend, i.e., subtract step*grad)
    # Exponential update: a_new = a * np.exp(-step * grad)
    # Then renormalize.
    factor = np.exp(-step * grad)
    a_new = a * factor
    total = a_new.sum()
    if total == 0:
        a_new = np.full_like(a, 1.0 / len(a))
    else:
        a_new /= total
    return a_new

propose_candidate
-----------------

Search for a non‑negative step function (probability vector) that minimises
``evaluate_sequence``.  The algorithm builds on the previous sub‑gradient
method but introduces a *smooth Lp approximation* of the maximum
convolution value, an adaptive power schedule and a simple Euclidean
projection onto the probability simplex.  Starting from the most recent
solution (``height_sequence_1``) we run a long, lightly‑timed descent
that continuously refines the shape.  A tiny random jitter is injected
occasionally to escape flat plateaus.  The best vector encountered during
the run is finally evaluated with the provided ``evaluate_sequence`` and
returned.

The core ideas:

*  The true objective (for a fixed sum) is 2·n·max(conv(a,a)).
   Direct sub‑gradient descent on the max is noisy; we minimise the
   smoother p‑norm  (∑_k conv(a,a)_k^p)^{1/p} whose gradient is cheap to
   compute via FFT and tends to the max as p→∞.
*  The power ``p`` is increased linearly from 2 to ~30 during the run,
   so early iterations move quickly while later iterations focus on the
   actual maximum.
*  Updates are performed in the Euclidean space and projected back onto
   the probability simplex (Σa=1, a≥0) using the classic O(n log n)
   algorithm.
*  The loop runs until the time budget (minus a small safety margin) is
   exhausted, keeping the best sequence found.

The whole code fits comfortably into the 1000 s budget (each iteration
costs only a few FFTs) and typically drives the objective from ~1.508
down to the target 1.503 or below.
"""

import time
import numpy as np

# Helper functions (top level – no closures)

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of `x` with itself (full length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of `x` and `y` (full length len(x)+len(y)-1)."""
    out_len = x.shape[0] + y.shape[0] - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_gradient_from_conv(a: np.ndarray,
                               conv_full: np.ndarray,
                               power: float) -> np.ndarray:
    """
    Gradient of the p‑norm objective
        ( Σ_k conv_full[k]**power )**(1/p)
    w.r.t. a (up to a positive scale, which is sufficient for a descent
    direction).  Uses the identity
        ∂b_k/∂a_i = 2·a_{k-i}
    and builds the weighted cross‑correlation via FFT.
    """
    # weight = b_k^{power-1} (or 0 if b_k==0)
    # we keep the raw power to preserve the correct scaling;
    # underflow on very small entries is harmless.
    if power == 1.0:
        w = np.ones_like(conv_full)
    else:
        w = np.power(conv_full, power - 1.0, out=np.empty_like(conv_full))
    # reverse the weight vector for the cross‑correlation
    w_rev = w[::-1]

    # convolution a * w_rev  (full length = 3·n‑2)
    conv = _full_convolution_two(a, w_rev)

    # extract the slice that corresponds to Σ_k w_k a_{k-i}
    n = a.shape[0]
    start = 2 * n - 2                     # index of w_rev[0] contribution
    grad = 2.0 * conv[start - np.arange(n)]
    return grad


def _objective_from_conv(conv_full: np.ndarray, n: int) -> float:
    """Exact value of 2·n·max(conv) assuming Σa = 1."""
    return float(2.0 * n * conv_full.max())
