    deadline = start + budget - 0.5
    # precompute log_i, log_ni
    i = np.arange(1, N+1, dtype=np.float64)
    log_i = np.log(i)
    log_ni = np.log(N - i + 1)

    best_val = float('inf')
    best_a = None

    # Optionally start with existing solution if present, else uniform.
    try:
        old = np.asarray(height_sequence_1, dtype=float)

        if old.shape[0] != N:
            # use interpolation
            new_x = np.linspace(0,1,N)
            old_x = np.linspace(0,1,old.shape[0])
            arr = np.interp(new_x, old_x, old)
            arr = np.maximum(arr,0)
            arr = arr / arr.sum()
        else:
            arr = np.maximum(old,0)
            arr = arr / arr.sum()
        val = _objective(arr)
        best_val, best_a = val, arr.copy()
        print(f"[Init] previous solution value={val}")
    except Exception:
        best_a = np.full(N, 1.0/N)
        best_val = _objective(best_a)
        print(f"[Init] uniform value={best_val}")

    # Random candidate sampling loop
    while True:
        now = time.time()
        if now >= deadline:
            break

        # Choose candidate type randomly weighted
        r = np.random.rand()
        if r < 0.6:
            # Beta
            logp = np.random.uniform(-9, 2)  # p ~ 1e-4..7.4
            logq = np.random.uniform(-9, 2)
            p = np.exp(logp)

            # compute log densities

            # shift
            log_vals -= np.max(log_vals)
            vals = np.exp(log_vals)
            a = vals / vals.sum()
        elif r < 0.8:
            # Mixture of two Betas
            # sample two parameters
            logp1 = np.random.uniform(-9, 2)
            logq1 = np.random.uniform(-9, 2)
            logp2 = np.random.uniform(-9, 2)
            logq2 = np.random.uniform(-9, 2)
            p1, q1 = np.exp(logp1), np.exp(logq1)
            p2, q2 = np.exp(logp2), np.exp(logq2)
            log_vals1 = (p1 - 1.0) * log_i + (q1 - 1.0) * log_ni

            log_vals1 -= np.max(log_vals1)
            log_vals2 -= np.max(log_vals2)
            vals1 = np.exp(log_vals1)
            vals2 = np.exp(log_vals2)
            a1 = vals1 / vals1.sum()
            a2 = vals2 / vals2.sum()
            alpha = np.random.rand()

"""
Propose a step‑function (non‑negative vector on the simplex) that
minimises the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is deliberately simple and robust:

1. Initialise a good starting point:
   – If a previous solution ``height_sequence_1`` exists, interpolate it.
   – Otherwise start from the uniform distribution.

2. Random‑search phase (≈ 30 % of the time budget):
   – Sample many candidate shapes drawn from flexible families:
       • Beta(p,q) densities (both shape parameters span many orders of magnitude)
       • Random mixtures of two Beta densities
       • Dirichlet vectors with a random concentration parameter per coordinate
   – Evaluate each candidate with a fast FFT‑based convolution.
   – Keep the best candidate seen so far.

3. Local refinement (≈ 60 % of the time budget):
   – Run a projected sub‑gradient descent on the exact objective.
     The sub‑gradient of  max(conv(a,a)) is computed analytically,
     the iterate is projected onto the probability simplex after each step.
   – The step‑size decays linearly from a moderate value to a tiny one.
   – Occasional jitter helps to escape plateaus.

4. Final secure call to ``evaluate_sequence`` and return the best vector.

The whole routine respects the 1000 s budget and prints progress
information useful for debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions ------------------------------------------------------
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)–1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    f(a)=2·n·max(conv(a,a)).
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        # indices where the maximum is attained
        max_idxs = np.flatnonzero(b == max_b)

        # sub‑gradient of the max‑convolution term
        grad = np.zeros_like(a)
        for k in max_idxs:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear step‑size schedule (from 0.3 down to 1e‑5)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG]   new best = {best_val:.6f}")

        # occasional random jitter
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Draw a (p,q) pair on a log‑scale and return the normalised
    Beta(p,q) shape on the discrete grid {1,…,n}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    logi = np.log(i)
    logni = np.log(n - i + 1)

    logp = rng.uniform(-9.2, 2.3)   # ln(1e‑4) … ln(10)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)

    log_vals = (p - 1.0) * logi + (q - 1.0) * logni
    log_vals -= np.max(log_vals)               # numerical stability
    vals = np.exp(log_vals)
    if vals.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return vals / vals.sum()
