"""
Improved search for a low‑autocorrelation non‑negative step function.

The algorithm works in three stages:

1️⃣  Initialise from a previously saved solution (if available) and
    explore a richer family of parameterised shapes:
    * Beta‑shaped (already used before)
    * Exponential decay a_i ∝ λ^i
    * Power‑law a_i ∝ (i+1)^‑α
    A dense grid together with a log‑uniform random sweep is performed.
    The best shape becomes the starting point.

2️⃣  Refined descent on the *exact* objective.
    Instead of moving mass only away from the *single* maximal convolution
    entry, we look at the K largest entries (K≈5) and build a binary mask
    w over the full convolution.  The gradient of the summed masked
    autocorrelation can be written analytically as a pair of convolutions,
    which we evaluate by FFT (O(n log n)).  A linearly decaying step‑size
    (0.3 → 1e‑5) together with Euclidean projection onto the probability
    simplex yields an updated distribution.  Small random jitter is added
    occasionally to escape plateaus.

3️⃣  The best sequence discovered is finally evaluated with the provided
    ``evaluate_sequence`` routine.

The whole search respects a 1000 s budget; the loop stops a little
earlier to leave time for the final secure evaluation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a vector with itself (length 2·len(x)‑1)
    using an FFT of the next power of two for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function *without* the outer 2·n factor,
    assuming Σ a_i = 1.  The returned value equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    Σ x_i = 1, x_i ≥ 0.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Normalised Beta shape:
        a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the parameters yield a zero vector.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _exp_shape(lam: float, n: int) -> np.ndarray | None:
    """
    Exponential decay shape a_i ∝ lam**i,   i = 0,…,n‑1 .
    """
    if lam <= 0.0 or lam >= 1.0:
        return None
    i = np.arange(n, dtype=np.float64)
    a = lam ** i
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _powerlaw_shape(alpha: float, n: int) -> np.ndarray | None:
    """
    Power‑law shape a_i ∝ (i+1)^{-alpha},   i = 0,…,n‑1 .
    """
    if alpha < 0.0:
        return None
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha)
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _gradient_mask(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_{k} mask_k * (a*a)_k  with respect to a.
    For a binary mask this equals
        g_i = 2 * Σ_k mask_k * a_{k-i} + 2 * Σ_k mask_k * a_{i-k}
    The two sums are computed with FFT‑based convolutions.
    """
    n = a.shape[0]
    L = mask.shape[0]                 # = 2·n‑1
    total_len = n + L - 1
    size = 1 << (total_len - 1).bit_length()

    a_fft = np.fft.rfft(a, size)
    w_fft = np.fft.rfft(mask, size)
    conv1 = np.fft.irfft(a_fft * w_fft, size)[:total_len]

    w_rev_fft = np.fft.rfft(mask[::-1], size)
    conv2 = np.fft.irfft(a_fft * w_rev_fft, size)[:total_len]

    idx = np.arange(n)
    grad = 2.0 * (conv1[idx] + conv2[L - 1 - idx])
    return grad
