            if two_i < L:
                b_new[two_i] += delta*delta
            if ij < L:
                b_new[ij] -= 2.0 * delta * delta
            # update a
            a[i] -= delta
            a[j] += delta

                cur_val = new_val

                    best_a = a.copy()
                no_improve = 0
                break
            else:

                a[i] += delta
                a[j] -= delta
                # continue to next j

        else:
            # if no j improves, break out after some attempts
            no_improve += 1
        if no_improve > IMPROVE_LIMIT:
            # break or random perturb
            break
    return best_a, best_val

"""
Optimiser for the self‑convolution max‑norm problem.

The pipeline consists of four stages that together respect a 1000 s budget:

  1. **Adam on a smooth max surrogate** – fast gradient‐based descent
     on a log‑sum‑exp approximation of `max(conv(a,a))`.  The schedule
     first uses a relatively large smoothing parameter and then slowly
     tightens it.

  2. **Simulated‑annealing (SA)** – a classic hill‑climber that moves
     a random amount of mass from a “bad” source index to a “good”
     target index, accepting worsening moves with a linearly decreasing
     temperature.

  3. **Exact projected sub‑gradient descent** – a classic sub‑gradient
     method on the *true* max‑convolution objective, using a simple
     simplex projection after each step.

  4. **Greedy analytic refinement** – a deterministic local optimiser.
     It repeatedly transfers the maximal feasible amount of mass from the
     index that contributes most to the current maximal convolution entry
     to the index whose partner value in that entry is smallest.  The
     required step size is computed analytically so that the peak is
     brought down exactly to the second‑largest convolution entry (with a
     safety factor).  This phase is very cheap (O(n) per iteration) and
     is able to shave the last few percent off the objective.

Finally a tiny mixture with the uniform distribution is tried – this
often yields a marginal but free improvement.

All steps operate on vectors normalised to sum = 1, and the objective
`evaluate_sequence` is homogeneous, so the scale is irrelevant.
The algorithm is deterministic apart from the random seeds used in the
SA phase; a NumPy RNG seeded from the global `seed` argument ensures
reproducibility.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)-1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Hand‑crafted initial families + random Dirichlet draws
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a good starting vector (sum = 1) among several families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(candidate):
        nonlocal best_a, best_val
        a = np.asarray(candidate, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform
    _try(np.ones(n))

    # symmetric exponential (U‑shaped)
    for r in (0.9999, 0.9995, 0.998, 0.996, 0.995, 0.99):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _try(a)

    # arcsine‑like (heavy at ends)
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(np.clip(x * (1 - x), eps, None))
    _try(a)

    # random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp surrogate of max(conv)."""
    # initialise the soft‑max parametrisation
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.005
    eps_start, eps_end = 0.2, 5e-5  # keep eps from collapsing too early

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 50000

    while time.time() < deadline:
        # convert logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {t:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # linear schedules, relative to total elapsed time of this phase
        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # ----- smooth max via log‑sum‑exp -----------------
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over convolution positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, rev(p))
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length = 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax (w.r.t. w)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.006, size=n)

        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {t:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing hill‑climbing (exact max)
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Hill‑climbing with a temperature‑controlled acceptance rule."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    T0 = cur_val * 0.025                     # initial temperature (2.5 % of start)
    max_frac_start, max_frac_end = 0.30, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)               # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        k_max = int(np.argmax(b))

        # ---- choose source i -------------------------------------------------
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # ---- choose target j --------------------------------------------------
        # pick an index that currently contributes little to the max entry
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0                                    # avoid source again
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----- incremental update of the convolution -------------------------
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA  {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            # revert move
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb the current state a little and continue
            print(f"[SA  {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase C – Projected sub‑gradient descent (exact max) – polishing
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n            # start step size ≈ half average entry
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry (ordered pairs)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # gradient step + simplex projection
        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # perturb and reset step size
            print(f"[SGD {it:7d}] perturb after stagnation")
            noise = np.random.normal(scale=0.010, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase D – Greedy analytic refinement (new version)
# ----------------------------------------------------------------------
def _greedy_refine(a0: np.ndarray, n: int,
                   start: float, deadline: float,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Deterministic moves that analytically reduce the current maximal
    convolution entry.  At each iteration we:
      * locate the index k_max attaining the current peak;
      * pick the source i with the largest contribution a_i * a_{k_max-i};
      * pick the target j whose *partner* a_{k_max-j} is smallest;
      * compute analytically the maximal admissible delta that would bring
        the peak down to the second‑largest convolution entry;
      * apply the move if it yields a strict improvement.
    The loop stops after the deadline or when no improving move can be found.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = a.copy()
    best_local_val = cur_val

    no_improve = 0
    IMPROVE_LIMIT = 30_000   # after many idle steps we give up

    while time.time() < deadline:
        k_max = int(np.argmax(b))
        M = b[k_max]

        # contribution of each i to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx[mask]]

        i = int(np.argmax(contrib))
        if a[i] <= 0.0:
            break

        # partner values a_{k_max - j} (use ∞ for illegal j)
        partner = np.full(n, np.inf)
        idx_t = k_max - np.arange(n)
        mask_t = (idx_t >= 0) & (idx_t < n)
        partner[mask_t] = a[idx_t[mask_t]]

        # indices sorted by partner (ascending); skip i
        cand_js = np.argsort(partner)
        moved = False

        # second‑largest convolution entry (excluding k_max)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()

        for j in cand_js:
            if j == i:
                continue
            # partner values for the chosen pair
            a_ki = a[k_max - i] if 0 <= k_max - i < n else 0.0
            a_kj = a[k_max - j] if 0 <= k_max - j < n else 0.0
            if a_ki <= a_kj + 1e-14:
                continue   # cannot reduce the peak with this pair

            # analytic delta that would bring b[k_max] down to second_max
            denom = 2.0 * (a_ki - a_kj) + 1e-12
            delta_needed = (M - second_max) / denom
            delta = min(delta_needed * 0.90, a[i])   # safety factor 0.9
            if delta <= 1e-12:
                continue

            # incremental update of convolution (in‑place)
            b_new = b.copy()
            b_new[j:j + n] += 2.0 * delta * a
            b_new[i:i + n] -= 2.0 * delta * a
            L = len(b_new)
            two_j = 2 * j
            two_i = 2 * i
            ij = i + j
            if two_j < L:
                b_new[two_j] += delta * delta
            if two_i < L:
                b_new[two_i] += delta * delta
            if ij < L:
                b_new[ij] -= 2.0 * delta * delta

            a[i] -= delta
            a[j] += delta

            new_val = 2.0 * n * b_new.max()
            if new_val < cur_val - 1e-14:
                # accept
                b = b_new
                cur_val = new_val
                if new_val < best_local_val - 1e-14:
                    best_local_val, best_local_a = new_val, a.copy()
                no_improve = 0
                moved = True
                break
            else:
                # revert and try next candidate
                a[i] += delta
                a[j] -= delta
                continue

        if not moved:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # No improvement for a while – exit
            break

    return best_local_a, best_local_val

# Small post‑processing: mix a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a little uniform mass (often helps)."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a step‑function under the 1000 s time budget.
    Returns a list of non‑negative floats that (approximately) minimise
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final call

    # problem size – 9000 offers a good trade‑off between expressiveness & speed
    n = 9000

    # ------------------------------------------------------------------
    # Initialisation – reuse a known solution if present, otherwise fallback
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈15 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.15 * (deadline - start_time)
    best_a, best_val = _adam_phase(best_a, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (≈55 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _simulated_annealing_phase(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        rng=rng,
        target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase C – Exact projected sub‑gradient descent (≈15 % of budget)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase D – Greedy analytic refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _greedy_refine(
        best_a, n,
        start=phase_c_deadline,
        deadline=deadline,
        best_val=best_val)

    # ------------------------------------------------------------------
    # Final tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
