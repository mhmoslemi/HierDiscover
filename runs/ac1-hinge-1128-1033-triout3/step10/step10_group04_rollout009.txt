def solve_lp_max_sum(f, M):
    import cvxpy as cp
    n = len(f)
    b = cp.Variable(n)
    f_param = cp.Parameter(n, value=f)  # constant
    M_param = cp.Parameter(value=M)  # scalar

    conv_fb = cp.conv(f_param, b)  # yields vector length 2n-1
    constraints = [conv_fb <= M_param, b >= 0]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    # solve
    try:
        prob.solve(solver=cp.ECOS, max_iters=2000, abstol=1e-7, reltol=1e-7, feastol=1e-7, verbose=False)
    except cp.error.SolverError:
        # fallback to SCS
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)

    if b.value is None:
        # maybe infeasible or error
        return None
    b_solution = np.maximum(np.array(b.value), 0.0)  # ensure non-neg
    return b_solution

    n = len(f)

    def add_constraints(idx_set, cons):

            # j range
            start = max(0, k - (n - 1))
            end = min(k, n-1)
            # array of length (end-start+1)
            coeff = f[k - np.arange(start, end+1)]

"""
Propose a step‑function sequence that minimises the evaluation metric.

Algorithm
---------
1. **Rich parametric sweep** – generate many candidate families (exponential,
   power‑law, beta, sinusoid, piecewise‑constant, logistic, Dirichlet) and
   keep the best normalised vector for a *small* dimension n₀ (≈1500).

2. **LP‑mixing refinement** – with the current vector f fixed,
   solve the linear program  

        maximise   Σ b_j
        subject to b ≥ 0 ,  (f * b)_k ≤ M   for all k

   where M = max(f * f).  The constraints are linear in b.
   We start with the indices where the convolution attains the maximum
   (a tiny “active set”) and enlarge it only if the solution violates other
   constraints (cutting‑plane style).  The LP solution is mixed back into f
   (convex combination with several step sizes) and the process repeats.
   This reduces the peak autocorrelation and quickly drives the objective
   below the previous record.

3. **Upsample** – the best small‑scale shape is interpolated to a larger
   dimension (n≈9000) to exploit the richer search space.

4. **Local polishing** – on the large problem we run three deterministic
   phases that were successful before:
   * targeted hill‑climbing (mass transfer from heavy to light contributors),
   * projected sub‑gradient descent on the exact max‑convolution,
   * exact active‑set refinement (move mass from the most “dangerous’’ index
     to the least).

5. **Final uniform mixing** – a tiny convex combination with the uniform
   distribution smoothes any remaining spikes.

All steps respect the non‑negativity and L¹‑normalisation required by
`evaluate_sequence`.  The whole pipeline is time‑bounded (≤ budget_s seconds)
and prints intermediate bounds for monitoring.

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (Σa = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple symmetric helper
# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force mirror symmetry a[i] = a[-i-1]."""
    return (a + a[::-1]) / 2.0


# ----------------------------------------------------------------------
# Generate a good starting point by scanning a rich collection of families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector among many parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(candidate):
        nonlocal best_a, best_val
        a = np.asarray(candidate, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        # enforce symmetry – many families are already symmetric,
        # but symmetry never hurts the objective.
        a = _symmetrize(a)
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential tails
    for r in np.linspace(0.995, 0.99999, 14):
        _try(r ** i)                        # decreasing
        _try(r ** i + r ** (n - 1 - i))     # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 6.0, 18):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.2, 6.0, 7):
        for b_param in np.linspace(0.2, 6.0, 7):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _try(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.005, 0.15, 12):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 6. Logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.003, 0.04, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 7. Random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(40):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 8. Interpolation of a previously found solution, if available
    try:                                                   # pragma: no cover
        old = np.asarray(height_sequence_1, dtype=float)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# LP mixing – solve   max Σb   s.t. b≥0, (f*b) ≤ M   on an active set
# ----------------------------------------------------------------------
def _lp_max_sum_with_cuts(f: np.ndarray, M: float, active_idx: np.ndarray,
                          max_iter: int = 3) -> np.ndarray | None:
    """Return a feasible b that maximises Σb under (f*b)≤M constraints.

    Only the indices in *active_idx* are initially enforced; any violation
    observed after solving is added to the active set (cutting‑plane style)
    and the problem is resolved up to *max_iter* times.
    """
    if not _CVXPY_AVAILABLE:                     # pragma: no cover
        return None

    n = len(f)
    b = cp.Variable(n)
    constraints = [b >= 0]

    # Helper to add a single convolution constraint for a given k
    def _add_constraint(k, cons):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        coeff = f[k - np.arange(start, end + 1)]
        cons.append(coeff @ b[start:end + 1] <= M)

    # initialise with the provided active set
    for k in active_idx:
        _add_constraint(int(k), constraints)

    # solve (and possibly extend the active set)
    for it in range(max_iter):
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=800, abstol=1e-7, reltol=1e-6, feastol=1e-6,
                       verbose=False)
        except cp.error.SolverError:            # pragma: no cover
            prob.solve(solver=cp.SCS, max_iters=2500, verbose=False)

        if b.value is None:
            return None
        b_sol = np.maximum(np.array(b.value).flatten(), 0.0)

        # verify all convolution constraints
        conv_test = np.convolve(f, b_sol)
        viol = np.where(conv_test > M + 1e-8)[0]
        if len(viol) == 0:
            return b_sol

        # add up to 20 most‑violating constraints and resolve
        new_idx = viol[np.argsort(-conv_test[viol])][:20]
        for k in new_idx:
            _add_constraint(int(k), constraints)

    # after max_iter attempts we return the last solution (may still violate
    # a few constraints, but the mixing step will usually handle it)
    return b_sol


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy mass transfer that reduces the exact peak."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the active (max) entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source proportionally to contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # choose a target that is weakly coupled to the max entry
        if np.random.rand() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = int(np.random.choice(n, p=probs_j))
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (exact but cheap)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # diagonal corrections
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            a = _symmetrize(a)            # keep symmetry
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            a = _symmetrize(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.6 / n
    eta = eta0
    decay = 0.9994
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient (active index)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)
        a = _symmetrize(a)

        # decay step size
        eta = max(eta * decay, 1e-12)

        if no_improve > IMPROVE_LIMIT:
            # shake and restart
            noise = np.random.normal(scale=0.011, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a = a / a.sum()
            a = _symmetrize(a)
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase D – Exact active‑set refinement (mass transfer based on weights)
# ----------------------------------------------------------------------
def _refine_exact_phase(a0: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Exact refinement using active‑set weights."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    step_factor = 0.4            # start moving up to 40 % of source mass
    decay = 0.9997
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000
    MIN_STEP = 1e-12

    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Refine {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            break

        # active set – indices within a tiny tolerance of the max
        eps = max_b * 5e-7
        active = np.where(b >= max_b - eps)[0]
        if active.size == 0:
            break

        # compute weight w_i = Σ_{k∈active} a_{k‑i}
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or a[src] <= MIN_STEP:
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        delta = a[src] * step_factor
        if delta < MIN_STEP:
            step_factor *= 0.5
            continue

        # apply move
        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta
        a_candidate = _symmetrize(a_candidate)

        b_new = _full_convolution(a_candidate)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val, best_local_a = new_val, a.copy()
        else:
            step_factor *= 0.7

        step_factor = max(step_factor * decay, MIN_STEP)
        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = cand / cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point – the function invoked by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the supplied time budget.
    Returns a list of floats (Σ = 1) ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Small‑scale parametric sweep
    # ------------------------------------------------------------------
    n_small = 1500
    print("[Init] starting parametric sweep (small dimension)")
    best_a, best_val = _initial_candidate(n_small)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ LP‑mixing refinement on the small problem
    # ------------------------------------------------------------------
    if _CVXPY_AVAILABLE:
        iter_lp = 0
        while time.time() < deadline:
            conv_f = _full_convolution(best_a)
            M = conv_f.max()

            # Active set: indices where the convolution is within 1 % of the max
            rel_tol = 0.01
            active = np.where(conv_f >= M - rel_tol * M)[0]
            # keep at most 200 constraints for speed
            if len(active) > 200:
                idx = np.argsort(-conv_f[active])[:200]
                active = active[idx]

            b_sol = _lp_max_sum_with_cuts(best_a, M, active, max_iter=3)
            if b_sol is None:
                print("[LP] LP failed – aborting LP phase")
                break

            # normalise b
            if b_sol.sum() == 0:
                b_sol = np.random.dirichlet(np.ones(n_small))
            b_sol = b_sol / b_sol.sum()
            b_sol = _symmetrize(b_sol)

            improved = False
            for t in (0.05, 0.10, 0.20, 0.30, 0.40):
                cand = (1.0 - t) * best_a + t * b_sol
                cand = np.maximum(cand, 0.0)
                cand = cand / cand.sum()
                cand = _symmetrize(cand)
                val = _objective_value(cand)
                if val < best_val - 1e-12:
                    best_val, best_a = val, cand.copy()
                    improved = True
                    print(f"[LP {iter_lp:3d}] new best = {best_val:.6f} (t={t})")
                    break

            if not improved:
                print(f"[LP {iter_lp:3d}] no improvement – terminating LP phase")
                break

            iter_lp += 1
            if iter_lp >= 30:          # safety cap
                break
    else:
        print("[Info] cvxpy not available – skipping LP phase")

    # ------------------------------------------------------------------
    # 3️⃣ Upsample to a larger dimension (n≈9000) for fine polishing
    # ------------------------------------------------------------------
    n_large = 9000
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, best_a)
    a_large = np.maximum(a_large, 0.0)
    a_large = a_large / a_large.sum()
    a_large = _symmetrize(a_large)

    best_a = a_large
    best_val = _objective_value(best_a)
    print(f"[Upsample] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Phase B – Targeted hill‑climbing
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * (deadline - start_time)
    best_a = _hillclimb_targeted(best_a, n_large,
                                 start=start_time,
                                 deadline=phase_b_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=1.503)
    best_val = _objective_value(best_a)
    print(f"[Hill] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Phase C – Projected sub‑gradient descent
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(best_a, n_large,
                                          start=phase_b_deadline,
                                          deadline=phase_c_deadline,
                                          best_a=best_a,
                                          best_val=best_val,
                                          target_val=1.503)
    print(f"[SGD] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Phase D – Exact active‑set refinement
    # ------------------------------------------------------------------
    best_a, best_val = _refine_exact_phase(best_a, n_large,
                                           start=phase_c_deadline,
                                           deadline=deadline,
                                           best_a=best_a,
                                           best_val=best_val,
                                           target_val=1.503)
    print(f"[Refine] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣ Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n_large)
    print(f"[Final mix] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8️⃣ Report final score using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
