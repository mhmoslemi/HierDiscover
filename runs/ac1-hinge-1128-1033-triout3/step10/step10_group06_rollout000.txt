"""
Optimiser for the non‑negative step‑function problem.

Approach
--------
1. **Rich parametric sweep** (≤ 30 s)  
   A broad collection of analytic families (exponential, power‑law,
   beta‑U‑shapes, arcsine, sine‑power, piecewise‑constant edges, random
   Dirichlet draws) is evaluated on a moderate length `n`.  
   The best‑found shape becomes the seed for the next phase.

2. **Exponential‑gradient descent** (≈ 80 % of the time budget)  
   For the current sequence `a` the index `k_max` where the convolution
   `a*a` attains its maximum is located.  
   The sub‑gradient of the peak is simply  
        g[i] = 2·a[k_max‑i]  (if 0≤k_max‑i<n, else 0).  
   A multiplicative update  
        a ← a·exp(‑ε·g) ,   a ← a / Σa
   moves mass away from the indices contributing heavily to the peak.
   The step size `ε` is annealed from 0.5 to 1e‑4.  Improvements are kept,
   occasional Gaussian perturbations escape plateaus, and after long
   stagnation a restart near the best solution is performed.

3. **Uniform‑mix polishing**  
   A tiny convex combination with the uniform distribution (α∈{0.001,…,0.02})
   is tried; the best among them is returned.

The optimiser respects the required signature `propose_candidate` and works
under the 1000 s time budget.  All sequences are normalised to sum = 1,
so the internal objective `2·n·max(conv)` equals the value returned by
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (full length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective for a sequence with Σ a = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _initial_candidate(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Search a wide collection of families and return the best seed."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)

    def _try(candidate):
        nonlocal best_a, best_val
        a = np.asarray(candidate, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                            # 1 … n

    # ---- 1. exponential tails (decreasing and symmetric) ----
    for r in np.linspace(0.995, 0.999999, 12):
        _try(r ** i)                      # decreasing
        _try(r ** i + r ** (n - 1 - i))   # symmetric U‑shape

    # ---- 2. power‑law tails (decreasing, symmetric) ----
    for p in np.linspace(0.1, 4.0, 20):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # ---- 3. inverse‑sqrt (special case) ----
    _try(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))

    # ---- 4. Beta U‑shapes (symmetric) ----
    beta_vals = np.linspace(0.07, 0.9, 18)
    x = (i + 0.5) / n                     # avoid exact 0/1
    for a_par in beta_vals:
        for b_par in beta_vals:
            a = (x ** (a_par - 1.0)) * ((1.0 - x) ** (b_par - 1.0))
            # clip possible infinities caused by very small params
            a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
            _try(a)

    # ---- 5. arcsine distribution (Beta(0.5, 0.5)) ----
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # ---- 6. sine‑power family ----
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 15):
        _try(base_sin ** p)

    # ---- 7. piecewise‑constant “high‑at‑edges” ----
    for k_frac in np.linspace(0.003, 0.08, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.08, 0.35, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # ---- 8. random Dirichlet draws with varied concentration ----
    rng = np.random.default_rng(123456)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for _ in range(30):
            a = rng.dirichlet(np.full(n, alpha))
            _try(a)

    # ---- 9. interpolate a previously found solution, if any ----
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    # stop early if time is running out
    if time.time() > deadline - 5.0:
        print("[Init] early stop of sweep due to time")
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – exponential‑gradient descent on the peak of the convolution
# ----------------------------------------------------------------------
def _apply_exp_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    One multiplicative update step.
    grad[i] = 2 * a[k_max - i] (zero if out of bounds).
    """
    n = a.shape[0]
    b = _full_convolution(a)
    k_max = int(np.argmax(b))
    # Compute gradient
    grad = np.zeros_like(a)
    idx = k_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    # 2*a[j] for valid positions
    grad[mask] = 2.0 * a[idx[mask]]

    # Multiplicative (exponential) update – always stays non‑negative
    a_new = a * np.exp(-eps * grad)
    # re‑normalise to the simplex
    a_new_sum = a_new.sum()
    if a_new_sum == 0.0:
        # fallback to uniform if everything collapsed (should not happen)
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new = a_new / a_new_sum
    return a_new


def _gradient_descent(a0: np.ndarray, start: float, deadline: float) -> tuple[np.ndarray, float]:
    """Iterative exponential‑gradient descent with annealed step size."""
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_val = _objective(cur_a)

    best_a = cur_a.copy()
    best_val = cur_val

    it = 0
    stagnation = 0
    # parameters controlling restarts
    RESTART_LIMIT = 250_000
    NO_IMPROVE_LIMIT = 80_000

    while time.time() < deadline:
        # progress‑dependent step size (linear anneal)
        progress = (time.time() - start) / (deadline - start)
        eps = 0.5 * (1.0 - progress) + 1e-4 * progress

        # propose a new point
        cand_a = _apply_exp_grad(cur_a, eps)
        cand_val = _objective(cand_a)

        if cand_val < cur_val - 1e-15:
            # accept improvement
            cur_a, cur_val = cand_a, cand_val
            stagnation = 0
            if cur_val < best_val - 1e-15:
                best_val, best_a = cur_val, cur_a.copy()
                print(f"[GD {it:8d}] new best = {best_val:.6f}")
        else:
            # no improvement
            stagnation += 1

        # occasional tiny random perturbation to escape flat regions
        if it % 3000 == 0 and it > 0:
            noise = np.random.normal(scale=0.015, size=n)
            pert = np.maximum(cur_a + noise, 0.0)
            pert_sum = pert.sum()
            if pert_sum > 0:
                pert = pert / pert_sum
                pert_val = _objective(pert)
                if pert_val < cur_val:
                    cur_a, cur_val = pert, pert_val
                    print(f"[GD {it:8d}] perturb improved → {cur_val:.6f}")

        # restart after long stagnation
        if stagnation > NO_IMPROVE_LIMIT:
            # restart near the best point with a small random kick
            cur_a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            cur_a = np.maximum(cur_a + noise, 0.0)
            cur_a_sum = cur_a.sum()
            if cur_a_sum > 0:
                cur_a = cur_a / cur_a_sum
                cur_val = _objective(cur_a)
                stagnation = 0
                print(f"[GD {it:8d}] restart after stagnation")
        # hard restart after huge lack of progress
        if it - RESTART_LIMIT > 0 and stagnation > RESTART_LIMIT:
            cur_a = best_a.copy()
            cur_a = _project_simplex(cur_a)  # ensure feasibility
            cur_val = _objective(cur_a)
            stagnation = 0
            print(f"[GD {it:8d}] hard restart")

        it += 1
        # cheap safeguard to leave a few seconds for polishing
        if time.time() > deadline - 4.0:
            break

    return best_a, best_val

# Phase 2 – tiny uniform mixing polish

def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Convex combination with the uniform distribution (tiny alphas)."""
    best_a = a.copy()
    best_val = _objective(a)
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = cand / cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

    Returns a list of floats whose sum ≈ 1, ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries gives a good trade‑off
    n = 6000

    # ------------------------------------------------------------------
    # Phase 0 – broad parametric sweep for a good seed
    # ------------------------------------------------------------------
    print("[Init] parametric sweep")
    best_a, best_val = _initial_candidate(n, deadline)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponential‑gradient descent refinement
    # ------------------------------------------------------------------
    print("[GD] start gradient descent")
    best_a, best_val = _gradient_descent(best_a, start_time, deadline)
    print(f"[GD] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – tiny uniform mixing polish
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided wrapper (just for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return plain Python list (the harness expects this)
    return best_a.tolist()
