def project_simplex(v):
    # v can be negative, returns projection onto simplex {x >= 0, sum=1}
    # Implementation from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that (approximately) minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1) either by
   – re‑interpolating the historic `height_sequence_1` if it exists, or
   – a cheap parametric sweep (exponential, power‑law, Hann‑window).
2. Optimise `a` on the simplex with a smoothed version of the
   max‑convolution objective.
   * For a temperature `β` we use `p_i ∝ exp(β·b_i)` where `b = a*a`.
   * The gradient of the soft‑max objective is
         ∇ = 2·(p * a_rev)[n‑1 : n‑1+n]      (convolution via FFT)
   * After a gradient step we project back onto the probability simplex.
   * `β` is increased slowly (approaching the true max), while the
     learning rate `η` decays.  Random jitter and occasional
     re‑initialisation from the current best avoid stagnation.
3. A tiny uniform mixing (α ∈ {0.001,…,0.020}) often improves the final
   bound.
4. Return the best vector seen as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – fall back to uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Renormalise to guard against numerical drift
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, size chosen once)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution of x and y using an FFT of given length."""
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    # keep only the real part (tiny imag. may appear due to numerical noise)
    return conv.real


def _full_conv_a(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using the shared FFT size."""
    Fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(Fa * Fa, n=size_fft)[: 2 * a.shape[0] - 1]
    return conv.real


# ----------------------------------------------------------------------
# Objective (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_val(a: np.ndarray, size_fft: int) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    b = _full_conv_a(a, size_fft)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Tiny uniform mixing (often reduces the max‑convolution a little)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, size_fft: int):
    """Try (1‑α)·a + α·uniform for a few α and keep the best."""
    n = a.shape[0]
    best_a = a
    best_val = _objective_val(a, size_fft)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)        # keeps sum = 1
        val = _objective_val(cand, size_fft)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main optimisation routine (soft‑max gradient descent on the simplex)
# ----------------------------------------------------------------------
def _optimize_on_simplex(
    a0: np.ndarray,
    size_fft: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the simplex using a smoothed max‑convolution.
    Returns the best vector seen and its objective value.
    """
    a = _project_simplex(a0.copy())
    n = a.shape[0]

    # schedules
    beta = 5.0            # start temperature for soft‑max
    beta_max = 200.0
    beta_factor = 1.005
    eta = 0.5             # learning rate
    eta_decay = 0.9999

    best_a = a.copy()
    best_val = _objective_val(a, size_fft)

    no_improve = 0
    IMPROVE_LIMIT = 25000
    it = 0

    while time.time() < deadline:
        # ----- current convolution & objective -----
        b = _full_conv_a(a, size_fft)            # length 2n‑1
        cur_val = float(2.0 * n * b.max())
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[opt {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ----- soft‑max probabilities -----
        shift = b.max()                          # for numerical stability
        exp_args = beta * (b - shift)            # ≤ 0
        p = np.exp(exp_args)
        s = p.sum()
        if s == 0.0:                             # extreme under‑flow -> use argmax
            p[:] = 0.0
            p[np.argmax(b)] = 1.0
            s = 1.0
        p = p / s

        # ----- gradient of the smoothed objective -----
        a_rev = a[::-1]
        # convolve p with a_rev -> length 3n‑2, we need indices n‑1 … n‑1+n‑1
        grad_full = _fft_convolve(p, a_rev, size_fft)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # shape (n,)

        # ----- gradient step + projection -----
        a = _project_simplex(a - eta * grad)

        # ----- schedules -----
        beta = min(beta * beta_factor, beta_max)
        eta *= eta_decay

        # ----- occasional jitter (helps escape shallow plateaus) -----
        if rng.random() < 0.002:
            a += rng.normal(scale=0.001, size=n)
            a = _project_simplex(a)

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            # re‑centre around current best with a moderate perturbation
            a = best_a + rng.normal(scale=0.01, size=n)
            a = _project_simplex(a)
            eta = 0.5
            beta = 5.0
            no_improve = 0
            print(f"[opt {it:7d}] perturb after stagnation")

        it += 1
        # safety break a little before deadline
        if time.time() > deadline - 0.15:
            break

    return best_a, best_val
