def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000  # or maybe 12000 to get more freedom.
    # initial a
    if 'height_sequence_1' exists:
        try:
            old = np.asarray(height_sequence_1, dtype=float)
            # Interpolate to length n
            old_len = old.shape[0]
            new_x = np.linspace(0,1,n)
            old_x = np.linspace(0,1,old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0)
            a0 /= a0.sum()
            a = a0
        except Exception:
            a = np.random.dirichlet(np.ones(n))
    else:
        a = np.random.dirichlet(np.ones(n))

    best_a = a.copy()
    best_val = float('inf')
    # compute initial objective after normalization
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()
    # convert to 2n?? Actually conv length 2n-1.
    best_val = float(2.0 * n * conv.max())
    print("[init] initial objective:", best_val)

    stagnation_counter = 0
    # set parameters
    eta_start, eta_end = 0.6, 0.001
    alpha_start, alpha_end = 0.5, 20.0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute conv of a
        conv = _full_convolution(a)
        cur_val = float(2.0 * n * conv.max())
        # update best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            stagnation_counter = 0
            print("[iter] new best", best_val, "t=", now-start)
        else:
            stagnation_counter += 1

        # slope for schedules
        prog = (now - start) / (deadline - start)
        eta = eta_start * (1 - prog) + eta_end * prog
        alpha = alpha_start * (1 - prog) + alpha_end * prog

        # smoothing
        conv_max = conv.max()
        # subtract conv_max to avoid overflow
        exp_weights = np.exp(alpha * (conv - conv_max))
        p = exp_weights / exp_weights.sum()

        # compute gradient
        a_rev = a[::-1]
        # convolution of p and a_rev
        # compute size: len(p) + len(a_rev) -1 = (2n-1)+(n)-1 = 3n-2
        # Use full convolution
        # Use FFT convolution via np.fft.rfft for speed
        size = 1 << ((len(p) + len(a_rev) - 1) - 1).bit_length()
        fp = np.fft.rfft(p, size)
        fa_rev = np.fft.rfft(a_rev, size)
        conv_pa = np.fft.irfft(fp * fa_rev, size)[:len(p)+len(a_rev)-1]
        # slice gradient part
        grad_slice = conv_pa[(n-1):(n-1+n)]
        grad = 2.0 * grad_slice

"""
Non‑negative step‑function optimiser (gradient‑descent on a smoothed max).

The algorithm:
* Initialise a probability vector a (Σ a = 1) – either the previous best,
  a Beta‑family seed or a random Dirichlet draw.
* Repeatedly:
    – compute the full self‑convolution b = a*a,
    – turn b into a soft‑max weight vector p (parameter α),
    – compute the gradient of the smoothed maximum:
          g_i = 2 * Σ_k p_k * a_{k‑i}    (indices outside [0,n) are ignored)
      which can be obtained by a single FFT convolution of p with the
      reversed a.
    – normalise g (divide by its maximum) and perform a multiplicative
      (entropy‑mirror) update   a ← a·exp(−η·g)   followed by renormalisation.
    – adapt η (stepsize) from a large value to a tiny one and increase α
      (soft‑max sharpness) during the run.
    – keep the best vector seen so far; if progress stalls,
      apply a small Gaussian perturbation around the current best.
* After the time budget, evaluate the best vector with the provided
  evaluate_sequence() and return it as a list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two 1‑D real arrays via FFT."""
    out_len = len(x) + len(y) - 1
    size = 1 << (out_len - 1).bit_length()         # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2·len(a)‑1)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – safeguard for numerical drift
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Beta‑family generator (shape p,q) – used for cheap initial seeds
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Return a normalised Beta‑shaped sequence of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a

    """

    start_time = time.time()
    # leave a margin for the final call to evaluate_sequence()
    deadline = start_time + budget_s - 0.5

    # Problem size
    # --------------------------------------------------------------

            best_a = a0.copy()
