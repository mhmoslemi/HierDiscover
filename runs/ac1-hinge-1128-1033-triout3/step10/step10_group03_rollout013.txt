"""
Annealed soft‑max + Adam optimizer for flattening the self‑convolution.

The idea is to treat the objective  max_k (a∗a)_k  as a limit of the
smooth “log‑sum‑exp” function.  For a given temperature τ,
    Lτ(a) = τ·log∑_k exp((a∗a)_k / τ)
has the same minimisers as the original problem, but its gradient is
well‑behavioured:

    ∂Lτ/∂a_i = 2·∑_j w_{i+j}·a_j ,   w_k = softmax((a∗a)_k / τ).

We anneal τ from a sizeable fraction of the current maximum down to a
tiny value, so the optimisation first smooths the objective (acting like
a “fairness” term) and then focuses on the true maximum.
Updates are performed with the Adam optimiser and after each step the
vector is projected back onto the probability simplex (non‑negative,
unit‑sum).  Several short restarts are performed within the 1000 s
budget, the best sequence found is finally evaluated with
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_self_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full convolution a*a using FFT of given padded size."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


def _conv_rev_w_fft(a_rev: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """Full convolution a_rev * w using FFT of given padded size."""
    fa_rev = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa_rev * fw, size)[: a_rev.shape[0] + w.shape[0] - 1]
    return conv


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Clamp to non‑negative and renormalise to sum 1 (simple simplex)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _objective_max(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) (sum of a is 1 by construction)."""
    n = a.shape[0]
    b = _conv_self_fft(a, 1 << ((2 * n - 2).bit_length()))
    return float(2.0 * n * b.max())

        n = 8000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective_max(best_seq)
        print("[init] uniform start – score =", best_score)

    # --------------------------------------------------------------
    # 1️⃣  Small pool of random starts – keep the best as a warm start
    # --------------------------------------------------------------
    for _ in range(4):
        cand = np.random.dirichlet(np.ones(n))
        val = _objective_max(cand)
        if val < best_score:
            best_seq = cand.copy()
            best_score = val
            print("[rand init] improved start – score =", best_score)

    # --------------------------------------------------------------
    # 2️⃣  Annealed soft‑max Adam optimisation (multiple short restarts)
    # --------------------------------------------------------------
    # pre‑compute FFT sizes (constants for the whole run)
    size_self = 1 << ((2 * n - 2).bit_length())          # for a*a
    size_grad = 1 << ((3 * n - 2).bit_length())          # for a_rev * w

    # configuration
    total_rest = 3                     # number of restarts
    lr_start, lr_end = 0.5, 0.01       # Adam learning‑rate schedule
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    jitter_prob = 0.001
    jitter_scale = 1e-3

    # remaining time split roughly equally among the restarts
    remaining_budget = deadline - time.time()
    time_per_restart = remaining_budget / (total_rest + 0.5)

    for r in range(total_rest):
        # check global time budget
        if time.time() >= deadline:
            break

        # initialise this restart
        if r == 0:
            a = best_seq.copy()
        else:
            a = np.random.dirichlet(np.ones(n))

        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it = 0
        restart_start = time.time()

        while True:
            now = time.time()
            if now >= deadline:
                break
            if now - restart_start > time_per_restart:
                # time allocated for this restart exhausted
                break

            # -----------------------------------------------------------------
            # compute b = a*a and the current maximum
            # -----------------------------------------------------------------
            b = _conv_self_fft(a, size_self)                # length 2n‑1
            max_b = b.max()

            # -------------------------------------------------------------
            # annealing schedule for τ (fraction of the current max)
            # -------------------------------------------------------------
            # progress ∈ [0,1] for the current restart
            prob = (now - restart_start) / time_per_restart
            alpha = max(0.5 * (1.0 - prob), 1e-3)           # start at 0.5·max, end ≈0.001·max
            tau = max_b * alpha + 1e-12                     # avoid exact zero

            # -------------------------------------------------------------
            # soft‑max weights over the convolution entries
            # -------------------------------------------------------------
            shifted = b - max_b                              # ≤ 0
            exp_scaled = np.exp(shifted / tau)              # stable (max exponent = 0)
            Z = exp_scaled.sum()
            w = exp_scaled / Z                               # length 2n‑1, sums to 1

            # -------------------------------------------------------------
            # gradient of the smoothed objective:
            #   ∂Lτ/∂a_i = 2·∑_j w_{i+j}·a_j
            # obtained via a single convolution a_rev * w
            # -------------------------------------------------------------
            a_rev = a[::-1]
            conv_rev = _conv_rev_w_fft(a_rev, w, size_grad)   # length 3n‑2
            grad = 2.0 * conv_rev[n - 1: n - 1 + n]            # length n

            # -------------------------------------------------------------
            # Adam update (bias‑corrected)
            # -------------------------------------------------------------
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # learning‑rate schedule (linear decay)
            lr = lr_start * (1.0 - prob) + lr_end * prob
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_candidate = a - step

            # simple simplex projection (clamp & renormalise)
            a_candidate = _simplex_projection(a_candidate)

            # occasional jitter to escape plateaus
            if np.random.rand() < jitter_prob:
                a_candidate += np.random.normal(scale=jitter_scale, size=n)
                a_candidate = _simplex_projection(a_candidate)

            a = a_candidate

            # -----------------------------------------------------------------
            # evaluate true objective (no smoothing) and store the best ever
            # -----------------------------------------------------------------
            cur_score = 2.0 * n * max_b
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[optim] new best {best_score:.6f} (restart {r}, iter {it})")
                if best_score <= 1.503:
                    break

        print(f"[restart {r}] finished – elapsed {time.time() - restart_start:.1f}s, "
              f"best score so far {best_score:.6f}")
