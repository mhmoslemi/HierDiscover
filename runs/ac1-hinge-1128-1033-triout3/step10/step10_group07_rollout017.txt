import time
import numpy as np

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _full_convolution(a):
    n = a.shape[0]
    size = _next_pow2(2*n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    return conv

def _objective(a):
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))

def _smooth_eg(a0, n, deadline, rng):
    a = a0.copy()
    eps = 1e-12
    log_w = np.log(a + eps)
    eta = 0.5
    decay_eta = 0.9999
    tau = 0.05
    decay_tau = 0.9995
    tau_min = 0.0005
    size_fft = 1 << ((3*n - 2).bit_length())
    best_a = a.copy()
    best_obj = _objective(a)
    no_improve = 0
    IMP_LIMIT = 30000
    iteration = 0
    while time.time() < deadline:
        # compute FFT of a
        Fa = np.fft.rfft(a, size_fft)
        # convolution for b
        conv_full = np.fft.irfft(Fa * Fa, size_fft)[:2*n-1]
        max_b = conv_full.max()
        # update best
        cur_obj = 2.0 * n * max_b / (a.sum()**2)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            print(f"[EG {iteration:7d}] new best {best_obj:.6f}, tau {tau:.4f}")
        else:
            no_improve += 1

        # softmax weights from conv
        # use max_b for numerical stability
        w = np.exp((conv_full - max_b) / tau)
        w_sum = w.sum()
        w /= w_sum

        # compute correlation gradient
        Fw = np.fft.rfft(w, size_fft)
        corr = np.fft.irfft(np.conj(Fa) * Fw, size_fft)
        start = 2*n - 2
        grad = 2.0 * corr[start:start + n]

        # EG update on log weights
        log_w -= eta * grad
        # small jitter
        if rng.random() < 0.001:
            log_w += rng.normal(scale=0.001, size=n)

        # project back onto simplex via exponentiation
        max_log = np.max(log_w)
        w_exp = np.exp(log_w - max_log)
        a = w_exp / w_exp.sum()

        # decay step size and temperature
        eta *= decay_eta
        tau = max(tau * decay_tau, tau_min)

        # stagnation restart
        if no_improve > IMP_LIMIT:

            eta = 0.5

            no_improve = 0
            print(f"[EG {iteration:7d}] stagnation restart")

        iteration += 1
        if time.time() > deadline - 0.2:

def _local_refine(a, n, best_obj, rng, max_iter=5000):
    best = a.copy()
    best_val = best_obj
    for _ in range(max_iter):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose a small delta
        delta = rng.normal(scale=0.001) * best.mean()
        # ensure positivity
        if delta > 0:
            delta = min(delta, best[j])
        else:
            delta = -min(-delta, best[i])
        if abs(delta) < 1e-12:
            continue
        a_new = best.copy()
        a_new[i] += delta
        a_new[j] -= delta
        # compute obj
        cur = _objective(a_new)
        if cur < best_val:
            best_val = cur
            best = a_new
    return best, best_val

"""
search for a non‑negative step function that minimises
        evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages

1. **Initialisation** – reuse a known good construction if available,
   otherwise start from a random Dirichlet draw or from a few hand‑crafted
   parametric families (exponential, power‑law, cosine window).

2. **Smoothed exponentiated‑gradient descent** – we repeatedly
   compute the full self‑convolution `b = a*a`.  Instead of the
   nonsmooth max we minimise a smooth approximation
   `Mτ(a) = τ·log(∑exp(b_i/τ))`.  The gradient of `Mτ` w.r.t. `a`
   is `2·(a ⋆ w)`, where `w` are the soft‑max weights
   `w_i = exp(b_i/τ)/∑exp(b_j/τ)` and `⋆` denotes the (non‑reversed)
   cross‑correlation.  The gradient is obtained cheap by a single FFT.
   Updates are performed on log‑weights (mirror descent on the simplex)
   with an exponentially decaying step size and a temperature τ that is
   slowly annealed.

3. **Post‑processing** – a tiny mixing with the uniform distribution and
   a cheap random‑pairwise local search are applied to polish the best
   candidate.

The whole routine respects the 1000 s budget, works on a single CPU
and uses only NumPy (no external I/O).  It improves over the previous
EG implementation by using a smooth surrogate of the max‑operator and
by a more aggressive annealing schedule, which helps to escape the
local minima that limited the earlier best score of ≈1.5088.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple uniform mixing (tiny α)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for several α and keep the best."""
    best_a, best_val = a, _objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric candidate generator (used when we have no previous solution)
# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, rng: np.random.Generator):
    """Yield a few simple families of vectors (not normalised)."""
    i = np.arange(n, dtype=np.float64)

    # 1. Monotone exponential decay a_i ∝ r**i  (no symmetry)
    for r in np.linspace(0.90, 0.9999, 10):
        a = np.power(r, i)
        yield a

    # 2. Symmetric exponential (the one previously used)
    for r in np.linspace(0.55, 0.9999, 10):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 3. Power‑law decay a_i ∝ (i+1)^‑p
    for p in np.linspace(0.5, 4.0, 10):
        a = 1.0 / np.power(i + 1.0, p)
        yield a

    # 4. Cosine‑window raised to a power β
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))          # Hann window
    for beta in np.linspace(0.5, 4.0, 8):
        a = np.power(w, beta)
        yield a

    # 5. A few random Dirichlet draws
    for _ in range(15):
        a = rng.dirichlet(np.ones(n))
        yield a


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_eg(
    a0: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using a soft‑max
    approximation of the max‑convolution.  Returns the best vector seen.
    """
    eps = 1e-12
    log_w = np.log(a0 + eps)          # initialise log‑weights
    eta = 0.5                         # initial step size
    decay_eta = 0.9999                # geometric decay
    tau = 0.05                        # initial temperature for soft‑max
    decay_tau = 0.9995
    tau_min = 5e-4

    size_fft = _next_pow2(3 * n - 2)   # enough for both conv & corr

    best_a = a0.copy()
    best_val = _objective(best_a)

    no_improve = 0
    IMP_LIMIT = 25000
    it = 0

    while time.time() < deadline:
        # ---- convolution of a with itself ----
        Fa = np.fft.rfft(a0, size_fft)
        conv_full = np.fft.irfft(Fa * Fa, size_fft)[: 2 * n - 1]

        # ---- current objective (exact) ----
        cur_val = _objective(a0)
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a0.copy()
            no_improve = 0
            print(f"[EG {it:7d}] new best = {best_val:.6f} (τ={tau:.5f})")
        else:
            no_improve += 1

        # ---- soft‑max weights for the smooth max ----
        max_b = conv_full.max()
        w_s = np.exp((conv_full - max_b) / tau)    # stable version
        w_s /= w_s.sum()                           # ∑w = 1

        # ---- gradient = 2·(a ⋆ w_s)  (cross‑correlation) ----
        Fw = np.fft.rfft(w_s, size_fft)
        corr = np.fft.irfft(np.conj(Fa) * Fw, size_fft)
        start = 2 * n - 2          # index where lag 0 starts
        grad = 2.0 * corr[start : start + n]

        # ---- mirror‑descent update on log‑weights ----
        log_w -= eta * grad

        # occasional jitter to leave shallow plateaus
        if rng.random() < 0.001:
            log_w += rng.normal(scale=0.001, size=n)

        # projection back onto the simplex (by exponentiation & renorm)
        max_log = np.max(log_w)
        w_exp = np.exp(log_w - max_log)
        a0 = w_exp / w_exp.sum()

        # ---- schedule updates ----
        eta *= decay_eta
        tau = max(tau * decay_tau, tau_min)

        # ---- stagnation handling ----
        if no_improve > IMP_LIMIT:
            log_w = np.log(best_a + eps)
            log_w += rng.normal(scale=0.02, size=n)
            eta = 0.5
            tau = max(tau, 0.01)
            no_improve = 0
            print(f"[EG {it:7d}] restart after stagnation")

        it += 1
        if time.time() > deadline - 0.2:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap local refinement (random pairwise transfers)
# ----------------------------------------------------------------------
def _local_refine(a: np.ndarray, n: int, rng: np.random.Generator,
                  max_iter: int = 6000) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass between random positions; keep improvements."""
    best = a.copy()
    best_val = _objective(best)

    for _ in range(max_iter):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a small transfer δ (drawn from a narrow Gaussian)
        delta = rng.normal(scale=0.002) * best.mean()
        if delta > 0:
            delta = min(delta, best[j])
        else:
            delta = -min(-delta, best[i])
        if abs(delta) < 1e-12:
            continue
        cand = best.copy()
        cand[i] += delta
        cand[j] -= delta
        # keep non‑negativity (tiny numerical drift is tolerated)
        if cand[i] < 0 or cand[j] < 0:
            continue
        cur = _objective(cand)
        if cur < best_val:
            best_val, best = cur, cand
    return best, best_val
