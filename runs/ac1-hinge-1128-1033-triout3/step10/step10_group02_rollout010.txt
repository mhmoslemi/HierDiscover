def _beta_shape(p, q, n):
    i = np.arange(1, n+1, dtype=np.float64) # 1-indexed to avoid zero.
    a = (i ** (p-1.0)) * ((n - i + 1) ** (q-1.0))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a_sum
    return a

def _adam_optimize_mod(a0, n, deadline, start_time, best_a, best_val):
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.9, 0.999
    lr_start, lr_end = 0.2, 0.001
    eps_start, eps_end = 0.5, 1e-7

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 50000  # avoid restart

    while time.time() < deadline:
        # decode a
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a_sum = a.sum()
        a = a / a_sum

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        _, grad = _smooth_max_and_grad(a, eps)

        # gradient through softmax
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        t += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w * grad_w)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.01, size=n)

        # do not restart
    return best_a, best_val

def _hillclimb_mod(a0, n, deadline, start_time, best_a, best_val):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.6, 0.001
    temp_start, temp_end = 0.05, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 80000
    iteration = 0

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # find index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source selection
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target selection bias towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n-1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_new = b.copy()
        # Adding contribution from j to all positions where a interacts
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # Quadratic term updates
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] stagnation, perturb at iteration {iteration}")

        iteration += 1

    return best_a, best_val

def _subgradient_polish_mod(a0, n, deadline, start_time, best_a, best_val, max_iters=500_000):
    a = a0.copy()
    cur_val = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        b = _full_convolution(a)
        max_b = b.max()
        # indices where max occurs
        idxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in idxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step size diminishing as 1/sqrt(iter)
        step = 0.5 / np.sqrt(it + 1)  # starting step 0.5, decays
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        if np.random.rand() < 0.001:
            a = np.maximum(a + np.random.normal(scale=0.01, size=n), 0.0)
            a = _project_simplex(a)
            cur_val = _objective(a)

        it += 1

"""
Propose a high‑quality step‑function (non‑negative vector a, Σa = 1) that
minimises the security metric

    evaluate_sequence(a) = 2·n·max(conv(a,a))

where conv(a,a) is the full linear convolution of a with itself.
The algorithm combines:

1.   A quick search over parametric Beta‑shapes (a_i ∝ i^{p‑1}(n‑i+1)^{q‑1})
     together with random Dirichlet and Beta draws.
2.   A smooth‑max Adam optimisation (log‑sum‑exp approximation of the
     max) – no restarts and a long, gently decaying learning‑rate schedule.
3.   An extended simulated‑annealing hill‑climber that moves a tunable
     fraction of mass between indices, with a slower temperature decay.
4.   A projected sub‑gradient polishing phase with a 1/√t step‑size.
5.   Final blending with the uniform distribution.

The whole routine respects the 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv(a,a)) and gradient of 2·n·smooth_max
    w.r.t. a.  Uses the log‑sum‑exp trick.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # soft‑max weights

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # includes factor 2·n
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric Beta shape (deterministic)
# ----------------------------------------------------------------------
def _beta_shape(p: float, q: float, n: int) -> np.ndarray:
    """Return the normalised Beta‑like shape a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)   # avoid the zero‑index singularity
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Random Beta family candidate (already used in the original code)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ln(1e‑4) .. ln(10)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max (no restarts)
# ----------------------------------------------------------------------
def _adam_optimize_mod(a0: np.ndarray, n: int,
                       deadline: float, start_time: float,
                       best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex for the loss L(a)=2·n·smooth_max(a)."""
    w = np.log(a0 + 1e-12)                # log‑softmax parameters

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-7

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 50000                      # keep moving, avoid restarts

    while time.time() < deadline:
        # decode logits to a probability vector
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            idle += 1

        # scheduling
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress

        # smooth‑max and its gradient
        _, grad = _smooth_max_and_grad(a, eps)   # already contains 2·n factor

        # chain rule through the soft‑max
        dot = np.dot(grad, a)
        grad_w = a * (grad - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to keep exploration alive
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.01, size=n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (fast O(n) updates)
# ----------------------------------------------------------------------
def _hillclimb_mod(a0: np.ndarray, n: int,
                   deadline: float, start_time: float,
                   best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """
    Extended hill‑climber.  Moves a (shrinking) fraction of mass from a
    “high‑contribution’’ index towards a “low‑contribution’’ index,
    with a slower temperature decay.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules (more aggressive at the start)
    max_frac_start, max_frac_end = 0.6, 0.001
    temp_start, temp_end = 0.05, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 80000
    iteration = 0

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index where the convolution attains its current maximum
        k_max = int(np.argmax(b))

        # contribution of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index – proportional to its contribution to the max entry
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – favour low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # Stagnation handling – jump back to the current best with a
        # small random perturbation.
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] stagnation – perturb (it={iteration})")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish_mod(a0: np.ndarray, n: int,
                           deadline: float, start_time: float,
                           best_a: np.ndarray, best_val: float,
                           max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Uses a diminishing step size ~ 1/√t for fine control.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        idxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in idxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step size ∝ 1/√(it+1)
        step = 0.4 / np.sqrt(it + 1.0)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional tiny jitter to escape flat plateaux
        if np.random.rand() < 0.001:
            a = np.maximum(a + np.random.normal(scale=0.01, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        n: int,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for several t‑values."""
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, 21):
        cand = t * a + (1.0 - t) * u
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget and
    return a list suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the final call

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise (previous solution, systematic Beta, random draws)
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf

    # 0.1 – reuse a previously discovered solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[Init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # 0.2 – a few deterministic symmetric Beta shapes (p = q)
    for p in (0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        a = _beta_shape(p, p, n)
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[Init] Beta(p={p:.2f}) value={best_val:.6f}")

    # 0.3 – random Beta draws
    for _ in range(40):
        a = _random_beta_candidate(n, rng)
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val
            print(f"[Init] random Beta improved value={best_val:.6f}")

    # 0.4 – Dirichlet draws with varied concentration
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            a = rng.dirichlet(np.full(n, alpha))
            val = _objective(a)
            if val < best_val:
                best_a, best_val = a, val
                print(f"[Init] Dirichlet(alpha={alpha}) value={best_val:.6f}")

    # fallback to uniform if nothing succeeded
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print(f"[Init] fallback uniform, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * budget_s
    best_a, best_val = _adam_optimize_mod(
        best_a, n,
        deadline=phase_a_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase B – Hill‑climbing (≈60 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _hillclimb_mod(
        best_a, n,
        deadline=phase_b_deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Phase C – Sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish_mod(
        best_a, n,
        deadline=deadline,
        start_time=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
    )
    best_a, best_val = _blend_with_uniform(best_a, n, best_val)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    # ensure the output is a plain list of floats
    return best_a.tolist()
