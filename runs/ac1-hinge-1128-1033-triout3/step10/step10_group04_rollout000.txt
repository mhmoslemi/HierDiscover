"""
propose_candidate
-----------------

A time‑bounded optimiser for the “max‑convolution” bound.

Strategy
--------
1. **Large problem size** – use a longer step‑function (≈15000 points)
   which gives a much smaller theoretical lower bound (≈1.0).

2. **Rich parametric sweep** – create a diverse pool of shapes
   (exponential, power‑law, beta‑like, sine‑power, piecewise‑constant,
   logistic, random Dirichlet) and keep the one with the smallest
   exact objective.

3. **Targeted hill‑climbing with annealing** – at each iteration
   locate the convolution entry that attains the current peak,
   compute for every index its *partner* contribution to that peak,
   and greedily transfer a fraction of mass from a high‑contributing
   index to a low‑contributing one.  The fraction is annealed from
   0.5→0.01 while a simple Metropolis acceptance rule occasionally
   permits uphill moves, helping to escape local minima.

4. **Stagnation handling** – if no improvement is seen for a while,
   inject a small Gaussian perturbation around the best solution
   discovered so far.

5. **Final uniform mixing** – a tiny convex combination with the
   uniform distribution (α∈{0.001,…,0.020}) often smooths residual spikes.

The whole routine respects the 1000 s budget, prints progress and
returns a list of non‑negative heights (∑ a = 1) ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())   # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def objective(a: np.ndarray) -> float:
    """
    Exact objective for a vector with Σa = 1:
        2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    max_b = full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidate(n: int, rng: np.random.Generator):
    """Search a large set of parametric families and keep the best."""
    best_a = np.full(n, 1.0 / n)
    best_val = objective(best_a)

    def try_candidate(arr):
        nonlocal best_a, best_val
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a /= s
        val = objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential tails (symmetrised)
    for r in np.linspace(0.995, 0.99999, 12):
        try_candidate(r ** i)                         # monotone
        try_candidate(r ** i + r ** (n - 1 - i))      # symmetric

    # 2. Power‑law tails
    for p in np.linspace(0.2, 8.0, 16):
        try_candidate(i1 ** (-p))
        try_candidate(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.2, 6.0, 6):
        for b_param in np.linspace(0.2, 6.0, 6):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            try_candidate(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 10):
        try_candidate(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.005, 0.12, 10):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.35, 8):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            try_candidate(a)

    # 6. Logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.003, 0.04, 5):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        try_candidate(a)
        try_candidate(np.flip(a))

    # 7. Random Dirichlet draws
    for _ in range(60):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main optimisation loop – targeted hill‑climbing with annealing
# ----------------------------------------------------------------------
def hillclimb_anneal(a0: np.ndarray, n: int,
                     start: float, deadline: float,
                     best_a: np.ndarray, best_val: float,
                     target_val: float = 1.503) -> np.ndarray:
    """
    Greedy mass transfer guided by the current convolution peak.
    The fraction of moved mass is annealed; occasional uphill moves
    are allowed with a simple temperature schedule.
    """
    a = a0.copy()
    conv = full_convolution(a)
    cur_obj = 2.0 * n * conv.max()
    best_local_a = best_a.copy()
    best_local_val = best_val

    # schedules
    max_frac_start, max_frac_end = 0.5, 0.005
    temp_start, temp_end = 0.02, 1e-6

    # stagnation control
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    rng = np.random.default_rng(12345)

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        # annealed parameters
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        temperature = temp_start * (1.0 - progress) + temp_end * progress

        # locate the current maximal convolution entry
        k_max = int(conv.argmax())

        # partner index for each position w.r.t. the active entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        # contributions = a[i] * a[partner]
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        # partner values (used as a weight for choosing a target)
        partner = np.zeros_like(a)
        partner[mask] = a[idx[mask]]

        # source = index with largest contribution (if all zero fall back to max a)
        if contrib.max() > 0:
            s = int(np.argmax(contrib))
        else:
            s = int(np.argmax(a))

        # target = index with smallest partner weight
        t = int(np.argmin(partner))
        if t == s:
            # choose the next smallest
            tmp = partner.copy()
            tmp[s] = np.inf
            t = int(np.argmin(tmp))

        if a[s] <= 1e-16:
            # nothing to move – random perturbation
            s = int(np.argmax(a))
            t = int(np.argmin(a))

        # draw a move size
        delta = a[s] * rng.random() * max_frac
        if delta <= 0:
            continue

        # ---- incremental update of the full convolution ----
        conv_new = conv.copy()
        # shift of a by t (adds 2*delta*a)
        conv_new[t:t + n] += 2.0 * delta * a
        # shift of a by s (subtracts 2*delta*a)
        conv_new[s:s + n] -= 2.0 * delta * a
        # pointwise terms from the moved mass
        dsq = delta * delta
        conv_new[2 * t] += dsq
        conv_new[2 * s] += dsq
        conv_new[t + s] -= 2.0 * dsq

        new_obj = 2.0 * n * conv_new.max()

        # accept if improvement or with Metropolis probability
        accept = False
        if new_obj < cur_obj - 1e-15:
            accept = True
        else:
            # uphill acceptance probability
            prob = np.exp(-(new_obj - cur_obj) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # perform the move
            a[s] -= delta
            a[t] += delta
            conv = conv_new
            cur_obj = new_obj

            if cur_obj < best_local_val - 1e-12:
                best_local_val, best_local_a = cur_obj, a.copy()
                no_improve = 0
                print(f"[HillClimb] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
            else:
                no_improve = 0
        else:
            no_improve += 1

        # stagnation handling – inject small Gaussian noise
        if no_improve > IMPROVE_LIMIT:
            # perturb the best solution found so far
            a = best_local_a.copy()
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            conv = full_convolution(a)
            cur_obj = 2.0 * n * conv.max()
            no_improve = 0
            print(f"[HillClimb] perturbing after stagnation (best={best_local_val:.6f})")

    return best_local_a


# ----------------------------------------------------------------------
# Final polishing – tiny uniform mixing
# ----------------------------------------------------------------------
def mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a small share of the uniform distribution."""
    best_a, best_val = a, objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point – used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the supplied time budget.
    Returns a list of floats (∑ = 1) that (hopefully) beats the current record.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # problem size – a few‑ten‑thousands gives a good trade‑off
    n = 15000

    rng = np.random.default_rng(seed + 999)

    # ------------------------------------------------------------------
    # 1️⃣ Initial parametric sweep
    # ------------------------------------------------------------------
    print("[Init] scanning parametric families …")
    best_a, best_val = _initial_candidate(n, rng)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Targeted hill‑climbing with annealing (major improvement phase)
    # ------------------------------------------------------------------
    print("[HillClimb] starting annealed greedy optimisation …")
    best_a = hillclimb_anneal(
        best_a, n,
        start=start_time,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,

    )

    # ------------------------------------------------------------------

    return best_a.tolist()
