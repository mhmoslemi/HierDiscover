        contrib[mask] = a[mask] * a[idx[mask]]
        # source i
        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break
        # candidate j list sorted by ascending contribution
        order = np.argsort(contrib)
        j = None
        for cand in order:
            if cand != i:
                # ensure improvement condition
                a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
                a_k_cand = a[k_max - cand] if 0 <= k_max - cand < n else 0.0
                if a_k_i > a_k_cand + 1e-12:
                    j = int(cand)
                    break
        if j is None:
            # if none found, try random j
            j_candidates = np.where(a > 0)[0]
            if len(j_candidates) <= 1:
                break
            j = int(rng.choice(j_candidates[j_candidates != i]))
        # compute second max after moving
        # get second max of b (excluding current max)
        # naive:
        second_max = np.max(np.concatenate([b[:k_max], b[k_max+1:]]))
        a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
        denom = 2.0 * (a_k_i - a_k_j)
        if denom <= 0:
            continue
        delta_needed = (b[k_max] - second_max) / denom
        # bound delta <= a[i]
        delta = min(delta_needed * 0.7, a[i])  # safety factor less than 1
        if delta <= 1e-15:
            continue
        # perform incremental update
        # update b for addition from j
        L = b.shape[0]
        # add from j shift
        end_j = min(j + n, L)
        len_j = end_j - j
        b[j:end_j] += 2.0 * delta * a[:len_j]
        # subtract from i shift
        end_i = min(i + n, L)
        len_i = end_i - i
        b[i:end_i] -= 2.0 * delta * a[:len_i]
        # self terms
        if 2*j < L:
            b[2*j] += delta * delta
        if 2*i < L:
            b[2*i] += delta * delta
        ij = i + j
        if ij < L:
            b[ij] -= 2.0 * delta * delta
        # update a
        a[i] -= delta
        a[j] += delta
        # compute new objective
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve_counter = 0
            print(f"[Iter {iter:6d}] new best = {best_val:.6f}")
        else:
            no_improve_counter += 1
        # stagnation handling: if no_improve_counter large, add small noise
        if no_improve_counter > 20000:
            # add noise to escape
            noise = rng.normal(scale=1e-4, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve_counter = 0
            print(f"[Iter {iter:6d}] perturb after stagnation")
        iter += 1
    # after loop, final mixing with uniform
    uniform = np.full(n, 1.0 / n)
    best_a2 = best_a
    best_val2 = best_val
    for alpha in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand = (1 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val2:
            best_val2 = val
            best_a2 = cand
            print(f"[Mix] alpha {alpha:.3f} improved to {best_val2:.6f}")
    # final evaluate using evaluate_sequence
    try:
        final_score = evaluate_sequence(best_a2.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    return best_a2.tolist()

"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a, a)) / (sum(a)**2)

using a simple “move‑mass from the worst to the best” greedy scheme.
The algorithm works on a moderate size (default n≈3000) so that many
cheap iterative updates can be performed in the 1000 s budget.

Core ideas
----------
* Start from the uniform distribution (sum = 1).
* Repeatedly locate the convolution entry with the largest value.
  Its value is a sum of products a[i]·a[k−i].  Pick the index i that
  contributes most to that entry and move a small amount of mass from i
  to an index j that contributes the least.
* The move is analytically chosen so that the current maximal
  convolution entry is reduced to (just below) the second largest entry.
* When progress stalls a tiny random perturbation is added and the
  process continues.
* Finally a tiny amount of uniform mass is mixed in – this often yields
  a marginal improvement.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, exact for our real vectors)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) is 1 throughout the algorithm)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector (sum=1)."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Greedy reduction of the maximal convolution entry
# ----------------------------------------------------------------------
def _greedy_reduce(a: np.ndarray, deadline: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a little mass from the index that contributes most
    to the current maximal convolution entry to the index that contributes
    the least.  The move is computed analytically so that the maximal entry
    drops towards the second‑largest entry.  Stops when time runs out or
    the target value is reached.
    """
    n = a.shape[0]
    b = _full_convolution(a)                # current full convolution
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng(12345)
    it = 0
    idle = 0
    IDLE_LIMIT = 20000                       # after many idle steps we jitter

    while time.time() < deadline and best_val > target_val:
        # --------------------------------------------------------------
        # locate the worst (largest) convolution entry
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))

        # contributions of each position i to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n, dtype=a.dtype)
        contrib[mask] = a[mask] * a[idx[mask]]

        # source i = index with largest contribution
        i = int(np.argmax(contrib))
        if a[i] <= 0.0:                     # nothing left to move
            break

        # target j = index with smallest contribution (different from i)
        # if several have the same tiny contribution we simply pick the first.
        order = np.argsort(contrib)
        j = None
        for cand in order:
            if cand != i:
                # we also want a reduction of the selected entry,
                # i.e. a[k_max‑i] should be larger than a[k_max‑cand].
                a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
                a_k_c = a[k_max - cand] if 0 <= k_max - cand < n else 0.0
                if a_k_i > a_k_c + 1e-12:
                    j = int(cand)
                    break
        if j is None:                       # fallback: any different index
            cand = (i + 1) % n
            j = int(cand)

        # --------------------------------------------------------------
        # compute the analytically optimal delta (with a safety factor)
        # --------------------------------------------------------------
        # second largest convolution value (excluding k_max)
        # (simple O(n) scan – cheap compared to the rest)
        second_max = np.max(np.concatenate([b[:k_max], b[k_max + 1:]]))

        a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
        denom = 2.0 * (a_k_i - a_k_j)
        if denom <= 0.0:
            # cannot improve with this pair – try the next iteration
            idle += 1
            it += 1
            continue

        # delta needed to make b[k_max] equal to second_max
        delta_needed = (b[k_max] - second_max) / denom
        # safety factor 0.7 keeps us away from overshooting
        delta = min(delta_needed * 0.7, a[i])

        if delta <= 1e-15:
            idle += 1
            it += 1
            continue

        # --------------------------------------------------------------
        # apply the incremental update to the convolution vector b
        # (the formulas are derived from (a‑δe_i+δe_j)*(a‑δe_i+δe_j))
        # --------------------------------------------------------------
        L = b.shape[0]

        # contribution from the new mass at position j
        end_j = min(j + n, L)
        len_j = end_j - j
        b[j:end_j] += 2.0 * delta * a[:len_j]

        # removal of the mass from position i
        end_i = min(i + n, L)
        len_i = end_i - i
        b[i:end_i] -= 2.0 * delta * a[:len_i]

        # self‑pair adjustments
        if 2 * j < L:
            b[2 * j] += delta * delta
        if 2 * i < L:
            b[2 * i] += delta * delta
        ij = i + j
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        # ------------------------------------------------------------------
        # update the probability vector
        # ------------------------------------------------------------------
        a[i] -= delta
        a[j] += delta

        # ------------------------------------------------------------------
        # evaluate new objective
        # ------------------------------------------------------------------
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[greedy {it:7d}] new best = {best_val:.6f}")
        else:
            idle += 1

        # ------------------------------------------------------------------
        # occasional perturbation if we have been idle for a while
        # ------------------------------------------------------------------
        if idle > IDLE_LIMIT:
            noise = rng.normal(scale=1e-4, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            idle = 0
            print(f"[greedy {it:7d}] perturb after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Mixing a tiny amount of uniform mass (helps a little)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Convex combination with a small uniform component."""
    n = a.shape[0]
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_a, best_val = cand, val
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a non‑negative step function under the 1000 s budget.
    Returns a list of floats (the heights) that (approximately) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size – 3000 gives a good trade‑off between expressiveness
    # and speed of the greedy loop.
    # ------------------------------------------------------------------
    n = 3000

    # ------------------------------------------------------------------
    # initialise: uniform vector perturbed by tiny noise
    # ------------------------------------------------------------------
    a0 = np.full(n, 1.0 / n, dtype=np.float64)
    a0 += rng.normal(scale=5e-5, size=n)        # break exact symmetry
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    # ------------------------------------------------------------------
    # greedy mass‑transfer phase (dominates the time budget)
    # ------------------------------------------------------------------
    best_a, best_val = _greedy_reduce(a0, deadline)

    # ------------------------------------------------------------------
    # final tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a)

    # ------------------------------------------------------------------
    # final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
