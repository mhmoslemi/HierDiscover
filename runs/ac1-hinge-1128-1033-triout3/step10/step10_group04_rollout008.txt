"""
search_step_function_v2.py

A time‑bounded optimiser for the “max‑convolution” bound.
Improvements over the original pipeline:

*   Added a richer family of parametric seeds (including a semi‑circular shape).
*   Performed several short Adam restarts with random Dirichlet starts.
*   Added a lightweight “random local search” that repeatedly transfers a
    small amount of mass from the most damaging index to a random one,
    accepting only improving moves.
*   Followed by an aggressive post‑processing stage:
    – convex mixing with uniform and with the reversed vector,
    – Gaussian smoothing with a schedule of decreasing σ,
    – additional mixing with the smoothed variants.
*   All phases respect the 1000 s budget, printing intermediate best values.

The returned sequence is non‑negative, sums to 1 and can be fed directly to

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families for an initial seed

def _initial_candidate(n: int):
    """Return the best vector among many hand‑crafted families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential tails
    for r in np.linspace(0.995, 0.99999, 14):
        _try(r ** i)                     # decreasing
        _try(r ** i + r ** (n - 1 - i))  # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 6.0, 18):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_p in np.linspace(0.2, 6.0, 7):
        for b_p in np.linspace(0.2, 6.0, 7):
            a = (i1 ** (a_p - 1.0)) * ((n - i) ** (b_p - 1.0))
            _try(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _try(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.005, 0.15, 12):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:

        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:

            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 6. Logistic tails
    for slope in np.linspace(0.003, 0.04, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 7. Semi‑circular shape (new)
    rad = (i - (n - 1) / 2.0) / (n / 2.0)
    semi = np.sqrt(np.maximum(0.0, 1.0 - rad ** 2))
    _try(semi)

    # 8. Random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(40):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 9. Interpolation of a previously found solution, if available
    try:

        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)

# Phase A – Adam on a smoothed surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:

    # work in logits → simplex via soft‑max
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.003
    eps_start, eps_end = 1e-1, 1e-13

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    IMPROVE_LIMIT = 80000

        # simplex point
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:

        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a
        p_rev = p[::-1]
        # convolution a * p_rev (full length 3n‑2) via FFT
        n_a = a.shape[0]
        n_p = p_rev.shape[0]
        out_len = n_a + n_p - 1
        size = 1 << (out_len - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv = np.fft.irfft(fa * fp, size)[:out_len]
        # keep the central segment that corresponds to indices 0 … n‑1
        grad_a = 4.0 * n * conv[n_a - 1: 2 * n_a - 1][::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy mass transfer that reduces the exact peak."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the active (max) entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source proportional to contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # choose a target that is weakly coupled to the max entry
        if np.random.rand() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = int(np.random.choice(n, p=probs_j))
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (exact but cheap)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – Projected sub‑gradient descent (exact objective)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.6 / n
    eta = eta0
    decay = 0.9994
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient (active index)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # decay step size
        eta = max(eta * decay, 1e-12)

        if no_improve > IMPROVE_LIMIT:
            # shake and restart
            noise = np.random.normal(scale=0.011, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase D – Exact active‑set refinement (mass transfer based on weights)
# ----------------------------------------------------------------------
def _refine_exact_phase(a0: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Exact refinement using active‑set weights."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    step_factor = 0.4            # start moving up to 40 % of source mass
    decay = 0.9997
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000
    MIN_STEP = 1e-12

    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Refine {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            break

        # active set – indices whose value is within a tiny tolerance of the max
        eps = max_b * 5e-7
        active = np.where(b >= max_b - eps)[0]
        if active.size == 0:
            break

        # compute weight w_i = Σ_{k∈active} a_{k‑i}
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or a[src] <= MIN_STEP:
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        delta = a[src] * step_factor
        if delta < MIN_STEP:
            step_factor *= 0.5
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        b_new = _full_convolution(a_candidate)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val, best_local_a = new_val, a.copy()
        else:
            step_factor *= 0.7

        step_factor = max(step_factor * decay, MIN_STEP)
        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: mixture with uniform and with reversed vector
# ----------------------------------------------------------------------
def _mix_with_uniform_and_reverse(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try mixing with uniform mass and with the reversed vector."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    rev = a[::-1]

    for alpha in np.linspace(0.001, 0.10, 20):
        # uniform mix
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_a, best_val = cand, val

        # reverse mix
        cand2 = (1.0 - alpha) * a + alpha * rev
        cand2 /= cand2.sum()
        val2 = _objective_value(cand2)
        if val2 < best_val:
            best_a, best_val = cand2, val2

    return best_a, best_val


# ----------------------------------------------------------------------
# Gaussian smoothing (same‑mode convolution)
# ----------------------------------------------------------------------
def _smooth(a: np.ndarray, sigma: float) -> np.ndarray:
    """Apply a (centered) Gaussian smoothing kernel to a."""
    n = a.shape[0]
    # kernel size ≈ 6σ, forced to be odd
    m = max(3, int(np.ceil(6 * sigma)))
    if m % 2 == 0:
        m += 1
    half = m // 2
    x = np.arange(-half, half + 1, dtype=float)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    sm = np.convolve(a, kernel, mode='same')
    sm = np.maximum(sm, 0.0)
    return sm


# ----------------------------------------------------------------------
# Phase E – Random local search (cheap mass transfers) until budget ends
# ----------------------------------------------------------------------
def _random_local_search(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing that moves a tiny fraction of mass."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.2, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # pick source heavily involved in the current maximum entry
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        tot = contrib.sum()
        if tot > 0.0:
            probs = contrib / tot
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # random target (different from i)
        j = int(np.random.randint(0, n))
        while j == i:
            j = int(np.random.randint(0, n))

        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # cheap incremental update of the convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[RandSearch {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional restart if we have been idle for long
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[RandSearch {it:7d}] restart after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point – invoked by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the supplied time budget.
    Returns a list of floats (Σ = 1) that (hopefully) beats the current record.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – a few thousand entries give a good trade‑off
    n = 9000

    # ------------------------------------------------------------------
    # 1️⃣  Initial parametric sweep
    # ------------------------------------------------------------------
    print("[Init] starting parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # 2️⃣  Phase A – multiple short Adam restarts (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * (deadline - start_time)
    trials = 0
    while time.time() < phase_a_deadline:
        if trials == 0:
            a0 = best_a
        else:
            a0 = np.random.dirichlet(np.ones(n))
        best_a, best_val = _adam_phase(
            a0, n,
            start=start_time,
            deadline=phase_a_deadline,
            best_a=best_a,
            best_val=best_val,
            target_val=TARGET,
        )
        trials += 1
    print(f"[Phase A] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Phase B – targeted hill‑climbing (≈25 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * (deadline - start_time)
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    best_val = _objective_value(best_a)
    print(f"[Phase B] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Phase C – projected sub‑gradient descent (≈15 % of budget)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase C] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Phase D – exact active‑set refinement (≈10 % of budget)
    # ------------------------------------------------------------------
    phase_d_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _refine_exact_phase(
        best_a, n,
        start=phase_c_deadline,
        deadline=phase_d_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase D] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Phase E – random local search (remaining time)
    # ------------------------------------------------------------------
    phase_e_deadline = deadline - 5.0   # keep a few seconds for final polishing
    best_a, best_val = _random_local_search(
        best_a, n,
        start=phase_d_deadline,
        deadline=phase_e_deadline,
        best_a=best_a,
        best_val=best_val,
    )
    print(f"[Phase E] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final polishing – mix with uniform / reversed & Gaussian smoothing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform_and_reverse(best_a, n)
    print(f"[Mix] after uniform/reverse = {best_val:.6f}")

    # Gaussian smoothing schedule
    sigma = max(1.0, n / 30.0)   # start relatively wide
    while time.time() < deadline - 1.0:
        cand = _smooth(best_a, sigma)
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[Smooth σ={sigma:.2f}] improved to {best_val:.6f}")
        sigma = max(1.0, sigma * 0.92)     # tighten gradually
        if sigma < 1.0:
            break

    print(f"[Final] best after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8️⃣  Report final score using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:       # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
