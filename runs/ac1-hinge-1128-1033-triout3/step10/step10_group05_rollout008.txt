def solve_lp(a, M):
    n = len(a)
    # Build sparse matrix
    rows = 2*n - 1
    A = sp.lil_matrix((rows, n))
    for i in range(n):
        A[i:i+n, i] = a
    A_csr = A.tocsr()
    g = cp.Variable(n, nonneg=True)
    constraints = [A_csr @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    if g.value is None:
        # fallback: uniform
        return np.full(n, 1.0/n)
    return np.array(g.value, dtype=float)

"""
A fast multi‑phase optimizer for the self‑convolution L∞‑problem.

Phases (all respect a 1000 s global budget):
  1.  Initialise from a family of analytically‑motivated shapes
      (arcsine‑type, power‑law, Dirichlet draws) and keep the best.
  2.  Refine the best candidate with a few steps of Adam on a
      smooth log‑sum‑exp surrogate of the max‑convolution.
  3.  An *exact* LP‑based update: for the current vector a compute the
      maximal convolution value M, solve a linear program that maximises
      Σ g under the constraint a * g ≤ M, normalise g and mix a←(1‑t)a+t g.
  4.  Projected sub‑gradient descent on the true max‑convolution objective.
  5.  Deterministic greedy refinement that moves mass from the index that
      contributes most to the maximal entry towards a low‑contributing index.
  6.  Tiny uniform mixing (often yields a marginal improvement).

The algorithm works on a moderate size (n≈800) – small enough for the LP
phase yet large enough to capture the “continuous” optimum.  The final
vector is returned directly; `evaluate_sequence` will be called by the
harness to obtain the score.
"""

import time
import numpy as np
import scipy.sparse as sp

# cvxpy is optional but available for the LP‑step.
import cvxpy as cp


# ----------------------------------------------------------------------
# Convolution / objective utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (sum need not be 1)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _full_convolution(a)
    return float(2.0 * a.shape[0] * b.max() / (s * s))


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 1 – families + random Dirichlet initial search
# ----------------------------------------------------------------------
def _initial_search(n: int, rng: np.random.Generator,
                    start: float, deadline: float) -> tuple[np.ndarray, float]:
    """Return a good starting vector (normalised sum=1)."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)

    eps = 1e-12
    x = (np.arange(n) + 0.5) / n

    # arcsine‑type (beta) + uniform mixing
    while time.time() < deadline:
        beta = rng.uniform(0.35, 0.80)
        mix = rng.uniform(0.0, 0.03)
        a = (x * (1 - x) + eps) ** (-beta)
        a = (1.0 - mix) * a + mix * np.ones(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # power‑law shape (p) + optional uniform mix
        p = rng.uniform(0.8, 2.5)
        mix = rng.uniform(0.0, 0.02)
        i = np.arange(n)
        a = (i + 1) ** (-p) + (n - i) ** (-p)
        a = (1.0 - mix) * a + mix * np.ones(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # a few Dirichlet draws
        a = rng.dirichlet(np.ones(n))
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # safety break if time is getting low
        if time.time() > deadline - 0.5:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Adam on a log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                start: float,
                deadline: float,
                best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth approximation of the max‑convolution."""
    n = a0.shape[0]
    # logits → simplex via soft‑max
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.006
    eps_start, eps_end = 0.15, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMP_LIMIT = 30000

    while time.time() < deadline:
        # current point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # schedules (linear in elapsed fraction)
        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over all positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a ( 2·n·∑_k p_k (a*a)_k ) = 4·n·conv(a, rev(p))
        p_rev = p[::-1]
        # convolution a * p_rev (full) – we need only the middle n entries
        conv_ab = _full_convolution(a) * 0.0  # placeholder to reuse utility
        # compute via FFT for speed
        size = 1 << (2 * n - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv_ab = np.fft.irfft(fa * fp, size)[:2 * n - 1]
        segment = conv_ab[n - 1: 2 * n - 1]          # length n
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if no_improve > IMP_LIMIT:
            # random restart
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Exact LP‑based update (iteration of the linear‑program
#           described in the problem statement)
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Return the (2n‑1)×n matrix B such that B@g = a∗g (full convolution)."""
    n = a.shape[0]
    rows = 2 * n - 1
    # LIL is convenient for column‑wise filling
    B = sp.lil_matrix((rows, n), dtype=np.float64)
    for i in range(n):
        B[i:i + n, i] = a
    return B.tocsr()


def _lp_step(a: np.ndarray, M: float) -> np.ndarray:
    """Solve max Σ g  s.t. B·g ≤ M, g≥0.  Return the (unnormalised) solution."""
    n = a.shape[0]
    B = _build_conv_matrix(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [B @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS,
                   warm_start=True,
                   max_iters=2000,
                   abstol=1e-9,
                   reltol=1e-9,
                   feastol=1e-9,
                   verbose=False)
    except Exception:
        # fallback – return uniform
        return np.full(n, 1.0 / n)

    if g.value is None:
        return np.full(n, 1.0 / n)
    return np.array(g.value, dtype=float)


def _lp_refine(a0: np.ndarray,
               start: float,
               deadline: float,
               best_val: float) -> tuple[np.ndarray, float]:
    """Iteratively apply the LP step and mix the result with a."""
    a = a0.copy()
    n = a.shape[0]
    step = 0.12                     # initial mixing weight
    it = 0
    while time.time() < deadline and it < 25:
        b = _full_convolution(a)
        M = b.max()
        g = _lp_step(a, M)
        if g.sum() == 0:
            break
        g = g / g.sum()
        # geometric decay of the mixing weight
        mix = step * (0.85 ** it)
        cand = (1 - mix) * a + mix * g
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        cand_val = _objective(cand)
        if cand_val < best_val - 1e-12:
            best_val, a = cand_val, cand.copy()
            print(f"[LP   {it:2d}] new best = {best_val:.6f}")
        else:
            # no improvement – stop early
            break
        it += 1
        if time.time() > deadline - 0.5:
            break
    return a, best_val


# ----------------------------------------------------------------------
# Phase 4 – Projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray,
                       start: float,
                       deadline: float,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    n = a.shape[0]
    eta0 = 0.5 / n
    eta = eta0
    decay = 0.9997
    no_improve = 0
    IMP_LIMIT = 200000
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max() / (a.sum() ** 2)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            it = 0
        else:
            no_improve += 1

        # sub‑gradient of the active (max) entry
        kmax = int(np.argmax(b))
        grad = np.zeros_like(a)
        i_min = max(0, kmax - (n - 1))
        i_max = min(n - 1, kmax)
        idx = np.arange(i_min, i_max + 1)
        jdx = kmax - idx
        mask = idx != jdx
        # off‑diagonal contributions counted twice
        grad[idx[mask]] += 2.0 * a[jdx[mask]]
        # diagonal (i == j) contributes only once
        diag = idx[~mask]
        if diag.size:
            grad[diag] += a[diag]

        # step + simplex projection
        a = _project_simplex(a - eta * grad)
        eta *= decay
        it += 1

        if no_improve > IMP_LIMIT:
            # perturb and restart step size
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
        if time.time() > deadline - 0.5:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 5 – Deterministic local refinement
# ----------------------------------------------------------------------
def _deterministic_refine(a0: np.ndarray,
                          start: float,
                          deadline: float,
                          best_val: float) -> tuple[np.ndarray, float]:
    """Greedy analytic moves that lower the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur = 2.0 * n * b.max() / (a.sum() ** 2)
    best = cur
    best_a = a.copy()
    it = 0
    while time.time() < deadline:
        k = int(np.argmax(b))
        # contributions of each i to entry k
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        partners = k - idx
        contrib = a[idx] * a[partners]

        i_src = idx[np.argmax(contrib)]
        if a[i_src] <= 0:
            break

        # choose target j that has smallest partner value
        partner_vals = a[partners]
        # we want a partner as small as possible while j != i_src
        cand_idxs = np.argsort(a)  # smallest a overall
        j_tgt = None
        for cand in cand_idxs:
            if cand != i_src:
                j_tgt = cand
                break
        if j_tgt is None:
            break

        # partner values for source and target
        a_k_i = a[k - i_src] if 0 <= k - i_src < n else 0.0
        a_k_j = a[k - j_tgt] if 0 <= k - j_tgt < n else 0.0
        if a_k_i <= a_k_j + 1e-12:
            break

        # second‑largest convolution value (excluding the current max)
        b_tmp = b.copy()
        b_tmp[k] = -np.inf
        second = b_tmp.max()
        # analytic delta to bring the max down to the second max
        denom = 2.0 * (a_k_i - a_k_j) + 1e-12
        delta_needed = (b[k] - second) / denom
        delta = min(delta_needed * 0.92, a[i_src])   # safety factor
        if delta <= 1e-12:
            break

        # incremental update of convolution
        b_old = b.copy()
        b[j_tgt:j_tgt + n] += 2.0 * delta * a
        b[i_src:i_src + n] -= 2.0 * delta * a
        L = len(b)
        if 2 * j_tgt < L:
            b[2 * j_tgt] += delta * delta
        if 2 * i_src < L:
            b[2 * i_src] += delta * delta
        ij = i_src + j_tgt
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i_src] -= delta
        a[j_tgt] += delta

        new_val = 2.0 * n * b.max() / (a.sum() ** 2)
        if new_val < cur - 1e-13:
            cur = new_val
            if new_val < best - 1e-13:
                best, best_a = new_val, a.copy()
                if best <= 1.503:
                    break
        else:
            # revert – move was not beneficial
            b = b_old
            a[i_src] += delta
            a[j_tgt] -= delta
            break

        it += 1
        if time.time() > deadline - 0.5:
            break
    return best_a, best


# ----------------------------------------------------------------------
# Phase 6 – Tiny uniform mixing (helps to shave off the last bits)
# ----------------------------------------------------------------------
def _mix_uniform(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    base = a.copy()
    best = _objective(base)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.006, 0.012, 0.025):
        cand = (1.0 - alpha) * base + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best:
            best, base = val, cand.copy()
    return base

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a non‑negative step function under a 1000 s time budget.
    Returns a list of floats that (approximately) minimise `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate n keeps the LP cheap and still expressive
    # ------------------------------------------------------------------
    n = 800

    # ------------------------------------------------------------------
    # Phase 1 – initialise from analytical families + Dirichlet draws
    # ------------------------------------------------------------------
    a, best_val = _initial_search(n, rng, start_time, deadline)

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smooth surrogate (≈15 % of budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.15 * (deadline - start_time)
    a, best_val = _adam_phase(a, start=start_time,
                              deadline=phase2_deadline,
                              best_val=best_val)

    # ------------------------------------------------------------------
    # Phase 3 – Linear‑program based mixing (≈20 % of budget)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.35 * (deadline - start_time)
    a, best_val = _lp_refine(a, start=phase2_deadline,
                             deadline=phase3_deadline,
                             best_val=best_val)

    # ------------------------------------------------------------------
    # Phase 4 – Exact projected sub‑gradient descent (≈35 % of budget)
    # ------------------------------------------------------------------
    phase4_deadline = start_time + 0.70 * (deadline - start_time)
    a, best_val = _subgradient_phase(a, start=phase3_deadline,
                                     deadline=phase4_deadline,
                                     best_val=best_val)

    # ------------------------------------------------------------------
    # Phase 5 – Deterministic greedy refinement (remaining time)
    # ------------------------------------------------------------------
    a, best_val = _deterministic_refine(a, start=phase4_deadline,
                                        deadline=deadline,
                                        best_val=best_val)

    # ------------------------------------------------------------------
    # Phase 6 – Tiny uniform mixing (final polishing)
    # ------------------------------------------------------------------
    a = _mix_uniform(a)
    best_val = _objective(a)

    # ------------------------------------------------------------------
    # Final report (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a.tolist()
