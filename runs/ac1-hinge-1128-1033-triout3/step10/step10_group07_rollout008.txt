def project_simplex(v):
    # ensure v sum to 1 and nonnegative.
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step‑function (probability vector) that minimizes

    2·n·max(conv(a,a))

where conv(a,a) is the full linear self‑convolution.
The search uses a softened “log‑sum‑exp” surrogate for the max,
gradient descent on the simplex (exponentiated‑gradient / mirror descent)
with an annealing schedule for the smoothing temperature.
A quick parametric sweep gives a good start, then the iterative refinement
continues until the time budget is exhausted.  Finally a tiny uniform
mixing is tried and the best vector found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: projection onto the probability simplex (euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero - fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, size chosen once)
# ----------------------------------------------------------------------
def _fft_size_for_length(L: int) -> int:
    """Return the next power‑of‑two >= L (for rfft)."""
    return 1 << (L - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of given size."""
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _gradient_via_softmax_weights(a: np.ndarray,
                                  w: np.ndarray,
                                  size: int) -> np.ndarray:
    """
    Compute 2 * correlation(w, a) where
        w : length 2·n-1  (weights from softmax)
        a : length n
    Returned gradient has shape (n,).
    """
    n = a.shape[0]
    # reverse a once (needed for correlation)
    a_rev = a[::-1]
    wf = np.fft.rfft(w, size)
    arf = np.fft.rfft(a_rev, size)
    corr = np.fft.irfft(wf * arf, size)
    start = n - 1
    grad = 2.0 * corr[start : start + n]
    return grad


# ----------------------------------------------------------------------
# Exact objective for a probability vector (sum == 1)
# ----------------------------------------------------------------------
def _exact_objective(a: np.ndarray, conv_size: int) -> float:
    """Return 2·n·max(conv(a,a)) (exact, assuming sum(a)=1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, conv_size)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Generate a few hand‑crafted candidates (used when no previous solution)
# ----------------------------------------------------------------------
def _candidate_generator(n: int, rng: np.random.Generator):
    """Yield a series of candidate probability vectors."""
    i = np.arange(n, dtype=np.float64)

    # 1. Symmetric exponential (U‑shaped)
    for r in np.linspace(0.55, 0.999, 20):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 2. Power‑law (edge‑peaked)
    for p in np.linspace(-4.0, 4.0, 20):
        j = np.arange(1, n + 1, dtype=np.float64)
        base = j * (n + 1 - j) + 1e-12
        a = np.power(base, -p)
        yield a

    # 3. Cosine‑window (Hann) with exponent
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))  # Hann
    for beta in np.linspace(0.5, 4.0, 12):
        a = np.power(w, beta)
        yield a

    # 4. Gaussian‑like bump (centre heavy)
    for sigma in (n / 6.0, n / 4.0):
        a = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
        yield a

    # 5. Random Dirichlet draws
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        yield a

# Main entry point required by the harness
# ----------------------------------------------------------------------

    """

        # use w_i ∝ exp((b_i - max)/tau) to keep numerics safe

            print(f"[it {it:07d}] jitter injected after stagnation")

                best_local_val = val
                best_local = cand.copy()
