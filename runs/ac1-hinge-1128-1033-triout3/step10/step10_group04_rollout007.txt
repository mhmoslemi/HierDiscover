"""
Propose a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)
for non‑negative a.  The algorithm works in three stages:

1️⃣  **Fast small‑scale search** – optimise a short vector (n≈1500) where
    each iteration is cheap.  We start from a rich parametric sweep,
    then run a greedy “move mass from the most harmful index to the
    least harmful” hill‑climber for most of the time budget.

2️⃣  **Upsampling** – linearly interpolate the best small vector to the
    target length (n=9000).  The scaling invariance of the objective
    guarantees that the up‑sampled shape is a good starting point.

3️⃣  **Large‑scale refinement** – run the same targeted hill‑climber on
    the full‑size vector and finish with a tiny uniform mixing step.

All updates keep the L1‑norm of the vector exactly one, so the objective
is simply 2·n·max(conv).  The code respects the 1000 s time limit and
prints intermediate bounds for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Generate a good starting point by scanning a rich collection of families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector among a large set of parametric families."""
    # start with the uniform vector
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential tails (decreasing & symmetric)
    for r in np.linspace(0.99, 0.9999, 12):
        _try(r ** i)                        # decreasing
        _try(r ** i + r ** (n - 1 - i))     # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 5.0, 15):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.2, 6.0, 6):
        for b_param in np.linspace(0.2, 6.0, 6):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 11):
        _try(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.005, 0.12, 10):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.35, 10):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:

            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    for slope in np.linspace(0.002, 0.04, 5):

        _try(a)
        _try(np.flip(a))

# ----------------------------------------------------------------------
# Targeted hill‑climbing – move mass from high‑contribution to low‑contribution slots
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy mass transfer that reduces the exact peak."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # schedule for the fraction of source mass that may be moved
    max_frac_start, max_frac_end = 0.5, 0.001
    # after how many idle steps we perturb the current best
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # progress in [0,1] used for the linear decay of max_frac
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index where the convolution is currently maximal
        k_max = int(np.argmax(b))

        # --- choose source index (large contribution to the active peak) ---
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(rng.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # --- choose a target index (weakly coupled to the active peak) ---
        if rng.random() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0  # never move mass back to the source immediately
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = int(rng.choice(n, p=probs_j))
            else:
                j = (i + rng.integers(1, n)) % n
        else:
            j = (i + rng.integers(1, n)) % n

        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # --- incremental update of the full convolution (exact) ---
        b_new = b.copy()
        # contributions of the moved mass with all other entries
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic self‑terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # accept move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb the current best to escape stagnation
            a = best_a.copy()
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
