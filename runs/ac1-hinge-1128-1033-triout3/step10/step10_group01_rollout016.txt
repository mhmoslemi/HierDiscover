"""
step_sequence_opt.py

A self‑contained optimizer for the “step‑function” challenge.

The goal is to minimise

    f(a) = 2·n·max_k (a∗a)_k / ( Σ a_i )² ,

where a is a non‑negative vector of length n.
Because the objective is invariant under a global scaling, we may (and do)
enforce Σ a_i = 1 and minimise

    F(a) = 2·n·max_k (a∗a)_k .

The optimiser consists of three independent phases:

1️⃣  **Initialisation** – reuse the historic best sequence (if present),
    otherwise obtain a quick candidate from a two‑parameter Beta family
    (a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}) and a few Dirichlet draws.

2️⃣  **Deterministic hill‑climbing with simulated‑annealing acceptance** –  
    at each iteration the convolution‐lag with maximal value is identified.
    The algorithm then moves a small amount of mass from the index that
    contributes most to that lag towards the index that contributes the least.
    The transferred fraction starts large (≈50 % of the source entry) and
    decays exponentially; a temperature schedule enables occasional uphill
    moves to escape local minima.

3️⃣  **Short projected‑gradient refinement** – after the hill‑climb has
    stalled a few simple gradient steps are taken to polish the solution.

The whole routine respects the 1000 s budget that the platform allocates.
The final vector is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())  # next power of two for FFT speed
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective value 2·n·max(conv(a,a)) assuming Σa = 1.
    The caller must guarantee a is non‑negative and sums to 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _beta_family_best(n: int) -> np.ndarray:
    """
    Sweep a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}.
    Returns the best vector (normalised to sum 1) found on the grid.
    """
    i = np.arange(1, n + 1, dtype=np.float64)

    # grid of exponents – a mix of fine and geometric spacing
    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()

    best_a = None
    best_val = np.inf

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a


def _initial_sequence(n: int) -> np.ndarray:
    """
    Produce a reasonable starting vector of length n.
    Preference order:
        1) Interpolated historic best (height_sequence_1 if it exists)
        2) Best Beta‑family member
        3) A few random Dirichlet draws
    The returned vector is normalised to sum 1.
    """
    # 1) try to reuse the historic solution
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.size == n:
            a0 = old.copy()
        else:
            # linear interpolation onto the new grid
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old.size)
            a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 /= a0.sum()
            print("[init] interpolated historic solution (n=%d)" % n)
            return a0
    except Exception:
        pass

    # 2) Beta family sweep
    try:
        a_beta = _beta_family_best(n)
        if a_beta is not None:
            print("[init] beta‑family sweep produced value %.6f" % _objective(a_beta))
            return a_beta
    except Exception as exc:
        print("[init] beta sweep failed:", exc)

    # 3) a couple of Dirichlet seeds – keep the best
    best_a = None
    best_val = np.inf
    for _ in range(3):
        a = np.random.dirichlet(np.ones(n))
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    if best_a is not None:
        print("[init] Dirichlet seed best value %.6f" % best_val)
        return best_a

    # fallback – uniform (never really reached)
    a = np.full(n, 1.0 / n)
    return a

# Core optimisation: deterministic hill‑climb with annealing
# ----------------------------------------------------------------------


def _hillclimb_anneal(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves mass from indices that dominate the current
    maximal convolution entry toward indices that contribute very little.
    The schedule for the transferred fraction as well as a simple
    temperature schedule are exponential in the elapsed proportion of the
    total budget.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)                     # full convolution, length 2·n‑1
    cur_val = 2.0 * n * b.max()
    best_a, best_val = a.copy(), cur_val

    # schedule parameters
    max_frac_start, max_frac_end = 0.5, 1e-4   # fraction of source mass to move
    temp_start, temp_end = 0.05, 1e-12       # annealing temperature

    # stagnation handling
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress ratio in [0,1]
        prog = (now - start_time) / (deadline - start_time)
        # linear interpolation of the two schedules
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # locate the current worst (maximal) convolution lag
        k_max = int(np.argmax(b))
        # indices that actually contribute to that lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        # contribution of each index to the active lag (a[i] * a[partner])
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        # ----- source selection -------------------------------------------------
        # pick the index with the largest contribution (deterministic, fast)
        if np.any(contrib > 0):
            i_src = int(np.argmax(contrib))
        else:
            # fallback – pick the globally largest element
            i_src = int(np.argmax(a))

        # ----- target selection -------------------------------------------------
        # pick the index with the smallest a[j] (excluding the source)
        tmp = a.copy()
        tmp[i_src] = np.inf                      # do not select the source itself
        j_tgt = int(np.argmin(tmp))

        # ----- amount to move ---------------------------------------------------
        if a[i_src] <= 0.0:
            # source exhausted – re‑sample a new source
            i_src = int(np.argmax(a))
            if a[i_src] <= 0.0:
                break
        delta = a[i_src] * max_frac * np.random.rand()
        if delta < 1e-15:
            # too tiny – skip this iteration
            iteration += 1
            continue

        # ----- incremental update of the convolution ---------------------------
        # The update follows exactly the derivation used in the reference code.
        b_new = b.copy()
        # subtract the contribution of the removed mass
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        # add the contribution of the added mass
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        # correct the self‑terms
        b_new[2 * i_src] += delta * delta
        b_new[2 * j_tgt] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- acceptance -------------------------------------------------------
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            # guard against tiny negative round‑off
            if a[i_src] < 0:
                a[i_src] = 0.0
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling --------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            # random perturbation around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – random perturb")

        # occasional renormalisation to fight floating point drift
        if iteration % 10_000 == 0:
            a /= a.sum()

        iteration += 1

    return best_a, best_val

def _simple_pgd(a0: np.ndarray,
                max_iters: int,
                step: float = 0.05) -> np.ndarray:
    """
    Very light projected‑gradient descent on
        F(a) = 2·n·max(conv(a,a))
    Uses the sub‑gradient of the active lag only.
    """
    a = a0.copy()
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())   # FFT size for fast convolutions

    for _ in range(max_iters):
        # compute convolution via FFT (fast)
        fx = np.fft.rfft(a, size)
        conv = np.fft.irfft(fx * fx, size)[:2 * n - 1]
        k = int(conv.argmax())
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        a_candidate = a - step * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate /= a_candidate.sum()
        a = a_candidate
    return a

    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a problem size.
    # The historic best used n=8000; we stay with that because it gives a
    # good trade‑off between solution quality and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initial candidate
    # ------------------------------------------------------------------
    a = _initial_sequence(n)
    cur_val = _objective(a)
    print(f"[start] initial objective = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Deterministic hill‑climbing with annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        a, cur_val = _hillclimb_anneal(
            a,
            deadline=deadline,
            start_time=start_time,
            target_val=1.503,
        )
        print(f"[hillclimb] best after climb = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Optional short projected‑gradient polishing
    # ------------------------------------------------------------------
    # Give a tiny budget (≈2 % of the total time) to smooth out any
    # leftover roughness.
    if time.time() < deadline - 0.2:
        pgd_iters = 2000
        a = _simple_pgd(a, max_iters=pgd_iters, step=0.05)
        cur_val = _objective(a)
        print(f"[pgd] polished objective = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation (the platform will also compute this,
    #     but doing it once here gives us a quick sanity check)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return a.tolist()
