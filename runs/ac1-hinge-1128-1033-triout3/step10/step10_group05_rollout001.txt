def _fine_tune(a0, n, start, deadline, best_val, best_a):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_val = best_val
    best_local_a = best_a.copy()
    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        # find top contributing indices
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]
        # get top source indices
        src_inds = np.argpartition(contrib, -10)[-10:]  # top 10
        # exclude zeros
        src_inds = src_inds[contrib[src_inds] > 0]
        if len(src_inds) == 0:
            break

        # define target candidates: smallest a values not in src
        target_inds = np.argpartition(a, 10)[:10]
        target_inds = [t for t in target_inds if t not in src_inds]
        if not target_inds:
            break

        best_move = None
        best_new_val = cur_val
        for i in src_inds:
            # compute a_k_i
            ak_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
            for j in target_inds:
                if i == j:
                    continue
                ak_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
                if ak_i <= ak_j + 1e-12:
                    continue
                # compute delta needed to bring current max down to second max (or small step)
                second_max = np.max(np.delete(b, k_max))
                denom = 2.0 * (ak_i - ak_j) + 1e-12
                delta_needed = (b[k_max] - second_max) / denom
                # bound delta <= a[i]
                delta = min(delta_needed * 0.9, a[i])
                if delta <= 1e-12:
                    continue
                # compute new b_max after move using incremental approach
                b_new = b.copy()
                # update conv for move i->j
                b_new[j:j+n] += 2.0 * delta * a
                b_new[i:i+n] -= 2.0 * delta * a
                L = len(b_new)
                two_j = 2*j
                two_i = 2*i
                ij = i + j
                if two_j < L:
                    b_new[two_j] += delta*delta
                if two_i < L:
                    b_new[two_i] += delta*delta
                if ij < L:
                    b_new[ij] -= 2.0*delta*delta
                new_val = 2.0 * n * b_new.max()
                if new_val < best_new_val - 1e-15:
                    best_new_val = new_val
                    best_move = (i, j, delta, b_new)
        if best_move is not None:
            i, j, delta, b = best_move
            a[i] -= delta
            a[j] += delta
            cur_val = best_new_val
            if cur_val < best_local_val - 1e-15:
                best_local_val = cur_val
                best_local_a = a.copy()
                # optionally print progress
                print(f"[Fine {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= 1.503:
                    break
        else:
            # no improvement; maybe random small perturb
            # choose random i, j
            i = np.random.choice(src_inds)
            j = np.random.choice([x for x in range(n) if x != i])
            delta = a[i] * 0.005  # small
            if delta < 1e-12:
                break
            # apply move
            b_old = b.copy()
            b[j:j+n] += 2.0 * delta * a
            b[i:i+n] -= 2.0 * delta * a
            L = len(b)
            two_j, two_i, ij = 2*j, 2*i, i+j
            if two_j < L: b[two_j] += delta*delta
            if two_i < L: b[two_i] += delta*delta
            if ij < L: b[ij] -= 2.0*delta*delta
            a[i] -= delta
            a[j] += delta
            cur_val = 2.0 * n * b.max()
            if cur_val < best_local_val - 1e-15:
                best_local_val = cur_val
                best_local_a = a.copy()
        it += 1
        # maybe also occasionally reset or jitter to escape plateau
        if it % 1000 == 0:
            # small random noise
            noise = np.random.normal(scale=1e-4, size=n)
            a = np.maximum(best_local_a + noise, 0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0*n*b.max()
    return best_local_a, best_local_val

"""
Improved multi‑phase optimizer for the min‑max self‑convolution problem.

The algorithm follows the same coarse‑grained stages as the reference
implementation (Adam smoothing → Simulated‑Annealing → Sub‑gradient
descent → simple deterministic refinement) and adds a **targeted greedy
refinement** phase.  In the final phase we repeatedly select a source
index that contributes most to the current maximal convolution entry
and evaluate a small set of candidate target indices.  For each pair we
compute analytically the maximal admissible mass transfer that would bring
the peak down to the second‑largest entry, apply the move that yields the
largest reduction, and keep the best vector seen.  This fine‑grained
search can shave off the remaining ~0.005 in the objective that the
previous pipeline left untouched.

All helper functions are defined at top level; the entry point required
by the competition harness is ``propose_candidate``.
"""

from __future__ import annotations

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)-1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Initial candidates – hand‑crafted families + random Dirichlet draws
# ----------------------------------------------------------------------
def _initial_candidate(n: int) -> tuple[np.ndarray, float]:
    """Return a good starting vector (sum = 1) among several families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(candidate):
        nonlocal best_a, best_val
        a = np.asarray(candidate, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform
    _try(np.ones(n))

    # symmetric exponential (U‑shaped)
    for r in (0.9999, 0.9995, 0.998, 0.996, 0.995):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _try(a)

    # arcsine‑like (heavy at ends)
    eps = 1e-12
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(np.clip(x * (1 - x), eps, None))
    _try(a)

    # random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(12):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val

# Phase A – Adam on a smooth surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp surrogate of max(conv)."""
    w = np.log(a0 + 1e-12)                     # logits → simplex via soft‑max

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.005
    eps_start, eps_end = 0.2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 50000

    it = 0
    while time.time() < deadline:
        # current point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.007:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val

# Phase B – Simulated‑annealing hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Hill‑climbing with a temperature‑controlled acceptance rule."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    T0 = cur_val * 0.025                     # initial temperature (2.5 % of start)
    max_frac_start, max_frac_end = 0.30, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)               # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        k_max = int(np.argmax(b))

        # ---- choose source i -------------------------------------------------
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # ---- choose target j --------------------------------------------------
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0                                    # avoid source again
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----- incremental update of the convolution -------------------------
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA  {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            # revert move
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the current state a little and continue
            print(f"[SA  {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val

def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n            # start step size ≈ half average entry
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # gradient step + simplex projection
        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # perturb and reset step size
            print(f"[SGD {it:7d}] perturb after stagnation")
            noise = np.random.normal(scale=0.010, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0

        it += 1

    return best_local_a, best_local_val

# Phase D – Deterministic local refinement (simple version)
# ----------------------------------------------------------------------
def _deterministic_refine(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Greedy moves that analytically reduce the current maximal convolution entry.
    Each move transfers mass from the index that contributes most to the
    maximal entry to an index with the smallest contribution.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_val = cur_val
    best_local_a = a.copy()

    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contribution of each i to the maximal entry
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]

        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break  # nothing left to move

        # pick target j with smallest a (heuristic)
        j_candidates = np.argsort(a)
        for cand in j_candidates:
            if cand != i:
                j = int(cand)
                break

        # contributions of i and j to the maximal entry
        a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
        if a_k_i <= a_k_j + 1e-12:
            # cannot reduce the maximal entry with this pair
            break

        # second‑largest convolution entry (excluding k_max)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max == -np.inf:
            second_max = 0.0

        # analytic delta that would bring b[k_max] down to second_max
        denom = 2.0 * (a_k_i - a_k_j) + 1e-12
        delta_needed = (b[k_max] - second_max) / denom
        delta = min(delta_needed * 0.9, a[i])          # safety factor 0.9
        if delta <= 1e-12:
            break

        # incremental convolution update (store a copy for possible revert)
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        if new_val < cur_val - 1e-14:
            cur_val = new_val
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                if best_local_val <= 1.503:
                    break
        else:
            # revert – move was not beneficial
            b = b_old
            a[i] += delta
            a[j] -= delta
            break

        it += 1
        if it % 2000 == 0:
            print(f"[Refine {it:7d}] best = {best_local_val:.6f}")

    return best_local_a, best_local_val

# Phase E – Targeted greedy refinement (fine‑grained)
# ----------------------------------------------------------------------
def _targeted_refine(a0: np.ndarray, n: int,
                     start: float, deadline: float,
                     best_a: np.ndarray, best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Exhaustive‑ish local search that evaluates a small set of candidate
    (source, target) pairs for each iteration.  For each pair we compute
    the maximal admissible delta that would bring the current peak down
    to the second‑largest entry and apply the pair that yields the largest
    reduction.  This phase is deliberately cheap (few candidates) but
    capable of shaving off the remaining few 1e‑4 of the objective.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = a.copy()
    best_local_val = best_val

    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        max_b = b[k_max]

        # contributions to the maximal entry
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]

        # source candidates: top 10 contributors (if any)
        src_inds = np.argpartition(contrib, -10)[-10:]
        src_inds = src_inds[contrib[src_inds] > 0]
        if src_inds.size == 0:
            break

        # target candidates: 10 smallest a values (excluding sources)
        target_inds = np.argpartition(a, 10)[:10]
        target_inds = [t for t in target_inds if t not in src_inds]
        if not target_inds:
            break

        # second largest convolution entry
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max == -np.inf:
            second_max = 0.0

        best_move = None
        best_new_val = cur_val

        # try all (i, j) pairs
        for i in src_inds:
            a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
            for j in target_inds:
                if i == j:
                    continue
                a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
                if a_k_i <= a_k_j + 1e-12:
                    continue
                denom = 2.0 * (a_k_i - a_k_j) + 1e-12
                delta_needed = (max_b - second_max) / denom
                delta = min(delta_needed * 0.9, a[i])
                if delta <= 1e-12:
                    continue

                # incremental update of convolution for this tentative move
                b_new = b.copy()
                b_new[j:j + n] += 2.0 * delta * a
                b_new[i:i + n] -= 2.0 * delta * a
                L = len(b_new)
                two_j = 2 * j
                two_i = 2 * i
                ij = i + j
                if two_j < L:
                    b_new[two_j] += delta * delta
                if two_i < L:
                    b_new[two_i] += delta * delta
                if ij < L:
                    b_new[ij] -= 2.0 * delta * delta

                new_val = 2.0 * n * b_new.max()
                if new_val < best_new_val - 1e-15:
                    best_new_val = new_val
                    best_move = (i, j, delta, b_new)

        if best_move is not None:
            i, j, delta, b = best_move
            a[i] -= delta
            a[j] += delta
            cur_val = best_new_val
            if cur_val < best_local_val - 1e-14:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[Fine {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            # No improving pair found – perform a tiny random jitter
            # to potentially escape a flat region.
            rng = np.random.default_rng()
            i = int(rng.choice(src_inds))
            j_candidates = [t for t in range(n) if t != i]
            j = int(rng.choice(j_candidates))
            delta = a[i] * 0.001
            if delta <= 0:
                break
            # apply move (always accepted because it's tiny)
            b[j:j + n] += 2.0 * delta * a
            b[i:i + n] -= 2.0 * delta * a
            L = len(b)
            two_j, two_i, ij = 2 * j, 2 * i, i + j
            if two_j < L:
                b[two_j] += delta * delta
            if two_i < L:
                b[two_i] += delta * delta
            if ij < L:
                b[ij] -= 2.0 * delta * delta
            a[i] -= delta
            a[j] += delta
            cur_val = 2.0 * n * b.max()
        it += 1

        # occasional clean‑up to keep numbers tidy
        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
    return best_local_a, best_local_val

# Small post‑processing: mix a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a little uniform mass (often helps)."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.15 * (deadline - start_time)
    best_a, best_val = _adam_phase(best_a, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _simulated_annealing_phase(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        rng=rng,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase C – Exact projected sub‑gradient descent (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase D – Simple deterministic refinement (≈5 % of the budget)
    # ------------------------------------------------------------------
    phase_d_deadline = start_time + 0.85 * (deadline - start_time)
    best_a, best_val = _deterministic_refine(
        best_a, n,
        start=phase_c_deadline,
        deadline=phase_d_deadline,
        best_val=best_val
    )

    # ------------------------------------------------------------------
    # Phase E – Targeted fine‑grained refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _targeted_refine(
        best_a, n,
        start=phase_d_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Final tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
```

**Explanation of the new components**

* **Targeted refinement (`_targeted_refine`)** –  
  Instead of moving mass arbitrarily, the routine inspects the peak
  convolution entry, picks the ten most harmful source indices, and
  evaluates a handful of candidate destinations (the ten smallest
  amplitudes).  For each pair it computes the maximal admissible
  transfer `delta` that would bring the peak down to the second‑largest
  entry, applies the best move, and records any improvement.  This
  focused search is cheap (≈ 100 candidate updates per iteration) yet
  can resolve the very fine adjustments needed to push the objective
  below the previous 1.508787 barrier.

* **Time‑budget redistribution** – the earlier phases now occupy
  85 % of the total budget, leaving the final ~15 % for the
  high‑precision targeted phase.

* **Minor tweaks** – tighter restart/perturbation thresholds and a
  final uniform mixing step, both of which were beneficial in the
  reference code.

The combination of coarse‑level global optimisation (Adam, SA, SGD) and
the precise greedy fine‑graining gives the algorithm a better chance
to beat the 1.503 target within the 1000 s limit.
