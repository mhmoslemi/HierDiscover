    # SA loop
    it = 0
    IMPROVE_LIMIT = maybe 50000.

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        # Temperature schedule: start T=0.01, end T=1e-5
        T = 0.01 * (1e-5/0.01) ** progress
        # step size schedule
        max_frac = 0.3 * (1 - progress) + 0.001 * progress

        # Determine peak index
        k_max = int(np.argmax(b))
        # compute contributions:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            prob_src = contrib / total_contrib
            i = int(np.random.choice(n, p=prob_src))
        else:
            i = int(np.argmax(a))

        # if small mass skip
        if a[i] <= 1e-14:
            continue

        # compute rev_contrib for target selection
        rev = np.zeros_like(a)
        rev[mask] = a[idx[mask]]
        # Inverse weight: high rev -> low probability
        inv_weights = 1.0 / (rev + 1e-12)
        inv_weights[i] = 0.0  # cannot select same
        prob_tgt = inv_weights / inv_weights.sum()
        j = int(np.random.choice(n, p=prob_tgt))

        # delta choose random fraction of source mass up to max_frac
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # apply incremental update to b
        b_new = b.copy()
        # slice updates
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        # single index updates
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # metropolis accept
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # update a, b, cur_val
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            # record best
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # optionally print
                if it % 1000 == 0:
                    print(f"[SA {it}] new best = {best_val:.6f}")

        it += 1

"""
Proposes a step function sequence minimising the convolution‑max bound.
Algorithm:
1. Random parametric search on a modest size (n0≈2000) to discover promising shapes.
2. Upsample the best shape to the target size (n≈9000).
3. Perform a long simulated‑annealing / targeted hill‑climbing phase that
   moves small amounts of mass from indices that contribute most to the current peak
   toward indices that contribute least.  Accepts occasional worsening moves
   according to a cooling schedule, allowing the procedure to escape local minima.
4. Finish with a tiny uniform mixing to smooth residual spikes.
The code is deliberately self‑contained and respects the 1000 s budget.
"""
import time
import numpy as np

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv

def _objective_value(a):
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)

def _initial_random_search(n, time_budget, seed=0):
    rng = np.random.default_rng(seed)
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)
    start = time.time()
    i = np.arange(n, dtype=float)
    while time.time() - start < time_budget:
        typ = rng.integers(0, 9)
        if typ == 0:  # exponential tail
            r = rng.uniform(0.995, 0.99999)
            a = r ** i
            if rng.random() < 0.5:
                a = a + a[::-1]
        elif typ == 1:  # power law
            p = rng.uniform(0.2, 6.0)
            a = (i + 1.0) ** (-p)
            if rng.random() < 0.5:
                a = a + a[::-1]
        elif typ == 2:  # beta family
            a_param = rng.uniform(0.2, 6.0)
            b_param = rng.uniform(0.2, 6.0)
            a = (i + 1.0) ** (a_param - 1.0) * (n - i) ** (b_param - 1.0)
        elif typ == 3:  # sine power
            base = np.sin(np.pi * (i + 1.0) / (n + 1.0))
            p = rng.uniform(0.1, 6.0)  # include sub‑unit powers
            a = base ** p
        elif typ == 4:  # logistic tail
            slope = rng.uniform(0.003, 0.04)
            a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
            if rng.random() < 0.5:
                a = a[::-1]
        elif typ == 5:  # piecewise high‑edges
            k_frac = rng.uniform(0.001, 0.12)
            k = max(1, int(k_frac * n))
            high_frac = rng.uniform(0.02, 0.45)
            total_high = 2.0 * k * high_frac
            if total_high >= 0.99:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
        elif typ == 6:  # random Dirichlet
            a = rng.dirichlet(np.ones(n))
        elif typ == 7:  # mixture of two random families
            # take two random Dirichlet vectors and mix
            a1 = rng.dirichlet(np.ones(n))
            a2 = rng.dirichlet(np.ones(n))
            alpha_mix = rng.uniform(0.0, 1.0)
            a = alpha_mix * a1 + (1 - alpha_mix) * a2
        elif typ == 8:  # random perturbation of current best
            a = best_a * (1 + 0.2 * rng.normal(size=n))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s <= 0.0:
                continue
            a /= s
        else:
            continue
        # normalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            continue
        a /= s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val

def _simulated_annealing(a, deadline, start_time, target_val=1.503):
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    no_improve = 0
    IMPROVE_LIMIT = 200_000
    it = 0

    while time.time() < deadline:
        # progress based schedules
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total
        # temperature (for Metropolis) decays exponentially
        T = 0.01 * (1e-5 / 0.01) ** progress
        # maximum fraction of source mass to move
        max_frac = 0.5 * (1 - progress) + 0.001 * progress

        # locate current peak
        k_max = int(np.argmax(b))

        # compute contribution of each index to this peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source index: weighted by contribution (or plain argmax)
        if total_contrib > 0.0 and np.random.rand() < 0.7:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(contrib))

        if a[i] <= 1e-15:
            # nothing to move, try next iteration
            continue

        # weights for targets: low contribution to the peak
        rev = np.zeros_like(a)
        rev[mask] = a[idx[mask]]
        # convert to probability inversely proportional to rev
        inv = 1.0 / (rev + 1e-12)
        inv[i] = 0.0
        inv_sum = inv.sum()
        if inv_sum > 0.0 and np.random.rand() < 0.7:
            prob_t = inv / inv_sum
            j = int(np.random.choice(n, p=prob_t))
        else:
            # fallback: choose far index
            j = (i + np.random.randint(1, n)) % n

        # ensure source != target
        if j == i:
            continue

        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental convolution update (exact)
        b_new = b.copy()
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0.0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                # optional printing
                if it % 1000 == 0:
                    print(f"[SA {it}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake: add small Gaussian noise to current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            # optional print
            print(f"[SA {it}] restart after stagnation")

        it += 1

    return best_a, best_val

def _mix_with_uniform(a):
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry point called by the evaluation harness.
    Returns a list of non‑negative heights that (hopefully) beats the current record.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # Phase 0 – coarse random search on a modest size.
    n_small = 2000
    # Allocate ~15 % of the budget for this phase.
    phase0_deadline = start_time + 0.15 * (deadline - start_time)
    best_small_a, best_small_val = _initial_random_search(n_small, phase0_deadline - start_time, seed=seed)
    print(f"[Phase0] best small objective = {best_small_val:.6f}")

    # Phase 1 – up‑sample to the target length.
    n = 9000
    # interpolate the best small shape onto the large grid
    x_small = np.linspace(0.0, 1.0, best_small_a.shape[0])
    x_large = np.linspace(0.0, 1.0, n)
    a_init = np.interp(x_large, x_small, best_small_a)
    a_init = np.maximum(a_init, 0.0)
    a_init /= a_init.sum()
    a = a_init.copy()
    best_val = _objective_value(a)
    print(f"[Phase1] after up‑sampling objective = {best_val:.6f}")

    # Phase 2 – long simulated‑annealing refinement.
    phase2_deadline = deadline   # use remaining budget
    a, best_val = _simulated_annealing(a, phase2_deadline, start_time)

    print(f"[Phase2] best after SA = {best_val:.6f}")

    # Phase 3 – tiny uniform mixing.
    a, best_val = _mix_with_uniform(a), _objective_value(a)
    print(f"[Final] after uniform mix = {best_val:.6f}")

    # Verify using the secure wrapper.
    try:
        final_score = evaluate_sequence(a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a.tolist()

Proposes a non‑negative step‑function (list of heights) that minimizes

    2·n·max(conv(a, a)) / (sum(a)^2)

where conv is the full linear convolution.  The algorithm works in three
stages:

1. **Random parametric search on a modest size (n₀≈2000).**  
   Many families (exponential, power‑law, beta, sine‑power, logistic,
   piecewise‑constant, Dirichlet, mixtures) are sampled and the best
   shape is kept.

2. **Upsampling to the target length (n≈9000) and simulated‑annealing.**  
   The shape found in step 1 is interpolated onto the larger grid.
   A long SA / targeted hill‑climbing loop repeatedly moves a small
   amount of mass from indices that contribute most to the current peak
   of the convolution to indices that contribute least.  Moves that
   worsen the objective are occasionally accepted according to a
   temperature schedule, allowing escape from local minima.  A second
   SA run from the uniform distribution is performed on the remaining
   time; the best of the two runs is retained.

3. **Final polishing.**  
   A tiny convex combination with the uniform distribution (α∈{0.001,…,0.020})

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------

    """Exact objective (sum is assumed to be 1)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)

# ----------------------------------------------------------------------
# Phase 1 – simulated‑annealing / targeted hill‑climbing on the large grid.
# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, deadline: float,
                         global_start: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA that drives down max(conv(a,a))."""
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_a = a.copy()
    best_val = cur_val

    no_improve = 0
    IMPROVE_LIMIT = 200_000
    it = 0

    while time.time() < deadline:
        # -------- schedule ----------
        elapsed = time.time() - global_start
        total = deadline - global_start
        progress = elapsed / total
        # temperature (Metropolis)
        T = 0.01 * (1e-5 / 0.01) ** progress
        # how much of the source mass we may move
        max_frac = 0.5 * (1.0 - progress) + 0.001 * progress

        # -------- locate the current peak ----------
        k_max = int(np.argmax(b))

        # contributions of each index to this peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        i = int(np.argmax(contrib))
        if a[i] <= 1e-15:
            # nothing to move – skip this iteration
            continue

        # reverse contribution (how strongly each index participates as the
        # *other* partner in the peak)
        rev = np.zeros_like(a)
        rev[mask] = a[idx[mask]]
        rev[i] = np.inf                      # avoid choosing the same index
        j = int(np.argmin(rev))
        if j == i:
            # fallback to a far index
            j = (i + np.random.randint(1, n)) % n

        # size of the transfer
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # ----- exact incremental update of the full convolution -----
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0.0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                if it % 1000 == 0:
                    print(f"[SA {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            # shake the current best a little and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {it:7d}] restart after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – tiny uniform mixing (final polishing)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> np.ndarray:
    """Convex combination of a with a tiny amount of the uniform vector."""
    n = a.shape[0]
    best_a, best_val = a.copy(), _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a
