def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 12000

    # init best_a
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        old_len = old.shape[0]
        # Interpolate linearly to new n
        old_x = np.linspace(0, 1, old_len)
        new_x = np.linspace(0, 1, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0)
        a0 /= a0.sum()
        best_a = a0.copy()
        best_val = _objective(best_a)
        print(f"[init] interpolated from previous length {old_len} to {n}, val={best_val:.6f}")
    except Exception:
        # fallback random
        best_a = np.random.dirichlet(np.ones(n))
        best_val = _objective(best_a)
        print(f"[init] random dirichlet, val={best_val:.6f}")

    # symmetrize best_a
    best_a = (best_a + best_a[::-1]) / 2.0
    best_a /= best_a.sum()

    # Mirror descent loop
    a = best_a.copy()
    stagnation_cnt = 0
    IMPROVE_LIMIT = 4000 # increase
    # schedule parameters
    eta0, eta1 = 0.6, 0.002
    lambda0, lambda1 = 1.0, 800.0
    # temperature for Metropolis (optional)
    T0, T1 = 1e-2, 1e-7

    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute conv and objective
        conv = _full_convolution(a)
        cur_val = float(2.0 * n * conv.max())

        # update best if improved
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            stagnation_cnt = 0
            print(f"[iter] new best {best_val:.6f} at {now - start_time:.1f}s")
        else:
            stagnation_cnt += 1

        # schedule
        prog = (now - start_time) / (deadline - start_time)
        eta = eta0 * (1 - prog) + eta1 * prog
        lam = lambda0 * (lambda1 / lambda0) ** prog
        T = T0 * (T1 / T0) ** prog

        # compute softmax weights
        # conv max for stability
        conv_max = conv.max()
        exp_args = lam * (conv - conv_max)
        # Clip perhaps to avoid overflow
        w = np.exp(exp_args)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # compute gradient g
        # cross_corr = sum_i a_i w_{i+j}
        cross = _cross_corr(w, a)  # shape n
        diag = w[2 * np.arange(n)] * a
        g = 2.0 * cross - diag

        # Mirror update (exponential)
        a_new = a * np.exp(-eta * g)
        # avoid all zeros
        if a_new.sum() == 0:
            a_new = np.random.dirichlet(np.ones(n))
        else:
            a_new /= a_new.sum()

        # enforce symmetry
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new /= a_new.sum()

        # evaluate candidate
        # compute objective for candidate
        conv_cand = _full_convolution(a_new)
        cand_val = float(2.0 * n * conv_cand.max())

        # metropolis acceptance
        accept = False
        if cand_val < cur_val - 1e-15:
            accept = True
        else:
            # probability
            prob = math.exp(-(cand_val - cur_val) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            # cur_val might be updated later

        # stagnation reset
        if stagnation_cnt > IMPROVE_LIMIT:
            # Gaussian perturbation around best
            sigma = 0.02  # smaller than before
            noise = np.random.normal(scale=sigma, size=n)
            a = np.maximum(best_a + noise, 0.0)

            stagnation_cnt = 0
            print(f"[reset] after stagnation at {now - start_time:.1f}s: val={float(2.0 * n * _full_convolution(a).max()):.6f}")

    # final evaluation using evaluate_sequence
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] score: {final_score:.6f}")

    return best_a.tolist()

"""
Non‑negative step‑function optimiser (mirror‑descent with soft‑max weighting).

The algorithm works on the simplex (Σ a_i = 1, a_i ≥ 0) and minimises

    2·n·max_k (a * a)[k]          ( = evaluate_sequence for sum(a)=1 )

by repeatedly smoothing the max‑operator with a temperature λ (soft‑max)
and performing an entropy‑mirror update

    a ← a ∘ exp(−η ∇F_λ(a))

where ∇F_λ(a) = 2 · cross_corr(w, a) and w is the soft‑max of the
self‑convolution a*a.  A schedule for the stepsize η and the
soft‑max sharpness λ drives the iterates from a smooth phase to a
high‑precision phase.  After each update the vector is symmetrised
(a_i = a_{n‑1‑i}) because the optimal solution is known to be symmetric.
Periodically a Gaussian perturbation is injected to escape stagnation.
The routine respects a 1000 s time budget and returns the best sequence
found (as a list of floats) which is finally evaluated with
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, full length)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Full self‑convolution a*a (length 2·n-1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Cross‑correlation needed for the gradient:
#   g_j = Σ_i a_i w_{i+j}   (j = 0,…,n‑1)
# ----------------------------------------------------------------------
def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_j = Σ_i a_i w_{i+j} for j = 0,…,len(a)-1.
    w has length 2·len(a)-1.
    """
    rev_w = w[::-1]                     # reverse w
    conv = _conv_fft(a, rev_w)          # length = n + (2n‑1) - 1 = 3n‑2
    start = len(w) - 1                  # = 2·n‑2
    # slice: start, start‑1, …, start‑n+1  →  n entries
    return conv[start:start - len(a):-1]


# ----------------------------------------------------------------------
# Objective for monitoring (sum(a)=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used only as a safety net
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
