"""
Propose a step‑function (non‑negative vector) that minimises
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

Algorithm
---------

* **Phase 0 – initial pool** – generate a large (few‑hundred) pool of
  candidates from simple parametric families (Beta, Dirichlet, power‑law,
  geometric, sorted / reversed variants).  Keep the best *K* (K≈10)
  according to the exact objective (2·n·max_b) under the normalisation
  sum(a)=1.

* **Phase 1 – multi‑start sub‑gradient polishing** – each of the
  *K* survivors is refined by projected sub‑gradient descent on the
  exact objective.  The step size decays slowly (0.5 → 1e‑6) and the
  routine runs until the overall time budget is exhausted or the
  allotted per‑candidate budget runs out.  The best result over all
  starts is retained.

* **Phase 2 – uniform offset optimisation** – for the current best
  vector we add a constant ``δ`` (subject to the hard cap of 1000) and
  optimise ``δ`` by a short golden‑section search.  Adding a constant
  reduces the ratio because the denominator grows quadratically while
  the numerator grows only linearly (theoretically optimal when
  ``(sum a)² = n·max_b``).  The search evaluates the true ``evaluate_sequence``
  value (fast FFT‑based implementation).

* **Phase 3 – final polishing** – a last short run of the sub‑gradient
  method refines the offset‑augmented vector.

All steps respect the 1000 s budget (a safety margin of 0.5 s is kept
for the final call to ``evaluate_sequence``).  The routine returns the
plain list of floats required by the grader.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple objective (exact, *without* sum‑normalisation)
# ----------------------------------------------------------------------
def _exact_obj(a: np.ndarray) -> float:
    """Returns 2·n·max(conv(a,a)) – the objective used while sum(a)=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _evaluate_score(a: np.ndarray) -> float:
    """Exact score as defined by ``evaluate_sequence`` (no clipping)."""
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:   # safety – mirrors evaluate_sequence behaviour
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent.  The vector is kept on the simplex
    (sum=1) because the objective is homogeneous – the scaling does not
    affect the final ratio.
    """
    a = a0.copy()
    n = a.shape[0]
    best_val = _exact_obj(a)
    best_a = a.copy()

    # Linear decay of the step size from 0.5 down to 1e‑6
    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        # all indices attaining the maximum
        max_idxs = np.where(b == max_b)[0]

        # sub‑gradient (see analysis notes)
        grad = np.zeros_like(a)
        for k in max_idxs:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step schedule
        prog = (now - start_time) / (deadline - start_time)
        step = 0.5 * (1.0 - prog) + 1e-6 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _exact_obj(a_new)
        if new_val < best_val:
            best_val, best_a = new_val, a_new.copy()
            # optional early‑stop when improvement is tiny
        # accept the move if it improves; otherwise keep old a
        if new_val < _exact_obj(a):
            a = a_new

        # occasional jitter to escape flats
        if np.random.rand() < 0.001:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate families (all produce vectors with sum = 1)
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ln(1e‑4) .. ln(10)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_dirichlet_candidate(n: int, alpha: float,
                               rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with scalar concentration."""
    a = rng.dirichlet(np.full(n, alpha))
    return a


def _power_law_candidate(n: int, exp: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-exp}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a /= a.sum()
    return a


def _geometric_candidate(n: int, r: float) -> np.ndarray:
    """a_i ∝ r^{i}."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Small helper: evaluate a constant offset added to a vector.
# ----------------------------------------------------------------------
def _score_with_offset(base: np.ndarray,
                       delta: float) -> float:
    """Score after adding a constant delta (clipping at 1000)."""
    a = base + delta
    # clipping required by evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)
    return _evaluate_score(a)


def _optimise_offset(base: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Golden‑section line‑search for the best constant offset."""
    # maximum offset without breaking the 1000 cap
    max_delta = 1000.0 - base.max()
    if max_delta <= 0.0:
        return base, _evaluate_score(base)

    # golden‑section parameters
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    a0, b0 = 0.0, max_delta
    # limit to a small number of evaluations (≈30)
    for _ in range(30):
        if time.time() >= deadline:
            break
        m1 = a0 + (1 - phi) * (b0 - a0)
        m2 = a0 + phi * (b0 - a0)
        f1 = _score_with_offset(base, m1)
        f2 = _score_with_offset(base, m2)
        if f1 < f2:
            b0 = m2
        else:
            a0 = m1
    delta_opt = (a0 + b0) / 2.0
    a_opt = np.clip(base + delta_opt, 0.0, 1000.0)
    return a_opt, _evaluate_score(a_opt)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose elements satisfy the constraints of
    ``evaluate_sequence`` and (hopefully) improve the published bound
    1.508498 → < 1.503.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Phase 0 – build a large pool of candidates
    # --------------------------------------------------------------
    pool = []

    # 0️⃣  Uniform (the obvious baseline)
    uniform = np.full(n, 1.0 / n)
    pool.append(uniform)

    # try to reuse any previously discovered solution
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        pool.append(a0)
        print("[Init] interpolated previous solution added to pool")
    except Exception:
        pass

    # 1️⃣  Random Beta family (increase count compared to the reference)
    for _ in range(200):
        cand = _random_beta_candidate(n, rng)
        pool.append(cand)
        pool.append(cand[::-1])                     # reversed version
        pool.append(np.sort(cand)[::-1])            # sorted descending

    # 2️⃣  Dirichlet draws with several concentration values
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(20):
            cand = _random_dirichlet_candidate(n, alpha, rng)
            pool.append(cand)
            pool.append(np.sort(cand)[::-1])        # force monotone shape

    # 3️⃣  Power‑law family (exponents on a log‑scale)
    for exp in np.logspace(-2, 2, 30):   # from 0.01 to 100
        pool.append(_power_law_candidate(n, exp))

    # 4️⃣  Geometric family (ratios close to 1)
    for r in np.linspace(0.90, 0.9999, 30):
        pool.append(_geometric_candidate(n, r))

    # ------------------------------------------------------------------
    # Evaluate the pool and keep the K best (K≈10)
    # ------------------------------------------------------------------
    best_val = np.inf
    best_a = None
    scored = []
    for a in pool:
        val = _exact_obj(a)            # sum(a) is 1 → objective = 2·n·max_b
        scored.append((val, a))
    scored.sort(key=lambda x: x[0])
    K = 10
    survivors = [a for _, a in scored[:K]]
    best_val, best_a = scored[0]
    print(f"[Init] best of pool = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – multi‑start sub‑gradient polishing (≈70 % of budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * budget_s
    per_candidate_deadline = phase1_deadline - 5.0   # keep a few seconds free
    # split remaining time roughly evenly among the survivors
    for idx, a_start in enumerate(survivors):
        now = time.time()
        if now >= per_candidate_deadline:
            break
        # give each start a proportional slice of the remaining time
        remaining_candidates = len(survivors) - idx
        sub_deadline = now + (per_candidate_deadline - now) / remaining_candidates
        a_polished, val_polished = _subgradient_polish(
            a_start,
            deadline=sub_deadline,
            start_time=now,
            max_iters=300_000,
        )
        if val_polished < best_val:
            best_val, best_a = val_polished, a_polished.copy()
            print(f"[Polish] new global best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – uniform offset optimisation (constant addition)
    # --------------------------------------------------------------
    # make sure we have some margin
    if time.time() < deadline - 30.0:
        best_a, best_val = _optimise_offset(best_a, deadline)
        print(f"[Offset] after offset optimisation score = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final short polishing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        final_a, final_val = _subgradient_polish(
            best_a,
            deadline=deadline - 2.0,
            start_time=time.time(),
            max_iters=200_000,
        )
        if final_val < best_val:
            best_a, best_val = final_a, final_val
            print(f"[Final polish] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation (the grader will also do this)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        # Should never happen – we keep the best we have
        print("[Final] evaluate_sequence raised:", exc)

    # Return a plain Python list of floats
    return best_a.tolist()
