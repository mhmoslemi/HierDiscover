"""
propose_candidate.py

A time‑bounded optimiser for the max‑convolution problem.

Strategy
--------
1. **Rich parametric sweep** – scan a wide collection of analytic families
   (exponential, power‑law, beta, logistic, piecewise‑constant) with
   far‑extended parameter ranges (including extreme U‑shapes).  The best
   normalised vector from this sweep seeds the optimiser.

2. **Deterministic active‑set refinement** – repeatedly move a small
   fraction of probability mass from the index that contributes most to
   the current maximal convolution entries to the index that contributes
   least.  The move is accepted only if the true objective decreases.
   The step size decays automatically and a tiny random shake‑up is
   performed after long stagnation.

3. **Final uniform mixing** – a convex combination with a minute share of
   the uniform distribution often removes residual spikes.

The whole pipeline respects the 1000 s budget, prints progress and
returns a list of non‑negative heights whose sum is 1 (the normalisation
required by `evaluate_sequence`).  The algorithm is deliberately
different from the previous implementation: it does not rely on Adam or
sub‑gradient descent, but on a greedy, analytically‑exact mass‑transfer
procedure driven by the active set of maximal convolution entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a vector that already satisfies Σa = 1.
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact 2·n·max(a*a) for a normalised (sum = 1) vector."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):

        if s <= 0:

        val = _objective_value(a)

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

# contributor to the “best” contributor.
# ----------------------------------------------------------------------

                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:

    Greedy mass‑transfer guided by the active set of maximal convolution
    entries.  The move is accepted only if it decreases the exact objective.
    """

    best_local_a = a.copy()
    best_local_val = best_val

            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            w[idx] += a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))

        # Guard against pathological selections
        if src == tgt or a[src] <= 1e-16:
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))
            if src == tgt:
                break

        delta = a[src] * step_factor
        if delta < 1e-14:
            step_factor = max(step_factor * decay, min_step_factor)
            continue

        # ----- compute new convolution after moving delta from src → tgt -----
        b_new = b.copy()
        # contribution loss from source
        start_src = src
        end_src = src + n
        b_new[start_src:end_src] -= 2.0 * delta * a
        # contribution gain at target
        start_tgt = tgt
        end_tgt = tgt + n
        b_new[start_tgt:end_tgt] += 2.0 * delta * a
        # quadratic delta² terms
        idx_2src = 2 * src
        if idx_2src < b_new.shape[0]:
            b_new[idx_2src] += delta * delta
        idx_2tgt = 2 * tgt
        if idx_2tgt < b_new.shape[0]:
            b_new[idx_2tgt] += delta * delta
        idx_sum = src + tgt
        if idx_sum < b_new.shape[0]:
            b_new[idx_sum] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-15:          # accept move
            a[src] -= delta
            a[tgt] += delta
            # Numerical clean‑up (avoid drift)
            a = np.maximum(a, 0.0)

            b = b_new
            cur_val = new_val
            no_improve = 0

            if cur_val < best_local_val - 1e-12:
                best_local_val, best_local_a = cur_val, a.copy()
                print(f"[RefineWeighted {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break

            # slightly enlarge step after a success
            step_factor = min(step_factor * 1.05, 0.9)
        else:
            # reject – shrink step
            step_factor = max(step_factor * 0.7, min_step_factor)
            no_improve += 1

        # occasional random shake‑up after prolonged stagnation
        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            step_factor = 0.5
            print(f"[RefineWeighted {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: convex combination with a tiny share of uniform mass.
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point – the function invoked by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the supplied time budget.
    Returns a list of floats (Σ = 1) that (hopefully) beats the current record.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem size – a few thousand entries give a good trade‑off.
    n = 9000

    # ------------------------------------------------------------------
    # 1️⃣ Initial parametric sweep (≈10 % of budget)
    # ------------------------------------------------------------------
    best_a, best_val = _initial_candidate_extended(n)

    # ------------------------------------------------------------------
    # 2️⃣ Deterministic active‑set refinement (≈80 % of budget)
    # ------------------------------------------------------------------
    refine_deadline = start_time + 0.88 * (deadline - start_time)
    best_a, best_val = _refine_weighted(
        best_a, n,
        start=start_time,
        deadline=refine_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
    )
    print(f"[Phase Refine] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final polishing – tiny uniform mixing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Report final score using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
