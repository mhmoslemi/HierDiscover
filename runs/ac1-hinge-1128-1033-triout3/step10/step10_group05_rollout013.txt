"""
Hybrid LP‑guided update + local refinement for the min‑max self‑convolution
problem.

The algorithm works as follows:

1.  Generate a portfolio of simple symmetric candidates (uniform,
    arcsine‑like, power‑law, exponential) and a few random Dirichlet draws.
2.  For each candidate run an iterative scheme:
        * compute the maximal self‑convolution entry M,
        * pick a handful of indices where (a*a) attains its largest values,
        * solve a tiny linear program that maximises the total mass of a new
          vector `b` while keeping the selected convolution entries below M,
          and also caps the total mass of `b` by a large bound (to keep the LP
          bounded when only a few constraints are active).
        * normalise `b` to sum 1 and replace the current vector by a convex
          combination `a ← (1‑t)·a + t·b` (t decays during the run).
        * keep the best vector seen so far.
   The LP has at most a few dozen rows, so it is solved instantly even for
   n≈20000.
3.  After the LP‑driven phase a short deterministic refinement moves mass
   from the index that contributes most to the current maximal convolution
   entry to the index that contributes the least, analytically reducing the
   peak.
4.  Finally a tiny blend with the uniform distribution is tried; this often
   improves the objective by a small amount.
5.  The best vector found (as a plain Python list) is returned.

All numbers are forced to be non‑negative; the routine respects the 1000 s
budget and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, b: np.ndarray | None = None) -> np.ndarray:
    """Full linear convolution a * b (or a * a if b is None)."""
    if b is None:
        b = a
    n = a.shape[0]
    m = b.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a normalised to sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple candidate generators (symmetric, sum = 1)
# ----------------------------------------------------------------------
def _candidate_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _candidate_arcsine(n: int) -> np.ndarray:
    # a_i ∝ 1 / sqrt(x_i (1−x_i)),  x_i = (i+0.5)/n
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = 1.0 / np.sqrt(np.clip(x * (1.0 - x), 1e-12, None))
    a /= a.sum()
    return a


def _candidate_power(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^p + (n-i)^p  (p < 0 gives U‑shape)."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.power(i, p) + np.power(n - i + 1, p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear programme: maximise total mass of b under a few active constraints
# ----------------------------------------------------------------------
def _solve_lp_max_mass(a: np.ndarray,
                       M: float,
                       active_idx: np.ndarray,
                       sum_bound: float = 1e6) -> np.ndarray | None:
    """Return a normalised vector b (sum = 1) solving the LP, or None on failure."""
    from scipy.optimize import linprog

    n = a.shape[0]
    K = active_idx.size
    # Build A_ub for the active convolution constraints
    A_ub = np.zeros((K + 1, n), dtype=np.float64)
    b_ub = np.empty(K + 1, dtype=np.float64)
    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # fill the row: coefficient for b_j is a_i where i = k - j
        for i in range(i_start, i_end + 1):
            j = k - i
            A_ub[row, j] = a[i]
        b_ub[row] = M
    # a global bound on the total mass of b (necessary for boundedness)
    A_ub[K, :] = 1.0
    b_ub[K] = sum_bound

    # maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    # variable bounds: b_i ≥ 0
    bounds = [(0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:   # pragma: no cover
        return None

    if not res.success:
        return None
    b = res.x
    total = b.sum()
    if total <= 0:
        return None
    b /= total
    return b


# ----------------------------------------------------------------------
# Iterative LP‑driven optimisation (single start vector)
# ----------------------------------------------------------------------
def _lp_iterative_optimize(a0: np.ndarray,
                           deadline: float,
                           max_iter: int = 10_000) -> tuple[np.ndarray, float]:
    """Run the LP‑guided update until time runs out; return best vector."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_value(a)

    # mixing step size (decays slowly)
    t = 0.25
    decay = 0.9995
    no_improve = 0
    IMPROVE_LIMIT = 5_000

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv_a = _full_convolution(a)
        M = conv_a.max()
        # Choose a small set of active constraints (top‑K entries)
        K = 5
        # indices of the largest K values (may contain duplicates)
        idx_top = np.argpartition(-conv_a, K - 1)[:K]
        # Solve the LP
        b = _solve_lp_max_mass(a, M, idx_top, sum_bound=1e6)
        if b is None:
            # fall back: simple gradient move (move mass from worst to best)
            # pick source i contributing most to M
            k_max = int(np.argmax(conv_a))
            idx = k_max - np.arange(a.shape[0])
            mask = (idx >= 0) & (idx < a.shape[0])
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[idx[mask]]
            i = int(np.argmax(contrib))
            # pick a target with tiny mass
            j = int(np.argmin(a + 1e-12))
            delta = min(a[i] * 0.5, 1e-3)
            a[i] -= delta
            a[j] += delta
            continue

        # convex combination
        a_new = (1.0 - t) * a + t * b
        val_new = _objective_value(a_new)

        if val_new < best_val - 1e-14:
            best_val = val_new
            best_a = a_new.copy()
            no_improve = 0
            print(f"[LP] new global best = {best_val:.6f}")
        else:
            no_improve += 1

        a = a_new
        t = max(0.01, t * decay)

        if no_improve > IMPROVE_LIMIT:
            # a small perturbation to escape a plateau
            noise = np.random.normal(scale=0.005, size=a.shape[0])
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            t = 0.25
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic local refinement (greedy mass transfer)
# ----------------------------------------------------------------------
def _deterministic_refine(a0: np.ndarray,
                          deadline: float,
                          best_val: float) -> tuple[np.ndarray, float]:
    """Greedy moves that analytically reduce the current maximal entry."""
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_local = cur_val
    it = 0

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contribution of each i to entry k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx[mask]]
        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break

        # target with smallest a (different from i)
        j = int(np.argsort(a)[0] if np.argsort(a)[0] != i else np.argsort(a)[1])

        # values that matter for the k_max entry
        a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0
        if a_k_i <= a_k_j + 1e-12:
            break

        # second largest convolution entry (excluding current max)
        tmp = b.copy()
        tmp[k_max] = -np.inf
        second = tmp.max()
        if second == -np.inf:
            second = 0.0

        # analytic delta to bring the max down towards the second max
        denom = 2.0 * (a_k_i - a_k_j) + 1e-12
        delta_needed = (b[k_max] - second) / denom
        delta = min(delta_needed * 0.9, a[i])
        if delta <= 1e-12:
            break

        # perform the move (incremental convolution update)
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        if new_val < cur_val - 1e-14:
            cur_val = new_val
            if new_val < best_local - 1e-14:
                best_local = new_val
                best_a = a.copy()
                if best_local <= best_val:
                    break
        else:
            # revert – move was not beneficial
            b = b_old
            a[i] += delta
            a[j] -= delta
            break

        it += 1
        if it % 2000 == 0:
            print(f"[Refine] best so far = {best_local:.6f}")

    return best_a, best_local


# ----------------------------------------------------------------------
# Tiny uniform mixing (helps to smooth the final vector)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function under the 1000 s budget.
    Returns a list of non‑negative floats that (approximately) minimise
    `evaluate_sequence`.
    """
    # ------------------------------------------------------------------
    # initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # keep a small safety margin

    # problem size – 20 k gives a fine resolution while staying fast
    n = 20000

    # ------------------------------------------------------------------
    # build a pool of starting vectors
    # ------------------------------------------------------------------
    candidates = []

    # reuse a previous construction if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 /= a0.sum()
            candidates.append(a0)
    except Exception:
        pass

    # hand‑crafted families
    candidates.append(_candidate_uniform(n))
    candidates.append(_candidate_arcsine(n))
    for p in (-0.9, -0.75, -0.6, -0.5, -0.4, -0.3):
        candidates.append(_candidate_power(n, p))
    for r in (0.9999, 0.9995, 0.999, 0.9985, 0.998):
        candidates.append(_candidate_exponential(n, r))

    # a few random Dirichlet draws
    for _ in range(5):
        a = rng.dirichlet(np.ones(n))
        candidates.append(a)

    # ------------------------------------------------------------------
    # run the LP‑based optimiser on each candidate (time‑sliced)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # allocate time equally among candidates, but leave room for final
    # refinement and mixing
    remaining = deadline - time.time()
    per_candidate = max(5.0, remaining / max(1, len(candidates) * 2))

    for idx, a_start in enumerate(candidates):
        if time.time() >= deadline:
            break
        t_end = min(deadline, time.time() + per_candidate)
        a_opt, val_opt = _lp_iterative_optimize(a_start, t_end)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Main] new global best = {best_val:.6f} (candidate {idx})")

    # ------------------------------------------------------------------
    # deterministic greedy refinement on the best vector so far
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5.0:
        best_a, best_val = _deterministic_refine(best_a, deadline, best_val)

    # ------------------------------------------------------------------
    # final tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    if best_a is not None:
        best_a, best_val = _mix_with_uniform(best_a)

    # ------------------------------------------------------------------
    # final sanity check (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist() if best_a is not None else []
