"""
Search for a step‑function (non‑negative sequence) that minimises the
`evaluate_sequence` metric.

The algorithm works in three stages:

1. **Fast parametric random search** –  
   Samples a few simple families (symmetric exponential,
   power‑law, sinusoidal and mixtures) on a moderate size
   (`n_search≈3000`).  The best shape is up‑sampled to the target
   length (`n=9000`).

2. **Deterministic greedy refinement** –  
   Repeatedly moves a tiny amount of mass from the index that
   contributes most to the current maximal convolution entry to an
   index that contributes the least, analytically reducing the
   maximal entry.  This is the same local‐move routine used in the
   reference code but applied only once after the warm‑start.

3. **Tiny uniform mixing** –  
   A final convex combination with a small amount of uniform mass
   often yields a marginal improvement.

The whole procedure respects the 1000 s time budget and returns a plain
list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length = 2·len(a)-1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact value of the evaluation metric for sum(a)=1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def shape_exponential(n: int, r: float) -> np.ndarray:
    """Symmetric exponential: a_i = r**i + r**(n-1-i)."""
    i = np.arange(n, dtype=np.float64)
    return np.power(r, i) + np.power(r, n - 1 - i)


def shape_powerlaw(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: a_i = (i+0.5)^(-p) + (n-i-0.5)^(-p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    return (i ** (-p)) + ((n - i + 0.5) ** (-p))


def shape_sine_power(n: int, q: float) -> np.ndarray:
    """Half‑wave sine raised to a power q."""
    x = np.arange(1, n + 1, dtype=np.float64) / (n + 1)
    return np.sin(np.pi * x) ** q


# ----------------------------------------------------------------------
# Deterministic greedy refinement (Phase D of the reference code)
# ----------------------------------------------------------------------
def _deterministic_refine(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Greedy moves that analytically reduce the maximal convolution entry.
    Operates until the time budget expires.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_val = cur_val
    best_local_a = a.copy()

    it = 0
    while time.time() < deadline:
        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each i to this entry
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]

        i = int(np.argmax(contrib))
        if a[i] <= 0.0:
            break

        # choose a target j with (very) small a value
        j_candidates = np.argsort(a)
        for cand in j_candidates:
            if cand != i:
                j = int(cand)
                break

        # values of the partner entries for i and j
        a_k_i = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_k_j = a[k_max - j] if 0 <= k_max - j < n else 0.0

        if a_k_i <= a_k_j + 1e-12:
            # cannot lower the maximal entry with this pair
            break

        # second‑largest convolution entry (excluding k_max)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max == -np.inf:
            second_max = 0.0

        # analytic delta that would bring b[k_max] down to second_max
        denom = 2.0 * (a_k_i - a_k_j) + 1e-12
        delta_needed = (b[k_max] - second_max) / denom
        delta = min(delta_needed * 0.9, a[i])          # safety factor 0.9
        if delta <= 1e-12:
            break

        # incremental convolution update
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        if new_val < cur_val - 1e-14:
            cur_val = new_val
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                if best_local_val <= 1.503:
                    break
        else:
            # move was not beneficial – revert
            b = b_old
            a[i] += delta
            a[j] -= delta
            break

        it += 1
        if it % 2000 == 0:
            print(f"[Refine {it:7d}] best = {best_local_val:.6f}")

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (helps escape shallow local minima)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a small amount of uniform mass."""
    best_a, best_val = a.copy(), _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises the step‑function within a 1000 s wall‑clock budget.
    Returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the final call
    n_final = 9000                          # target length used by the reference

    # ------------------------------------------------------------------
    # Initialise with a previously found solution if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if prev.size == 0:
            raise Exception("empty")
        # resample/interpolate to the target length
        x_old = np.linspace(0.0, 1.0, prev.shape[0])
        x_new = np.linspace(0.0, 1.0, n_final)
        a_prev = np.interp(x_new, x_old, prev)
        a_prev = np.maximum(a_prev, 0.0)
        a_prev /= a_prev.sum()
        best_a = a_prev
        best_val = _objective_value(best_a)
        print(f"[init] previous solution value = {best_val:.6f}")
    except Exception:
        # fallback: uniform distribution
        best_a = np.full(n_final, 1.0 / n_final)
        best_val = _objective_value(best_a)
        print(f"[init] uniform solution value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fast random search over simple parametric families (≈45 % of budget)
    # ------------------------------------------------------------------
    n_search = 3000
    best_search_a = None
    best_search_raw = float("inf")   # raw objective on the small size (only for guidance)
    search_deadline = start_time + 0.45 * (deadline - start_time)

    while time.time() < search_deadline:
        family = rng.choice(["exp", "pow", "sine", "mix"])
        if family == "exp":
            # r very close to 1 gives a gentle U‑shape
            r = 1.0 - (rng.random() ** 3) * 0.01   # r ∈ [0.99, 1)
            a = shape_exponential(n_search, r)
        elif family == "pow":
            p = rng.uniform(0.1, 3.0)
            a = shape_powerlaw(n_search, p)
        elif family == "sine":
            q = rng.uniform(0.5, 4.0)
            a = shape_sine_power(n_search, q)
        else:   # mix of exponential and power‑law
            r = 1.0 - (rng.random() ** 3) * 0.01
            p = rng.uniform(0.1, 3.0)
            w = rng.random()
            a = w * shape_exponential(n_search, r) + (1.0 - w) * shape_powerlaw(n_search, p)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            continue
        a /= s

        # quick evaluation on the small size
        max_b = _full_convolution(a).max()
        raw_val = 2.0 * n_search * max_b
        if raw_val < best_search_raw:
            best_search_raw = raw_val
            best_search_a = a.copy()
            print(f"[search] new best raw value {best_search_raw:.6f} (family={family}) "
                  f"t={time.time() - start_time:.1f}s")

    # ------------------------------------------------------------------
    # If a better shape was found, up‑sample it to the final length
    # ------------------------------------------------------------------
    if best_search_a is not None:
        # up‑sample (linear interpolation) to the target dimension
        x_old = np.linspace(0.0, 1.0, best_search_a.shape[0])
        x_new = np.linspace(0.0, 1.0, n_final)
        a_up = np.interp(x_new, x_old, best_search_a)
        a_up = np.maximum(a_up, 0.0)
        a_up /= a_up.sum()
        cand_val = _objective_value(a_up)
        print(f"[init] up‑sampled candidate value = {cand_val:.6f}")
        if cand_val < best_val:
            best_a = a_up
            best_val = cand_val
            print("[init] adopting up‑sampled candidate as new best")

    # ------------------------------------------------------------------
    # Deterministic greedy refinement (remaining budget)
    # ------------------------------------------------------------------
    try:
        best_a, best_val = _deterministic_refine(best_a, n_final,
                                                 start=start_time,
                                                 deadline=deadline,
                                                 best_val=best_val)
        print(f"[refine] after deterministic phase value = {best_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[refine] exception:", exc)

    # ------------------------------------------------------------------
    # Tiny uniform mixing (final polishing)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n_final)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
