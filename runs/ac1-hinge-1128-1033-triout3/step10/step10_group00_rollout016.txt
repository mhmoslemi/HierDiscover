"""
Propose a non‑negative step function that minimises
`evaluate_sequence`.  The optimisation is performed on a simplex
(Σ a_i = 1, a_i ≥ 0) with a smoothed version of the ℓ∞–norm of the
self‑convolution.  At each iteration we

1. compute the full convolution b = a * a,
2. build a probability vector w that concentrates on indices where b is
   large (exponential weighting with a temperature that is annealed),
3. obtain the exact gradient of the smoothed objective:
        g_j = 2 Σ_l a_l w_{j+l}               (cross‑correlation)
   which is evaluated by a second FFT‑based convolution,
4. apply an entropy‑mirror (multiplicative) update
        a ← a·exp(−η·g)   and renormalise,
5. accept the move greedily or with a small Metropolis probability
   (simulated‑annealing) to escape shallow plateaus,
6. periodically inject Gaussian noise if the algorithm stalls.

The schedule for the step‑size η, the weighting sharpness γ and the
Metropolis temperature T are all linearly annealed so that the method
starts aggressive and finishes fine‑grained.  The algorithm keeps the
best sequence seen during the run and returns it at the end of the
budget (≈1000 s).  If a previous construction `height_sequence_1` is
available it is interpolated to the chosen length and used as a warm
start.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# FFT‑based full linear convolution (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (full length)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution objective.
#   g_j = 2 * Σ_l a_l * w_{j+l} .   (cross‑correlation a with w)
# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Given a (length n) and a weight vector w (length 2n‑1),
    return the gradient g (length n)."""
    n = a.shape[0]
    rev = a[::-1]                                   # reversed a
    out_len = rev.shape[0] + w.shape[0] - 1
    size = 1 << (out_len - 1).bit_length()
    fr = np.fft.rfft(rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fr * fw, size)[:out_len]
    # the slice that corresponds to Σ_l a_l * w_{j+l}
    start = w.shape[0] // 2                         # = n‑1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
# Simple beta‑family generator (used for a cheap initial sweep)
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Return a normalised Beta‑shaped probability vector of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a
