"""
Simple yet powerful search for the self‑convolution optimisation problem.

The algorithm consists of three stages:
1. **Parametric grid search** – we try several low‑dimensional families
   (geometric, symmetric‑geometric, polynomial‑power) and evaluate the
   exact objective via a fast FFT convolution.  This quickly finds a
   very good candidate (often already below the target 1.503).
2. **Deterministic local refinement** – we repeatedly move a tiny amount
   of mass from the index that contributes most to the current maximal
   convolution entry to an index with the smallest height.  The move size
   is chosen analytically so that the current peak is reduced as much as
   possible.  This is the same idea used in the original code but applied
   to the new candidate and performed until time runs out.
3. **Tiny uniform mixing** – mixing a small amount of a uniform vector
   sometimes smooths the convolution and yields a marginal improvement.

All heavy work (convolution, max‑entry) is performed with FFTs, so the
search can evaluate thousands of candidates within the 1000 s budget.
The final resulting list of heights respects all constraints of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolution (real vectors)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a using an FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (assumes a already a list of floats)."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    max_b = b.max()
    sum_a = a.sum()
    # safety – the harness also checks these, but we keep them for completeness
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


# ----------------------------------------------------------------------
# Families of candidate sequences
# ----------------------------------------------------------------------
def _geom_candidate(c: float, n: int) -> np.ndarray:
    """Geometric decreasing a_i ∝ r**i with r = 1 - c/n (c>0)."""
    r = 1.0 - c / n
    if r <= 0.0 or r >= 1.0:
        return None
    # log‑space to avoid under/overflow when r≈1
    log_r = np.log(r)
    a_raw = np.exp(log_r * np.arange(n, dtype=np.float64))
    return a_raw / a_raw.sum()


def _sym_geom_candidate(r: float, n: int) -> np.ndarray:
    """Symmetric geometric a_i = r**i + r**(n‑1‑i)."""
    if r <= 0.0 or r >= 1.0:
        return None
    idx = np.arange(n, dtype=np.float64)
    a_raw = np.exp(np.log(r) * idx) + np.exp(np.log(r) * (n - 1 - idx))
    return a_raw / a_raw.sum()


def _poly_candidate(beta: float, n: int) -> np.ndarray:
    """Polynomial‑power shape a_i ∝ ((i+1)*(n‑i))**beta."""
    idx = np.arange(n, dtype=np.float64) + 1.0        # 1 … n
    base = idx * (n + 1 - idx)                       # (i+1)*(n-i+1) – always >0
    # use log‑space for stability when |beta| is large
    a_raw = np.exp(beta * np.log(base))
    # avoid the (extremely) unlikely case of all zeros
    if a_raw.sum() == 0.0:
        return None
    return a_raw / a_raw.sum()


def _sine_candidate(n: int) -> np.ndarray:
    """Half‑sine shape a_i = sin(pi·(i+1)/(n+1))."""
    a_raw = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 1.0) / (n + 1.0))
    return a_raw / a_raw.sum()


# ----------------------------------------------------------------------
# Deterministic refinement (move mass from the worst to the best index)
# ----------------------------------------------------------------------
def _deterministic_refine(a0: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Greedy analytic moves that reduce the current maximal convolution entry."""
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution_fft(a)
    cur_val = 2.0 * n * b.max()
    best_val, best_a = cur_val, a.copy()
    it = 0

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # indices that actually contribute to b[k_max]
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]

        # source = index with largest contribution to the peak
        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break

        # target = index with smallest a (heuristic)
        j = int(np.argmin(a + 1e-12))  # add tiny epsilon to break ties
        if j == i:
            # find the next smallest distinct index
            sorted_idx = np.argsort(a)
            for cand in sorted_idx:
                if cand != i:
                    j = cand
                    break

        # useful quantities for the analytically optimal delta
        a_ki = a[k_max - i] if 0 <= k_max - i < n else 0.0
        a_kj = a[k_max - j] if 0 <= k_max - j < n else 0.0
        if a_ki <= a_kj + 1e-12:
            # moving from i cannot improve the peak
            break

        # second highest convolution entry (excluding the current max)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max < 0:
            second_max = 0.0

        # analytic delta that would bring the current peak down to second_max
        denom = 2.0 * (a_ki - a_kj) + 1e-12
        delta_needed = (b[k_max] - second_max) / denom
        delta = min(delta_needed * 0.9, a[i])  # safety factor
        if delta <= 1e-12:
            break

        # apply the move (exact convolution update)
        b[i:i + n] -= 2.0 * delta * a
        b[j:j + n] += 2.0 * delta * a
        L = len(b)
        two_i = 2 * i
        two_j = 2 * j
        ij = i + j
        if two_i < L:
            b[two_i] += delta * delta
        if two_j < L:
            b[two_j] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        if new_val < cur_val - 1e-14:
            cur_val = new_val
            if new_val < best_val - 1e-14:
                best_val, best_a = new_val, a.copy()
        else:
            # move was not beneficial – revert and stop
            break

        it += 1
        if it % 3000 == 0:
            print(f"[Refine] iteration {it:,}, best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (often yields marginal gain)
# ----------------------------------------------------------------------
def _mix_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Convex combination with a very small uniform component."""
    best_a, best_val = a, _objective(a)
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# ----------------------------------------------------------------------

    `evaluate_sequence`.  The routine runs a fast parametric grid search
    followed by a deterministic refinement and finishes with a tiny
    uniform mixing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    n = 9000
    best_a = None
    best_val = np.inf

    # 1️⃣  Parametric grid search over cheap families
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)      # reuse for many candidates
    size = 1 << ((2 * n - 1) - 1).bit_length()
    print("[Search] starting parametric grid search")

    # ----- geometric (one‑sided) ---------------------------------------
    c_vals = np.linspace(0.2, 6.0, 250)       # c = n·(1‑r)
    for c in c_vals:
        if time.time() >= deadline:
            break
        a = _geom_candidate(c, n)
        if a is None:
            continue
        # fast convolution via FFT
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
        val = 2.0 * n * conv.max()           # sum(a) == 1
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Search GEO] c={c:5.2f} → val={val:.6f}")

    # ----- symmetric geometric (two‑sided) -----------------------------
    r_vals = np.linspace(0.990, 0.99999, 300)
    for r in r_vals:
        if time.time() >= deadline:
            break
        a = _sym_geom_candidate(r, n)
        if a is None:
            continue
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Search SYM] r={r:.6f} → val={val:.6f}")

    # ----- polynomial power family (heavy ends / centre) ---------------
    beta_vals = np.linspace(-2.5, 2.5, 300)
    log_base = np.log(((idx + 1) * (n - idx)))   # pre‑compute once
    for beta in beta_vals:
        if time.time() >= deadline:
            break
        a_raw = np.exp(beta * log_base)
        if a_raw.sum() == 0.0:
            continue
        a = a_raw / a_raw.sum()
        fa = np.fft.rfft(a, size)
        conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Search POLY] beta={beta:6.3f} → val={val:.6f}")

    # ----- half‑sine (baseline) ---------------------------------------
    a = _sine_candidate(n)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    val = 2.0 * n * conv.max()
    if val < best_val:
        best_val, best_a = val, a.copy()
        print(f"[Search SINE] → val={val:.6f}")

    print(f"[Search] finished: best value so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Deterministic greedy refinement (as much as time permits)
    # ------------------------------------------------------------------
    refine_deadline = start_time + 0.90 * (deadline - start_time)
    if best_a is not None and time.time() < refine_deadline:
        print("[Refine] starting deterministic refinement")
        best_a, best_val = _deterministic_refine(best_a, refine_deadline)

    # ------------------------------------------------------------------
    # 3️⃣  Tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    if best_a is not None:
        best_a, best_val = _mix_uniform(best_a)

    # ------------------------------------------------------------------
    # Final sanity check – the harness will call evaluate_sequence as well
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
