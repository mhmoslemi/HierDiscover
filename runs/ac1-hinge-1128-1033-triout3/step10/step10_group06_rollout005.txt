"""
propose_candidate
~~~~~~~~~~~~~~~~~

A two‑stage optimizer for the step‑function “max‑convolution” problem.

Stage 0 – a dense parametric sweep (exponential, power‑law, symmetric Beta,
piecewise‑edge) over a fine grid.  Every candidate is projected onto the
probability simplex (Σ a = 1) and evaluated by an exact FFT‑based
convolution, keeping the best seed.

Stage 1 – deterministic hill‑climbing.  From the seed we repeatedly move a
small amount of mass from the index that contributes most to the current
peak of the self‑convolution to the index with the smallest weight.
The convolution is updated incrementally (O(n) vectorised operations) and
a move is accepted only if the objective strictly improves.  The step size
decreases linearly with the remaining time, allowing a fine‑grained polish
in the final seconds.

A tiny convex combination with the uniform distribution gives a final
“polish”.  The returned list satisfies all validation checks of
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself (full length = 2·len(a)‑1)
    using a power‑of‑two FFT for speed and exactness (up to floating
    point rounding).  The input is assumed to be a 1‑D float array.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """
    Exact objective value for a vector a with Σ a = 1:

        2·n·max( a * a ),

    where * is the linear convolution.
    """
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w ≥ 0 , Σ w = 1 }.
    """
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(v.size) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Stage 0 – exhaustive parametric sweep for a strong seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int) -> tuple[np.ndarray, float]:
    """Search a rich collection of parametric families and keep the best."""
    best_a = np.full(n, 1.0 / n)   # uniform (objective = 2)
    best_val = _objective_value(best_a)

    def _try_raw(arr_raw):
        nonlocal best_a, best_val
        a = np.asarray(arr_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)            # 0 … n‑1
    i1 = i + 1.0                              # 1 … n

    # 1. Exponential families (symmetric)
    for r in np.linspace(0.990, 0.999999, 300):
        _try_raw(r ** i + r ** (n - 1 - i))

    # 2. Power‑law families (symmetric)
    for p in np.linspace(0.05, 4.0, 300):
        _try_raw(i1 ** (-p) + (n - i) ** (-p))

    # 3. Symmetric Beta densities (α = β < 1 gives a “U” shape)
    for alpha in np.linspace(0.05, 0.95, 300):
        a = (i1 ** (alpha - 1.0)) * ((n - i) ** (alpha - 1.0))
        _try_raw(a)

    # 4. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.004, 0.12, 30):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.03, 0.45, 30):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try_raw(a)

    # 5. Random Dirichlet draws for diversity
    rng = np.random.default_rng(123456)
    for _ in range(150):
        _try_raw(rng.dirichlet(np.ones(n)))

    # 6. Interpolate a previously found solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try_raw(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 1 – deterministic hill‑climbing refinement
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer: each step moves a small fraction of the
    probability mass from the index that contributes most to the current
    maximal convolution bin to the index with the smallest weight.
    Only improving moves are kept.
    """
    a = a0.copy()
    b = _full_convolution(a)                 # length 2n‑1
    cur_val = 2.0 * n * b.max()

    # keep the best seen during the whole hill‑climb
    best_local_a = best_a.copy()
    best_local_val = best_val

    # counters for stagnation detection
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # choose the peak of the convolution and its contribution array
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))                     # index of current max
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)        # indices that appear in this bin
        contrib = a[idx] * a[k_max - idx]             # contribution of each index
        src = int(idx[np.argmax(contrib)])            # source of largest contribution

        if a[src] <= 1e-16:                           # nothing to move
            break

        # target: smallest weight (but not the same as src)
        target = int(np.argmin(a))
        if target == src:
            # second smallest
            order = np.argsort(a)
            target = int(order[1])

        # --------------------------------------------------------------
        # step size – linearly shrinking with time
        # --------------------------------------------------------------
        progress = (time.time() - start) / (deadline - start)
        max_frac = 0.06 * (1.0 - progress) + 0.0005 * progress   # from ~6 % → 0.05 %
        delta = a[src] * max_frac
        if delta <= 0.0:
            break

        # --------------------------------------------------------------
        # incremental update of the full convolution (vectorised)
        # --------------------------------------------------------------
        b_new = b.copy()
        # remove mass from src
        b_new[src:src + n] -= 2.0 * delta * a
        # add mass to target
        b_new[target:target + n] += 2.0 * delta * a
        # correct the diagonal/self‑terms and the cross‑term (src,target)
        b_new[2 * src] += delta * delta
        b_new[2 * target] += delta * delta
        b_new[src + target] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # --------------------------------------------------------------
        # accept only if the objective strictly improves
        # --------------------------------------------------------------
        if new_val < cur_val - 1e-15:
            a[src] -= delta
            a[target] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_local_val - 1e-15:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[HC {it:7d}] new best = {best_local_val:.6f}")
        else:
            no_improve += 1

        # periodic full recompute to avoid drift from rounding errors
        if it % 4000 == 0:
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        # simple restart after long stagnation
        if no_improve > 150_000:
            a = best_local_a.copy()
            # add tiny Gaussian noise and re‑normalise
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HC {it:7d}] restart after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Tiny uniform mixing polish
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a tiny share of the uniform distribution."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val
