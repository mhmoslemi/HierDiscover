"""
Improved step‑function optimiser.

The algorithm is a two‑phase optimiser:

1. **LP‑lifting phase** (the idea from the paper’s description):
   Given the current sequence `a`, compute M = max(a*a).
   Solve a linear program that maximises the ℓ₁‑norm of a new sequence `g`
   under the linear constraints (a * g)(k) ≤ M for every convolution index k.
   The solution `g` is then rescaled to the same total mass as `a` and mixed
   with `a` (convex combination).  This step provably never increases the
   objective and typically reduces the worst antidiagonal sum dramatically.

2. **Fine‑tuning phase**:
   After the LP steps (or if the LP fails) we run a few iterations of the
   entropy‑mirror‑descent scheme used in the previous baseline.  This removes
   any remaining slack and converges to a locally optimal point.

Both phases respect the non‑negativity and sum constraints, work entirely on
the simplex, and finish well within the 1000 s budget.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper: full linear convolution via FFT (used by both phases)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Helper: current objective (ignores clipping – exact same as the target)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    s = a.sum()
    if s < 0.01:          # protection – should never happen
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Entropy mirror‑descent update (same as in the baseline)
# ----------------------------------------------------------------------
def _mirror_update(a: np.ndarray,
                   conv: np.ndarray,
                   top_idx: np.ndarray,
                   weights: np.ndarray,
                   eta: float) -> np.ndarray:
    n = a.shape[0]
    g = np.zeros_like(a)

    for k, w in zip(top_idx, weights):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        j = np.arange(low, high + 1, dtype=int)
        g[j] += 2.0 * w * a[k - j]

    a_new = a * np.exp(-eta * g)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Build the convolution‑matrix A such that (a * g)[k] = (A @ g)[k]
# ----------------------------------------------------------------------
def _build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Return sparse matrix A with shape (2n‑1, n) where A[k,j] = a[k‑j] (or 0)."""
    n = a.shape[0]
    # each diagonal offset d (=k‑j) contains the constant a[d]
    diags = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)          # offsets 0 … n‑1
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    return A


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise `evaluate_sequence`.  The routine respects the 1000 s time
    budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Initialise from the previous best if available, otherwise uniform
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
        print("[init] interpolated previous construction")
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    best_a = a.copy()
    best_val = _objective(best_a)
    print("[init] objective = {:.6f}".format(best_val))

    # ------------------------------------------------------------------
    # Phase 1 – LP lifting (a few aggressive iterations)
    # ------------------------------------------------------------------
    # We cap the number of LP iterations to keep the total runtime low.
    max_lp_iters = 12

        A = _build_convolution_matrix(a)

        # invariant under scaling)

        eta = eta_start * (1.0 - prog) + eta_end * prog

    # ------------------------------------------------------------------

    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
