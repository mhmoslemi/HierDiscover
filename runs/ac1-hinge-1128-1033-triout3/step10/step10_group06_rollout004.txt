"""
Step‑function optimizer.

The algorithm consists of three stages:

1️⃣  A comprehensive parametric sweep (exponential, power‑law,
    two‑parameter power‑product, Beta, sine‑power, piece‑wise‑constant
    and random Dirichlet draws) provides a solid seed.

2️⃣  A long‑running simulated‑annealing / hill‑climbing phase.
    At each iteration we locate the current convolution peak,
    move a small amount of probability mass from the index that
    contributes most to that peak to an index that contributes least,
    and accept the move according to a Metropolis rule.
    The temperature and the maximal transferred fraction decay
    linearly with the remaining time budget.

3️⃣  A tiny convex blend with the uniform distribution is performed
    as a final polish.

All operations respect the required interface `propose_candidate`.
The code runs under the 1000 seconds budget and returns a list of
non‑negative floats (Σ ≈ 1) that minimizes `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(b)/sum(a)² for a given vector a."""
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – broad parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Search many parametric families for a low‑objective seed."""
    rng = np.random.default_rng(123456)

    # start from uniform – it is already a decent baseline
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)

    def _maybe_update(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)            # 0 … n‑1
    i1 = i + 1.0                              # 1 … n

    # 1. Simple exponential tails (left‑hand and symmetric)
    for r in np.linspace(0.96, 0.9999, 18):
        _maybe_update(r ** i)                         # decreasing left
        _maybe_update(r ** i + r ** (n - 1 - i))      # symmetric U

    # 2. Power‑law tails (left‑hand and symmetric)
    for p in np.linspace(0.1, 3.0, 30):
        _maybe_update(i1 ** (-p))
        _maybe_update(i1 ** (-p) + (n - i) ** (-p))

    # 3. Two‑parameter power product a_i = (i+1)^(-p)*(n-i)^(-q)
    p_vals = np.linspace(0.05, 1.5, 12)
    q_vals = np.linspace(0.05, 1.5, 12)
    for p in p_vals:
        for q in q_vals:
            a = (i1 ** (-p)) * ((n - i) ** (-q))
            _maybe_update(a)

    # 4. Beta densities – grid over (α,β) with emphasis near 0
    alpha_vals = np.concatenate((
        np.linspace(0.05, 0.3, 6),
        np.linspace(0.3, 1.5, 6)
    ))
    beta_vals = alpha_vals.copy()
    for a_param in alpha_vals:
        for b_param in beta_vals:
            # unnormalised beta density on {0,…,n‑1}
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _maybe_update(a)

    # 5. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 14):
        _maybe_update(base_sin ** p)

    # 6. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.003, 0.05, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.06, 0.3, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _maybe_update(a)

    # 7. Random Dirichlet draws for diversification
    for _ in range(120):
        _maybe_update(rng.dirichlet(np.ones(n)))

    # 8. Interpolation of a previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer SA that works directly with the exact max‑convolution.
    A small amount of probability mass is moved from a source index that
    heavily contributes to the current peak to a target that contributes
    little.  Acceptance follows a Metropolis rule with a linearly cooling
    temperature.
    """
    a = a0.copy()
    # initial full convolution
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = best_a.copy()
    best_local_val = best_val

    # schedule parameters
    max_frac_start, max_frac_end = 0.45, 0.001     # maximal transferred fraction
    iter_cnt = 0

    while time.time() < deadline:
        # ---- progressive schedule (linear in remaining time) ----
        progress = (time.time() - start) / (deadline - start)   # 0 → 1
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        # temperature proportional to current value
        T0 = 0.03 * cur_val
        T = max(T0 * (1.0 - progress), 1e-12)

        # ---- locate the current maximal convolution bin ----
        k_max = int(np.argmax(b))

        # ---- source index (large contribution to the peak) ----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            # nothing useful to move from here
            continue

        # ---- target index (little contribution to the peak) ----
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        # weight inversely with rev_contrib (avoid zero division)
        w = 1.0 / (rev_contrib + 1e-14)
        w[i] = 0.0                       # do not pick the same index immediately
        if w.sum() > 0.0:
            probs_j = w / w.sum()
            j = int(np.random.choice(n, p=probs_j))
        else:
            j = (i + np.random.randint(1, n)) % n

        if j == i:
            continue

        # ---- transfer amount (a small fraction of a[i]) ----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the full convolution ----
        # copy to avoid mutating current b on a reject
        b_new = b.copy()
        # removal from i
        b_new[i:i + n] -= 2.0 * delta * a
        # addition to j
        b_new[j:j + n] += 2.0 * delta * a
        # diagonal terms and cross term
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ---- Metropolis acceptance ----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            # apply accepted move permanently
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if cur_val < best_local_val - 1e-15:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[SA {iter_cnt:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    # we have reached the target, stop early
                    break
        # periodic re‑normalisation to curb rounding errors
        if iter_cnt % 6000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
        iter_cnt += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase 2 – deterministic hill‑climbing polish (tiny steps)
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, n: int,
                start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                max_iters: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Very small deterministic moves that never increase the objective.
    Used as a final safeguard after the stochastic phase.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = best_a.copy()
    best_local_val = best_val

    it = 0
    while time.time() < deadline and it < max_iters:
        k_max = int(np.argmax(b))

        # source = index with maximal contribution to peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        i = int(np.argmax(contrib))

        # target = index with minimal contribution (avoid i)
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        rev_contrib[i] = np.inf  # exclude source
        j = int(np.argmin(rev_contrib))

        if a[i] <= 1e-16 or i == j:
            break

        delta = a[i] * 0.001   # tiny redistribution
        # incremental update
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if cur_val < best_local_val:
                best_local_val = cur_val
                best_local_a = a.copy()
        else:
            # no improvement – break early
            break
        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small final polishing – blend with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a = a.copy()
    best_val = _objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ ≈ 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # Problem size – a few thousand entries is a good trade‑off.
    # We start with a modest size for speed and later upscale.
    n_small = 4000
    n_large = 9000

    # --------------------------------------------------------------
    # Phase 0 – parametric sweep on a smaller problem (fast)
    # --------------------------------------------------------------
    print("[Init] parametric sweep (small size)")
    best_a_small, best_val_small = _initial_candidate(n_small)
    print(f"[Init] best after sweep (n={n_small}) = {best_val_small:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – simulated‑annealing on the small size
    # --------------------------------------------------------------
    best_a_small, best_val_small = _simulated_annealing(
        best_a_small, n_small,
        start=start_time,
        deadline=deadline - 30.0,   # reserve ~30 s for up‑scaling & final polish
        best_a=best_a_small,
        best_val=best_val_small,
        target_val=1.503,
    )
    print(f"[SA small] best = {best_val_small:.6f}")

    # --------------------------------------------------------------
    # Phase 1b – deterministic hill‑climb (tiny refinements)
    # --------------------------------------------------------------
    best_a_small, best_val_small = _hill_climb(
        best_a_small, n_small,
        start=start_time,
        deadline=deadline - 20.0,
        best_a=best_a_small,
        best_val=best_val_small,
    )
    print(f"[Hill small] best = {best_val_small:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – upscale to the target size (linear interpolation)
    # --------------------------------------------------------------
    # Preserve shape while increasing resolution.
    old_x = np.linspace(0.0, 1.0, best_a_small.shape[0])
    new_x = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(new_x, old_x, best_a_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    best_val_large = _objective(a_large)
    print(f"[Upscale] objective after interpolation (n={n_large}) = {best_val_large:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing on the large size (shortened schedule)
    # --------------------------------------------------------------
    best_a_large, best_val_large = _simulated_annealing(
        a_large, n_large,
        start=start_time,
        deadline=deadline - 7.0,
        best_a=a_large,
        best_val=best_val_large,
        target_val=1.503,
    )
    print(f"[SA large] best = {best_val_large:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – deterministic hill‑climb on large size (fast)
    # --------------------------------------------------------------
    best_a_large, best_val_large = _hill_climb(
        best_a_large, n_large,
        start=start_time,
        deadline=deadline - 3.0,
        best_a=best_a_large,
        best_val=best_val_large,
    )
    print(f"[Hill large] best = {best_val_large:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final uniform mixing polish
    # --------------------------------------------------------------
    best_a_large, best_val_large = _mix_with_uniform(best_a_large, n_large)
    print(f"[Final] best after mixing = {best_val_large:.6f}")

    # --------------------------------------------------------------
    # Final evaluation using the provided wrapper (just for logging)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a_large.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a_large.tolist()
