def _targeted_random_move(a, b, n, best_val, start, deadline):
    while time < deadline:
        # Determine current max index:
        k_max = np.argmax(b)
        # contributions:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            src = int(np.random.choice(n, p=contrib/total_contrib))
        else:
            src = int(np.argmax(a))
        # target selection: index with minimum a:
        tgt = int(np.argmin(a))
        # possibly random mutate target:
        if np.random.rand() < 0.3:
            # pick random target not src
            candidates = np.arange(n)
            candidates = candidates[candidates != src]
            tgt = int(np.random.choice(candidates))
        # delta:
        progress = (time.time() - start) / (deadline - start)
        max_frac = 0.5 * (1 - progress) + 0.001 * progress
        delta = a[src] * max_frac * np.random.rand()
        if delta <= 0:
            continue
        # apply incremental update (as before)
        b_new = b.copy()
        b_new[src:src+n] -= 2.0 * delta * a
        b_new[tgt:tgt+n] += 2.0 * delta * a
        b_new[2*src] += delta * delta
        b_new[2*tgt] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

            # optionally log

"""
step_function_optimizer.py

A time‑bounded optimizer for the “max‑convolution” objective.

Strategy
--------
1. **Rich parametric sweep** – enumerate many classical window families
   (exponential, power‑law, beta, sine‑power, Hann/Hamming/Blackman,
   Kaiser, logistic) together with random Dirichlet draws.  The best
   normalised vector from this pool seeds the later phases.

2. **Multi‑start Adam** – run a few short Adam optimisations from many
   random starting points (Dirichlet draws).  The surrogate is a
   log‑sum‑exp approximation of the max‑convolution with an annealed
   temperature, allowing the optimiser to escape shallow local minima.

3. **Projected sub‑gradient descent** – a classic sub‑gradient method
   on the exact max‑convolution, with a diminishing step‑size and
   periodic random perturbations to avoid stagnation.

4. **Intensive random hill‑climbing** – many fast, cheap moves that
   transfer a small fraction of mass from indices that heavily
   contribute to the current peak toward indices with little mass.
   Incremental updates of the full convolution avoid recomputing FFTs.

5. **Uniform mixing** – a final convex blend with a tiny amount of the
   uniform distribution may smooth residual spikes.

All phases respect a global 1000 s budget.  Progress and intermediate
best values are printed for monitoring.  The returned list sums to 1
(and therefore can be fed directly to `evaluate_sequence`).
"""
import time
import numpy as np
from scipy import special  # for Kaiser window

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Generate a good starting point by scanning many parametric families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector among a large set of parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential tails
    for r in np.linspace(0.995, 0.99999, 12):
        _try(r ** i)                        # monotone decreasing
        _try(r ** i + r ** (n - 1 - i))     # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 6.0, 14):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes (including extreme parameters)
    for a_param in np.linspace(0.1, 8.0, 9):
        for b_param in np.linspace(0.1, 8.0, 9):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power family (typical window shapes)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _try(base_sin ** p)

    # 5. Classical window functions
    # Hann
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    _try(hann)
    # Hamming
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    _try(hamming)
    # Blackman
    blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
                + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    _try(blackman)
    # Kaiser (several beta values)
    for beta in np.linspace(2.0, 12.0, 5):
        kaiser = special.i0(beta * np.sqrt(1.0 - ((2.0 * i) / (n - 1) - 1.0) ** 2))
        kaiser /= kaiser.max()
        _try(kaiser)

    # 6. Logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.003, 0.04, 6):
        a1 = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a1)
        _try(np.flip(a1))

    # 7. Gaussian blobs (different widths)
    for sigma in np.linspace(0.05, 0.35, 6):
        centre = (n - 1) / 2.0
        gauss = np.exp(-0.5 * ((i - centre) / (sigma * n)) ** 2)
        _try(gauss)

    # 8. Random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(80):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 9. Interpolation of a previously found construction (if present)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   start: float, deadline: float,
                   best_a: np.ndarray, best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth log‑sum‑exp surrogate."""
    # work in the log‑softmax (logits) domain
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.004
    eps_start, eps_end = 1e-1, 1e-13

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 70_000
    it = 0

    while time.time() < deadline:
        # ----- point on simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- annealing schedules -----
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # ----- gradient of smooth surrogate w.r.t. a -----
        p_rev = p[::-1]

        n_a = a.shape[0]
        n_p = p_rev.shape[0]
        out_len = n_a + n_p - 1
        size = 1 << (out_len - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv = np.fft.irfft(fa * fp, size)[:out_len]
        # slice to obtain gradient of length n
        grad_a = 4.0 * n_a * conv[n_a - 1: 2 * n_a - 1][::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.009:
            w += np.random.normal(scale=0.012, size=n)

        if no_improve > IMPROVE_LIMIT:
            # restart from fresh random logits
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.25
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray, n: int,
                          start: float, deadline: float,
                          best_a: np.ndarray, best_val: float,
                          target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the exact max‑convolution."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.7 / n               # initial step‑size
    eta = eta0
    decay = 0.9993
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 130_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient (active lag)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        eta = max(eta * decay, 1e-13)

        if no_improve > IMPROVE_LIMIT:
            # shake the current best when stuck
            rng = np.random.default_rng()
            noise = rng.normal(scale=0.009, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Intensive random hill‑climbing (mass‑transfer moves with incremental updates)
# ----------------------------------------------------------------------
def _intensive_hillclimb(a_start: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Very cheap O(n) move set:
    - pick a source index that contributes heavily to the current peak,
    - pick a target with little mass,
    - transfer a random fraction of the source,
    - accept only if the exact objective improves.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 250_000
    MIN_DELTA = 1e-15

    while time.time() < deadline:
        # schedule a decreasing maximal transfer fraction
        progress = (time.time() - start) / (deadline - start)
        max_frac = 0.45 * (1.0 - progress) + 0.001 * progress

        # locate the lag with maximal convolution value
        k_max = int(np.argmax(b))

        # contribution of each index to that lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source: weighted by contribution (if any), otherwise by mass
        if total_contrib > 0.0:
            src = int(np.random.choice(n, p=contrib / total_contrib))
        else:
            src = int(np.argmax(a))

        # target: pick a low‑mass index (deterministic min or a random low‑mass)
        if np.random.rand() < 0.35:
            # sample a low‑mass index with probability inversely proportional to a
            w = 1.0 / (a + 1e-12)
            w[src] = 0.0
            tgt = int(np.random.choice(n, p=w / w.sum()))
        else:
            tgt = int(np.argmin(a))

        if src == tgt or a[src] <= MIN_DELTA:
            continue

        delta = a[src] * max_frac * np.random.rand()
        if delta < MIN_DELTA:
            continue

        # incremental update of convolution (exact, O(n))
        b_new = b.copy()
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[2 * src] += delta * delta
        b_new[2 * tgt] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # accept move
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # gentle shake and continue
            rng = np.random.default_rng()
            noise = rng.normal(scale=0.006, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
