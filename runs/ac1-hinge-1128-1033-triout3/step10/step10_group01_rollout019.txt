"""
Step‑function optimizer for the `evaluate_sequence` challenge.

The goal is to minimise

    2·n·max_k (a∗a)_k / (∑ a)^2 ,

where a is a non‑negative vector of length *n*.
The algorithm works in three stages and respects a 1000 s wall‑clock budget.

1️⃣  **Rich initialisation** – several parametric families (Beta,
    triangular “tent”, sinusoidal and exponential) are sampled on a fine
    grid plus a few random Dirichlet draws.  The best of those seeds is
    taken forward.

2️⃣  **Smoothed Adam optimisation** – we minimise a smooth proxy of the
    max‑convolution using the log‑sum‑exp (soft‑max) with an Adam‑style
    update and Euclidean projection onto the probability simplex.
    The soft‑max temperature is slowly increased so that the optimiser
    focuses on the true maximum as the run proceeds.

3️⃣  **Intensive annealed hill‑climbing** – a mass‑transfer Monte‑Carlo
    scheme (very similar to the original “mass‑transfer anneal”) refines
    the best candidate until the deadline.

The code is deliberately written with only NumPy (no external LP
solvers) and makes heavy use of FFT‑based convolutions for speed.
All helper functions are top‑level as required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families ---------------------------------------------------
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a, best_val = None, np.inf
    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 32),
        np.geomspace(3.0, 12.0, 8)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val

# ----------------------------------------------------------------------
# Smoothed Adam optimiser (soft‑max surrogate of the max)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimiser on the soft‑max approximation of max(conv).
    Keeps the iterate on the probability simplex.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.12               # a little larger than the simple PGD step
    best_a, best_val = a.copy(), _objective(a)

    # FFT size (next power of two >= 2·n‑1)
    size = 1 << ((2 * n - 2).bit_length())
    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- current objective -----
        fx = np.fft.rfft(a, size)
        conv = np.fft.irfft(fx * fx, size)[:2 * n - 1]
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max

        # ----- soft‑max over the convolution entries -----
        # temperature grows with the iteration count so that we focus
        # increasingly on the true maximum.
        # We use a scale that makes the exponent numerically stable.
        beta = 8.0 / (cur_max + 1e-12)            # ~8 · 1/scale
        w = np.exp(beta * (conv - cur_max))      # the max entry gets weight 1
        w_sum = w.sum()
        p = w / w_sum

        # ----- gradient of the soft‑max surrogate -----
        # grad_i = 2 * Σ_k p_k * a_{k-i}
        fp = np.fft.rfft(p, size)
        grad_full = np.fft.irfft(fx * fp, size)[:2 * n - 1]
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # centred slice

        # ----- Adam update (minimise) -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= a_sum

        # ----- evaluate new point -----
        fx2 = np.fft.rfft(a, size)
        conv2 = np.fft.irfft(fx2 * fx2, size)[:2 * n - 1]
        val = 2.0 * n * conv2.max()

        if val < best_val - 1e-12:
            best_val, best_a = val, a.copy()
            print(f"[Adam] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # occasional stochastic perturbation to escape plateaus
        if t % 4000 == 0:
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            m.fill(0.0)
            v.fill(0.0)

    return best_a, best_val

        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source: either the presently dominant contributors or a random one
        if total_contrib > 0 and np.random.rand() < 0.85:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target weights – favour indices that receive little contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback – any other index
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # ----- update the convolution incrementally -----
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        elif T > 0.0:
            prob = math.exp(-(new_val - cur_val) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # restart from best with a gentle perturbation
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s time
    budget and prints intermediate statistics.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # problem sizes – we explore a few, spending more time on the largest
    sizes = [4000, 8000, 12000, 16000]

    global_best_a = None
    global_best_val = np.inf
    global_best_n = None

    for n in sizes:
        # ------------------------------------------------------------------
        # 0️⃣  Warm‑start from a previously found solution (if any)
        # ------------------------------------------------------------------
        a0 = None
        try:
            old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print(f"[init] interpolated previous solution to n={n}")
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 1️⃣  Parameter‑family sweep + Dirichlet seeds
        # ------------------------------------------------------------------
        best_a, best_val = None, np.inf

        # ---- beta family ------------------------------------------------
        try:
            beta_a, beta_val = _beta_family_best(n)
            if beta_val < best_val:
                best_a, best_val = beta_a, beta_val
                print(f"[init] beta‑family (n={n}) value={best_val:.6f}")
        except Exception as exc:
            print(f"[init] beta‑family failed for n={n}:", exc)

        # ---- tent (triangular) -----------------------------------------
        try:
            tent_a, tent_val = _tent_family_best(n)
            if tent_val < best_val:
                best_a, best_val = tent_a, tent_val
                print(f"[init] tent‑family (n={n}) value={best_val:.6f}")
        except Exception as exc:
            print(f"[init] tent‑family error:", exc)

        # ---- sinusoidal -------------------------------------------------
        try:
            sin_a, sin_val = _sin_family_best(n)
            if sin_val < best_val:
                best_a, best_val = sin_a, sin_val
                print(f"[init] sin‑family (n={n}) value={best_val:.6f}")
        except Exception as exc:
            print(f"[init] sin‑family error:", exc)

        # ---- exponential ------------------------------------------------
        try:
            exp_a, exp_val = _exp_family_best(n)
            if exp_val < best_val:
                best_a, best_val = exp_a, exp_val
                print(f"[init] exp‑family (n={n}) value={best_val:.6f}")
        except Exception as exc:
            print(f"[init] exp‑family error:", exc)

        # ---- a few random Dirichlet draws --------------------------------
        for _ in range(4):
            rand_a = np.random.dirichlet(np.ones(n))
            val = _objective(rand_a)
            if val < best_val:
                best_a, best_val = rand_a, val
                print(f"[init] Dirichlet seed improved to {best_val:.6f}")

        # ---- use the interpolated a0 if it beats the current best -------
        if a0 is not None:
            val0 = _objective(a0)
            if val0 < best_val:
                best_a, best_val = a0, val0
                print(f"[init] interpolated start beats others, value={best_val:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Smoothed Adam optimisation (time‑boxed)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # Give each size at most 35 s (or a share of the remaining budget)
        adam_deadline = time.time() + min(35.0, remaining * 0.3)
        a_refined, val_refined = _adam_optimize(best_a, n, adam_deadline, start_time)
        print(f"[Adam] n={n} best after refinement = {val_refined:.6f}")

        if val_refined < global_best_val:
            global_best_val, global_best_a, global_best_n = (
                val_refined, a_refined.copy(), n
            )

        # early exit if we already beat the target
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final intensive hill‑climbing (uses whatever time is left)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            global_best_n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Secure final evaluation (optional, for debugging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
