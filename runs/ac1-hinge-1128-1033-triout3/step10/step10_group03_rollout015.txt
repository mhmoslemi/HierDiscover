def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))
```

"""
Propose a step‑function (non‑negative vector) that minimizes the
evaluation function

    2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the full linear convolution of the vector with itself.
The algorithm works in three phases:

0️⃣  Initialise from the previously saved solution (if any).
1️⃣  Explore the two‑parameter Beta family (dense grid + log‑uniform random
    sampling) and a few Dirichlet draws – the best of these is taken as
    a starting point.
2️⃣  Run a simulated‑annealing style mass‑transfer search: repeatedly move a
    random amount of mass from one index to another, accepting improvements
    or (with a temperature‑dependent probability) deteriorations.
3️⃣  Finish with a short projected sub‑gradient descent (as in the reference
    code) to clean up any remaining wiggles.

All helpers are top‑level functions; the only public entry point is
`propose_candidate`.  The routine respects the 1000 s budget, prints
intermediate progress and finally returns the best sequence it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / objective helpers
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself, using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact value of the objective used by `evaluate_sequence`
    when sum(a) == 1 (the optimiser works on the simplex, so this holds).
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σx=1 , x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family generator
# ----------------------------------------------------------------------


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta‑shaped vector a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence discovered before
    the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # length of the vector – a good compromise between expressive power
    # and speed of the FFT based convolution.
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣  Initialise from a previous solution when possible
    # --------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)          # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Broad search in the Beta family + a few Dirichlet draws
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf

    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    # dense grid for (p,q)
    p_grid = np.concatenate((
        np.geomspace(0.001, 0.1, 8, endpoint=False),
        np.geomspace(0.1, 5.0, 12, endpoint=False),
        np.geomspace(5.0, 20.0, 5),
    ))
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is None:
                continue
            val = _objective(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[grid] new best = {best_val:.6f} (p={p:.4g}, q={q:.4g})")
        # early‑exit guard
        if time.time() >= deadline - 30.0:
            break

    # random log‑uniform sampling in the (p,q) plane
    rng = np.random.default_rng()
    random_trials = 8000
    log_min, log_max = np.log(1e-3), np.log(20.0)
    for _ in range(random_trials):
        p, q = np.exp(rng.uniform(log_min, log_max, size=2))
        cand = _beta_shape(p, q, n)
        if cand is None:
            continue
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[rand] new best = {best_val:.6f} (p={p:.4g}, q={q:.4g})")
        if time.time() >= deadline - 30.0:
            break

    # a few Dirichlet draws for extra diversity
    for _ in range(6):
        cand = rng.dirichlet(np.ones(n))
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[dir] new best = {best_val:.6f}")

    # fallback – uniform distribution (guaranteed feasible)
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] uniform start, value=", best_val)

    # --------------------------------------------------------------
    # 2️⃣  Simulated‑annealing / mass‑transfer local search
    # --------------------------------------------------------------
    cur = best_a.copy()
    cur_val = best_val
    best_seq = cur.copy()
    best_score = cur_val

    # annealing schedule parameters
    temp_start = 0.2
    temp_end = 1e-5
    move_start = 0.5      # up‑to 50 % of the source entry may be moved
    move_end = 0.001

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start_time) / (deadline - start_time)

        # temperature and step‑size decay (linear)
        T = temp_start * (1.0 - progress) + temp_end * progress
        max_move = move_start * (1.0 - progress) + move_end * progress

        # propose a neighbour by moving mass from i → j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur[i] == 0.0:
            continue
        # amount to move: a random fraction of cur[i] bounded by max_move
        delta = cur[i] * max_move * rng.random()
        if delta <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection is unnecessary – sum is preserved and both entries stay ≥0

        cand_val = _objective(cand)

        # accept if better, otherwise accept with a Boltzmann probability
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(cand_val - cur_val) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_val = cand, cand_val
            if cand_val < best_score:
                best_score, best_seq = cand_val, cand.copy()
                print(f"[SA] iter {it:06d} new global best = {best_score:.6f}")

        # early exit if we have already reached the target
        if best_score <= 1.503:
            print("[SA] target reached")
            break

        it += 1

    # --------------------------------------------------------------
    # 3️⃣  Short projected sub‑gradient descent for polishing
    # --------------------------------------------------------------
    a = best_seq.copy()
    best_seq = a.copy()
    best_score = best_score

    step_start = 0.1
    step_end = 1e-5
    jitter_prob = 0.002
    jitter_scale = 5e-4
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start_time) / (deadline - start_time)

        # compute convolution and current objective value
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_score:
            best_score, best_seq = cur_val, a.copy()
            print(f"[subgrad] iter {iteration} new best = {best_score:.6f}")
            if best_score <= 1.503:
                break

        # index where the current maximum occurs
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # linear step‑size schedule
        step = step_start * (1.0 - progress) + step_end * progress
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        # occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            a_candidate += np.random.normal(scale=jitter_scale, size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate
        iteration += 1

    # --------------------------------------------------------------
    # 4️⃣  Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
