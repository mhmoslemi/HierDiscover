"""
A fast multi‑stage optimizer for the step‑function “max‑convolution’’ problem.

* Stage 0 – cheap seeding: draw a bunch of random Beta / Dirichlet vectors
  (n≈2000) and keep the best one.
* Stage 1 – mirror‑descent on the exact sub‑gradient of
  f(a)=2·n·max(conv(a,a)).  The update is an exponentiated‑gradient step
  (a_i ← a_i·exp(−η·∇_i f)) followed by a projection onto the simplex.
  η decays as 1/√t.  This yields a fast “first‑order’’ descent that works
  directly on the non‑smooth objective.
* Stage 2 – upscale to the full size (n=8000) by linear interpolation,
  renormalise and perform a brief hill‑climber / sub‑gradient polish to
  wipe out any remaining spikes.
* The routine respects the 1000 s budget, prints intermediate bounds and
  finally returns the best sequence found (as a plain ``list[float]``).

The algorithm differs from the reference implementation by using a
mirror‑descent (exponentiated‑gradient) loop on the exact max‑convolution
sub‑gradient instead of Adam on a smooth approximation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length = 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta‑family generator
# ----------------------------------------------------------------------
def _random_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw (p,q) on a log‑scale and return the normalised Beta shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    logp = rng.uniform(-9.2, 2.3)   # ln(1e‑4) .. ln(10)
    logq = rng.uniform(-9.2, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Mirror‑descent on the exact sub‑gradient of f(a)=2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _mirror_descent(a0: np.ndarray,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    eta0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *exact* objective.
    The gradient of the max‑convolution is obtained via a tiny FFT
    (indicator of the active indices).
    """
    a = a0.copy()
    n = a.shape[0]
    it = 0
    # pre‑allocate buffer for the indicator vector (2·n‑1)
    ind_buf = np.zeros(2 * n - 1, dtype=np.float64)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- evaluate current point -----
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[MD]  new best = {best_val:.6f}  iter={it}")

        # ----- compute sub‑gradient of max_b -----
        ks = np.where(b == max_b)[0]            # active convolution indices
        ind_buf.fill(0.0)
        ind_buf[ks] = 1.0                       # indicator of active ks
        # reverse the indicator (needed for the convolution trick)
        rev = ind_buf[::-1]
        conv = _conv_fft(a, rev)                # length 3·n‑2
        # extract the slice that corresponds to the sub‑gradient
        idx = (2 * n - 2) - np.arange(n)        # (2n‑2)‑i
        grad_b = 2.0 * conv[idx]                # sub‑gradient of max_b
        grad_f = 2.0 * n * grad_b                # gradient of the full objective

        # ----- exponentiated‑gradient update (mirror descent) -----
        it += 1
        eta = eta0 / np.sqrt(it)                # simple 1/√t schedule
        a = a * np.exp(-eta * grad_f)          # stay non‑negative
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s                               # project onto simplex

        # occasional jitter to shake off plateaus
        if np.random.rand() < 0.008:
            a = np.maximum(a + np.random.normal(scale=0.001, size=n), 0.0)
            a = a / a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Fast O(n) hill‑climber (tiny mass moves)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly moves a small chunk of mass from a “source’’
    index to a “target’’ index, accepting moves that lower the objective or
    that pass a simulated‑annealing test.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-14
    no_improve = 0
    IMPROVE_LIMIT = 25000

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = 0.6 * (1 - prog) + 0.001 * prog
        T = 0.03 * (0.01) ** prog               # geometric temperature drop

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # choose source index proportional to its contribution
        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # choose a target that currently matters little
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform (excluding i)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        # positions where a contributes to b: additive 2*delta*a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # self‑interaction corrections
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            # Metropolis acceptance
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # stochastic restart – recenter around current best with a tiny jitter
            a = best_a.copy()
            a = np.maximum(a + np.random.normal(scale=0.004, size=n), 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print("[Hill] stagnation – random perturbation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient polishing (few passes)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective, using a linear
    step‑size schedule.  The routine stops early when the overall time
    budget is exhausted.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_b = b.max()
        ks = np.where(b == max_b)[0]

        # build the indicator for the active ks (same trick as in mirror descent)
        ind = np.zeros(2 * n - 1, dtype=np.float64)
        ind[ks] = 1.0
        rev = ind[::-1]
        conv = _conv_fft(a, rev)
        idx = (2 * n - 2) - np.arange(n)
        grad_b = 2.0 * conv[idx]                # sub‑gradient of max_b
        grad_f = 2.0 * n * grad_b                # sub‑gradient of the full objective

        # linear schedule – starts around 0.25 and decays to 1e‑5
        progress = (now - start_time) / (deadline - start_time)
        step = 0.25 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad_f
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SubG] new best = {best_val:.6f}")

        # occasional jitter to escape flat regions
        if np.random.rand() < 0.0015:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple blending with the uniform distribution
# ----------------------------------------------------------------------
def _blend_with_uniform(a: np.ndarray,
                        cur_best_val: float,
                        max_iters: int = 15) -> tuple[np.ndarray, float]:
    """Try convex combinations t·a + (1‑t)·u for a few t‑values."""
    n = a.shape[0]
    u = np.full(n, 1.0 / n)
    best_a, best_val = a.copy(), cur_best_val
    for t in np.linspace(0.0, 1.0, max_iters):
        cand = t * a + (1.0 - t) * u
        cand = cand / cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a tiny margin for the final ``evaluate_sequence`` call
    deadline = start_time + budget_s - 0.4

    # --------------------------------------------------------------
    # 0️⃣  Small‑scale seeding (n≈2000)
    # --------------------------------------------------------------
    n_small = 2000
    best_a = None
    best_val = np.inf

    # uniform starter
    a = np.full(n_small, 1.0 / n_small)
    val = _objective(a)
    best_a, best_val = a.copy(), val
    print(f"[Init] uniform (n={n_small}) value={best_val:.6f}")

    # a handful of deterministic shapes
    i = np.arange(n_small, dtype=np.float64)
    # sine shape
    a_sin = np.sin(np.pi * (i + 0.5) / n_small)
    a_sin = a_sin / a_sin.sum()
    val = _objective(a_sin)
    if val < best_val:
        best_a, best_val = a_sin.copy(), val
        print(f"[Init] sine shape value={best_val:.6f}")

    # quadratic (parabolic) shape
    a_quad = (i + 1) * (n_small - i)
    a_quad = a_quad / a_quad.sum()
    val = _objective(a_quad)
    if val < best_val:
        best_a, best_val = a_quad.copy(), val
        print(f"[Init] quadratic shape value={best_val:.6f}")

    # random Beta draws
    for _ in range(120):
        cand = _random_beta_candidate(n_small, rng)
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val
            print(f"[Init] random Beta improved value={best_val:.6f}")

    # Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 2.0, 5.0):
        for _ in range(4):
            cand = rng.dirichlet(np.full(n_small, alpha))
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand.copy(), val
                print(f"[Init] Dirichlet(alpha={alpha}) improved value={best_val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Mirror‑descent on the small problem (≈30 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _mirror_descent(best_a,
                                       deadline=phase1_deadline,
                                       start_time=start_time,
                                       best_a=best_a,
                                       best_val=best_val)

    # --------------------------------------------------------------
    # 2️⃣  Optional hill‑climb on the small problem (≈10 % of budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.40 * budget_s
    best_a, best_val = _hillclimb(best_a,
                                  deadline=phase2_deadline,
                                  start_time=phase1_deadline,
                                  best_a=best_a,
                                  best_val=best_val)

    # --------------------------------------------------------------
    # 3️⃣  Upscale to the full size (n≈8000) and fine‑tune
    # --------------------------------------------------------------
    n_large = 8000
    # linear interpolation on the unit interval
    old_x = np.linspace(0.0, 1.0, n_small)
    new_x = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(new_x, old_x, best_a)
    a_large = np.maximum(a_large, 0.0)
    a_large = a_large / a_large.sum()
    best_a, best_val = a_large.copy(), _objective(a_large)
    print(f"[Scale] up‑scaled to n={n_large}, value={best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Mirror‑descent on the large problem (≈40 % of budget)
    # --------------------------------------------------------------
    phase3_deadline = start_time + 0.80 * budget_s
    best_a, best_val = _mirror_descent(best_a,
                                       deadline=phase3_deadline,
                                       start_time=phase2_deadline,
                                       best_a=best_a,
                                       best_val=best_val,
                                       eta0=0.09)

    # --------------------------------------------------------------
    # 5️⃣  Hill‑climbing polish (≈10 % of budget)
    # --------------------------------------------------------------
    phase4_deadline = deadline
    best_a, best_val = _hillclimb(best_a,
                                  deadline=phase4_deadline,
                                  start_time=phase3_deadline,
                                  best_a=best_a,
                                  best_val=best_val)

    # --------------------------------------------------------------
    # 6️⃣  Sub‑gradient polishing (a few passes)
    # --------------------------------------------------------------
    best_a, best_val = _subgradient_polish(best_a,
                                           deadline=deadline,
                                           start_time=phase4_deadline,
                                           best_a=best_a,
                                           best_val=best_val,
                                           max_iters=140_000)

    # --------------------------------------------------------------
    # 7️⃣  Blend with uniform (helps remove occasional spikes)
    # --------------------------------------------------------------
    best_a, best_val = _blend_with_uniform(best_a, best_val)

    # --------------------------------------------------------------
    # Final secure evaluation (using the provided helper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
