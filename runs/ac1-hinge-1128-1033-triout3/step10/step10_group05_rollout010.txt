"""
LP‑guided mass‑redistribution optimizer for the self‑convolution bound.

The algorithm works with a moderate length (≈2000) to stay fast.
For a given non‑negative vector ``a`` (∑a = 1) we repeatedly:

1. compute the self‑convolution and its maximal entry M,
2. pick the K largest convolution positions (they are the most
   restrictive constraints for a new vector g),
3. solve a tiny linear program  maximise  Σ g  subject to
      a * g ≤ M   (only at the selected K positions)  and
      Σ g ≤ (2·n‑1)·M  (global bound derived from adding all constraints),
   with g ≥ 0.
   The LP pushes mass onto indices that hardly affect the current peak.
4. scale the LP solution back to the same ℓ₁‑norm as ``a`` and
   convex‑combine it with ``a`` (mixing factor t).  If the objective
   improves we keep the new vector.
5. Occasionally perform a tiny uniform mixing which is known to
   smooth remaining irregularities.

The whole loop runs until the time budget (≈1000 s) expires.
The best vector found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Convolution utilities (FFT, full linear convolution a*a)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()               # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective evaluation (identical to evaluate_sequence up to the clipping)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)²)."""
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    s = a.sum()
    if s < 1e-12:                     # should never happen for a proper candidate
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Initialise a smooth arcsine‑like step function (sum = 1)
# ----------------------------------------------------------------------
def _initial_arcsine(n: int) -> np.ndarray:
    """Arcsine‑like shape: a_i ∝ 1/√(x_i·(1‑x_i)), normalised to ℓ₁‑norm 1."""
    eps = 1e-12
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Build the (K+1)×n constraint matrix for the LP
# ----------------------------------------------------------------------
def _build_lp_matrix(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Return a dense matrix A where row k corresponds to the linear
    constraint (a * g)_{idxs[k]} ≤ M.  The final row (sum constraint)
    must be appended by the caller.
    """
    n = a.shape[0]
    K = len(idxs)
    A = np.zeros((K, n), dtype=np.float64)

    for r, k in enumerate(idxs):
        # indices j such that 0 ≤ j < n and 0 ≤ k‑j < n
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        # slice of columns that are active in this row
        js = np.arange(start, end + 1, dtype=int)
        # corresponding a‑coefficients
        A[r, js] = a[k - js]

    return A


# ----------------------------------------------------------------------
# Linear‑programming step: maximise Σg under a*g ≤ M (selected positions)
# ----------------------------------------------------------------------
def _lp_max_sum(a: np.ndarray, M: float, top_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve  maximize Σ_j g_j   subject to   (a*g)_k ≤ M for k∈top_idx
    and   Σ_j g_j ≤ (2·n‑1)·M,   g ≥ 0.
    Returns the optimal g (ℓ₁‑normalised to its own sum) or None on failure.
    """
    n = a.shape[0]
    # Build constraint matrix for the selected K positions
    A_conv = _build_lp_matrix(a, top_idx)               # shape (K, n)

    # Global sum‑bound row
    sum_row = np.ones((1, n), dtype=np.float64)

    # Stack all rows together
    A_ub = np.vstack([A_conv, sum_row])                # (K+1) × n
    b_ub = np.concatenate([np.full(len(top_idx), M, dtype=np.float64),
                           np.array([(2 * n - 1) * M], dtype=np.float64)])

    # Linear objective: maximise Σg  → minimise –Σg
    c = -np.ones(n, dtype=np.float64)

    # Use the HiGHS interior‑point solver (fast for small dense problems)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})

    if not res.success:
        return None

    g = res.x
    # Numerical noise may produce tiny negatives – clip
    g = np.maximum(g, 0.0)
    total = g.sum()
    if total < 1e-12:
        return None
    # Return g as‑is (its own ℓ₁‑norm will be re‑scaled later)
    return g


# ----------------------------------------------------------------------
# Small uniform mixing (helps to smooth tiny irregularities)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> np.ndarray:
    """Convexly blend a with a little uniform mass; keep the best blend."""
    best = a.copy()
    best_val = _objective(best)
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best = cand
    return best

            cand = (1.0 - t) * a + t * g_scaled
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum == 0:
                continue
            cand /= cand_sum
            cand_val = _objective(cand)
            if cand_val < cur_val - 1e-14:
                a = cand
                improved = True
                break
        if not improved:
            # if none of the tested mixings improved, fall back to a milder step
            a = (1.0 - 0.05) * a + 0.05 * g_scaled
            a = np.maximum(a, 0.0)
            a /= a.sum()

    # ------------------------------------------------------------------
    # Final polishing: a deterministic local refinement (same spirit as
    # the original code but very lightweight – one pass of mass transfer)
    # ------------------------------------------------------------------
    def _deterministic_refine(a_vec: np.ndarray) -> np.ndarray:
        """Move a tiny amount of mass from the worst contributor to the
        smallest coefficient; this is cheap and often yields a marginal gain."""
        a_loc = a_vec.copy()
        n_loc = a_loc.shape[0]
        for _ in range(2000):
            b_loc = _full_convolution(a_loc)
            k_max = int(b_loc.argmax())
            # contributions to the maximal entry
            idx = np.arange(n_loc)
            mask = (k_max - idx >= 0) & (k_max - idx < n_loc)
            contrib = np.zeros_like(a_loc)
            contrib[mask] = a_loc[idx[mask]] * a_loc[k_max - idx[mask]]

            i = int(contrib.argmax())
            if a_loc[i] <= 0:
                break
            # pick the smallest index different from i
            j = int(np.argsort(a_loc)[0] if np.argsort(a_loc)[0] != i else np.argsort(a_loc)[1])

            # analytical delta to lower the peak (same formula as in the
            # reference implementation)
            a_k_i = a_loc[k_max - i] if 0 <= k_max - i < n_loc else 0.0
            a_k_j = a_loc[k_max - j] if 0 <= k_max - j < n_loc else 0.0
            if a_k_i <= a_k_j + 1e-12:
                break
            second_max = np.partition(b_loc, -2)[-2]    # second largest entry
            denom = 2.0 * (a_k_i - a_k_j) + 1e-12
            delta_needed = (b_loc[k_max] - second_max) / denom
            delta = min(delta_needed * 0.9, a_loc[i])
            if delta <= 1e-12:
                break

            # perform the move (incrementally update the convolution not needed)
            a_loc[i] -= delta
            a_loc[j] += delta
            a_loc = np.maximum(a_loc, 0.0)
            a_loc /= a_loc.sum()
        return a_loc

    a = _deterministic_refine(a)

    # ------------------------------------------------------------------
    # Final tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    a = _mix_with_uniform(a)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a.tolist()
