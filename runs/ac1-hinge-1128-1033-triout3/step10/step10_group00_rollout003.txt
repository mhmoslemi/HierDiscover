"""
Non‑negative step‑function optimiser (Adam‑based mirror descent).

The algorithm works on the simplex (Σ a_i = 1, a_i ≥ 0).  It
initialises a smooth “Beta” shape and then runs a large number of
Adam updates on the logarithmic parameters (z_i = log a_i).  The
gradient of the objective L(a)=max_k (a∗a)_k is a sub‑gradient of the
self‑convolution; we take the M largest convolution entries, weight
them proportionally and accumulate their contributions.  The update is
performed in the log‑domain (soft‑max) which automatically respects the
constraints.  A simple stagnation detector injects Gaussian noise
around the current best solution to escape local minima.  The routine
stops just before the time budget expires and reports the best
sequence found via ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution via FFT (real‑valued input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power‑of‑two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum == 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(a∗a)  (the quantity returned by evaluate_sequence for sum(a)=1)."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple Beta‑family (used for a fast deterministic seed)
# ----------------------------------------------------------------------
def _beta_family(n: int, p: float, q: float) -> np.ndarray:
    """Normalised Beta‑shaped sequence of length n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the surrounding framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s
    time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few‑tens of thousands gives a finer discretisation
    # ------------------------------------------------------------------
    n = 12000                     # can be tuned; still fast with FFT

    # ------------------------------------------------------------------
    # 0️⃣  Load a previous solution if it exists (fallback to a Beta seed)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        best_a = a0.copy()
        best_val = _objective(a0)
        print("[init] interpolated previous solution – val={:.6f}".format(best_val))
    except Exception:
        pass

    # If no previous solution, use a symmetric Beta distribution as seed
    if best_a is None:
        a0 = _beta_family(n, p=2.5, q=2.5)   # smooth, centrally‑peaked shape
        best_a = a0.copy()
        best_val = _objective(a0)
        print("[init] Beta(2.5,2.5) seed – val={:.6f}".format(best_val))

    # ------------------------------------------------------------------
    # 1️⃣  Adam optimisation in log‑probability space
    # ------------------------------------------------------------------
    # log‑parameters (soft‑max guarantees positivity and Σa=1)
    z = np.log(best_a + 1e-32)           # avoid log(0)

    # Adam bookkeeping
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # algorithmic knobs
    M_top = 64                           # number of top convolution entries
    stagnation_cnt = 0
    IMPROVE_LIMIT = 3000                # after this many non‑improvements we randomise
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1

        # ----- (a) compute current probability vector and its convolution -----
        a = np.exp(z - np.max(z))        # stability shift
        a /= a.sum()
        conv = _full_convolution(a)
        cur_max = conv.max()
        cur_obj = float(2.0 * n * cur_max)

        # ----- (b) keep track of the global best -----
        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_a = a.copy()
            stagnation_cnt = 0
            print("[iter] new best = {:.6f} at t={:.1f}s".format(
                best_val, now - start_time))
        else:
            stagnation_cnt += 1

        # ----- (c) sub‑gradient: use the M largest convolution entries -----
        # (weights proportional to the entry magnitude)
        top_idx = np.argpartition(-conv, M_top - 1)[:M_top]
        top_vals = conv[top_idx]
        w_sum = top_vals.sum()
        if w_sum == 0.0:
            # fall back to uniform weighting
            weights = np.full(M_top, 1.0 / M_top, dtype=np.float64)
        else:
            weights = top_vals / w_sum

        g_a = np.zeros_like(a)            # gradient w.r.t. a
        for k, wk in zip(top_idx, weights):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            if low > high:
                continue
            j = np.arange(low, high + 1, dtype=np.int64)
            g_a[j] += 2.0 * wk * a[k - j]

        # ----- (d) convert gradient to log‑space (soft‑max Jacobian) -----
        # dL/ dz_i = a_i * (dL/da_i - ⟨dL/da, a⟩)
        dot = np.dot(g_a, a)
        g_z = a * (g_a - dot)

        # ----- (e) Adam update -----
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple linear learning‑rate schedule
        prog = (now - start_time) / (deadline - start_time)
        lr = 0.5 * (1.0 - prog) + 0.001 * prog

        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- (f) occasional random perturbation to escape stagnation -----
        if stagnation_cnt > IMPROVE_LIMIT:
            # perturb around the best solution seen so far
            noise = np.random.normal(scale=0.08, size=n)
            z = np.log(best_a + 1e-32) + noise
            m.fill(0.0)
            v.fill(0.0)
            stagnation_cnt = 0
            print("[reset] perturbation after stagnation – best_val={:.6f}".format(best_val))

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation (uses the original evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
