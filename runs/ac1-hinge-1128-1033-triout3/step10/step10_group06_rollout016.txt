"""
Proposes a low‑peak step function for ``evaluate_sequence``.

Algorithm
---------
1. **Initial seed** – a discrete arcsine law (Beta(0.5,0.5)) that is known
   to give a low autocorrelation peak.
2. **Exponentiated‑gradient descent** – minimise a smooth approximation of
   ``max_k (a*a)_k``.
   * At each iteration the full self‑convolution `b = a∗a` is computed.
   * A soft‑max distribution `p` over the convolution values is built
     (`λ` grows from a small to a large value, turning the smooth
     approximation into the true maximum).
   * The gradient of the smoothed objective w.r.t. `a` is
     `∇ = 2·corr(p, a)`, which is obtained efficiently by a single FFT
     based convolution.
   * An exponentiated‑gradient update `a_i ← a_i·exp(−η·∇_i)` preserves
     non‑negativity; after the update the vector is projected back onto
     the probability simplex.
   * Step‑size `η` decays as `η₀/√t` and a tiny random perturbation is
     added after long periods without improvement to avoid stagnation.
3. **Final polishing** – a tiny convex blend with the uniform distribution
   (α∈{0.001,…,0.020}) is tested; the best of these candidates is returned.

The whole optimisation runs inside the allotted time budget
(`budget_s` seconds, default 1000 s).  All numbers are kept non‑negative,
normalised to sum = 1, and the final sequence is returned as a plain

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – exact, O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two 1‑D real vectors via FFT."""
    n1 = x.shape[0]
    n2 = y.shape[0]
    out_len = n1 + n2 - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[:out_len]
    return conv


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a∗a."""
    return _full_convolution(a, a)

    """Projection of a vector onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (exact, using the same normalisation as evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a), assuming Σ a = 1."""
    n = a.shape[0]
    max_b = _self_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on a smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _optimize_egd(a0: np.ndarray, n: int,
                  start: float, deadline: float,
                  rng: np.random.Generator,
                  target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Minimise ``2·n·max(a∗a)`` by exponentiated gradient descent.
    The soft‑max temperature λ grows from ``λ_start`` to ``λ_end``.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    # hyper‑parameters (tuned for the 1000 s budget)
    eta0 = 0.9                 # initial step‑size factor
    lambda_start = 5.0        # start of soft‑max temperature
    lambda_end = 2000.0       # final temperature – sharpens towards the true max
    max_no_improve = 4000     # after this many idle iterations we perturb

    it = 0
    no_improve = 0

    while time.time() < deadline:
        # ---- progress dependent parameters ---------------------------------
        progress = (time.time() - start) / (deadline - start)
        lam = lambda_start + (lambda_end - lambda_start) * progress
        eta = eta0 / math.sqrt(it + 1.0)

        # ---- current convolution and objective -----------------------------
        b = _self_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        # record best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[EGD {it:6d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- smooth max via soft‑max weights --------------------------------
        # shift by the maximum to avoid overflow
        w = np.exp(lam * (b - max_b))
        p = w / w.sum()                    # probability distribution on 0..2n‑2

        # ---- gradient = 2·corr(p, a) --------------------------------------
        # corr(p, a)[i] = Σ_j a_j · p_{i+j}, i = 0..n‑1
        a_rev = a[::-1]
        conv = _full_convolution(p, a_rev)      # length = 3n‑2
        grad = 2.0 * conv[n - 1: n - 1 + n]      # slice i = 0..n‑1

        # ---- exponentiated‑gradient update ---------------------------------
        a = a * np.exp(-eta * grad)
        # re‑normalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # ---- occasional random jitter to escape flat regions ---------------
        if no_improve >= max_no_improve:
            noise = rng.normal(scale=0.001, size=n)
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            no_improve = 0
            print(f"[EGD {it:6d}] jitter after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing polish (same idea as the reference code)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few convex blends with the uniform distribution."""
    best_a = a.copy()
    best_val = _objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = cand / cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Polish] alpha={alpha:.3f} val={val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function within the given time budget.
    Returns a list of floats (Σ ≈ 1) that minimises ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # a small safety margin

    # problem size – a few thousand entries are a sweet spot
    n = 9000

    # ------------------------------------------------------------------
    # Phase 0 – deterministic arcsine start (a very good seed)
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    init_val = _objective(a0)
    print(f"[Init] arcsine seed value = {init_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent (smooth max → true max)
    # ------------------------------------------------------------------
    a_opt, val_opt = _optimize_egd(a0, n, start_time, deadline, rng,
                                   target_val=1.503)
    print(f"[EGD] best after optimisation = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final tiny blend with the uniform distribution
    # ------------------------------------------------------------------
    a_opt, val_opt = _mix_with_uniform(a_opt, n)
    print(f"[Final] best after uniform mixing = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Log the score as seen by the provided wrapper (purely for information)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_opt.tolist())
        print(f"[final] evaluate_sequence reported = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return a_opt.tolist()
