def _exp_gradient_descent_phase(a0, n, start, deadline, best_a, best_val):
    """Exponentiated gradient descent minimizing max convolution."""
    a = a0.copy()
    log_a = np.log(np.maximum(a, 1e-300))

    t = 0
    eta0 = 0.5
    no_improve = 0
    IMPROVE_LIMIT = 200000
    while time.time() < deadline:
        # compute convex objective
        b = _full_convolution(a)
        cur_val = _objective_value(a)  # 2*n*max(b) as sum=1
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[EG  {t:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step-size schedule
        eta = eta0 / np.sqrt(t + 1)
        log_a -= eta * grad
        # stabilize
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        if no_improve > IMPROVE_LIMIT:
            # random perturbation to escape stagnation
            print(f"[EG  {t:7d}] perturb after stagnation")
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            log_a = np.log(np.maximum(a, 1e-300))
            no_improve = 0

        t += 1

    return best_a, best_val

def _deterministic_refine2(a0, n, start, deadline, best_val):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = _objective_value(a)
    best_local_a = a.copy()
    best_local_val = cur_val
    it = 0
    while time.time() < deadline:
        k_max = int(np.argmax(b))
        # compute contributions
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        # source index i with largest contribution
        i = int(np.argmax(contrib))
        if a[i] <= 0:
            break
        # get a_{k-i}
        a_ki = a[k_max - i] if 0 <= k_max - i < n else 0.0

        # second max (excluding current max)
        sec_max = np.max(np.where(np.arange(b.size) == k_max, -np.inf, b))

        # candidate targets: smallest a entries
        j_candidates = np.argsort(a)[:10]  # top 10 smallest
        best_delta = 0.0
        best_j = None
        best_new_val = cur_val
        for j in j_candidates:
            if j == i:
                continue
            a_kj = a[k_max - j] if 0 <= k_max - j < n else 0.0
            diff = a_ki - a_kj
            if diff <= 0:
                continue
            denom = 2.0 * diff
            delta_needed = (b[k_max] - sec_max) / denom
            delta = min(delta_needed * 0.9, a[i])
            if delta <= 1e-12:
                continue
            # Simulate new max value after moving delta from i to j
            # Approximate new max:
            new_val_est = (b[k_max] - 2 * delta * diff)  # ignore square terms
            if new_val_est < best_new_val:
                best_new_val = new_val_est
                best_delta = delta
                best_j = j

        if best_j is None:
            # no beneficial move, break
            break

        # apply the move exactly with incremental update
        j = best_j
        delta = best_delta
        # incremental update on convolution
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = _objective_value(a)
        cur_val = new_val
        if new_val < best_local_val - 1e-14:
            best_local_val, best_local_a = new_val, a.copy()
            if best_local_val <= 1.503:
                break

        it += 1
        if it % 2000 == 0:
            print(f"[Refine2 {it:7d}] best = {best_local_val:.6f}")

    return best_local_a, best_local_val

"""
Propose a step‑function (non‑negative list) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

Strategy
---------
1. Build many hand‑crafted candidates (uniform, sine‑powers,
   exponential tails, Beta‑shapes, triangular, quadratic, random Dirichlet)
   and keep the best one (fast, O(n log n) per candidate).

2. Coarse optimisation – Adam on a smooth log‑sum‑exp surrogate of the
   max‑convolution (Phase A).  This drives the solution into a good basin.

3. Fine optimisation – Exponentiated‑gradient descent on the exact
   sub‑gradient of the maximal convolution entry (Phase B).  This works
   directly on the simplex and can keep reducing the peak.

4. Greedy polishing – deterministic moves that analytically lower the
   current maximal convolution entry.  For each iteration a source index
   with the largest contribution to the peak is chosen, and the best
   target among the smallest entries is selected (Phase C).

5. Tiny uniform mixing (often helpful for the final evaluation).

All phases respect the 1000 s budget and continuously keep the best
vector found so far.  The final list is returned directly; the harness
will call ``evaluate_sequence`` on it.

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length = 2·len(a)-1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial candidates – a richer suite of families + Dirichlet draws
# ----------------------------------------------------------------------
def _initial_candidate_extended(n: int):
    """Return the best hand‑crafted start vector (sum = 1)."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(candidate):
        nonlocal best_a, best_val
        a = np.asarray(candidate, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # ------------------------------------------------------------------
    # 1. Uniform
    # ------------------------------------------------------------------
    _try(np.ones(n))

    # ------------------------------------------------------------------
    # 2. Sine families  (a_i = sin^p)
    # ------------------------------------------------------------------
    xs = np.arange(1, n + 1) / (n + 1)
    base = np.sin(np.pi * xs)
    for p in (0.5, 1.0, 2.0, 3.0):
        _try(base ** p)

    # ------------------------------------------------------------------
    # 3. Exponential tails (double‑ended)
    # ------------------------------------------------------------------
    for r in (0.9999, 0.9995, 0.998, 0.996, 0.99, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # ------------------------------------------------------------------
    # 4. Beta‑shapes (symmetric)  a_i ∝ x^{α‑1}(1‑x)^{α‑1}
    # ------------------------------------------------------------------
    x = (np.arange(n) + 0.5) / n
    for alpha in (0.2, 0.3, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0):
        a = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
        _try(a)

    # ------------------------------------------------------------------
    # 5. Triangular (peak in centre)
    # ------------------------------------------------------------------
    tri = 1.0 - np.abs(np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)
    _try(tri)

    # ------------------------------------------------------------------
    # 6. Quadratic (parabolic) shape
    # ------------------------------------------------------------------
    quad = 1.0 - ((np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)) ** 2
    quad = np.clip(quad, 0, None)
    _try(quad)

    # ------------------------------------------------------------------
    # 7. Gaussian centred shape
    # ------------------------------------------------------------------
    mid = (n - 1) / 2.0
    for sigma in (n / 6.0, n / 4.0, n / 8.0):
        a = np.exp(-0.5 * ((np.arange(n) - mid) / sigma) ** 2)
        _try(a)

    # ------------------------------------------------------------------
    # 8. Random Dirichlet draws
    # ------------------------------------------------------------------
    rng = np.random.default_rng(12345)
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth (log‑sum‑exp) surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp surrogate of max(conv)."""
    w = np.log(a0 + 1e-12)          # logits
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.005
    eps_start, eps_end = 0.2, 1e-12
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 50000
    it = 0
    while time.time() < deadline:
        # simplex point
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # ---------- smooth max ----------
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # ---------- gradient w.r.t. a ----------
        p_rev = p[::-1]
        conv_ab = np.fft.irfft(np.fft.rfft(a) * np.fft.rfft(p_rev))
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter and restart
        if np.random.rand() < 0.007:
            w += np.random.normal(scale=0.008, size=n)
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_descent_phase(a0: np.ndarray, n: int,
                                start: float, deadline: float,
                                best_a: np.ndarray, best_val: float,
                                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex with the entropy mirror map.
    Uses the exact sub‑gradient of the maximal convolution entry.
    """
    a = a0.copy()
    log_a = np.log(np.maximum(a, 1e-300))   # keep a strictly >0 for log
    t = 0
    eta0 = 0.5
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        cur_val = _objective_value(a)
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[EG  {t:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # sub‑gradient of the current maximal entry
        b = _full_convolution(a)
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step size schedule
        eta = eta0 / np.sqrt(t + 1.0)
        log_a -= eta * grad
        # normalise in log‑space for numerical stability
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if no_improve > IMPROVE_LIMIT:
            print(f"[EG  {t:7d}] perturb after stagnation")
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            log_a = np.log(np.maximum(a, 1e-300))
            no_improve = 0

        t += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Deterministic greedy refinement (enhanced candidate selection)
# ----------------------------------------------------------------------
def _deterministic_refine2(a0: np.ndarray, n: int,
                          start: float, deadline: float,
                          best_val: float) -> tuple[np.ndarray, float]:
    """
    Greedy analytical moves.  For the current peak index we look at the
    source with the largest contribution and test several low‑mass
    targets, picking the move that most reduces the peak.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = _objective_value(a)
    best_local_a = a.copy()
    best_local_val = cur_val
    it = 0

    while time.time() < deadline:
        k_max = int(np.argmax(b))

        # contribution of each i to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        i = int(np.argmax(contrib))
        if a[i] <= 0.0:
            break

        a_ki = a[k_max - i] if 0 <= k_max - i < n else 0.0

        # second‑largest convolution entry (excluding the current max)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        if second_max == -np.inf:
            second_max = 0.0

        # candidate targets = a‑indices with smallest mass (try a few)
        j_candidates = np.argsort(a)[:12]          # 12 smallest entries
        best_delta = 0.0
        best_j = None
        best_new_peak = cur_val

        for j in j_candidates:
            if j == i:
                continue
            a_kj = a[k_max - j] if 0 <= k_max - j < n else 0.0
            diff = a_ki - a_kj
            if diff <= 0.0:
                continue
            denom = 2.0 * diff + 1e-12
            delta_needed = (b[k_max] - second_max) / denom
            delta = min(delta_needed * 0.9, a[i])
            if delta <= 1e-12:
                continue
            # cheap estimate of the new peak after the move
            new_peak_est = b[k_max] - 2.0 * delta * diff
            if new_peak_est < best_new_peak:
                best_new_peak = new_peak_est
                best_delta = delta
                best_j = j

        if best_j is None:
            # no profitable move found – stop
            break

        j = best_j
        delta = best_delta

        # ---- apply the move (exact convolution update) ----
        # update convolution (linear time, using the old a)
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        L = len(b)
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        if two_j < L:
            b[two_j] += delta * delta
        if two_i < L:
            b[two_i] += delta * delta
        if ij < L:
            b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = _objective_value(a)
        cur_val = new_val
        if new_val < best_local_val - 1e-14:
            best_local_val, best_local_a = new_val, a.copy()
            if best_local_val <= 1.503:
                break

        it += 1
        if it % 2000 == 0:
            print(f"[Refine2 {it:7d}] best = {best_local_val:.6f}")

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: mix a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a little uniform mass (often helps)."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a step‑function under the given time budget.
    Returns a list of non‑negative floats that approximately minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    # Choose a problem size with a good trade‑off between expressiveness
    # and computational cost.
    n = 9000

    # --------------------------------------------------------------
    # Initialise – use the previously found construction if available,
    # otherwise fall back to an extensive hand‑crafted search.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    except Exception:
        a0, _ = _initial_candidate_extended(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * (deadline - start_time)
    best_a, best_val = _adam_phase(best_a, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase B – Exponentiated‑gradient descent (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.75 * (deadline - start_time)
    best_a, best_val = _exp_gradient_descent_phase(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase C – Deterministic greedy refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _deterministic_refine2(
        best_a, n,
        start=phase_b_deadline,
        deadline=deadline,
        best_val=best_val
    )

    # ------------------------------------------------------------------
    # Final tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call ``evaluate_sequence``)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
