"""
Propose a non‑negative step function that (approximately) minimises

        2·n·max(conv(a,a)) / (sum(a)²)

using a smoothed‑max Adam optimiser with continuation on the smoothing
parameter β.  The method works on the probability simplex (∑a = 1) and
periodically mixes the current best candidate with a tiny uniform
distribution to escape shallow plateaus.  Multiple cheap initialisations
are tried (exponential, power‑law, cosine‑window, Dirichlet draws and
simple monotone ramps).  The whole search respects the 1000 s budget,
runs on a single CPU and uses only NumPy (no external files).  The
final candidate is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast for the chosen n)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _smooth_max_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (b, grad) where
        b = a*a (full convolution)
        grad = ∇_{a} (smooth max_{k} b_k) using soft‑max weights with parameter β.
    The gradient formula is:
        grad_i = 2· Σ_j w_{i+j}·a_j,
    where w_k = exp(β·b_k) / Σ_l exp(β·b_l).
    """
    n = a.shape[0]

    # --- b = a*a --------------------------------------------------------
    b = _full_convolution(a)

    # --- soft‑max weights ------------------------------------------------
    max_b = b.max()
    # stability: subtract the max before exponentiation
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:        # numerical safeguard
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # --- gradient = 2·(w ⋆ a_rev) (correlation) -------------------------
    a_rev = a[::-1]
    # length of w⋆a_rev = len(w) + len(a_rev) - 1 = (2n‑1) + n‑1 = 3n‑2
    out_len = len(w) + len(a_rev) - 1
    size = 1 << ((out_len - 1).bit_length())
    fw = np.fft.rfft(w, size)
    fav = np.fft.rfft(a_rev, size)
    corr_full = np.fft.irfft(fw * fav, size)
    start = n - 1                         # slice that yields n entries
    grad = 2.0 * corr_full[start:start + n]
    return b, grad


def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())

# Small helper families for initial candidates
# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator):
    """Yield a collection of cheap, diverse probability vectors of length n."""
    # 1. Uniform
    yield np.full(n, 1.0 / n)

    # 2. Pure exponential decay (asymmetric)
    for r in np.linspace(0.90, 0.9999, 12):
        a = np.power(r, np.arange(n, dtype=np.float64))
        a /= a.sum()
        yield a

    # 3. Power‑law decay
    for p in np.linspace(0.5, 3.0, 12):
        a = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        a /= a.sum()
        yield a

    # 4. Cosine‑window (Hann) with shape exponent β
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    for beta in np.linspace(0.5, 4.0, 12):
        a = np.power(w, beta)
        a /= a.sum()
        yield a

    # 5. Simple monotone ramps
    a = np.linspace(1.0, 0.0, n)
    a /= a.sum()
    yield a
    a = np.linspace(0.0, 1.0, n)
    a /= a.sum()
    yield a

    # 6. Random Dirichlet draws (few samples)
    for _ in range(25):
        a = rng.dirichlet(np.ones(n))
        yield a


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed‑max objective (simple continuation on β)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   beta_start: float,
                   beta_factor: float,
                   beta_max: float,
                   lr: float,
                   max_iters: int,
                   rng: np.random.Generator,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max of the convolution.
    Returns the best vector seen (projected onto the simplex) and its exact
    objective value.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_value(best_a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    beta = beta_start
    for it in range(1, max_iters + 1):
        if time.time() > deadline - 0.2:
            break

        # gradient of the smoothed max
        _, grad = _smooth_max_gradient(a, beta)

        # Adam update (minimisation → step opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step

        # projection onto the probability simplex (non‑negative, sum = 1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # occasional random jitter to escape flat regions
        if rng.random() < 0.004:
            a += rng.normal(scale=0.001, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # evaluate the exact (non‑smooth) objective occasionally
        if it % 50 == 0:
            cur_val = _objective_value(a)
            if cur_val < best_val - 1e-14:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam] iter {it:6d}  β={beta:.2f}  best={best_val:.6f}")

        # continuation on β (gradually sharpen the max)
        beta = min(beta * beta_factor, beta_max)

    # final check
    final_val = _objective_value(a)
    if final_val < best_val:
        best_val, best_a = final_val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (helps to break ties)
# ----------------------------------------------------------------------
def _tiny_mixing(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Mix (1‑α)·a + α·uniform for a few α and keep the best result."""
    n = a.shape[0]
    best_a = a
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.006, 0.01, 0.015, 0.02):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] Re‑used height_sequence_1")
    except Exception:
        # cheap parametric sweep
        best_a = None
        best_val = np.inf
        for cand in _initial_candidates(n, rng):
            val = _objective_value(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
        a0 = best_a
        print(f"[init] Parametric sweep best ≈ {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation with β‑continuation.
    # --------------------------------------------------------------
    # hyper‑parameters (chosen to work well for n≈9000)
    beta_start = 1.0          # start with a very smooth approximation
    beta_factor = 1.08        # geometric increase each iteration
    beta_max = 300.0          # final sharpening (approaches true max)
    lr = 0.12                 # Adam base learning rate
    max_iters_per_run = 18000

    # Run a few independent restarts (each consumes a slice of the budget)
    best_overall_a = a0
    best_overall_val = _objective_value(a0)

    # time budgeting helpers
    remaining = deadline - time.time()
    # Allocate roughly equal time to three restarts; adapt if we run out.
    while time.time() < deadline - 0.5:
        # limit the number of iterations by the remaining time
        max_iters = max_iters_per_run
        # run optimisation from the current best seed (adds jitter each restart)
        a_restart = best_overall_a + rng.normal(scale=0.001, size=n)
        a_restart = np.maximum(a_restart, 0.0)
        a_restart /= a_restart.sum()

        a_opt, val_opt = _adam_optimize(
            a_restart,
            beta_start=beta_start,
            beta_factor=beta_factor,
            beta_max=beta_max,
            lr=lr,
            max_iters=max_iters,
            rng=rng,
            deadline=deadline,
        )
        if val_opt < best_overall_val:
            best_overall_val, best_overall_a = val_opt, a_opt.copy()
            print(f"[restart] new global best = {best_overall_val:.6f}")

        # If the remaining time is too small we break
        if time.time() > deadline - 5.0:
            break

    # --------------------------------------------------------------
    # Phase 2 – Tiny uniform mixing (often gives a final small win)
    # --------------------------------------------------------------
    best_overall_a, best_overall_val = _tiny_mixing(best_overall_a)

    # --------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_overall_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_overall_a.tolist()
