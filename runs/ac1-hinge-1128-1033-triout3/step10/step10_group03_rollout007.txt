"""
Propose a candidate step‑function minimizing ``evaluate_sequence``.

The method proceeds in three phases:

1. Initialise from the stored solution (if any).  Fallback to a
   strongly “U‑shaped’’ Beta distribution.
2. Run a short random search (Dirichlet draws) to obtain a decent
   starting point.
3. Perform a long projected heavy‑ball optimisation on a smooth
   approximation of  ``max(conv(a,a))``.
   The smooth approximation uses a soft‑max (temperature α that grows
   from 0.5 to 80).  Each iteration:
        * compute b = a∗a,
        * weight the lags w_i = softmax(α·b_i),
        * gradient ≈ 2·conv(w,a) (first n entries),
        * heavy‑ball update,
        * Euclidean projection onto the probability simplex.
   Occasional jitter and a tiny learning‑rate schedule help escape
   plateaus.  The best vector encountered is finally evaluated with
   ``evaluate_sequence`` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` and `y` using real FFT."""
    out_len = len(x) + len(y) - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Convenient wrapper for self‑convolution."""
    return _convolve_fft(x, x)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex Σx=1, x≥0. """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the total mass is zero.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a little margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise from a previously saved solution, else use a U‑shape
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        print("[init] interpolated previous solution")
    except Exception:
        # A strongly U‑shaped beta distribution (p,q<1) tends to have a low max‑lag.
        a = _beta_shape(0.2, 0.2, n)
        if a is None:
            a = np.full(n, 1.0 / n)
        print("[init] fallback U‑shape")

    # ------------------------------------------------------------------
    # 1️⃣  Quick random Dirichlet exploration (first ≈30 s)
    # ------------------------------------------------------------------
    rand_deadline = start_time + min(30.0, budget_s * 0.03)
    best_a = a.copy()
    best_val = float(2 * n * _full_convolution(best_a).max())
    print(f"[init] initial value = {best_val:.6f}")

    while time.time() < rand_deadline and time.time() < deadline:
        cand = np.random.dirichlet(np.ones(n))
        cur = float(2 * n * _full_convolution(cand).max())
        if cur < best_val:
            best_val, best_a = cur, cand.copy()
            print(f"[rand] new best = {best_val:.6f}")

    a = best_a.copy()   # start optimisation from the best found so far

    # ------------------------------------------------------------------
    # 2️⃣  Heavy‑ball optimisation on a smoothed max‑convolution objective
    # ------------------------------------------------------------------
    # Momentum term (heavy‑ball)
    momentum = 0.9
    v = np.zeros_like(a)

    # Learning‑rate schedule (linear decay)
    step_start = 0.5
    step_end = 1e-5

    # Soft‑max temperature schedule
    alpha_start = 0.5
    alpha_end = 80.0

    # Small random jitter to escape plateaus
    jitter_prob = 0.001
    jitter_scale = 0.001

    iteration = 0
    last_improve_iter = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # progress in [0,1]
        progress = (now - start_time) / (deadline - start_time)

        # ---------- compute objective and possibly update best ----------
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            last_improve_iter = iteration
            print(f"[opt] new best = {best_val:.6f}  iter={iteration}")
            if best_val <= 1.503:
                print("[opt] target reached")
                break

        # ---------- smoothed gradient ----------
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        scaled = alpha * b
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)      # stabilise
        w = exp_scaled / exp_scaled.sum()            # soft‑max weights, sum=1

        # gradient = 2 * conv(w, a) (first n entries)
        grad_full = _convolve_fft(w, a)
        grad = 2.0 * grad_full[:n]

        # ---------- step‑size ----------
        step = step_start * (1.0 - progress) + step_end * progress

        # ---------- heavy‑ball update ----------
        v = momentum * v - step * grad
        a_candidate = a + v

        # occasional jitter
        if np.random.rand() < jitter_prob:
            a_candidate += np.random.normal(scale=jitter_scale, size=n)

        # project back onto the probability simplex
        a = _project_simplex(a_candidate)

        iteration += 1

        # safety break if we have been idle for a very long time
        if iteration - last_improve_iter > 50000 and progress > 0.9:
            # no improvement for many iterations near the end – stop early
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    # Return the best sequence we have (as plain Python list)
    return best_a.tolist()

def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` and `y` using a real FFT."""
    out_len = len(x) + len(y) - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the self‑convolution x∗x."""
    return _convolve_fft(x, x)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """
    Return a normalised Beta‑shaped vector:
        a_i ∝ i^{p‑1} (n‑i+1)^{q‑1},   i = 1,…,n .
    Returns None if the total mass is zero.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a
