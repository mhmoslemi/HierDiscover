def conv_matrix(f):
    n = f.shape[0]
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    return A
```

Now define function `solve_lp_for_g(f, M)`:

```
def solve_lp_for_g(f, M):
    n = f.shape[0]
    A = conv_matrix(f)   # shape (2n-1, n)
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # minimize -sum(g)

    # Use bounds (0, None)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # maybe return f as fallback
        return f.copy()
    g = res.x
    return g
```

def _initial_candidate(n):
    # generate various families and return best
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def consider(arr):
        nonlocal best_a, best_val
        a = np.asarray(arr, dtype=np.float64)
        if a.shape != (n,):
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # Exponential tails
    for r in np.linspace(0.995, 0.999999, 12):
        consider(r ** i)
        consider(r ** i + r ** (n - 1 - i))

    # Power law
    for p in np.linspace(0.1, 3.0, 15):
        consider(i1 ** (-p))
        consider(i1 ** (-p) + (n - i) ** (-p))

    # Beta
    beta_vals = np.linspace(0.2, 1.2, 10)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            consider(a)

    # Arcsine
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    consider(arcs)

    # Sine power
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 8):
        consider(base_sin ** p)

    # Piecewise constant high edges
    for k_frac in np.linspace(0.005, 0.08, 8):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.08, 0.3, 6):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            consider(a)

    # Random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(60):
        consider(rng.dirichlet(np.ones(n)))

    # Interpolate previous solution if defined
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        consider(a0)
    except Exception:
        pass

    return best_a, best_val

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    # Round small negatives to zero for numeric stability
    conv[conv < 0] = 0.0
    return conv

"""
Propose a step‑function (non‑negative list) that minimizes
    2·n·max(self‑convolution) / (sum a)²

The algorithm proceeds in four stages:

1. Broad parametric sweep  – generate many candidate shapes
   (exponential, power‑law, beta, arcsine, sine‑power, piecewise‑edge,
   random Dirichlet).  The best is kept as the seed.

2. Gradient‑descent refinement – repeatedly move probability mass
   away from indices that contribute most to the current maximal
   convolution bin.  The direction is the sub‑gradient of the L∞‑norm
   of the self‑convolution; a line‑search with projection onto the
   simplex yields monotone improvement.

3. Simulated‑annealing (optional) – a lightweight mass‑transfer SA
   helps escape shallow local minima if the gradient phase stalls.

4. Tiny uniform mixing – final convex combination with the uniform
   distribution often yields a modest extra drop.

All steps respect the required interface and run within the given
time budget (default 1000 s).  The code uses only NumPy and SciPy
(and the already‑imported evaluate_sequence) and prints progress
statistics.

# ----------------------------------------------------------------------
# Convolution utilities (FFT – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    # eliminate possible tiny negative artefacts
    conv[conv < 0] = 0.0
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a normalized vector (Σ a = 1)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Projection onto the probability simplex Σ w = 1, w≥0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Search a rich collection of parametric families for a low‑objective seed."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _consider(arr):
        nonlocal best_a, best_val
        a = np.asarray(arr, dtype=np.float64)
        if a.shape != (n,):
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a /= s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)            # 0 … n‑1
    i1 = i + 1.0                              # 1 … n

    # 1. Exponential tails (left‑hand and symmetric)
    for r in np.linspace(0.995, 0.999999, 12):
        _consider(r ** i)                     # decreasing
        _consider(r ** i + r ** (n - 1 - i))  # symmetric U‑shape

    # 2. Power‑law tails (left‑hand and symmetric)
    for p in np.linspace(0.1, 3.0, 15):
        _consider(i1 ** (-p))
        _consider(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta family (covers many shapes)
    beta_vals = np.linspace(0.2, 1.2, 9)
    for a_par in beta_vals:
        for b_par in beta_vals:
            a = (i1 ** (a_par - 1.0)) * ((n - i) ** (b_par - 1.0))
            _consider(a)

    # 4. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _consider(arcs)

    # 5. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 8):
        _consider(base_sin ** p)

    # 6. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.005, 0.08, 8):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.08, 0.3, 6):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _consider(a)

    # 7. Random Dirichlet draws for diversity
    rng = np.random.default_rng(123456)
    for _ in range(60):
        _consider(rng.dirichlet(np.ones(n)))

# ----------------------------------------------------------------------
# Phase 1 – gradient‑descent refinement (sub‑gradient of max‑convolution)
# ----------------------------------------------------------------------
def _gradient_refine(a0: np.ndarray, n: int, start: float, deadline: float):
    """
    Repeatedly move mass away from indices that heavily contribute to
    the current maximal convolution bin.  Each trial is projected onto
    the simplex; a simple back‑tracking line search guarantees monotone
    improvement.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_value(a)
    cur_val = best_val
    it = 0
    stagnation = 0

    while time.time() < deadline:
        it += 1

        # ---- current convolution and active indices ----
        b = _full_convolution(a)
        M = b.max()
        eps = 1e-12 * M
        K = np.where(b >= M - eps)[0]       # all nearly‑maximal bins

        # ---- sub‑gradient of the max‑convolution ----
        grad = np.zeros_like(a)
        for k in K:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad[idx] += a[k - idx]

        # ---- line search for a beneficial step ----
        step = 0.5
        improved = False
        for _ in range(10):
            cand = a - step * grad
            cand = _project_simplex(cand)
            cand_val = _objective_value(cand)

            if cand_val + 1e-14 < cur_val:           # strict improvement
                a = cand
                cur_val = cand_val
                improved = True
                break
            elif cand_val < cur_val:                 # non‑strict (numerical)
                a = cand
                cur_val = cand_val
                improved = True
                break
            else:
                step *= 0.5

        if improved:
            stagnation = 0
            if cur_val + 1e-15 < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[GD {it:6d}] new best = {best_val:.6f}")
        else:
            stagnation += 1
            # occasional random perturbation to escape plateaus
            if stagnation >= 2000:
                noise = np.random.normal(scale=0.01, size=n)
                a = a + noise
                a = np.maximum(a, 0.0)
                a = _project_simplex(a)
                cur_val = _objective_value(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[GD {it:6d}] after noise best = {best_val:.6f}")
                stagnation = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – light simulated‑annealing mass‑transfer (fallback)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Small‑scale SA that moves a fraction of probability mass from a
    high‑contributing source index to a low‑contributing target.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = best_a.copy()
    best_local_val = best_val

    max_frac_start, max_frac_end = 0.6, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T0 = 0.02 * cur_val
        T = max(T0 * (1.0 - progress), 1e-12)

        # index of the current maximal convolution bin
        k_max = int(np.argmax(b))

        # ---- choose a source index i (large contribution) ----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # ---- choose a target index j (small contribution) ----
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0.0:
            probs_j = w / w.sum()
            j = int(np.random.choice(n, p=probs_j))
        else:
            j = (i + np.random.randint(1, n)) % n

        if j == i:
            continue

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the full convolution -----
        b_new = b.copy()
        # removal from i
        b_new[i:i + n] -= 2.0 * delta * a
        # addition at j
        b_new[j:j + n] += 2.0 * delta * a
        # corrections for the diagonal terms
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_local_val - 1e-15:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        # occasional re‑normalisation
        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        # restart after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_local_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val

# Phase 3 – final uniform mixing polish
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a tiny share of the uniform distribution."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val
