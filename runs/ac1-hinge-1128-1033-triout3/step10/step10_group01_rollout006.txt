import time

def _full_convolution(x):
    n = x.shape[0]
    out_len = 2*n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _conv_rev(a, w):
    # a_rev = a[::-1], conv_rev = a_rev * w
    n = a.shape[0]
    a_rev = a[::-1]
    l_w = w.shape[0]
    out_len = n + l_w - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    # slice n-1 to n-1+n
    start = n - 1
    return conv[start:start + n]

def _objective_from_conv(conv, n):
    return float(2.0 * n * conv.max())

def _objective(a):
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])

def _project_simplex(v):
    # Euclidean projection onto simplex sum=1, v>=0
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_softmax(a, alpha):
    # a shape (n,)
    n = a.shape[0]
    # compute full convolution a*a
    conv = _full_convolution(a)
    # compute softmax weights
    maxc = conv.max()
    # stabilize:
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    # compute gradient
    grad = 2.0 * _conv_rev(a, w)
    return grad, conv

"""
Smooth‑max Adam optimiser for the step‑function attack.

The pipeline:
 1️⃣  Create a diverse set of initial sequences (Gaussian / Laplace /
     power‑law families, Dirichlet draws, interpolation of the previous
     best solution if available).
 2️⃣  Optimise a *soft‑max surrogate* of the max‑convolution using
     Adam with Euclidean projection onto the probability simplex.
     The surrogate
         f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)_k)
     approaches the true objective as α→∞, while remaining smooth for
     moderate α.  α is increased linearly during the run.
 3️⃣  A short hill‑climbing phase (mass‑transfer) polishes the best
     candidate.
 4️⃣  The final sequence is returned; the host will evaluate it with
     ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_rev(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·∑_i a_i·w_{i+j} for j=0..n‑1.
    Implements the gradient of the soft‑max surrogate.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    l_w = w.shape[0]
    out_len = n + l_w - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    start = n - 1
    return conv[start:start + n]        # length n


# ----------------------------------------------------------------------
# Objective helpers (exact, scale‑invariant)
# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact value 2·n·max(a∗a) given the full convolution."""
    return float(2.0 * n * conv.max())


def _objective(a: np.ndarray) -> float:
    """Exact objective for a (assumed non‑negative, sum>0)."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.shape[0])


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, Σx = 1} (Euclidean)."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max surrogate gradient
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Computes normalized soft‑max weights w_k ∝ exp(α·conv_k)."""
    maxc = conv.max()
    # shift for numerical stability
    exp_vals = np.exp(alpha * (conv - maxc))
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0:
        # fallback – uniform weights
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / sum_exp


# ----------------------------------------------------------------------
# Adam‑style optimiser on the soft‑max surrogate
# ----------------------------------------------------------------------
def _smooth_adam_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          start_time: float,
                          alpha_start: float = 5.0,
                          alpha_end: float = 200.0,
                          lr0: float = 0.1,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser of the smooth objective.
    Returns the best sequence found (sum=1) and its exact objective.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_conv = _full_convolution(a)
    cur_val = _objective_from_conv(cur_conv, n)
    best_a = a.copy()
    best_val = cur_val

    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        alpha = alpha_start + (alpha_end - alpha_start) * prog

        # soft‑max weights and gradient
        w = _softmax_weights(cur_conv, alpha)          # shape (2n‑1,)
        grad = 2.0 * _conv_rev(a, w)                    # shape (n,)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 / (np.sqrt(v_hat) + eps)

        a_candidate = a - step * m_hat
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = _project_simplex(a_candidate)

        cand_conv = _full_convolution(a_candidate)
        cand_val = _objective_from_conv(cand_conv, n)

        # accept only if we do not increase the exact objective
        if cand_val <= cur_val:
            a = a_candidate
            cur_conv = cand_conv
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[Adam] iter {t:7d} α={alpha:6.1f} val={best_val:.6f}")
                if best_val <= 1.503:
                    break
        else:
            # no improvement – keep the old point
            # (momentum vectors are left untouched so the direction continues)
            pass

        # occasional random perturbation to escape plateaus
        if t % 5000 == 0:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            cur_conv = _full_convolution(a)
            cur_val = _objective_from_conv(cur_conv, n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Small parametric families for initialisation
# ----------------------------------------------------------------------
def _parametric_candidates(n: int):
    """Yield a handful of hand‑crafted shapes (Gaussian, Laplace, power)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Gaussian families
    for sigma_factor in (0.18, 0.22, 0.26, 0.30):
        sigma = sigma_factor * n
        a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Laplace (double‑exponential) families
    for scale_factor in (0.12, 0.16, 0.20):
        scale = scale_factor * n
        a = np.exp(-np.abs(i - mid) / scale)
        if a.sum() > 0:
            a /= a.sum()
            yield a

    # Power‑law / Beta‑like families
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = (i + 1) ** (p - 1.0) * (n - i) ** (p - 1.0)
        if a.sum() > 0:
            a /= a.sum()
            yield a


# ----------------------------------------------------------------------
# Interpolation of a previously found sequence (if present)
# ----------------------------------------------------------------------
def _interpolate_previous(prev_seq, n):
    try:
        old = np.asarray(prev_seq, dtype=float)
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
        return a0
    except Exception:
        return None


# ----------------------------------------------------------------------
# Light‑weight hill‑climbing (mass‑transfer) for final polishing
# ----------------------------------------------------------------------
def _final_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Very simple hill‑climbing that moves a small amount of mass from the
    most “guilty’’ index (largest contribution to the current max lag) to a
    low‑contribution index.  Stops early if the target value is reached.
    """
    b = _full_convolution(a)
    cur_val = _objective_from_conv(b, n)
    best_val = cur_val
    best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # locate the lag giving the maximal autocorrelation
        k_max = int(np.argmax(b))

        # indices that participate in this lag
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        # contribution of each position to the offending lag
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source: preferably a heavy contributor
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(contrib))

        if a[src] < 1e-12:
            continue

        # target: a location that is *not* a heavy contributor
        inv = contrib.max() - contrib + 1e-12
        inv[src] = 0.0
        tgt = np.random.choice(n, p=inv / inv.sum())

        # move a small random fraction of the source mass
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        b_new = _full_convolution(a_new)
        new_val = _objective_from_conv(b_new, n)

        if new_val < cur_val:
            # accept move
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] iter {iteration:7d} val={best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # random jitter around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = _objective_from_conv(b, n)
            no_improve = 0

        iteration += 1

    return best_a, best_val
