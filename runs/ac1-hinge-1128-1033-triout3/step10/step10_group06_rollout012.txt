def gradient_descent(a_init, n, start, deadline, best_val, best_a):

    a = a_init.copy()
    cur_val = best_val
    best_local_a = a.copy()
    best_local_val = cur_val

    # schedule for p values (large p approximates max)
    p_vals = [2,4,8,16,32,64,128]  # maybe final 256.

    for p in p_vals:
        if time.time() >= deadline:
            break
        # compute weight exponent for smoothing.
        # We'll use power smoothing: w = b**(p-1)
        # For large p, large numbers maybe overflow; scale by max.
        # We'll compute in log domain to avoid overflow:
        # w = np.exp((p-1)*(np.log(b) - np.max(np.log(b)))) -> normalized later.
        # Because b values may be very small; log(b) negative large; exponent may underflow.
        # Better to compute w = b**(p-1) - but if b tiny, w might underflow to 0; okay if we add epsilon to b.
        # We'll simply compute w = (b + 1e-30)**(p-1) then normalize.
        # Could use np.power.

        # iteration loop
        it = 0
        while time.time() < deadline:
            # compute conv
            b = _full_convolution(a)
            max_b = b.max()
            cur_val = 2.0 * n * max_b
            # update best as needed
            if cur_val < best_local_val - 1e-12:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[GD p={p}] new best local {best_local_val:.6f} it {it}")
                if best_local_val < best_val:
                    best_val = best_local_val
                    best_a = best_local_a.copy()
                    print(f"[GD] global best updated {best_val:.6f}")

            # compute smoothing weights
            if p <= 64:
                # use power smoothing
                # add epsilon to avoid zero
                w_unnorm = np.power(b + 1e-30, p-1)
            else:
                # use exponential smoothing (log-sum-exp) with eps coefficient
                eps = 0.01 * p  # so eps growing
                # compute scaled values to avoid overflow
                t = eps * (b - b.max())
                w_unnorm = np.exp(t)
            # Normalize
            w_sum = w_unnorm.sum()
            if w_sum == 0:
                break
            w = w_unnorm / w_sum

            # compute gradient g_i = 2 * sum_k w_k * a_{k-i}
            # Compute convolution of a with reversed w
            conv_w_rev = np.convolve(a, w[::-1])
            g = 2.0 * conv_w_rev[n-1:2*n-1][::-1]  # shape n

            # Determine step size candidates
            # Base step size choose scaling factor based on typical magnitude
            # We'll try a few step sizes decreasing as p increases
            step_factors = [0.5, 1.0, 2.0, 3.0]  # multiply with some base
            # Compute base step based on 1/(norm(g)+1e-12)
            base = 0.01 / (np.linalg.norm(g) + 1e-12)  # small

            improved = False
            for factor in step_factors:
                eta = base * factor
                cand = a - eta * g
                # Project onto simplex
                cand = _project_simplex(cand)
                # compute objective
                b_cand = _full_convolution(cand)
                cand_val = 2.0 * n * b_cand.max()
                if cand_val < cur_val - 1e-12:
                    # accept
                    a = cand
                    cur_val = cand_val
                    improved = True
                    break
            if not improved:
                # If no improvement with simple step, maybe reduce eta further
                # try smaller steps
                for factor in [0.1, 0.05, 0.01]:
                    eta = base * factor
                    cand = a - eta * g
                    cand = _project_simplex(cand)
                    b_cand = _full_convolution(cand)
                    cand_val = 2.0 * n * b_cand.max()
                    if cand_val < cur_val - 1e-12:
                        a = cand
                        cur_val = cand_val
                        improved = True
                        break
                if not improved:
                    # if still no improvement, break inner loop for this p
                    break
            it += 1

    return best_local_a, best_local_val

        best_a, best_val = _simulated_annealing(best_a, n,

    # final evaluate
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()

"""
Propose a non‑negative step‑function (probability vector) that lowers the
objective 2·n·max(conv(a,a))/ (∑a)².  The algorithm consists of three phases:

1.  A broad parametric sweep (exponential, power‑law, Beta, sine‑power,
    piecewise‑constant, random Dirichlet) to obtain a decent seed.
2.  A projected gradient descent that minimizes a smooth approximation of the
    max‑convolution (using power‑ or exponential smoothing).  Gradients are
    computed exactly via FFT‑based convolutions and the iterate is projected
    onto the probability simplex.
3.  A short simulated‑annealing polishing step followed by a tiny uniform‑mix
    polish.

All heavy work is done with FFTs, the simplex projection is O(n log n) and the
total runtime is bounded by the supplied time budget (≈1000 s).  The final
sequence is returned as a plain list; ``evaluate_sequence`` is only called at
the end for logging.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (exact)
# ----------------------------------------------------------------------
def _conv_linear(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n1 = a.shape[0]
    n2 = b.shape[0]
    out_len = n1 + n2 - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return _conv_linear(a, a)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto { w ≥ 0, Σ w = 1 }."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective for a probability vector (Σ a = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Phase 0 – rich parametric sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Search a variety of parametric families and keep the best vector."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try_raw(arr_raw):
        nonlocal best_a, best_val
        a = np.asarray(arr_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                           # 1 … n

    # 1. Exponential tails (decreasing / symmetric)
    for r in np.linspace(0.995, 0.999999, 20):
        _try_raw(r ** i)                         # pure decline
        _try_raw(r ** i + r ** (n - 1 - i))      # symmetric U

    # 2. Power‑law tails (decline / symmetric)
    for p in np.linspace(0.1, 4.0, 35):
        _try_raw(i1 ** (-p))
        _try_raw(i1 ** (-p) + (n - i) ** (-p))

    # 3. Inverse‑sqrt (special power‑law)
    _try_raw(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))

    # 4. Beta densities – extended range, including very sharp U‑shapes
    beta_vals = np.concatenate((
        np.linspace(0.02, 0.09, 8),
        np.linspace(0.1, 0.5, 9),
        np.linspace(0.6, 1.5, 10)
    ))
    for a_param in beta_vals:
        for b_param in beta_vals:
            a_vec = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try_raw(a_vec)

    # 5. Arcsine distribution (Beta(0.5, 0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try_raw(arcs)

    # 6. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.2, 5.5, 14):
        _try_raw(base_sin ** p)

    # 7. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.003, 0.08, 10):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.04, 0.35, 10):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a_vec = np.full(n, low)
            a_vec[:k] = high_frac
            a_vec[-k:] = high_frac
            _try_raw(a_vec)

    # 8. Random Dirichlet draws for diversity
    rng = np.random.default_rng(123456)
    for _ in range(120):
        _try_raw(rng.dirichlet(np.ones(n)))

    # 9. Interpolate a previous solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try_raw(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – projected gradient descent with smooth max approximation
# ----------------------------------------------------------------------
def _gradient_descent(seed_a: np.ndarray, n: int,
                     start: float, deadline: float,
                     best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv(a,a)) by projected gradient descent.
    The smoothing is performed with a power‑law (b^{p-1}) that gradually
    concentrates on the active peak.  After each update the vector is projected
    onto the simplex.
    """
    a = seed_a.copy()
    cur_val = best_val
    best_local_a = best_a.copy()
    best_local_val = best_val

    # schedule of smoothing exponents – larger p focuses on the true max
    p_schedule = [2, 4, 8, 16, 32, 64, 128]

    for p in p_schedule:
        if time.time() >= deadline:
            break
        # inner loop: stop when no improvement for a few tries or time runs out
        stagnation = 0
        while time.time() < deadline:
            # -------------------------------------------------------------
            # current objective
            b = _full_convolution(a)
            max_b = b.max()
            cur_val = 2.0 * n * max_b

            if cur_val < best_local_val - 1e-12:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[GD p={p}] new local best = {best_local_val:.6f}")
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[GD] global best improved to {best_val:.6f}")

            # -------------------------------------------------------------
            # compute smoothing weights
            if p <= 64:
                # power smoothing – very cheap
                w_unnorm = np.power(b + 1e-30, p - 1)
            else:
                # exponential (log‑sum‑exp) smoothing for very large p
                eps = 0.005 * p
                t = eps * (b - b.max())
                w_unnorm = np.exp(t)
            w_sum = w_unnorm.sum()
            if w_sum == 0.0:
                break
            w = w_unnorm / w_sum

            # -------------------------------------------------------------
            # gradient of the smoothed max: ∇ = 2 * (a ⋆ w)   (⋆ = cross‑correlation)
            conv_rev = _conv_linear(a, w[::-1])
            g = 2.0 * conv_rev[n - 1 : n - 1 + n][::-1]   # length n

            # -------------------------------------------------------------
            # step‑size candidates (scale inversely with ‖g‖)
            norm_g = np.linalg.norm(g) + 1e-12
            base_eta = 0.01 / norm_g
            improved = False
            for factor in (1.0, 0.5, 2.0, 3.0):
                eta = base_eta * factor
                cand = a - eta * g
                cand = _project_simplex(cand)
                b_cand = _full_convolution(cand)
                cand_val = 2.0 * n * b_cand.max()
                if cand_val < cur_val - 1e-12:
                    a = cand
                    cur_val = cand_val
                    improved = True
                    break
            if not improved:
                # try a few smaller steps before giving up on this p
                for factor in (0.2, 0.1, 0.05):
                    eta = base_eta * factor
                    cand = a - eta * g
                    cand = _project_simplex(cand)
                    b_cand = _full_convolution(cand)
                    cand_val = 2.0 * n * b_cand.max()
                    if cand_val < cur_val - 1e-12:
                        a = cand
                        cur_val = cand_val
                        improved = True
                        break
            if not improved:
                stagnation += 1
                if stagnation >= 40:      # give up on this p
                    break
            else:
                stagnation = 0            # reset counter after any improvement

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing polish (identical to the earlier version)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Mass‑transfer simulated annealing (exact max‑convolution)."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = best_a.copy()
    best_local_val = best_val

    max_frac_start, max_frac_end = 0.6, 0.001
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T0 = 0.02 * cur_val
        T = max(T0 * (1.0 - progress), 1e-12)

        # index where current maximal convolution occurs
        k_max = int(np.argmax(b))

        # ----- choose a source index (high contribution to the peak) -----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))
        if a[i] <= 1e-16:
            continue

        # ----- choose a target index (low contribution) -----
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0.0:
            probs_j = w / w.sum()
            j = int(np.random.choice(n, p=probs_j))
        else:
            j = (i + np.random.randint(1, n)) % n
        if j == i:
            continue

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the full convolution -----
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_local_val - 1e-15:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        # occasional re‑normalisation to curb round‑off
        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        # restart after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_local_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {it:7d}] perturbing after stagnation")
        it += 1

    return best_local_a, best_local_val

# Phase 3 – tiny uniform mixing polish
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a small share of the uniform distribution."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val
