"""
Step‑function optimizer.

We search for a non‑negative vector a (Σa = 1) that minimises

    2·n·max(a * a)         (a * a = full linear convolution of a with itself)

The code works in three stages:

1. **Rich parametric sweep** – a large collection of simple families
   (exponential, power‑law, Beta, edge‑heavy piecewise‑constant,
   random Dirichlet, arcsine) is evaluated (FFT based) and the best
   candidate is kept.  The sweep is stopped early if the overall time
   budget is exhausted.

2. **Targeted hill‑climbing / simulated annealing** – from the best seed
   we repeatedly move a small amount of probability mass from an index
   that contributes most to the current maximal convolution bin to an
   index that contributes very little.  The move is accepted with a
   Metropolis rule (temperature proportional to the current objective).
   The convolution is updated incrementally; a full recompute is done
   periodically to avoid drift.  After a long stagnation we perturb the
   current best and continue.  A second run starting from a random
   Dirichlet mixture is performed if time permits.

3. **Uniform polishing** – a tiny convex combination with the uniform
   distribution (α∈{0.001,…,0.020}) is tried; the best mixture is kept.

All operations respect the simplex constraints, run within the given
time budget (≤ 1000 s) and finally return a list of floats ready for
`evaluate_sequence`.
"""

import time

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """
    Exact objective for a vector with Σ a = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, n: int,
                start: float, deadline: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Deterministic‑stochastic hill‑climbing.
    Moves a small amount of mass from an index that heavily contributes
    to the current maximal convolution bin to an index that contributes
    very little.  Acceptance follows a Metropolis rule.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_a = a.copy()
    best_val = cur_val

    # parameters governing the schedule
    max_frac_start, max_frac_end = 0.8, 0.001
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # progress‑dependent schedules
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T0 = 0.02 * cur_val
        T = max(T0 * (1.0 - progress), 1e-12)

        # locate the (one) maximal bin
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # --- choose source index (high contribution) -----------------
        if total_contrib > 0.0:
            # take the top 1 % of contributors (at least 2 indices)
            top_k = max(2, n // 100)
            top_idxs = np.argpartition(contrib, -top_k)[-top_k:]
            probs_src = contrib[top_idxs]
            prob_sum = probs_src.sum()
            if prob_sum > 0.0:
                probs_src = probs_src / prob_sum
                i = int(np.random.choice(top_idxs, p=probs_src))
            else:
                i = int(np.random.choice(top_idxs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            # nothing to move from this index
            it += 1
            continue

        # --- choose target index (low contribution) ------------------
        bottom_k = max(2, n // 100)
        bottom_idxs = np.argpartition(contrib, bottom_k)[:bottom_k]
        # avoid picking the same index
        if i in bottom_idxs:
            bottom_idxs = bottom_idxs[bottom_idxs != i]
            if bottom_idxs.size == 0:
                bottom_idxs = np.arange(n)
                bottom_idxs = bottom_idxs[bottom_idxs != i]
        j = int(np.random.choice(bottom_idxs))

        if j == i:
            it += 1
            continue

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            it += 1
            continue

        # ----- incremental update of the full convolution -----
        # copy once per trial (still cheap)
        b_new = b.copy()
        # subtraction of the old i‑contributions
        b_new[i:i + n] -= 2.0 * delta * a
        # addition of the new j‑contributions
        b_new[j:j + n] += 2.0 * delta * a
        # quadratic self‑terms and cross‑term correction
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val - 1e-15:
                best_val = cur_val
                best_a = a.copy()
                print(f"[HC {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        # restart after long stagnation
        if no_improve > IMPROVE_LIMIT:
            # perturb the current best slightly and continue
            a = best_a.copy()
            noise = np.random.normal(scale=0.004, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HC {it:7d}] perturbing after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small final polishing – blend with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """
    Convex combination with the uniform distribution; useful for a final
    tweak that often reduces the peak a little further.
    """
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the allotted time budget.
    Returns a list of non‑negative floats (Σ≈1) ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # Problem size – 12 k entries balances quality and speed
    n = 12000

    # ------------------------------------------------------------------
    # Phase 0 – extensive parametric sweep
    # ------------------------------------------------------------------
    print("[Init] parametric sweep")
    best_a, best_val = _initial_candidate(n, deadline)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – hill‑climbing / simulated annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb(best_a, n, start_time, deadline,
                                       target_val=1.503)
        print(f"[HC1] best = {best_val:.6f}")

    # Optional second run from a fresh random start (if time permits)
    if time.time() + 5.0 < deadline:
        rng = np.random.default_rng(seed + 123)
        a_rand = rng.dirichlet(np.ones(n))
        a_rand = a_rand / a_rand.sum()
        # blend a bit with the current best to stay in a promising region
        a0 = 0.6 * best_a + 0.4 * a_rand
        a0 = _project_simplex(a0)
        a2, val2 = _hill_climb(a0, n, start_time, deadline,
                               target_val=1.503)
        print(f"[HC2] best = {val2:.6f}")
        if val2 < best_val:
            best_a, best_val = a2, val2

    # ------------------------------------------------------------------
    # Phase 2 – tiny uniform mixing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _mix_with_uniform(best_a, n)
        print(f"[Polish] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided wrapper (just for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
