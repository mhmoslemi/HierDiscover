def _max_convolution_geometric(n: int, r: float) -> float:
    if r == 1.0:
        # uniform
        return 1.0 / n
    # compute sum a_i
    # a_i = r^i
    # sum_r = (1 - r**n) / (1 - r)
    sum_r = (1 - r**n) / (1 - r)
    # compute b_k for k in 0..2n-2
    # For k <= n-1: count = k+1
    # For k > n-1: count = 2n-1 - k
    # We can compute using vectorized array
    ks = np.arange(2*n - 1)
    counts = np.where(ks < n, ks + 1, 2*n - 1 - ks)
    # r^k
    rk = r**ks
    b = rk * counts / (sum_r**2)
    return b.max()
```

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    n = 8000

    # initial candidates maybe few random shapes etc.
    best_val = np.inf
    best_a = None

    # 1) Geometric search
    def obj_for_c(c):
        r = 1.0 - c / n
        if r <= 0:
            return np.inf
        # compute sum of powers
        sum_r = (1 - r**n) / (1 - r)
        # counts
        ks = np.arange(2 * n - 1)
        counts = np.where(ks < n, ks + 1, 2 * n - 1 - ks)
        # use log to avoid underflow? Not needed.
        rpow = r**ks
        b = rpow * counts / (sum_r**2)
        max_b = b.max()
        return 2.0 * n * max_b

    # coarse grid
    cs = np.linspace(0.1, 8.0, 200)
    for c in cs:
        val = obj_for_c(c)
        if val < best_val:
            best_val = val
            best_c = c
    # refine around best_c using golden-section
    a_low = max(0.01, best_c - 1.0)
    a_high = best_c + 1.0
    phi = (np.sqrt(5) - 1) / 2  # about 0.618
    x1 = a_low + (1 - phi) * (a_high - a_low)
    x2 = a_low + phi * (a_high - a_low)
    f1 = obj_for_c(x1)
    f2 = obj_for_c(x2)
    while (a_high - a_low) > 1e-4 and time.time() < deadline:
        if f1 < f2:
            a_high = x2
            x2 = x1
            f2 = f1
            x1 = a_low + (1 - phi) * (a_high - a_low)
            f1 = obj_for_c(x1)
        else:
            a_low = x1
            x1 = x2
            f1 = f2
            x2 = a_low + phi * (a_high - a_low)
            f2 = obj_for_c(x2)
    best_c = (a_low + a_high) / 2
    best_val = obj_for_c(best_c)

    # construct a from best_c:
    r = 1.0 - best_c / n
    if r <= 0:
        r = 0.0
    # a_i = r**i
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0/n)
    else:
        a /= s
    cur_val = _objective(a)  # compute exact using conv (or analytic)
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    # Optional: small hillclimb refinement using existing code but limited iterations
    # Use only if time left > something
    # We'll reuse _hillclimb but with small budget
    remaining = deadline - time.time()
    if remaining > 30:  # use some time for hillclimbing
        sub_deadline = time.time() + min(remaining*0.8, 200)  # Up to 200 sec
        best_a, best_val = _hillclimb(best_a, n,
                                      deadline=sub_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val)
    # final
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[Final] reported score = {final_score:.6f}")

    return best_a.tolist()

      subject to  (a * g)_k ≤ M   for the *active* convolution
                  constraints (those k where a*a is near its current peak M)
                  and g ≥ 0.
  This is exactly the LP described in the reference text – it tells us how
  much total mass we could add without breaking the current worst‑case bound.
* Blend the current distribution a with the normalised LP solution ĝ :
      a ← (1‑t)·a + t·ĝ    (t≈0.5, reduced if the new max grows).
  The blend strictly reduces (or at least does not increase) the maximum
  convolution value, driving the objective down.
* After a few LP steps a short hill‑climbing polish (the fast O(n) move
  from the previous code) is applied, then the final sequence is returned.

All heavy work stays well below the 1000 s budget; the LP works on a very
small set of “active’’ constraints (≈30 rows), so the linear program is tiny
even for n=8000.  The routine prints progress and the final reported score.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simple hill‑climbing (fast O(n) updates) – same as previous code
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               n: int,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules
    max_frac_start, max_frac_end = 0.7, 0.001
    temp_start, temp_end = 0.02, 1e-8

    no_improve = 0
    IMPROVE_LIMIT = 60000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill]   new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill]   stagnation – perturb (it={iteration})")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Build a sparse matrix that encodes (a * g)_k ≤ M for a selected set of k’s
# ----------------------------------------------------------------------
def _build_constraint_matrix(a: np.ndarray, active_ks: np.ndarray) -> sparse.csr_matrix:
    """
    Returns a sparse matrix A where  A @ g  gives the convolution entries
    (a * g)_k for each k in `active_ks`.  The matrix has shape
    (len(active_ks), len(a)).
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []

    for row_id, k in enumerate(active_ks):
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # i runs over indices of a that actually appear in this convolution term
        i_vals = np.arange(i_low, i_high + 1)
        j_vals = k - i_vals               # corresponding columns for g
        rows.extend([row_id] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(a[i_vals].tolist())

    return sparse.csr_matrix((data, (rows, cols)), shape=(len(active_ks), n))


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                n: int,
                deadline: float,
                start_time: float,
                best_a: np.ndarray,
                best_val: float,
                max_active: int = 30,
                blend_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    One iteration of the LP improvement:
        * find the current maximum M,
        * pick the top `max_active` convolution indices as active constraints,
        * solve maximise Σ g_j  subject to (a * g)_k ≤ M on those rows,
        * blend the normalised solution into `a`.
    Returns possibly improved (a, value).
    """
    # current convolution and bound
    b = _full_convolution(a)
    M = b.max()
    # choose active constraints – those where b is close to the peak
    # keep at most `max_active` rows, always include the extremes (0 and 2n‑2)
    idx_sorted = np.argsort(-b)          # descending
    active = idx_sorted[:max_active].tolist()
    for k in (0, 2 * n - 2):
        if k not in active:
            active.append(k)

    active_arr = np.array(active, dtype=int)

    # Build sparse A matrix for active rows
    A = _build_constraint_matrix(a, active_arr)

    # Linear program: maximise Σ g_j  (i.e. minimise -Σ g_j)
    c = -np.ones(n, dtype=float)

    # Upper bounds: A @ g ≤ M
    ub = np.full(A.shape[0], M, dtype=float)

    # Non‑negativity bounds on g
    bounds = [(0, None) for _ in range(n)]

    # Solve
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max(1.0, deadline - time.time())})

    if res.status != 0:
        # infeasible or other issue – give up this LP step
        print("[LP]   linprog status:", res.status, res.message)
        return a, best_val

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1.0 + 1e-9:
        # no useful mass increase possible
        print("[LP]   no increase (sum(g)={:.6f})".format(sum_g))
        return a, best_val

    # Normalise the LP solution to a probability vector
    g_norm = g / sum_g

    # Blend – we try a few decreasing blend factors if the max grows
    for t in (blend_factor, blend_factor * 0.5, blend_factor * 0.25):
        a_new = (1 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        val = _objective(a_new)
        if val < best_val:
            print(f"[LP]   improvement: val={val:.6f} (t={t:.3f})")
            return a_new, val

    # No improvement – keep the old vector
    return a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (tries to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size
    # ------------------------------------------------------------------
    n = 8000

    #    otherwise fall back to a random Beta shape.
    # ------------------------------------------------------------------

        cand = np.interp(new_x, old_x, old)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective(cand)
        best_a, best_val = cand, val
        print(f"[Init] interpolated previous solution, val={best_val:.6f}")
    except Exception:
        pass

    if best_a is None:
        # Random Beta (log‑uniform p,q) – a cheap family that often yields a good start
        i = np.arange(1, n + 1, dtype=np.float64)
        logp = np.random.uniform(-9.2, 2.3)
        logq = np.random.uniform(-9.2, 2.3)
        p, q = np.exp(logp), np.exp(logq)
        cand = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective(cand)
        best_a, best_val = cand, val
        print(f"[Init] random Beta start, val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Repeated LP improvement steps (fast, each ~0.1 s)
    # ------------------------------------------------------------------
    lp_iters = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        if lp_iters >= 15:               # safeguard against infinite loops
            break
        best_a, best_val = _lp_improve(best_a, n, deadline, start_time,
                                       best_a, best_val,
                                       max_active=30, blend_factor=0.5)
        lp_iters += 1

    # ------------------------------------------------------------------
    # 2️⃣  Optional short hill‑climbing polish (≈20 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        hill_deadline = time.time() + min(remaining * 0.8, 200.0)
        best_a, best_val = _hillclimb(best_a, n,
                                      deadline=hill_deadline,
                                      start_time=start_time,
                                      best_a=best_a,
                                      best_val=best_val)

    except Exception as exc:  # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)
