def _gradient_weighted(a, b, topk, T):
    n = a.shape[0]
    # find topk indices
    idxs = np.argpartition(-b, topk-1)[:topk]
    # stable: get actual values for those indices
    # compute weights
    max_b = b.max()
    # compute raw weights
    raw = np.exp((b[idxs] - max_b) / T)
    w = raw / raw.sum()
    grad = np.zeros_like(a)
    arange = np.arange(n)
    for k, weight in zip(idxs, w):
        shift = k - arange  # array length n
        mask = (shift >= 0) & (shift < n)
        grad[mask] += weight * 2.0 * a[shift[mask]]
    return grad
```

"""
Propose a step‐function (non‑negative vector) that minimises
evaluate_sequence(sequence) under a 1000 s time budget.

Algorithm
---------
1. **Initial pool** – start from any previously saved solution
   (`height_sequence_1`) and from a collection of Beta‑shaped vectors
   a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} (dense grid + random log‑uniform sampling).
   The best of these becomes the baseline.

2. **Weighted sub‑gradient descent** – we minimise
      F(a) = 2·n·max(conv(a,a))   (with Σa = 1)
   using a smoothed version of the max:
        w_k ∝ exp((b_k‑max(b))/T)      (softmax over the convolution)
   Only the top‑K convolution entries are kept (K≈5) to keep the
   gradient cheap.
   A momentum term (β≈0.9) and a linearly decaying step size
   (0.2 → 1e‑5) are employed.  After each step we project back onto
   the probability simplex.

3. **Local hill‑climbing** – with the remaining time we perform
   cheap random swaps of a tiny mass ε between two coordinates;
   accept the swap only if the objective improves.

All arithmetic is performed in double precision and uses FFT‑based
convolution for O(n log n) cost per iteration (n=8000 by default).

The best vector found before the deadline is returned as a plain
Python list; `evaluate_sequence` will be called once on it for the
final score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta‑shaped vector or None if degenerate."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _weighted_gradient(a: np.ndarray, b: np.ndarray,
                       topk: int, temperature: float) -> np.ndarray:
    """
    Gradient of a soft‑max weighted sum of convolution entries.
    Only the top‑k entries of b are used.
    """
    n = a.shape[0]
    # indices of the k largest convolution values
    idxs = np.argpartition(-b, topk - 1)[:topk]
    max_b = b.max()
    # soft‑max weights (stable)
    raw = np.exp((b[idxs] - max_b) / temperature)
    weights = raw / raw.sum()

    grad = np.zeros_like(a)
    ar = np.arange(n)

    for k, w in zip(idxs, weights):
        shift = k - ar          # k - i
        mask = (shift >= 0) & (shift < n)
        # Each contribution is 2 * a[shift] (standard sub‑gradient)
        grad[mask] += w * 2.0 * a[shift[mask]]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a small margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between expressivity and speed
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Gather initial candidates (previous solution, Beta‑grid, random)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # 0a – try to reuse a previously saved solution
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)
        best_a, best_val = a0, val
        print("[init] interpolated previous solution, value={:.6f}".format(val))
    except Exception:
        pass

    # 0b – dense grid over the Beta family
    p_grid = np.concatenate((
        np.geomspace(0.001, 0.1, 8, endpoint=False),
        np.geomspace(0.1, 5.0, 12, endpoint=False),
        np.geomspace(5.0, 20.0, 5),
    ))
    q_grid = p_grid.copy()
    for p in p_grid:
        for q in q_grid:
            cand = _beta_shape(p, q, n)
            if cand is None:
                continue
            val = _objective(cand)
            if val < best_val:
                best_a, best_val = cand, val
                print(f"[grid] new best={val:.6f} (p={p:.4g}, q={q:.4g})")
    # 0c – random log‑uniform sampling in the Beta family
    log_min, log_max = np.log(1e-3), np.log(20.0)
    for _ in range(8000):
        logp = rng.uniform(log_min, log_max)
        logq = rng.uniform(log_min, log_max)
        p, q = np.exp(logp), np.exp(logq)
        cand = _beta_shape(p, q, n)
        if cand is None:
            continue
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[rand] new best={val:.6f} (p={p:.4g}, q={q:.4g})")
        if time.time() >= deadline - 30.0:   # keep ~30 s for refinement
            break

    # 0d – a handful of Dirichlet draws for diversity
    for _ in range(6):
        cand = rng.dirichlet(np.ones(n))
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[dir] new best={val:.6f}")

    # fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] uniform start, value={:.6f}".format(best_val))

    # ------------------------------------------------------------------
    # 1️⃣  Weighted sub‑gradient descent with momentum and temperature annealing
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_seq = a.copy()
    best_score = best_val

    # hyper‑parameters
    step_start = 0.20
    step_end = 1e-5
    temperature_start = 0.015
    temperature_end = 1e-6
    momentum_beta = 0.90
    topk = 5                 # number of convolution entries influencing the gradient

    # momentum buffer
    v = np.zeros_like(a)

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress in [0,1]
        progress = (now - start_time) / (deadline - start_time)

        # current convolution and objective
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        # record best
        if cur_val < best_score:
            best_score, best_seq = cur_val, a.copy()
            print(f"[subgrad] iter {iteration:5d} new best={cur_val:.6f}")

        # compute weighted gradient
        T = temperature_start * (1.0 - progress) + temperature_end * progress
        grad = _weighted_gradient(a, b, topk=topk, temperature=T)

        # momentum update
        v = momentum_beta * v + (1.0 - momentum_beta) * grad

        # step‑size schedule (linear decay)
        step = step_start * (1.0 - progress) + step_end * progress

        # gradient step + simplex projection
        a_candidate = a - step * v
        a_candidate = np.maximum(a_candidate, 0.0)        # safeguard before projection
        a_candidate = _project_simplex(a_candidate)

        a = a_candidate
        iteration += 1

        # occasional small random jitter to escape flat plateaus
        if rng.random() < 0.001:
            a += rng.normal(scale=0.001, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # 2️⃣  Final hill‑climbing phase (tiny random swaps)
    # ------------------------------------------------------------------
    # Use the remaining time (≈30 s) for cheap local improvements.
    hill_deadline = deadline
    rng = np.random.default_rng(seed + 999)  # different stream
    while time.time() < hill_deadline:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # move a tiny amount from i to j (but not more than what i holds)
        eps = rng.uniform(0.0, min(a[i], 1e-4))
        if eps == 0.0:
            continue
        a_new = best_seq.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # evaluate
        val = _objective(a_new)
        if val < best_score:
            best_score, best_seq = val, a_new.copy()
            print(f"[hill] improved to {val:.6f} (swap {i}->{j})")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
