    deadline = start + budget_s - 0.5

    i = np.arange(n, dtype=np.float64) + 1.0

    # beta family couple of parameters
    def beta_seq(p,q):
        i = np.arange(1,n+1, dtype=np.float64)
        a = (i**(p-1)) * ((n - i + 1)**(q-1))
        a = a / a.sum()
        return a

    for p,q in [(0.5,0.5),(1.2,1.2),(2.0,0.5),(0.5,2.0)]:

    # random Dirichlet seeds
    for _ in range(2):

    best_a = None

"""
Smooth‑max projected gradient descent for the step‑function “max‑convolution”
challenge.

The algorithm builds a probability vector a (sum = 1, a_i≥0) on a given length n.
It minimises   f(a)=max_k (a∗a)_k   by replacing the non‑differentiable max
with a smooth Log‑Sum‑Exp surrogate.
At each iteration the gradient of the surrogate is obtained by a single FFT‑
based convolution, the iterate is updated and projected back onto the simplex.
Several simple initialisations (uniform, inverse‑square‑root, a few Beta
families and random Dirichlet draws) are tried; the best result found before
the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_linear(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution of u (len L) with v (len M) using FFT."""
    L = u.shape[0]
    M = v.shape[0]
    out_len = L + M - 1
    size = 1 << ((out_len - 1).bit_length())
    fu = np.fft.rfft(u, size)
    fv = np.fft.rfft(v, size)
    return np.fft.irfft(fu * fv, size)[:out_len]


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# Smooth‑max projected gradient descent
# ----------------------------------------------------------------------
def _smooth_pgd_optimize(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the log‑sum‑exp surrogate of max_k (a∗a)_k.
    The iterates are always kept on the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    # parameters
    beta = 1.0                     # start with a very smooth surrogate
    beta_max = 2000.0              # final temperature (almost the true max)
    step = 0.3                     # initial step size
    min_step = 1e-12
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    # pre‑allocate FFT size for the gradient convolution (w length = 2n‑1)
    w_len = 2 * n - 1
    grad_fft_size = 1 << ((w_len + n - 1 - 1).bit_length())

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- current convolution and objective -----------------------
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        max_b = b.max()

        # ----- smooth max weights ------------------------------------
        # w_k ∝ exp(beta * (b_k - max_b))  – numerically stable
        exp_arg = beta * (b - max_b)
        # to avoid overflow we clip
        np.clip(exp_arg, a_min=-700, a_max=700, out=exp_arg)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        w /= w_sum

        # ----- gradient (2 * (w * reverse(a))) -----------------------
        # compute linear convolution of w with reversed a
        a_rev = a[::-1]
        fw = np.fft.rfft(w, grad_fft_size)
        fa = np.fft.rfft(a_rev, grad_fft_size)
        conv = np.fft.irfft(fw * fa, grad_fft_size)[:w_len + n - 1]
        # entry i of the gradient corresponds to slice n‑1 : n‑1+n
        grad = 2.0 * conv[n-1:n-1+n]

        # ----- line‑search -------------------------------------------------
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            a_cand = a - step * factor * grad
            a_cand = _project_simplex(a_cand)
            b_cand = _full_convolution(a_cand)
            cand_val = 2.0 * n * b_cand.max()
            if cand_val < cur_val - 1e-15:
                a = a_cand
                cur_val = cand_val
                improved = True
                if cand_val < best_val - 1e-12:
                    best_val, best_a = cand_val, a.copy()
                    print(f"[SmoothPGD] new global best = {best_val:.6f}")
                    if best_val <= target_val:
                        return best_a, best_val
                break

        if not improved:
            step *= 0.5
            no_improve += 1
            if step < min_step:
                break
        else:
            # increase step a little after a success
            step = min(step * 1.07, 0.5)
            no_improve = 0

        # ----- temperature schedule ---------------------------------------
        beta = min(beta * 1.02, beta_max)

        # ----- stagnation handling -----------------------------------------
        if no_improve > IMPROVE_LIMIT:
            # random perturbation of the current iterate
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            step = 0.3
            no_improve = 0
            print(f"[SmoothPGD] stagnation – random perturb")
