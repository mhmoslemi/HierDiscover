def toeplitz_mul(w, a):
    # w length L = 2n-1
    n = len(a)
    w_rev = w[::-1]
    size = 1 << ((len(w_rev) + n - 1 - 1).bit_length())
    fw = np.fft.rfft(w_rev, size)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fw * fa, size)[:len(w_rev)+n-1]
    # take slice indices (len(w_rev)-1) - np.arange(n)
    L = len(w_rev)
    out = conv[L-1 - np.arange(n)]
    return out

def _adam_optimize(a0, n, deadline, start_time, init_lr=0.05, init_alpha=5.0, final_alpha=300.0):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = _objective(a)
    while time.time() < deadline:
        t += 1
        # compute conv, weights, gradient
        conv = _full_convolution(a)
        # compute softmax weights for current alpha
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = init_alpha * (final_alpha / init_alpha) ** progress
        # stable softmax
        max_c = conv.max()
        exp_term = np.exp(alpha * (conv - max_c))
        w = exp_term / exp_term.sum()
        # gradient
        grad = 2.0 * _toeplitz_mul(w, a)  # shape n
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # projection onto simplex
        a_new = _project_simplex(a_new)
        # evaluate
        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            print(f"[Adam {t}] new best = {best_val:.6f}")
        a = a_new
    return best_a, best_val

"""
A two‑phase optimizer for the step‑function challenge.

Phase 1 – smooth Adam descent:
    Minimise a soft‑max (log‑sum‑exp) surrogate for
    max(a∗a) while staying on the probability simplex.
    The surrogate becomes tighter as the schedule “α”
    grows, allowing large exploratory steps early and
    precise sharpening later.  Gradients are obtained
    analytically via a Toeplitz‑matrix identity and
    computed with FFTs (O(n log n)).

Phase 2 – intensive hill‑climbing:
    Refines the result with the classic mass‑transfer
    annealing used in earlier submissions.

The routine respects the 1000 s budget and returns a
list of non‑negative heights.
"""
import time
import math
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y (length len(x)+len(y)-1)."""
    nx = x.shape[0]
    ny = y.shape[0]
    out_len = nx + ny - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(a∗a) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)

def _toeplitz_mul(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute (Toeplitz(w) @ a), where Toeplitz(w)_{i,j}=w_{i+j}.
    Implemented via a single FFT‑based convolution.
    The result is a vector of length len(a).
    """
    n = a.shape[0]
    L = w.shape[0]                     # = 2*n-1
    w_rev = w[::-1]
    conv = _full_convolution_two(w_rev, a)   # length = L+n-1 = 3n-2
    indices = (L - 1) - np.arange(n)         # descending slice
    return conv[indices]

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   init_lr: float = 0.12,
                   init_alpha: float = 5.0,
                   final_alpha: float = 250.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        f_α(a) = (1/α) log Σ exp(α·(a∗a))
    The surrogate approaches the true max for α→∞.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-10
    best_a, best_val = a.copy(), _objective(a)
    t = 0
    while time.time() < deadline:
        t += 1
        # current convolution
        conv = _full_convolution(a)

        # schedule α (exponential annealing)
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = init_alpha * (final_alpha / init_alpha) ** progress

        # soft‑max weights
        max_c = conv.max()
        exp_term = np.exp(alpha * (conv - max_c))       # stability trick
        w = exp_term / exp_term.sum()

        # gradient = 2 * Toeplitz(w) @ a
        grad = 2.0 * _toeplitz_mul(w, a)

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        val = _objective(a_new)
        if val < best_val:
            best_val, best_a = val, a_new.copy()
            print(f"[Adam {t:7d}] α={alpha:.1f}  new best = {best_val:.6f}")

        a = a_new

    return best_a, best_val

def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local annealing move that transfers a small amount
    of mass from an index heavily involved in the current
    maximal autocorrelation to a relatively quiet index.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source selection – biased towards large contribution entries
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target – favour indices that are currently quiet
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback: uniform random distinct index
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # mass to move
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update convolution efficiently
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # escape stagnation
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – random perturb")

        iteration += 1

    return best_a, best_val

    beta1, beta2 = 0.9, 0.999
    eps = 1e-10
    best_a, best_val = a.copy(), _objective(a)
    t = 0

    while time.time() < deadline:
        t += 1
        # current self‑convolution
        conv = _full_convolution(a)

        # exponential schedule for α
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = init_alpha * (final_alpha / init_alpha) ** progress

        # soft‑max weights w_i ∝ exp(α·conv_i)
        max_c = conv.max()
        exp_term = np.exp(alpha * (conv - max_c))     # subtract max for stability
        w = exp_term / exp_term.sum()

        # gradient = 2·Toeplitz(w)·a
        grad = 2.0 * _toeplitz_mul(w, a)

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        val = _objective(a_new)
        if val < best_val:
            best_val, best_a = val, a_new.copy()
            print(f"[Adam {t:7d}] α={alpha:.1f}  new best = {best_val:.6f}")

        a = a_new

    return best_a, best_val

# Intensive simulated‑annealing hill‑climbing (identical to the reference)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local annealing move that transfers a small amount of mass from a
    heavily‑contributing index to a quieter one.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – preferably one with large contribution
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – prefers positions with little contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # amount of mass to move
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update the convolution tail efficiently
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # escape stagnation by adding a small Gaussian jitter
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – random perturb")

        iteration += 1

    return best_a, best_val

# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s
    budget, starts from the previous construction if available,
    runs a smooth Adam descent followed by a refined hill‑climb, and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final call

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – reuse the best known construction if it exists.
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        n = a0.shape[0]
        print(f"[init] using previous solution, n={n}")
    except Exception:
        n = 8000
        a0 = np.random.dirichlet(np.ones(n))
        print(f"[init] random start, n={n}")

    # ------------------------------------------------------------------
    # 0b️⃣  Quick random‑seed improvement (cheap Dirichlet samples)
    # ------------------------------------------------------------------
    best_a, best_val = a0.copy(), _objective(a0)
    for _ in range(5):
        cand = np.random.dirichlet(np.ones(n))
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[init] Dirichlet seed improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Phase 1 – smooth Adam descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        phase1_deadline = time.time() + min(300.0, remaining * 0.6)
        a_opt, val_opt = _adam_optimize(best_a, n, phase1_deadline, start_time)
        print(f"[phase 1] best objective = {val_opt:.6f}")
    else:
        a_opt, val_opt = best_a, best_val

    # ------------------------------------------------------------------
    # 2️⃣  Phase 2 – intensive annealed hill‑climb
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_opt, val_opt = _hillclimb_anneal(
            a_opt,
            n,
            deadline=deadline,
            start_time=start_time,
            best_a=a_opt,
            best_val=val_opt,
            target_val=1.503,
        )
        print(f"[phase 2] final best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation (only for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_opt.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return a_opt.tolist()
