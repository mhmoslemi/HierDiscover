Improved search for a non‑negative step function.

The algorithm builds on the previous approach but adds two key ideas:

1️⃣  A *soft‑max* smoothing of the “max” in the objective.
    Instead of using only the single lag where the autocorrelation is maximal,
    we compute a weighted combination of all lags with weights
    w_k ∝ exp(β·b_k).  The temperature β is increased gradually,
    turning the smooth objective into the original one.
    This provides dense gradient information and avoids getting stuck early.

2️⃣  Adaptive optimisation with Adam (first‑ and second‑moment estimates) on the
    simplex.  After each Adam step the vector is projected back onto the
    probability simplex Σa=1, a≥0.  The step‑size decays linearly and a tiny
    random jitter is added occasionally to escape plateaus.

The search proceeds in three stages:

* Initialise from the previously computed solution (if present) and explore
  the two‑parameter Beta family plus a few Dirichlet draws.
* Run the Adam optimiser with the soft‑max smoothing, slowly raising β.
  The best sequence seen during this phase is stored.
* Finally perform a short hard‑max sub‑gradient refinement (identical to the
  original code) to polish the result.

All work respects a 1000 s time budget; the final sequence is evaluated with
``evaluate_sequence`` before returning.

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of `x` with itself (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()                 # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective_from_a(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv(a,a))  (assumes Σa = 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _beta_shape(p: float, q: float, n: int) -> np.ndarray | None:
    """Return a normalised Beta‑shaped vector a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    total = a.sum()
    if total <= 0.0:
        return None
    a /= total
    return a


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max over β·b."""
    # shift by max to avoid overflow
    mb = b.max()
    exp_term = np.exp(beta * (b - mb))
    return exp_term / exp_term.sum()


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of 2·n·∑ w_k·b_k  where b = a*a.
    Using the identity  ∂b_k/∂a_i = a_{k-i} (zero outside range).
    This is exactly the linear cross‑correlation of a and w.
    """
    n = a.shape[0]
    L = w.shape[0]                       # = 2·n‑1
    size = 1 << ((n + L - 1).bit_length())   # enough for linear correlation
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    # Linear correlation (a ⋆ w)[i] = Σ_j a_j * w_{i+j}
    corr = np.fft.irfft(A * np.conj(W), size)[:n]
    return 2.0 * n * corr

            print(f"[adam] new best = {best_val:.6f} (iter {iteration})")
            # early exit if we have already reached the official target
            if best_val <= 1.503:
                break

        a = a_candidate
        iteration += 1
        beta = min(beta_max, beta * beta_mul)

    # --------------------------------------------------------------
    # 2️⃣  Brief hard‑max sub‑gradient polishing (identical to the earlier code)
    # --------------------------------------------------------------
    a = best_seq.copy()
    sub_iter = 0
    step_start = 0.1
    step_end = 1e-5
    jitter_prob = 0.001
    jitter_scale = 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break
        # convolution and exact objective
        b_full = _full_convolution(a)
        max_b = b_full.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            print(f"[subgrad] new best = {best_val:.6f} (sub‑iter {sub_iter})")
            if best_val <= 1.503:
                break

        # index of the current maximal autocorrelation entry
        k_max = int(np.argmax(b_full))

        # sub‑gradient of the max entry (only the active lag)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step‑size schedule (linear decay)
        progress = (now - start_time) / (deadline - start_time)
        step = step_start * (1.0 - progress) + step_end * progress

        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        if np.random.rand() < jitter_prob:
            a_candidate = a_candidate + np.random.normal(scale=jitter_scale, size=n)
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_simplex(a_candidate)

        a = a_candidate
        sub_iter += 1

    # --------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = np.inf

    return best_seq.tolist()
