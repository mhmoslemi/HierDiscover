"""
Propose a candidate step‐function that minimizes
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq))²

The algorithm works on the probability simplex (sum = 1) because the
objective is scale‑invariant.  It uses a smoothed “soft‑max’’ of the
full convolution and optimises the logits z = log a by Adam.  The
smoothing parameter α is increased from 1 to ≈200 during the run,
so the optimisation first sees a smooth landscape and finally focuses
on the true maximal entry.  After the Adam phase a short entropy‑mirror
refinement based on the exact maximal convolution entry is performed
to polish the result.

All heavy work (two FFT‑based full convolutions) is O(n log n) and
fits comfortably into the 1000 s budget for n≈8000.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolutions
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two different vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Mirror‑descent polishing (exact max)
# ----------------------------------------------------------------------
def _max_refine(a: np.ndarray, steps: int = 1500, eta_start: float = 0.5) -> np.ndarray:
    """Entropy‑mirror descent on the exact max‑entry of the convolution."""
    n = a.shape[0]
    a = a.copy()
    for t in range(steps):
        conv = _full_convolution(a)
        k = int(conv.argmax())
        # gradient of max‑entry w.r.t. a (sub‑gradient)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            idx = np.arange(lo, hi + 1, dtype=int)
            g[idx] = 2.0 * a[k - idx]
        # decreasing step size
        eta = eta_start * (1.0 - t / steps)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s time
    budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.8

    # ------------------------------------------------------------------
    # Problem size – 8000 works well and is the size of the previous best
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a (probability vector).  Prefer the previous best if
    # ``height_sequence_1`` exists, otherwise fall back to a smooth Beta
    # shape.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        x_new = np.linspace(0.0, 1.0, n)
        x_old = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(x_new, x_old, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        # a smooth, centrally‑peaked Beta shape (p = q = 2)
        i = np.arange(1, n + 1, dtype=np.float64)
        p = q = 2.0
        a0 = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        a0 = a0 / a0.sum()
        print("[init] generated Beta(2,2) start")

    # work in the log‑domain to keep positivity automatically
    eps = 1e-30
    z = np.log(a0 + eps)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # learning‑rate schedule (from 0.1 down to 0.004)
    lr_start, lr_end = 0.10, 0.004

    # best record (objective based on 2·n·maxconv, i.e. the exact
    # evaluate_sequence value because the sum is 1)
    best_a = None
    best_obj = np.inf
    best_iter = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (Adam on soft‑max of the convolution)
    # ------------------------------------------------------------------
    it = 0
    stagnation = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress fraction (0 → 1)
        prog = (now - start) / (deadline - start)

        # schedule learning rate and smoothing parameter α
        lr = lr_start * (1.0 - prog) + lr_end * prog
        # α grows from about 1 to ~200 (smooth → hard max)
        alpha = 1.0 + 199.0 * (prog ** 1.5)

        # ----------------------------------------------------------------
        # Recover a from logits, enforce exact sum‑to‑1
        # ----------------------------------------------------------------
        a = np.exp(z)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----------------------------------------------------------------
        # Full self‑convolution (needed for the soft‑max)
        # ----------------------------------------------------------------
        conv = _full_convolution(a)                # length 2n‑1

        # ----------------------------------------------------------------
        # Soft‑max of the convolution (stable)
        # ----------------------------------------------------------------
        maxc = conv.max()
        exp_scaled = np.exp(alpha * (conv - maxc))
        sumexp = exp_scaled.sum()
        weights = exp_scaled / sumexp               # Σ weights = 1
        log_sumexp = maxc + np.log(sumexp)
        obj = 2.0 * n * (log_sumexp / alpha)       # smooth approximation of evaluate_sequence

        # ----------------------------------------------------------------
        # Keep the best exact value seen so far
        # ----------------------------------------------------------------
        if obj < best_obj - 1e-12:
            # compute the exact (non‑smoothed) objective for reporting
            exact_obj = 2.0 * n * conv.max()
            best_obj = exact_obj
            best_a = a.copy()
            best_iter = it
            stagnation = 0
            print(f"[iter] new best O={best_obj:.6f}  it={it}  t={now-start:.1f}s")
        else:
            stagnation += 1

        # ----------------------------------------------------------------
        # Gradient w.r.t. a : g = 4·n * Σ_k weights_k * a_{k-i}
        # This is a convolution of rev(a) with the weight vector.
        # ----------------------------------------------------------------
        rev_a = a[::-1]
        conv_rev = _full_convolution_two(rev_a, weights)   # length 3n‑2
        g_a = 4.0 * n * conv_rev[n - 1:n - 1 + n]           # slice of length n

        # ----------------------------------------------------------------
        # Chain rule through the softmax (a = softmax(z))
        # ----------------------------------------------------------------
        dot = np.dot(g_a, a)
        grad_z = a * (g_a - dot)

        # ----------------------------------------------------------------
        # Adam update on the logits
        # ----------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----------------------------------------------------------------
        # Simple escape mechanism if we have been stuck for a while
        # ----------------------------------------------------------------
        if stagnation >= 3000:
            # add a tiny random jitter to the logits and reset counters
            z += np.random.normal(scale=0.02, size=n)
            stagnation = 0
            print(f"[reset] jitter added at it={it}")

    # ------------------------------------------------------------------
    # Polishing phase: entropy‑mirror descent on the exact max entry
    # ------------------------------------------------------------------
    if best_a is not None:
        a_refined = _max_refine(best_a, steps=1200, eta_start=0.45)
        conv_ref = _full_convolution(a_refined)
        obj_ref = 2.0 * n * conv_ref.max()
        if obj_ref < best_obj:
            best_obj = obj_ref
            best_a = a_refined
            print(f"[polish] improved to O={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final secure evaluation using the provided wrapper
    # ------------------------------------------------------------------
    final_seq = best_a.tolist() if best_a is not None else a0.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_score:.6f} (approx O={best_obj:.6f})")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
