"""
Propose a step‑function that (approximately) minimizes
    2·n·max(conv(a,a)) / (∑a)²
(the objective used by ``evaluate_sequence``).

The optimisation proceeds in three stages:

1.  **Initialisation** – a cheap parametric sweep (exponential,
    power‑law, cosine‑window, Beta‑distribution and a few Dirichlet draws)
    is performed on the target length (default 9000).  The best candidate
    of this sweep seeds the optimiser.

2.  **Smoothed max‑gradient descent** – we minimise a smooth approximation
    of the max using a soft‑max (log‑sum‑exp) with a temperature that is
    increased during the run.  The gradient of the smoothed objective is
    obtained analytically and evaluated efficiently via FFT‐based
    convolution / correlation.  After each gradient step the iterate is
    projected onto the probability simplex (non‑negative, sum = 1).  An
    adaptive step‑size (multiplicative increase/decrease) and a simple
    stagnation‑re‑initialisation scheme are used.

3.  **Tiny uniform mixing** – after the main descent a convex combination
    with a tiny amount of the uniform distribution is examined
    (α∈{0.001,…,0.020}) and the best is kept.

All heavy linear algebra uses NumPy’s FFT (rfft/irfft) and the code
automatically respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (single length for the whole run)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT of size L."""
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)
    return conv[: 2 * a.shape[0] - 1]


def _correlate_with_weights(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the correlation
        g[i] = Σ_j a[j] * w[i+j]   (0 ≤ i < len(a))
    where w has length 2·len(a)-1.
    Implemented via convolution with the reversed weight vector.
    Returns an array of length len(a).
    """
    # reverse w
    w_rev = w[::-1]
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w_rev, L)
    corr_full = np.fft.irfft(fa * fw, L)  # length L
    start = 2 * a.shape[0] - 2               # index of w_rev's last element
    return corr_full[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x≥0, Σx=1})
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (exact, assuming Σa = 1)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the smoothed max (soft‑max) objective
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, beta: float, L: int) -> np.ndarray:
    """
    Compute ∇_a  (log‑sum‑exp(β·conv(a)) / β) .
    The returned gradient already includes the factor 2 coming from ∂c_k/∂a_i.
    """
    # convolution a*a
    conv = _full_convolution(a, L)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (stable)
    w = np.exp(beta * (conv - max_c))
    sum_w = w.sum()
    if sum_w == 0.0:
        # degenerate case – fall back to plain sub‑gradient at max location
        k_max = int(np.argmax(conv))
        grad = np.zeros_like(a)
        idx = k_max - np.arange(a.shape[0])
        mask = (idx >= 0) & (idx < a.shape[0])
        grad[mask] = 2.0 * a[idx[mask]]
        return grad
    # correlation of a with the (normalised) weight vector
    corr = _correlate_with_weights(a, w / sum_w, L)      # length n
    return 2.0 * corr


# ----------------------------------------------------------------------
# Parametric candidate generator (used for the seed)
# ----------------------------------------------------------------------
def _parametric_candidates(n: int, rng: np.random.Generator):
    """Yield a handful of hand‑crafted families of probability vectors."""
    # 1. Exponential‑U shape
    for r in np.linspace(0.55, 0.9995, 30):
        i = np.arange(n, dtype=np.float64)
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 2. Power‑law (edge‑peaked)
    for p in np.linspace(-4.0, 4.0, 30):
        i = np.arange(1, n + 1, dtype=np.float64)
        base = i * (n + 1 - i) + 1e-12
        a = np.power(base, -p)
        yield a

    # 3. Cosine‑window (Hann) with exponent β
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    for beta in np.linspace(0.5, 4.0, 20):
        a = np.power(w, beta)
        yield a

    # 4. Beta‑distribution shapes (continuous analogue)
    xs = (np.arange(n) + 0.5) / n
    for a_param in np.linspace(0.3, 5.0, 10):
        for b_param in np.linspace(0.3, 5.0, 10):
            a = (xs ** (a_param - 1)) * ((1 - xs) ** (b_param - 1))
            yield a

    # 5. Random Dirichlet draws
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        yield a


# ----------------------------------------------------------------------
# Tiny uniform mixing (helps to move off shallow plateaux)
# ----------------------------------------------------------------------
def _tiny_uniform_mix(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for a few tiny α values and keep the best."""
    best_a, best_val = a, _objective_exact(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_onto_simplex(cand)   # ensures exact sum‑1
        val = _objective_exact(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s time budget.
    Returns a list of floats (the heights) that (approximately) minimise
    ``evaluate_sequence``.
    """
    # ----- reproducibility -------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ----- problem size ----------------------------------------------------
    n = 9000
    # FFT size that safely covers both convolutions (a*a) and a*w_rev
    L = _next_pow2(3 * n - 2)

    # ----- initial candidate ------------------------------------------------
    try:
        # reuse a previously discovered sequence if the harness provides it
        old = np.asarray(height_sequence_1, dtype=float)          # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_onto_simplex(a0)
        print("[init] Re‑used height_sequence_1")
    except Exception:
        best_a = None
        best_val = np.inf
        for cand in _parametric_candidates(n, rng):
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s <= 0.0:
                continue
            cand = cand / s
            val = _objective_exact(cand)
            if val < best_val:
                best_val, best_a = val, cand.copy()
        if best_a is None:                     # a very unlikely fallback
            best_a = np.full(n, 1.0 / n)
            best_val = _objective_exact(best_a)
        a0 = best_a
        print(f"[init] Parametric sweep gave initial value = {best_val:.6f}")

    # ----- optimisation parameters -----------------------------------------
    eta = 0.1                     # initial step size for gradient descent
    beta_start = 1.0
    beta_max = 30.0
    stagnation_limit = 5000      # iterations without improvement before re‑init
    no_improve = 0
    best_a = a0.copy()
    best_obj = _objective_exact(best_a)

    it = 0
    while time.time() < deadline:
        # ---- 1. current convolution and exact objective --------------------
        conv = _full_convolution(a0, L)                # length 2n‑1
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a0.copy()
            no_improve = 0
            print(f"[it {it:7d}] new best = {best_obj:.6f}")
        else:
            no_improve += 1

        # ---- 2. temperature (β) schedule ---------------------------------
        elapsed = time.time() - start_time
        total = deadline - start_time
        beta = beta_start + (beta_max - beta_start) * min(1.0, elapsed / total)

        # ---- 3. compute gradient of smoothed max -------------------------
        grad = _softmax_gradient(a0, beta, L)

        # ---- 4. gradient step with projection -----------------------------
        a_candidate = a0 - eta * grad
        a_candidate = _project_onto_simplex(a_candidate)

        # ---- 5. evaluate candidate (cheap exact objective) ---------------
        conv_cand = _full_convolution(a_candidate, L)
        cand_obj = 2.0 * n * conv_cand.max()

        # ---- 6. accept / reject + adaptive step size ----------------------
        if cand_obj < cur_obj:
            a0 = a_candidate
            eta *= 1.02                     # gently accelerate after a success
            if eta > 2.0:
                eta = 2.0
        else:
            # reject the move, keep old a0
            eta *= 0.5
            if eta < 1e-9:
                # re‑initialise around the best seen so far
                a0 = best_a + rng.normal(scale=0.01, size=n)
                a0 = np.maximum(a0, 0.0)
                a0 = _project_onto_simplex(a0)
                eta = 0.1
                no_improve = 0
                print(f"[it {it:7d}] re‑initialised after stagnation")

        # ---- 7. stagnation handling --------------------------------------
        if no_improve > stagnation_limit:
            a0 = best_a + rng.normal(scale=0.02, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 = _project_onto_simplex(a0)
            eta = 0.1
            no_improve = 0
            print(f"[it {it:7d}] perturb after {stagnation_limit} iters without improvement")

        it += 1

        # safety break a bit before the hard deadline
        if time.time() > deadline - 0.2:
            break

    # ----- tiny uniform mixing ---------------------------------------------
    best_a, best_obj = _tiny_uniform_mix(best_a, n)
    print(f"[final mix] best value after mixing = {best_obj:.6f}")

    # ----- final evaluation (the external harness will call it as well) -----
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
