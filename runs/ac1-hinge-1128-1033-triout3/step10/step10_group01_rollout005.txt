"""Optimizes a non‑negative step‑function to minimize
evaluate_sequence() using a pipeline:
1) parametric β‑family + random Dirichlet seeds,
2) projected sub‑gradient descent (PGD),
3) exponentiated‑gradient descent on the simplex,
4) intensive simulated‑annealing hill‑climbing.
All stages are time‑boxed and respect the 1000 s budget."""
import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a normalized sequence (sum == 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective = 2 * n * max(conv(a, a)), assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Two‑parameter Beta‑family sweep
# ----------------------------------------------------------------------
def _beta_family_best(n: int):
    """Search a∝i^{p‑1}(n‑i+1)^{q‑1} for the best objective."""
    best_a = None
    best_val = np.inf

    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float):
    a = a0.copy()
    size = 1 << ((2 * n - 2).bit_length())
    best_a = a.copy()
    best_val = _objective(a)

    step = 0.02
    min_step = 1e-12
    iter_cnt = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current convolution
        fx = np.fft.rfft(a, size)
        conv = np.fft.irfft(fx * fx, size)[:2 * n - 1]
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        kmax = int(conv.argmax())

        # sub‑gradient of the active entry
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # try a step
        a_candidate = a - step * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:
            a_candidate = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_candidate /= s

        # evaluate
        fx_new = np.fft.rfft(a_candidate, size)
        conv_new = np.fft.irfft(fx_new * fx_new, size)[:2 * n - 1]
        new_val = 2.0 * n * conv_new.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[PGD {iter_cnt:7d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    break
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.5
            no_improve += 1
            if step < min_step:
                break

        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            step = 0.02
            no_improve = 0
            print(f"[PGD {iter_cnt:7d}] stagnation – random perturb")
        iter_cnt += 1

    return best_a, best_val

# Exponentiated‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _eg_optimize(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float):
    """Deterministic mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    size = 1 << ((2 * n - 2).bit_length())
    eta0 = 0.6               # initial learning rate
    min_eta = 1e-6
    no_improve = 0
    IMPROVE_LIMIT = 120_000

    iter_cnt = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # convolution and max
        fx = np.fft.rfft(a, size)
        conv = np.fft.irfft(fx * fx, size)[:2 * n - 1]
        max_b = conv.max()
        # indices with value almost equal to max (relative tolerance 1e-8)
        rel_tol = 1e-8
        active = np.where(conv >= max_b - rel_tol * max_b)[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])

        # aggregate sub‑gradient over the active set
        grad = np.zeros_like(a)
        for k in active:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        grad = grad / float(active.size)

        # scheduled learning rate
        prog = (now - start_time) / (deadline - start_time)
        eta = eta0 * (1.0 - prog) + min_eta

        # exponentiated update
        a_new = a * np.exp(-eta * grad)
        s = a_new.sum()
        if s == 0.0 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-15:
            a = a_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[EG {iter_cnt:7d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    break
        else:
            no_improve += 1
            # occasionally try a small random perturbation to escape plateaus
            if no_improve > IMPROVE_LIMIT:
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                cur_val = _objective(a)
                no_improve = 0
                print(f"[EG {iter_cnt:7d}] stagnation – perturb")
        iter_cnt += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (as in the reference)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # fast update of the convolution vector
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

"""Optimises a non‑negative step function to minimise evaluate_sequence().

Pipeline (time‑boxed):
0️⃣  Warm‑start from any previously found solution (if available).
1️⃣  Initialise from a two‑parameter Beta family, a few Dirichlet draws and,
    if SciPy is present, Kaiser windows.
2️⃣  Projected sub‑gradient descent (PGD) on the simplex.
3️⃣  Exponentiated‑gradient (mirror‑descent) refinement.
4️⃣  Intensive simulated‑annealing hill‑climbing (same core as the reference).

All stages respect the 1000 s total budget.  The returned list is already
normalised (sum ≈ 1) and consists of non‑negative floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution via FFT (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a normalised sequence (Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Two‑parameter Beta family sweep
# ----------------------------------------------------------------------
def _beta_family_best(n: int):
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float):
    a = a0.copy()
    size = 1 << ((2 * n - 2).bit_length())
    best_a = a.copy()
    best_val = _objective(a)

    step = 0.02
    min_step = 1e-12
    iter_cnt = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        fx = np.fft.rfft(a, size)
        conv = np.fft.irfft(fx * fx, size)[:2 * n - 1]
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        kmax = int(conv.argmax())

        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        a_candidate = a - step * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:
            a_candidate = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_candidate /= s

        fx_new = np.fft.rfft(a_candidate, size)
        conv_new = np.fft.irfft(fx_new * fx_new, size)[:2 * n - 1]
        new_val = 2.0 * n * conv_new.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[PGD {iter_cnt:7d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    break
            step = min(step * 1.07, 0.5)
        else:
            step *= 0.5
            no_improve += 1
            if step < min_step:
                break

        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            step = 0.02
            no_improve = 0
            print(f"[PGD {iter_cnt:7d}] stagnation – random perturb")
        iter_cnt += 1

    return best_a, best_val

# Exponentiated‑gradient (mirror‑descent) on the simplex
# ----------------------------------------------------------------------
def _eg_optimize(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float):
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    size = 1 << ((2 * n - 2).bit_length())
    eta0 = 0.6
    min_eta = 1e-6
    no_improve = 0
    IMPROVE_LIMIT = 120_000
    iter_cnt = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        fx = np.fft.rfft(a, size)
        conv = np.fft.irfft(fx * fx, size)[:2 * n - 1]
        max_b = conv.max()
        rel_tol = 1e-8
        active = np.where(conv >= max_b - rel_tol * max_b)[0]
        if active.size == 0:
            active = np.array([int(conv.argmax())])

        grad = np.zeros_like(a)
        for k in active:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]
        grad = grad / float(active.size)

        prog = (now - start_time) / (deadline - start_time)
        eta = eta0 * (1.0 - prog) + min_eta

        a_new = a * np.exp(-eta * grad)
        s = a_new.sum()
        if s == 0.0 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_new /= s

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-15:
            a = a_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[EG {iter_cnt:7d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    break
        else:
            no_improve += 1
            if no_improve > IMPROVE_LIMIT:
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                cur_val = _objective(a)
                no_improve = 0
                print(f"[EG {iter_cnt:7d}] stagnation – perturb")
        iter_cnt += 1

    return best_a, best_val

# Intensive simulated‑annealing hill‑climbing (reference implementation)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val
