"""
Propose a low‑objective step function.

Algorithm
---------
1. **Rich parametric initialisation** – explore a handful of classic families
   (exponential, power‑law, beta, sine‑power, logistic, piecewise‑constant)
   together with a set of random Dirichlet draws.  All candidates are
   normalised to sum = 1 and the best one (smallest true objective) is kept.

2. **Adam on a smooth max‑convolution surrogate** – re‑parameterise the
   vector by logits `w` (soft‑max → simplex) and minimise the log‑sum‑exp
   approximation of `max(a*a)`.  The temperature `eps` is annealed from
   `1e‑1` to `1e‑13` while the learning‑rate decays from `0.5` to `0.001`.
   The gradient of the surrogate is obtained analytically via an FFT‑based
   convolution (`grad_a = 4·n·conv(a, softmax(b)/eps reversed)`).  Adam
   updates are performed with bias correction; occasional jitter and a
   restart on stagnation help escape shallow minima.

3. **Exact active‑set refinement** – after the Adam phase the true
   auto‑convolution `b = a*a` is computed.  All positions whose value lies
   within a tiny tolerance of the current maximum form the *active set*.
   For each index `i` we compute a weight `w_i = Σ_{k∈active} a_{k‑i}`.
   Mass is transferred from the index with largest weight to the one with
   smallest weight (a fraction of the source mass).  The move is accepted
   only if the exact objective improves; otherwise the step size shrinks.
   This loop drives the peak of the convolution down further.

4. **Final polishing** – a tiny convex combination with the uniform
   distribution (`α∈{0.001,0.003,0.005,0.010,0.020}`) is tried; such a
   smoothing can remove residual spikes.

The total time budget is split roughly 80 % for Adam and 20 % for the
exact refinement.  The whole pipeline respects the 1000 s limit,
periodically checks the wall‑clock and prints progress.  The returned
list sums to 1, contains only non‑negative numbers and can be fed directly
to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of a and b via FFT (both 1‑D)."""
    out_len = a.shape[0] + b.shape[0] - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant; we keep sum(a)=1)
# ----------------------------------------------------------------------
def objective(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(conv) when sum(a)=1."""
    n = a.shape[0]
    max_b = full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for a quick good starting point
# ----------------------------------------------------------------------
def initial_candidate(n: int, rng: np.random.Generator):
    """Return best vector among a collection of hand‑crafted families."""
    best_a = np.full(n, 1.0 / n)
    best_val = objective(best_a)

    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.shape != (n,):
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0
    # 1. Exponential tails
    for r in np.linspace(0.995, 0.99999, 12):
        try_candidate(r ** i)                              # decreasing
        try_candidate(r ** i + r ** (n - 1 - i))           # symmetric U‑shape
    # 2. Power‑law tails
    for p in np.linspace(0.2, 6.0, 16):
        try_candidate(i1 ** (-p))
        try_candidate(i1 ** (-p) + (n - i) ** (-p))
    # 3. Beta distributions
    for a_param in np.linspace(0.2, 6.0, 6):
        for b_param in np.linspace(0.2, 6.0, 6):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            try_candidate(a)
    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 10):
        try_candidate(base_sin ** p)
    # 5. Piecewise constant “high‑at‑edges”
    for k_frac in np.linspace(0.005, 0.12, 10):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.35, 10):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            try_candidate(a)
    # 6. Logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.002, 0.04, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        try_candidate(a)
        try_candidate(np.flip(a))
    # 7. Random Dirichlet draws (various concentrations)
    for concentration in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(30):
            a = rng.dirichlet(np.full(n, concentration))
            try_candidate(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimisation on a smooth log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def adam_phase(a0: np.ndarray,
               start: float,
               deadline: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Adam on log‑sum‑exp surrogate of the max‑convolution."""
    n = a0.shape[0]
    # initialise logits (soft‑max parametrisation)
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.5, 0.001
    eps_start, eps_end = 1e-1, 1e-13

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    it = 0
    while time.time() < deadline:
        # current point on simplex
        w_shift = w - np.max(w)          # numeric stability
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = objective(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # schedule for eps and lr (linear in log‑time)
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max -----
        b_full = full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()           # soft‑max over convolution entries

        # gradient of surrogate w.r.t a:
        #   ∂/∂a 2·n·∑ p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_full = conv_fft(a, p_rev)       # length = n + (2n-1) -1 = 3n-2
        # slice corresponding to indices 0 … n‑1 (then reverse)
        grad_a = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]

        # back‑prop through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update (bias‑corrected)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.02, size=n)

        # restart if stuck for a long time
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact active‑set refinement (mass transfer guided by active indices)
# ----------------------------------------------------------------------
def refine_exact(a0: np.ndarray,
                 start: float,
                 deadline: float,
                 best_a: np.ndarray,
                 best_val: float) -> tuple[np.ndarray, float]:
    """Iteratively move mass from high‑weight to low‑weight indices."""
    n = a0.shape[0]
    a = a0.copy()
    step_factor = 0.35               # initial fraction of source mass to move
    decay = 0.9993
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    it = 0

    while time.time() < deadline:
        b = full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Refine {it:7d}] new best = {best_val:.6f}")
            if best_val <= 1.503:
                break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # stagnation – stop the refinement
            break

        # active set: indices whose value is within a tiny tolerance of the peak
        eps_active = max_b * 5e-7
        active = np.where(b >= max_b - eps_active)[0]
        if active.size == 0:
            break

        # compute weight w_i = Σ_{k∈active} a_{k-i}
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or a[src] < 1e-15:
            # fallback to a simple random move
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        delta = a[src] * step_factor
        if delta <= 0.0:
            step_factor *= 0.7
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        new_val = objective(a_candidate)

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
        else:
            # reject – shrink step size
            step_factor *= 0.7

        step_factor = max(step_factor * decay, 1e-12)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small uniform mixing (final polishing)
# ----------------------------------------------------------------------
def mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few tiny convex combinations with the uniform distribution."""
    best_a, best_val = a, objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
