Propose a step‑function sequence that improves the current bound (≈1.50878).

Approach
--------
1. **Rich initialisation** – try a handful of parametric families
   (uniform, exponential, power‑law, beta, random Dirichlet) and keep the
   best after normalising the sum to 1.

2. **Simulated annealing** – repeatedly move a small amount of mass from a
   “source’’ index to a “target’’ index, updating the full convolution
   incrementally.  A Metropolis criterion with a linearly annealed temperature
   allows occasional uphill moves, helping escape local minima.

   *Targeted moves* (≈60 % of iterations) pick the source proportionally to
   its contribution to the currently maximal convolution entry and pick a
   target that contributes little to that entry.

3. **Exact active‑set refinement** – after the time budget for annealing is

import time

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector with sum(a)=1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Mix a tiny amount of the uniform distribution (helps smooth spikes)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a small uniform component."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing with targeted mass moves
# ----------------------------------------------------------------------
def _anneal(a0: np.ndarray, n: int, start: float, deadline: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style optimisation of the exact objective."""
    cur_a = a0.copy()
    cur_b = _full_convolution(cur_a)
    cur_val = 2.0 * n * cur_b.max()

    best_a, best_val = cur_a.copy(), cur_val

    # Schedules (linear annealing)
    max_frac_start, max_frac_end = 0.30, 0.001
    temp_start, temp_end = 0.02, 1e-7

    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        temp = temp_start * (1.0 - progress) + temp_end * progress

        # --------------------------------------------------------------
        # Choose move (targeted with probability 0.6)
        # --------------------------------------------------------------
        if np.random.rand() < 0.6:
            # focus on the current maximal convolution entry
            k_max = int(np.argmax(cur_b))

            # contribution of each index to that entry
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib = np.zeros_like(cur_a)
            contrib[mask] = cur_a[mask] * cur_a[idx[mask]]
            total_contrib = contrib.sum()
            if total_contrib > 0.0:
                probs_i = contrib / total_contrib
                i = int(np.random.choice(n, p=probs_i))
            else:
                i = int(np.argmax(cur_a))

            # choose a target that is weakly coupled to the active entry
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = cur_a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = int(np.random.choice(n, p=probs_j))
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            # completely random pair
            i = np.random.randint(0, n)
            j = np.random.randint(0, n - 1)
            if j >= i:
                j += 1

        if cur_a[i] <= 1e-20:
            continue

        delta = cur_a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # Incremental update of the full convolution (exact)
        # --------------------------------------------------------------
        b_new = cur_b.copy()
        b_new[j:j + n] += 2.0 * delta * cur_a
        b_new[i:i + n] -= 2.0 * delta * cur_a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temp > 0.0:
                prob = np.exp(-(new_val - cur_val) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_b = b_new
            cur_val = new_val

            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, cur_a.copy()
                if it % 1000 == 0:
                    print(f"[Anneal {it:8d}] new best = {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact active‑set refinement (deterministic mass transfer)
# ----------------------------------------------------------------------
def _refine_exact(a0: np.ndarray, n: int, start: float, deadline: float) -> tuple[np.ndarray, float]:
    """Deterministic refinement using active‑set weights."""
    a = a0.copy()
    best_a, best_val = a.copy(), _objective_value(a)

    step_factor = 0.4            # initial fraction of source mass to move
    decay = 0.9997
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[Refine {it:8d}] new best = {best_val:.6f}")

        # active set – entries within a tiny tolerance of the max
        eps = max_b * 5e-7
        active = np.where(b >= max_b - eps)[0]
        if active.size == 0:
            break

        # compute weight w_i = Σ_{k∈active} a_{k‑i}
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += a[k - idx]

        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or a[src] <= 0.0:
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        delta = a[src] * step_factor
        if delta <= 0.0:
            break

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        b_new = _full_convolution(a_candidate)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
        else:
            step_factor *= 0.7  # shrink step if not successful

        step_factor = max(step_factor * decay, 1e-12)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the given time budget.
    Returns a list of floats (Σ ≈ 1) that can be fed to `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – a few thousand entries is a good trade‑off
    n = 9000

    print("[Init] generating initial candidate")
    a0, val0 = _initial_candidate(n)
    print(f"[Init] initial objective = {val0:.6f}")

    # ------------------------------------------------------------------
    # Phase A – simulated annealing (majority of the budget)
    # ------------------------------------------------------------------
    anneal_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _anneal(a0, n, start=start_time, deadline=anneal_deadline)
    print(f"[Anneal] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – exact active‑set refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _refine_exact(best_a, n,
                                     start=anneal_deadline,
                                     deadline=deadline)
    print(f"[Refine] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – tiny uniform mixing (polishing)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Mix] best after uniform mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (via the provided wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
