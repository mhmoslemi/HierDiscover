"""
A fast multi‑phase search for a non‑negative step function that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

(the scaling of a is irrelevant, we normalise sum(a)=1).

Strategy
--------
1. **Coarse parametric sweep** on a modest size (n0≈2000).  Families
   include exponential U‑shapes, power‑law, beta‑type, cosine windows and
   a few Dirichlet draws.  The ten best candidates are kept.

2. **Projected sub‑gradient descent** (mirror‑descent on the simplex)
   from each of those seeds.  The sub‑gradient of the max‑convolution
   entry is computed analytically, a step is taken, the vector is
   projected back onto the probability simplex.  Steps are adaptively
   reduced; occasional random jitters escape plateaus.

3. **Upscaling** – the best low‑dimensional solution is linearly
   interpolated to a larger dimension (N≈8000) where a final short
   refinement is performed.

4. **Uniform mixing** – a tiny convex combination with the uniform
   distribution often lowers the peak further.

All phases respect the 1000 s overall wall‑clock budget.  The final
sequence (a list of Python floats) is returned; the evaluation harness
will call `evaluate_sequence` on it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using an FFT of the next power of two."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return _full_convolution(a).max()


# ----------------------------------------------------------------------
# Objective (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Compute 2·n·max(conv(a,a)).  The sum of a is assumed to be 1."""
    n = a.shape[0]
    max_b = _max_convolution(a)
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Helper: keep a sorted list of the K best candidates seen so far
# ----------------------------------------------------------------------
def _update_best(best_list, val, vec, K):
    """Insert (val,vec) into best_list (sorted low→high) keeping size ≤K."""
    if len(best_list) < K:
        best_list.append((val, vec.copy()))
        best_list.sort(key=lambda x: x[0])
    else:
        if val < best_list[-1][0]:
            best_list[-1] = (val, vec.copy())
            best_list.sort(key=lambda x: x[0])
    return best_list


# ----------------------------------------------------------------------
# Parameter‑family generator (used for the coarse sweep)
# ----------------------------------------------------------------------
def _parametric_candidates(n: int, rng: np.random.Generator):
    """Yield a stream of candidate vectors built from simple parametric families."""
    i = np.arange(n, dtype=np.float64)

    # 1. Symmetric exponential (U‑shaped) a_i ∝ r**i + r**(n‑1‑i)
    for r in np.linspace(0.55, 0.9999, 40):
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 2. Symmetric power‑law a_i ∝ ((i+1)*(n-i))**(-p)
    for p in np.linspace(-4.0, 4.0, 40):
        base = (i + 1) * (n - i) + 1e-12
        a = np.power(base, -p)
        yield a

    # 3. Beta‑type: a_i ∝ (i+1)^{α‑1} * (n‑i)^{β‑1}
    alphas = np.linspace(0.2, 3.0, 6)
    betas  = np.linspace(0.2, 3.0, 6)
    for α in alphas:
        for β in betas:
            a = np.power(i + 1, α - 1.0) * np.power(n - i, β - 1.0)
            yield a

    # 4. Cosine‑window with exponent β
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))   # Hann window
    for β in np.linspace(0.5, 4.0, 20):
        a = np.power(w, β)
        yield a

    # 5. Random Dirichlet draws (few)
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        yield a


# ----------------------------------------------------------------------
# Projected sub‑gradient / mirror descent on the probability simplex
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Perform many projected sub‑gradient steps starting from a0.
    Returns the best vector seen (sum = 1, non‑negative).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    # step‑size schedule
    eta = 0.5
    decay = 0.99997
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 8000

    while time.time() < deadline:
        # compute full convolution once
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            # occasional progress report
            if it % 2000 == 0:
                print(f"[SGD {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # index of the maximal convolution entry
        k_max = int(np.argmax(conv))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        # sub‑gradient of max‑convolution entry (see analysis)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # gradient step + projection onto simplex
        a -= eta * grad
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            # fallback to uniform if everything collapsed
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # decay step size
        eta *= decay

        # occasional jitter to escape plateaus
        if rng.random() < 0.002:
            jitter = rng.normal(scale=0.006, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a /= a.sum()

        # restart after too many non‑improving steps
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            eta = 0.5
            no_improve = 0
            print(f"[SGD {it:7d}] restart after stagnation")

        it += 1
        # safety break
        if time.time() > deadline - 0.1:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (often reduces the peak slightly)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Blend a little with the uniform distribution and keep the better."""
    best_a = a
    best_val = _objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.006, 0.010, 0.015, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low value of
    evaluate_sequence().  The routine respects a 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Phase 0 – choose a modest base size for the cheap sweep
    # ------------------------------------------------------------------
    n0 = 2000                     # fast enough for many candidates
    print(f"[init] base size n0 = {n0}")

    # try to reuse a previously found solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n0)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] Re‑used height_sequence_1 for warm start")
        candidates = [( _objective(a0), a0 )]
    except Exception:
        candidates = []

    # ------------------------------------------------------------------
    # Phase 1 – coarse parametric sweep (retain the ten best seeds)
    # ------------------------------------------------------------------
    TOP_K = 10
    best_list = []   # list of (value, vector)

    for cand in _parametric_candidates(n0, rng):
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s
        val = _objective(cand)
        best_list = _update_best(best_list, val, cand, TOP_K)

    # If we also have a warm start, insert it:
    if candidates:
        val0, vec0 = candidates[0]
        best_list = _update_best(best_list, val0, vec0, TOP_K)

    print(f"[sweep] retained {len(best_list)} best candidates after coarse sweep")
    for i, (v, _) in enumerate(best_list):
        print(f"   candidate {i:02d}:  obj = {v:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – local refinement on the small problem (sub‑gradient descent)
    # ------------------------------------------------------------------
    refined = []
    for idx, (val, vec) in enumerate(best_list):
        # allocate a share of the remaining time to each seed
        remaining = deadline - time.time()
        if remaining < 2.0:
            break
        # give each seed at most 0.12 * total budget
        seed_deadline = time.time() + min(remaining * 0.12, 45.0)
        print(f"[refine] start sub‑gradient on seed {idx} (deadline in {seed_deadline - time.time():.1f}s)")
        a_ref, v_ref = _subgradient_descent(vec, seed_deadline, rng)
        refined.append((v_ref, a_ref))
        print(f"[refine] seed {idx} finished: best = {v_ref:.6f}")

    # include also the raw sweep candidates that were not refined (in case time ran out)
    if not refined:
        refined = best_list

    # pick the overall best from the refined pool
    refined.sort(key=lambda x: x[0])
    best_a_small, best_val_small = refined[0][1], refined[0][0]
    print(f"[best_small] best value after refinement = {best_val_small:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – upscale to a larger dimension and final fine‑tuning
    # ------------------------------------------------------------------
    N_large = 8000
    print(f"[upscale] interpolating to larger size N = {N_large}")
    x_small = np.linspace(0.0, 1.0, n0)
    x_large = np.linspace(0.0, 1.0, N_large)
    a_large = np.interp(x_large, x_small, best_a_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()

    # Allocate remaining time for the final refinement (≈30 % of budget)
    final_deadline = deadline - 0.5
    print(f"[final] start sub‑gradient on large problem (will stop at {final_deadline - time.time():.1f}s left)")
    a_final, val_final = _subgradient_descent(a_large, final_deadline, rng)

    # ------------------------------------------------------------------
    # Phase 4 – tiny uniform mixing and final evaluation
    # ------------------------------------------------------------------
    a_final, val_final = _mix_with_uniform(a_final, a_final.shape[0])
    print(f"[final] objective after mixing = {val_final:.6f}")

    # Optional sanity check using the provided evaluate_sequence
    try:
        reported = evaluate_sequence(a_final.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_final.tolist()
