"""
Optimisation of max‑convolution of a non‑negative step function.

The objective is  2·n·max(conv(a,a)) / (∑a)²  (scale‑invariant).
We work on the probability simplex (∑a = 1, a ≥ 0).

Algorithm
---------
1. **Multiple restarts** – different simple seeds (uniform, sinusoid,
   exponential).  The best vector found across all restarts is kept.

2. **Smoothed‑max (mirror) descent** – minimise a smooth surrogate of
   the max‑convolution.
   * For large smoothing ε we use a soft‑max (log‑sum‑exp) weighting.
   * In the final stage we focus on the true peak by weighting only the
     top‑k convolution entries (hard‑max approximation).
   * Updates are multiplicative (mirror‑descent) and a Euclidean projection
     onto the simplex keeps the iterates feasible.

   The schedule contains many more iterations than the reference code
   (≈15000 FFT steps) so the solution can converge further.

3. **Anti‑symmetric re‑ordering** – after each ε‑stage we reorder the
   vector by pairing the largest values with the smallest ones.  The
   rearrangement inequality guarantees that this never increases the
   maximal convolution entry.

4. **Hill‑climbing refinement** – cheap local search (random swaps,
   tiny mass transfers) that is run after every descent run and finally
   on the overall best vector.

All heavy work (FFT‑based gradient steps) is performed without calling
`evaluate_sequence`; the true objective is queried only occasionally to
track progress.  The routine respects a 1000 s budget and returns the
best sequence it has discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·b_k w.r.t. a,
    where b = a*a (linear convolution) and p ≥ 0, Σp = 1.

    Result: g_i = Σ_k p_k·a_{k‑i}   (out‑of‑range indices are zero).
    Computed with a single FFT.
    """
    n = a.shape[0]
    # reverse the weight vector (convolution with a corresponds to correlation)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # keep only the segment that corresponds to indices n‑1 … 2n‑2
    seg = conv[n - 1 : 2 * n - 1]
    g = seg[::-1]                            # reverse back
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                         # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]                   # large → left side
        b[-1 - i] = s[i]                   # small → right side
    if n % 2:
        b[half] = s[half]
    return b

    Perform a long mirror‑descent run with a multi‑stage smoothing schedule.
    Returns the best vector found (as an ndarray) and its true objective.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent init] objective = {best_val:.6f}")

    # schedule: (ε, number of inner iterations, initial η)
    schedule = [

        (0.2, 1200, 0.8),

        (0.05,2400, 0.6),
        (0.02,3000, 0.5),
        (0.01,3000, 0.4),
        (None,2000, 0.3)   # hard‑max (top‑k) stage

            else:

                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[reorder eps {eps} it {it}] improved to {best_val:.6f}")

        # final re‑ordering for the stage (cannot increase the max)
        a = _anti_symmetric_reorder(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Light local search (hill climbing)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float, n: int):
    """
    Random swaps / tiny mass transfers, kept only if they strictly improve
    the true objective.  The loop runs until the time budget is exhausted.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        # decide move type
        if rng.random() < 0.3:                     # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # transfer a small amount
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.05 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] improved to {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem dimension – a few thousand coefficients give a good trade‑off
    n = 8000

    best_a = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Multiple restarts (different initialisations)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 200 and restart < 12:
        restart += 1

        # ----------------------------------------------------------------
        # Choose initial vector
        # ----------------------------------------------------------------
        if restart == 1 and 'height_sequence_1' in globals():
            # initialise from the previously found construction (if available)
            old = np.asarray(globals()['height_sequence_1'], dtype=float)
            if old.shape[0] != n:
                # interpolate to required length
                old_x = np.linspace(0.0, 1.0, old.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a = a0 / s if s > 0.0 else np.full(n, 1.0 / n)
            init_name = "previous"
        else:
            strat = rng.choice(["uniform", "sin", "exp"])
            if strat == "uniform":
                a = np.full(n, 1.0 / n)
                init_name = "uniform"
            elif strat == "sin":
                idx = np.arange(n) + 0.5
                a = np.sin(np.pi * idx / n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                init_name = "sinusoid"
            else:   # exponential random
                a = rng.exponential(scale=1.0, size=n)
                a = a / a.sum()
                init_name = "exponential"

        # ----------------------------------------------------------------
        # Run the long mirror‑descent phase
        # ----------------------------------------------------------------
        a_opt, val_opt = _refined_descent(a, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[global] new best {best_val:.6f} (restart {restart}, init={init_name})")

        # ----------------------------------------------------------------
        # Light hill‑climbing after each descent (if time permits)
        # ----------------------------------------------------------------
        if time.time() < deadline - 30:
            a_ref, val_ref = _hill_climb_refine(best_a, best_val,
                                                deadline, n)
            if val_ref < best_val:
                best_val = val_ref
                best_a = a_ref.copy()
                print(f"[global] post‑hill improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the overall best vector
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5:
        best_a, best_val = _hill_climb_refine(best_a, best_val,
                                              deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
