def _gradient_weighted(a, w_rev):
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1
    seg = conv[start:start + n]
    return 2.0 * seg

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5

    # define n (target length)
    n_target = 12000

    # init a
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)
        if a0.shape[0] != n_target:
            # linear interpolation
            old_x = np.linspace(0, 1, a0.shape[0])
            new_x = np.linspace(0, 1, n_target)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        a0 = rng.exponential(scale=1.0, size=n_target)

    a0 = a0 / a0.sum()
    best_a = a0.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n_target} eval={best_val}")

    # Phase 0: random restarts
    for i in range(6):
        if time.time() >= deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n_target)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand {i}] best={best_val}")

    # Phase 1: smoothed max descent
    a = best_a.copy()
    it = 0
    max_iters = 200_000
    last_improve = 0
    # hyperparameters
    alpha0 = 0.6
    beta_start = 1.0
    beta_end = 2000.0
    beta_ramp = 50_000
    reorder_interval = 5000

    while time.time() < deadline - 30 and it < max_iters:
        # compute conv and max
        conv = _full_convolution(a)
        max_c = conv.max()

        # compute smoothing weight
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp)

            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                last_improve = it

            i, j = rng.integers(0, n_target, size=2)

                new_a[i], new_a[j] = new_a[j], new_a[i]
