"""
Non‑negative step‑function optimizer (multi‑start simulated‑annealing).

The algorithm proceeds as follows:

1.  Choose a moderate problem size (n ≈ 5 000).  A smaller n makes
    each iteration cheap while the objective 2·n·max(conv) depends only on
    the shape of the distribution (≈ 2·C for a distribution with
    max_b ≈ C/n).

2.  Build a small pool of promising initial candidates:
    * interpolate the previous best solution (if available);
    * a dense sweep over the 2‑parameter Beta family;
    * several Dirichlet draws with different concentration parameters.
    The best of these seeds is kept as the current global best.

3.  Repeatedly launch a short hill‑climbing run (simulated‑annealing
    style) from a perturbed copy of the global best.  The move rule
    transfers a random amount of mass from an index that contributes a
    lot to the *few* largest convolution entries to an index that
    contributes little.  The fraction of transferred mass and the
    Metropolis temperature decay slowly, allowing a long exploration
    phase.

4.  Whenever a run discovers a better sequence the global best is
    updated.  After the time budget is exhausted the best sequence is
    returned and finally evaluated with ``evaluate_sequence``.

The code respects all constraints of the task and runs under the
1000 s budget on a single core.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution (FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a simplex‑normalised a."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Project onto probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Richer Beta‑family sweep (more parameter values than the previous version)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a 2‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}
    over a broad grid of (p,q).  Returns best vector and its objective."""
    best_a = None
    best_val = np.inf

    # a mixture of linear, logarithmic and geometric grids
    p_vals = np.unique(np.concatenate([
        np.linspace(0.1, 1.0, 9),                 # fine near 0‑1
        np.geomspace(1.0, 12.0, 12)[1:],         # 1 → 12 geometric
        np.linspace(12.0, 30.0, 6)                # very heavy tails
    ]))
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing move
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly moves a small amount of mass from an
    index heavily involved in the top‑K convolution entries to an index
    with little involvement.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule parameters (slow decay → long exploration)
    max_frac_start, max_frac_end = 0.90, 0.001   # move fraction
    temp_start, temp_end = 0.05, 1e-8            # Metropolis temperature
    top_k = 5                                    # number of max entries to watch

    no_improve = 0
    IMPROVE_LIMIT = 180_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear interpolation of schedule variables
        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # ------------------------------------------------------------------
        # Identify the K largest convolution entries and their per‑index
        # contributions.
        # ------------------------------------------------------------------
        # np.argpartition is O(n) and gives the indices of the top‑K values.
        top_idxs = np.argpartition(b, -top_k)[-top_k:]
        contrib = np.zeros(n, dtype=np.float64)

        for k_max in top_idxs:
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib[mask] += a[mask] * a[idx[mask]]

        total_contrib = contrib.sum()

        # source index – weighted by contribution, occasional fallback
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – biased towards low contribution
        tgt_weights = np.zeros(n, dtype=np.float64)
        mask_pos = contrib > 0
        tgt_weights[mask_pos] = 1.0 / (contrib[mask_pos] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (different from source)
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # amount of mass to move (fraction of source)
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # ------------------------------------------------------------------
        # Incremental convolution update (O(n) arithmetic)
        # ------------------------------------------------------------------
        b_new = b.copy()

        # add contribution of the transferred mass at the target position
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        # remove contribution at the source position
        b_new[i_src:i_src + n] -= 2.0 * delta * a

        # diagonal / double‑count corrections
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # Stagnation handling – random perturbation around the current best
        # ------------------------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a_interp = np.interp(new_x, old_x, old)
        a_interp = np.maximum(a_interp, 0.0)
        a_interp /= a_interp.sum()
        seeds.append(a_interp)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # b) a dense sweep over the Beta family
    try:
        beta_a, beta_val = _beta_family_best(n)
        seeds.append(beta_a)
        print("[init] beta‑family sweep completed")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # c) a handful of Dirichlet draws with diverse concentration
    conc_levels = [0.05, 0.2, 0.8, 1.5, 4.0]
    for conc in conc_levels:
        a_dir = np.random.dirichlet(np.full(n, conc))
        seeds.append(a_dir)

    # evaluate all collected seeds and keep the best one
    for a in seeds:
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[init] seed value = {best_val:.6f}")

    # safety fallback (should never be needed)
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] fallback random seed, value =", best_val)

    # ------------------------------------------------------------------
    # 1️⃣  Repeated multi‑start hill‑climbing until time runs out
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break
        remaining = deadline - now
        if remaining < 8.0:               # not enough time for another run
            break

        # start from a perturbed copy of the current global best
        a_start = best_a.copy()
        perturb = np.random.normal(scale=0.025, size=n)
        a_start = np.maximum(a_start + perturb, 0.0)
        a_start /= a_start.sum()

        # allocate at most 30 s for a single hill‑climb attempt
        sub_deadline = now + min(30.0, remaining)

        a_res, val_res = _hillclimb_anneal(
            a_start,
            n,
            deadline=sub_deadline,
            start_time=now,
            best_a=best_a,
            best_val=best_val,
            target_val=1.503,
        )

        if val_res < best_val - 1e-12:
            best_val, best_a = val_res, a_res.copy()
            print(f"[global] new global best = {best_val:.6f}")

        if best_val <= 1.503:
            print("[global] target reached, stopping early")
            break

    # ------------------------------------------------------------------
    # 2️⃣  Final evaluation (for logging purposes only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
```

This implementation builds on the previous hill‑climbing idea but
introduces several refinements:

* a richer Beta‑family search,
* multiple diverse seeds (including Dirichlet draws),
* a top‑K‑aware source/target selection that looks at the few largest
  convolution entries rather than a single one,
* a slower decay of the move‑size and temperature to keep the search
  active for the whole budget,
* periodic random perturbations to escape stagnation,
* a multi‑start outer loop that repeatedly restarts from a noisy copy of
  the best‑so‑far solution.

All helpers are top‑level functions, no closures are used, and the
routine respects the 1000 s time limit while aiming to beat the target
bound of 1.503.
