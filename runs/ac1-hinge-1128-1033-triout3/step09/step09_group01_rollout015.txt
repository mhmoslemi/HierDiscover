    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 8000

    # Get starting solution: either from previous best (height_sequence_1) interpolated, else Beta family sweep.
    # Acquire initial a (norm sum=1)

    best_a = a0
    best_val = objective(best_a)

    iteration = 0

    eta_start = 0.5  # this will be multiplied with typical gradient; but we need scaling relative to gradient magnitude.
    eta_end = 0.01
    # We'll also consider using line search: compute new objective after step; if not improved we shrink step.

    while time.time() < deadline:

        c = _full_convolution(a)
        max_c = c.max()
        k = np.argmax(c)

"""
Non‑negative step‑function optimizer (projected gradient + refined hill‑climb).

The routine builds several diverse starts (previous solution, an extended
Beta‑family sweep and a handful of Dirichlet draws) for a problem size
n≈12 000.  Each start is processed by a simple projected‑gradient descent
that reduces the maximal self‑convolution entry.  The gradient is the
sub‑gradient of the current maximal convolution bin; after a gradient step
the vector is projected onto the probability simplex (Σa=1, a≥0).  A
momentum term accelerates convergence.  When the descent stagnates a
small random perturbation is added to escape local minima.

The best vector found by the gradient phase is finally polished with the
intensive simulated‑annealing hill‑climber from the previous version.
All work finishes well before the 1000 s budget, after which the best
sequence is returned (as a plain list of floats).
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# 1.  Convolution helpers (FFT – full linear convolution)
# --------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of `x` with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# --------------------------------------------------------------
# 2.  Objective (assumes Σa = 1)
# --------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# --------------------------------------------------------------
# 3.  Projection onto the probability simplex (Σx = 1, x ≥ 0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# 4.  Extended Beta‑family sweep (cheap parametric search)
# --------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # a richer grid – more points and also very small exponents
    p_vals = np.unique(np.concatenate([
        np.geomspace(0.01, 0.5, 12),          # very heavy edges
        np.linspace(0.1, 1.0, 9),
        np.geomspace(1.0, 12.0, 12)
    ]))
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# --------------------------------------------------------------
# 5.  Projected gradient descent with momentum
# --------------------------------------------------------------
def _grad_descend(a0: np.ndarray,
                  deadline: float,
                  start_time: float,
                  max_iters: int = 200_000,
                  step_start: float = 1.0,
                  step_end: float = 1e-3,
                  momentum: float = 0.9,
                  stagnation_patience: int = 3_000) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent on the objective 2·n·max(conv(a,a)).
    Uses the sub‑gradient of the current maximal convolution entry and a
    Euclidean projection onto the simplex.  Momentum is employed to
    accelerate convergence.  When no improvement is seen for a while a
    tiny random perturbation is added to escape plateaus.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a, best_val = a.copy(), cur_val

    # momentum buffer
    v = np.zeros_like(a)

    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iters:
        # linear schedule for the step size
        prog = (time.time() - start_time) / (deadline - start_time)
        step = step_start * (step_end / step_start) ** prog

        # compute current convolution and the index of the maximal bin
        b = _full_convolution(a)
        max_c = b.max()
        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal bin
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        # derivative: a_i + a_{k-i} (or 2·a_i when i == k-i)
        grad[mask] = a[mask] + a[idx[mask]]

        # momentum update
        v = momentum * v - step * grad
        a_new = a + v
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < best_val - 1e-12:
            best_val, best_a = new_val, a_new.copy()
            print(f"[PGD {it:6d}] new best = {best_val:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # if we are stuck for a while, add a tiny jitter and reset momentum
        if no_improve >= stagnation_patience:
            noise = np.random.normal(scale=1e-4, size=n)
            a_new = np.maximum(best_a + noise, 0.0)
            a_new /= a_new.sum()
            v.fill(0.0)
            no_improve = 0
            print(f"[PGD {it:6d}] stagnation – perturb")

        a = a_new
        it += 1
    return best_a, best_val


# --------------------------------------------------------------
# 6.  Intensive hill‑climbing (same as the previous version, but
#     re‑implemented here to avoid closure capture)
# --------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local stochastic search that repeatedly moves a small amount of mass
    from an index that heavily contributes to the current maximal
    convolution entry to one that contributes little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        # contribution of each position to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – weighted by contribution
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – biased towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val
