# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (useful for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    The result has length len(x)+len(y)-1 and is returned as a real array.
    """
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: len(x) + len(y) - 1]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0.0, out=conv)
    return conv


def _maxconv_objective(a: np.ndarray) -> tuple[float, float]:
    """
    Returns (max_convolution, objective_value) for a given non‑negative vector.
    The objective is the same as evaluate_sequence().
    """
    n = a.shape[0]
    conv = _fft_convolve(a, a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    # sum(a) is expected to be 1, but keep the formula general
    s = a.sum()
    if s < 0.01:        # same protection as evaluate_sequence()
        return max_c, np.inf
    obj = float(2.0 * n * max_c / (s * s))
    return max_c, obj


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce a new vector where the largest values are placed opposite the
    smallest ones (pairwise).  This often flattens the self‑convolution.
    """
    n = a.shape[0]
    s = np.sort(a)  # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]          # big → left side
        b[-1 - i] = s[i]          # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` within the allotted time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # a few seconds margin

    # ------------------------------------------------------------------
    # Problem dimension – a larger n gives more freedom but costs more FFT.
    # 12000 is a comfortable compromise for a 1000 s budget.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise a few candidate vectors and keep the best one.
    # ------------------------------------------------------------------
    candidates = []

    # 1) historic construction, if available
    seq1 = globals().get("height_sequence_1")
    if seq1 is not None:
        try:
            a = np.asarray(seq1, dtype=float)
            if a.shape[0] != n:
                # linear interpolation to the target length
                old_x = np.linspace(0.0, 1.0, a.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                a = np.interp(new_x, old_x, a)
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a)
        except Exception:
            pass

    # 2) uniform vector (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # 3) a few random Dirichlet draws
    for _ in range(3):
        a = rng.exponential(scale=1.0, size=n)
        a = a / a.sum()
        candidates.append(a)

    # pick the best among the initial candidates
    best_a = None
    best_obj = np.inf
    for a in candidates:
        _, obj = _maxconv_objective(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    print(f"[init] n={n}, best initial objective = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    size_fft = _next_pow2(3 * n - 2)          # size sufficient for all convolutions
    beta_max = 2000.0                         # final smoothing strength
    beta_ramp = 100_000                       # iterations to reach beta_max
    max_iter = 400_000
    lr = 0.1                                 # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999                # Adam moments
    eps = 1e-8

    # initialise Adam state on log‑probability parameters
    u = np.log(best_a + 1e-30)                # log‑softmax parameters
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    last_improve_iter = 0
    it = 0

    # ------------------------------------------------------------------
    # Main loop – Adam on the smooth max of the self‑convolution
    # ------------------------------------------------------------------
    while time.time() < deadline and it < max_iter:
        # --- recover probability vector a from logits u ---
        a = np.exp(u)
        a_sum = a.sum()
        a = a / a_sum

        # --- FFT of a (reused for both convolutions) ---
        fft_a = np.fft.rfft(a, size_fft)

        # --- self‑convolution a*a (needed for the smooth max) ---
        conv_full = np.fft.irfft(fft_a * fft_a, size_fft)[: 2 * n - 1]
        max_c = conv_full.max()

        # --- schedule for the smoothing parameter β ---
        if it < beta_ramp:
            beta = 1.0 + (beta_max - 1.0) * (it / beta_ramp)
        else:
            beta = beta_max

        # --- soft‑max weights on the convolution entries ---
        w = np.exp(beta * (conv_full - max_c))   # max entry gets weight 1
        w_sum = w.sum()
        if w_sum == 0.0:                         # numerical safeguard
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # --- gradient of the smooth objective (2 * a * w_rev) ---
        w_rev = w[::-1]
        fft_w = np.fft.rfft(w_rev, size_fft)
        conv_aw = np.fft.irfft(fft_a * fft_w, size_fft)[: len(w_rev) + n - 1]
        start = len(w_rev) - 1
        grad_a = 2.0 * conv_aw[start:start + n]   # shape (n,)

        # --- gradient w.r.t. logits u (softmax Jacobian) ---
        avg = np.dot(a, grad_a)
        grad_u = a * (grad_a - avg)

        # --- Adam update on u ---
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr / (np.sqrt(v_hat) + eps)
        u = u - step * m_hat

        # keep logits from drifting too far (prevents overflow in exp)
        if np.max(u) - np.min(u) > 120.0:
            u -= np.max(u)        # shift without changing the soft‑max

        # ------------------------------------------------------------------
        # Logging & occasional structural moves
        # ------------------------------------------------------------------
        if it % 200 == 0:
            # objective for the current a (max_c already known)
            cur_obj = 2.0 * n * max_c / (a_sum * a_sum)
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve_iter = it
                print(f"[iter {it}] new best objective = {best_obj:.6f}")

            # every few thousand steps try the anti‑symmetric reorder
            if it % 2000 == 0 and it > 0:
                a_ord = _anti_symmetric_reorder(a)
                _, obj_ord = _maxconv_objective(a_ord)
                if obj_ord < best_obj - 1e-12:
                    best_obj = obj_ord
                    best_a = a_ord.copy()
                    # reset the optimisation state to the reordered vector
                    u = np.log(a_ord + 1e-30)
                    m.fill(0.0)
                    v.fill(0.0)
                    last_improve_iter = it
                    print(f"[reorder {it}] improved objective = {best_obj:.6f}")

        # ------------------------------------------------------------------
        # Escape from stagnation
        # ------------------------------------------------------------------
        if it - last_improve_iter > 50_000:
            # add a small Gaussian perturbation to the logits
            u += rng.normal(scale=0.05, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = it
            print(f"[perturb {it}] added Gaussian noise to logits")

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – cheap stochastic hill‑climbing refinement
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_obj = best_obj
    while time.time() < deadline:
        # Choose a random local move
        if rng.random() < 0.3:                         # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                          # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        _, new_obj = _maxconv_objective(new_a)
        if new_obj < cur_obj - 1e-12:
            cur_a = new_a
            cur_obj = new_obj
            best_a = new_a.copy()
            best_obj = new_obj
            print(f"[refine] improved to {cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (using the official function) and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_a.tolist()
