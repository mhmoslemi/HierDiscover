def _annealing_refine(a, cur_val, deadline):
    best_a = a.copy()
    best_val = cur_val
    T_initial = 0.01  # maybe objective scale? Actually objective is ~1.5, so T initial ~0.01?

    rng = np.random.default_rng()

"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm Overview
------------------
* **Multiple restarts** – try several initialisations (uniform, sinusoid,
  random exponential, and the previously discovered `height_sequence_1` if
  available).

* **Smoothed‑max descent** – minimise a log‑sum‑exp surrogate of the
  peak of the self‑convolution.  The smoothing parameter ε is reduced
  through a long schedule down to 0.001, and a multiplicative
  mirror‑descent update keeps the vector non‑negative and on the
  probability simplex.

* **Anti‑symmetric re‑ordering** – after each ε‑stage (and periodically
  inside) the vector is reordered so that the largest values are placed
  opposite the smallest ones.  By the rearrangement inequality this never
  increases the maximal convolution value.

* **Local refinement** – a cheap hill‑climber (swaps / tiny mass
  transfers) is applied, followed by a simulated‑annealing phase that
  accepts occasional worsening moves to escape shallow local minima.

The routine respects the 1000 s time budget, printing progress and
returning the best sequence found.  All numbers are non‑negative and
normalised to sum = 1 (the objective is scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Returns a non‑negative array.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)   # eliminate tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·b_k w.r.t. a,
    where b = a*a (linear convolution) and p is a weight vector
    (Σp = 1, p ≥ 0).

    Result: g_i = Σ_k p_k·a_{k‑i}  (zero outside the support).
    Computed with a single FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv[n - 1 : 2 * n - 1]          # indices that correspond to a
    g = seg[::-1]                         # reverse back
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest ones (pairwise).
    This never worsens the maximal convolution (rearrangement inequality).
    """
    n = a.shape[0]
    s = np.sort(a)                 # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]           # large → left side
        b[-1 - i] = s[i]           # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Smoothed‑max descent (multiplicative mirror‑descent)
# ----------------------------------------------------------------------
def _smoothed_descent(a_start: np.ndarray, deadline: float):
    """
    Minimise a log‑sum‑exp surrogate of `max(a*a)`.
    Returns the best vector encountered and its true evaluation.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent init] objective = {best_val:.6f}")

    # Extended schedule down to ε = 0.001
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    total_iters = 4000                     # total inner FFT steps
    it_per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        eta = 0.75                         # base step size for this ε

        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:
                break

            # ---- convolution and soft‑max weights ----
            conv = _full_convolution(a)          # length 2n‑1
            max_conv = conv.max()
            shift = max_conv                     # for numerical stability
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()        # Σp = 1, p ≥ 0

            # ---- gradient of surrogate ----
            grad = 2.0 * _gradient_weighted_peaks(a, p)

            # ---- multiplicative mirror‑descent step ----
            delta = -eta * grad
            delta = np.clip(delta, -20.0, 0.0)    # avoid overflow
            a = a * np.exp(delta)
            a = _project_simplex(a)              # keep Σa = 1, a ≥ 0

            # ---- occasional evaluation of true objective ----
            if it % 20 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.4f} it {it}] new best = {best_val:.6f}")

                # re‑order every 200 steps (cheap and never harms)
                if it % 200 == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    val_ord = evaluate_sequence(a_ord.tolist())
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[reorder eps {eps:.4f} it {it}] improved to {best_val:.6f}")

            # slowly decay the step size (helps stability for small ε)
            eta *= 0.9995

        # final re‑ordering for this ε (guaranteed non‑increase)
        a = _anti_symmetric_reorder(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Light hill‑climbing (strict improvements)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float, n: int):
    """
    Random swaps or tiny mass transfers, kept only if they strictly
    improve the true objective.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        if rng.random() < 0.25:                 # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                   # transfer a small amount
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.05 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # renormalise onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing style refinement (allows temporary worsening)
# ----------------------------------------------------------------------
def _anneal_refine(a: np.ndarray, cur_val: float,
                   deadline: float):
    """
    Random mass transfers with Metropolis acceptance.
    The temperature decays slowly; the loop stops near the deadline.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]
    T = 0.02                     # initial temperature (≈ 1 % of objective)
    decay = 0.9999               # exponential decay per iteration

    while time.time() < deadline - 1.0:
        i, j = rng.integers(0, n, size=2)
        if i == j or best_a[i] <= 1e-15:
            continue
        # transfer up to 30 % of the donor's mass
        delta = best_a[i] * (0.02 + 0.28 * rng.random())
        new_a = best_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s == 0.0:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        diff = val - best_val
        if diff < 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            # accept
            best_val = val
            best_a = new_a
            if diff < 0:
                print(f"[anneal] improved to {best_val:.6f}")
        T *= decay

    return best_a, best_val

    n = 8000

        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        sum0 = a0.sum()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_a = None

    # ------------------------------------------------------------------

    init_strategies = ["uniform", "sin", "rand"]
    restart = 0
    max_restarts = 6

    while time.time() < deadline - 200 and restart < max_restarts:
        restart += 1

        # ----------------------------------------------------------------

        else:

    except Exception as exc:   # pragma: no cover

    return best_a.tolist()
