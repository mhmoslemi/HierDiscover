"""
Propose a non‑negative step function (list of floats) that tries to minimise
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Initialisation** – take the historic construction if available,
   otherwise draw a random Dirichlet vector (exponential draws normalised).

2. **Adam‑accelerated exponential‑mirror descent** – at each iteration we
   compute the self‑convolution `c = a * a`.  A smooth approximation of the
   maximum is built with weights `w = exp(beta·(c‑max(c)))`.  The (weighted)
   gradient of `∑ w_k·c_k` w.r.t. `a` can be obtained by a single convolution.
   The gradient is fed to an Adam optimiser that updates the *log* of `a`
   multiplicatively and re‑projects onto the simplex.  The smoothing parameter
   `beta` is increased from 1 to a few thousand while the learning rate is
   slowly decayed.  Every few thousand iterations an “interleaved‑extreme”
   re‑ordering (largest values placed alternately at the two ends) is tried;
   it often lowers the peak of the convolution.

3. **Fast stochastic local search** – starting from the best vector found,
   many cheap local moves (pairwise swaps and tiny mass transfers) are tried.
   The objective is evaluated with a custom FFT‑based routine (`evaluate_fast`)
   which is orders of magnitude faster than the reference `evaluate_sequence`.
   Strict improvements are kept.

All heavy work is performed with FFTs; only the final candidate is checked
with the supplied `evaluate_sequence`.  The routine respects a 1000 s time
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)      # eliminate tiny negatives
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to `a`.
    Implements g_i = 2·(a * w_rev)_i using a single convolution.
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1                 # = 2n‑2
    seg = conv[start:start + n]                # length n
    return 2.0 * seg


def evaluate_fast(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` – computes
    2·n·max(a*a) / (sum(a)^2) using FFTs.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


def _interleaved_extreme_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values alternately at the leftmost and rightmost
    positions (large‑small‑large‑small …).  This tends to keep large‑large
    products from accumulating at a single convolution index.
    """
    vals = np.sort(a)[::-1]          # descending
    n = a.shape[0]
    b = np.empty_like(a)
    left, right = 0, n - 1
    idx = 0
    while left <= right:
        if idx % 2 == 0:
            b[left] = vals[idx]
            left += 1
        else:
            b[right] = vals[idx]
            right -= 1
        idx += 1
    return b


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` within the allotted time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep a few seconds margin

    # ------------------------------------------------------------------
    # Problem size – 8000 coefficients give a good balance.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from historic construction when possible.
    # ------------------------------------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, a)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        # random Dirichlet (exponential draws) – already non‑negative
        a = rng.exponential(scale=1.0, size=n)
        a = np.maximum(a, 0.0)
        a /= a.sum()

    # ------------------------------------------------------------------
    # Keep the best solution seen so far.
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_fast(best_a)
    best_it = 0
    print(f"[init] n={n}, fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – few random restarts to diversify the start.
    # ------------------------------------------------------------------
    for tr in range(4):
        if time.time() >= deadline - 150:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_fast(a_rand)
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best_fast = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam accelerated exponential‑mirror descent.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    max_iter = 500_000

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Schedules
    alpha0 = 0.3                 # base learning rate (for log‑updates)
    beta_start = 1.0
    beta_end = 2500.0
    beta_ramp = 40_000          # iterations over which β grows

    while time.time() < deadline - 30 and it < max_iter:
        # ----- convolution and smooth weights -----
        conv = _full_convolution(a)          # length 2n‑1
        max_c = conv.max()

        # smooth weight vector approximating the max
        delta = conv - max_c                 # ≤ 0
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp)
        w = np.exp(beta * delta)            # w[max]=1, others ≤1
        w_rev = w[::-1]

        # ----- gradient of the weighted sum -----
        g = _gradient_weighted(a, w_rev)    # shape (n,)

        # ----- Adam update on log‑variables -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # a simple learning‑rate decay
        lr = alpha0 * (0.98 ** (it // 2000))

        delta_log = -lr * m_hat / (np.sqrt(v_hat) + eps)
        # keep exponentials from exploding
        delta_log = np.clip(delta_log, -15.0, 15.0)

        a = a * np.exp(delta_log)

        # ----- simplex projection -----
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # ----- logging & occasional reordering -----
        if it % 200 == 0:
            cur_val = evaluate_fast(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                best_it = it
                print(f"[gd iter {it}] new best_fast = {best_val:.6f}")

            # try the interleaved‑extreme ordering every 2000 iters
            if it % 2000 == 0:
                a_ord = _interleaved_extreme_reorder(a)
                val_ord = evaluate_fast(a_ord)
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    best_it = it
                    print(f"[gd reorder {it}] improved_fast = {best_val:.6f}")

        # ----- stagnation handling -----
        if (it - best_it) > 20_000:
            # blend with a fresh random Dirichlet vector to escape plateau
            noise = rng.exponential(scale=1.0, size=n)
            noise /= noise.sum()
            a = 0.7 * a + 0.3 * noise
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            best_it = it
            print(f"[gd perturb at {it}]")

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – fast stochastic local search (swap / tiny transfer).
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    while time.time() < deadline:
        if rng.random() < 0.3:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.04 * rng.random()   # up to 4 % shift
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = evaluate_fast(new_a)
        if new_val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = new_a.copy()
                print(f"[refine] new best_fast = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine and return.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] official evaluation = {final_score:.6f}")
    return best_a.tolist()
