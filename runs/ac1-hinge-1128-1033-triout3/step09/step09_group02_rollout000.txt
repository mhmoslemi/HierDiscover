"""
Simple yet powerful optimizer for the step‑function “max‑convolution” problem.

Strategy
--------
1. **Dense parametric sweep** – construct many candidate probability vectors
   using a rich set of families (exponential, power‑law, Beta‑distribution,
   an explicit arcsine shape, and random Dirichlet draws).  The sweep is
   intentionally dense for the Beta parameters because the optimum lies
   near the singular Beta(½,½) shape.

2. **Targeted hill‑climbing** – starting from the best candidate, repeatedly
   move a small amount of mass from an index that contributes heavily to the
   current maximal convolution entry to an index that contributes very little.
   The move is performed with an exact O(n) update of the convolution, so
   millions of steps fit in the time budget.

3. **Final uniform mixing** – a tiny convex mixture with the uniform vector
   often smooths remaining peaks.

All steps respect the simplex (non‑negative entries, sum = 1) and
the `evaluate_sequence` wrapper.  The algorithm runs for the full
budget (default 1000 s) and prints progress information.

Author: an expert software developer & inequality specialist.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective handling (exact, no internal rescaling)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)^2)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x ≥ 0, Σx = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric sweep – returns (best_vector, best_value)
# ----------------------------------------------------------------------

    rng = np.random.default_rng(123456)

    # start from uniform – this gives objective 2.0 (the worst baseline)
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(candidate: np.ndarray):
        nonlocal best_a, best_val
        if candidate.ndim != 1 or candidate.shape[0] != n:
            return
        if np.any(candidate < 0):
            return

        if s <= 0:
            return

        val = _objective_value(candidate)
        if val < best_val:
            best_val, best_a = val, candidate.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0

    # ----- 1. Exponential tails (both monotone and symmetric) -----

# ----------------------------------------------------------------------
# Targeted hill‑climbing on the exact objective (fast O(n) updates)
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               deadline: float,
               start_time: float) -> tuple[np.ndarray, float]:
    """Greedy redistribution of mass from “heavy” to “light” indices."""
    a = a_start.copy()
    n = a.shape[0]

    # initial convolution and objective
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a, best_val = a.copy(), cur_val

    # schedule for the maximal transferable fraction
    max_frac_start, max_frac_end = 0.45, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # linearly decay the allowed step‑size
        progress = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))                     # location of the peak

        # ---- source index (high contribution to the current peak) ----
        idx = k_max - np.arange(n)                    # partners that pair with i
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # ---- target index (as little partner as possible) ----
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.full(n, np.inf)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        # ignore the source itself
        rev_contrib[i] = np.inf
        # pick among the 5 best candidates to keep randomness
        cand_js = np.argpartition(rev_contrib, 5)[:5]
        j = int(np.random.choice(cand_js))

        # ---- choose a transfer amount ----
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (exact) ----
        b_new = b.copy()
        # contributions coming from the moved mass (linear term)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:                # accept only if it improves
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---- occasional perturbation after long stagnation ----
        if no_improve > IMPROVE_LIMIT:
            # small Gaussian shake, re‑normalize, recompute convolution
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

# Tiny uniform mixing (final polishing)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Convex combination with a minute share of the uniform distribution."""
    n = a.shape[0]
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.007, 0.010, 0.018):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    all security checks inside `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 12 k entries is a good balance between expressivity
    # and speed (FFT length ≈ 2^15).
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep – pick a promising starting shape
    # ------------------------------------------------------------------
    print("[Init] running dense parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Hill‑climbing refinement (major part of the budget)
    # ------------------------------------------------------------------
    phase_hill_deadline = start_time + 0.85 * (deadline - start_time)
    best_a, best_val = _hillclimb(best_a, phase_hill_deadline, start_time)
    print(f"[Hillclimb] best after hill = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final uniform mixing (tiny tweak)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Verify with the provided evaluation wrapper (optional, for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
