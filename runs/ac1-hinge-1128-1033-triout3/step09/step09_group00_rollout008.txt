import time, numpy as np

def full_convolution(a):
    n = len(a)
    out_len = 2*n-1
    size = 1 << (out_len-1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:out_len]
    return conv

def objective(a):
    n = len(a)
    b = full_convolution(a)
    return 2.0 * n * b.max()

def max_convolution_gradient(a):
    n = len(a)
    b = full_convolution(a)
    max_b = b.max()
    ks = np.where(b == max_b)[0]
    grad = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        i_idx = np.arange(i_start, i_end+1)
        j_idx = k - i_idx
        grad[i_idx] += 2.0 * a[j_idx]
    return grad, max_b

def exp_gradient_optimize(a0, deadline, start_time):
    a = a0.copy()

        # we bias toward lower contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[i_start:i_end+1] = 1.0 / (contrib[i_start:i_end+1] + 1e-12)
        tgt_weights[i] = 0.0
        sum_w = tgt_weights.sum()

        else:

            j = (i + np.random.randint(1, n)) % n
        if a[i] <= 1e-12:

        if delta <= 0:

        # recompute b new via incremental update with corrected term
        b_new = b.copy()

        # quadratic terms
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta   # corrected
        # also need to adjust for self contributions at positions where i==j? Already motion ensures i!=j
        # compute new val
        new_max = b_new.max()
        new_val = 2.0 * n * new_max
        # Metropolis criterion
        accept = False
        if new_val < cur_val:
            accept = True
        else:

            if np.random.rand() < np.exp(-(new_val - cur_val) / Tcur):
                accept = True
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[Hill {iter:6d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    break
        else:
            no_improve += 1
        if no_improve > 80000:

            a += np.random.normal(scale=0.02, size=n)
            a = np.maximum(a, 0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iter:6d}] perturb")
        iter += 1
    return best_a, best_val

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – cheap parametric search:
   * Interpolate the previous solution (if ``height_sequence_1`` is defined).
   * Scan a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}.
   * Add a few random Beta and Dirichlet draws.
   * Keep the best shape (its sum is 1).

2. **Entropic mirror descent (exponential‑gradient)** – main optimisation:
   * Objective:  f(a)=2·n·max(conv(a,a)).
   * The sub‑gradient of max(conv) is computed analytically.
   * Update in log‑space:  w ← log a – η·∇f(a) (entropy mirror descent).
   * η decays as η₀/√t, η₀≈0.5.
   * Restart with a small jitter if no improvement for many steps.

3. **Corrected simulated‑annealing hill‑climbing**:
   * Re‑implemented the O(n) incremental convolution update
     (the previous code missed a factor 2 in the quadratic term).
   * Moves mass from indices contributing heavily to the maximal
     convolution entry toward low‑contribution indices.
   * Temperature and maximal transfer fraction are annealed.
   * Stagnation triggers a random perturbation of the current best.

4. **Deterministic sub‑gradient polishing**:
   * Projected sub‑gradient descent on the exact max‑convolution.
   * Linear step‑size schedule, occasional jitter to escape flats.

The whole procedure respects the 1000 s budget, prints progress, and
finally returns the best sequence found (as a plain Python list)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (a * a)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (under Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Gradient of max‑convolution (sub‑gradient of max(b))
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute a sub‑gradient of  max(conv(a,a))  and return (grad, max_b).
    grad_i = Σ_{k∈K} 2·a_{k‑i}   where K = {k | conv(a,a)_k = max_b}.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    ks = np.where(b == max_b)[0]

    grad = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k - i_idx
        grad[i_idx] += 2.0 * a[j_idx]
    return grad, max_b


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric sweeps (Beta family + random draws)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Exhaustive grid sweep over a two‑parameter Beta family."""
    best_a, best_val = None, np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _beta_random_search(n: int, trials: int = 250) -> tuple[np.ndarray, float]:
    """Random search in the Beta family (log‑uniform p,q)."""
    best_a, best_val = None, np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    rng = np.random.default_rng()
    for _ in range(trials):
        logp = rng.uniform(-6.9, 2.3)   # ln(0.001) … ln(10)
        logq = rng.uniform(-6.9, 2.3)
        p, q = np.exp(logp), np.exp(logq)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – Entropic mirror descent (exponential‑gradient)
# ----------------------------------------------------------------------
def _exponential_gradient_optimize(a0: np.ndarray,
                                   start_time: float,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex with the negative entropy
    (a_i ← a_i·exp(−η·∇f)).  Stops when the time budget expires.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a, best_val = a.copy(), _objective(a)
    eta0 = 0.5
    it = 0
    idle = 0
    MAX_IDLE = 8000

    while True:
        now = time.time()
        if now >= deadline:
            break

        grad, max_b = _max_conv_gradient(a)           # sub‑gradient of max(conv)
        grad_obj = 2.0 * n * grad                     # gradient of the full objective
        eta = eta0 / np.sqrt(it + 1)

        # entropy‑mirror update in log‑space
        w = np.log(a + 1e-30) - eta * grad_obj
        a = np.exp(w)
        s = a.sum()
        if s == 0.0:                                 # fallback to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[EG {it:6d}] new best = {best_val:.6f}")
        else:
            idle += 1

        # simple restart if no progress for a long while
        if idle > MAX_IDLE:
            print(f"[EG {it:6d}] restart after {idle} idle steps")
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = np.maximum(a, 0)
            a /= a.sum()
            idle = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Simulated‑annealing hill climbing (fixed quadratic term)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      start_time: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast O(n) local search that moves a chunk of mass from a source
    index (high contribution to the current maximal convolution entry)
    towards a low‑contribution target index.  The incremental convolution
    update is now mathematically correct (the 2·δ² term is restored).
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a, best_val = a.copy(), cur_val

    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.02, 1e-8
    no_improve = 0
    IMPROVE_LIMIT = 80000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k_max - i_idx
        contrib = np.zeros_like(a)
        contrib[i_idx] = a[i_idx] * a[j_idx]
        total_contrib = contrib.sum()

        # source index – proportional to its contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # target index – bias toward low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[i_start:i_end + 1] = 1.0 / (contrib[i_start:i_end + 1] + 1e-14)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (excluding i)
            j = (i + np.random.randint(1, n)) % n

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- exact (incremental) convolution update -----
        b_new = b.copy()
        # linear part due to moving δ
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic part (corrected factor 2)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[Hill {it:6d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.02, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:6d}] perturb due to stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        start_time: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Stops when the time budget expires.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a, best_val = a.copy(), cur_val
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # sub‑gradient of max(conv)
        grad, _ = _max_conv_gradient(a)          # returns 2·a_{k‑i} summed over argmax
        grad_obj = 2.0 * n * grad                # gradient of full objective

        # linear step size (decreases to ~1e‑5)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.3 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad_obj
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-12:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a_new.copy()
                print(f"[Sub {it:6d}] new best = {best_val:.6f}")

        # occasional jitter
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

        it += 1

    return best_a, best_val

    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣   Initialise – try to reuse a previous solution
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweeps
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf

        val = _objective(a0)

        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    shape_a, shape_val = _beta_family_best(n)
    if shape_val < best_val:

    if rand_val < best_val:

    for _ in range(4):
        cand = np.random.dirichlet(np.ones(n))
        cand_val = _objective(cand)

            print(f"[init] Dirichlet seed improved value={best_val:.6f}")

    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()

    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s

    phase2_deadline = start_time + 0.85 * budget_s

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish(
        best_a, phase2_deadline, deadline
    )

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:

        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
