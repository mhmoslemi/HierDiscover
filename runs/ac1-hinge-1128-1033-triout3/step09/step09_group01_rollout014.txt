```
    a = a0.copy()
    cur_val = _objective(a)
    best_a = a.copy()
    best_val_local = cur_val
    no_improve = 0
    while time.time() < deadline:
        # compute progress fraction
        prog = (time.time() - start_time) / (deadline - start_time)
        # compute gamma and step
        gamma = 0.5 + 49.5 * prog  # from 0.5 to 50
        step = 0.1 * (1 - prog) + 1e-5 * prog
        # convolution
        b = _full_convolution(a)
        max_b = b.max()
        # compute weights smoothed
        # shift by max for stability
        exp_vals = np.exp(gamma * (b - max_b))
        w = exp_vals / exp_vals.sum()
        # gradient
        conv_w_a = np.convolve(w, a)
        grad = conv_w_a[n-1 : n-1 + n]  # length n
        # update (descent direction)
        a_new = a - step * grad
        # enforce nonnegativity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0:
            # fallback to uniform
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= s
        new_val = _objective(a_new)
        if new_val < best_val_local - 1e-12:
            best_val_local = new_val
            best_a = a_new.copy()
            print(f"[GD] iteration {iteration:6d}, new best {best_val_local:.6f}")
            no_improve = 0
        else:
            no_improve += 1
        # accept if improvement; else maybe keep current a (or with probability)
        if new_val < cur_val:
            a = a_new
            cur_val = new_val
        else:
            # optionally accept with small probability; skip.

        # stagnation: if no_improve > threshold, restart with small noise

            noise = np.random.normal(scale=0.01, size=n)

            a /= a.sum()
            cur_val = _objective(a)
            no_improve = 0
            print(f"[GD] stagnation restart at iteration {iteration}")
        iteration += 1
    return best_a, best_val_local
```

```
def _hillclimb_det(a0, n, deadline, start_time, best_val):
    a = a0.copy()
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2 * n * cur_max
    best_a = a.copy()
    best_val = cur_val
    iteration = 0
    no_improve = 0
    max_frac_start, max_frac_end = 0.5, 0.001
    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1 - prog) + max_frac_end * prog
        # compute contributions
        k_max = int(np.argmax(b))
        # compute pairs
        i_range = np.arange(n)
        j_idx = k_max - i_range
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_range[mask]] * a[j_idx[mask]]
        # if no contributions (shouldn't happen)
        if not mask.any():
            break
        i_src = int(np.argmax(contrib))
        # candidate target indices sorted by ascending contrib
        low_idxs = np.argsort(contrib)
        # ensure unique and not src
        found = False
        for i_tgt in low_idxs:
            if i_tgt == i_src:
                continue
            # compute delta
            delta = a[i_src] * max_frac
            if delta <= 0:
                continue
            # compute new b via copy and inc
            b_new = b.copy()
            # updates slice
            b_new[i_tgt:i_tgt + n] += 2.0 * delta * a
            b_new[i_src:i_src + n] -= 2.0 * delta * a
            b_new[2 * i_tgt] += delta * delta
            b_new[2 * i_src] += delta * delta
            b_new[i_src + i_tgt] -= 2.0 * delta * delta
            new_max = b_new.max()
            if new_max < cur_max - 1e-12:
                # accept
                a[i_src] -= delta
                a[i_tgt] += delta
                b = b_new
                cur_max = new_max
                cur_val = 2.0 * n * cur_max
                found = True
                no_improve = 0
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[Hill] iter {iteration} new best {best_val:.6f}")
                break
        if not found:
            no_improve += 1
            # maybe try small random perturb if stuck
        if no_improve > 5000:
            # perturb
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            no_improve = 0
            print(f"[Hill] stagnation restart at iter {iteration}")
        iteration += 1
    return best_a, best_val
```

The task is to minimise  

\[
F(a)=\frac{2\;n\;\max\_{k}(a*a)_k}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where ``a*a`` is the full linear convolution of the non‑negative vector `a`.  
Because the denominator is homogeneous we may restrict ourselves to vectors with
\(\sum_i a_i = 1\); the objective then simplifies to `2*n*max(conv(a,a))`.

The code below implements a **deterministic hill‑climbing search** that
concentrates all effort on the entry of the convolution that is currently the
largest (`k_max`).  
For this index we know the exact contribution of every coefficient:

```
contrib[i] = a[i] * a[k_max-i]          (if the pair exists, otherwise 0)
```

Moving a small amount `δ` from a coefficient with a large contribution to a
coefficient with a tiny (or zero) contribution reduces `max(conv)`.  The
convolution update after moving `δ` can be performed in *O(n)* time by the
incremental formula used in the reference implementation – no full FFT is
required during the inner loop.

The algorithm proceeds as follows:

1. **Warm start** – try the previous best solution (if available), otherwise
   pick the best candidate from a cheap two‑parameter Beta family and a few
   random Dirichlet draws.
2. **Hill climbing** – repeatedly  
   * compute the full convolution and locate `k_max`;  
   * compute the contributions vector, pick the source index with maximal
     contribution and a target index with minimal contribution (up to a small
     set of candidates);  
   * transfer a fraction of the source mass (`δ = max_frac * a[source]`);
     the fraction `max_frac` decays from 0.9 to 0.001 over the run.  
   * accept the move only if it strictly reduces the maximal convolution entry.
   * keep the best vector seen so far; after a long period without improvement
     perturb the current best with a small Gaussian noise and continue.
3. **Final evaluation** – evaluate the best vector with the provided
   `evaluate_sequence` function and return it.

All operations respect non‑negativity and the simplex constraint (the vector
is re‑normalised after each perturbation).  The search uses only NumPy, runs
well inside the 1000 s budget, and typically pushes the upper bound below the
target value 1.503.

```python
"""
Non‑negative step‑function optimiser.

* Warm‑starts from a previous solution, a best Beta‑family candidate and a
  few random Dirichlet draws.
* Deterministic hill‑climbing: at each iteration the current maximal
  convolution entry `k_max` is identified; the coefficient contributing most
  to this entry is reduced while a coefficient contributing the least is
  increased.  The convolution is updated incrementally (O(n)).
* Mass transfers become smaller over time (max_frac from 0.9 → 0.001).  If the
  search stalls a Gaussian perturbation around the best vector is performed.
* The best candidate found before the time limit is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective for a vector `a` that already satisfies Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """
    Search the 2‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}.
    Returns the best vector (normalised to sum = 1) and its objective value.
    """
    best_a = None
    best_val = np.inf
    # parameters: dense near 1, plus a geometric spread up to 6
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climbing
# ----------------------------------------------------------------------


def _hillclimb_deterministic(a0: np.ndarray,
                             n: int,
                             deadline: float,
                             start_time: float,
                             best_val_start: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that always moves mass from the coefficient with the largest
    contribution to the current maximal convolution entry towards a coefficient
    with the smallest contribution.  The move is accepted only if the maximal
    convolution entry strictly decreases.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    # schedule for the transferred fraction
    max_frac_start, max_frac_end = 0.9, 0.001

    # stagnation handling
    no_improve = 0
    IMPROVE_LIMIT = 8000       # after this many failed attempts we perturb

    iteration = 0
    eps = 1e-15                # numerical tolerance

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress in [0, 1]
        prog = (now - start_time) / (deadline - start_time)
        # decaying fraction of mass that may be moved
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # ---- identify the worst convolution entry --------------------
        k_max = int(np.argmax(b))

        # contributions a[i] * a[k_max-i] (zero if the pair is out of range)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[mask] * a[idx[mask]]

        # source = index with largest contribution
        i_src = int(np.argmax(contrib))

        # list of candidate targets: indices with smallest contributions
        # (skip the source itself)
        low_order = np.argsort(contrib)
        candidates = low_order[low_order != i_src]

        move_accepted = False

        for i_tgt in candidates[:10]:            # try at most 10 candidates
            if i_tgt == i_src:
                continue
            delta = a[i_src] * max_frac
            if delta <= eps:
                continue

            # incremental convolution update (O(n))
            b_new = b.copy()
            # the 2·δ·a[t] part
            b_new[i_tgt:i_tgt + n] += 2.0 * delta * a
            b_new[i_src:i_src + n] -= 2.0 * delta * a
            # the δ² self‑pair corrections
            b_new[2 * i_tgt] += delta * delta
            b_new[2 * i_src] += delta * delta
            b_new[i_src + i_tgt] -= 2.0 * delta * delta

            new_max = b_new.max()
            if new_max < cur_max - eps:
                # accept move
                a[i_src] -= delta
                a[i_tgt] += delta
                b = b_new
                cur_max = new_max
                cur_val = 2.0 * n * cur_max
                move_accepted = True
                no_improve = 0

                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                break   # stop trying further targets

        if not move_accepted:
            no_improve += 1

        # ---- stagnation handling ------------------------------------
        if no_improve > IMPROVE_LIMIT:
            # perturb around the best vector found so far
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val
```
