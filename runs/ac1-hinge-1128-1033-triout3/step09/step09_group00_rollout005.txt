"""
Optimise a non‑negative step function under a 1000 s budget.

Main idea
---------
1. Initialise from the previous solution when available.
2. Perform an extensive search in the two‑parameter Beta family
   a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}.  Both a dense grid and a large
   log‑uniform random sweep are used; the best shape becomes the
   starting point.
3. Run a long projected sub‑gradient descent on the exact
   objective   f(a)=2·n·max(conv(a,a)).
   At every iteration we
   * evaluate the full convolution (FFT),
   * locate the index where the maximum is attained,
   * move a little mass away from the indices that contribute most to
     that entry,
   * project back onto the probability simplex.
   The step size decays linearly from 0.3 to 1e‑5 and a tiny jitter
   is added occasionally to escape flat regions.
4. After the time limit the best sequence found is returned and
   finally evaluated with ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

3️⃣  Run a long projected sub‑gradient descent on the exact objective
    f(a)=2·n·max(conv(a,a)).  In each iteration we:

      * locate the lag where the maximum is attained,
      * move a little mass away from the indices that contribute most to

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]

    return w
