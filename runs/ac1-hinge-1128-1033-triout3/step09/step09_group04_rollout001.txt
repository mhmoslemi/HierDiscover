```
def proj_simplex(v):
    # v is 1-d np.ndarray
    if np.sum(v) == 1 and np.all(v >= 0): return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1 # zero-indexed
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

assistantfinal**Solution Overview**

The code below searches a non‑negative step function `a` (length `n`) that minimises  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}\;,
\]

which is exactly the value returned by `evaluate_sequence`.  
Because the objective is invariant under scaling, we keep the vector on the
probability simplex (`a_i≥0`, `∑a_i=1`).  

**Main ideas**

1. **Smooth approximation of the max** –  
   `softmaxγ(b)=exp(γ·b)/∑exp(γ·b)` gives a differentiable surrogate for the
   maximal convolution entry.  We start with a small `γ` (very smooth) and
   increase it linearly to a larger value, ending with an almost exact max.

2. **Projected gradient descent** –  
   The gradient of the smoothed objective is  
   \[
   \nabla_a F_\gamma(a)=2\;\bigl(a\star w\bigr),\qquad 
   w=\operatorname{softmax}_\gamma(a*a),
   \]
   where `⋆` denotes the (full) cross‑correlation.  The gradient is computed
   with two FFT‑based convolutions (O(n log n)).  After a step we project the
   point back onto the simplex.

3. **Stagnation handling** –  
   If no improvement is seen for many iterations we add a tiny Gaussian
   jitter to escape flat regions.

4. **Random restarts & tiny uniform mixing** –  
   The outer loop repeatedly starts from a random Dirichlet vector (or a
   jittered copy of the current best) and runs the inner optimiser for a
   short time slice.  The globally best vector is kept.  
   Finally a few convex combinations with the uniform vector are tested.

The whole routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found as a plain Python list.

```python
"""
Projected gradient descent with a smoothed max‑convolution objective.

* The objective is 2·n·max(conv(a,a)) under the simplex constraint
  (∑a = 1, a_i ≥ 0).  Scaling does not affect the value, therefore we keep
  the vector on the simplex throughout the search.

* A soft‑max (temperature γ) provides a smooth approximation of the max.
  γ is increased linearly from γ_start to γ_end, turning the smooth problem
  into the original one.

* The gradient of the smoothed objective is
      ∇Fγ(a) = 2 * (a_rev  ★  w)          (★ = full correlation)
  where w = softmaxγ(conv(a,a)).  Both convolutions are computed with FFTs,
  which makes a single iteration O(n log n).

* After each step we project onto the simplex (non‑negative, sum = 1).  If
  the algorithm stagnates for many iterations we inject a tiny Gaussian
  jitter.  Multiple short random restarts are performed until the time budget
  expires, and a final tiny mixing with the uniform vector is tried.

The function `propose_candidate` is the entry point required by the
competition harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex
# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform vector
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1] - 1             # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise exactly to 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n, dtype=v.dtype)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Full linear convolution using FFT (real input → real output)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """
    Return the full linear convolution a*a (length 2·len(a)-1).
    `size` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Smoothed objective and its gradient
# ----------------------------------------------------------------------
def smoothed_objective_and_grad(
    a: np.ndarray,
    n: int,
    size_conv: int,
    size_grad: int,
    gamma: float,
) -> tuple[float, np.ndarray]:
    """
    Compute the smoothed objective value
        Fγ(a) = 2·n· ( 1/γ * log Σ exp(γ·(a*a)) )
    and its gradient w.r.t. a (projected later).
    """
    # 1) convolution a*a
    b = full_convolution(a, size_conv)      # length 2n-1
    max_b = b.max()

    # 2) softmax weights (stable)
    shifted = gamma * (b - max_b)            # max becomes 0
    exp_shift = np.exp(shifted)
    w = exp_shift / exp_shift.sum()          # length 2n-1, sum = 1

    # 3) gradient = 2 * (a_rev ★ w)
    a_rev = a[::-1]
    # compute full correlation using FFT
    w_fft = np.fft.rfft(w, size_grad)
    rev_fft = np.fft.rfft(a_rev, size_grad)
    corr_full = np.fft.irfft(w_fft * rev_fft, size_grad)
    # slice the middle n entries (indices n-1 … 2n-2)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]

    # smoothed objective value (the quantity returned by evaluate_sequence)
    # Using the identity: max_b ≤ (1/γ)·log Σ exp(γ·b) ≤ max_b + log(2n-1)/γ
    # We can simply evaluate the true objective (max) because scaling is invariant.
    # For the optimiser we return the true objective to keep the same scale.
    obj = 2.0 * n * max_b
    return obj, grad

# ----------------------------------------------------------------------

    size_grad = 1 << ((n + conv_len - 2).bit_length())   # ≥ 3n‑2

    obj, _ = smoothed_objective_and_grad(cur, n, size_conv, size_grad, gamma_start)
    best_val = obj
    no_improve = 0
    rng = np.random.default_rng()

        # linearly interpolated hyper‑parameters
        t_now = time.time()
        total = deadline - t_now + (t_now - deadline)  # dummy to avoid divide by zero
        # progress is relative to the remaining time slice
        progress = (t_now - (deadline - (deadline - t_now))) / (deadline - (deadline - (deadline - t_now)))  # always 0, replace with simpler:
        # Simpler: compute based on elapsed time inside this run
        # (we keep the start time of the run separately)
        # For robustness we recompute progress as:
        # fraction of the slice that has elapsed
        # Use a static start time saved the first time we enter.
        # We'll instead compute via a closure variable; to avoid that we simply
        # keep progress monotone by counting iterations.

        # (the step size is small enough that a coarse schedule works)
        # Here we use a decreasing schedule:
        #   eta = eta_start * (1 - decay) + eta_end * decay
        #   gamma = gamma_start * (1 - decay) + gamma_end * decay
        # where decay = iteration / max_iter_est (approx 200000)
        # To avoid tracking max_iter we use time‑based decay:
        now = time.time()
        # progress ∈ [0,1] across the whole allowed time
        # (global_time_fraction helps to keep schedules consistent)
        global_start = run_optimiser.global_start_time
        global_deadline = run_optimiser.global_deadline
        progress = (now - global_start) / (global_deadline - global_start)
        gamma = gamma_start + progress * (gamma_end - gamma_start)
        eta = eta_start + progress * (eta_end - eta_start)

        # --------------------------------------------------------------
        # 1) evaluate objective and gradient
        # --------------------------------------------------------------
        obj, grad = smoothed_objective_and_grad(cur, n, size_conv, size_grad, gamma)

        # keep the best seen in this run
        if obj < best_val - 1e-15:
            best_val, best_a = obj, cur.copy()
            no_improve = 0
            print(f"[run] iter {int(now):05d} new best = {best_val:.6f}")
            # early stop if we already beat the global target
            if best_val <= 1.503:
                break
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 2) gradient step + simplex projection
        # --------------------------------------------------------------
        step = eta * grad
        cand = cur - step
        # enforce non‑negativity before projection (helps numeric stability)
        cand = np.maximum(cand, 0.0)
        cur = proj_simplex(cand)

        # --------------------------------------------------------------
        # 3) occasional jitter to escape stagnation
        # --------------------------------------------------------------
        if no_improve >= stagnation_limit:
            noise = rng.normal(scale=0.001, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur = proj_simplex(cur)
            no_improve = 0
            print(f"[run] jitter after stagnation at iter {int(now):05d}")

    # Return the best vector found in this run and its objective value.
    return best_a, best_val


# Store global timestamps for schedule computation (set by propose_candidate)
run_optimiser.global_start_time = 0.0
run_optimiser.global_deadline = 1.0


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The method is a time‑bounded series of projected‑gradient runs with a
    smooth max‑convolution surrogate.  It returns the best sequence as a plain
    Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin
    run_optimiser.global_start_time = start_time
    run_optimiser.global_deadline = deadline

    # ------------------------------------------------------------------
    # Problem size – a balance between quality and speed
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise with the previous solution if available, otherwise random
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        # a fresh Dirichlet draw (concentration 1 → uniform on the simplex)
        a0 = np.random.default_rng(seed).dirichlet(np.ones(n))

    best_a = a0.copy()
    # evaluate the true objective once (equivalent to evaluate_sequence)
    conv_len = 2 * n - 1
    size_conv = 1 << (conv_len - 1).bit_length()
    b0 = full_convolution(best_a, size_conv)
    best_val = 2.0 * n * b0.max()
    print(f"[init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned empirically)
    # ------------------------------------------------------------------
    gamma_start = 0.5          # smoothness at the beginning
    gamma_end = 30.0           # almost exact max at the end
    eta_start = 0.8            # aggressive first steps
    eta_end = 0.001            # fine‑grained final steps
    stagnation_limit = 25000   # jitter after this many non‑improving steps

    rng = np.random.default_rng(seed + 12345)
    restart_counter = 0

    # ------------------------------------------------------------------
    # Main restart loop – each slice gets a few seconds of work
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # give each restart a modest budget (0.5 s … 5 s)
        slice_budget = max(0.5, min(5.0, 0.015 * remaining))

        slice_start = time.time()
        slice_deadline = slice_start + slice_budget

        # 0.5 probability: start from the global best with a tiny jitter
        # 0.5 probability: fresh Dirichlet draw
        if rng.random() < 0.5:
            a_start = best_a.copy()
            noise = rng.normal(scale=0.001, size=n)
            a_start = np.maximum(a_start + noise, 0.0)
            a_start = proj_simplex(a_start)
        else:
            alpha = 10.0 ** rng.uniform(-2, 2)          # log‑uniform concentration
            a_start = rng.dirichlet(alpha * np.ones(n))

        # run the optimiser on this slice
        a_local, val_local = run_optimiser(
            a_start,
            n,
            slice_deadline,
            best_a,
            best_val,
            gamma_start,
            gamma_end,
            eta_start,
            eta_end,
            stagnation_limit=stagnation_limit,
        )

        # keep the global best
        if val_local < best_val - 1e-12:
            best_val, best_a = val_local, a_local.copy()
            print(f"[restart {restart_counter:03d}] new global best = {best_val:.6f}")

        # early exit if we have reached the target
        if best_val <= 1.503:
            print("[target] objective reached, terminating early.")
            break

        restart_counter += 1

    # ------------------------------------------------------------------
    # Final tiny mixing with the uniform vector (often reduces the peak a bit)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=float)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * best_a + alpha * uniform
        cand = proj_simplex(cand)                     # keep on simplex
        # true objective (no need for evaluate_sequence inside the loop)
        b = full_convolution(cand, size_conv)
        val = 2.0 * n * b.max()
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[mix] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final verification using the provided secure wrapper (optional)
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list (the harness expects this)
    return best_a.tolist()
```
