def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, scipy.optimize, scipy.sparse as sp
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # Choose n based on time: if budget > 600 seconds, use n=4000 else n=2000
    if budget_s > 600:
        n = 4000
    else:
        n = 2000
    # Helper: objective returns float.
    def obj(a):
        return float(2.0 * n * np.max(np.convolve(a, a)) / (np.sum(a)**2))
    # Generate initial candidates.
    best_a = None
    best_val = np.inf
    def try_candidate(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=float)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s   # scale to unit sum (scale invariant)
        val = obj(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    # Uniform
    try_candidate(np.ones(n))
    # Half-sine
    try_candidate(np.sin(np.pi * (np.arange(1, n+1) / (n+1))))
    # Exponential V shape
    for r in (0.5, 0.7, 0.8, 0.9):
        i = np.arange(n)
        a = r ** i + r ** (n - 1 - i)
        try_candidate(a)
    # Power law: a_i = (i+1)^p * (n-i)^p
    for p in (0.2, 0.5, 1.0):
        i = np.arange(n)
        a = ((i+1) * (n-i)) ** p
        try_candidate(a)
    # Random Dirichlet
    for _ in range(10):
        a = rng.dirichlet(np.ones(n))
        try_candidate(a)
    # Ensure we have a candidate
    if best_a is None:
        best_a = np.full(n, 1.0/n)
        best_val = obj(best_a)
    print(f"[init] best value {best_val:.6f}")
    # Phase 1: LP mixing iterations
    def lp_step(a, t_mix=0.3):
        n = a.shape[0]
        M = np.max(np.convolve(a, a))
        # Build sparse matrix
        rows = []
        cols = []
        data = []
        # For each shift k
        for k in range(2*n-1):
            i_start = max(0, k-(n-1))
            i_end = min(k, n-1)
            # i loops over a indices where a_i contributes
            for i in range(i_start, i_end+1):
                j = k - i
                rows.append(k)
                cols.append(j)
                data.append(a[i])

        bounds = [(0, None)] * n

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:

            return a
        g0 = res.x

"""
search_min_max_conv.py

Search for a non‑negative step‑function that minimises the evaluation
function

    2 * n * max(conv(a, a)) / (sum(a)**2)

(the objective is scale‑invariant, so we keep sum(a)=1).

The algorithm consists of three stages:

1. **Initial palette** – a handful of hand‑crafted families (uniform,
   half‑sine, exponential “V‑shapes”, power‑law, Dirichlet draws) are
   examined and the best candidate is kept.

2. **LP improvement loop** – given the current vector `a` and its maximal
   self‑convolution `M = max(conv(a,a))`, we solve the linear program

        maximise   Σ g_j
        subject to  conv(a, g) ≤ M   (component‑wise)
                    g_j ≥ 0

   The solution `g` is rescaled to the same ℓ₁‑norm as `a` and mixed back
   via a convex combination `a ← (1‑t)·a + t·g`.  The LP never increases
   the maximal convolution, and a few such mixes already reduce the
   objective dramatically.

3. **Exact sub‑gradient polishing** – we finish with a projected
   sub‑gradient descent on the true max‑convolution objective and a final
   tiny uniform mixing.

All work respects the 1000 s budget and returns the best sequence found.
"""

def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1)×n matrix A that maps g ↦ conv(a, g).

    Row k of A contains a_i at column j = k‑i whenever 0 ≤ i,j < n,
    i.e. (A·g)[k] = Σ_i a_i·g_{k‑i}.
    """
    n = a.shape[0]
    # pre‑allocate arrays for the sparse representation
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = i + np.arange(n)          # k = i + j
        cols[pos:pos + n] = np.arange(n)              # column j
        data[pos:pos + n] = a[i]                      # coefficient a_i
        pos += n

    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_improve(a: np.ndarray, mix: float = 0.35) -> np.ndarray:
    """
    One LP‑based improvement step.

    1) M = max(conv(a,a)).
    2) Solve maximise Σ g_j  s.t. conv(a,g) ≤ M, g ≥ 0.
    3) Rescale g to have the same ℓ₁‑norm as a.
    4) Return the convex mixture (1‑mix)·a + mix·g, projected onto the
       probability simplex.
    """
    n = a.shape[0]
    M = _max_convolution(a)

    # Build the linear constraints A·g ≤ M.
    A = _build_conv_matrix(a)
    c = -np.ones(n)                     # maximise Σ g → minimise -Σ g
    bounds = [(0, None)] * n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Solve the LP (highs is fast for moderate sizes).
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=bounds, method='highs',
                                options={'presolve': True})
    if not res.success:
        # If the LP fails, simply return the original vector.
        return a

    g0 = res.x
    sum_g = g0.sum()
    if sum_g == 0.0:
        return a

    # Rescale to keep the ℓ₁‑norm unchanged.
    g = g0 * (a.sum() / sum_g)

    # Convex combination.
    cand = (1.0 - mix) * a + mix * g
    cand = np.maximum(cand, 0.0)
    cand /= cand.sum()
    return cand


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto {w ≥ 0, Σ w = 1}.
    """
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_descent(a: np.ndarray,
                         max_iter: int = 50000,
                         step0: float = 0.5) -> np.ndarray:
    """
    Exact projected sub‑gradient descent on the true max‑convolution
    objective (no surrogate).  The vector a is assumed to have unit sum.
    """
    n = a.shape[0]
    best_val = _objective(a)
    best_a = a.copy()

    eta = step0 / n          # initial step size ≈ half the average entry
    decay = 0.9995

    for it in range(max_iter):
        if it % 5000 == 0 and time.time() > _deadline - 1.0:
            break

        b = np.convolve(a, a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SGD {it:06d}] new best = {best_val:.6f}")

        k = int(np.argmax(b))

        # Sub‑gradient of the maximal entry (convex because all a ≥ 0)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # Gradient step + projection onto the probability simplex
        v = a - eta * grad
        a = _project_onto_simplex(v)

        eta *= decay

    return best_a

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function under the given time budget.

    Returns a list of non‑negative floats that (approximately) minimise
    `evaluate_sequence`.  The implementation follows the three‑stage
    strategy described in the module docstring.
    """
    global _deadline  # used by the sub‑gradient routine for early exit
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    _deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 2000 gives a good trade‑off between expressive power
    # and LP tractability for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # 1️⃣  Initialise with a hand‑crafted palette
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    def _consider(v):
        nonlocal best_a, best_val
        v = np.asarray(v, dtype=float)
        if v.shape != (n,):
            return
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0:
            return
        v /= s
        val = _objective(v)
        if val < best_val:
            best_val = val
            best_a = v.copy()

    # uniform
    _consider(np.ones(n))

    # half‑sine
    _consider(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))

    # exponential V‑shapes
    for r in (0.5, 0.7, 0.8, 0.9):
        i = np.arange(n, dtype=float)
        _consider(r ** i + r ** (n - 1 - i))

    # power‑law V‑shapes
    for p in (0.2, 0.5, 1.0):
        i = np.arange(n, dtype=float)
        _consider(((i + 1) * (n - i)) ** p)

    # random Dirichlet seeds
    for _ in range(15):
        _consider(rng.dirichlet(np.ones(n)))

    # try the globally stored candidate if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old = np.maximum(old, 0.0)
        old = old / old.sum()
        if old.size != n:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            old = np.interp(x_new, x_old, old)
            old = np.maximum(old, 0.0)
            old = old / old.sum()
        _consider(old)
    except Exception:
        pass

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)

    print(f"[init] best objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  LP improvement loop
    # ------------------------------------------------------------------
    lp_rounds = 12
    mix_factor = 0.35
    for it in range(lp_rounds):
        if time.time() > _deadline - 5.0:
            break
        cand = _lp_improve(best_a, mix=mix_factor)
        val_cand = _objective(cand)
        if val_cand < best_val:
            best_val = val_cand
            best_a = cand
            print(f"[LP  {it:02d}] new best = {best_val:.6f}")
        else:
            # keep the candidate as a new base for the next round – it may
            # open a different facet of the feasible region.
            best_a = cand
        mix_factor = max(0.1, mix_factor * 0.95)   # slowly decay the step size

    # ------------------------------------------------------------------
    # 3️⃣  Exact sub‑gradient polishing
    # ------------------------------------------------------------------
    best_a = _subgradient_descent(best_a)

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny uniform mixing (often grants a marginal gain)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_a = cand

    # ------------------------------------------------------------------
    # Report the final value (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
