import numpy as np
import time

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _objective(a, n):
    s = a.sum()
    if s < 0.01:
        return np.inf
    a_norm = a / s
    maxb = _full_convolution(a_norm).max()
    return float(2 * n * maxb)

    # time split
    param_deadline = start + 0.75 * (deadline - start)

    while time.time() < param_deadline:

            # log-uniform lam in [1e-6, 5]
            lam = 10 ** rng.uniform(-6, 0.5)   # lam in [1e-6, ~3.16]

# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

        else:
            ks = np.argpartition(b, -top_k)[-top_k:]

        g = np.zeros_like(cur)

        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            # contribution 2 * cur[k - idx] (see gradient of a_i*a_{k-i})
            g[idx] += 2.0 * cur[k - idx]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005

        cur = cur * np.exp(-eta * g)

        # renormalise to Σa = 1
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional jitter if stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.009, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb – deterministic mass transfer
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     start_frac: float = 0.30,
                     end_frac: float = 0.001,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move mass from the worst contributor to the best; accept only improvements."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing of the transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # active peak index
        k = int(np.argmax(b))

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev = np.zeros_like(cur)
        rev[idx] = cur[k - idx]

        src = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[src] = np.inf
        tgt = int(np.argmin(rev_tmp))

        if cur[src] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        new_val = 2.0 * n * _full_convolution(cur).max()

        if new_val < cur_val - 1e-15:
            # keep the move
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Uses a fast random‑parameter sweep followed by a
    short exponentiated‑gradient + hill‑climb refinement.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 123)

    start_time = time.time()
    deadline = start_time + budget_s - 0.4          # tiny safety margin
    n = 8000                                        # length of the vector
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Initial uniform candidate
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_val = _objective(best_a, n)
    print(f"[Init] uniform value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – fast random parametric search (≈70 % of total budget)
    # ------------------------------------------------------------------
    param_deadline = start_time + 0.70 * (deadline - start_time)

    while time.time() < param_deadline:
        typ = rng.choice(['exp', 'pow', 'dir'])
        if typ == 'exp':
            # λ in [1e‑6, 1]  (log‑uniform)
            lam = 10 ** rng.uniform(-6, 0)
            a = np.exp(-lam * np.arange(n, dtype=np.float64))
        elif typ == 'pow':
            # exponent p in [0, 8]
            p = rng.uniform(0, 8)
            a = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        else:  # Dirichlet draw with random concentration
            log_alpha = rng.uniform(-2, 2)          # α ∈ [10⁻², 10²]
            alpha = 10.0 ** log_alpha
            a = rng.dirichlet(alpha * np.ones(n))

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            continue
        a = a / s

        # try the raw ordering
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Param] new best ({typ}) = {best_val:.6f}")

        # also try the decreasingly sorted version (large‑small pairing)
        if not np.all(np.diff(a) <= 0):
            a_desc = np.sort(a)[::-1]
            val_desc = _objective(a_desc, n)
            if val_desc < best_val:
                best_val, best_a = val_desc, a_desc.copy()
                print(f"[Param] new best (sorted {typ}) = {best_val:.6f}")

        if best_val <= TARGET:
            print("[Target reached during param sweep]")
            break

    # ------------------------------------------------------------------
    # Stage 2 – refined exponentiated‑gradient descent (≈20 % of budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:          # give the refinement a couple of seconds at least
        expgrad_deadline = time.time() + 0.5 * remaining
        best_a, best_val = _expgrad_phase(
            best_a, n,
            start_time,
            expgrad_deadline,
            best_a,
            best_val,
            top_k=20,
            target_val=TARGET
        )

        # ------------------------------------------------------------------
        # Stage 3 – greedy hill‑climb + tiny uniform mix (≈10 % of budget)
        # ------------------------------------------------------------------
        hill_deadline = time.time() + (deadline - time.time())
        best_a, best_val = _hillclimb_phase(
            best_a, n,
            start_time,
            hill_deadline,
            best_a,
            best_val,
            start_frac=0.30,
            end_frac=0.001,
            target_val=TARGET
        )

        best_a, best_val = _mix_with_uniform(best_a, n)
        print(f"[Final] best value after refinement = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final verification via the provided secure wrapper (optional)
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list (sum ≈ 1)
    return best_a.tolist()
