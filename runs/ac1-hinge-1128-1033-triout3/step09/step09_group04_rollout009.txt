Implement projection:

def project_to_simplex(v):
    # Based on algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    it = 0
    no_improve = 0
    while time.time() < deadline:
        # compute b
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Softmax {it}] new best = {best_val:.6f}")
            if best_val <= target:
                break
        else:
            no_improve += 1

        # compute beta based on progress
        progress = (time.time() - start_time) / (deadline - start_time)
        beta = 1.0 + 999.0 * progress  # from 1 to 1000

        # compute softmax weights
        exponents = beta * (b - maxb)  # shift by max to avoid overflow
        w_unscaled = np.exp(exponents)

        # g = 2 * conv[n-1: n-1 + n]

        # We'll allocate size dynamic maybe next power of two.

def _gradient_softmax(a, w):
    # returns gradient vector (size len(a)), using FFT
    n = a.shape[0]

    # reversed a
    a_rev = a[::-1]
    size = 1 << ((len(w) + n - 1) - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)[:len(w)+n-1]
    # slice indices n-1 to n-1 + n
    g = 2.0 * conv[n-1: n-1 + n]   # shape n
    return g

def _softmax_phase(a0, n, start_time, deadline, best_a, best_val, target_val=1.503):
    cur = a0.copy()
    it = 0
    no_improve = 0
    max_beta = 5000.0  # might use high value for final iteration
    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute b and objective
        b = _full_convolution(cur)   # length 2n -1
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = cur.copy()
            no_improve = 0
            print(f"[Softmax {it}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # progress fraction
        progress = (now - start_time) / (deadline - start_time)
        # beta schedule: start ~1, grow to max_beta
        beta = 1.0 + (max_beta - 1.0) * progress ** 2  # maybe quadratic to accelerate later

        # compute softmax weights
        # Shift by maxb to avoid overflow
        exponents = beta * (b - maxb)
        w_unscaled = np.exp(exponents)
        w = w_unscaled / w_unscaled.sum()

        # compute gradient
        g = _gradient_softmax(cur, w)

        # step size schedule: from 0.5 -> 0.001
        eta = (1.0 - progress) * 0.5 + progress * 0.001

        # gradient descent update with projection onto simplex
        new_a = cur - eta * g
        cur = _project_to_simplex(new_a)

        # jitter if stuck
        if no_improve > 200_000:
            noise = np.random.normal(scale=0.005, size=n)
            cur = _project_to_simplex(cur + noise)
            no_improve = 0
            print(f"[Softmax {it}] jitter after stagnation")
        it += 1
    return best_a, best_val

def _project_to_simplex(v):
    # implementation returning projection onto simplex sum=1, v>=0
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.full(n, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _upscale(a_small, n_big):
    m = a_small.shape[0]
    x_small = np.linspace(0, 1, m)
    x_big = np.linspace(0, 1, n_big)
    a_big = np.interp(x_big, x_small, a_small)
    # ensure positivity
    a_big = np.maximum(a_big, 0.0)
    # renorm
    a_big /= a_big.sum()
    return a_big

"""
A new optimisation pipeline for the step‑function problem.

The algorithm works in three stages:

1. **Coarse search (small dimension).**
   - Use a short, smoothed‐gradient descent that minimises a
     “soft‑max” approximation of the maximal convolution entry.
   - Many random Dirichlet initialisations explore the space quickly.
   - The best coarse shape is interpolated to the final dimension.

2. **Fine tuning (full dimension).**
   - Run the same smoothed‑gradient descent for a longer
     time, then a deterministic hill‑climb that moves mass from the
     worst contributor to the best (as in the reference solution).
   - Finish with a tiny mixing with the uniform vector.

3. **Fast restarts.**
   - With the remaining budget, repeatedly initialise a random vector,
     perform a short gradient descent, a brief hill‑climb and mixing,
     keeping the globally best candidate.

The gradient of the soft‑max loss w.r.t. the coefficients can be written
as a convolution, which we evaluate with FFTs (≈ O(N log N)).  All
updates keep the vector on the probability simplex by projecting onto
the simplex after each step.

The code respects the 1000 s budget, uses only `numpy` and the
provided `evaluate_sequence`, and prints intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)-1) using FFT.
    Assumes a is real and non‑negative.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective returned by ``evaluate_sequence`` for a
    **L1‑normalised** vector a of length n.
    """
    s = a.sum()
    if s <= 0:
        return float("inf")
    a_norm = a / s
    maxb = _full_convolution(a_norm).max()
    return float(2.0 * n * maxb)      # denominator = (sum a)^2 = 1


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max loss w.r.t. a.

    a : vector of length n (assumed normalised)
    w : soft‑max weights of length 2·n‑1 (sum to 1)
    Returns a vector of length n.
    """
    n = a.shape[0]
    # reverse a for the correlation
    a_rev = a[::-1]
    size = 1 << ((len(w) + n - 1) - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv_full = np.fft.irfft(fw * fa, size)[: len(w) + n - 1]
    # slice the part that corresponds to the gradient
    g = 2.0 * conv_full[n - 1: n - 1 + n]
    return g


def _softmax_descent(
    a0: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    target_val: float = 1.503,
) -> np.ndarray:
    """
    Smoothed gradient descent that minimises a soft‑max approximation
    of the maximal convolution entry.
    Returns the final iterate (still L1‑normalised).
    """
    cur = a0.copy()
    max_beta = 200.0                     # final “sharpness” of the soft‑max
    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective (for early exit)
        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()
        if cur_val <= target_val:
            break

        # progress in [0,1]
        prog = (now - start_time) / (deadline - start_time)

        # soft‑max temperature (linear schedule)
        beta = 1.0 + (max_beta - 1.0) * prog

        # soft‑max weights (stable formulation)
        shift = b.max()
        exps = np.exp(beta * (b - shift))
        w = exps / exps.sum()

        # gradient w.r.t. a
        g = _gradient_softmax(cur, w)

        # step‑size schedule (aggressive → fine)
        eta = (1.0 - prog) * 0.5 + prog * 0.001

        # projected gradient step
        nxt = cur - eta * g
        cur = _project_to_simplex(nxt)

    return cur


def _hillclimb_phase(
    a_start: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
    start_frac: float = 0.30,
    end_frac: float = 0.001,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """Same deterministic hill‑climb as the reference implementation."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        prog = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - prog) + end_frac * prog

        k = int(np.argmax(b))

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev = np.zeros_like(cur)
        rev[idx] = cur[k - idx]

        src = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[src] = np.inf
        tgt = int(np.argmin(rev_tmp))

        if cur[src] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = _project_to_simplex(cur + noise)
                no_improve = 0
                print(f"[HillClimb {it}] jitter after stagnation")
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        cur[src] -= delta
        cur[tgt] += delta

        new_val = 2.0 * n * _full_convolution(cur).max()
        if new_val < cur_val - 1e-15:
            # keep the move
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try a few tiny convex combinations with the uniform vector."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_to_simplex(cand)   # re‑normalise safely
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


def _upscale(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from a smaller to a larger length."""
    m = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, m)
    x_big = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    return a_big

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence`` within the allocated time budget.
    Returns the best list found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 123)

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Keep any previously found sequence (if the global variable exists)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:
        old = None

    # ------------------------------------------------------------------
    # Coarse phase (small dimensionality → cheap exploration)
    # ------------------------------------------------------------------
    n_coarse = 2000
    if old is not None:
        # down‑sample the old solution to the coarse grid
        a0_coarse = np.interp(
            np.linspace(0.0, 1.0, n_coarse),
            np.linspace(0.0, 1.0, old.shape[0]),
            old,
        )
        a0_coarse = np.maximum(a0_coarse, 0.0)
        a0_coarse /= a0_coarse.sum()
    else:
        a0_coarse = np.full(n_coarse, 1.0 / n_coarse)

    best_a_coarse = a0_coarse.copy()
    best_val_coarse = _objective(best_a_coarse, n_coarse)
    print(f"[Coarse init] best_val = {best_val_coarse:.6f}")

    # Allocate ~15 % of the total budget to the coarse search
    coarse_deadline = start_time + 0.15 * (deadline - start_time)

    while time.time() < coarse_deadline:
        # random Dirichlet start (log‑uniform concentration)
        log_alpha = rng.uniform(-2, 2)          # α ∈ [10⁻², 10²]
        alpha = 10.0 ** log_alpha
        a = rng.dirichlet(alpha * np.ones(n_coarse))

        # a short soft‑max descent (≈0.2 s or until the slice expires)
        slice_deadline = min(coarse_deadline, time.time() + 0.25)
        a = _softmax_descent(a, n_coarse, time.time(), slice_deadline)

        val = _objective(a, n_coarse)
        if val < best_val_coarse:
            best_val_coarse, best_a_coarse = val, a.copy()
            print(f"[Coarse] new best = {best_val_coarse:.6f}")

    # ------------------------------------------------------------------
    # Upscale to the final resolution
    # ------------------------------------------------------------------
    n_fine = 8000
    best_a = _upscale(best_a_coarse, n_fine)
    best_val = _objective(best_a, n_fine)
    print(f"[Fine init] best_val = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Warm‑up fine optimisation (longer soft‑max descent + hill‑climb)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    warmup_deadline = time.time() + 0.20 * remaining

    best_a = _softmax_descent(best_a, n_fine, time.time(), warmup_deadline)
    best_val = _objective(best_a, n_fine)

    best_a, best_val = _hillclimb_phase(
        best_a,
        n_fine,
        time.time(),
        warmup_deadline,
        best_a,
        best_val,
        start_frac=0.30,
        end_frac=0.001,
    )
    best_a, best_val = _mix_with_uniform(best_a, n_fine)
    print(f"[Post‑warmup] best_val = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fast restarts with the remaining budget
    # ------------------------------------------------------------------
    restart_counter = 0
    TARGET = 1.503

    while time.time() < deadline:
        now = time.time()
        remaining = deadline - now
        # allocate a small slice (at least 0.6 s) for each restart
        slice_budget = max(0.6, 0.008 * remaining)
        slice_deadline = now + slice_budget

        # ---- random initialisation (Dirichlet) ----
        log_alpha = rng.uniform(-2, 2)
        alpha = 10.0 ** log_alpha
        a = rng.dirichlet(alpha * np.ones(n_fine))

        # ---- short soft‑max descent ----
        a = _softmax_descent(a, n_fine, now, slice_deadline)

        # ---- brief hill‑climb ----
        a, _ = _hillclimb_phase(
            a,
            n_fine,
            time.time(),
            deadline,
            best_a,
            best_val,
            start_frac=0.25,
            end_frac=0.001,
        )

        # ---- final tiny mixing ----
        a, val = _mix_with_uniform(a, n_fine)

        # ---- global update ----
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Restart {restart_counter}] new global best = {best_val:.6f}")
            if best_val <= TARGET:
                print("[Target reached] exiting early.")
                break

        restart_counter += 1

    # ------------------------------------------------------------------
    # Final verification (optional, uses the provided wrapper)
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list (sum ≈ 1)
    return best_a.tolist()
