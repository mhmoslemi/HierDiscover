def _full_convolution_fft(a):
    n = a.shape[0]
    out_len = 2*n-1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:out_len]
    return conv

"""
Projective gradient descent optimizer for step function sequence.

We minimize the evaluation function which depends on max of self‑convolution.
The algorithm:
1. Warm‑start from previous best or from a Beta‑family parametric sweep.
2. Perform a continuation method: minimise a smooth approximation of the
   max using projected gradient descent.  The smooth approximation is the
   log‑sum‑exp (soft‑max) of the convolution values.  α controls the
   smoothness and is increased slowly.
3. When time permits a final intensive hill‑climbing phase (similar to
   simulated‑annealing) refines the solution.
All operations use FFT based convolutions for O(n log n) time.
"""

import time
import math
import numpy as np

def _full_convolution_self(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _full_convolution_two(x, y):
    out_len = len(x) + len(y) - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _objective(a):
    n = a.shape[0]
    b = _full_convolution_self(a)
    return float(2.0 * n * b.max())

def _beta_family_best(n):
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)
    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val

def _resize_sequence(a, new_n):
    old_n = len(a)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 0:
        a_new /= a_new.sum()
    else:
        a_new = np.random.dirichlet(np.ones(new_n))
    return a_new

def _smooth_descent(a0, n, deadline, start_time, best_a, best_val):
    """
    Projected gradient descent on the smoothed objective
    f_alpha(a) = max(b) + log(sum(exp(alpha*(b-max))))/alpha.
    Alpha is increased slowly (continuation).
    """
    a = a0.copy()
    cur_val = _objective(a)
    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()
    # hyper‑parameters
    alpha = 1.0
    max_alpha = 200.0
    step = 0.5
    min_step = 1e-7
    iter_no_improve = 0
    IMPROVE_LIMIT = 2000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute convolution and soft‑max weights
        b = _full_convolution_self(a)
        max_b = b.max()
        # softmax weights w (sum to 1)
        exp_arg = alpha * (b - max_b)            # shift for stability
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * conv(w, a_rev) on the appropriate slice
        a_rev = a[::-1]
        c = _full_convolution_two(w, a_rev)
        grad = 2.0 * c[n-1:n-1+n]

        # try a step; simple back‑tracking line search
        step_try = step
        improved = False
        for _ in range(12):
            a_new = a - step_try * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 0.0:
                a_new /= s
            else:
                a_new = np.random.dirichlet(np.ones(n))
            new_val = _objective(a_new)
            if new_val < cur_val - 1e-12:
                improved = True
                break
            step_try *= 0.5

        if improved:
            a = a_new
            cur_val = new_val
            step = min(step_try * 1.2, 1.0)      # slowly increase step
            iter_no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[GD {iteration:6d}] time={now-start_time:.1f}s best={best_val:.6f} alpha={alpha:.2f} step={step:.4g}")
            # increase smoothness a little
            if iteration % 50 == 0:
                alpha = min(alpha * 1.05, max_alpha)
        else:
            # no improvement – shrink step and possibly perturb
            step = max(step * 0.5, min_step)
            iter_no_improve += 1
            if iter_no_improve >= IMPROVE_LIMIT:
                # random perturbation around current best
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                cur_val = _objective(a)
                iter_no_improve = 0
                print(f"[GD {iteration:6d}] stagnation – perturb")
        iteration += 1
    return best_a, best_val

def _hillclimb_anneal(a0, n, deadline, start_time, best_a, best_val, target_val=1.503):
    """
    Adapted version of the previous simulated‑annealing hill‑climber.
    It moves a small amount of mass from an index that contributes heavily
    to the current maximal convolution entry to an index that contributes
    little.  This phase is executed only if time permits after the gradient
    descent.
    """
    a = a0.copy()
    b = _full_convolution_self(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # efficient update of convolution
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution_self(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1
    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Entry point for the evaluation framework.

    The routine:
    1. Warm‑starts from a previous solution (``height_sequence_1``) when
       available, otherwise falls back to a two‑parameter Beta family search.
    2. Performs a continuation‑based projected gradient descent that
       minimises a smooth approximation of the max convolution.
    3. If time permits, refines the result with the simulated‑annealing hill‑climber.
    4. Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.5

    n = 8000

    # ---- 0️⃣ Warm start -------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        a0 = _resize_sequence(old, n)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    best_a, best_val = None, np.inf
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from previous best, value={best_val:.6f}")

    # beta‑family sweep
    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_val < best_val:
            best_a, best_val = beta_a, beta_val
            print(f"[init] beta‑family improved value={best_val:.6f}")
    except Exception as exc:

"""
Projected gradient descent optimiser for the step‑function attack.

The evaluation function depends only on the shape of the sequence
(because the objective is homogeneous of degree 0).  The algorithm
proceeds in three stages:

1️⃣ **Warm‑start** – use a previously found solution (global variable
   ``height_sequence_1``) if available, otherwise obtain a good start
   from a two‑parameter Beta family or from random Dirichlet draws.

2️⃣ **Continuation gradient descent** – minimise a smooth surrogate of
   the objective,
        f_α(a) = max(conv(a,a)) + log(∑exp(α·(conv‑max)))/α,
   which converges to the true max when the parameter α → ∞.
   The gradient of f_α is
        ∇f_α = 2·conv(w, a_rev)   (restricted to the original length),
   where w is the soft‑max weights of the convolution values.
   The method repeatedly projects the updated vector onto the simplex
   (non‑negative, unit sum) and slowly increases α.  A simple
   back‑tracking scheme adapts the step size.

3️⃣ **Optional hill‑climbing** – if time remains, a simulated‑annealing
   style local search refines the best sequence further (identical to
   the previously supplied hill‑climber).

All convolutions are performed with FFTs (O(n log n)).  The routine
respects the 1000 s budget and returns the best sequence found as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def _full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two possibly different vectors."""
    out_len = len(x) + len(y) - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (invariant to scaling)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)).  The vector a is assumed non‑negative."""
    n = a.shape[0]
    b = _full_convolution_self(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simple parametric sweep (Beta family) – cheap warm start
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search in the family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Rescaling / interpolation helper
# ----------------------------------------------------------------------
def _resize_sequence(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(a) to new_n, then normalise to sum = 1."""
    old_n = a.shape[0]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        a_new = np.random.dirichlet(np.ones(new_n))
    return a_new


# ----------------------------------------------------------------------
# Continuation – projected gradient descent on a smooth max surrogate
# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max‑convolution using
    projected gradient descent.  The smoothness parameter α is increased
    gradually (continuation).  The current best solution is returned.
    """
    a = a0.copy()
    cur_val = _objective(a)
    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()

    # hyper‑parameters
    alpha = 1.0
    max_alpha = 200.0
    step = 0.5               # initial learning rate
    min_step = 1e-7
    iter_no_improve = 0
    IMPROVE_LIMIT = 2500     # after this many non‑improving steps we perturb

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # 1) convolution and soft‑max weights
        b = _full_convolution_self(a)
        max_b = b.max()
        # w_i = exp(α·(b_i−max_b)) / Σ_j exp(α·(b_j−max_b))
        exp_arg = alpha * (b - max_b)          # shift for numerical stability
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 2) gradient = 2·conv(w, a_rev) restricted to original indices
        a_rev = a[::-1]
        c = _full_convolution_two(w, a_rev)    # length 3n‑2
        grad = 2.0 * c[n-1:n-1+n]               # slice of length n

        # 3) projected gradient step
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            a_new = np.random.dirichlet(np.ones(n))

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-15:          # accept improvement
            a = a_new
            cur_val = new_val
            step = min(step * 1.2, 1.0)        # gently increase step
            iter_no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[GD {iteration:6d}] time={now-start_time:.1f}s "
                      f"best={best_val:.6f} α={alpha:.2f} step={step:.4g}")
            # slowly increase the smoothness parameter
            if iteration % 30 == 0:
                alpha = min(alpha * 1.04, max_alpha)
        else:
            # reject – shrink step
            step = max(step * 0.5, min_step)
            iter_no_improve += 1
            if iter_no_improve >= IMPROVE_LIMIT:
                # perturb around the current best to escape stagnation
                a = best_a.copy()
                noise = np.random.normal(scale=0.02, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                cur_val = _objective(a)
                iter_no_improve = 0
                print(f"[GD {iteration:6d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (same spirit as the previous version)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Refine a solution by moving a small amount of mass from indices that
    strongly contribute to the current maximal convolution entry to indices
    that contribute little.  The schedule is stretched to the remaining
    time budget.
    """
    a = a0.copy()
    b = _full_convolution_self(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of the (first) maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each position to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – bias towards large contribution
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – bias towards small contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # amount of mass to move
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # update convolution efficiently (O(n))
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution_self(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val
