"""
A fast mirrored‑Adam optimiser for the max‑convolution objective.

Key ideas
---------
* Work on the probability simplex (Σa = 1, a ≥ 0) – scaling does not affect the
  evaluation, so we can keep the sum fixed.
* Replace the nonsmooth max‑convolution by a smooth log‑sum‑exp surrogate.
  The surrogate is gradually “hardened’’ by decreasing the temperature ε
  (0.5 → … → 0.001).  For the smallest ε we switch to a hard‑max weighting,
  i.e. the true sub‑gradient of max(b).
* Optimise the surrogate with a multiplicative Adam update
      a_i ← a_i·exp(−η·m̂_i/√v̂_i),
  which stays on the simplex after a single Euclidean projection.
  Adam’s moment estimates give a stable, self‑tuning step size.
* Periodically reorder the vector anti‑symmetrically (largest ↔ smallest);
  the rearrangement inequality guarantees that the true max‑convolution never
  increases.
* After the descent a cheap simulated‑annealing hill‑climber performs random
  swaps / tiny mass transfers and accepts occasional uphill moves.
* Multiple restarts (uniform / sinusoid / exponential) are tried until the
  time budget (≈1000 s) is almost exhausted.  The best vector found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise sometimes yields tiny negatives.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ_k p_k·b_k with respect to a,
    where b = a*a (convolution) and p is a probability vector (Σp=1, p≥0).

    The formula reduces to  Σ_k p_k·a_{k-i}.  One FFT suffices.
    """
    n = a.shape[0]
    p_rev = p[::-1]                     # reverse for convolution
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # The useful slice corresponds to positions n‑1 … 2n‑2 (inclusive)
    seg = conv[n - 1 : 2 * n - 1]
    # Reverse back to obtain the gradient vector
    return seg[::-1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest ones (pairwise).
    The rearrangement inequality guarantees that this never worsens
    the max‑convolution.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Weight‑construction helpers (soft‑ and hard‑max)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Soft‑max weights p ∝ exp(ε·conv) (numerically stable)."""
    shift = conv.max()
    exp_vals = np.exp(eps * (conv - shift))
    p = exp_vals / exp_vals.sum()
    return p


def _hardmax_weights(conv: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """Hard‑max: uniform weight on all indices whose value is within tol of the max."""
    max_val = conv.max()
    idx = np.where(conv >= max_val - tol)[0]
    p = np.zeros_like(conv)
    if idx.size:
        p[idx] = 1.0 / idx.size
    else:               # degenerate – should never happen
        p[conv.argmax()] = 1.0
    return p


# ----------------------------------------------------------------------
# Adam‑style multiplicative mirror update
# ----------------------------------------------------------------------
def _adam_update(a: np.ndarray, grad: np.ndarray,
                 m: np.ndarray, v: np.ndarray,
                 t: int, eta: float,
                 beta1: float = 0.9, beta2: float = 0.999,
                 eps_adam: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray, int]:
    """
    Perform one Adam‑style multiplicative mirror‑descent step.
    Returns the updated (a, m, v, t).
    """
    t += 1
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    # Bias‑correction
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    # Multiplicative factor (clipped to avoid overflow/underflow)
    delta = -eta * m_hat / (np.sqrt(v_hat) + eps_adam)
    delta = np.clip(delta, -20.0, 20.0)
    a = a * np.exp(delta)

    # Back to the simplex (preserves non‑negativity)
    a = _project_simplex(a)
    return a, m, v, t


# ----------------------------------------------------------------------
# Main optimisation loop (mirrored Adam on the log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _adam_descent(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise the smoothed maximum of the convolution using Adam on the
    simplex.  Returns the best vector encountered and its true evaluation.
    """
    a = a_start.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] objective = {best_val:.6f}")

    # Temperature schedule (large → small).  Tiny eps allows us to use the true
    # sub‑gradient of the maximum.
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    # Rough target: ~250‑300 inner iterations per ε‑stage (≈ 2700 total).
    it_per_eps = 250

    for eps in eps_schedule:
        eta = 0.7                       # initial step size for this stage
        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:
                break

            # --- 1️⃣ convolution and weighting --------------------------------
            conv = _full_convolution(a)

            if eps >= 0.02:            # softmax (smooth surrogate)
                p = _softmax_weights(conv, eps)
            else:                      # hardmax – true sub‑gradient of max
                p = _hardmax_weights(conv, tol=1e-12)

            # --- 2️⃣ gradient of the surrogate (2· because ∂b_k/∂a_i = 2·a_{k-i})
            grad = 2.0 * _gradient_weighted_peaks(a, p)

            # --- 3️⃣ Adam multiplicative update --------------------------------
            a, m, v, t = _adam_update(a, grad, m, v, t, eta)

            # --- 4️⃣ occasional bookkeeping ------------------------------------
            if it % 50 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.4f} it {it}] new best = {best_val:.6f}")

            # --- 5️⃣ anti‑symmetric re‑ordering (helps bound) -----------------
            if it % 200 == 0:
                a = _anti_symmetric_reorder(a)

            # --- 6️⃣ slow decay of the learning rate ---------------------------
            eta *= 0.998

        # End of ε‑stage – final re‑ordering (cannot increase the max)
        a = _anti_symmetric_reorder(a)

        # If time runs out early we exit the outer loop as well.
        if time.time() >= deadline - 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (tiny swaps / mass transfers)
# ----------------------------------------------------------------------
def _annealing_refine(a: np.ndarray, cur_val: float,
                      deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing local search: random swaps or small mass moves.
    Accepts uphill moves with a temperature that decays linearly towards zero.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    start = time.time()
    total = max(deadline - start - 2.0, 0.0)

    # Temperature schedule: start relatively warm, finish cold.
    temp_start = 0.05
    temp_end = 1e-5

    while time.time() < deadline - 2.0:
        # Linear interpolation of temperature
        elapsed = time.time() - start
        frac = elapsed / total if total > 0 else 1.0
        temp = (1.0 - frac) * temp_start + frac * temp_end

        # --- propose a move -------------------------------------------------
        if rng.random() < 0.30:        # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                          # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.07 * rng.random()   # up to 7 % of source
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Keep feasibility (non‑negative, same total mass)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0:
            continue
        new_a /= s

        # --- evaluate -------------------------------------------------------
        val = evaluate_sequence(new_a.tolist())
        delta_val = val - best_val

        # Accept if better, otherwise with Metropolis probability
        if delta_val < -1e-12:
            best_a, best_val = new_a, val
            print(f"[anneal] improved to {best_val:.6f}")
        else:
            if rng.random() < np.exp(-delta_val / temp):
                best_a, best_val = new_a, val

        # Occasionally try the anti‑symmetric re‑ordering (free move)
        if rng.random() < 0.02:
            cand = _anti_symmetric_reorder(best_a)
            cand_val = evaluate_sequence(cand.tolist())
            if cand_val < best_val:
                best_a, best_val = cand, cand_val
                print(f"[anneal] reorder improved to {best_val:.6f}")

    return best_a, best_val
