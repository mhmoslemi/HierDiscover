def _full_corr(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    # cross-correlation of a (len n) and b (len m) returning length n+m-1
    # cross-correlation a ⋆ b at lag i: sum_j a_j * b_{i+j}
    # Use FFT: corr = ifft(FFT(a[::-1], size) * FFT(b, size))
    n = a.shape[0]; m = b.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, size)
    B = np.fft.rfft(b, size)
    conv = np.fft.irfft(A * B, size)[:out_len]
    return conv

Implementation of projection onto simplex:
def _project_onto_simplex(v):
    """Project vector v onto simplex {x >=0, sum x =1}."""
    # Implementation due to Wang & Carreira-Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Step‑function optimizer (enhanced).

We explore a richer family of parametric shapes (exponential,

# Fast linear convolution (FFT, full length)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)−1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Full cross‑correlation a ⋆ b (length len(a)+len(b)−1)
# ----------------------------------------------------------------------
def _full_correlation(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Cross‑correlation (a ⋆ b)[i] = Σ_j a_j · b_{i+j}."""
    n, m = a.shape[0], b.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, size)
    B = np.fft.rfft(b, size)
    corr = np.fft.irfft(A * B, size)[:out_len]
    return corr


# ----------------------------------------------------------------------
# Objective used during optimisation (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a normalized vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σx = 1}."""
    # Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper families used for the random sweep
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    return a / a.sum()


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    return a / a.sum()


def _gen_power(n: int, p: float) -> np.ndarray:
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    return a / a.sum()


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    return a / a.sum()


# ----------------------------------------------------------------------
# Initial deterministic sweep (quick families)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n or np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective(vec, n)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    rng = np.random.default_rng(12345)
    idx = np.arange(n)

    # a few hand‑crafted patterns
    try_update(np.arange(n, 0, -1, dtype=float))
    try_update(np.maximum(idx + 1, n - idx))
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99,
              0.98, 0.95):
        i = np.arange(n, dtype=np.float64)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # a few random Dirichlet draws
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 0 – systematic sweep of symmetric exponentials (very cheap)
# ----------------------------------------------------------------------
def _sweep_symmetric_exp(n: int, best_a: np.ndarray, best_val: float,
                         deadline: float):
    """Search r∈[low,high] for the symmetric (r^i+r^{n‑1‑i}) family."""
    # we focus on the region where r is very close to 1.
    lows, highs = 0.998, 0.99995
    nr = 400                     # resolution – cheap under the time budget
    for r in np.linspace(lows, highs, nr):
        a = (np.power(r, np.arange(n, dtype=np.float64)) +
             np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64)))
        a = a / a.sum()
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – L2‑smoothing (projected gradient descent)
# ----------------------------------------------------------------------
def _l2_smoothing_phase(a0: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Project‑gradient descent on  Σ (conv(a,a)−μ)²  with μ=1/(2n−1)."""
    cur = a0.copy()
    mu = 1.0 / (2 * n - 1)                 # ideal constant value of convolution
    it = 0
    no_improve = 0
    IMPATIENT = 30_000

    while time.time() < deadline:
        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[L2  {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # gradient of L2 loss  L = Σ (b-μ)²  ->  ∇L = 4·(cur ⋆ (b-μ))
        diff = b - mu
        corr = _full_correlation(cur, diff)          # length 3n‑2
        grad = 4.0 * corr[n - 1:n - 1 + n]           # central n entries

        # step‑size schedule (large at start, tiny near the end)
        progress = (time.time() - start) / (deadline - start)
        eta = (1.0 - progress) * 0.5 + progress * 0.001

        # additive update followed by projection onto the simplex
        cand = cur - eta * grad
        cur = _simplex_projection(cand)

        if no_improve > IMPATIENT:
            # a little random jitter to escape possible plateaus
            noise = np.random.normal(scale=0.01, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur = _simplex_projection(cur)
            no_improve = 0
            print(f"[L2  {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – exponentiated‑gradient descent on the max‑convolution
# ----------------------------------------------------------------------
def _expgrad_phase(a0: np.ndarray, n: int,
                   start: float, deadline: float,
                   best_a: np.ndarray, best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    while time.time() < deadline:
        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]               # a_{k‑i}
        g = 2.0 * rev                           # sub‑gradient of the max

        progress = (time.time() - start) / (deadline - start)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005

        cur = cur * np.exp(-eta * g)
        cur /= cur.sum()
        if cur.sum() == 0.0:
            cur = np.full(n, 1.0 / n)

        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.008, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – deterministic greedy hill‑climbing
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while time.time() < deadline:
        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))               # index with biggest contribution
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))            # index with smallest contribution

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur = _simplex_projection(cur)
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            it += 1
            continue

        cur[i] -= delta
        cur[j] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-15:
            cur_val = new_val
        else:
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 4 – final uniform mixing (tiny blend with the flat vector)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _simplex_projection(cand)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under a time budget.
    Returns the best list of floats (Σa = 1) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – a tiny buffer before the judge kills the process
    total_deadline = start_time + budget_s - 0.5
    total_time = total_deadline - start_time

    # Problem size – 8000 provides a good trade‑off.
    n = 8000

    # ------------------------------------------------------------------
    # Initialise (reuse a previous solution if present)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase‑0 – random exploration of rich parametric families
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    random_deadline = start_time + 0.15 * total_time

    while time.time() < random_deadline:
        choice = rng.integers(0, 5)   # five families
        if choice == 0:                     # pure exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_exp(n, r)
        elif choice == 1:                   # reversed exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_rev_exp(n, r)
        elif choice == 2:                   # mixture of two exponentials
            r1 = rng.uniform(0.90, 0.99999)
            r2 = rng.uniform(0.90, 0.99999)
            w = rng.random()
            cand = _gen_mix_exp(n, r1, r2, w)
        elif choice == 3:                   # pure power‑law
            p = rng.uniform(-5.0, 5.0)
            cand = _gen_power(n, p)
        else:                               # power‑law × exponential
            p = rng.uniform(-5.0, 5.0)
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_power_exp(n, p, r)

        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Random] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑0‑b – systematic sweep of symmetric exponentials (cheap)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 0.20 * total_time
    best_a, best_val = _sweep_symmetric_exp(n, best_a, best_val, sweep_deadline)

    # ------------------------------------------------------------------
    # Phase‑1 – L2 smoothing (projected gradient descent)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.45 * total_time
    best_a, best_val = _l2_smoothing_phase(
        best_a, n,
        start=start_time,
        deadline=phase1_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase‑2 – exponentiated‑gradient descent (max‑convolution)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * total_time
    best_a, best_val = _expgrad_phase(
        best_a, n,
        start=start_time,
        deadline=phase2_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase‑3 – deterministic greedy hill‑climbing
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.95 * total_time
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=start_time,
        deadline=phase3_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase‑4 – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification with the provided security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
