def expgrad_opt(a0, n, deadline, top_k=5, start_eta=0.4, end_eta=1e-5, target_val=1.503):
    a = np.copy(a0)
    best_a = a.copy()
    best_val = objective(a, n)
    it = 0
    no_improve = 0
    IMPATIENT = 200000
    while time.time() < deadline:
        b = full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b  # since sum a approx 1
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[ExpGrad {it}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1
        # top indices
        if top_k == 1:
            ks = [int(np.argmax(b))]
        else:
            ks = np.argpartition(b, -top_k)[-top_k:]
        g = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi+1)
            g[idx] += 2.0 * a[k - idx]
        # step size schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * start_eta + progress * end_eta
        a = a * np.exp(-eta * g)
        # renorm
        total = a.sum()
        if total <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total
        if no_improve >= IMPATIENT:
            # add jitter
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            no_improve = 0
            print(f"[ExpGrad {it}] jitter after stagnation")
        it += 1
    return best_a, best_val

def hillclimb_opt(a0, n, deadline, start_frac=0.3, end_frac=0.001, target_val=1.503):
    a = np.copy(a0)
    best_a = a.copy()
    best_val = objective(a, n)
    it = 0
    no_improve = 0
    IMPATIENT = 200000
    while time.time() < deadline:
        b = full_convolution(a)
        max_idx = int(np.argmax(b))
        max_b = b[max_idx]
        cur_val = 2.0 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[HillClimb {it}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1
        # compute rev contributions to max_idx
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi+1)
        rev = np.zeros_like(a)
        rev[idx] = a[max_idx - idx]
        src = int(np.argmax(rev))
        # target is argmin rev (excluding src)
        rev_tmp = rev.copy()
        rev_tmp[src] = np.inf
        tgt = int(np.argmin(rev_tmp))
        if a[src] <= 1e-15:
            # jitter if needed
            if no_improve >= IMPATIENT:
                noise = np.random.normal(scale=0.01, size=n)
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                no_improve = 0
                print(f"[HillClimb {it}] jitter")
            it += 1
            continue
        # fraction schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        frac = start_frac * (1 - progress) + end_frac * progress
        delta = a[src] * frac
        if delta <= 0:
            it += 1
            continue
        # trial move
        a[src] -= delta
        a[tgt] += delta
        b_new = full_convolution(a)
        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-12:
            # accept and continue; keep best updated later
            pass
        else:
            # revert
            a[src] += delta
            a[tgt] -= delta
        it += 1
    return best_a, best_val

"""
search for a non‑negative step function that minimises `evaluate_sequence`.

The strategy improves over the reference implementation in several ways:

1. **Rich parametric seeding** – a collection of analytically‑motivated
   families (power‑law, double‑exponential tails, V‑shape, etc.) is scanned
   quickly; the best of these seeds is used as the starting point.

2. **Aggressive exponentiated‑gradient phase** – we use a larger active set
   (top‑K≈5) and a smooth linear decay of the learning rate from 0.3 to
   1e‑5.  Stagnation triggers a small Gaussian jitter.

3. **Extended deterministic hill‑climb** – the move fraction is annealed
   from 30 % to 0.1 % and the transfer is accepted only when it truly
   reduces the objective.

4. **Iterative refinement & cheap random restarts** – after an initial
   long optimisation (≈40 % of the budget) we perform many short restarts
   with fresh Dirichlet draws; each restart runs a brief gradient phase,
   a hill‑climb and a tiny uniform mixing.

5. **Time‑aware scheduling** – every helper receives the global start
   time and a per‑phase deadline, guaranteeing termination before the
   global 1000 s budget.

The code respects all sandbox constraints (no I/O, top‑level functions only,
no lambdas) and prints progress information.  The returned list is a plain
Python list of floats; the final call to `evaluate_sequence` is only for
diagnostic output.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (full linear convolution via FFT)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (exactly the same as `evaluate_sequence`)
# ----------------------------------------------------------------------
def objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a) / (sum a)^2  (identical to evaluate_sequence)."""
    b = full_convolution(a)
    max_b = b.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple parametric families – quick initial seeding
# ----------------------------------------------------------------------
def generate_initial_candidate(n: int):
    """Return a promising start vector obtained by scanning a few families."""
    best_a = np.full(n, 1.0 / n)
    best_val = objective(best_a, n)

    def consider(v):
        nonlocal best_a, best_val
        v = np.asarray(v, dtype=np.float64)
        if v.ndim != 1 or v.shape[0] != n:
            return
        if np.any(v < 0):
            return
        v_sum = v.sum()
        if v_sum <= 0:
            return
        v = v / v_sum
        val = objective(v, n)
        if val < best_val:
            best_val, best_a = val, v.copy()

    rng = np.random.default_rng(0)

    # power‑law / U‑shaped families (negative exponents)
    for p in np.linspace(-3.0, -0.5, 6):
        x = (np.arange(1, n + 1, dtype=np.float64) ** p) + \
            (np.arange(n, 0, -1, dtype=np.float64) ** p)
        consider(x)

    # double‑exponential tails (mass concentrated near the ends)
    for sigma in (5, 10, 20, 50, 100, 200, 500, 1000):
        i = np.arange(n, dtype=np.float64)
        x = np.exp(-i / sigma) + np.exp(-(n - 1 - i) / sigma)
        consider(x)

    # V‑shape (linear decreasing from the centre)
    x = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    consider(x)

    # sqrt‑U shape (edges heavier than centre)
    x = np.sqrt(np.arange(1, n + 1)) + np.sqrt(np.arange(n, 0, -1))
    consider(x)

    # piecewise constant + end spikes
    end_len = max(1, n // 20)
    a = np.full(n, 0.6 / (n - 2 * end_len))
    a[:end_len] = (0.4 / 2) / end_len
    a[-end_len:] = (0.4 / 2) / end_len
    consider(a)

    # a few random Dirichlet draws
    for _ in range(5):
        consider(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (aggregated sub‑gradient from top‑K peaks)
# ----------------------------------------------------------------------
def expgrad_opt(a0: np.ndarray,
                n: int,
                start_time: float,
                deadline: float,
                top_k: int = 5,
                start_eta: float = 0.30,
                end_eta: float = 1e-5,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Multiplicative updates using the sub‑gradient of the largest convolution peaks."""
    cur = a0.copy()
    best_a = cur.copy()
    best_val = objective(cur, n)

    it = 0
    no_improve = 0
    IMPATIENT = 200_000          # after this many stagnant steps add jitter

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = full_convolution(cur)
        max_b = b.max()
        cur_val = 2.0 * n * max_b / (cur.sum() ** 2)

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- aggregated sub‑gradient from the top‑K active peaks ----
        if top_k == 1:
            ks = [int(np.argmax(b))]
        else:
            ks = np.argpartition(b, -top_k)[-top_k:]

        g = np.zeros_like(cur)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            # contribution from a_{k-i}
            g[idx] += 2.0 * cur[k - idx]

        # linear decay of the learning rate
        progress = (now - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * start_eta + progress * end_eta

        # exponentiated update (keeps non‑negativity)
        cur = cur * np.exp(-eta * g)

        # renormalise to Σa = 1
        s = cur.sum()
        if s <= 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= s

        # occasional jitter to escape plateaus
        if no_improve >= IMPATIENT:
            noise = np.random.normal(scale=0.01, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climb – move mass from the biggest to the
# smallest contributor of the active peak.
# ----------------------------------------------------------------------
def hillclimb_opt(a0: np.ndarray,
                  n: int,
                  start_time: float,
                  deadline: float,
                  start_frac: float = 0.30,
                  end_frac: float = 0.001,
                  target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Mass transfer moves that only accept strict improvement."""
    cur = a0.copy()
    best_a = cur.copy()
    best_val = objective(cur, n)

    it = 0
    no_improve = 0
    IMPATIENT = 150_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = full_convolution(cur)
        k = int(np.argmax(b))               # index of current maximal convolution entry
        cur_val = 2.0 * n * b.max() / (cur.sum() ** 2)

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal the transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # range of indices that contribute to b[k]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = np.zeros_like(cur)
        contrib[idx] = cur[k - idx]

        src = int(np.argmax(contrib))
        # smallest contribution (do not pick src again)
        contrib_tmp = contrib.copy()
        contrib_tmp[src] = np.inf
        tgt = int(np.argmin(contrib_tmp))

        if cur[src] <= 1e-15:
            # source depleted – occasional shake‑up
            if no_improve >= IMPATIENT:
                noise = np.random.normal(scale=0.008, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            it += 1
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            it += 1
            continue

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        b_new = full_convolution(cur)
        new_val = 2.0 * n * b_new.max() / (cur.sum() ** 2)

        if new_val < cur_val - 1e-15:
            # accept – nothing else to do
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def mix_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a handful of convex combinations with the uniform vector."""
    best_a, best_val = a, objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
