def _full_convolution(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = 1 << ((out_len-1).bit_length())
    # we can use np.fft.rfft for speed.
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:out_len]
    return conv

def _smooth_gradient_phase(a0, n, start, deadline, best_a, best_val, target_val=1.503, alpha=1500.0, top_k=5):
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 150000
    while True:
        now = time.time()
        if now >= deadline:
            break
        conv = _full_convolution(cur)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothGrad {it:7d}] new best = {best_val:.6f}")
            if cur_val <= target_val:
                break
        else:
            no_improve += 1

        # Determine top indices
        # Use argpartition for speed
        if top_k >= len(conv):
            top_idxs = np.arange(len(conv))
        else:
            top_idxs = np.argpartition(conv, -top_k)[-top_k:]
        # Compute weights
        # Compute exp(alpha * (conv_i - maxc)) safe: <=0 => <=1
        w = np.exp(alpha * (conv[top_idxs] - maxc))
        w_sum = np.sum(w)
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0/len(w))
        else:
            w_norm = w / w_sum

        # accumulate gradient
        g = np.zeros(n, dtype=np.float64)
        for idx, weight in zip(top_idxs, w_norm):
            rev = _rev_vector(cur, idx)
            g += weight * (2.0 * rev)

        # learning rate schedule
        progress = (now - start) / (deadline - start)
        eta = (1 - progress) * 0.08 + progress * 0.0005  # more aggressive start

        # multiplicative update
        # ensure non-neg and not zero; we add small epsilon before exponent to avoid overflow
        cur = cur * np.exp(-eta * g)
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        if no_improve > IMPATIENT:
            # jitter
            noise = np.random.normal(scale=0.005, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SmoothGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val

"""
step‑function optimizer – enhanced version.

The optimizer works on a fixed length `n` (default 8000).  It starts
from a reasonable seed (the previously discovered `height_sequence_1`,
or a quick parametric sweep), then performs several complementary
phases:

* **Phase 0 – random parametric families** – many inexpensive samples of
  exponentials, power‑laws, mixtures and Gaussians.

* **Phase A – exponentiated‑gradient descent** – a classic multiplicative
  sub‑gradient update that drives the maximum of the self‑convolution
  down.

* **Phase B – deterministic greedy hill‑climb** – moves a small chunk
  of mass from a large‑gradient entry to a low‑gradient entry.

* **Phase C – smoothed‑gradient descent** – instead of using the single
  tightest convolution index, it builds a soft‑max over the top few
  indices and follows the resulting averaged sub‑gradient (still
  multiplicative).

* **Phase D – linear‑programming refinement** – solves a tiny LP that
  maximises the total mass of a partner vector `g` under the current
  convolution bound, then mixes a fraction of `g` into the current
  candidate.

* **Phase E – tiny uniform mixing** – adds a small amount of the uniform
  vector, which is known to improve the final bound slightly.

All phases respect the global time budget (`budget_s`, default 1000 s)
and keep the best sequence found.  The returned list has sum ≈ 1,
so `evaluate_sequence` returns the same value as the internal
objective `2·n·max(conv(a,a))`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:out_len]
    return conv


def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalized vector."""
    return float(2.0 * n * _full_convolution(a).max())


def _rev_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given convolution index k, return the vector `rev`
    with rev[i] = a[k‑i] (zero when the index is out of range).
    """
    n = a.shape[0]
    rev = np.zeros(n, dtype=a.dtype)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    rev[idx] = a[k - idx]
    return rev


def _lp_constraint_row(a: np.ndarray, k: int) -> np.ndarray:
    """Row of the LP matrix for the constraint (a * g)[k] ≤ M."""
    n = a.shape[0]
    row = np.zeros(n, dtype=np.float64)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    row[idx] = a[k - idx]
    return row


# ----------------------------------------------------------------------
# Parametric families (used in Phase 0)
# ----------------------------------------------------------------------


def _gen_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power(n: int, p: float) -> np.ndarray:
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)                     # stabilize
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs((2.0 * i) / (n - 1) - 1.0)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)

# Phase 0 – random exploration of rich families
# ----------------------------------------------------------------------


def _random_exploration(n: int,
                        rng: np.random.Generator,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """
    Draw many random parameterisations (exponential, power‑law,
    mixtures, Gaussian, triangular, Dirichlet) until `deadline`.
    """
    while time.time() < deadline:
        choice = rng.integers(0, 7)  # seven families

        if choice == 0:  # pure exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_exp(n, r)
        elif choice == 1:  # reversed exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_rev_exp(n, r)
        elif choice == 2:  # mixture of two exponentials
            r1 = rng.uniform(0.90, 0.99999)
            r2 = rng.uniform(0.90, 0.99999)
            w = rng.random()
            cand = _gen_mix_exp(n, r1, r2, w)
        elif choice == 3:  # power‑law
            p = rng.uniform(-5.0, 5.0)
            cand = _gen_power(n, p)
        elif choice == 4:  # power‑law × exponential
            p = rng.uniform(-5.0, 5.0)
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_power_exp(n, p, r)
        elif choice == 5:  # Gaussian bell
            # sigma between 5% and 30% of n
            sigma = rng.uniform(0.05 * n, 0.30 * n)
            cand = _gen_gaussian(n, sigma)
        else:  # triangular shape (deterministic)
            cand = _gen_triangular(n)

        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Random] new best = {best_val:.6f}")

        # occasional Dirichlet draws (heavy‑tailed)
        if rng.random() < 0.03:
            alpha = rng.uniform(0.1, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            val = _objective(cand, n)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Dirichlet] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – exponentiated‑gradient descent (sub‑gradient at the max)
# ----------------------------------------------------------------------


def _expgrad_phase(a0: np.ndarray,
                   n: int,
                   start: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 180_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        k = int(np.argmax(b))
        rev = _rev_vector(cur, k)
        g = 2.0 * rev  # sub‑gradient

        progress = (now - start) / (deadline - start)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005

        cur = cur * np.exp(-eta * g)
        total = cur.sum()
        cur = cur / total if total != 0.0 else np.full(n, 1.0 / n)

        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.008, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val

# Phase B – deterministic greedy hill‑climbing
# ----------------------------------------------------------------------


def _greedy_hillclimb(a_start: np.ndarray,
                      n: int,
                      start: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k = int(np.argmax(b))
        rev = _rev_vector(cur, k)

        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            cur_val = new_val
        else:
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – smoothed‑gradient descent (soft‑max over top indices)
# ----------------------------------------------------------------------


def _smooth_gradient_phase(a_start: np.ndarray,
                           n: int,
                           start: float,
                           deadline: float,
                           best_a: np.ndarray,
                           best_val: float,
                           target_val: float = 1.503,
                           alpha: float = 1500.0,
                           top_k: int = 5) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 150_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(cur)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # pick the `top_k` largest convolution entries
        if top_k >= conv.size:
            idxs = np.arange(conv.size)
        else:
            idxs = np.argpartition(conv, -top_k)[-top_k:]

        # soft‑max weights (all ≤ 1, 1 for the maximum)
        weights = np.exp(alpha * (conv[idxs] - maxc))
        w_sum = weights.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(weights, 1.0 / weights.size)
        else:
            w_norm = weights / w_sum

        # averaged sub‑gradient
        g = np.zeros(n, dtype=np.float64)
        for idx, w in zip(idxs, w_norm):
            rev = _rev_vector(cur, int(idx))
            g += w * (2.0 * rev)

        progress = (now - start) / (deadline - start)
        eta = (1.0 - progress) * 0.08 + progress * 0.0005

        cur = cur * np.exp(-eta * g)
        total = cur.sum()
        cur = cur / total if total != 0.0 else np.full(n, 1.0 / n)

        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.005, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SmoothGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – LP refinement (tiny linear program using the tightest constraints)
# ----------------------------------------------------------------------


def _lp_refine_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503,
                     max_iter: int = 5,
                     mix_weight: float = 0.2) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0

    while it < max_iter:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(cur)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            print(f"[LPrefine {it}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # choose a few (up to 5) binding constraints
        top_k = min(5, conv.size)
        idxs = np.argpartition(conv, -top_k)[-top_k:]

        A = np.vstack([_lp_constraint_row(cur, int(k)) for k in idxs])
        b = np.full(A.shape[0], maxc)

        c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  ⇔ minimise -sum(g)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"[LPrefine {it}] LP failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0.0:
            break
        g = g / g.sum()

        # convex combination
        cur = (1.0 - mix_weight) * cur + mix_weight * g
        it += 1

    # final check
    conv = _full_convolution(cur)
    cur_val = 2.0 * n * conv.max()
    if cur_val < best_val:
        best_val, best_a = cur_val, cur.copy()
        print(f"[LPrefine final] new best = {best_val:.6f}")

    return best_a, best_val

# Phase E – tiny uniform mixing
# ----------------------------------------------------------------------


def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)

    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

def _initial_candidate(n: int):
    """Return a decent start vector and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective(vec, n)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    idx = np.arange(n)

    # simple deterministic families
    try_update(np.arange(n, 0, -1, dtype=float))                 # decreasing
    try_update(np.maximum(idx + 1, n - idx))                    # V‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))         # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))  # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=np.float64)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # a few random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val

def propose_candidate(seed: int = 42,

    """

    n = 8000

    # Initialisation (reuse previous solution if present)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)

    # ------------------------------------------------------------------
    # Phase 0 – random exploration of rich parametric families
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase B – deterministic greedy hill‑climbing
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase C – smoothed‑gradient descent (soft‑max over top indices)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.85 * (deadline - start_time)
    best_a, best_val = _smooth_gradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
        alpha=1800.0,
        top_k=7
    )

    # ------------------------------------------------------------------
    # Phase D – LP refinement (tiny linear program)
    # ------------------------------------------------------------------
    phase_d_deadline = start_time + 0.95 * (deadline - start_time)
    best_a, best_val = _lp_refine_phase(
        best_a, n,
        start=phase_c_deadline,
        deadline=phase_d_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
        max_iter=6,
        mix_weight=0.15
    )

    # ------------------------------------------------------------------
    # Phase E – final uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification with the provided security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:                                   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
