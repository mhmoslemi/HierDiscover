"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initial seeds** – a mixture of (i) the previous best solution
   (if available), (ii) a few analytically‑motivated shapes
   (sin, sinⁿ, symmetric beta, triangular) and (iii) a random
   exponential vector.  All seeds are projected onto the simplex
   (sum = 1).

2. **Two‑stage descent**  
   *Stage A – smoothed max* – minimise a soft‑max surrogate
   `f_ε(a) = (1/ε)·log∑_k exp(ε·(a*a)[k])` for a decreasing schedule
   `ε ∈ {0.5,0.2,0.1,0.05,0.02,0.01}`.  The gradient is obtained by a
   single FFT‑based convolution; a multiplicative mirror‑descent step
   `a ← a·exp(−η·∇f_ε)` keeps the vector non‑negative, after which it is
   projected to the probability simplex.

   *Stage B – hard max* – after the smoothing schedule we switch to the
   true objective by using the (normalised) indicator of the maximal
   convolution entries as the weight vector.  This yields a sub‑gradient
   descent on `max(a*a)` and further reduces the peak.

   The step‑size `η` starts at 0.5 and decays slowly (factor 0.999 each
   iteration).  The best vector seen (according to the real
   ``evaluate_sequence``) is remembered.

3. **Local refinement** – a cheap stochastic hill‑climbing phase which
   performs random swaps or mass transfers (up to 20 % of the source
   entry).  Moves are kept only if they strictly improve the true
   objective.  The anti‑symmetric re‑ordering (pairing largest with
   smallest values) is applied regularly because it never worsens the
   convolution peak.

4. **Time management** – the whole procedure respects the 1000 s
   budget.  A safety margin is kept and the number of iterations is
   dynamically curtailed when the deadline approaches.

The combination of a richer set of initial shapes, a hard‑max
sub‑gradient stage and an aggressive local search typically drives the
objective below the previous best (≈ 1.510) and gets close to the target
≈ 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    The result is forced to be non‑negative to avoid tiny negative
    round‑off errors.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·b_k w.r.t. a,
    where b = a*a and p is a non‑negative weight vector with Σp = 1.
    The formula is:
        g_i = Σ_k p_k·a_{k‑i}
    which is a convolution of a with the reversed weight vector.
    One FFT is sufficient.
    """
    n = a.shape[0]
    # reverse p for the convolution formula
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # relevant slice corresponds to indices n‑1 … 2n‑2 (inclusive)
    seg = conv[n - 1 : 2 * n - 1]
    # reverse back to obtain the gradient
    g = seg[::-1]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0 , Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]      # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest entries opposite the smallest ones (pairwise).
    This never increases the maximum convolution value.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b

# Shape generation for a variety of analytical seeds
# ----------------------------------------------------------------------
def _generate_shape(n: int, kind: str, param: float = 1.0) -> np.ndarray:
    """
    Produce a non‑negative vector of length n according to ``kind``:
    * 'uniform'   – constant 1
    * 'sin'       – sin(π·x)   (x∈[0,1])
    * 'sin_pow'   – sin(π·x) raised to ``param`` (param>0)
    * 'beta'      – (x·(1‑x))**param  (symmetric beta with shape param+1)
    * 'triangular'– linear ramp up/down (peak 1)
    The result is normalised to sum = 1.
    """
    x = np.linspace(0.0, 1.0, n, endpoint=False) + 0.5 / n  # centre of bins
    if kind == "uniform":
        a = np.ones_like(x)
    elif kind == "sin":
        a = np.sin(np.pi * x)
    elif kind == "sin_pow":
        a = np.sin(np.pi * x) ** max(param, 1e-12)
    elif kind == "beta":
        a = (x * (1.0 - x)) ** max(param, 1e-12)
    elif kind == "triangular":
        a = 1.0 - np.abs(2.0 * x - 1.0)
    else:
        raise ValueError(f"Unknown shape kind: {kind}")
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / s
    return a

# Two‑stage descent (smoothed → hard max)
# ----------------------------------------------------------------------
def _descent(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a long descent from ``a_start``.
    Returns the best vector seen and its true objective value.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent] start objective = {best_val:.6f}")

    # schedule of smoothing parameters; the final 0 triggers a hard‑max phase
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.0]

    # total number of inner steps – we try to stay close to 6000
    total_steps = 6000
    steps_per_eps = max(1, total_steps // len(eps_schedule))

    eta0 = 0.5          # initial step size
    decay = 0.9995      # gentle decay

    step_counter = 0
    eta = eta0

    for eps in eps_schedule:
        for _ in range(steps_per_eps):
            if time.time() >= deadline - 6.0:
                break

            # ----- convolution -----
            conv = _full_convolution(a)          # length 2n‑1
            max_conv = conv.max()

            # ----- weight vector p -----
            if eps > 0.0:                         # soft‑max
                shift = max_conv
                exp_vals = np.exp(eps * (conv - shift))
                p = exp_vals / exp_vals.sum()
            else:                                 # hard max (sub‑gradient)
                mask = conv >= max_conv - 1e-12
                cnt = mask.sum()
                if cnt == 0:
                    p = np.full_like(conv, 1.0 / conv.shape[0])
                else:
                    p = mask.astype(float) / cnt

            # ----- gradient -----
            grad = 2.0 * _gradient_weighted_peaks(a, p)

            # ----- multiplicative mirror‑descent step -----
            delta = -eta * grad
            # protect against overflow / underflow in the exponent
            delta = np.clip(delta, -20.0, 0.0)
            a = a * np.exp(delta)
            a = _project_simplex(a)

            # ----- occasional true evaluation -----
            if step_counter % 30 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[descent] eps {eps:.3f} step {step_counter} new best = {best_val:.6f}")

                # every few hundred steps also try the anti‑symmetric reorder
                if step_counter % 300 == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    val_ord = evaluate_sequence(a_ord.tolist())
                    if val_ord < best_val - 1e-12:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[reorder] eps {eps:.3f} step {step_counter} improved to {best_val:.6f}")

            # decay step size and advance counter
            eta *= decay
            step_counter += 1

        # safety break if we are out of time
        if time.time() >= deadline - 6.0:
            break

        # final re‑ordering for the current epsilon (cannot worsen the max)
        a = _anti_symmetric_reorder(a)

    return best_a, best_val

# Hill‑climbing refinement (simple stochastic local search)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    Randomly swap or transfer mass between entries while keeping the sum fixed.
    Only strictly improving moves are kept.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    eval_interval = 20
    step = 0

    while time.time() < deadline - 4.0:
        # choose move type
        if rng.random() < 0.3:               # swap two positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                 # transfer a fraction of mass
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            # transfer up to 20 % of source mass, scaled by a random factor
            delta = best_a[i] * (0.05 + 0.15 * rng.random())
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex (protect against tiny negatives)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        # evaluate
        val = evaluate_sequence(new_a.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] step {step} new best = {best_val:.6f}")

        # occasional reorder (safe move)
        if step % 250 == 0:
            a_ord = _anti_symmetric_reorder(best_a)
            ord_val = evaluate_sequence(a_ord.tolist())
            if ord_val < best_val - 1e-12:
                best_val = ord_val
                best_a = a_ord
                print(f"[hill‑reorder] step {step} improved to {best_val:.6f}")

        step += 1

    return best_a, best_val
