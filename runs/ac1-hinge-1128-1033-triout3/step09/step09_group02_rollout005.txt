"""
search_step_function.py

A fast, multi‑restart optimizer for the convolution‑max objective.

Strategy
--------
1. Work with a modest size (n = 1024, a power of two) – full FFT convolutions
   are cheap, allowing many iterations within the 1000 s budget.
2. Exhaustively scan several parametric families (exponential, power law,
   beta‑distribution, sine‑power, piecewise‑constant, logistic).  The scan
   uses a very fine grid and also draws random Dirichlet vectors with tiny
   concentration parameters, guaranteeing a strong initial candidate.
3. Repeatedly perform cheap hill‑climbing moves:
      * Move a small fraction of mass from an index that heavily contributes
        to the current peak of the convolution to an index that contributes
        little.
      * The convolution is updated in O(n) using the exact incremental formula
        (no FFT per move).
      * Each accepted move is kept; if no improvement occurs for a long
        stretch we perturb the current best and restart.
4. After hill‑climbing we run a projected sub‑gradient descent on the exact
   objective (the sub‑gradient is known analytically at the active shift).
5. Finally we mix a tiny amount of the uniform distribution and keep the
   best among a few mixing ratios.
6. The algorithm respects the 1000 s time limit, prints progress and
   returns the best non‑negative list found (the list length is 1024,
   but any length would be accepted by the evaluation harness).

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, size a power of two)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (full length 2·n‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective handling (Σa = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a probability vector (Σ a = 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Hill‑climbing with exact O(n) convolution updates
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               target_val: float,
               start: float,
               deadline: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """Greedy mass transfer that strictly improves the exact objective."""
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        # compute the active shift
        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – weighted by contribution (more contribution ⇒ larger chance)
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # target index – bias towards indices that *avoid* the current peak
        if np.random.rand() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        # fraction of mass to move – decays linearly with elapsed time
        progress = (time.time() - start) / (deadline - start)
        max_frac = 0.3 * (1.0 - progress) + 0.001 * progress   # from 0.3 → 0.001
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of the convolution (exact, O(n))
        b_new = b.copy()
        # contributions from moving delta from i to j (see original optimizer)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        idx_2j = 2 * j
        idx_2i = 2 * i
        if idx_2j < b_new.size:
            b_new[idx_2j] += delta * delta
        if idx_2i < b_new.size:
            b_new[idx_2i] += delta * delta
        idx_ij = i + j
        if idx_ij < b_new.size:
            b_new[idx_ij] -= delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-14:                # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # restart if stuck for a long while
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbation after stagnation")

        it += 1

    return best_a, best_val

# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a_start: np.ndarray,
                         target_val: float,
                         start: float,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient method with simplex projection."""
    a = a_start.copy()
    n = a.shape[0]
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n
    eta = eta0
    decay = 0.9996
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 180_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient at the active shift
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # shake & restart
            noise = np.random.normal(scale=0.009, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbation after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: convex mixture with a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Try a few convex combinations with the uniform distribution."""
    best_a, best_val = a, _objective_value(a)
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    for alpha in np.arange(0.0, 0.022, 0.002):   # 0, 0.002, …, 0.02
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
