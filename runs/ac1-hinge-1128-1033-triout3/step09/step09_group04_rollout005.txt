import time
import numpy as np

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _objective_full(a):
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max() / (a.sum()**2))

def _project_simplex(v):
    """Project v onto the probability simplex (nonnegative, sum=1)."""
    # standard algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _pgd_optimize(a_init, deadline, best_val, best_a, top_k=1):
    a = a_init.copy()
    n = a.shape[0]
    iter = 0
    no_improve = 0
    IMPATIENT = 5000  # steps without improve before jitter
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute convolution
        b = _full_convolution(a)
        maxb = b.max()
        # objective
        S = a.sum()
        cur_val = 2.0 * n * maxb / (S * S)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[PGD {iter:6d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # identify active index
        if top_k == 1:
            ks = [int(np.argmax(b))]
        else:
            ks = np.argpartition(b, -top_k)[-top_k:]

        # compute subgradient of max with respect to a: combine contributions from each active k equally?
        grad = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += a[k - idx]  # derivative of b_k w.r.t a_i

        # If multiple ks, average
        grad = grad / len(ks)

        S = a.sum()
        # gradient of full objective f = 2n * maxb / S^2:
        # ∂f/∂a = (2n / S^2) * grad - (4n * maxb / S^3) * 1
        grad_f = (2.0 * n / (S * S)) * grad - (4.0 * n * maxb / (S**3)) * np.ones_like(a)

        # step size
        progress = iter / 100000.0  # arbitrary scaling

        # could also limit step size

        # project onto simplex

        # jitter if stuck

            a = _project_simplex(a)

def _project_simplex(v):
    # based on algorithm from https://arxiv.org/pdf/1101.6081.pdf
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step function that minimises the objective
``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** – start from the previously found sequence
   (global variable ``height_sequence_1``) if it exists, otherwise fall back
   to a uniform vector of length ``n = 8000``.

2. **Projected gradient descent (PGD)** – the true objective is

       f(a) = 2·n·max_k (a*a)_k / (∑a)^2 .

   For the currently maximal convolution index `k*` the sub‑gradient is
   `∂(a*a)_k*/∂a_i = a_{k*‑i}` (when the index exists).  Taking the derivative
   of `f` gives a simple explicit gradient.  Each iteration updates

       a ← a – η·grad
       a ← clip(a,0,∞) and renormalise to Σa = 1

   This keeps the iterate on the probability simplex without an expensive
   projection step.

   • A decreasing step‑size `η = η0 / sqrt(t+1)` (η₀≈0.4) is used.
   • When progress stalls a small Gaussian jitter is added.
   • To avoid being stuck at a single peak the gradient can be averaged over
     the `top_k` largest convolution entries (default 3).

   The routine runs for a *time slice* (seconds) and returns the best vector
   found in that slice.

3. **Random restarts** – the total budget (≈ 1000 s) is divided into an
   initial long PGD phase (≈ 30 % of the time) and many cheap restarts.
   Each restart draws a random Dirichlet initial vector with a
   log‑uniform concentration parameter (α∈[10⁻²,10²]), runs a short PGD,
   optionally mixes a tiny amount of the uniform vector, and updates the
   global best if an improvement is obtained.

4. **Final mixing** – a few convex combinations with the uniform vector are
   tried; this can sometimes lower the peak a little.

The method works directly on the full 8000‑dimensional problem, uses only
`numpy` and `time`, and respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (no normalisation, matches evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective value returned by evaluate_sequence for a given vector."""
    n = a.shape[0]
    maxb = _full_convolution(a).max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * maxb / (s * s))


# ----------------------------------------------------------------------
# Tiny uniform mixing (as in the reference code)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combos (1‑α)·a + α·uniform, keep the best."""
    best_a, best_val = a, _objective(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected Gradient Descent on the simplex (additive updates)
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  deadline: float,
                  top_k: int = 3,
                  eta0: float = 0.4,
                  impatient: int = 15000) -> tuple[np.ndarray, float]:
    """
    Run projected gradient descent on the simplex for the remaining time.
    Returns the best vector discovered in the allotted slice.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    stagnation = 0

    while time.time() < deadline:
        # full convolution and maximal entry
        conv = _full_convolution(a)
        maxb = conv.max()

        # indices of the K largest convolution entries
        if top_k == 1:
            ks = [int(np.argmax(conv))]
        else:
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # sub‑gradient of the maximal convolution entry
        g = np.zeros_like(a)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += a[k - idx]                     # ∂(a*a)_k / ∂a_i
        g = g / len(ks)                               # average if several ks

        # full objective gradient (including denominator)
        S = a.sum()
        grad = (2.0 * n / (S * S)) * g - (4.0 * n * maxb / (S ** 3)) * np.ones_like(a)

        # step size schedule
        eta = eta0 / np.sqrt(it + 1)

        # additive update, clip to non‑negative and renormalise (simplex projection)
        a = a - eta * grad
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:                     # safety fallback
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # evaluate current objective
        cur_val = _objective(a)
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, a.copy()
            stagnation = 0
        else:
            stagnation += 1

        # occasional jitter to escape flat regions
        if stagnation > impatient:
            noise = np.random.normal(scale=0.006, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()
            stagnation = 0
            print(f"[PGD {it:6d}] jitter after stagnation")

        it += 1

    return best_a, best_val
