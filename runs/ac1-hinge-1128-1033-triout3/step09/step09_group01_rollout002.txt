"""
Step‑function optimizer.

We start from several seeds (previous best, Beta‑family, Dirichlet draws) and
run a projected sub‑gradient descent on the maximal self‑convolution entry.
After a time‑budget split we switch to a more aggressive hill‑climbing
procedure (mass‑transfer simulated‑annealing) to polish the solution.
Both stages keep the sequence on the probability simplex (non‑negative,
sum = 1) so that the external ``evaluate_sequence`` simply returns
``2·n·max(conv)``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assuming Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Beta‑family sweep (cheap parametric search)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient of the active (max) convolution entry
# ----------------------------------------------------------------------
def _grad_max_entry(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the scalar max entry b_k = (a*a)_k with respect to a.
    The formula is g_j = a_{k-j} (if defined) + a_{j-k} (if defined).
    """
    n = a.shape[0]
    g = np.zeros(n, dtype=a.dtype)

    # first term: a_{k-j} for j where 0 <= k-j < n
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = a[k - idx]

    # second term: a_{j-k} for j where 0 <= j-k < n  (only if k <= n-1)
    if k <= n - 1:
        # j runs from k to n-1
        length = n - k
        g[k:] += a[:length]

    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_val: float,
                         target_val: float = 1.503,
                         max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent.
    Keeps the iterate on the simplex (Σa = 1, a ≥ 0) and returns the best
    point seen before the deadline.
    """
    a = a0.copy()
    cur_val = _objective(a)
    best_a = a.copy()
    best = cur_val
    iteration = 0

    # initial step size – tuned experimentally
    eta0 = 0.4

    while True:
        now = time.time()
        if now >= deadline or iteration >= max_iter:
            break

        # compute full convolution and locate the active index
        conv = _full_convolution(a)
        k = int(np.argmax(conv))

        # gradient of the active entry
        grad = _grad_max_entry(a, k)

        # diminishing step size
        step = eta0 / math.sqrt(iteration + 1.0)

        # gradient step (we minimize, so move opposite to gradient)
        a = a - step * grad

        # keep the point on the probability simplex
        a = _project_simplex(a)

        cur_val = _objective(a)

        if cur_val < best - 1e-12:
            best = cur_val
            best_a = a.copy()
            print(f"[SGD {iteration:7d}] new best = {best:.6f}")
            if best <= target_val:
                break

        iteration += 1

    return best_a, best


# ----------------------------------------------------------------------
# Intensive hill‑climbing (mass‑transfer simulated annealing)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Based on the earlier mass‑transfer annealing, but kept as a separate
    function so we can call it after the sub‑gradient phase.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule – deliberately stretched so the search lasts until the deadline
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        # contribution of each position to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – draw proportional to its contribution
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – biased towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

"""
Step‑function optimizer (sub‑gradient + mass‑transfer annealing).

The routine works in two phases:

1️⃣  **Projected sub‑gradient descent** – minimise the maximal entry of the
    self‑convolution (i.e. minimise 2·n·max(conv)) while keeping the
    sequence on the probability simplex (Σa = 1, aᵢ ≥ 0).  The gradient of
    the active entry is computed analytically, a diminishing step size is
    used and after each step the point is projected onto the simplex.

2️⃣  **Intensive hill‑climbing** – a simulated‑annealing style mass‑transfer
    algorithm (very similar to the previous baseline) tuned on the remaining
    time budget.  It reliably fine‑tunes the solution found by the first
    phase.

Both phases start from a collection of seeds (previous best solution,
a Beta‑family sweep and a few random Dirichlet draws).  The best point is
kept and finally passed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Full self‑convolution (FFT, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return ``x`` convolved with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (under Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x | x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Two‑parameter Beta‑family sweep (cheap parametric search)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a family a_i ∝ i^{p‑1} (n‑i+1)^{q‑1} and return the best."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient of the active (max) convolution entry
# ----------------------------------------------------------------------
def _grad_max_entry(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of b_k = (a*a)_k with respect to a.
    g_j = a_{k-j} (if defined) + a_{j-k} (if defined).
    """
    n = a.shape[0]
    g = np.zeros(n, dtype=a.dtype)

    # term a_{k-j}
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] = a[k - idx]

    # term a_{j-k} (only when k ≤ n‑1)
    if k <= n - 1:
        length = n - k
        g[k:] += a[:length]

    return g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         start_time: float,
                         best_val: float,
                         target_val: float = 1.503,
                         max_iter: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent with Euclidean simplex projection.
    Returns the best vector found before the deadline.
    """
    a = a0.copy()
    cur_val = _objective(a)
    best_a = a.copy()
    best = cur_val
    iteration = 0
    eta0 = 0.4                       # initial step‑size, tuned empirically

    while True:
        now = time.time()
        if now >= deadline or iteration >= max_iter:
            break

        conv = _full_convolution(a)
        k = int(np.argmax(conv))

        grad = _grad_max_entry(a, k)

        step = eta0 / math.sqrt(iteration + 1.0)
        a = a - step * grad
        a = _project_simplex(a)

        cur_val = _objective(a)

        if cur_val < best - 1e-12:
            best = cur_val
            best_a = a.copy()
            print(f"[SGD {iteration:7d}] new best = {best:.6f}")
            if best <= target_val:
                break

        iteration += 1

    return best_a, best


# ----------------------------------------------------------------------
# Mass‑transfer simulated‑annealing hill‑climbing (identical to baseline)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Fine‑tune a sequence by moving small mass between indices."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) that aims to minimise
    ``evaluate_sequence``.  The routine respects the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # Problem size – a few‑ten‑thousands give a good trade‑off.
    n = 12000

    # --------------------------------------------------------------
    # 0️⃣  Build an initial pool of candidate sequences
    # --------------------------------------------------------------
    candidates = []

    # previous best (warm‑start)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        candidates.append(a0)
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # Beta‑family sweep (cheap parametric family)
    try:
        beta_a, beta_val = _beta_family_best(n)
        candidates.append(beta_a)
        print(f"[init] beta‑family candidate value={beta_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # a few random Dirichlet seeds for diversity
    for _ in range(4):
        rand_a = np.random.dirichlet(np.ones(n))
        candidates.append(rand_a)

    # select the best among the initial pool
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[init] candidate value={val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Projected sub‑gradient descent (≈60 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.60 * (deadline - start_time)
    best_a, best_val = _subgradient_descent(best_a,
                                            deadline=phase1_deadline,
                                            start_time=start_time,
                                            best_val=best_val,
                                            target_val=1.503)

    # --------------------------------------------------------------
    # 2️⃣  Intensive hill‑climbing for the remaining time
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb_anneal(best_a,
                                         n=n,
                                         deadline=deadline,
                                         start_time=start_time,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # --------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # expose the solution for a possible warm‑start in a later run
    global height_sequence_1
    height_sequence_1 = best_a.tolist()    # type: ignore[assignment]

    return best_a.tolist()
