            if step < min_step:

        # Optional break if no progress for many iterations
    return best_a, best_val

"""
Simple projected‑gradient + hill‑climbing optimizer for the step‑function
challenge.

The optimizer works in three phases:

1️⃣  **Broad initialization** – for a few problem sizes (4000, 8000,
    12000) we scan a two‑parameter Beta family
    (a_i ∝ i^{p‑1}·(n‑i+1)^{q‑1}) together with a couple of random
    Dirichlet draws.  The best sequence found so far becomes the warm‑start.

2️⃣  **Projected gradient descent** – the objective is
    f(a)=2·n·max_k (a∗a)_k with the simplex constraint Σa=1, a≥0.
    The sub‑gradient of the active convolution entry k* is
    g_j=2·a_{k*‑j} (zero outside the support).  After a gradient step the
    iterate is projected back onto the probability simplex (simple
    non‑negative scaling).  A basic back‑tracking line search adjusts the
    step size.  This phase quickly reduces the maximal autocorrelation.

3️⃣  **Intensive hill‑climbing** – the previous implementation
    (mass‑transfer annealing) refines the solution further.

All work respects the 1000 s time budget; the final list of floats is
returned as the candidate.  The code prints intermediate objective values
to aid debugging.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Beta‑family sweep (denser grid than the previous version)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    # grid spanning values below and above 1 (U‑shaped and bell‑shaped)
    p_vals = np.concatenate([
        np.linspace(0.1, 3.0, 30),
        np.geomspace(3.0, 12.0, 7)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex
# ----------------------------------------------------------------------
def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on  f(a)=2·n·max(a∗a)  with simplex constraint.
    Uses a simple back‑tracking line‑search and Euclidean projection.
    """
    a = a0.copy()
    size = 1 << ((2 * n - 2).bit_length())   # FFT size (next power of two)
    best_a, best_val = a.copy(), _objective(a)

    step = 0.02          # initial step size
    min_step = 1e-12
    iter_cnt = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current convolution and max index
        fx = np.fft.rfft(a, size)
        conv = np.fft.irfft(fx * fx, size)[:2 * n - 1]
        max_b = conv.max()
        cur_val = 2.0 * n * max_b
        kmax = int(conv.argmax())

        # sub‑gradient of the active term
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # try a descent step
        a_candidate = a - step * grad
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s == 0.0:
            # failed step – random restart
            a_candidate = np.random.dirichlet(np.ones(n))
            s = 1.0
        a_candidate /= s

        # evaluate new point
        fx_new = np.fft.rfft(a_candidate, size)
        conv_new = np.fft.irfft(fx_new * fx_new, size)[:2 * n - 1]
        new_val = 2.0 * n * conv_new.max()

        # acceptance test (pure descent, no temperature)
        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[PGD {iter_cnt:7d}] new best = {best_val:.6f}")
                # early exit if we already beat the target
                if best_val <= 1.503:
                    break
            # slowly increase step size after success
            step = min(step * 1.07, 0.5)
        else:
            # reject – shrink step size
            step *= 0.5
            no_improve += 1
            if step < min_step:
                break

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            step = 0.02
            no_improve = 0
            print(f"[PGD {iter_cnt:7d}] stagnation – random perturb")

        iter_cnt += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Intensive simulated‑annealing hill‑climbing (unchanged core)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from indices that
    dominate the current maximal convolution entry to indices that
    contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

        # ------------------------------------------------------------------
        a0 = None
        try:
            old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print(f"[init] interpolated previous solution to n={n}")
        except Exception:
            pass

        # ------------------------------------------------------------------
        # 1️⃣  Cheap parametric sweep (Beta family) + a few Dirichlet seeds
        # ------------------------------------------------------------------
        best_a, best_val = None, np.inf

        try:
            beta_a, beta_val = _beta_family_best(n)
            if beta_val < best_val:
                best_a, best_val = beta_a, beta_val
                print(f"[init] beta‑family (n={n}) value={best_val:.6f}")
        except Exception as exc:
            print(f"[init] beta‑family failed for n={n}:", exc)

        # Add a couple of random Dirichlet seeds for diversity
        for _ in range(2):
            rand_a = np.random.dirichlet(np.ones(n))
            val = _objective(rand_a)
            if val < best_val:
                best_a, best_val = rand_a, val
                print(f"[init] Dirichlet seed improved to {best_val:.6f}")

        # Use the interpolated solution if it exists and beats the beta one
        if a0 is not None:
            val0 = _objective(a0)
            if val0 < best_val:
                best_a, best_val = a0, val0
                print(f"[init] interpolated start beats others, value={best_val:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Projected‑gradient refinement (time‑boxed)
        # ------------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # we give each size at most 30 s (or the remaining budget if smaller)
        pgd_deadline = time.time() + min(30.0, remaining * 0.3)
        a_refined, val_refined = _pgd_optimize(best_a, n, pgd_deadline, start_time)
        print(f"[PGD] n={n} best after refinement = {val_refined:.6f}")

        if val_refined < global_best_val:
            global_best_val, global_best_a, global_best_n = val_refined, a_refined.copy(), n

        # If we already beat the target we can stop early
        if global_best_val <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Final intensive hill‑climbing (uses the remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5 and global_best_a is not None:
        global_best_a, global_best_val = _hillclimb_anneal(
            global_best_a,
            global_best_n,
            deadline=deadline,
            start_time=start_time,
            best_a=global_best_a,
            best_val=global_best_val,
            target_val=1.503,
        )
        print(f"[final hillclimb] best value = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()
