assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (the step heights) that minimises  

\[
F(a)=\frac{2 n\; \max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the linear convolution of `a` with itself.
Because the objective is scale‑invariant we can keep `∑a = 1`.  
The problem is smooth except for the `max` operation.  We therefore

1. **Smooth the max** with a log‑sum‑exp (`softmax`) surrogate.
2. **Mirror‑descent** on the simplex (multiplicative updates + Euclidean projection).
3. **Gradually sharpen** the surrogate (decrease ε) and run many
   iterations (≈ 2 × 10⁴ FFT steps – still well below the 1000 s budget).
4. **Final refinement**: use the true max (ε = 0) as a sub‑gradient for a
   short “max‑descent” phase.
5. **Local hill‑climbing**: random swaps / tiny mass transfers.
6. **Anti‑symmetric reordering** after each ε‑stage (pair the largest with the
   smallest entry) – this never worsens the maximal convolution value.

All heavy work (convolution, gradient) is performed with the FFT, so each
iteration is cheap.  The algorithm is fully deterministic aside from the
random seeds, and it prints progress so you can see how the bound evolves.

```python
"""
propose_candidate – minimise `evaluate_sequence` by smoothed‑max mirror descent.

Algorithm
---------
* Work on the probability simplex (∑a = 1, a_i ≥ 0).
* For a sequence a, the convolution b = a*a is obtained with an FFT.
* Replace max(b) by a log‑sum‑exp surrogate:
        f_ε(a) = (1/ε)·log Σ_k exp(ε·b_k)
  Its gradient is a single FFT‑based convolution.
* Multiplicative mirror‑descent step:
        a ← a · exp(−η·∇f_ε)   ,   a ← proj_simplex(a)
* Decrease ε from 0.8 down to 0.001, performing many inner iterations.
* After the smoothing stages perform a few “max‑subgradient” steps
  (ε = 0) to tighten the true objective.
* Interleave anti‑symmetric reordering (largest ↔ smallest) – provably
  never increases max(b).
* Finish with a cheap hill‑climbing local search (swaps and tiny mass
  transfers).

The routine respects the 1000 s budget, runs several restarts, and returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas – top‑level as required)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padded FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)      # eliminate tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·(a*a)[k] w.r.t. a .
    p must be a (2n‑1) vector with Σp = 1, p ≥ 0.
    Returns g_i = Σ_k p_k·a_{k-i}.
    Computed with a single FFT.
    """
    n = a.shape[0]
    # reverse p for convolution with a (see derivation in the analysis)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # relevant slice corresponds to indices n‑1 … 2n‑2
    seg = conv[n - 1 : 2 * n - 1]
    # reverse back to obtain the gradient
    g = seg[::-1]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { x ≥ 0, Σx = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]            # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest ones.
    This reordering never increases max(a*a).
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) mirror descent
# ----------------------------------------------------------------------
def _smoothed_descent(a_start: np.ndarray,
                     deadline: float,
                     eps_schedule,
                     it_per_eps: int,
                     eta_start: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Runs the mirror‑descent loop for the whole ε‑schedule.
    Returns the best vector encountered and its true evaluation.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent init] objective = {best_val:.6f}")

    eta = eta_start

    for eps in eps_schedule:
        # inner loop for the current smoothing parameter
        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:
                break

            # convolution and soft‑max weights
            conv = _full_convolution(a)          # length 2n‑1
            max_conv = conv.max()
            shift = max_conv
            # stable log‑sum‑exp
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()        # Σp = 1, p ≥ 0

            # gradient of the smooth surrogate (factor 2 because ∂(a*a)/∂a = 2·shifted a)
            grad = 2.0 * _gradient_weighted_peaks(a, p)

            # multiplicative mirror‑descent step (clipped to avoid overflow)
            delta = -eta * grad
            delta = np.clip(delta, -20.0, 0.0)    # exp(‑20) ≈ 2·10⁻⁹
            a = a * np.exp(delta)
            a = _project_simplex(a)

            # occasional true‑objective check
            if it % 30 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.3f} it {it}] new best = {best_val:.6f} (max conv {max_conv:.6e})")

                # anti‑symmetric reordering every 200 steps
                if it % 200 == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    val_ord = evaluate_sequence(a_ord.tolist())
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[reorder eps {eps:.3f} it {it}] improved to {best_val:.6f}")

            # slowly decay step size (helps stability for very small ε)
            eta *= 0.9995

        # final re‑ordering for this ε‑stage (cannot increase the max)
        a = _anti_symmetric_reorder(a)

        # adapt number of inner iterations if we are running out of time
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Max‑subgradient refinement (ε = 0)
# ----------------------------------------------------------------------
def _max_subgradient_refine(a_start: np.ndarray,
                            deadline: float,
                            max_iters: int = 2000,
                            eta: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Performs a few steps using the true max as the objective
    (sub‑gradient of max(b)).
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[max‑refine init] objective = {best_val:.6f}")

    for it in range(max_iters):
        if time.time() >= deadline - 5.0:
            break

        conv = _full_convolution(a)
        k = int(np.argmax(conv))

        # one‑hot weight vector at the maximal index
        p = np.zeros_like(conv)
        p[k] = 1.0
        grad = 2.0 * _gradient_weighted_peaks(a, p)

        delta = -eta * grad
        delta = np.clip(delta, -20.0, 0.0)
        a = a * np.exp(delta)
        a = _project_simplex(a)

        # occasional evaluation
        if it % 50 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[max‑refine it {it}] new best = {best_val:.6f}")

        # decay step size gently
        eta *= 0.9997

    return best_a, best_val


# ----------------------------------------------------------------------
# Light hill‑climbing local search
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray,
                       cur_val: float,
                       deadline: float,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Random swaps and tiny mass transfers, kept only if they improve the
    true objective.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        if rng.random() < 0.3:                     # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # transfer a small amount
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.03 * rng.random()   # up to 3 % of the source mass
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto the simplex (numerical safety)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine respects the given time budget (default 1000 s) and returns
    a list of floats (the best sequence found).
    """
    # ------------------------------------------------------------------
    # Basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a 5‑second safety margin for the final evaluation/return
    deadline = start_time + budget_s - 5.0

    # Problem size – 8000 gives a good trade‑off between expressiveness
    # and speed with the FFT based code.
    n = 8000

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if present
    # ------------------------------------------------------------------

        if old.shape[0] != n:
            # linear interpolation to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        a0 = None

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Restart loop – a handful of different seeds / initialisations
    # ------------------------------------------------------------------
    init_strategies = ["uniform", "sin", "random"]
    restart = 0
    max_restarts = 5
    while time.time() < deadline - 180 and restart < max_restarts:
        restart += 1

        # ----------------------------------------------------------------
        # Initialise a vector
        # ----------------------------------------------------------------
        if a0 is not None and restart == 1:
            a = a0.copy()
            init_name = "from_previous"
        else:
            strat = rng.choice(init_strategies)
            if strat == "uniform":
                a = np.full(n, 1.0 / n)
                init_name = "uniform"
            elif strat == "sin":
                idx = np.arange(n) + 0.5
                a = np.sin(np.pi * idx / n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                init_name = "sinusoid"
            else:  # random exponential
                a = rng.exponential(scale=1.0, size=n)
                a /= a.sum()
                init_name = "random"

        # ----------------------------------------------------------------
        # Smoothed‑max mirror descent
        # ----------------------------------------------------------------
        eps_schedule = [0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
                        0.005, 0.002, 0.001]
        # Allocate about 2500 iterations per ε‑stage – total ≈ 28 k steps
        it_per_eps = max(200, int(2500 // len(eps_schedule)))
        a_opt, val_opt = _smoothed_descent(a, deadline, eps_schedule, it_per_eps)

        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[global] new best {best_val:.6f} (restart {restart}, init={init_name})")

        # ----------------------------------------------------------------
        # Max‑subgradient refinement (focuses on the true max)
        # ----------------------------------------------------------------
        if time.time() < deadline - 120:
            a_ref, val_ref = _max_subgradient_refine(best_a, deadline,
                                                     max_iters=2000,
                                                     eta=0.25)
            if val_ref < best_val:
                best_val = val_ref
                best_a = a_ref.copy()
                print(f"[global] after max‑refine best = {best_val:.6f}")

        # ----------------------------------------------------------------
        # Light hill‑climb after each restart (if time permits)
        # ----------------------------------------------------------------
        if time.time() < deadline - 30:
            a_hill, val_hill = _hill_climb_refine(best_a, best_val,
                                                  deadline, n)
            if val_hill < best_val:
                best_val = val_hill
                best_a = a_hill.copy()
                print(f"[global] post‑hill improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final intensive hill‑climb on the current best vector
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5:
        best_a, best_val = _hill_climb_refine(best_a, best_val,
                                              deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
```
