def solve_lp(f, M, max_val=1000.0):
    n = f.shape[0]
    # Build matrix A: shape (2n-1, n)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i+n, i] = f
    # define variable
    g = cp.Variable(n)
    constraints = [g >= 0, g <= max_val, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # Set solver parameters for speed
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-6, reltol=1e-6)
    if g.value is None:
        # fallback: zero vector
        return np.zeros(n)
    return np.array(g.value).astype(float)

"""
Propose a step‑function (list of non‑negative floats) that minimises
`evaluate_sequence`.  

The search consists of three stages:

1. **LP‑boost (n≈2000)** – Starting from a random (or historic) vector `f`
   we repeatedly solve a linear program that maximises the ℓ¹‑norm of a
   partner vector `g` under the constraint `f*g ≤ max(f*f)`.  Mixing `g`
   into `f` increases the total sum without raising the current peak of the
   self‑convolution.

2. **Smooth mirror‑descent (n≈8000)** – The boosted vector is up‑sampled to
   the full length.  A smooth approximation of the max (log‑sum‑exp) is
   used as the objective; its gradient is obtained by a single FFT‑based
   convolution.  Multiplicative mirror‑descent updates keep the vector on
   the simplex while driving the peak down.

3. **Stochastic hill‑climbing** – A cheap local search (random swaps &
   tiny mass transfers) finishes the optimisation.

All heavy work (FFT, LP) is performed without calling `evaluate_sequence`.
Only occasional calls are made for logging and the final verification.
The routine respects the 1000 s time budget.
"""

import time
import numpy as np
import cvxpy as cp

MAX_VAL = 1000.0          # upper bound enforced by evaluate_sequence
SMALL_N = 2000            # size for the LP phase (fast LPs)
BIG_N = 8000              # final size (used by the original run)

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear self‑convolution using a real FFT (returns length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate spurious tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a*a)_i w.r.t. a (weights are given reversed).
    Implements 2·(a * w_rev) where * is linear convolution.
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1                 # index of first relevant entry
    seg = conv[start:start + n]                # length n
    return 2.0 * seg


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    s = np.sort(a)                     # ascending
    n = a.shape[0]
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left
        b[-1 - i] = s[i]               # small → right
    if n % 2:
        b[half] = s[half]
    return b


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = f * g (linear convolution).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float, max_val: float = MAX_VAL) -> np.ndarray:
    """
    Linear programme:
        maximise   Σ_j g_j
        subject to g_j ≥ 0, g_j ≤ max_val,
                   (f * g)_k ≤ M  for all k.
    Returns the optimal g (as a NumPy array).  If the solver fails, returns zeros.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                # (2n‑1) × n
    g = cp.Variable(n)

    constraints = [
        g >= 0,
        g <= max_val,
        A @ g <= M
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-6, reltol=1e-6, feastol=1e-6, verbose=False)
    except Exception:                         # fallback to no‑solution
        return np.zeros(n)

    if g.value is None:
        return np.zeros(n)
    return np.asarray(g.value).reshape(-1)


def _compute_objective(a: np.ndarray) -> float:
    """
    Fast surrogate for `evaluate_sequence`:
        2·n·max(a*a) / (sum(a)²)   with clipping to [0,MAX_VAL].
    Returns np.inf if the sum is too small.
    """
    a = np.clip(a, 0.0, MAX_VAL)
    total = a.sum()
    if total < 0.01:
        return np.inf
    conv = _full_convolution(a)
    max_b = conv.max()
    n = a.shape[0]
    return 2.0 * n * max_b / (total * total)


def _upscale(a: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(a) to target_n points."""
    old_x = np.linspace(0.0, 1.0, a.shape[0])
    new_x = np.linspace(0.0, 1.0, target_n)
    return np.interp(new_x, old_x, a)
