"""
A three‑stage optimizer for the “max‑convolution” bound.

1️⃣  Coarse global search – a low‑dimensional piecewise‑linear shape is
    optimised with differential evolution on a small problem (n≈2000).
    The best shape is up‑sampled to the full size (≈9000) and normalised.

2️⃣  Mid‑range smoothing – Adam optimisation on a log‑sum‑exp surrogate
    of the max‑convolution (identical to the previous “Phase A” but with
    a longer schedule and a slightly higher learning‑rate).

3️⃣  Exact polishing – a combination of hill‑climbing (exact convolution
    updates) and projected sub‑gradient descent, followed by a tiny mix
    with the uniform distribution.

The routine returns a non‑negative list whose sum is arbitrary
(the evaluator rescales it internally).  The whole pipeline is timed
so that it stops just before the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy.optimize import differential_evolution

# -------------------------------------------------------------
# Convolution helpers (FFT based)
# -------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# -------------------------------------------------------------
# Exact objective (the evaluator uses the same formula)
# -------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# -------------------------------------------------------------
# Projection onto the probability simplex
# -------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Low‑dimensional parametric family (piecewise‑linear)
# -------------------------------------------------------------
def _piecewise_linear_shape(heights: np.ndarray, length: int) -> np.ndarray:
    """
    Build a non‑negative vector of given length by linear interpolation
    of the supplied anchor heights.
    """
    k = len(heights)
    anchors_x = np.linspace(0, length - 1, k)
    xs = np.arange(length)
    a = np.interp(xs, anchors_x, heights)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(length, 1.0 / length)
    else:
        a /= s
    return a


# -------------------------------------------------------------
# Global search on a small problem using DE
# -------------------------------------------------------------
def _de_initial_candidate(small_n: int, n_large: int, seed: int = 42):
    """
    Run a cheap differential‑evolution search on a piecewise‑linear
    family (k anchors) for a small problem size `small_n`.
    The best shape is then interpolated to the full size `n_large`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    k = 15                     # number of anchors – a good trade‑off
    bounds = [(0.0, 1.0)] * k  # heights are non‑negative (scale irrelevant)

    # -----------------------------------------------------------------
    # Objective for DE – works on the *small* problem size.
    # -----------------------------------------------------------------
    def de_obj(h):
        a = _piecewise_linear_shape(np.array(h, dtype=float), small_n)
        return _objective_value(a)

    # Run DE – we keep the iteration count modest so that it finishes fast.
    result = differential_evolution(
        de_obj,
        bounds,
        strategy="best1bin",
        maxiter=150,
        popsize=12,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=False,
        init='latinhypercube',
        atol=1e-12,
        tol=1e-12,
        disp=False,
    )

    best_heights = np.array(result.x)
    # Upsample to the full size
    a_large = _piecewise_linear_shape(best_heights, n_large)
    best_val = _objective_value(a_large)
    return a_large, best_val


# -------------------------------------------------------------
# Phase A – Adam on a smooth log‑sum‑exp surrogate
# -------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smoothed max‑convolution."""
    # logits → simplex via soft‑max
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.004          # a bit more aggressive than before
    eps_start, eps_end = 0.8, 1e-7

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80_000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient w.r.t. a of 2·n·∑ p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)               # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.009, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.25
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# -------------------------------------------------------------
# Phase B – Targeted hill‑climbing (exact objective)
# -------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy transfer of a small amount of mass from high‑contrib. indices."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the *current* maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source proportional to its contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # choose a target that *does not* appear in the active pair
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        # weight inversely to rev_contrib (prefer indices that hardly affect the max)
        w_target = 1.0 / (rev_contrib + 1e-12)
        w_target[i] = 0.0
        if w_target.sum() > 0.0:
            probs_j = w_target / w_target.sum()
            j = np.random.choice(n, p=probs_j)
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # exact linear‑time update of the convolution (see original code)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb current best and restart hill‑climbing
            a = best_a.copy()
            noise = np.random.normal(scale=0.018, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# -------------------------------------------------------------
# Phase C – Projected sub‑gradient descent (exact max)
# -------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.6 / n          # a slightly larger initial step
    eta = eta0
    decay = 0.9997
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient at the active index
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # step‑size decay
        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # random shake + reset step‑size
            noise = np.random.normal(scale=0.011, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# -------------------------------------------------------------
# Tiny uniform mixing – final polish
# -------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a tiny share of the uniform distribution."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the allotted time.
    Returns a list of non‑negative floats (any total sum is fine)
    that minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # 1️⃣  Global low‑dimensional search (DE) on a tiny problem.
    # ------------------------------------------------------------------
    small_n = max(1500, n // 6)                     # ≈1500 points – cheap
    print("[Init] running DE on small problem")
    a_de, val_de = _de_initial_candidate(small_n, n, seed=seed)
    best_a = a_de
    best_val = val_de
    print(f"[Init] DE best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Add a handful of classic parametric candidates (exponential,
    #     beta, etc.) as a safety net – very cheap.
    # ------------------------------------------------------------------
    def _try_candidate(arr):
        nonlocal best_a, best_val
        arr = np.asarray(arr, dtype=float)
        if arr.ndim != 1 or arr.shape[0] != n:
            return
        if np.any(arr < 0):
            return
        s = arr.sum()
        if s <= 0:
            return
        arr = arr / s
        val = _objective_value(arr)
        if val < best_val:
            best_val, best_a = val, arr.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # a few cheap families
    for r in (0.9998, 0.9999):
        _try_candidate(r ** i)                         # decreasing expo
    for p in (0.8, 1.2, 2.0):
        _try_candidate(i1 ** (-p))                     # power‑law
    for a_param in (0.8, 1.2):
        for b_param in (0.8, 1.2):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try_candidate(a)

    print(f"[Init] after cheap sweep best = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # 3️⃣  Phase A – Adam on the smooth surrogate (≈45 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase A] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Phase B – Targeted hill‑climbing (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.80 * (deadline - start_time)
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    best_val = _objective_value(best_a)
    print(f"[Phase B] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Phase C – Projected sub‑gradient descent (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase C] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final polishing – tiny uniform mixing.
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final evaluation via the secure wrapper.

    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
