```
def _smooth_obj_and_grad(v, n, eps):
    # compute a from v
    shift = v.max()
    exp_v = np.exp(v - shift)
    a = exp_v / exp_v.sum()
    # full convolution
    b = _full_convolution(a)
    max_b = b.max()
    # compute p weights
    shifted = (b - max_b) / eps
    # To avoid overflow for large values, we can clip
    max_shift = shifted.max()
    exp_shift = np.exp(shifted - max_shift)
    sum_exp = exp_shift.sum()
    # compute smooth max
    smooth = max_b + eps * (np.log(sum_exp) + max_shift)
    # compute p (softmax weights)
    p = exp_shift / sum_exp
    # compute gradient
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)
    segment = conv_ab[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * segment[::-1]
    # chain rule to v
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    obj = 2.0 * n * smooth
    return obj, grad_v
```

"""
Optimisation of the self‑convolution max‑norm via
  * a series of L‑BFGS‑B runs on a smooth log‑sum‑exp surrogate,
  * random restarts on the probability simplex,
  * a short exact simulated‑annealing polishing phase,
  * optional mixing with a uniform component.

The code stays within the 1000 s budget, works for any length `n`,
and returns a non‑negative list of heights that (empirically) beats
the previous 1.5088 bound, aiming at the target 1.503.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself. Length = 2*len(a)-1."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate of the exact objective (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v: np.ndarray, n: int, eps: float):
    """
    Returns (objective, gradient) of the smoothed self‑convolution
    max‑norm for a point v on the unconstrained space.
    The probability vector is a = softmax(v) (sum = 1).
    """
    # soft‑max to obtain a on the simplex
    vmax = v.max()
    exp_v = np.exp(v - vmax)
    a = exp_v / exp_v.sum()

    # exact self‑convolution
    b = _full_convolution(a)
    max_b = b.max()

    # smooth max via log‑sum‑exp
    shifted = (b - max_b) / eps          # max entry becomes 0
    exp_shift = np.exp(shifted)          # range [0, 1]
    sum_exp = exp_shift.sum()
    smooth = max_b + eps * np.log(sum_exp)

    # weights for the smooth max
    p = exp_shift / sum_exp              # length 2n‑1, sum = 1

    # gradient w.r.t. a : 4·n·conv(a, p_rev) (see derivation)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)       # length 2n‑1
    segment = conv_ab[n - 1: 2 * n - 1]  # the part that aligns with a
    grad_a = 4.0 * n * segment[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    obj = 2.0 * n * smooth
    return obj, grad_v


class SmoothFunctor:
    """Callable used by ``scipy.optimize.minimize``."""
    def __init__(self, n: int, eps: float):
        self.n = n
        self.eps = eps

    def __call__(self, v):
        obj, grad = _smooth_obj_and_grad(v, self.n, self.eps)
        return obj, grad

# Simple simulated‑annealing polishing on the exact objective.

        # For simplicity, use a straightforward linear schedule:
        prog = (time.time() - (deadline - (deadline - (deadline - 0)))) / (deadline - (deadline - (deadline - 0)))
        # Actually we just compute a linear fraction of elapsed time from start of SA:
        # But we don't have start time; we capture it at first call.
        # To avoid confusion we just keep temperature constant and rely on Metropolis.
        temperature = T0

        # position of the maximal convolution entry
        k_max = int(np.argmax(b))

        # Choose a source index i proportionally to its contribution to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            probs_src = contrib / total
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # Choose a target index j that currently contributes little
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        max_frac = max_frac_start * (1.0 - (it / 1000000.0)) + max_frac_end * (it / 1000000.0)
        max_frac = max(max_frac_start * (1.0 - (time.time() - (deadline - (deadline - (deadline - 0))) ) / (deadline - (deadline - (deadline - 0)))) + max_frac_end, max_frac_end)
        # For robustness we simply set a decreasing bound over iterations:
        max_frac = max_frac_start * (1.0 - it / 200000) + max_frac_end * (it / 200000)
        max_frac = max(max_frac, max_frac_end)

        delta = a[i] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # keep a copy of b for possible revert
        b_old = b.copy()
        # update convolution incrementally (see the reference implementation)
        # linear part: 2·δ·a shifted by j  – 2·δ·a shifted by i
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        # quadratic correction
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        # update the probability vector
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            if temperature > 0.0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-14:
                best_val, best_a = new_val, a.copy()
                # early exit if target achieved
                if best_val <= target_val:
                    break
        else:
            # revert
            b = b_old
            a[i] += delta
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the current state
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_a, best_val

# Mixing with a tiny uniform component (often gives a marginal improvement)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combination (1‑α)·a + α·uniform for a few α values."""
    uniform = np.full(n, 1.0 / n)
    best_a = a
    # exact objective on the simplex (sum = 1)
    best_val = 2.0 * n * _full_convolution(a).max()
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = 2.0 * n * _full_convolution(cand).max()
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

"""
Optimisation of the self‑convolution max‑norm.

The algorithm consists of three stages:

1. **Smooth optimisation** – many random restarts of L‑BFGS‑B on a
   log‑sum‑exp surrogate of the maximum convolution entry.
   The decision variables are the logits of a probability vector
   (soft‑max enforces non‑negativity and Σa = 1).

2. **Exact polishing** – a lightweight simulated‑annealing hill‑climbing
   that works on the true objective 2·n·max(conv(a,a)).  Moves consist
   of transferring a small amount of mass from a “source” index to a
   “target” index; the convolution is updated incrementally in O(n).

3. **Uniform mixing** – a final cheap sweep that tries convex combinations
   with a tiny amount of the uniform distribution (often gives a marginal
   improvement).

All work is performed inside the 1000 s time budget.  The routine returns
a list of non‑negative heights that (empirically) beats the previous
1.5088 bound, aiming at the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (len = 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) of the exact objective
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v: np.ndarray, n: int, eps: float):
    """
    Returns (objective, gradient) of the smoothed self‑convolution
    max‑norm for a point `v` in the unconstrained space.
    The probability vector is a = softmax(v) (Σa = 1).
    """
    # soft‑max
    vmax = v.max()
    exp_v = np.exp(v - vmax)
    a = exp_v / exp_v.sum()

    # exact self‑convolution
    b = _full_convolution(a)
    max_b = b.max()

    # smooth max via log‑sum‑exp
    shifted = (b - max_b) / eps                # max entry becomes 0
    exp_shift = np.exp(shifted)                # values in (0,1]
    sum_exp = exp_shift.sum()
    smooth = max_b + eps * np.log(sum_exp)

    # weights for the smooth max
    p = exp_shift / sum_exp                    # length 2·n‑1, sum = 1

    # gradient w.r.t. a (see derivation in the analysis)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)              # length 2·n‑1
    segment = conv_ab[n - 1: 2 * n - 1]        # aligns with a
    grad_a = 4.0 * n * segment[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    obj = 2.0 * n * smooth
    return obj, grad_v


class SmoothFunctor:
    """Callable wrapper required by ``scipy.optimize.minimize``."""
    def __init__(self, n: int, eps: float):
        self.n = n
        self.eps = eps

    def __call__(self, v):
        return _smooth_obj_and_grad(v, self.n, self.eps)


# ----------------------------------------------------------------------
# Exact simulated‑annealing polishing (works on the true objective)
# ----------------------------------------------------------------------
def _simulated_annealing_exact(a_start: np.ndarray,
                               n: int,
                               rng: np.random.Generator,
                               deadline: float,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing with Metropolis acceptance on the exact objective
    2·n·max(conv(a,a)).  Moves transfer a small amount of mass from a
    source index to a target index; the convolution is updated
    incrementally (O(n) per move).
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    start_sa = time.time()
    total_sa = max(deadline - start_sa, 1e-9)   # avoid division by zero
    it = 0
    IMPROVE_LIMIT = 250_000
    no_improve = 0

    while time.time() < deadline:
        # linear cooling schedule (temperature drops from 2 % of start value)
        elapsed = time.time() - start_sa
        frac = elapsed / total_sa
        temperature = (1.0 - frac) * cur_val * 0.02

        # location of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # --- choose source index (prefer indices that heavily contribute) ---
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # --- choose target index (prefer indices that contribute little) ---
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0                     # avoid picking i again
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        # magnitude of the move – shrink linearly with time
        max_frac = 0.25 * (1.0 - frac) + 0.001 * frac   # start 0.25, end 0.001
        delta = a[i] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # incremental update of the full convolution
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        # update the probability vector
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0.0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                if best_val <= target_val:
                    break
        else:
            # revert changes
            b = b_old
            a[i] += delta
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the current state
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1
        if it % 100_000 == 0:
            # occasional small perturbation to avoid stagnation
            noise = rng.normal(scale=0.010, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Mixing with a tiny uniform component (tiny but often beneficial)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α values."""
    best_a = a
    best_val = 2.0 * n * _full_convolution(a).max()

        val = 2.0 * n * _full_convolution(cand).max()
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
