"""
step_optimize.py

A fast, gradient‑based search for a non‑negative step function that
minimises `evaluate_sequence`.  The algorithm consists of three phases:

1. **Initialisation** – several restarts: uniform, sinusoidal,
   random exponential and a heavy‑edge pattern.  All candidates are
   projected onto the probability simplex (Σa = 1).

2. **Additive smoothed‑max descent** – we minimise the
   log‑sum‑exp surrogate of the maximum convolution peak.
   For a temperature ε we use the softmax weights
        p_k ∝ exp(ε·(b_k−max(b))) ,   b = a*a .
   The gradient of the surrogate is obtained with a single FFT
   (the function `_gradient_weighted_peaks`).  An additive mirror‑
   descent step `a ← a – η·∇` is performed and the vector is projected
   back onto the simplex.  The temperature is decreased through a
   schedule reaching ε = 0.001, with a slowly shrinking step size.
   After each ε‑stage an anti‑symmetric re‑ordering (largest ↔ smallest)
   is applied – this never increases the convolution peak.

3. **Hill‑climbing refinement** – cheap local moves (random swaps or
   tiny mass transfers) are accepted only if they strictly improve the
   true objective.  The routine runs until the remaining time is
   exhausted.

The whole search respects the 1000 s budget, prints periodic progress
information and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Small utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding in FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·b_k w.r.t. a,
    where b = a*a and p is a non‑negative weight vector (Σp = 1).

    Result: g_i = Σ_k p_k·a_{k‑i}   (indices outside [0,n) are treated as 0).
    Computed with a single FFT.
    """
    n = a.shape[0]
    # reverse the weight vector for convolution
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # slice that corresponds to indices n‑1 … 2n‑2 (inclusive)
    seg = conv[n - 1 : 2 * n - 1]
    # reverse back to obtain the gradient vector
    g = seg[::-1]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest entries opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b

def _smoothed_descent_additive(a_start: np.ndarray, deadline: float):
    """
    Minimise a log‑sum‑exp surrogate of the max‑convolution by
    additive mirror‑descent.  Returns the best vector found and its
    true evaluation.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent init] objective = {best_val:.6f}")

    # Temperature schedule – we go down to ε = 0.001
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    # Rough number of inner iterations per ε (larger for smaller ε)
    iter_per_eps = [700, 1000, 1200, 1500, 1800, 2500, 3000, 3500, 4000]

    for eps, max_it in zip(eps_schedule, iter_per_eps):
        # step size – a decreasing schedule tied loosely to ε
        eta = 0.8 * eps          # larger ε allows a larger step
        it = 0
        while it < max_it and time.time() < deadline - 5.0:

            conv = _full_convolution(a)          # length 2n‑1
            max_conv = conv.max()

            p = exp_vals / exp_vals.sum()        # Σp = 1, p ≥ 0

            grad = 2.0 * _gradient_weighted_peaks(a, p)   # shape (n,)

            a = _project_simplex(a_new)

def _hill_climb_refine(a: np.ndarray, cur_val: float,

    kept only if they strictly improve the true objective.

    best_val = cur_val
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        if rng.random() < 0.25:                         # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                            # transfer a small amount
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.06 * rng.random()      # up to 6 %
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto simplex (non‑negative + sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # Problem size – a few thousand coefficients balance quality & speed.
    n = 10000

    # ------------------------------------------------------------------
    # If a previously discovered construction is available, reuse it.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum > 0:
            a0 = a0 / a0_sum
        else:
            a0 = np.full(n, 1.0 / n)
    except Exception:
        a0 = None

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Restart loop – different random seeds / simple patterns
    # ------------------------------------------------------------------
    init_strategies = ["uniform", "sin", "rand", "edge"]
    restart = 0
    while time.time() < deadline - 200 and restart < 8:
        restart += 1

        # ----------------------------------------------------------------
        # Choose initial vector
        # ----------------------------------------------------------------
        if a0 is not None and restart == 1:
            a = a0.copy()
            init_name = "from_previous"
        else:
            strat = rng.choice(init_strategies)
            if strat == "uniform":
                a = np.full(n, 1.0 / n)
                init_name = "uniform"
            elif strat == "sin":
                idx = np.arange(n) + 0.5
                a = np.sin(np.pi * idx / n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                init_name = "sinusoid"
            elif strat == "edge":
                # heavy at the ends: a_i ∝ 1/((i+1)*(n-i))
                i = np.arange(n)
                w = 1.0 / ((i + 1.0) * (n - i))
                a = w / w.sum()
                init_name = "edge_heavy"
            else:   # random exponential
                a = rng.exponential(scale=1.0, size=n)
                a = a / a.sum()
                init_name = "random"

        # ----------------------------------------------------------------
        # Run smoothed‑max descent from this start
        # ----------------------------------------------------------------
        a_opt, val_opt = _smoothed_descent_additive(a, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[global] new best {best_val:.6f} (restart {restart}, init={init_name})")

        # ----------------------------------------------------------------
        # Short hill‑climb after each descent (if time permits)
        # ----------------------------------------------------------------
        if time.time() < deadline - 30:
            a_ref, val_ref = _hill_climb_refine(a_opt, val_opt, deadline)
            if val_ref < best_val:
                best_val = val_ref
                best_a = a_ref.copy()
                print(f"[global] post‑hill improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polish: one extra hill‑climb on the best vector
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final evaluation & return the list of floats
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
```

This implementation expands on the previous approach by:

* Using an **additive** mirror‑descent update (instead of multiplicative) which gives a more direct move on the simplex.
* Introducing a **finer temperature schedule** that reaches ε = 0.001 together with many more inner iterations.
* Adding an extra **edge‑heavy initialization** to escape the uniform‑like basin.
* Performing **more frequent re‑ordering** and a final **hill‑climbing** phase.

All heavy lifting (FFT‑based gradient computation) is performed without calling `evaluate_sequence`; the true objective is queried only intermittently, keeping the runtime well within the 1000 s budget while aiming for an objective below the target 1.503.
