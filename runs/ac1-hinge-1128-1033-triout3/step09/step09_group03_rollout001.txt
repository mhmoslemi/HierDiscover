"""
Multi‑restart, multi‑phase optimizer for the min‑max self‑convolution problem.

* Uses several independent runs (different problem sizes, random seeds).
* Each run consists of three phases:
    – Adam on a smooth log‑sum‑exp surrogate (quick coarse optimisation).
    – Simulated‑annealing hill‑climbing with large “mass‑transfer” moves.
    – Exact projected sub‑gradient descent for polishing.
* The best sequence seen over all runs is returned.
* A small post‑processing mixes a tiny amount of uniform mass, which
  often yields a marginal improvement.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length = 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (probability‑simplex normalisation)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Hand‑crafted families – give the optimiser a good launch point
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector (Σ=1) among a few parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n or np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform
    _try(np.ones(n))

    # symmetric exponential (U‑shaped)
    for r in (0.9999, 0.9995, 0.998, 0.995, 0.99):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _try(a)

    # sine with exponent >1 (flatter interior)
    for p in (1.2, 1.5, 2.0, 3.0):
        a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))) ** p
        _try(a)

    # parabola (i·(n‑i))
    a = (np.arange(n) + 1) * (n - np.arange(n))
    _try(a)

    # random Dirichlet draws – a few quick samples
    rng = np.random.default_rng(12345)
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # random Beta‑shaped draws (continuous analogue of a Beta pdf)
    for _ in range(10):
        a = np.arange(1, n + 1)
        a = a ** rng.integers(1, 5)
        _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth surrogate of max(conv)."""
    w = np.log(a0 + 1e-12)                  # logits → simplex via soft‑max
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.003
    eps_start, eps_end = 5e-1, 5e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80000

    it = 0
    while time.time() < deadline:
        # current point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules
        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape flat regions
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Classic SA that moves a random amount of mass from a source index i
    to a target index j. The move is accepted if it improves the objective,
    otherwise with a Metropolis probability based on a temperature schedule.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    # temperature is a fraction of the current objective – linear cooling
    T0 = cur_val * 0.018                     # ~2 % of start value
    max_frac_start, max_frac_end = 0.70, 0.04
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)      # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # most contributing source index for the current maximal entry
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # target tends to be a location with little contribution
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----- incremental convolution update -----
        b_old = b.copy()
        # linear part
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        # quadratic part
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        # update the probability vector
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA  {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            # revert
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake – add modest Gaussian noise and re‑normalise
            print(f"[SA  {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.018, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val

# Phase C – Exact projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.9 / n            # initial step size, roughly the average entry size
    eta = eta0
    decay = 0.9992
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 250_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry (2·a shifted)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step & projection onto the simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # decay step size
        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # small perturb and reset step size
            print(f"[SGD {it:7d}] perturb after stagnation")
            noise = np.random.normal(scale=0.010, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (often a marginal boost)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α values."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Run a single optimisation instance (one problem size)

def _run_once(n: int, time_budget: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Execute all three phases for a given n within the supplied budget."""
    start_time = time.time()
    deadline = start_time + time_budget

    # initialise
    a0, _ = _initial_candidate(n)
    best_a = a0.copy()
    best_val = _objective_value(best_a)

    # Phase A – Adam (≈15 % of budget)
    phase_a_deadline = start_time + 0.15 * (deadline - start_time)
    best_a, best_val = _adam_phase(best_a, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=1.503)

    # Phase B – Simulated annealing (≈70 % of budget)
    phase_b_deadline = start_time + 0.85 * (deadline - start_time)
    best_a, best_val = _simulated_annealing_phase(best_a, n,
                                                  start=phase_a_deadline,
                                                  deadline=phase_b_deadline,
                                                  best_a=best_a,
                                                  best_val=best_val,
                                                  rng=rng,
                                                  target_val=1.503)

    # Phase C – Sub‑gradient polishing (remaining time)
    best_a, best_val = _subgradient_phase(best_a, n,
                                          start=phase_b_deadline,
                                          deadline=deadline,
                                          best_a=best_a,
                                          best_val=best_val,
                                          target_val=1.503)

    # tiny uniform mixing
    best_a, best_val = _mix_with_uniform(best_a, n)

    return best_a, best_val

        # pick a problem size – moderate enough to be fast, large enough for expressive power
        n = int(rng.integers(2500, 8000))
        print(f"[run {run_counter}] n={n}, budget={run_time:.1f}s")
        a, val = _run_once(n, run_time, rng)

        # evaluate with the official evaluator (extra safety against clamping)
        try:
            eval_val = evaluate_sequence(a.tolist())
        except Exception:
            eval_val = float('inf')
        print(f"[run {run_counter}] evaluated value = {eval_val:.6f}")

        if eval_val < best_val:
            best_val, best_seq = eval_val, a.tolist()
            print(f"[run {run_counter}] *** NEW GLOBAL BEST = {best_val:.6f}")

        run_counter += 1

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a simple uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print("[fallback] using uniform distribution")

    # final report
    print(f"[final] best value over all runs = {best_val:.6f}")
    return best_seq
