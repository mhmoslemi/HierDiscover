"""
Hybrid optimisation for the step‑function objective.

The algorithm consists of three stages that together explore the
search‑space more aggressively than the previous approach:

1. **Exponentiated‑gradient descent** (multiplicative update) using the
   top‑K active convolution peaks (K=5).  A linear step‑size schedule
   drives the updates from a large learning rate (≈0.12) to a very fine
   one (≈5e‑4).  The iterate is kept on the probability simplex.

2. **Deterministic hill‑climbing** – repeatedly transfer a fraction of
   the mass from the index that contributes most to the current peak
   to the index that contributes the least.  The transfer fraction
   anneals from 30 % to 0.1 % and a move is kept only if it strictly
   improves the objective.

3. **Simulated‑annealing refinement** – a long SA phase that proposes
   random mass‑transfer moves (source → target).  Moves are accepted
   either when they improve the objective or with a Metropolis
   probability that decays with a temperature schedule
   (T≈0.01·best → 1e‑6).  The SA loop also injects occasional jitter
   to escape plateaus.

All three phases respect the global time budget (≈1000 s) and keep
track of the best vector found.  A final tiny mixing with the uniform
vector is applied before the sequence is returned.

The implementation works for any length; the default length is
8000, which matches the previous best solution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)−1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes sum(a) == 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` when Σa = 1."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Helper: sub‑gradient of the max‑convolution entry (aggregated over top‑K)
# ----------------------------------------------------------------------
def _subgradient_topk(a: np.ndarray, n: int, top_k: int) -> np.ndarray:
    """Return an aggregated sub‑gradient of the max(b) w.r.t. a."""
    b = _full_convolution(a)
    if top_k == 1:
        ks = [int(np.argmax(b))]
    else:
        ks = np.argpartition(b, -top_k)[-top_k:]
    g = np.zeros_like(a)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] += 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase
# ----------------------------------------------------------------------
def _expgrad_phase(
    a0: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
    top_k: int = 5,
) -> tuple[np.ndarray, float]:
    """Multiplicative sub‑gradient descent (exponentiated gradient)."""
    cur = a0.copy()
    no_improve = 0
    IMPATIENT = 250_000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # objective of the current iterate
        cur_val = _objective(cur, n)

        # record best
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # sub‑gradient from the top‑K active peaks
        g = _subgradient_topk(cur, n, top_k)

        # linear learning‑rate schedule (0.12 → 5e‑4)
        progress = (now - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005

        # exponentiated update
        cur = cur * np.exp(-eta * g)

        # projection onto the simplex Σa = 1
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional jitter if stagnating
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.009, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climbing phase
# ----------------------------------------------------------------------
def _hillclimb_phase(
    a_start: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
    start_frac: float = 0.30,
    end_frac: float = 0.001,
) -> tuple[np.ndarray, float]:
    """Move mass from the worst contributor to the best; accept only improvements."""
    cur = a_start.copy()
    no_improve = 0
    IMPATIENT = 300_000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        cur_val = _objective(cur, n)

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # anneal transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # locate the current maximal convolution index
        b = _full_convolution(cur)
        k = int(np.argmax(b))

        # indices that participate in that peak
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)

        # contribution of each a_i to the peak (a_{k-i})
        contrib = np.zeros_like(cur)
        contrib[idx] = cur[k - idx]

        src = int(np.argmax(contrib))
        # target is the index that contributes the least (excluding src)
        tmp = contrib.copy()
        tmp[src] = np.inf
        tgt = int(np.argmin(tmp))

        if cur[src] <= 1e-15:
            # source exhausted – add jitter after long stagnation
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            it += 1
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            it += 1
            continue

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        new_val = _objective(cur, n)

        if new_val < cur_val - 1e-15:
            # accept the move (already applied)
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing_phase(
    a_start: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Long SA phase using random mass‑transfer moves."""
    cur = a_start.copy()
    cur_val = _objective(cur, n)

    # temperature schedule: start at ~1 % of current best, end at 1e‑6
    T_start = max(1e-4, 0.01 * cur_val)
    T_end = 1e-6

    rng = np.random.default_rng(12345)
    no_improve = 0
    IMPATIENT = 200_000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear temperature decay
        progress = (now - start_time) / (deadline - start_time)
        T = (1.0 - progress) * T_start + progress * T_end

        # pick source (must have non‑zero mass) and target
        src = rng.integers(0, n)
        while cur[src] < 1e-12:
            src = rng.integers(0, n)
        tgt = rng.integers(0, n - 1)
        if tgt >= src:
            tgt += 1

        # transfer fraction (annealed: large early, tiny later)
        frac = rng.random() * (0.30 * (1.0 - progress) + 0.001 * progress)
        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        # apply trial move
        cur[src] -= delta
        cur[tgt] += delta

        new_val = _objective(cur, n)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, cur.copy()
                print(f"[SA {it:7d}] new best = {best_val:.6f}")
            no_improve = 0
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta
            no_improve += 1

        # occasional jitter to shake out of plateaus
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.001, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            cur_val = _objective(cur, n)
            no_improve = 0
            print(f"[SA {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector (final polishing)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
