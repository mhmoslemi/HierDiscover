def proj_simplex(v):
    v = np.maximum(v, 0)
    if v.sum() == 0:
        return np.full_like(v, 1.0 / len(v))
    # use algorithm for projecting onto simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, len(v)+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a better step‑function sequence.

The algorithm is a multi‑phase optimiser that works inside the given
time budget (≈1000 s).  It starts from the best known construction (the
global variable ``height_sequence_1`` if present) and improves it with
the following steps:

1. **Adam‑accelerated exponentiated‑gradient descent** –
   a multiplicative mirror‑descent update using the average sub‑gradient
   of the K largest convolution entries (K≈20).  Adam‑style moment
   estimates give an adaptive per‑coordinate step size.

2. **Deterministic hill‑climb** – repeatedly transfer a small fraction
   of the mass from the index that contributes most to the active peak
   to the index that contributes least; the move is kept only if it
   improves the objective.

3. **Fast random restarts** – each restart draws a Dirichlet‑distributed
   start vector, runs a short Adam‑exp‑grad phase followed by a brief
   hill‑climb, and finally a tiny random‑perturbation refinement.

4. **Uniform mixing** – the best vector is blended a few times with the
   uniform distribution; this can smooth isolated spikes.

All vectors are kept on the probability simplex (non‑negative,
sum = 1).  The current best vector and its objective are printed
throughout the run.  The final sequence is returned as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective for a vector normalised to sum = 1
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)   # denominator (sum_a)^2 == 1


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (non‑negative, sum=1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # ensure non‑negative first; then use the O(n log n) algorithm
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, v.shape[0] + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam‑accelerated exponentiated‑gradient descent (top‑K sub‑gradient)
# ----------------------------------------------------------------------
def _adam_expgrad_phase(a0: np.ndarray,
                        n: int,
                        start_time: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        top_k: int = 20,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam momentum using the average sub‑gradient of the
    top‑K convolution entries.  The update is multiplicative and the vector
    is re‑normalised to keep it on the simplex.
    """
    cur = a0.copy()
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        # record best
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            print(f"[AdamExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # ---- sub‑gradient from the K largest convolution entries ----
        ks = np.argpartition(b, -top_k)[-top_k:]
        g = np.zeros_like(cur)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * cur[k - idx]          # contribution a_i a_{k-i}
        g /= float(top_k)                         # average over the active set

        # ---- Adam moment updates ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # linear step‑size schedule (aggressive → fine)
        progress = (now - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * 0.12 + progress * 0.001

        # multiplicative Adam update (preserves non‑negativity)
        cur = cur * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # re‑normalise onto the simplex
        cur = _project_simplex(cur)

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb – deterministic mass transfer
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     start_frac: float = 0.30,
                     end_frac: float = 0.001,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Move a fraction of the mass from the index that contributes most to the
    current maximal convolution entry to the index that contributes least.
    Accept only strict improvements.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if cur_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # locate the active peak
        k = int(np.argmax(b))

        # contributions of each index to this peak (a_{k-i})
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = cur[k - idx]

        src = int(idx[np.argmax(contrib)])          # largest contribution
        tgt_candidates = np.where(contrib == contrib.min())[0]
        # pick the farthest index among the minima to diversify moves
        tgt = int(idx[tgt_candidates[-1]])

        if cur[src] <= 1e-15:
            # if source emptied, shake a little
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.005, size=n)
                cur = _project_simplex(cur + noise)
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            it += 1
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            it += 1
            continue

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        new_val = 2.0 * n * _full_convolution(cur).max()
        if new_val < cur_val - 1e-15:
            # accept
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny random perturbations – simple simulated‑annealing style move
# ----------------------------------------------------------------------
def _perturb_phase(a_start: np.ndarray,
                   n: int,
                   start_time: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Apply small Gaussian jitter and keep the change if it improves the
    objective.  The jitter amplitude decays with time.
    """
    cur = a_start.copy()
    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # decreasing noise amplitude
        progress = (now - start_time) / (deadline - start_time)
        sigma = (1.0 - progress) * 0.01 + progress * 0.001
        noise = np.random.normal(scale=sigma, size=n)
        cand = _project_simplex(cur + noise)
        cand_val = 2.0 * n * _full_convolution(cand).max()
        cur_val = 2.0 * n * _full_convolution(cur).max()
        if cand_val < cur_val - 1e-15:
            cur = cand
            if cand_val < best_val - 1e-15:
                best_val, best_a = cand_val, cand.copy()
                print(f"[Perturb {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.005, 0.010, 0.020, 0.030):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

        # final uniform mixing
        a, val = _mix_with_uniform(a, n)

        # global update
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Restart {restart_counter:3d}] new global best = {best_val:.6f}")

        if best_val <= TARGET:
            print("[Target reached] exiting early.")
            break

        restart_counter += 1

    # ------------------------------------------------------------------
    # Final verification (optional)
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
