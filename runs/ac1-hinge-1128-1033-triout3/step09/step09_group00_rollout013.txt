"""
Improved search for a low‑∞‑norm auto‑convolution step function.

The algorithm consists of three phases:

1. **Broad cheap exploration** – scans a flexible two‑parameter Beta family,
   random Beta draws, Dirichlet draws (including highly sparse
   Dirichlet with concentration <1) and a simple power‑law family.
   All candidates are normalised (∑a = 1) and the best one is kept.

2. **Local stochastic optimisation** – a customised simulated‑annealing
   hill‑climber that repeatedly moves a small amount of mass from an index
   that contributes heavily to the current maximal convolution entry to an
   index that contributes little.  The convolution update is performed
   incrementally (O(n)) and the quadratic correction is exact
   (the original code missed a factor 2).  The temperature and maximal
   transfer fraction are annealed aggressively, with occasional jitter to
   escape plateaus.

3. **Deterministic sub‑gradient polishing** – projected sub‑gradient descent
   on the exact objective `2·n·max(conv(a,a))`.  The sub‑gradient is derived
   analytically (the gradient is a sum of shifted copies of `a`).  A simple
   linear step‑size schedule finishes the optimisation.

All phases respect the 1000 s time budget and the simplex constraint
(`a_i ≥ 0, Σ a_i = 1`).  The final sequence is returned as a plain Python
list, ready for `evaluate_sequence`.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (cheap exhaustive / random search)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> Tuple[np.ndarray, float]:
    """Exhaustive sweep of a two‑parameter Beta family."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _beta_random_search(n: int, trials: int = 2000) -> Tuple[np.ndarray, float]:
    """Randomly sample p,q on a log‑scale and keep the best Beta shape."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    rng = np.random.default_rng()
    for _ in range(trials):
        logp = rng.uniform(-6.9, 2.3)   # ln(0.001) … ln(10)
        logq = rng.uniform(-6.9, 2.3)
        p = np.exp(logp)
        q = np.exp(logq)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


def _random_powerlaw_search(n: int, trials: int = 2000) -> Tuple[np.ndarray, float]:
    """Sample a_i ∝ i^p (p uniform in [-5,5]) and keep the best."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    rng = np.random.default_rng()
    for _ in range(trials):
        p = rng.uniform(-5.0, 5.0)

        s = a.sum()
        if s == 0.0:

        a /= s

        if val < best_val:

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (corrected convolution update)
# ----------------------------------------------------------------------
def _hillclimb_corrected(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_a: np.ndarray,
                         best_val: float,
                         target_val: float = 1.503) -> Tuple[np.ndarray, float]:
    """Fast stochastic local search with an exact O(n) convolution update."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule parameters
    max_frac_start, max_frac_end = 0.6, 0.001
    temp_start, temp_end = 0.05, 1e-7
    no_improve = 0
    IMPROVE_LIMIT = 50000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to this entry (a_i * a_{k_max-i})
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – probability ∝ contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – bias towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (excluding i)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # --- incremental convolution update (exact) ---
        b_new = b.copy()
        # linear part   (+2·Δ·a)   with original a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic correction (exact)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – jitter around the current best
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – jitter")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503,
                        max_iters: int = 30000) -> Tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact objective."""
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # compute convolution and its maximal entries
        b = _full_convolution(a)
        max_val = b.max()
        k_maxes = np.where(b == max_val)[0]

        # sub‑gradient of the max‑convolution
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step‑size schedule (large at the beginning, vanishing later)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.4 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Subgrad {it:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break

        # occasional tiny jitter to escape flat regions
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val

        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        best_a, best_val = a0.copy(), _objective(a0)
        print("[init] interpolated previous solution, value={:.6f}".format(best_val))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap wide‑range exploration
    # ------------------------------------------------------------------
    # – Beta family grid
    try:
        a_beta, v_beta = _beta_family_best(n)
        if v_beta < best_val:
            best_a, best_val = a_beta, v_beta
            print("[init] beta‑family grid improved value={:.6f}".format(best_val))
    except Exception as exc:
        print("[init] beta‑family grid failed:", exc)

    # – Random Beta draws
    try:
        a_rand, v_rand = _beta_random_search(n, trials=2000)
        if v_rand < best_val:
            best_a, best_val = a_rand, v_rand
            print("[init] beta‑random improved value={:.6f}".format(best_val))
    except Exception as exc:
        print("[init] beta‑random failed:", exc)

    # – Random power‑law shapes
    try:
        a_pow, v_pow = _random_powerlaw_search(n, trials=2000)
        if v_pow < best_val:
            best_a, best_val = a_pow, v_pow
            print("[init] power‑law improved value={:.6f}".format(best_val))
    except Exception as exc:
        print("[init] power‑law search failed:", exc)

    # – Dirichlet samples with low concentration (sparse seeds)
    for _ in range(8):
        try:
            alpha = np.random.uniform(0.01, 0.5)   # encourages sparsity
            a_dir = np.random.dirichlet(np.full(n, alpha))
            v_dir = _objective(a_dir)
            if v_dir < best_val:
                best_a, best_val = a_dir, v_dir
                print("[init] Dirichlet improved value={:.6f}".format(best_val))
        except Exception:
            continue

    # fall back to a uniform random start if nothing succeeded
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] random fallback, value={:.6f}".format(best_val))

    # ------------------------------------------------------------------
    # Phase A – Stochastic hill‑climbing (majority of the time)
    # ------------------------------------------------------------------
    hill_deadline = start_time + 0.78 * budget_s
    best_a, best_val = _hillclimb_corrected(
        best_a, n,
        deadline=hill_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
    )
    print("[phase A] hill‑climb finished, best = {:.6f}".format(best_val))

    # ------------------------------------------------------------------
    # Phase B – Deterministic sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline=deadline,
        start_time=hill_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
        max_iters=30000,
    )
    print("[phase B] sub‑gradient polishing finished, best = {:.6f}".format(best_val))

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score = {:.6f}".format(final_score))
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
