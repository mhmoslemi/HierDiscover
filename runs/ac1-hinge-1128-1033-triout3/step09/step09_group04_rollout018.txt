"""
Improvements over the previous search:
* Uses a *smooth* approximation of the max‑convolution (soft‑max with a
  temperature that slowly grows).  This gives a well‑behaved gradient that
  takes *all* convolution peaks into account, not just the top‑K ones.
* Gradient is obtained by a single FFT‑based correlation (O(n log n) per
  iteration) and applied with an exponentiated‑gradient (mirror descent)
  update that automatically respects non‑negativity and the simplex.
* A simple Adam‑style momentum scheme is added to accelerate convergence.
* After the smooth phase a deterministic hill‑climb (as in the reference
  implementation) cleans up the remaining small irregularities.
* The whole budget (≈1000 s) is split into a warm‑up phase, a hill‑climb
  phase and many cheap random restarts, each using a short smooth‑optimise
  followed by a brief hill‑climb.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (linear full convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear full convolution a*a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _correlate(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation (a ⋆ w).  Result length = len(a)+len(w)-1.
    The slice ``[len(w)-1 : len(w)-1+len(a)]`` contains
    ``sum_j a_j * w_{i+j}`` for i=0..len(a)-1, which is exactly the
    gradient of  Σ_k ρ_k·b_k  w.r.t. a_i when ρ_k are fixed weights.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr_full = np.fft.irfft(np.conj(fa) * fw, size)[: n + m - 1]
    start = m - 1
    return corr_full[start:start + n]   # length n


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simple parametric families – used only for an initial fallback
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector (uniform or a few hand‑crafted shapes)."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(candidate):
        nonlocal best_a, best_val
        cand = np.asarray(candidate, dtype=np.float64)
        if cand.ndim != 1 or cand.shape[0] != n:
            return
        if np.any(cand < 0):
            return
        if cand.sum() <= 0:
            return
        cand = cand / cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()

    rng = np.random.default_rng(0)

    # a couple of deterministic shapes
    idx = np.arange(n, dtype=float)
    try_update(np.ones(n) / n)                                # uniform
    try_update(np.sqrt(idx + 1))                               # gentle rising
    try_update(np.cos(np.pi * idx / (n - 1)) ** 2 + 0.1)       # cosine‑ish
    try_update(np.exp(- (idx - n / 2) ** 2 / (2 * (n / 8) ** 2)))  # Gaussian

    # a few random Dirichlet draws
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth optimisation – soft‑max of the convolution peaks
# ----------------------------------------------------------------------
def _smooth_opt_phase(a0: np.ndarray,
                      n: int,
                      start_time: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Minimises a smoothed version of the max‑convolution.
    Uses exponentiated‑gradient updates with Adam‑style momentum.
    """
    cur = a0.copy()
    # Adam parameters
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it = 0
    # iteration budget – we just run until the deadline
    while True:
        now = time.time()
        if now >= deadline:
            break

        # current convolution and objective
        conv = _full_convolution(cur)
        maxb = conv.max()
        cur_val = 2.0 * n * maxb

        # record best
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # ----- soft‑max weights -------------------------------------------------
        # temperature schedule – start gentle, finish sharp
        # progress is measured against the overall time budget of this phase
        progress = (now - start_time) / (deadline - start_time + 1e-12)
        beta = 5.0 + progress * 195.0          # 5 → 200

        # stable soft‑max
        shift = maxb
        exp_terms = np.exp(beta * (conv - shift))
        w = exp_terms / exp_terms.sum()                     # shape 2n‑1

        # ----- gradient (correlation) -----------------------------------------
        grad = 2.0 * _correlate(cur, w)                     # shape n

        # ----- Adam‑style step size -------------------------------------------
        it1 = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)

        # learning‑rate schedule (aggressive → fine)
        eta = (1.0 - progress) * 0.15 + progress * 0.001

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # exponentiated update (keeps positivity)
        cur = cur * np.exp(-step)

        # re‑normalise to the simplex
        s = cur.sum()
        if s == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb – deterministic mass transfer
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     start_frac: float = 0.30,
                     end_frac: float = 0.001,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move a fraction of the mass from the largest‑contributing index
    to the smallest one; accept only strict improvements."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(cur)
        maxb = conv.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Hill {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        progress = (now - start_time) / (deadline - start_time + 1e-12)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        lo = max(0, k - (n - 1))

        idx = np.arange(lo, hi + 1)

        src = int(idx[np.argmax(contrib)])          # biggest offender
        # target = index with smallest contribution (excluding src)
        tmp = contrib.copy()
        tmp[src - lo] = np.inf
        tgt = int(idx[np.argmin(tmp)])

        if cur[src] <= 1e-15:
            # source exhausted – occasional random jitter
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[Hill {it:7d}] jitter after stagnation")
            it += 1
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            it += 1
            continue

        cur[src] -= delta
        cur[tgt] += delta

        # accept only if it improves the objective
        conv_new = _full_convolution(cur)
        new_val = 2.0 * n * conv_new.max()
        if new_val >= cur_val - 1e-15:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the provided
    `evaluate_sequence`.  The method consists of:

    1.  Warm‑up with a smooth‑max optimisation (soft‑max temperature ramps up).
    2.  Deterministic hill‑climb to clean up remaining irregularities.
    3.  Tiny mixing with the uniform vector.
    4.  Many inexpensive random restarts, each applying a short
        smooth optimisation followed by a brief hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4          # safety margin
    n = 8000                                        # vector length

    # ------------------------------------------------------------------
    # Initialise from the previous best if available
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the current length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    TARGET = 1.503
    print(f"[Init] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑0 : smooth optimisation + hill‑climb
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.30 * (deadline - start_time)

    best_a, best_val = _smooth_opt_phase(
        best_a, n,
        start_time, phase0_deadline,
        best_a, best_val,
        target_val=TARGET
    )

    # hill‑climb to polish
    best_a, best_val = _hillclimb_phase(
        best_a, n,
        phase0_deadline, deadline,
        best_a, best_val,
        start_frac=0.30,
        end_frac=0.001,
        target_val=TARGET
    )
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Subsequent fast restarts
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 123)
    restart_counter = 0

    while time.time() < deadline:
        remaining = deadline - time.time()
        # each restart gets at most ~0.7 % of the remaining time,
        # but never less than 0.5 s.
        slice_budget = max(0.5, 0.007 * remaining)

        rs = time.time()
        r_deadline = rs + slice_budget

        # random Dirichlet start (log‑uniform concentration)
        log_alpha = rng.uniform(-2, 2)               # α ∈ [10⁻², 10²]
        alpha = 10.0 ** log_alpha
        a = rng.dirichlet(alpha * np.ones(n))

        # brief smooth optimisation (no more than a few hundred iterations)
        a, _ = _smooth_opt_phase(
            a, n,
            rs, r_deadline,
            best_a,
            best_val,
            target_val=TARGET
        )

        # short hill‑climb
        a, _ = _hillclimb_phase(
            a, n,
            rs, r_deadline,
            best_a,
            best_val,
            start_frac=0.25,
            end_frac=0.001,
            target_val=TARGET
        )

        # final tiny mixing
        a, val = _mix_with_uniform(a, n)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Restart {restart_counter:3d}] new global best = {best_val:.6f}")

        if best_val <= TARGET:
            print("[Target reached] exiting early.")
            break

        restart_counter += 1

    # ------------------------------------------------------------------
    # Final verification via the supplied secure wrapper (optional)
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return plain Python list (sum will be ≈1)
    return best_a.tolist()
