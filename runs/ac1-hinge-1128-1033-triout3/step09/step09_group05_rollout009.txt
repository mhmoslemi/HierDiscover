import time, numpy as np

def _full_convolution(a): ...
def _objective(a, n): ...

def proj_simplex(v):
    # Standard algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _softmax_gradient_step(a, n, beta, eta):
    # compute b
    b = _full_convolution(a)
    maxb = b.max()
    # weights
    w = np.exp(beta * (b - maxb))
    # compute gradient via convolution:
    w_rev = w[::-1]
    # compute convolution a * w_rev
    size = 1 << ((n + len(w_rev) - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv_full = np.fft.irfft(fa * fw, size)[: n + len(w_rev) - 1]
    offset = len(w_rev) - 1
    g = 2.0 * conv_full[offset - np.arange(n)]

    a_new = a * np.exp(-eta * g)

    s = a_new.sum()
    if s == 0:

"""
Optimiser for the step‑function attack.

The routine builds a large (n≈8000) non‑negative vector a with Σa=1.
It combines three phases:

1. **Exploratory random families** – quickly samples a rich
   collection of parametric shapes (exponential, reversed exponential,
   power‑law, power‑law×exponential and mixtures).  The best
   candidate is kept.

2. **Soft‑max gradient descent** – minimises a smooth approximation of
   max(a∗a).  For a given temperature β we set
      w_k ∝ exp(β·( (a∗a)_k – max(a∗a) )),
   then the gradient of the weighted sum is
      g = 2· (a ⋆ w)   (⋆ = correlation).
   An exponentiated‑gradient step a←a·exp(−η·g) is performed and the
   vector is projected back onto the simplex.  β is increased from a
   gentle value (~0.5) to a large one (~200) while η decays, so the
   algorithm first smooths the landscape and then concentrates on the
   true max‑norm.

3. **Fine deterministic hill‑climbing** – after the smooth phase we
   polish the solution with the classic “move mass from a high‑contrib
   index to a low‑contrib one” move, using a slowly decreasing move
   fraction.  A tiny final mixing with the uniform vector is applied.

All heavy work uses FFT‑based full convolutions (O(n log n)) and the
whole procedure respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution using FFT (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (full length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective used during optimisation (Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (optional)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are zero – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for the initial random sweep
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power(n: int, p: float) -> np.ndarray:
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Quick deterministic families – used for a guaranteed decent start
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective(vec, n)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    rng = np.random.default_rng(12345)

    idx = np.arange(n)
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # U‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99,
              0.98, 0.95):
        i = np.arange(n, dtype=np.float64)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val

def _random_mix_candidate(n: int, rng: np.random.Generator):

    a = np.zeros(n, dtype=np.float64)
    n_comp = rng.integers(1, 4)            # 1, 2 or 3 components
    for _ in range(n_comp):
        fam = rng.choice(['exp', 'rev', 'pow', 'powexp'])
        if fam == 'exp':

            part = _gen_exp(n, r)
        elif fam == 'rev':
            r = rng.uniform(0.5, 0.99999)
            part = _gen_rev_exp(n, r)
        elif fam == 'pow':
            p = rng.uniform(-5.0, 5.0)
            part = _gen_power(n, p)
        else:  # powexp
            p = rng.uniform(-5.0, 5.0)
            r = rng.uniform(0.5, 0.99999)
            part = _gen_power_exp(n, p, r)
        w = rng.random()
        a += w * part
    s = a.sum()
    if s == 0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# Phase‑1 – soft‑max (smooth) gradient descent
# ----------------------------------------------------------------------
def _softmax_descent_phase(a0: np.ndarray, n: int,
                           start: float, deadline: float,
                           best_a: np.ndarray, best_val: float,
                           target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(a∗a) by exponentiated‑gradient.
    β (temperature) rises from β0 to β_max, η (step) falls from η0 to η_min.
    """
    cur = a0.copy()
    best_local_a = best_a.copy()
    best_local_val = best_val

    # pre‑compute FFT size that accommodates a (len n) and w (len 2n‑1)
    fft_size = 1 << ((3 * n - 2).bit_length())
    it = 0
    IMPATIENT = 150_000            # jitter after stagnation
    no_improve = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----------- compute convolution a*a and soft‑max weights ----------
        a_fft = np.fft.rfft(cur, fft_size)
        # a*a
        conv_full = np.fft.irfft(a_fft * a_fft, fft_size)[: 2 * n - 1]
        maxb = conv_full.max()

        # scaled soft‑max weights (numerically stable)
        beta = 0.5 + (now - start) / (deadline - start) * (200.0 - 0.5)
        w = np.exp(beta * (conv_full - maxb))          # w ∈ (0,1]

        # ----------- gradient of weighted sum -----------------------------
        w_rev = w[::-1]                                 # reverse for correlation
        w_fft = np.fft.rfft(w_rev, fft_size)
        corr = np.fft.irfft(a_fft * w_fft, fft_size)[: n + len(w_rev) - 1]
        offset = len(w_rev) - 1                         # = 2n‑2
        g = 2.0 * corr[offset - np.arange(n)]

        # ----------- exponentiated‑gradient update -----------------------
        progress = (now - start) / (deadline - start)
        eta = 0.3 * (1.0 - progress) + 0.001 * progress   # 0.3 → 0.001
        # avoid overflow in the exponential
        g = np.clip(g, -1e4, 1e4)
        cur = cur * np.exp(-eta * g)
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s if s > 0 else np.full_like(cur, 1.0 / n)

        cur_val = 2.0 * n * maxb
        if cur_val < best_local_val - 1e-15:
            best_local_val, best_local_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SoftGrad {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve > IMPATIENT:
            # add a tiny isotropic jitter to escape plateaus
            noise = np.random.normal(scale=0.004, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SoftGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_local_a, best_local_val

# Phase‑2 – deterministic greedy hill‑climbing (fine polishing)
# ----------------------------------------------------------------------
def _fine_hillclimb(a_start: np.ndarray, n: int,
                    start: float, deadline: float,
                    best_a: np.ndarray, best_val: float,
                    target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    best_local_a = best_a.copy()
    best_local_val = best_val
    it = 0
    no_improve = 0
    IMPATIENT = 250_000
    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-15:
            best_local_val, best_local_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillFine {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        # locate the index k where the convolution attains its maximum
        k = int(np.argmax(b))
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rev = np.zeros(n, dtype=np.float64)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]               # contributions to the max entry

        # pick i from the upper 20 % of rev, j from the lower 20 %
        high_thr = np.percentile(rev, 80)
        low_thr = np.percentile(rev, 20)
        high_candidates = np.where(rev >= high_thr)[0]
        low_candidates = np.where(rev <= low_thr)[0]
        if high_candidates.size == 0 or low_candidates.size == 0:
            # fall back to the classic extreme‑pair choice
            i = int(np.argmax(rev))
            tmp = rev.copy()
            tmp[i] = np.inf
            j = int(np.argmin(tmp))
        else:
            i = int(np.random.choice(high_candidates))
            j = int(np.random.choice(low_candidates))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.003, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillFine {it:7d}] jitter after stagnation")
            continue

        # move a fraction of a_i to a_j
        progress = (now - start) / (deadline - start)
        max_frac = 0.5 * (1.0 - progress) + 0.001 * progress
        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            it += 1
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        b_new = _full_convolution(cand)
        new_val = 2.0 * n * b_new.max()

        # accept only improving moves
        if new_val < cur_val - 1e-15:
            cur = cand
            cur_val = new_val
        it += 1

    return best_local_a, best_local_val

# Phase‑3 – final tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    # ------------------------------------------------------------------
    phase2_deadline = deadline
    best_a, best_val = _fine_hillclimb(
        best_a, n,
        start=phase1_deadline,
        deadline=phase2_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase‑3 – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification with the provided security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
