"""
A fast random‑search optimizer for the `evaluate_sequence` objective.

The optimizer works in two stages inside the allotted wall‑clock budget:

1. **Broad stochastic sweep** – generate thousands of candidate
   step‑functions from a handful of expressive parametric families
   (logistic, shifted power‑law, beta, piecewise‑constant, Dirichlet).
   Each candidate is normalised to sum = 1 and its exact objective
   (via an FFT‑based convolution) is evaluated.  The best candidate so
   far is kept.

2. **Targeted hill‑climbing** – starting from the best candidate,
   repeatedly relocate a tiny amount of mass from a “heavy” index to a
   “light” index, updating the full convolution in O(n) time (the same
   update logic used in the reference implementation).  Only moves
   that strictly improve the objective are accepted; after a long
   stagnation period the current best is perturbed and the search
   continues.

The routine respects the 1000 s time budget, prints progress, and
finally returns the best sequence found (as a list of non‑negative
floats).  `evaluate_sequence` is called only once at the end to obtain
the official score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast exact objective (sum of the sequence is assumed to be 1)
# ----------------------------------------------------------------------
def _objective_fft(a: np.ndarray) -> float:
    """Exact `evaluate_sequence` objective for a vector already normalised."""
    n = a.shape[0]
    # next power‑of‑two size for an efficient FFT
    size = 1 << ((2 * n - 2).bit_length())
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Helper: incremental convolution update used by hill‑climbing
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution via FFT (used only for verification)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _incremental_update(b: np.ndarray, a: np.ndarray,
                        i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a copy of `b` (the full convolution) after moving `delta`
    mass from index `i` to index `j` in `a`.  The update follows the
    derivation used in the reference hill‑climber and runs in O(n).
    """
    n = a.shape[0]
    b_new = b.copy()

    # contribution of the moved mass with every entry of a
    b_new[j : j + n] += 2.0 * delta * a
    b_new[i : i + n] -= 2.0 * delta * a

    # self‑interaction terms (appear only once in the full convolution)
    b_new[2 * j] += delta * delta
    b_new[2 * i] += delta * delta
    b_new[i + j] -= 2.0 * delta * delta

    return b_new


# ----------------------------------------------------------------------
# Random‑parameter families
# ----------------------------------------------------------------------
def _candidate_logistic(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotone logistic decay a_i = 1/(1+exp(slope*(i‑shift)))."""
    slope = rng.uniform(0.001, 0.05)
    shift = rng.uniform(0.0, n - 1.0)
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp(slope * (i - shift)))
    return a


def _candidate_shifted_power(n: int, rng: np.random.Generator) -> np.ndarray:
    """Shifted power‑law a_i = (i + delta + 1)^{-p}."""
    p = rng.uniform(0.1, 4.0)          # exponent
    delta = rng.uniform(0.0, 5000.0)   # shift of the origin
    i = np.arange(n, dtype=float)
    a = (i + delta + 1.0) ** (-p)
    return a


def _candidate_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑shaped density on the discrete grid."""
    a_param = rng.uniform(0.2, 5.0)
    b_param = rng.uniform(0.2, 5.0)
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (a_param - 1.0)) * ((n + 1 - i) ** (b_param - 1.0))
    return a


def _candidate_piecewise(n: int, rng: np.random.Generator) -> np.ndarray:
    """Two‑level piecewise‑constant: high on the edges, low in the middle."""
    k_frac = rng.uniform(0.01, 0.20)            # edge width as a fraction of n
    k = max(1, int(round(k_frac * n)))
    high_frac = rng.uniform(0.05, 0.50)        # mass on each edge cell
    total_high = 2 * k * high_frac
    if total_high >= 1.0:
        total_high = 0.99
    low = (1.0 - total_high) / (n - 2 * k)
    a = np.full(n, low, dtype=float)
    a[:k] = high_frac
    a[-k:] = high_frac
    return a


def _candidate_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random draw from a symmetric Dirichlet (the baseline “random” family)."""
    return rng.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Hill‑climbing on the exact objective (mirrors the reference implementation)
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               deadline: float,
               best_val: float,
               best_a: np.ndarray) -> np.ndarray:
    """
    Greedy mass‑transfer hill‑climbing.  The loop runs until `deadline`
    is reached or until progress stalls for a long period.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * a.shape[0] * b.max()

    n = a.shape[0]
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # pick a source index weighted by its contribution to the current maximal entry
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # choose a target – favour indices that hardly affect the current max
        if np.random.rand() < 0.43:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = int(np.random.choice(n, p=probs_j))
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        # transfer a small random fraction
        delta = a[i] * np.random.rand() * 0.005   # very gentle moves
        if delta <= 0.0:
            continue

        b_new = _incremental_update(b, a, i, j, delta)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill   {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the current best a little and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill   {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The method combines a large random‑parametric sweep with a short
    deterministic hill‑climb.  It respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    n = 9000                                    # length giving a good trade‑off
    print("[Init] start – uniform baseline")
    best_a = np.full(n, 1.0 / n, dtype=float)
    best_val = _objective_fft(best_a)
    print(f"[Init] uniform objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – massive random‑parameter sweep
    # ------------------------------------------------------------------
    family_funcs = [_candidate_logistic,
                    _candidate_shifted_power,
                    _candidate_beta,
                    _candidate_piecewise,
                    _candidate_dirichlet]

    it = 0
    while time.time() < deadline * 0.60:        # first ~60 % of the budget
        fn = rng.choice(family_funcs)
        cand = fn(n, rng)

        # protect against NaNs / negatives (just in case)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s                                 # normalise to sum = 1

        val = _objective_fft(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Rand {it:7d}] new best = {best_val:.6f}")
        it += 1

    # ------------------------------------------------------------------
    # Stage 2 – targeted hill‑climbing from the best found so far
    # ------------------------------------------------------------------
    print("[Stage B] starting hill‑climb")
    best_a = _hillclimb(best_a,
                        deadline=deadline,
                        best_val=best_val,
                        best_a=best_a)

    # final official evaluation (the wrapper applies the clipping/safety checks)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # The harness expects a plain list of floats
    return best_a.tolist()
