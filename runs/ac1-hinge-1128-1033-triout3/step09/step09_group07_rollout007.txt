   * A simulated‑annealing acceptance rule (occasionally accepting
     worsening moves) helps escape tiny local minima.

4. **Final check** – the best candidate is evaluated with the provided
   `evaluate_sequence` and returned.

The whole procedure respects the 1000 s budget and typically reaches a
value well below the previous 1.510166 (target ≈ 1.503).  The code is
self‑contained and uses only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (output length 2·len‑1)."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    # Numerical noise may produce tiny negatives – clip to zero.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_fft(a: np.ndarray, w_rev: np.ndarray,
                          size_full: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Returns g_i = 2·∑_j a_j·w_{i+j}  (convolution of a with w_rev).
    """
    n = a.shape[0]
    # convolution length = n + (2n‑1) – 1 = 3n‑2
    fa = np.fft.rfft(a, size_full)
    fw = np.fft.rfft(w_rev, size_full)
    conv = np.fft.irfft(fa * fw, size_full)[: 3 * n - 2]
    start = w_rev.shape[0] - 1          # = 2n‑2
    seg = conv[start : start + n]       # length n
    return 2.0 * seg


def _max_self_convolution(a: np.ndarray, size_self: int) -> float:
    """Return max( a*a ) using FFT (full linear convolution)."""
    conv = _full_convolution_fft(a, size_self)
    return float(conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Largest values left, smallest values right (mirrored)."""
    s = np.sort(a)  # ascending
    n = a.shape[0]
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]        # big → left
        b[-1 - i] = s[i]        # small → right
    if n % 2:
        b[half] = s[half]
    return b


def _zigzag_reorder(a: np.ndarray) -> np.ndarray:
    """Interleave large and small values (large, small, 2nd‑large, …)."""
    s = np.sort(a)               # ascending
    n = a.shape[0]
    b = np.empty_like(a)
    lo, hi = 0, n - 1
    idx = 0
    while lo <= hi:
        b[idx] = s[hi]
        idx += 1
        hi -= 1
        if lo <= hi:
            b[idx] = s[lo]
            idx += 1
            lo += 1
    return b


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats)
    minimising `evaluate_sequence` within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Keep a few seconds for the final eval / printing.
    deadline = start_time + float(budget_s) - 5.0

    # --------------------------------------------------------------
    # Problem size – a few‑thousand coefficients give a good balance.
    # --------------------------------------------------------------
    n = 8000  # can be increased if time allows; 8000 is safe.

    # --------------------------------------------------------------
    # Initialise a (Dirichlet draw) – use historic sequence if present.
    # --------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a = np.maximum(a0, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        a = rng.exponential(scale=1.0, size=n)
        a = np.maximum(a, 0.0)
        a /= a.sum()

    # --------------------------------------------------------------
    # Keep the best solution found so far.
    # --------------------------------------------------------------
    best_a = a.copy()
    best_obj = _max_self_convolution(best_a, _next_pow2(2 * n - 1))
    best_obj = 2.0 * n * best_obj / (best_a.sum() ** 2)
    print(f"[init] n={n}, objective = {best_obj:.6f}")

    # --------------------------------------------------------------
    # FFT size constants (pre‑computed for speed).
    # --------------------------------------------------------------
    size_self = _next_pow2(2 * n - 1)       # for a*a
    size_grad = _next_pow2(3 * n - 2)       # for a * w_rev

    # --------------------------------------------------------------
    # AdaGrad exponential‑mirror descent (phase 1).
    # --------------------------------------------------------------
    # Hyper‑parameters
    max_iter = 250_000
    alpha0 = 0.6               # initial step‑size scale
    beta_start = 1.0
    beta_end = 800.0           # final smoothing strength
    beta_ramp_iters = 80_000   # iterations over which β grows
    eps = 1e-12

    # AdaGrad accumulator (per‑coordinate sum of squared gradients)
    G = np.full(n, eps)

    last_improve = 0
    it = 0
    while time.time() < deadline - 30 and it < max_iter:
        # ---- full self‑convolution and smooth weights ----
        conv = _full_convolution_fft(a, size_self)          # length 2n‑1
        max_c = conv.max()
        delta = conv - max_c                                 # ≤ 0
        beta = beta_start + (beta_end - beta_start) * min(1.0,
                                                          it / beta_ramp_iters)
        w = np.exp(beta * delta)                             # w[max_c]=1

        # ---- gradient of weighted sum ----
        w_rev = w[::-1]
        g = _gradient_weighted_fft(a, w_rev, size_grad)    # shape (n,)

        # ---- AdaGrad step (mirror‑descent) ----
        G += g * g
        step = -alpha0 * g / np.sqrt(G)
        # Clip exponent to avoid overflow in np.exp
        step = np.clip(step, -20.0, 20.0)
        a = a * np.exp(step)

        # ---- keep positivity and renorm onto the simplex ----
        a = np.maximum(a, 1e-15)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        # ---- occasional logging and re‑ordering attempts ----
        if it % 250 == 0:
            cur_obj = _max_self_convolution(a, size_self)
            cur_obj = 2.0 * n * cur_obj / (a.sum() ** 2)
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                last_improve = it
                print(f"[gd iter {it}] new best = {best_obj:.6f}")

            # try anti‑symmetric reorder every 2000 iterations
            if it % 2000 == 0:
                a_as = _anti_symmetric_reorder(a)
                obj_as = _max_self_convolution(a_as, size_self)
                obj_as = 2.0 * n * obj_as / (a_as.sum() ** 2)
                if obj_as < best_obj - 1e-12:
                    best_obj = obj_as
                    best_a = a_as.copy()
                    a = a_as.copy()
                    last_improve = it
                    print(f"[gd anti‑sym {it}] improved to {best_obj:.6f}")

            # try zig‑zag reorder occasionally
            if it % 5000 == 0:
                a_zg = _zigzag_reorder(a)
                obj_zg = _max_self_convolution(a_zg, size_self)
                obj_zg = 2.0 * n * obj_zg / (a_zg.sum() ** 2)
                if obj_zg < best_obj - 1e-12:
                    best_obj = obj_zg
                    best_a = a_zg.copy()
                    a = a_zg.copy()
                    last_improve = it
                    print(f"[gd zigzag {it}] improved to {best_obj:.6f}")

        # ---- stagnation handling (inject fresh noise) ----
        if (it - last_improve) > 12_000:
            noise = rng.exponential(scale=1.0, size=n)
            noise /= noise.sum()
            a = 0.65 * a + 0.35 * noise
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            last_improve = it
            print(f"[gd perturb {it}]")

        it += 1

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing based permutation & mass transfer.
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_obj = best_obj
    temperature = cur_obj * 0.02    # initial temperature
    cooling = 0.9997                # exponential cooling factor
    attempts = 0

    while time.time() < deadline:
        attempts += 1
        # ---- pick a move ----
        if rng.random() < 0.25:                # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                   # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-15:
                continue
            # transfer a fraction (up to 5 %) of source mass
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto simplex (numerical safety)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # evaluate true objective quickly (no clamping needed)
        new_obj = _max_self_convolution(new_a, size_self)
        new_obj = 2.0 * n * new_obj / (new_a.sum() ** 2)

        # ---- Metropolis acceptance ----
        if new_obj < cur_obj - 1e-12:
            cur_a = new_a
            cur_obj = new_obj
            if new_obj < best_obj - 1e-12:
                best_obj = new_obj
                best_a = new_a.copy()
                print(f"[refine] new best = {best_obj:.6f} (attempt {attempts})")
        else:
            # accept worse move with probability exp(-Δ/T)
            prob = np.exp(-(new_obj - cur_obj) / max(temperature, 1e-12))
            if rng.random() < prob:
                cur_a = new_a
                cur_obj = new_obj

        # cool down
        temperature *= cooling
        # keep temperature from collapsing to 0 too fast
        if temperature < 1e-8:
            temperature = 1e-8

    # --------------------------------------------------------------
    # Final evaluation via the provided evaluate_sequence.
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_a.tolist()
