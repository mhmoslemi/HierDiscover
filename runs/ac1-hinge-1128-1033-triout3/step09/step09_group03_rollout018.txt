    start = time.time()
    deadline = start + budget_s - 0.5

    a_best = np.full(n, 1.0 / n)
    best_val = _objective_value(a_best)

    # -- Phase 0: parametric sweep
    # Exponential r
    for r in np.linspace(0.90, 0.99999, 60):  # 60 values
        if time.time() >= deadline: break
        i = np.arange(n)
        a = r**i + r**(n-1-i)
        a = a / a.sum()
        val = _objective_value(a)
        if val < best_val:

        val = _objective_value(a)

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

# convolution utilities
def _full_convolution(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _conv_fft(x, y):
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _objective_value(a):
    # assumes a already normalized to sum = 1
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())

# projection
def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# initial families
def _initial_candidate(n, rng):
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0) or np.sum(a) <= 0:
            return
        a = a / a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # Uniform
    _try(np.ones(n))

    # Exponential family (U‑shaped)
    for r in np.linspace(0.90, 0.99999, 80):
        i = np.arange(n, dtype=np.float64)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # Power‑law family a_i = ((i+1)*(n-i))**p
    for p in np.linspace(-0.05, -2.0, 60):
        i = np.arange(n, dtype=np.float64)
        prod = (i + 1) * (n - i)
        a = prod ** p
        _try(a)

    # Dirichlet draws with varied concentration
    for alpha in [0.01, 0.03, 0.07, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(20):
            a = rng.dirichlet(np.full(n, alpha))
            _try(a)

    return best_a, best_val

# Adam on smooth surrogate
def _adam_phase(a0, n, start, deadline, best_a, best_val, target_val=1.503):
    w = np.log(a0 + 1e-12)   # logits
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.003     # more aggressive schedule
    eps_start, eps_end = 0.2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 30000

    it = 0
    while time.time() < deadline:
        # Bring logits to simplex (softmax)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            #print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()

        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.004, size=n)

        if no_improve > IMPROVE_LIMIT:
            #print(f"[Adam {it}] restart")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val

# Simulated‑annealing hill‑climbing
def _simulated_annealing_phase(a0, n, start, deadline,
                               best_a, best_val,
                               rng, target_val=1.503):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    T0 = cur_val * 0.025                # a bit hotter
    max_frac_start, max_frac_end = 0.45, 0.001
    IMPROVE_LIMIT = 500_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)      # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        k_max = int(np.argmax(b))

        # compute contribution to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0.0:
            probs_src = contrib / total
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # choose a target opposite to the source
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # incremental update of convolution
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                #print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # moderate perturbation
            #print(f"[SA {it}] perturb")
            noise = rng.normal(scale=0.012, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val

# Exact sub‑gradient descent (projected)
def _subgradient_phase(a0, n, start, deadline,
                       best_a, best_val,
                       target_val=1.503):
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.6 / n          # a bit larger than before
    eta = eta0
    decay = 0.9985
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            #print(f"[SGD {it}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        jdx = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[jdx]

        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # add a small jitter and reset step size
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            #print(f"[SGD {it}] perturb")

        it += 1

    return best_local_a, best_local_val

def _mix_with_uniform(a, n):
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Heuristic optimisation for the min‑max self‑convolution problem.
    The routine consists of a broad parametric sweep, a short Adam phase,
    an aggressive simulated‑annealing phase and a final projected
    sub‑gradient polishing step.  It respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – a compromise between expressive power and speed
    n = 9000

    # ----------------------------------------------------------------------
    # Phase 0 – a large random / parametric pre‑search
    # ----------------------------------------------------------------------
    a0, best_val = _initial_candidate(n, rng)
    best_a = a0.copy()
    # ----------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈ 12 % of the budget)
    # ----------------------------------------------------------------------
    deadline_a = start_time + 0.12 * (deadline - start_time)
    best_a, best_val = _adam_phase(best_a, n, start=start_time,
                                   deadline=deadline_a,
                                   best_a=best_a, best_val=best_val)

    # ----------------------------------------------------------------------
    # Phase B – Simulated‑annealing (≈ 68 % of the budget)
    # ----------------------------------------------------------------------
    deadline_b = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _simulated_annealing_phase(best_a, n,
                                                  start=deadline_a,
                                                  deadline=deadline_b,
                                                  best_a=best_a,
                                                  best_val=best_val,
                                                  rng=rng)

    # ----------------------------------------------------------------------
    # Phase C – Exact projected sub‑gradient descent (remaining time)
    # ----------------------------------------------------------------------
    best_a, best_val = _subgradient_phase(best_a, n,
                                          start=deadline_b,
                                          deadline=deadline,
                                          best_a=best_a,
                                          best_val=best_val)

    # ----------------------------------------------------------------------
    # Tiny uniform mixing – often yields a marginal improvement
    # ----------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ----------------------------------------------------------------------
    # Final sanity check (the harness will also call evaluate_sequence)
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()

Heuristic optimisation for the min‑max self‑convolution problem.

The routine consists of:
  1. A broad parametric sweep (exponential, power‑law and Dirichlet families)

  2. A short Adam phase on a log‑sum‑exp surrogate of the max‑convolution.

  4. Projected sub‑gradient descent for a final polishing pass.
  5. A tiny uniform mixing step (often yields a marginal gain).

The whole procedure respects the 1000 s time budget and returns a list of
non‑negative floats that (approximately) minimises `evaluate_sequence`.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (exact, assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto {w ≥ 0, Σ w = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – expansive parametric sweep (exponential, power‑law, Dirichlet)
# ----------------------------------------------------------------------
def _initial_candidate(n: int, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Return the best vector among a wide set of hand‑crafted families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0) or a.sum() <= 0:
            return
        a = a / a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform (baseline)
    _try(np.ones(n))

    # --- exponential U‑shaped families ---
    for r in np.linspace(0.90, 0.99999, 80):
        i = np.arange(n, dtype=np.float64)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # --- power‑law families a_i = ((i+1)*(n-i))**p, p < 0 ---
    for p in np.linspace(-0.05, -2.0, 60):
        i = np.arange(n, dtype=np.float64)
        prod = (i + 1) * (n - i)
        a = prod ** p
        _try(a)

    # --- Dirichlet draws with many concentration parameters ---
    for alpha in [0.01, 0.03, 0.07, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(20):
            a = rng.dirichlet(np.full(n, alpha))
            _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp approximation of the max‑convolution."""
    w = np.log(a0 + 1e-12)                      # logits → simplex via soft‑max
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.003              # more aggressive learning‑rate schedule
    eps_start, eps_end = 0.2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 30000

    it = 0
    while time.time() < deadline:
        # current point on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑ p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)               # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # tiny jitter to escape flat regions
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.004, size=n)

        if no_improve > IMPROVE_LIMIT:
            # restart from a fresh random log‑vector
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Classic hill‑climbing with a temperature‑controlled acceptance
    probability for uphill moves.  The full convolution is updated
    incrementally in O(n) per trial move.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    T0 = cur_val * 0.025        # a bit hotter than the original
    max_frac_start, max_frac_end = 0.45, 0.001
    IMPROVE_LIMIT = 500_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)          # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # --- source selection (probability proportional to contribution) ---
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # --- target selection (prefer indices that contribute little) ---
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----- apply the move (incremental convolution update) -----
        b_old = b.copy()               # cheap O(n) copy for easy revert
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                if best_local_val <= target_val:
                    break
        else:
            # revert everything
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the state a little and continue
            noise = rng.normal(scale=0.012, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase C – Exact projected sub‑gradient descent (classic polishing)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.6 / n            # initial step size (≈ half the average entry)
    eta = eta0
    decay = 0.9985
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        jdx = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[jdx]

        # step & projection onto the simplex
        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # small perturbation and reset step size
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: mixing a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α values."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step function under the 1000 s time budget.
    Returns a list of non‑negative floats that (approximately) minimise
    `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a problem size – 9000 offers a good trade‑off.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # Phase 0 – broad parametric sweep (very cheap compared to the budget)
    # ------------------------------------------------------------------
    a0, best_val = _initial_candidate(n, rng)
    best_a = a0.copy()

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈12 % of the budget)
    # ------------------------------------------------------------------
    deadline_a = start_time + 0.12 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        best_a, n,
        start=start_time,
        deadline=deadline_a,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (≈68 % of the budget)
    # ------------------------------------------------------------------
    deadline_b = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _simulated_annealing_phase(
        best_a, n,
        start=deadline_a,
        deadline=deadline_b,
        best_a=best_a,
        best_val=best_val,
        rng=rng,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Phase C – Exact sub‑gradient descent for polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=deadline_b,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Final tiny uniform mixing (often gives a marginal improvement)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
