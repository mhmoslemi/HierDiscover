"""
propose_candidate
=================

A two‑phase stochastic optimiser for the “step‑function” problem.

Phase 0 – cheap exploration
    * Interpolate a previously stored solution (if available).
    * Sample many Beta‑shaped profiles a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} with
      log‑uniform p,q and keep the best.
    * Add a handful of Dirichlet draws for diversity.

Phase 1 – projected sub‑gradient descent
    * The exact objective is  f(a)=2·n·max_k (a∗a)_k   (Σa_i = 1, a_i≥0).
    * For the currently maximal convolution index k* the sub‑gradient is
          ∂f/∂a_i = 4·n·a_{k*‑i}   (if 0≤k*‑i<n, else 0).
    * We move a ← a – η·∂f, then Euclidean‑project onto the probability
      simplex.  The step size η decreases linearly from 2·10⁻² to 10⁻⁶.
    * If no improvement is seen for a few hundred iterations we
      perturb the best known point and continue.

All heavy operations (convolution, projection) use fast FFT‑based code.
The routine respects the 1000 s budget, reports intermediate best
values and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution (FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective  f(a) = 2·n·max(a∗a)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random Beta‑family candidate (cheap)
# ----------------------------------------------------------------------
def _beta_random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a single Beta‑shaped sequence a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    # log‑uniform p,q in a reasonably wide range
    logp = rng.uniform(-4.0, 4.0)
    logq = rng.uniform(-4.0, 4.0)
    p = np.exp(logp)
    q = np.exp(logq)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    if not np.all(np.isfinite(a)) or a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the time budget.
    Returns the best sequence (as a Python list) found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # problem size – a compromise between expressivity and speed
    n = 2000

    # ------------------------------------------------------------------
    # Phase 0 – cheap initialisation
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # uniform start
    best_val = _objective(best_a)
    print(f"[init] uniform start, value={best_val:.6f}")

    # 0.1 try to reuse a previously stored solution (if the variable exists)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]

        old_x = np.linspace(0.0, 1.0, old_len)

        cand_sum = cand.sum()
        if cand_sum > 0:

        if a is None:

        val = _objective(a)

    print(f"[init] after Beta sweep, best value={best_val:.6f}")

        val = _objective(a)
        if val < best_val:

    a = best_a.copy()
