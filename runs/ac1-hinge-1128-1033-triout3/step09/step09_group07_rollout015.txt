def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _full_convolution(x):

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear (zero‑padded) self‑convolution of a real vector."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative rounding errors can appear – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σₖ wₖ·(a∗a)ₖ w.r.t. `a`.  Implements
    gᵢ = 2·∑ⱼ aⱼ·w_{i+j}  via one FFT convolution.
    """
    n = a.shape[0]
    wlen = w_rev.shape[0]                # = 2·n‑1
    conv_len = n + wlen - 1              # = 3·n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = wlen - 1                     # index where the valid segment begins
    seg = conv[start:start + n]          # length n
    return 2.0 * seg


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values on the left side and the smallest on the right."""
    n = a.shape[0]
    s = np.sort(a)                       # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]                 # large → left
        b[-1 - i] = s[i]                 # small → right
    if n % 2:
        b[half] = s[half]
    return b


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Uses the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few quick random restarts to avoid a pathological start.
    # ------------------------------------------------------------------
    for tr in range(5):
        if time.time() > deadline - 150:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed exponential‑mirror descent.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    max_iter = 200_000
    alpha0 = 0.5
    beta_start = 1.0
    beta_end = 2000.0
    beta_ramp_iters = 50_000          # linear increase of β over this many iterations
    last_improve_it = 0

    while time.time() < deadline - 30 and it < max_iter:
        # (a ∗ a)
        conv = _full_convolution(a)
        max_conv = conv.max()
        delta = conv - max_conv                     # ≤ 0
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)
        w = np.exp(beta * delta)                    # soft‑max weights (max weight = 1)

        # gradient of the weighted sum
        g = _gradient_weighted(a, w[::-1])

        # step‑size (slow decay)
        alpha = alpha0 / (1.0 + it * 1e-4)

        # multiplicative mirror‑descent update
        delta_exp = -alpha * g
        delta_exp = np.clip(delta_exp, -20.0, 20.0)   # avoid overflow
        a = a * np.exp(delta_exp)

        # keep positivity and re‑normalise onto the simplex
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # ------------------------------------------------------------------
        # Logging, occasional re‑ordering and stagnation handling.
        # ------------------------------------------------------------------
        if it % 500 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                last_improve_it = it
                print(f"[gd iter {it}] new best = {best_val:.6f}")

            # Try the anti‑symmetric re‑ordering every 5 000 iterations
            if it % 5000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    last_improve_it = it
                    print(f"[gd reorder {it}] improved to {best_val:.6f}")

        # If we have seen no improvement for a while, inject fresh randomness
        if (it - last_improve_it) > 20_000:
            noise = rng.exponential(scale=1.0, size=n)
            noise /= noise.sum()
            a = 0.7 * a + 0.3 * noise
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            last_improve_it = it
            print(f"[gd perturbed at {it}]")

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – fine sub‑gradient descent on the *exact* maximum.
    # ------------------------------------------------------------------
    a = best_a.copy()
    it2 = 0
    max_iter2 = 150_000
    eta0 = 0.1
    eps_rel = 1e-9
    last_improve2 = 0

    while time.time() < deadline - 10 and it2 < max_iter2:
        conv = _full_convolution(a)
        max_conv = conv.max()
        # active set: all entries whose value is within eps of the maximum
        eps = max(eps_rel * max_conv, 1e-12)
        mask = (conv >= max_conv - eps).astype(float)   # 1 for active positions
        g = _gradient_weighted(a, mask[::-1])          # sub‑gradient (average over active)

        eta = eta0 / np.sqrt(it2 + 1)                  # classical diminishing step size
        a_new = a - eta * g
        a_new = _project_onto_simplex(a_new)

        val = evaluate_sequence(a_new.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            last_improve2 = it2
            print(f"[subgd {it2}] new best = {best_val:.6f}")
        else:
            a = a_new

        # occasional re‑ordering
        if it2 % 5000 == 0:
            a_ord = _anti_symmetric_reorder(a)
            val_ord = evaluate_sequence(a_ord.tolist())
            if val_ord < best_val - 1e-12:
                best_val = val_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                print(f"[subgd reorder {it2}] improved to {best_val:.6f}")

        # shake the solution if it stalls
        if (it2 - last_improve2) > 15_000:
            # pick a relatively large entry and a relatively small one
            idx_large = np.argsort(-a)[:max(1, n // 200)]
            idx_small = np.argsort(a)[:max(1, n // 200)]
            i = rng.choice(idx_large)
            j = rng.choice(idx_small)
            delta = min(a[i], 0.05 * a[i]) * rng.random()
            a[i] -= delta
            a[j] += delta
            a = _project_onto_simplex(a)
            last_improve2 = it2
            print(f"[subgd shake {it2}]")

        it2 += 1

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing refinement.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    while time.time() < deadline:
        if rng.random() < 0.3:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.03 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = _project_onto_simplex(new_a)
        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            best_a = new_a.copy()
            best_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_a.tolist()
