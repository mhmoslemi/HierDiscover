def _smoothed_descent_custom(a_start, deadline, total_iters=6000):
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = 2 * n * _full_convolution(a).max()   # initial objective

    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    it_per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        eta = 0.7
        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:
                break

            conv = _full_convolution(a)
            max_conv = conv.max()

            shift = max_conv
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()

            grad = 2.0 * _gradient_weighted_peaks(a, p)

            delta = -eta * grad
            delta = np.clip(delta, -20.0, 0.0)
            a = a * np.exp(delta)
            a = _project_simplex(a)

            if it % 20 == 0:
                cur_val = 2 * n * max_conv
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.4f} it {it}] new best = {best_val:.6f}")

                if it % 200 == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    conv_ord = _full_convolution(a_ord)
                    val_ord = 2 * n * conv_ord.max()
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[reorder eps {eps:.4f} it {it}] improved to {best_val:.6f}")

            eta *= 0.9995

        a = _anti_symmetric_reorder(a)

    return best_a, best_val

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initial seeds** – a few deterministic and random shapes (uniform,
   sinusoidal, exponential decays, previous solution).  All seeds are
   projected onto the probability simplex and anti‑symmetrically
   reordered (largest ↔ smallest) because this never increases the
   convolution peak.

2. **Smoothed‑max descent** – minimise the log‑sum‑exp surrogate of
   ``max(conv)`` using a multiplicative mirror‑descent step.
   A long schedule of smoothing parameters
   ``ε ∈ [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]``
   is used, with many iterations (≈ 8 000 FFT steps) so that the
   algorithm converges close to the true‐max objective.

3. **Targeted local refinement** – after each descent the index of the
   current convolution maximum is identified.  A small amount of mass is
   moved from the pair that contributes most to that maximum to a
   location whose partner value in the same lag is tiny.  The move is
   kept only if the true objective (computed via the cheap FFT‑based
   convolution) improves.

4. **Fast hill‑climb** – a final light stochastic local search (random
   swaps / tiny transfers) using the same cheap internal evaluation.
   ``evaluate_sequence`` is called only once at the very end.

The routine respects a 1000 s wall‑clock budget and returns the best
sequence found (a list of floats).  All numbers are non‑negative and are
implicitly normalised to sum = 1, which makes the objective
scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted‑soft‑max surrogate.
    For the surrogate  f_ε(a)= (1/ε)·log Σ_k exp(ε·c_k)  (c_k = (a*a)[k]),
    the gradient w.r.t. a is 2·(a * p) where * denotes the convolution‑like
    operation implemented below.
    """
    n = a.shape[0]
    # reverse the weight vector as required by the derivation
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # slice the part that corresponds to the true gradient
    seg = conv[n - 1 : 2 * n - 1]
    return seg[::-1]                       # reverse back


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { x ≥ 0 , Σ x = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest ones.
    Guarantees that the convolution peak at lag n‑1 cannot increase.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Smoothed‑max descent (multiplicative mirror‑descent)
# ----------------------------------------------------------------------
def _smoothed_descent_custom(a_start: np.ndarray,
                             deadline: float,
                             total_iters: int = 8000):
    """
    Minimise a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector encountered and the corresponding internal
    objective 2·n·max_conv (i.e. the value *before* the final clipping
    performed by ``evaluate_sequence``).
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a).max()
    print(f"[descent init] objective ≈ {best_val:.6f}")

    # smoothing schedule – large → small
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01,
                    0.005, 0.002, 0.001]

    it_per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        eta = 0.7                     # initial step size for this ε‑stage

        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:
                break

            conv = _full_convolution(a)          # length 2n‑1
            max_conv = conv.max()

            # stable softmax weights
            shift = max_conv
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()        # Σp = 1, p≥0

            # gradient of the smooth surrogate
            grad = 2.0 * _gradient_weighted_peaks(a, p)

            # multiplicative mirror‑descent step
            delta = -eta * grad
            delta = np.clip(delta, -20.0, 0.0)    # avoid overflow/underflow
            a = a * np.exp(delta)
            a = _project_simplex(a)               # enforce sum = 1, a≥0

            # occasional checks
            if it % 20 == 0:
                cur_val = 2.0 * n * max_conv
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.4f} it {it}] new best ≈ {best_val:.6f}")

                if it % 200 == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    val_ord = 2.0 * n * _full_convolution(a_ord).max()
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[reorder eps {eps:.4f} it {it}] improved to {best_val:.6f}")

            eta *= 0.9995        # slow decay for stability at small ε

        # final re‑ordering for this ε‑stage
        a = _anti_symmetric_reorder(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted local refinement (directly reduces the true max convolution)
# ----------------------------------------------------------------------
def _targeted_refine(a: np.ndarray,
                     cur_val: float,
                     deadline: float):
    """
    Greedy reduction of the peak of a*a.
    Repeatedly moves a small amount of mass away from the pair that
    contributes most to the current maximal convolution value.
    Uses only FFT‑based convolutions, so it is cheap and does not call
    ``evaluate_sequence`` inside the loop.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        conv = _full_convolution(best_a)
        max_conv = conv.max()
        k = int(np.argmax(conv))                 # lag where the maximum occurs

        # Range of i for which a_i * a_{k-i} participates in conv[k]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        products = best_a[i_range] * best_a[j_range]

        # index of the largest contributing pair
        idx_max_local = int(np.argmax(products))
        i_max = int(i_range[idx_max_local])
        j_max = int(j_range[idx_max_local])

        # Find a partner index i2 whose counterpart a_{k-i2} is minimal
        partner_vals = np.full(n, np.inf)
        valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
        partner_vals[valid] = best_a[k - np.arange(n)[valid]]
        partner_vals[i_max] = np.inf            # exclude the current worst i

        i2 = int(np.argmin(partner_vals))
        if partner_vals[i2] == np.inf:
            break                               # no eligible move found

        # Transfer a modest fraction of mass from i_max to i2
        delta = min(best_a[i_max] * 0.05, 0.02)  # keep steps small
        if delta <= 0:
            break

        new_a = best_a.copy()
        new_a[i_max] -= delta
        new_a[i2] += delta

        # re‑normalise (projection onto the simplex)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        new_val = 2.0 * n * _full_convolution(new_a).max()
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = new_a
            print(f"[targeted] improved to {best_val:.6f}")

        # occasional random perturbation to escape tiny plateaus
        if rng.random() < 0.01:
            # random tiny shuffle
            p = rng.integers(0, n, size=2)
            new_a[p[0]], new_a[p[1]] = new_a[p[1]], new_a[p[0]]
            new_a = _project_simplex(new_a)

    return best_a, best_val

# Fast hill‑climbing (uses internal convolution, no costly evaluate calls)

def _hill_climb_refine_fast(a: np.ndarray,
                            cur_val: float,
                            deadline: float,
                            n: int):
    """
    Very light stochastic local search: random swaps or tiny mass
    transfers, kept only if they strictly improve the internal objective
    2·n·max_conv.  ``evaluate_sequence`` is called only once at the end.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline - 1.0:
        if rng.random() < 0.3:
            # random swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # tiny transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-12:
                continue
            delta = best_a[i] * 0.05 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        val = 2.0 * n * _full_convolution(new_a).max()
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] new best = {best_val:.6f}")

    return best_a, best_val

        # --- targeted refinement --------------------------------------------------
        if time.time() < deadline - 10:
            a_ref, val_ref = _targeted_refine(best_a, best_val, deadline)
            if val_ref < best_val:
                best_val = val_ref
                best_a = a_ref.copy()
                print(f"[global] targeted improved to {best_val:.6f}")

        # early exit if we have already crossed the target bound
        if best_val <= 1.503:
            print("[global] target bound reached, exiting early")
            break

    # ----------------------------------------------------------------------
    # Final fast hill‑climb (uses internal evaluation only)
    # ----------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5:
        best_a, best_val = _hill_climb_refine_fast(best_a, best_val,
                                                   deadline, n)

    # ----------------------------------------------------------------------
    # Final evaluation via the provided ``evaluate_sequence`` (once)
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
```
