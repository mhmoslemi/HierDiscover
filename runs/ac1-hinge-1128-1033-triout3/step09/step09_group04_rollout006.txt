assistantfinal**Solution Overview**

The task is to minimise  

\[
F(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a` is a non‑negative step function (a list of heights) and  
`(a*a)` denotes the full linear convolution of `a` with itself.  
After normalising `\sum_i a_i = 1` the denominator is constant, therefore we
only have to minimise the maximal convolution entry  
`max_b = max_k (a*a)_k`.

The code below implements a *continuous* optimisation that works directly on
the probability simplex:

1. **Soft‑max (smooth‑max) surrogate** – instead of the non‑differentiable
   `max`, we minimise the smooth approximation  

   \[
   S_\beta(a)=\log\!\Bigl(\sum_k e^{\beta\,(a*a)_k}\Bigr)/\beta,
   \]

   whose gradient is a weighted combination of all convolution entries.
2. **Exponentiated‑gradient (multiplicative) update** – the gradient is
   applied in log‑space (`a_i ← a_i·exp(-η·∇_i)`), which guarantees
   non‑negativity and automatically stays on the simplex after normalisation.
3. **Adaptive schedules** –  
   *β* grows slowly (the surrogate becomes sharper) and the step size η
   decays from a fairly aggressive value to a very fine one.  
   If progress stalls a tiny Gaussian jitter is added.
4. **Weighted hill‑climb refinement** – after the gradient phase we look at
   the convolution entries that are within a tiny relative tolerance of the
   current maximum, compute for each index its *total contribution* to those
   peaks and move a small fraction of probability mass from the worst
   contributor to the best one.
5. **Random restarts** – the whole routine is repeated many times with
   randomly drawn Dirichlet initialisations (concentration drawn log‑uniformly);
   the best vector found over the whole budget is kept.
6. **Final uniform mixing** – a few convex combinations with the uniform
   vector are tried; this can occasionally smooth a remaining spike.

All heavy work (convolutions, correlations) is performed with FFTs, which
makes each iteration fast even for vectors of length 8000.  
The algorithm respects the 1000 s time limit, prints intermediate progress,
and finally returns a Python list of non‑negative heights.

```python
"""
propose_candidate – minimise evaluate_sequence by smooth‑max gradient descent
and a weighted hill‑climb.  The method works on the probability simplex,
uses FFT‑based convolutions, adaptive soft‑max temperature and step‑size,
and many cheap random restarts.  The final vector is returned as a plain
list of non‑negative floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# FFT‑based convolution helpers (full linear convolution)
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Return the full linear convolution a * a (length 2·len(a)‑1)
    using FFT.  a is assumed to be 1‑D real.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k‑i}   (i = 0 … n‑1).
    This is the cross‑correlation of a with the weight vector w.
    Both a (length n) and w (length 2·n‑1) are real.
    """
    n = a.shape[0]
    a_rev = a[::-1]                       # reversed a
    # convolution length = n + (2n‑1) – 1 = 3n‑2
    size = 1 << ((n + w.shape[0] - 1).bit_length())
    fa_rev = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv = np.fft.irfft(fa_rev * fw, size)
    start = n - 1
    return conv[start:start + n]          # central n entries


# ------------------------------------------------------------
# Local objective (sum of a is assumed to be 1)
# ------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by evaluate_sequence for a normalised vector."""
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ------------------------------------------------------------
# Smooth‑max surrogate and exponentiated‑gradient step
# ------------------------------------------------------------
def _smoothmax_step(a: np.ndarray,
                    n: int,
                    beta: float,
                    eta: float,
                    jitter: bool = False) -> np.ndarray:
    """
    Perform one multiplicative update using the gradient of the
    smooth‑max S_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k).
    """
    # 1. convolution
    b = _full_convolution(a)

    # 2. soft‑max weights (numerically stable)
    max_b = b.max()
    # shift to avoid overflow
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    w /= w_sum                         # normalise – only relative weights matter

    # 3. gradient of the surrogate:  ∇_i S = 2·Σ_k w_k·a_{k‑i}
    grad = 2.0 * _correlation(a, w)

    # 4. multiplicative (exponentiated) update
    a_new = a * np.exp(-eta * grad)

    # optional jitter to escape plateaux
    if jitter:
        noise = np.random.normal(scale=0.006, size=a.shape)
        a_new = np.maximum(a_new + noise, 0.0)

    # renormalise to Σ a = 1
    tot = a_new.sum()
    if tot == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= tot
    return a_new


# ------------------------------------------------------------
# Weighted hill‑climb (uses a set of near‑max peaks)
# ------------------------------------------------------------
def _weighted_hillclimb(a: np.ndarray,
                        n: int,
                        start_frac: float = 0.30,
                        end_frac: float = 0.001,
                        rel_tol: float = 0.9995,
                        max_iters: int = 5000) -> np.ndarray:
    """
    Move a small fraction of mass from the index that contributes most to the
    current near‑max peaks to the index that contributes least.
    Only moves that strictly improve the true objective are kept.
    """
    cur = a.copy()
    best_val = _objective(cur, n)
    best = cur.copy()

    for it in range(max_iters):
        b = _full_convolution(cur)
        max_b = b.max()
        # indices whose value is within rel_tol of the maximum
        active = np.where(b >= rel_tol * max_b)[0]
        if active.size == 0:
            break

        # weight vector for the active set (uniform over the set)
        w_active = np.zeros_like(b)
        w_active[active] = 1.0 / active.size

        # contribution of each coordinate to the active peaks
        contrib = 2.0 * _correlation(cur, w_active)

        src = int(np.argmax(contrib))
        tgt = int(np.argmin(contrib))
        if src == tgt:
            break

        # annealed transfer fraction
        progress = it / max_iters
        frac = start_frac * (1.0 - progress) + end_frac * progress

        delta = cur[src] * frac
        if delta <= 0.0:
            break

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        # re‑normalise (should stay at 1, but numerical errors may appear)
        cur /= cur.sum()

        val = _objective(cur, n)
        if val < best_val - 1e-15:
            best_val, best = val, cur.copy()
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta
            cur /= cur.sum()    # keep normalisation

    return best


# ------------------------------------------------------------
# Tiny uniform mixing
# ------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> np.ndarray:
    """Try a few convex combinations (1‑α)·a + α·uniform."""
    best = a.copy()
    best_val = _objective(best, n)
    uniform = np.full(n, 1.0 / n)

    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best = val, cand
    return best


# ------------------------------------------------------------
# Single optimisation run (one random restart)
# ------------------------------------------------------------
def _optimise_one(rng: np.random.Generator,
                  n: int,
                  start_time: float,
                  deadline: float,
                  init_vec: np.ndarray = None) -> tuple[np.ndarray, float]:
    """
    Return the best vector found in a time slice.
    If init_vec is supplied it is used; otherwise a random Dirichlet draw
    (with a log‑uniform concentration) is generated.
    """
    # ----- initial vector -------------------------------------------------
    if init_vec is None:
        # concentration α ∈ [10⁻², 10²] (log‑uniform)
        alpha = 10.0 ** rng.uniform(-2, 2)
        a = rng.dirichlet(alpha * np.ones(n))
    else:
        a = init_vec.copy()
        a = np.maximum(a, 0.0)
        a /= a.sum()

    best_a = a.copy()
    best_val = _objective(best_a, n)

    # ----- schedules ------------------------------------------------------
    beta = 5.0          # start temperature for smooth‑max
    eta_start = 0.12   # aggressive start step‑size
    eta_end = 5e-4     # fine final step‑size

    it = 0
    stall_counter = 0
    IMPATIENT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress for schedules
        progress = (now - start_time) / (deadline - start_time)
        beta = min(beta * 1.01, 200.0)          # slowly sharpen the surrogate
        eta = (1.0 - progress) * eta_start + progress * eta_end

        # one gradient step
        jitter = stall_counter > IMPATIENT
        a = _smoothmax_step(a, n, beta, eta, jitter=jitter)

        # evaluate true objective
        val = _objective(a, n)

        if val < best_val - 1e-15:
            best_val, best_a = val, a.copy()
            stall_counter = 0
            print(f"[Grad {it:7d}] new best = {best_val:.6f}")
        else:
            stall_counter += 1

        it += 1

    # ----- deterministic refinement ----------------------------------------
    best_a = _weighted_hillclimb(best_a, n, max_iters=4000)
    best_a = _mix_with_uniform(best_a, n)
    best_val = _objective(best_a, n)
    return best_a, best_val
```
