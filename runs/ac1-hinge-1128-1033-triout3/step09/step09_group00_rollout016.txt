"""
Greedy sub‑gradient descent for the max‑auto‑convolution problem.

- Initialise a candidate with a quick sweep of the Beta‑family,
  a few random Beta draws and a handful of Dirichlet samples.
- Repeatedly:
    * compute the full self‑convolution b = a∗a,
    * locate all indices where b attains its maximum,
    * obtain the sub‑gradient g of max(b) efficiently
      via a convolution with the binary mask of those indices,
    * move a small fraction of mass from the index with the
      largest gradient (most responsible for the peak) to the index
      with the smallest gradient (least responsible),
    * accept the move only if the objective improves,
      otherwise shrink the transferred fraction or perform a tiny
      random jitter.
- If the algorithm stagnates for a while a small random perturbation
  of the best‑so‑far vector is applied.
- The procedure runs until the 1000 s budget expires (a short safety
  margin is kept).  The final vector is returned as a list; it is
  already normalised to sum to 1, which makes ``evaluate_sequence``
  compute exactly ``2·n·max(conv(a,a))``.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Convolution helpers (FFT, full linear convolution)
# --------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()            # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# --------------------------------------------------------------
# Simple objective (sum of a is 1 ⇒ evaluate = 2·n·max(b))
# --------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# --------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Families of candidate shapes (cheap to evaluate)
# --------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    best_a = None
    best_val = np.inf
    # grid on p,q (including very small values)
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _beta_random_search(n: int, trials: int = 250) -> tuple[np.ndarray, float]:
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    rng = np.random.default_rng()
    for _ in range(trials):
        logp = rng.uniform(-6.9, 2.3)   # ln(0.001) .. ln(10)
        logq = rng.uniform(-6.9, 2.3)
        p = np.exp(logp)
        q = np.exp(logq)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# --------------------------------------------------------------
# Greedy coordinate descent (main optimisation loop)
# --------------------------------------------------------------
def _greedy_descent(a: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float,
                    target_val: float = 1.503,
                    stagnation_limit: int = 15000) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a chunk of mass from the index with the largest
    sub‑gradient (most responsible for the current peak) towards the
    index with the smallest sub‑gradient.
    """
    cur_a = a.copy()
    cur_val = _objective(cur_a)
    no_improve = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # full convolution and mask of maximal entries
        b = _full_convolution(cur_a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        # update global best
        if cur_val < best_val - 1e-13:
            best_val, best_a = cur_val, cur_a.copy()
            print(f"[Greedy] new best = {best_val:.6f}")
            no_improve = 0
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # binary mask of the active constraints
        mask = (b == max_b).astype(float)

        # sub‑gradient of max(b) w.r.t. a:
        #   grad_j = 2 * Σ_{k: b_k=max} a_{k-j}
        # computed as a convolution of a with the reversed mask
        rev_mask = mask[::-1]
        conv = _conv_fft(cur_a, rev_mask)               # length 3n‑2
        segment = conv[n - 1:2 * n - 1]                 # n entries
        grad = 2.0 * segment[::-1]                     # reverse to match indices

        # source = index with largest gradient, target = smallest
        i_src = int(np.argmax(grad))
        i_tgt = int(np.argmin(grad))

        # schedule for transfer fraction (large early, tiny later)
        progress = (now - start_time) / (deadline - start_time)
        max_frac = 0.6 * (1.0 - progress) + 0.02 * progress

        improved = False
        # try a few decreasing fractions until an improvement occurs
        for frac in (max_frac, max_frac * 0.5, max_frac * 0.25,
                     max_frac * 0.125, max_frac * 0.0625):
            if cur_a[i_src] <= 0.0:
                break
            delta = cur_a[i_src] * frac
            if delta < 1e-14:
                continue
            a_candidate = cur_a.copy()
            a_candidate[i_src] -= delta
            a_candidate[i_tgt] += delta
            # feasibility is guaranteed (non‑negative, sum unchanged)
            new_val = _objective(a_candidate)
            if new_val < cur_val - 1e-12:
                cur_a = a_candidate
                cur_val = new_val
                improved = True
                break

        if not improved:
            # tiny random jitter to escape flat regions
            i_rand, j_rand = np.random.choice(n, 2, replace=False)
            if cur_a[i_rand] > 0:
                delta = cur_a[i_rand] * 0.02
                a_tmp = cur_a.copy()
                a_tmp[i_rand] -= delta
                a_tmp[j_rand] += delta
                tmp_val = _objective(a_tmp)
                if tmp_val < cur_val:
                    cur_a = a_tmp
                    cur_val = tmp_val
                    improved = True

        if not improved and no_improve > stagnation_limit:
            # restart from the best vector with a small random perturbation
            cur_a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            cur_a = np.maximum(cur_a + noise, 0.0)
            cur_a /= cur_a.sum()
            no_improve = 0
            print("[Greedy] stagnation – random perturbation")
    return best_a, best_val

# Main entry point required by the evaluation framework
# --------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # problem size – a compromise between expressivity and speed
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise a – cheap parametric families + random draws
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # uniform start (fallback)
    uniform = np.full(n, 1.0 / n)
    best_a, best_val = uniform, _objective(uniform)

    # Beta family grid search
    try:
        cand_a, cand_val = _beta_family_best(n)
        if cand_val < best_val:
            best_a, best_val = cand_a, cand_val
            print(f"[Init] beta‑grid improved value = {best_val:.6f}")
    except Exception as exc:
        print("[Init] beta‑grid failed:", exc)

    # Random Beta draws (more extensive than before)
    try:
        cand_a, cand_val = _beta_random_search(n, trials=400)
        if cand_val < best_val:
            best_a, best_val = cand_a, cand_val
            print(f"[Init] beta‑random improved value = {best_val:.6f}")
    except Exception as exc:
        print("[Init] beta‑random failed:", exc)

    # Edge‑heavy power‑law shapes
    i = np.arange(1, n + 1, dtype=float)
    for p in (0.2, 0.5, 0.8, 1.2, 2.0):
        a = (i ** (-p) + (n - i + 1) ** (-p))
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a.copy(), val
            print(f"[Init] edge‑power p={p:.2f} improved value = {best_val:.6f}")

    # Half‑sine shape (positive on whole interval)
    a = np.sin(np.pi * i / (n + 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = _objective(a)
    if val < best_val:
        best_a, best_val = a.copy(), val
        print(f"[Init] sine‑shape improved value = {best_val:.6f}")

    # A few Dirichlet draws for extra diversity
    for _ in range(6):
        cand = np.random.dirichlet(np.ones(n))
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand.copy(), val
            print(f"[Init] Dirichlet seed improved value = {best_val:.6f}")

    print(f"[Init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Greedy sub‑gradient descent (the core optimisation)
    # ------------------------------------------------------------------
    best_a, best_val = _greedy_descent(
        a=best_a,
        n=n,
        deadline=deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
        stagnation_limit=15000
    )

    # ------------------------------------------------------------------
    # 2️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
