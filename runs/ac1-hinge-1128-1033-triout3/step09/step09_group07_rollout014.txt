def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of a real vector `x`.
    Length of the result is 2·len(x)‑1.
    """
    n = x.shape[0]

    np.maximum(conv, 0.0, out=conv)      # eliminate tiny negatives
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of Duchi et al.
    """
    n = v.shape[0]
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]                # descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:

    This often spreads large masses to the extremes of the vector.

        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b

    cur_val = best_val
    refine_it = 0
    while time.time() < deadline:
        # pick operation
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.04 * rng.random()   # up to 4 % of the source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex (keeps sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            best_a = new_a.copy()
            best_val = val
            last_improve_it = it
            print(f"[refine {refine_it}] new best = {cur_val:.6f}")
        else:
            # occasional uphill acceptance (simulated annealing)
            T = 1e-4 * np.exp(-refine_it / 50000.0)
            if rng.random() < np.exp(-(val - cur_val) / max(T, 1e-12)):
                cur_a = new_a
                cur_val = val
        refine_it += 1

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_a.tolist()
