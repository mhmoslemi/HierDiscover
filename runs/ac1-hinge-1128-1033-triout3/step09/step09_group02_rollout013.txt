def _adam_optimize(a0, n, start_time, deadline, best_a, best_val):
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    while time.time() < deadline:
        # compute a via softmax
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()

        b = _full_convolution(a)
        cur_val = 2 * n * b.max()

        # track best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # print

        # check progress for schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        eps = 0.1 * (1e-12/0.1)**progress   # using geometric schedule: start 0.1 to 1e-12
        lr = 0.5 * (0.001/0.5)**progress   # start high 0.5 to low 0.001

        # compute softmax of (b - b_max)/eps
        b_max = b.max()
        shifted = (b - b_max) / eps
        weights = np.exp(shifted - np.max(shifted))  # numeric stable
        p = weights / weights.sum()

        # gradient
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]  # as earlier

        # backprop through softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w * grad_w)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.01:
            w += 0.01 * np.random.randn(n)

    return best_a, best_val
```

"""
Optimisation of a non‑negative step function under the evaluate_sequence
objective.  The algorithm proceeds in three stages:

1. **Broad parametric search** – many analytic families (exponential,
   power‑law, beta, sine‑powers, piece‑wise constant, random Dirichlet)
   are sampled on a fine grid.  The best candidate (normalised to sum = 1)
   is kept.

2. **Adam optimisation on a smooth log‑sum‑exp surrogate** – the
   max‑convolution is approximated by a soft‑max with a temperature that
   decays from 1e‑1 to 1e‑12.  Adam updates are performed on the logits
   (log‑softmax parametrisation) of the sequence.

3. **Targeted stochastic hill‑climbing** – small amounts of mass are
   moved from indices that contribute most to the current maximal
   convolution entry to indices that contribute least.  An incremental
   convolution update makes each move O(n).  Occasionally a noisy
   perturbation restarts the walk if progress stalls.

A final tiny mixing with the uniform distribution is performed,
after which the result is passed to the provided ``evaluate_sequence``
function and returned as a plain list of floats.
"""
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = rfft(a, size)
    conv = irfft(af * af, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = rfft(x, size)
    fy = rfft(y, size)
    conv = irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (for sequences normalised to sum=1)
# ----------------------------------------------------------------------
def _objective_normed(a: np.ndarray) -> float:
    """Exact objective for a vector with Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families – extensive sweep for a good starting point
# ----------------------------------------------------------------------
def _initial_candidate(n: int, rng: np.random.Generator):
    """Return a good initial probability vector found by a dense parametric sweep."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_normed(best_a)

    def _try_raw(v_raw):
        nonlocal best_a, best_val
        a = np.asarray(v_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0.0:

        a = a / s
        val = _objective_normed(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)                # 0 … n‑1
    i1 = i + 1.0                                   # 1 … n

    # 1. Exponential tails (both decreasing and symmetric)
    for r in np.logspace(np.log10(0.999), np.log10(0.9999999), 20):
        _try_raw(r ** i)                        # monotone decreasing
        _try_raw(r ** i + r ** (n - 1 - i))     # symmetric U‑shape

    # 2. Power‑law tails (wide range, fine granularity)
    for p in np.logspace(-1.5, 1.0, 30):            # 0.03 … 10

        _try_raw(i1 ** (-p) + (n - i) ** (-p))

            _try_raw(a)

# ----------------------------------------------------------------------
# Phase A – Adam on a smooth log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int,
                start: float, deadline: float,
                best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp approximation of the max‑convolution.
    """
    # logits → simplex via soft‑max
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    while time.time() < deadline:
        # probability vector on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_normed(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            print(f"[Adam] new best = {best_val:.6f}")

        # schedule
        prog = (time.time() - start) / (deadline - start)
        eps = 0.1 * (1e-12 / 0.1) ** prog          # 0.1 → 1e‑12
        lr = 0.5 * (0.001 / 0.5) ** prog           # 0.5 → 0.001

        # soft‑max over convolution entries
        b_full = _full_convolution(a)
        b_max = b_full.max()
        shifted = (b_full - b_max) / eps
        # numerical stabilisation
        shifted -= np.max(shifted)
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a 2·n·∑ p_k (a*a)_k = 4·n·conv(a, rev(p))
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)             # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.01, size=n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted stochastic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray, n: int,
                     start: float, deadline: float,
                     best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic moves that transfer a small amount of mass from
    indices that contribute most to the current maximal convolution entry
    to indices that contribute least.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = _objective_normed(a)

    # mass‑transfer schedule: start relatively aggressive, end very small
    max_frac_start, max_frac_end = 0.5, 0.0005
    stall_counter = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        k_max = int(np.argmax(b))

        # contributions of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick source index (biased by its contribution)
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-18:
            continue

        # pick a destination index (prefer indices that hardly affect the max entry)
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0.0 and np.random.rand() < 0.55:
            probs_j = w / w.sum()
            j = np.random.choice(n, p=probs_j)
        else:
            j = np.random.randint(0, n)

        # amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear time)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # acceptance – always accept improvement, otherwise accept with a tiny probability
        if new_val < cur_val - 1e-15 or np.random.rand() < 0.001:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            stall_counter = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
                if best_val <= 1.50:
                    break
        else:
            stall_counter += 1

        if stall_counter > IMPROVE_LIMIT:
            # perturb current best slightly to escape a plateau
            a = best_a.copy()
            a += np.random.normal(scale=0.015, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective_normed(a)
            stall_counter = 0
            print("[Hill] perturbing after stagnation")

    return best_a, best_val

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a, best_val = a, _objective_normed(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.007, 0.015, 0.025):

        cand /= cand.sum()
        val = _objective_normed(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
