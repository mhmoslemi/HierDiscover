"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm (≈ 1000 s budget)
--------------------------------
1. **Problem size** – try a moderately large vector (default 4000).  The
   objective 2·n·max(conv)/(Σa)² is (almost) scale‑invariant, so the
   exact length is not critical; 4000 gives a good trade‑off between
   expressive power and speed.

2. **Initialisation** – several restarts with simple seeds:
   uniform, exponential, sinusoid, and (if present) the previous best
   `height_sequence_1`.  All vectors are projected onto the probability
   simplex (Σa = 1, a ≥ 0).

3. **Smoothed‑max descent** – minimise the smooth log‑sum‑exp surrogate
   fε(a) = (1/ε)·log Σ exp(ε·(a*a)[k]) using a multiplicative mirror‑
   descent step (a ← a·exp(−η·∇fε)).  The gradient of the surrogate is
   obtained with a single FFT‑based convolution.  The smoothing
   parameter ε is decreased from 0.5 down to 0.001, giving a coarse‑to‑fine
   optimisation.  After each ε‑stage an *interleaved* re‑ordering of the
   coefficients (largest ↔ smallest alternating at the two ends) is
   applied – this spreads large values and often reduces the true max‑
   convolution beyond the simple anti‑symmetric ordering used previously.

4. **Exact‑max sub‑gradient descent** – once the smooth phases finish,
   the true maximum index is located and a sub‑gradient of that peak
   (g_i = 2·a_{k−i}) is used for a few thousand mirror‑descent steps.
   This hones the worst convolution value directly.

5. **Hill‑climbing refinement** – cheap local moves (random swaps,
   tiny mass transfers) are accepted only if they strictly improve the
   true objective.  The search is performed until the time budget is
   exhausted.

6. **Book‑keeping** – the best vector seen over all restarts and all
   phases is tracked; the final value is obtained with the official
   `evaluate_sequence` call and returned as a plain list.

All heavy work (FFT‑based gradients, surrogate evaluation) is done
without calling `evaluate_sequence`; the true objective is queried only
occasionally to avoid unnecessary work.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·(a*a)[k] w.r.t. a (ignores factor 2).

    Result: g_i = Σ_k p_k·a_{k-i}   (out‑of‑range indices are zero).
    Computed with one FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]                      # reverse the weight vector
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # indices n‑1 … 2n‑2 correspond to the required slice
    seg = conv[n - 1 : 2 * n - 1]
    # reverse back to obtain the gradient
    return seg[::-1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _interleaved_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest, second largest, … alternating at the left and
    right ends of the array.  This spreads large values and tends to
    reduce the maximal autocorrelation peak.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]          # descending
    out = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for v in sorted_desc:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def _compute_objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a)/ (sum a)^2 (sum should be 1)."""
    conv = _full_convolution(a)
    max_b = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Smoothed‑max descent (softmax surrogate)
# ----------------------------------------------------------------------
def _smoothed_descent(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _compute_objective(a)
    print(f"[smoothed] start objective = {best_val:.6f}")

    # Smoothing schedule (coarse → fine)
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    total_iters = 15000                     # total inner steps across all eps
    it_per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        eta = 0.7
        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:
                break

            # --- surrogate gradient ---
            conv = _full_convolution(a)
            max_conv = conv.max()
            # stable softmax
            shift = max_conv
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()
            grad = 2.0 * _gradient_weighted_peaks(a, p)   # shape (n,)

            # --- mirror‑descent update (multiplicative) ---
            delta = -eta * grad
            delta = np.clip(delta, -20.0, 0.0)            # avoid overflow
            a = a * np.exp(delta)
            a = _project_simplex(a)

            # --- occasional re‑ordering & true evaluation ---
            if (it % 25) == 0:
                a_ord = _interleaved_reorder(a)
                obj_ord = _compute_objective(a_ord)
                if obj_ord < best_val - 1e-12:
                    best_val = obj_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[reorder eps {eps:.3f} it {it}] improved = {best_val:.6f}")

                if (it % 200) == 0:
                    # true objective via the provided evaluator (infrequent)
                    true_val = evaluate_sequence(a.tolist())
                    if true_val < best_val - 1e-12:
                        best_val = true_val
                        best_a = a.copy()
                        print(f"[true eps {eps:.3f} it {it}] new best = {best_val:.6f}")

            # --- step‑size decay ---
            eta *= 0.9995

        # final re‑ordering for this ε‑stage
        a = _interleaved_reorder(a)

        # one more true check after the stage
        cur_true = evaluate_sequence(a.tolist())
        if cur_true < best_val - 1e-12:
            best_val = cur_true
            best_a = a.copy()
            print(f"[stage end eps {eps:.3f}] best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution peak
# ----------------------------------------------------------------------
def _max_subgradient_descent(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _compute_objective(a)

    max_iters = 5000
    for t in range(max_iters):
        if time.time() >= deadline - 5.0:
            break

        conv = _full_convolution(a)
        k = int(np.argmax(conv))          # index of the current maximum

        # gradient of (a*a)[k] w.r.t. a
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            i_idx = np.arange(lo, hi + 1)
            j_idx = k - i_idx
            grad[i_idx] = 2.0 * a[j_idx]

        # mirror‑descent step (step size shrinks slowly)
        eta = 0.6 / (1.0 + t * 0.0005)
        delta = -eta * grad
        delta = np.clip(delta, -20.0, 0.0)
        a = a * np.exp(delta)
        a = _project_simplex(a)

        # occasional re‑ordering and check
        if (t % 100) == 0:
            a = _interleaved_reorder(a)
            val = _compute_objective(a)
            if val < best_val - 1e-12:
                best_val = val
                best_a = a.copy()
                print(f"[subgrad it {t}] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Light hill‑climb (swap / tiny transfer) – strictly improving moves
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    best_a = a.copy()
    best_val = cur_val
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        # choose move type
        if rng.random() < 0.3:
            # swap two positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # move a tiny amount from i to j
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.03 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection back onto the simplex (numeric safety)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        val = _compute_objective(new_a)
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] improved = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0     # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the allotted time.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Try to load a previously discovered construction (if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum > 0:
            a0 = a0 / a0_sum
        else:
            a0 = np.full(n, 1.0 / n)
    except Exception:
        a0 = None

    best_a = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Search loop – a few restarts with varied seeds
    # ------------------------------------------------------------------
    init_strategies = ["uniform", "sin", "rand"]
    restart = 0
    while time.time() < deadline - 250 and restart < 6:
        restart += 1

        # ----- select initial vector -------------------------------------------------
        if a0 is not None and restart == 1:
            a = a0.copy()
            init_name = "previous"
        else:
            strat = rng.choice(init_strategies)
            if strat == "uniform":
                a = np.full(n, 1.0 / n)
                init_name = "uniform"
            elif strat == "sin":
                idx = np.arange(n) + 0.5
                a = np.sin(np.pi * idx / n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                init_name = "sinusoid"
            else:
                a = rng.exponential(scale=1.0, size=n)
                a = a / a.sum()
                init_name = "random"

        print(f"\n[restart {restart}] init = {init_name}, start obj = {_compute_objective(a):.6f}")

        # ----- smoothed‑max descent ---------------------------------------------------
        a_opt, val_opt = _smoothed_descent(a, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[global] after smoothed descent: new best = {best_val:.6f}")

        # ----- sub‑gradient descent on the exact max ----------------------------------
        if time.time() < deadline - 150:
            a_sub, val_sub = _max_subgradient_descent(a_opt, deadline)
            if val_sub < best_val:
                best_val = val_sub
                best_a = a_sub.copy()
                print(f"[global] after sub‑grad: new best = {best_val:.6f}")

        # ----- short hill‑climb (if time permits) ------------------------------------
        if time.time() < deadline - 80:
            a_hc, val_hc = _hill_climb_refine(best_a, best_val, deadline)
            if val_hc < best_val:
                best_val = val_hc
                best_a = a_hc.copy()
                print(f"[global] after hill‑climb: new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polish on the best vector seen so far
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final evaluation with the official black‑box (required for the harness)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best objective reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = np.inf

    return best_a.tolist()
