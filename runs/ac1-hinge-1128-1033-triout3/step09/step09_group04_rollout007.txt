def simplex_projection(v):
    # ensure sum to 1 and nonnegative
    # algorithm: sort descending, find rho, compute theta.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _simplex_projection(v):
    """Project vector v onto probability simplex (non-negative, sum=1)."""
    # ensure numpy array
    v = np.asarray(v, dtype=np.float64)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:

        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

"""
Propose a step‑function (non‑negative vector) that minimizes the
`evaluate_sequence` objective.

The search consists of:

* **Warm‑up phase** – a smooth (soft‑max) gradient descent on a
  probability‑simplex using an exponential‑weighted sub‑gradient.
  The smoothing parameter α is increased from a small value to a large
  one, so the updates initially follow a smooth landscape and finish
  by approximating the true max‑norm.

* **Hill‑climb phase** – a fast deterministic local move that repeatedly
  transfers a fraction of the mass from the index that contributes most
  to the current worst convolution entry to the index that contributes
  the least.  The transfer fraction anneals from 30 % to 0.1 %.

* **Random restarts** – many short restarts, each starting from a
  random Dirichlet draw (with a log‑uniform concentration) or from the
  current best vector (interpolated to the working length).  Each restart
  runs a brief smooth‑gradient phase followed by a short hill‑climb and a
  tiny mixture with the uniform vector.

All operations respect the overall time budget (≈1000 s).  The best
vector found is returned as a plain Python list.  The code makes no
filesystem or network accesses and only uses the already‑imported
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full length = 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _linconv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of arbitrary vectors a and b (FFT implementation).
    """
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (onto the probability simplex Σx = 1, x≥0)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending order
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                    # all non‑positive → uniform
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)

# Simple deterministic candidates (fallback)

def _initial_candidate(n: int):
    """Return a reasonable starting vector (uniform or a few hand‑crafted shapes)."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)
    rng = np.random.default_rng(0)

    def try_update(c):
        nonlocal best_a, best_val
        cand = np.asarray(c, dtype=np.float64)
        if cand.ndim != 1 or cand.shape[0] != n:
            return
        if np.any(cand < 0):
            return
        cand = _simplex_projection(cand)
        val = _objective(cand, n)
        if val < best_val:
            best_a, best_val = cand, val

    idx = np.arange(n, dtype=float)
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # V‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic decay
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends
    for _ in range(8):
        try_update(rng.dirichlet(np.ones(n)))
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent on the L∞‑norm of the autocorrelation
# ----------------------------------------------------------------------
def _smooth_grad_phase(a0: np.ndarray,
                       n: int,
                       start_time: float,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smoothed approximation of max(conv(a,a)).
    Uses a soft‑max with a linearly increasing temperature α.
    """
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        conv = _full_convolution(cur)
        maxb = conv.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # progress in [0,1]
        progress = (now - start_time) / (deadline - start_time)

        # α moves from 5 (smooth) to 200 (sharp)
        alpha = 5.0 * (1.0 - progress) + 200.0 * progress

        # soft‑max weights (stable computation)
        c = conv.max()
        exp_term = np.exp(alpha * (conv - c))
        w = exp_term / exp_term.sum()

        # gradient = 2 * correlation(a, w)
        w_rev = w[::-1]
        grad_full = _linconv_fft(cur, w_rev)          # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)              # slice that yields Σ a_i w_{i+m}
        grad = 2.0 * grad_full[idx]

        # step‑size schedule (large → small)
        eta = 0.7 * (1.0 - progress) + 0.001 * progress

        # gradient step + projection onto simplex
        cur = cur - eta * grad
        cur = _simplex_projection(cur)

        # jitter if stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.005, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur = _simplex_projection(cur)
            no_improve = 0
            print(f"[SmoothGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     start_frac: float = 0.30,
                     end_frac: float = 0.001,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move a fraction of mass from the worst contributor to the best.
       Accept only strictly improving moves."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealed transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # locate the active peak
        k = int(np.argmax(b))

        # slice a_{k-i}
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev = np.zeros_like(cur)
        rev[idx] = cur[k - idx]

        src = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[src] = np.inf
        tgt = int(np.argmin(rev_tmp))

        if cur[src] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur = _simplex_projection(cur)
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        # trial transfer
        cur[src] -= delta
        cur[tgt] += delta

        new_val = 2.0 * n * _full_convolution(cur).max()
        if new_val < cur_val - 1e-12:
            # accepted move (already applied)
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test few convex combinations (1‑α)·a + α·uniform)."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _simplex_projection(cand)
        val = _objective(cand, n)
        if val < best_val:
            best_a, best_val = cand, val
    return best_a, best_val

            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 = _simplex_projection(a0)
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    TARGET = 1.503
    print(f"[Init] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Warm‑up: a longer smooth‑gradient phase + hill‑climb
    # ------------------------------------------------------------------
    warm_deadline = start_time + 0.25 * (deadline - start_time)
    best_a, best_val = _smooth_grad_phase(best_a, n,
                                          start_time, warm_deadline,
                                          best_a, best_val,
                                          target_val=TARGET)
    best_a, best_val = _hillclimb_phase(best_a, n,
                                        warm_deadline, deadline,
                                        best_a, best_val,
                                        start_frac=0.30, end_frac=0.001,
                                        target_val=TARGET)
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Subsequent fast restarts
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    restart_counter = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a small slice (at least 0.6 s) for each restart
        slice_budget = max(0.6, 0.008 * remaining)

        restart_start = time.time()
        restart_deadline = restart_start + slice_budget

        # --- random Dirichlet initialisation (log‑uniform concentration) ---
        log_alpha = rng.uniform(-2, 2)          # α ∈ [10⁻², 10²]
        alpha = 10.0 ** log_alpha
        a = rng.dirichlet(alpha * np.ones(n))

        # --- brief smooth gradient descent ---
        a, _ = _smooth_grad_phase(a, n,
                                  restart_start, restart_deadline,
                                  best_a, best_val,
                                  target_val=TARGET)

        # --- short hill‑climb ---
        a, _ = _hillclimb_phase(a, n,
                                restart_start, restart_deadline,
                                best_a, best_val,
                                start_frac=0.20, end_frac=0.001,
                                target_val=TARGET)

        # --- tiny uniform mixing ---
        a, val = _mix_with_uniform(a, n)

        # --- global update ---
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Restart {restart_counter:3d}] new global best = {best_val:.6f}")
            if best_val <= TARGET:
                print("[Target reached] exiting early.")
                break

        restart_counter += 1

    # ------------------------------------------------------------------
    # Final verification via the supplied secure wrapper (optional)
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list
    return best_a.tolist()
