def _swap_phase(a0, n, start, deadline, best_a, best_val):
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = a.copy()
    best_local_val = cur_val

    # Temperature schedule: start T0 maybe (best_val - 1.5) * some factor?
    # Or use a fraction of best_val.
    T0 = best_val * 0.1  # maybe.
    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T0 * (1.0 - progress)
        # choose i and j
        # bias: choose source i proportionally to contribution at max:
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0 and np.random.rand() < 0.7:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.random.rand() * n)
        # target j choose among low contributions:
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        # Choose j with probability inversely proportional:
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0:
            probs_j = w / w.sum()
            j = np.random.choice(n, p=probs_j)
        else:
            j = (i + np.random.randint(1, n)) % n

        if i == j:
            continue

        # compute delta = a[j] - a[i] (swap)
        delta = a[j] - a[i]
        if abs(delta) < 1e-15:
            continue

        # compute new convolution quickly
        b_new = b.copy()

        src = j
        tgt = i
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a

"""

-----------------

A multi‑stage optimizer for the step‑function evaluation task.

The optimiser runs the following pipeline within a 1000 s time budget:

1. **Broad parametric sweep** – a rich family of analytic shapes
   (exponential, power‑law, beta, Gaussian, sine‑power, logistic,
   piecewise constant) plus many random Dirichlet draws.  The best
   normalised vector from this sweep is kept.

2. **Adam on a smooth log‑sum‑exp surrogate** – minimises a differentiable
   approximation of ``max(conv)`` while staying on the probability
   simplex.

3. **Targeted hill‑climbing** – greedy transfer of a small amount of mass
   from indices that contribute most to the current maximal convolution
   entry to indices that contribute little.

4. **Projected sub‑gradient descent** – exact sub‑gradient updates for the
   true max‑convolution objective.

5. **Swap‑annealing** – a simulated‑annealing phase that repeatedly swaps
   two positions (i.e. moves the whole difference of their heights) and
   accepts the swap if the objective improves or with a temperature‑based
   probability.  The swap move is implemented with an O(n) incremental
   convolution update.

6. **Tiny uniform mixing** – a final convex combination with a small
   amount of the uniform distribution to shave off any remaining peak.

The routine returns a non‑negative list whose sum equals 1, ready to be
passed to ``evaluate_sequence``.  All steps keep the vector on the simplex
and respect the security checks of the evaluation function.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # not used directly but kept for compatibility


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective handling (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector already normalised to Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families – exhaustive sweep for a good starting point
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector found among a very diverse set of families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)                # 0 … n‑1
    i1 = i + 1.0                                 # 1 … n

    # 1. Exponential tails (wide range)
    for r in np.logspace(np.log10(0.9), np.log10(0.99999), 15):
        _try(r ** i)                            # monotone decreasing
        _try(r ** i + r ** (n - 1 - i))         # symmetric U‑shape

    # 2. Power‑law tails (wide range)
    for p in np.linspace(0.1, 10.0, 15):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shaped densities (broader parameter grid)
    for a_param in np.linspace(0.1, 10.0, 10):
        for b_param in np.linspace(0.1, 10.0, 10):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Gaussian‑type families
    x = (i - (n - 1) / 2.0) / n
    for sigma in np.linspace(0.02, 0.30, 12):
        a = np.exp(-0.5 * (x / sigma) ** 2)
        _try(a)

    # 5. Laplace (double‑exponential) tails
    for tau in np.linspace(0.02, 0.30, 12):
        a = np.exp(-np.abs(x) / tau)
        _try(a)

    # 6. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _try(base_sin ** p)

    # 7. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.005, 0.15, 12):
        k = max(1, int(round(k_frac * n)))
        for high_frac in np.linspace(0.02, 0.40, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 8. Logistic‑type tails (asymmetric but can be mirrored)
    for slope in np.linspace(0.003, 0.04, 8):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 9. Random Dirichlet draws (enhanced diversity)
    rng = np.random.default_rng(12345)
    for _ in range(200):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 10. Interpolate previous solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp approximation of the max‑convolution."""
    w = np.log(a0 + 1e-12)          # logits → simplex via soft‑max

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.14, 0.004
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 70_000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to shake out of plateaux
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy transfer of a small amount of mass from high‑contrib. indices."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.6, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source – weighted by contribution
        if total_contrib > 0.0 and np.random.rand() < 0.7:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        # target – bias towards indices that hardly affect the max entry
        if np.random.rand() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear‑time)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb the current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n
    eta = eta0
    decay = 0.9996
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 180_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient at the active index
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # gradual step‑size decay
        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # shake & restart
            noise = np.random.normal(scale=0.009, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase D – Swap‑annealing (simulated annealing on permutations)
# ----------------------------------------------------------------------
def _swap_phase(a0: np.ndarray, n: int,
                start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Simulated‑annealing on swaps of whole elements (full difference)."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = a.copy()
    best_local_val = cur_val

    # Temperature schedule – start relatively high, cool to near zero
    T0 = max(0.1 * best_val, 1e-3)
    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T0 * (1.0 - progress)  # linear cooling

        # Identify the current maximal convolution index
        k_max = int(np.argmax(b))

        # Compute contributions to this entry (helps bias the move)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # Choose source index – likely a high‑contributor
        if total_contrib > 0 and np.random.rand() < 0.75:
            probs = contrib / total_contrib
            src = np.random.choice(n, p=probs)
        else:
            src = int(np.random.rand() * n)

        # Choose target index – preferably low contribution (inverse weighting)
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[src] = 0.0
        if w.sum() > 0 and np.random.rand() < 0.85:
            probs_t = w / w.sum()
            tgt = np.random.choice(n, p=probs_t)
        else:
            tgt = (src + np.random.randint(1, n)) % n

        if src == tgt:
            continue

        delta = a[tgt] - a[src]          # moving entire difference -> swap
        if abs(delta) < 1e-15:
            continue

        # Incremental convolution update for the swap
        b_new = b.copy()
        # source = src, target = tgt (move delta from src to tgt)
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Acceptance test
        accept = False
        if new_val < cur_val - 1e-12:
            accept = True
        elif T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
            accept = True

        if accept:
            # Perform the swap (equivalently: a[tgt] += delta; a[src] -= delta)
            a[tgt] += delta
            a[src] -= delta
            b = b_new
            cur_val = new_val

            if cur_val < best_local_val - 1e-12:
                best_local_val, best_local_a = cur_val, a.copy()
                print(f"[Swap {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break

        # Periodically refresh the full convolution to suppress drift
        if it % 2500 == 0:
            b = _full_convolution(a)

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem size – a compromise between representation power and speed.
    n = 6000

    # ------------------------------------------------------------------
    # Initialise a – exhaustive parametric sweep.
    # ------------------------------------------------------------------
    print("[Init] starting parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase A] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – targeted hill‑climbing (≈25 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * (deadline - start_time)
    best_a = _hillclimb_targeted(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    best_val = _objective_value(best_a)
    print(f"[Phase B] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – projected sub‑gradient descent (≈20 % of budget)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase C] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – swap‑annealing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _swap_phase(
        best_a, n,
        start=phase_c_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase D] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
