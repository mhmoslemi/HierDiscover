    best_val = cur_val
    best_a = a.copy()
    no_improve = 0

        # compute contributions
        idx = np.arange(n)
        kminus = k - idx

        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[kminus[mask]]
        total = contrib.sum()
        # choose source with probability proportional to contrib (heavy)
        if total > 0:
            src = np.random.choice(n, p=contrib/total)
        else:
            src = np.argmax(contrib)
        # choose target with probability inversely proportional to contrib
        inv = np.where(mask, 1.0/(contrib+1e-12), 0)
        inv[src] = 0
        if inv.sum() > 0:
            tgt = np.random.choice(n, p=inv/inv.sum())
        else:
            # fallback: random different index
            tgt = (src + np.random.randint(1,n))%n
        delta = a[src] * np.random.rand() * max_frac

        a[tgt] += delta
        # maintain nonnegative and renormalize? For better stability, after move we renormalize sum? But we keep sum=1 because we transferred mass, not scaled.

                best_a = a.copy()

            progress = (time.time() - start_time) / (deadline - start_time)

                # keep old b.

def _full_convolution(x):
    n = len(x)
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _beta_family_search(n, time_remaining):
    start = time.time()
    best_a = None
    best_val = np.inf
    i = np.arange(1, n+1, dtype=np.float64)

    p_grid = np.linspace(0.2, 0.8, 31)
    for p in p_grid:
        for q in p_grid:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0:
                continue
            a /= s
            val = _objective(a)

def _hillclimb_improved(a0, n, deadline, start_time, best_a, best_val, target_val=1.503):
    a = a0.copy()
    cur_val = _objective(a)
    no_improve = 0
    IMPROVE_LIMIT = 200_000
    # schedule parameters
    max_frac_start = 0.5
    max_frac_end = 0.001
    temp_start = 0.2
    temp_end = 1e-8

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # update schedule
        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # compute convolution and max location
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # compute contributions
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index selection
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        # target index selection (prefer low contribution)
        inv_weights = np.zeros_like(a)
        inv_weights[mask] = 1.0 / (contrib[mask] + 1e-12)
        inv_weights[i_src] = 0.0
        sum_inv = inv_weights.sum()
        if sum_inv > 0:
            j_tgt = np.random.choice(n, p=inv_weights / sum_inv)
        else:
            # fallback uniform distinct index
            j_tgt = (i_src + np.random.randint(1, n)) % n

        if i_src == j_tgt:
            continue

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= 1e-14:
            continue

        # Apply move
        a[i_src] -= delta
        a[j_tgt] += delta

        new_val = _objective(a)

        # acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # revert
            a[i_src] += delta
            a[j_tgt] -= delta
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # perturbation
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            else:
                a = np.random.dirichlet(np.ones(n))
            cur_val = _objective(a)
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1
    return best_a, best_val

"""
Step‑function optimizer for ``evaluate_sequence``.

The algorithm works in three stages:

1️⃣  **Warm‑start** – reuse a previously found solution (if present) or
    generate a collection of promising candidates:
        * a short deterministic sweep of the two‑parameter Beta family
          a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} (including the special cases
          i^{-1/2}, (i·(n‑i+1))^{-1/2}, and the uniform shape);
        * a few random Beta draws and random mixtures of two Beta
          distributions;
        * a handful of Dirichlet draws.

   The best of those candidates is kept as the initial point.

2️⃣  **Improved stochastic hill‑climbing** – starting from the best
    candidate we repeatedly move a small amount of mass from an index
    that heavily contributes to the current maximal convolution entry to
    an index that contributes little.  The move size (``max_frac``) and
    a Metropolis temperature decay very slowly, giving a long
    exploration phase.  When progress stalls we add a small Gaussian
    perturbation and continue.  The routine respects the simplex
    constraints (non‑negative, sum = 1) and stores the globally best
    sequence seen.

3️⃣  **Final evaluation** – the obtained sequence is passed to
    ``evaluate_sequence`` for the official score and stored in the
    global variable ``height_sequence_1`` for possible reuse.

The whole search respects a 1000 s time budget (0.5 s are reserved
for the final evaluation call).  The implementation uses only NumPy
(and the standard library) and runs on a single CPU core.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families (Beta, arcsine‑like, sin, etc.)
# ----------------------------------------------------------------------
def _beta_candidate(i: np.ndarray, n: int, p: float, q: float) -> np.ndarray:
    """Return a normalized Beta‑family sequence for given (p,q)."""
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def _mixture_candidate(i: np.ndarray,
                       n: int,
                       p1: float, q1: float,
                       p2: float, q2: float,
                       alpha: float) -> np.ndarray:
    """Mixture of two Beta families (weight alpha for the first)."""
    a1 = _beta_candidate(i, n, p1, q1)
    a2 = _beta_candidate(i, n, p2, q2)
    if a1 is None or a2 is None:
        return None
    mix = alpha * a1 + (1.0 - alpha) * a2
    # both components are already normalized, so mix sums to 1
    return mix


def _extra_candidates(n: int) -> List[np.ndarray]:
    """A few handcrafted families that are known to be good baselines."""
    i = np.arange(1, n + 1, dtype=np.float64)
    cand = []

    # left‑only power‑law (p=0.5, q=1)
    a = 1.0 / np.sqrt(i)
    a /= a.sum()
    cand.append(a)

    # symmetric arcsine‑like (p=q=0.5)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    a /= a.sum()
    cand.append(a)

    # pure sine shape (positive half‑wave)
    a = np.sin(np.pi * i / (n + 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    cand.append(a)

    # cosine‑shaped (non‑negative half)
    a = np.cos(np.pi * (i - 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    cand.append(a)

    return cand


# ----------------------------------------------------------------------
# Search over the Beta family (deterministic grid + random draws)
# ----------------------------------------------------------------------
def _beta_family_search(n: int, time_budget: float) -> Tuple[np.ndarray, float]:
    """
    Search the 2‑parameter Beta family for a good start.
    Returns (best_sequence, best_value).  The function respects the
    supplied time_budget (seconds) and stops early if time runs out.
    """
    start = time.time()
    i = np.arange(1, n + 1, dtype=np.float64)

    best_a = None
    best_val = np.inf

    # 1. fine grid around the region where the previous run performed well
    p_grid = np.linspace(0.2, 0.8, 31)           # includes p≈0.5
    for p in p_grid:
        for q in p_grid:
            a = _beta_candidate(i, n, p, q)
            if a is None:
                continue
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()

    # 2. random draws (including occasional mixtures)
    trials = 2500
    for _ in range(trials):
        now = time.time()
        if now - start > time_budget:
            break

        # 30 % chance to use a mixture of two Beta families
        if np.random.rand() < 0.3:
            p1 = 10 ** np.random.uniform(np.log10(0.01), np.log10(20.0))
            q1 = 10 ** np.random.uniform(np.log10(0.01), np.log10(20.0))
            p2 = 10 ** np.random.uniform(np.log10(0.01), np.log10(20.0))
            q2 = 10 ** np.random.uniform(np.log10(0.01), np.log10(20.0))
            alpha = np.random.rand()
            a = _mixture_candidate(i, n, p1, q1, p2, q2, alpha)
            if a is None:
                continue
        else:
            p = 10 ** np.random.uniform(np.log10(0.01), np.log10(20.0))
            q = 10 ** np.random.uniform(np.log10(0.01), np.log10(20.0))
            a = _beta_candidate(i, n, p, q)
            if a is None:
                continue

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Improved stochastic hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb_improved(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.48) -> Tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that repeatedly moves a small amount of
    mass from a highly‑contributing index to a low‑contributing one.
    The schedule for the maximal transfer fraction and the temperature
    is deliberately stretched so that the search lasts until the
    deadline.
    """
    a = a0.copy()
    cur_val = _objective(a)
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    # schedule parameters (very slow decay)
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    iteration = 0
    eps = 1e-12
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # convolution and location of the current maximum
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – probability proportional to its contribution
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        # target index – inverse weighting (prefer low contribution)
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (different from source)
            j_tgt = (i_src + np.random.randint(1, n)) % n

        if i_src == j_tgt:
            continue

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # perform the move
        a[i_src] -= delta
        a[j_tgt] += delta

        new_val = _objective(a)

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # revert the move
            a[i_src] += delta
            a[j_tgt] -= delta
            no_improve += 1

        # stagnation handling – inject a small Gaussian perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.random.dirichlet(np.ones(n))
            cur_val = _objective(a)
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Produce a non‑negative step function (list of floats) that aims to
    minimise ``evaluate_sequence``.  The routine respects the 1000 s
    time budget.
    """
    np.random.seed(seed)

    start_time = time.time()
    # leave a small margin for the final secure call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 4000          # can be increased at the cost of speed

    # ------------------------------------------------------------------
    # 0️⃣  Warm‑start from a previously best solution (if available)
    # ------------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0:
            a0 /= s
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Gather a collection of promising candidates
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf

    # start from the interpolated solution (if any)
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from previous best, value={best_val:.6f}")

    # deterministic & random search in the Beta family
    time_for_beta = min(120.0, max(0.0, deadline - time.time() - 0.5))
    beta_a, beta_val = _beta_family_search(n, time_for_beta)
    if beta_val < best_val:
        best_a, best_val = beta_a, beta_val
        print(f"[init] beta family improved value={best_val:.6f}")

    # extra handcrafted families (arcsine‑like, sine, etc.)
    for cand in _extra_candidates(n):
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[init] extra family improved value={best_val:.6f}")

    # a few random Dirichlet draws for diversity
    for _ in range(4):
        rand_a = np.random.dirichlet(np.ones(n))
        val = _objective(rand_a)
        if val < best_val:
            best_a, best_val = rand_a, val
            print(f"[init] random Dirichlet improved value={best_val:.6f}")

    # fallback (should never happen)
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print(f"[init] fallback random, value={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Intensive hill‑climbing (the rest of the budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_improved(best_a, n,
                                           deadline=deadline,
                                           start_time=start_time,
                                           best_a=best_a,
                                           best_val=best_val,
                                           target_val=1.48)

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation & expose the solution globally
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # store for possible reuse in a later run
    global height_sequence_1
    height_sequence_1 = best_a.tolist()    # type: ignore[assignment]

    return best_a.tolist()
