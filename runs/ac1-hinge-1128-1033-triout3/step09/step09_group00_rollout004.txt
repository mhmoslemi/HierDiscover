      where k* are the indices attaining the current maximum.
    * A soft‑max (log‑softmax) parametrisation a = softmax(w) guarantees the
      simplex constraint Σa=1 automatically.
    * Adam updates (β₁=0.90, β₂=0.999) are applied to the logits w using the
      sub‑gradient transformed through the soft‑max Jacobian.
    * If no improvement is observed for a configurable number of steps the
      optimiser is re‑initialised (random logits) to escape plateaus.

3️⃣  Optional hill‑climbing (local mass transfer) – only a few quick passes
    * After the Adam phase a lightweight simulated‑annealing move set is
      performed; it moves a small fraction of probability mass from an index
      that heavily contributes to the current maximal convolution entry to an
      index that contributes little.  This often polishes the solution.

The whole routine respects the 1000 s budget and finally returns the best
sequence found (as a plain Python list).  All numbers are non‑negative and the
sequence sum is 1, but evaluate_sequence will still apply its safety checks.
"""

import time
import numpy as np

# ------------------------------------------------------------
# FFT based full linear convolution (a with itself)
# ------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the linear convolution of x with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ------------------------------------------------------------
# Sub‑gradient of the exact objective (max‑convolution)
# ------------------------------------------------------------
def _max_convolution_and_subgrad(a: np.ndarray, n: int):
    """
    Compute max convolution value and the sub‑gradient of
    F(a) = 2·n·max(a∗a) w.r.t. a.

    Returns:
        max_val : float  – the maximal convolution entry (before multiplying 2·n)
        grad    : np.ndarray – same shape as a, the sub‑gradient of 2·n·max_val
    """
    b = _full_convolution(a)
    max_val = b.max()
    # indices where the maximum is attained (allow tiny tolerance)
    eps = 1e-15
    max_idx = np.where(b >= max_val - eps)[0]

    grad = np.zeros_like(a)
    # contribution of each maximal index
    idx_arr = np.arange(n)
    for k in max_idx:
        # i + j = k  =>  j = k - i
        j = k - idx_arr
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]          # derivative of a_i*a_j term
    # factor 2·n from the outer objective
    grad *= 2.0 * n
    return max_val, grad


# ------------------------------------------------------------
# Adam optimiser on the simplex via log‑softmax parametrisation
# ------------------------------------------------------------
def _adam_subgrad_optimize(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           start_time: float,
                           best_a: np.ndarray,
                           best_val: float,
                           target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam on the exact max‑convolution objective using sub‑gradients.
    The parametrisation a = softmax(w) enforces a_i ≥ 0 and Σa_i = 1.
    """
    # initialise logits (add a tiny epsilon to avoid log(0))
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.0015          # a little more aggressive than before
    max_idle = 8_000                         # restart after this many idle steps

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---------- decode logits ----------
        w_shift = w - np.max(w)               # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()                          # soft‑max

        # ---------- evaluate ----------
        max_conv, _ = _max_convolution_and_subgrad(a, n)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # ---------- schedule ----------
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---------- sub‑gradient ----------
        _, grad_a = _max_convolution_and_subgrad(a, n)

        # Chain rule through soft‑max (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-10)

        # occasional jitter to shake off tiny plateaus
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.025, size=n)

        # restart if idle for too long
        if idle > max_idle:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ------------------------------------------------------------
# Light hill‑climbing (mass transfer) – optional polishing
# ------------------------------------------------------------
def _simple_hillclimb(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Very fast simulated‑annealing style move: take a fraction of probability mass
    from an index that heavily contributes to the current maximum convolution
    entry and give it to an index that contributes little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # annealing schedules (aggressive early, vanishing later)
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.025, 1e-9

    stagnation = 0
    STAG_LIMIT = 15_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate a (possibly non‑unique) maximal convolution index
        k_max = int(np.argmax(b))

        # contribution of each index to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source – probability proportional to contribution
        if total > 0:
            src = np.random.choice(n, p=contrib / total)
        else:
            src = np.argmax(a)

        if a[src] <= eps:
            continue

        # target – prefer small contribution (inverse weighting)
        inv = np.zeros_like(a)
        inv[mask] = 1.0 / (contrib[mask] + eps)
        inv[src] = 0.0
        wsum = inv.sum()
        if wsum > 0:
            tgt = np.random.choice(n, p=inv / wsum)
        else:
            # uniform fallback excluding src
            cand = np.delete(np.arange(n), src)
            tgt = np.random.choice(cand)

        # amount of mass to move
        delta = a[src] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # update convolution incrementally (O(n) update)
        b_new = b.copy()
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            stagnation = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            stagnation += 1

        # occasional random perturbation if stuck
        if stagnation > STAG_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.03, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            stagnation = 0
            print(f"[Hill {iteration:7d}] perturb due to stagnation")
        iteration += 1

    return best_a, best_val


# ------------------------------------------------------------
# Very cheap search over the Beta family (dense grid)
# ------------------------------------------------------------
def _beta_grid_search(n: int) -> tuple[np.ndarray, float]:
    """Search a fine grid of (p,q)∈[0.1,2] with step 0.05 in the Beta family."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)

    p_vals = np.arange(0.1, 2.001, 0.05)
    q_vals = p_vals.copy()

    for p in p_vals:
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            # exact objective (Σa = 1)
            b = _full_convolution(a)
            cur = 2.0 * n * b.max()
            if cur < best_val:
                best_val, best_a = cur, a.copy()
    return best_a, best_val

        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------
    # 1️⃣  Beta‑family sweep (cheap, gives a decent deterministic start)
    # ------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:
        val = 2.0 * n * _full_convolution(a0).max()
        best_a, best_val = a0.copy(), val

    # run a fairly dense beta grid; this is cheap compared to the later phases
    grid_a, grid_val = _beta_grid_search(n)
    if grid_val < best_val:
        best_a, best_val = grid_a, grid_val
        print(f"[init] beta‑grid improved value={best_val:.6f}")

    # a handful of Dirichlet draws for extra stochastic variety
    for _ in range(6):
        cand = np.random.dirichlet(np.ones(n))
        val = 2.0 * n * _full_convolution(cand).max()
        if val < best_val:
            best_a, best_val = cand, val
            print(f"[init] Dirichlet seed improved value={best_val:.6f}")

    if best_a is None:
        # fallback – uniform random vector
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = 2.0 * n * _full_convolution(best_a).max()
        print("[init] fallback random start, value=", best_val)

    # ------------------------------------------------------------
    # 2️⃣  Adam optimisation on the *exact* max‑convolution (sub‑gradient)
    # ------------------------------------------------------------

        deadline=phase_adam_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )
    print(f"[phase A] finish – best value = {best_val:.6f}")

    # ------------------------------------------------------------
    # 3️⃣  Light hill‑climbing polishing (the remaining time)
    # ------------------------------------------------------------
    best_a, best_val = _simple_hillclimb(
        best_a, n,
        deadline=deadline,
        start_time=phase_adam_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )
    print(f"[phase B] final best value = {best_val:.6f}")

        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
