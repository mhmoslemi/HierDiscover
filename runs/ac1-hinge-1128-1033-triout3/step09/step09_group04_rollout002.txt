def project_onto_simplex(v):
    # ensure non-negative sum to 1
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

```
def simplex_projection(v):
    n = v.shape[0]
    if v.sum() == 0:
        return np.full(n, 1.0 / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Softmax‑smoothed Adam descent for the max‑convolution problem.

The algorithm works inside the 1000 s budget.  For a vector `a`
on the probability simplex (non‑negative, sum = 1) we minimise

    2·n·max_k (a * a)_k

where `*` is the linear convolution.
The max entry is approximated by a smooth soft‑max with a
temperature that is annealed during optimisation.
Gradients are obtained analytically (the derivative of a
convolution peak is a shifted copy of the current vector) and
computed efficiently with FFT‑based correlation.
A classic Adam update keeps a well‑behaved step‑size,
after which the iterate is projected back onto the simplex.
If the optimisation stalls, a small Gaussian jitter is added
and the Adam moments are reset.
The best vector seen is remembered, after the main loop a brief
mix with the uniform vector is tried, and the final value is
checked with the supplied `evaluate_sequence`.

The code is self‑contained, uses only `numpy` and `time`,
and respects the required function signature `propose_candidate`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al.)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform vector
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution / correlation using a shared FFT size
# ----------------------------------------------------------------------
def _fft_size_for_length(L: int) -> int:
    """Return the next power‑of‑two size ≥ L (for real FFT)."""
    return 1 << (L - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Return the linear convolution a*a (length 2·len(a)-1) using FFT size L.
    L must be ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * a.shape[0] - 1]
    return conv


def _correlation_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation of a (len n) with w (len m) using FFT size L.
    Returns the full correlation of length n+m‑1.
    """
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    corr = np.fft.irfft(fa * np.conj(fw), L)[: a.shape[0] + w.shape[0] - 1]
    return corr


# ----------------------------------------------------------------------
# Gradient of the soft‑max‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, b: np.ndarray, tau: float) -> np.ndarray:
    """
    Given a (length n) and its self‑convolution b (length 2n‑1),
    compute the gradient of
        F_tau(a) = tau * log Σ_k exp(b_k / tau)
    using FFT‑based correlation.
    """
    n = a.shape[0]
    max_b = b.max()
    # Soft‑max weights (numerically stable)
    w = np.exp((b - max_b) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Correlation a with w gives ∂b_k/∂a_i summed with weights w_k
    # gradient = 2 * (a ⋆ w)   (⋆ = linear correlation)
    L = _fft_size_for_length(n + w.shape[0] - 1)
    corr = _correlation_fft(a, w, L)                     # length = n + (2n‑1) - 1 = 3n‑2
    start = w.shape[0] - 1                               # index where k=0 aligns
    grad = 2.0 * corr[start : start + n]                 # shape (n,)
    return grad


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector (optional final tweak)
# ----------------------------------------------------------------------
def _tiny_uniform_mix(a: np.ndarray, n: int) -> np.ndarray:
    """Test a few small convex combinations with the uniform vector."""
    best = a
    best_val = 2.0 * n * _full_convolution_fft(best, _fft_size_for_length(2 * n - 1)).max()
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _simplex_projection(cand)
        val = 2.0 * n * _full_convolution_fft(cand, _fft_size_for_length(2 * n - 1)).max()
        if val < best_val:
            best_val, best = val, cand
    return best

# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Uses a soft‑max‑smoothed Adam descent
    (FFT based) with occasional jitter and a final tiny uniform mix.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # safety margin

    # --------------------------------------------------------------
    # Problem size: a slightly larger n gives a tighter theoretical bound.
    # --------------------------------------------------------------
    n = 10000

    # --------------------------------------------------------------
    # Initialise from the previous best if it exists, otherwise random.
    # --------------------------------------------------------------
    try:

        if old.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 = _simplex_projection(a0)
    except Exception:
        # random Dirichlet start (concentration ≈ 1)
        rng = np.random.default_rng(seed + 123)
        a0 = rng.dirichlet(np.ones(n))

    cur = a0.copy()
    best = cur.copy()
    # initial objective
    L_conv = _fft_size_for_length(2 * n - 1)
    b = _full_convolution_fft(cur, L_conv)
    best_val = 2.0 * n * b.max()
    print(f"[Init] objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------------
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.1

    # temperature schedule for the soft‑max
    tau0 = 0.05
    tau_min = 1e-6

    # stagnation handling
    last_improve = 0
    IMPATIENT = 200_000  # iterations without improvement → jitter

    # Pre‑compute the FFT size needed for both convolution and correlation.
    # For correlation we need at least n + (2n‑1) - 1 = 3n‑2 points.
    L_fft = _fft_size_for_length(3 * n - 2)

    t = 0
    while time.time() < deadline:

        fa = np.fft.rfft(cur, L_fft)

        # ---- self‑convolution (to obtain the max‑entry) ----
        conv_full = np.fft.irfft(fa * fa, L_fft)[: 2 * n - 1]
        max_b = conv_full.max()
        obj = 2.0 * n * max_b

        # ---- record best ----
        if obj < best_val - 1e-12:
            best_val = obj
            best = cur.copy()
            last_improve = t
            print(f"[Iter {t:7d}] new best = {best_val:.6f}")
            if best_val <= 1.503:
                print("[Target reached] exiting early.")
                break

        # ---- soft‑max temperature (exponential decay) ----
        progress = (time.time() - start_time) / (deadline - start_time)
        tau = max(tau_min, tau0 * (0.5 ** (t / 5000.0)))   # halve every ~5k iters

        # ---- soft‑max weights over all convolution entries ----
        w = np.exp((conv_full - max_b) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth approximation ----
        # reuse fa, compute FFT of weights
        fw = np.fft.rfft(w, L_fft)
        corr_full = np.fft.ifft(fa * np.conj(fw), L_fft).real
        # length 3n‑2, slice the part that corresponds to the required gradient
        start = (2 * n - 1) - 1                     # = 2n‑2
        grad = 2.0 * corr_full[start : start + n]   # shape (n,)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cur = cur - step

        # ---- projection back onto the simplex ----
        cur = _simplex_projection(cur)

        # ---- jitter if we have been stagnant for a long time ----
        if t - last_improve > IMPATIENT:
            rng = np.random.default_rng(seed + t)
            cur += rng.normal(scale=0.002, size=n)
            cur = np.maximum(cur, 0.0)
            cur = _simplex_projection(cur)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = t
            print(f"[Iter {t}] jitter after stagnation")

    # --------------------------------------------------------------
    # Final tiny mixing with the uniform vector (often gives a tiny gain)
    # --------------------------------------------------------------
    best = _tiny_uniform_mix(best, n)

    # --------------------------------------------------------------
    # Verify with the provided secure wrapper (optional, for logging)
    # --------------------------------------------------------------

        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return as plain Python list (sum ≈ 1)
    return best.tolist()
