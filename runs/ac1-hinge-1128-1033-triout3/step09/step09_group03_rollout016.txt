"""
A multi‑restart stochastic optimizer for the “min‑max self‑convolution” problem.

The algorithm works in three stages, but it repeats the first two stages
many times with different random starts:

1. **Adam on a smooth (log‑sum‑exp) surrogate** – quickly drives the
   solution towards a basin of attraction.
2. **Simulated‑annealing hill‑climbing** – explores the neighbourhood of
   the current best with occasional uphill moves; the temperature schedule
   is aggressive so that larger moves are accepted early on.
3. **Projected sub‑gradient descent** – a short polishing pass that uses
   the exact non‑smooth objective.

Multiple short runs (a few seconds each) are performed until the time
budget is almost exhausted; the best solution found over all runs is
returned.  All vectors are kept on the probability simplex (non‑negative,
sum = 1) and the final candidate is mixed with a tiny amount of uniform
mass, which often yields a marginal improvement.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length = 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution."""
    w = np.log(a0 + 1e-12)  # logits → simplex via soft‑max
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 40000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules
        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # tiny jitter to escape plateaus
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.006, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing with a temperature‑controlled acceptance rule.
    The temperature is set higher than in the reference implementation to
    allow more aggressive exploration.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    # A slightly hotter start (5 % of the current value) and a broader max‑step
    T0 = cur_val * 0.05
    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)               # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # locate the dominant convolution entry
        k_max = int(np.argmax(b))

        # --- source selection (indices that heavily contribute to the maximum) ---
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # --- target selection (prefer indices that contribute little) ---
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----- incremental convolution update -----
        b_old = b.copy()
        # linear part 2·δ·a shifted by j  –  2·δ·a shifted by i
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        # quadratic part
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        # update probability vector
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0.0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA  {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            # revert
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the current state a bit and continue
            print(f"[SA  {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.018, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase C – Exact projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 250_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturb after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: mixing a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α values."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Helper: build a few deterministic seeds
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector (sum = 1) among several simple families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform
    _try(np.ones(n))

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _try(a)

    # monotone exponential (no symmetry)
    for r in (0.9999, 0.9995, 0.999, 0.998):
        a = r ** np.arange(n, dtype=float)
        _try(a)

    # random Dirichlet draws – a few quick samples
    rng = np.random.default_rng(12345)
    for _ in range(15):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val

        phase_b_deadline = time.time() + 0.60 * (deadline - time.time())
        best_a, best_val = _simulated_annealing_phase(best_a, n,
                                       start=time.time(),
                                       deadline=phase_b_deadline,
                                       best_a=best_a,
                                       best_val=best_val,
                                       rng=rng,
                                       target_val=TARGET)

        best_a, best_val = _subgradient_phase(best_a, n,
                                       start=time.time(),
                                       deadline=deadline,
                                       best_a=best_a,
                                       best_val=best_val,
                                       target_val=TARGET)

    # ------------------------------------------------------------------
    # Small post‑processing: mix with a tiny amount of uniform mass
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
