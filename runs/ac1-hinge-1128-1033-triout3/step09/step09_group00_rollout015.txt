def _random_beta_array(n, p, q):
    i = np.arange(1, n+1)
    a = (i ** (p-1.0)) * ((n - i + 1) ** (q-1.0))
    s = a.sum()
    if s <= 0:
        a = np.ones_like(a)
        s = a.sum()
    return a / s

    phase0_end = start + 0.30 * budget_s

        m = np.random.choice([1,2,3], p=[0.5,0.3,0.2])
        # sample weights
        weights = np.random.dirichlet(np.ones(m))
        # sample component parameters
        comp_arrays = []
        for _ in range(m):
            # sample p, q log-uniform between 1e-3 and 10

            p = np.exp(logp)
            q = np.exp(logq)
            comp = _random_beta_array(n_small, p, q)
            comp_arrays.append(comp)
        a_small = np.zeros(n_small)
        for w, comp in zip(weights, comp_arrays):

    hill_deadline = start_time + 0.80 * budget_s
    a = best_a.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n_full * b.max()
    no_improve = 0
    IMPROVE_LIMIT = 40000
    iteration = 0
    eps = 1e-14

    while time.time() < hill_deadline:
        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a_i to that entry
        idx = k_max - np.arange(n_full)
        mask = (idx >= 0) & (idx < n_full)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        tot_contrib = contrib.sum()

        # source index – biased by contribution
        if tot_contrib > 0:
            src_probs = contrib / tot_contrib
            i = np.random.choice(n_full, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – bias towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n_full, p=tgt_weights / wsum)
        else:
            j = np.random.randint(n_full - 1)
            if j >= i:
                j += 1

        # fraction of mass to move (decays linearly)
        progress = (time.time() - start_time) / (hill_deadline - start_time)
        max_frac = 0.8 * (1.0 - progress) + 0.001 * progress
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # propose a new candidate
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        b_candidate = _full_convolution(a_candidate)
        new_val = 2.0 * n_full * b_candidate.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # temperature schedule (exponential decay)
            T = 0.02 * (0.001 / 0.02) ** progress
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / (T + 1e-12))
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_candidate
            b = b_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    break
        else:
            no_improve += 1

        # stagnation handling – random perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.03, size=n_full)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n_full * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient polishing (remaining time)
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        max_val = b.max()
        ks = np.where(b == max_val)[0]          # all active constraints
        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n_full)
            mask = (idx >= 0) & (idx < n_full)
            grad[mask] += 2.0 * a[idx[mask]]

        # step‑size schedule (larger early, vanishing later)
        progress = (time.time() - start_time) / (deadline - start_time)
        step = 0.4 * (1.0 - progress) + 1e-5 * progress
        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[Subgrad {it:6d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    break

        # occasional jitter to escape flat plateaus
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n_full), 0.0)
            a = _project_simplex(a)

        it += 1

    # --------------------------------------------------------------
    # Final evaluation (official score)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as e:           # pragma: no cover
        print("[final] evaluate_sequence raised:", e)
        final_score = None

    return best_a.tolist()
