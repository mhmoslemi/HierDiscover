"""
A two‑stage optimizer for the min‑max self‑convolution problem.

Stage 1 – Smooth surrogate optimisation.
    * Variables are logits w with a = softmax(w) (ensures non‑negativity &
      Σa = 1 automatically).
    * The objective 2·n·max(conv(a,a)) is approximated by a log‑sum‑exp
      (a smooth max).  The surrogate is minimised by L‑BFGS‑B with exact
      gradients (derived analytically).  The temperature ε is reduced
      geometrically, using the solution of the previous stage as a warm
      start.

Stage 2 – Exact local polishing.
    * A fast O(n) simulated‑annealing hill‑climber (the same incremental
      update as the reference implementation) finishes the optimisation.

The code respects the 1000 s budget, works for any n (default 10000)
and returns a list of non‑negative floats ready for evaluate_sequence().
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities – FFT based, O(n log n)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (len = 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry and re‑normalise to Σa = 1."""
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Smooth surrogate optimisation (L‑BFGS‑B)
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray, eps: float, maxiter: int,
                     deadline: float) -> np.ndarray:
    """
    Minimise the smooth surrogate 2·n·ε·log‑sum‑exp(b/ε)
    where b = conv(a,a) and a = softmax(w).
    Returns the probability vector a found (or the input if time runs out).
    """
    n = a0.size
    # initialise logits from a0 (add a tiny epsilon to avoid log(0))
    w = np.log(a0 + 1e-12)

    # ------------------------------------------------------------------
    # Function returning (f, grad_w) for SciPy L‑BFGS‑B
    # ------------------------------------------------------------------
    def fun_and_grad(w_vec):
        # softmax → a
        w_shift = w_vec - np.max(w_vec)
        a = np.exp(w_shift)
        a /= a.sum()

        # convolution of a with itself
        b = _full_convolution(a)

        # log‑sum‑exp of b / eps (smooth max)
        max_b = b.max()
        shifted = (b - max_b) / eps
        exp_shifted = np.exp(shifted)
        Z = exp_shifted.sum()
        # smooth objective
        f = 2.0 * n * (max_b + eps * np.log(Z))

        # soft‑max weights over positions (probability distribution)
        p = exp_shifted / Z                     # length 2·n‑1, Σp = 1

        # gradient wrt a: 4·n·conv(a, p_rev) on the "valid" slice
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]    # length n
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through soft‑max (log‑it)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        return f, grad_w

    # ------------------------------------------------------------------
    # Run L‑BFGS‑B; respect the global deadline.
    # ------------------------------------------------------------------
    options = {"maxiter": maxiter, "gtol": 1e-12, "disp": False}
    # SciPy's minimize will call fun_and_grad repeatedly; we break early
    # if the global deadline is reached by raising a RuntimeError.
    def callback(xk):
        if time.time() > deadline:
            raise RuntimeError("time out")
    try:
        res = minimize(fun_and_grad, w, method="L-BFGS-B", jac=True,
                       callback=callback, options=options)
        w_opt = res.x
    except RuntimeError:
        # timed out – fall back to the current w
        w_opt = w

    # convert final logits to probability vector
    w_shift = w_opt - np.max(w_opt)
    a_opt = np.exp(w_shift)
    a_opt /= a_opt.sum()
    return a_opt


# ----------------------------------------------------------------------
# Exact local polishing – fast simulated annealing (same idea as reference)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, start: float, deadline: float,
                         rng: np.random.Generator,
                         target_val: float = 1.503) -> np.ndarray:
    """
    Small‑temperature hill‑climber that works with an O(n) incremental
    update of the convolution.  Stops early if the target value is met.
    """
    n = a0.size
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_a = a.copy()
    best_val = cur_val

    T0 = cur_val * 0.02          # initial temperature = 2 % of current value
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)                # linear cooling
        # how much mass we are willing to move in this iteration
        max_frac = 0.25 * (1.0 - prog) + 0.001 * prog

        # locate the (one) index where the convolution is maximal
        k_max = int(np.argmax(b))

        # indices that directly contribute to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            # source – pick proportionally to its contribution to the max entry
            src = rng.choice(n, p=contrib / total)
        else:
            src = int(np.argmax(a))

        # target – pick a place that contributes little (inverse weighting)
        inv = 1.0 / (contrib + 1e-12)
        inv[src] = 0.0
        tgt = rng.choice(n, p=inv / inv.sum())

        if src == tgt:
            continue

        delta = a[src] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----- incremental update of the full convolution (O(n)) -----
        b_old = b.copy()
        # linear part 2·δ·a shifted by tgt  – 2·δ·a shifted by src
        b[tgt:tgt + n] += 2.0 * delta * a
        b[src:src + n] -= 2.0 * delta * a
        # quadratic part
        two_tgt = 2 * tgt
        two_src = 2 * src
        ts = src + tgt
        b[two_tgt] += delta * delta
        b[two_src] += delta * delta
        b[ts] -= 2.0 * delta * delta

        # update the probability vector
        a[src] -= delta
        a[tgt] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis rule for uphill moves
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-14:
                best_val, best_a = new_val, a.copy()
                print(f"[SA] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # revert
            b = b_old
            a[src] = a[src] + delta
            a[tgt] = a[tgt] - delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the state a bit and continue
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a step‑function under a 1000 s budget.
    Returns a list of non‑negative floats (the heights).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final evaluation
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 10 000 gives a good trade‑off between expressiveness
    # and per‑iteration cost on modern hardware.
    # --------------------------------------------------------------
    n = 10000

    # --------------------------------------------------------------
    # Initial construction – symmetric exponential (U‑shaped)
    # --------------------------------------------------------------
    r = 0.998
    i = np.arange(n, dtype=float)
    a0 = r ** i + r ** (n - 1 - i)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    a0 = _symmetrize(a0)                         # enforce exact symmetry

    # --------------------------------------------------------------
    # A quick random‑search warm‑start (cheap, improves the baseline)
    # --------------------------------------------------------------
    best_a = a0
    best_val = _objective_value(best_a)
    for _ in range(12):
        if time.time() > deadline:
            break
        cand = rng.dirichlet(np.ones(n))
        cand = _symmetrize(cand)
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[random] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – Smooth surrogate optimisation (L‑BFGS‑B)
    # --------------------------------------------------------------
    eps_schedule = [
        1e-1, 5e-2, 2e-2, 1e-2, 5e-3,
        2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5
    ]
    # allocate about 60 % of remaining time to this stage
    stage1_deadline = start_time + 0.60 * (deadline - start_time)
    for eps in eps_schedule:
        if time.time() > stage1_deadline:
            break
        # allow up to 200 LBFGS iterations (will stop earlier if converged)
        a_opt = _smooth_optimize(best_a, eps, maxiter=200,
                                 deadline=stage1_deadline)
        a_opt = _symmetrize(a_opt)
        val = _objective_value(a_opt)
        if val < best_val:
            best_val, best_a = val, a_opt.copy()
            print(f"[L‑BFGS ε={eps:.1e}] new best = {best_val:.6f}")
        # keep the best vector as the start for the next (colder) ε
        # (already done because we reuse best_a)

    # --------------------------------------------------------------
    # Stage 2 – Exact local polishing (fast SA)
    # --------------------------------------------------------------
    # use the rest of the budget for SA
    stage2_deadline = deadline
    best_a = _simulated_annealing(best_a,
                                  start=start_time,
                                  deadline=stage2_deadline,
                                  rng=rng,
                                  target_val=1.503)

    # --------------------------------------------------------------
    # Final symmetric clean‑up (tiny uniform mixing)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010):
        cand = (1.0 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[mix α={alpha:.3f}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation (for the competition harness)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Ensure the returned list complies with the security checks:
    # non‑negative, finite numbers, etc.
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a = best_a / total

    return best_a.tolist()
