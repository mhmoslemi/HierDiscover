def annealing_phase(a, start_time, deadline, best_val, best_a):
    n = len(a)
    b = _full_convolution(a)
    cur_val = 2 * n * b.max()
    best_val_local = best_val
    best_a_local = best_a.copy()
    # schedule parameters
    start_frac = 0.5
    end_frac = 0.001
    start_temp = 0.02  # maybe adjust
    end_temp = 1e-6
    iteration = 0
    while time.time() < deadline:
        progress = (time.time() - start_time) / (deadline - start_time)
        max_frac = start_frac * (1 - progress) + end_frac * progress
        temp = start_temp * ((end_temp / start_temp) ** progress)
        # pick source index i based on weight proportional to a (mass) to transfer from high mass areas.
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-16:
            continue
        # pick target j, random not equal i
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # propose delta
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0:
            continue
        # Compute new convolution b_new incrementally
        b_new = b.copy()
        # overlay updates
        b_new[j:j+n] += 2 * delta * a
        b_new[i:i+n] -= 2 * delta * a
        idx2j = 2*j
        idx2i = 2*i
        idxij = i + j
        # ensure indices within range; they are.
        b_new[idx2j] += delta * delta
        b_new[idx2i] += delta * delta
        b_new[idxij] -= delta * delta

        new_val = 2 * n * b_new.max()

        # acceptance
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            # accept
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val_local:
                best_val_local = new_val
                best_a_local = a.copy()
                # print progress
                print(f"[Anneal {iteration}] new best = {best_val_local:.6f}")
        iteration += 1
    return best_a_local, best_val_local

---------

1. **Block representation** – the full vector of length `n` is built by

   (piecewise constant, exponential tails, etc.).

    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
