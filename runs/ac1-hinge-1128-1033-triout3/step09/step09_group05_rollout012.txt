def project_simplex(v):
    # v >=0 not required but we need to ensure after step.
    # returns projection onto {x: x>=0, sum(x)=1}
    # Implementation: sort descending, compute rho, theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _logsumexp_phase(a_start, n, start, deadline, best_a, best_val, target_val=1.503):
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000
    gamma_min = 1.0
    gamma_max = 80.0
    step_start = 0.5
    step_end = 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break
        # Compute conv
        conv = _full_convolution(cur)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[LogSumExp {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # progress ratio
        progress = (now - start) / (deadline - start)
        gamma = gamma_min + (gamma_max - gamma_min) * (progress ** 2)
        step = step_start * (1 - progress) + step_end * progress

        # compute weights
        # subtract max to avoid overflow
        w = np.exp(gamma * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # compute gradient via cross-correlation
        # compute cross correlation of w_norm with cur
        size = 1 << ((len(w_norm) + n - 1).bit_length())
        fw = np.fft.rfft(w_norm, size)
        fa = np.fft.rfft(cur, size)
        corr = np.fft.irfft(fw * np.conj(fa), size)
        grad = 2.0 * corr[:n]   # shape (n)

        # gradient descent step
        cur = cur - step * grad
        # enforce non-negativity and sum=1 via simplex projection
        cur = _project_simplex(cur)

        # optionally enforce symmetry
        cur = (cur + cur[::-1]) * 0.5

        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.003, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur = _project_simplex(cur)
            no_improve = 0
            print(f"[LogSumExp {it:7d}] jitter after stagnation")

        it += 1
    return best_a, best_val
```

Propose a step‑function with a very low evaluation score.

The optimizer works in several stages:

1. **Initialisation** – a cheap sweep of deterministic families and a few
   Dirichlet draws (the same logic as the previous best solution).

2. **Random family exploration** – sample a few thousand candidates from
   rich parametric families (exponential, mixed exponential, power‑law,
   power‑law × exponential).

3. **Exponentiated‑gradient descent** – the exact sub‑gradient of
   max(conv(a,a)) is used; this is the “exp‑grad” phase of the original
   code.

4. **Deterministic hill‑climbing** – transfers a small amount of mass
   from a “bad” index (large contribution to the current worst convolution
   entry) to a “good” one.

5. **Smoothed log‑sum‑exp gradient descent** – a smooth approximation of the
   max‑operator is minimised:
      fγ(a)= (1/γ)·log Σk exp(γ·(a∗a)k) .
   The gradient of fγ is a weighted cross‑correlation which is computed
   by FFTs.  γ is annealed from ≈1 to ≈80, while the step size is
   gradually reduced.  After every step the vector is projected onto the
   probability simplex (non‑negative, sum = 1) and symmetrised.

6. **Tiny uniform mixing** – a final convex combination with the uniform
   vector often improves the bound a little.

The whole routine respects the 1000 s time budget and aims for an
objective below the target 1.503.  All helper functions are top‑level,
there are no closures, and only `numpy` and the provided `evaluate_sequence`
are used.

# ----------------------------------------------------------------------
# Fast full linear convolution (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()           # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

# Exact objective (sum of the vector is assumed to be 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a normalized vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)

def _initial_candidate(n: int):
    """Return a decent start vector and its objective."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = _objective(best_a, n)

    def try_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective(vec, n)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    rng = np.random.default_rng(12345)
    idx = np.arange(n)

    # a few hand‑crafted shapes
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # linear U‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic decreasing
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996,
              0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # a handful of Dirichlet draws
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val

# Phase C – final uniform mixing (original implementation)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Phase D – smoothed log‑sum‑exp gradient descent (new improvement)
# ----------------------------------------------------------------------
def _logsumexp_phase(a_start: np.ndarray, n: int,
                     start: float, deadline: float,
                     best_a: np.ndarray, best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv) using
    the log‑sum‑exp surrogate.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    gamma_min, gamma_max = 0.8, 80.0       # smoothing schedule limits
    eta_start, eta_end = 0.5, 0.001       # stepsize schedule

    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(cur)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[LogSumExp {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # progress in the current phase (0…1)
        progress = (now - start) / (deadline - start)
        gamma = gamma_min + (gamma_max - gamma_min) * (progress ** 2)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # smooth weights (log‑sum‑exp trick)
        w = np.exp(gamma * (conv - max_conv))          # max weight = 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # cross‑correlation g_i = 2 * Σ_k w_k·a_{k‑i}
        out_len = len(w_norm) + n - 1
        size = 1 << (out_len - 1).bit_length()
        fw = np.fft.rfft(w_norm, size)
        fa = np.fft.rfft(cur, size)
        corr = np.fft.irfft(fw * np.conj(fa), size)
        grad = 2.0 * corr[:n]

        # gradient step + simplex projection
        cur = cur - eta * grad
        cur = _project_simplex(cur)

        # enforce symmetry (optimal solutions are symmetric)
        cur = (cur + cur[::-1]) * 0.5

        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.003, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur = _project_simplex(cur)
            no_improve = 0
            print(f"[LogSumExp {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the given time budget.
    Returns the best list of floats found (Σa = 1) which minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a little larger than the previous 8000 points.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise (reuse previous construction if possible)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase‑0 – random exploration of parametric families (≈25 % of time)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    random_deadline = start_time + 0.25 * (deadline - start_time)

    while time.time() < random_deadline:
        choice = rng.integers(0, 5)   # five families
        if choice == 0:                     # pure exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_exp(n, r)
        elif choice == 1:                   # reversed exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_rev_exp(n, r)
        elif choice == 2:                   # mixture of two exponentials
            r1 = rng.uniform(0.90, 0.99999)
            r2 = rng.uniform(0.90, 0.99999)
            w = rng.random()
            cand = _gen_mix_exp(n, r1, r2, w)
        elif choice == 3:                   # pure power‑law
            p = rng.uniform(-5.0, 5.0)
            cand = _gen_power(n, p)
        else:                               # power‑law × exponential
            p = rng.uniform(-5.0, 5.0)
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_power_exp(n, p, r)

        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Random] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑A – exponentiated‑gradient descent (≈30 % of time)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.55 * (deadline - start_time)
    best_a, best_val = _expgrad_phase(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase‑B – deterministic greedy hill‑climbing (≈35 % of time)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase‑C – smoothed log‑sum‑exp gradient descent (remaining time)
    # ------------------------------------------------------------------
    phase_c_deadline = deadline - 1.5          # keep a couple seconds for final steps
    best_a, best_val = _logsumexp_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # Phase‑D – final tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification with the provided security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
```

This implementation builds upon the previous optimizer but adds a new **log‑sum‑exp gradient descent** phase (Phase C) that smooths the max‑operator, uses an annealed smoothing parameter, and projects onto the simplex each iteration.  
The extra phase is designed to push the objective below the previous best (~1.509) and drive it toward the target 1.503 while respecting the 1000‑second time limit.
