Implement projection onto simplex: algorithm by Duchi et al. (2008). Works for any vector.

Pseudo:

def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv/(np.arange(len(v))+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    # returns projection onto simplex {x: x>=0, sum x = 1}
    v = np.asarray(v, dtype=float)
    # ensure shape (n,)
    if v.ndim != 1:
        v = v.ravel()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:

        return np.full_like(v, 1.0 / len(v))

    w = np.maximum(v - theta, 0)

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of a real vector (length 2·len(x)‑1) using FFT.
    Small negative round‑off values are clipped to zero.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·(a*a)_i w.r.t. a.
    The formula is g = 2·(a * w_rev) where * denotes linear convolution.
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3·n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1                 # = 2·n‑2
    seg = conv[start:start + n]                # length n
    return 2.0 * seg


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    v = np.asarray(v, dtype=float)
    if v.ndim != 1:
        v = v.ravel()
    n = v.shape[0]

    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # eliminate tiny numerical drift
    if w.sum() == 0:
        return np.full_like(v, 1.0 / n)
    w /= w.sum()
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence` within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a few seconds free for final evaluation/return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 20 000 coefficients give a good trade‑off.
    # ------------------------------------------------------------------
    n = 20000

    # ------------------------------------------------------------------
    # Initialise – use historic construction if available, otherwise Dirichlet.
    # ------------------------------------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, a)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        a = rng.exponential(scale=1.0, size=n)
        a = np.maximum(a, 0.0)
        a /= a.sum()

    # ------------------------------------------------------------------
    # Store the best solution seen so far.
    # ------------------------------------------------------------------
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"[init] n={n}, objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # A few random restarts to avoid an unlucky start.
    # ------------------------------------------------------------------
    for tr in range(5):
        if time.time() > deadline - 200:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        cur = 2.0 * n * _full_convolution(a_rand).max()
        if cur < best_val:
            best_val = cur
            best_a = a_rand.copy()
            a = a_rand.copy()
            print(f"[rand init {tr}] new best ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam projected gradient descent on a smooth max.
    # ------------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    t = 0

    # hyper‑parameters
    lr0 = 0.5                       # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    beta_start = 0.1
    beta_end = 2000.0
    beta_ramp = 120_000            # iterations over which β grows
    max_iter = 300_000
    last_improve = 0
    best_obj = best_val

    while time.time() < deadline and t < max_iter:
        t += 1

        # ----- compute self‑convolution and smooth weights -----
        conv = _full_convolution(a)
        max_c = conv.max()
        cur_obj = 2.0 * n * max_c

        # update best if we have a strict improvement
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = t
            print(f"[gd iter {t}] new best ≈ {best_obj:.6f}")

        # smoothing parameter β
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / beta_ramp)

        # weights that approximate the max (largest entry → 1)
        w = np.exp(beta * (conv - max_c))          # shape 2n‑1, w[max]=1

        # ----- gradient of the smooth objective -----
        g = _gradient_weighted(a, w[::-1])        # shape n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat

        # ----- projection onto the simplex (keeps sum = 1, non‑negative) -----
        a = _project_simplex(a_new)

        # ----- occasional re‑ordering (anti‑symmetric) -----
        if t % 5000 == 0:
            a_ord = _anti_symmetric_reorder(a)
            obj_ord = 2.0 * n * _full_convolution(a_ord).max()
            if obj_ord < best_obj - 1e-12:
                best_obj = obj_ord
                best_a = a_ord.copy()
                a = a_ord.copy()
                last_improve = t
                print(f"[gd reorder {t}] improved to ≈ {best_obj:.6f}")

        # ----- stagnation handling – blend with fresh random Dirichlet -----
        if t - last_improve > 30_000:
            noise = rng.exponential(scale=1.0, size=n)
            noise /= noise.sum()
            a = 0.7 * a + 0.3 * noise
            a = _project_simplex(a)
            last_improve = t
            print(f"[gd perturbed at {t}]")

        # ----- light logging (call the heavy evaluate_sequence only rarely) -----
        if t % 10_000 == 0:
            eval_val = evaluate_sequence(a.tolist())
            if eval_val < best_obj:
                best_obj = eval_val
                best_a = a.copy()
                print(f"[eval] iteration {t}, external score = {eval_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – fast hill‑climbing refinement (random swaps / mass transfers).
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_obj

    while time.time() < deadline:
        # choose an elementary local move
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = _project_simplex(new_a)

        # fast objective evaluation (no call to evaluate_sequence)
        new_obj = 2.0 * n * _full_convolution(new_a).max()

        if new_obj < cur_val - 1e-12:
            cur_a = new_a
            cur_val = new_obj
            best_a = new_a.copy()
            best_obj = new_obj
            print(f"[refine] improvement → {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided routine and return.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] reported score = {final_score:.6f}")
    return best_a.tolist()
