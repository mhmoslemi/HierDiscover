def smooth_max_descent(a0, n, start_time, deadline, best_a, best_val):
    a = a0.copy()
    iteration = 0
    beta = 5.0
    while time.time() < deadline:
        # compute b
        b = full_convolution(a)
        max_b = b.max()
        cur_val = 2 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[SmoothGrad {iteration}] new best {best_val:.6f}")
            if cur_val <= 1.503:
                break
        # compute weight vector
        # at first iterations, small beta for exploration
        # schedule beta increase
        elapsed = (time.time() - start_time) / (deadline - start_time)
        beta = 5.0 + 95.0 * elapsed  # grows from 5 to 100
        # compute w
        w = np.exp(beta * (b - max_b))  # max exponent 0
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        conv = full_convolution(a, w_rev)  # length 3n-2
        idx_base = 2 * n - 2
        g = 2.0 * conv[idx_base - np.arange(n)]
        # step size schedule
        eta = (1.0 - elapsed) * 0.5 + elapsed * 0.0005
        # multiplicative update
        a = a * np.exp(-eta * g)
        # Renormalize
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total
        iteration += 1
    return best_a, best_val

def full_convolution(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv

"""
Optimise a non‑negative step function to minimise ``evaluate_sequence``.
The new optimiser replaces the “top‑K” sub‑gradient by a *smooth max*:
for a vector a (∑a=1) we consider

    Lβ(a) = (1/β)·log(∑_k exp(β·b_k)),   b = a∗a (full linear convolution)

Lβ → max(b) when β→∞, but for finite β the gradient uses *all* convolution
entries, which gives a much more informative direction than the previous
top‑K heuristic.  The gradient can be written in closed form

    ∂Lβ/∂a_i = 2·∑_j a_j·w_{i+j} ,

where w_k ∝ exp(β·b_k) and the sum runs over all admissible indices.
Both convolutions (a∗a and a∗w_rev) are performed with FFTs, so each
iteration costs O(n log n) and we can afford thousands of iterations in the
1000 s budget.

The optimiser consists of three stages:
  1. A *warm‑up* smooth‑gradient descent (large time slice, slowly decreasing
     step size and increasing β) to obtain a good starting point.
  2. A deterministic hill‑climb that moves a small fraction of mass from the
     worst contributor of the current peak to the best contributor.
  3. A few tiny convex mixes with the uniform vector.

Multiple random restarts (Dirichlet draws with a log‑uniform concentration
parameter) are interleaved in the remaining budget; each restart runs a short
smooth‑gradient phase followed by the hill‑climb and final mixing.  The best
vector seen across **all** restarts is returned.

All vectors are kept on the probability simplex (non‑negative, sum = 1) so
the value of ``evaluate_sequence`` is invariant under scaling, and the final
call to ``evaluate_sequence`` validates the result.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution via FFT (returns the full length 2·len(a)‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, b: np.ndarray | None = None) -> np.ndarray:
    """Linear convolution of a with b (or a with itself if b is None)."""
    if b is None:
        b = a
    n = a.shape[0]
    m = b.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective used inside the optimiser (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a∗a)  (identical to evaluate_sequence for a normalised vector)."""
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Smooth‑max gradient descent
# ----------------------------------------------------------------------
def _smooth_descent_phase(a0: np.ndarray,
                          n: int,
                          start_time: float,
                          deadline: float,
                          best_a: np.ndarray,
                          best_val: float,
                          target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Multiplicative update using the gradient of
    Lβ(a) = (1/β)·log∑exp(β·b_k)  where b = a∗a.
    β grows from β_start to β_end; the step size η decays linearly.
    """
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    # schedules
    beta_start, beta_end = 1.0, 200.0   # start smooth, end close to true max
    eta_start, eta_end = 0.30, 0.0005

    while True:
        now = time.time()
        if now >= deadline:
            break

        # convolution and objective (exact max for early stopping)
        b = _full_convolution(cur)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # progress in [0,1]
        prog = (now - start_time) / (deadline - start_time)
        beta = beta_start * (1.0 - prog) + beta_end * prog
        eta = eta_start * (1.0 - prog) + eta_end * prog

        # smooth weights (soft‑max) – normalised to sum to 1
        w = np.exp(beta * (b - max_b))          # max exponent = 0 → numeric safe
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient: g_i = 2·∑_j a_j·w_{i+j}
        w_rev = w[::-1]                         # reversal needed for the convolution formula
        conv = _full_convolution(cur, w_rev)    # length = 3·n‑2
        idx_base = 2 * n - 2
        g = 2.0 * conv[idx_base - np.arange(n)]

        # multiplicative (exponentiated) update – stays in the simplex automatically
        cur = cur * np.exp(-eta * g)

        # renormalise to Σa = 1
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional random jitter if we are stuck
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.008, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[Smooth {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climb (identical to the one used previously)
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     start_frac: float = 0.30,
                     end_frac: float = 0.001,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Mass transfer from the worst to the best contributor of the active peak."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Hill   {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal the transfer fraction
        prog = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - prog) + end_frac * prog

        # index of the current maximal convolution entry
        k = int(np.argmax(b))

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)

        # contribution of each index to the peak (a_{k-i})
        contrib = cur[k - idx]

        src = int(np.argmax(contrib))
        rev = contrib.copy()
        rev[src] = np.inf
        tgt = int(np.argmin(rev))

        if cur[src] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[Hill   {it:7d}] jitter after stagnation")
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # keep the move
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector (keeps the simplex constraint)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to obtain a deterministic start vector (uniform or simple shapes)
# ----------------------------------------------------------------------
def _initial_candidate(n: int) -> tuple[np.ndarray, float]:
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)
    rng = np.random.default_rng(0)

    # a few deterministic shapes
    for cand in [
        np.arange(n, 0, -1, dtype=float),                     # strictly decreasing
        np.maximum(np.arange(n, dtype=float) + 1,
                   n - np.arange(n, dtype=float)),            # V‑shape
        (np.arange(n, 0, -1, dtype=float) ** 2),             # quadratic decay
        np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))),      # half‑wave sine
        1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)),   # cosine ends
    ]:
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()

    # a few random Dirichlet draws
    for _ in range(8):
        cand = rng.dirichlet(np.ones(n))
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()

    return best_a, best_val
