def smoothed_descent_new(a0, eps_schedule, eta_initial, iters_per_eps):
    a = a0.copy()
    w = np.log(a + 1e-30)
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    for eps idx...
        for it in range(iters_per_eps):
            if out of time break
            conv = compute convolution
            max_conv = conv.max()
            # compute softmax weights
            shift = np.max(conv)  # for stability
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()
            # gradient
            grad = 2.0 * gradient_weighted_peaks(a, p)  # shape n
            # mirror descent step in log-space
            w -= eta * grad
            a = np.exp(w)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full_like(a, 1.0 / len(a))
                w = np.log(a)
                continue
            a /= a_sum
            # occasionally evaluate true objective
            if it % 10 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val:
                    best_val = cur_val; best_a = a.copy()
            # occasionally reorder
            if it % 200 == 0:
                a = anti_symmetric_reorder(a)
                w = np.log(a + 1e-30)
        # after eps stage reorder again
        a = anti_symmetric_reorder(a)
        w = np.log(a + 1e-30)
    return best_a, best_val

"""
Improved search for a step‑function that minimises `evaluate_sequence`.

Algorithm Overview
-------------------
1. **Mirror‑descent on a smooth approximation of the max‑convolution**  
   * Soft‑max surrogate:   f_ε(a) = (1/ε)·log Σ exp(ε·(a*a)).  
   * Gradient of the surrogate is obtained with a single FFT‑based
     convolution (weighted peaks).  
   * Updates are performed in log‑space (entropic mirror descent) :
        w ← w – η·∇f_ε ,   a = softmax(w) .
   * The smoothing parameter ε follows a decreasing schedule
     (0.5 → … → 0.001) to first smooth the landscape and finally focus on
     the true ℓ∞‑norm.

2. **Anti‑symmetric re‑ordering**  
   After each ε‑stage (and regularly inside the loop) the vector is
   reordered by pairing large and small entries.  This never increases the
   convolution peak (rearrangement inequality) and helps escape plateaus.

3. **Multiple restarts**  
   Several diverse initialisations (uniform, sinusoidal, exponential,
   mixed spikes) plus an optional warm‑start from a previously found
   construction (`height_sequence_1`).  Each restart runs the descent
   described above.

4. **Local hill‑climbing**  
   A cheap stochastic local search (random swaps / tiny mass transfers)
   is applied to the best vector of each restart and finally to the
   global best.  Only strictly improving moves are kept.

The routine respects a 1000 s time budget, prints intermediate progress
and finally returns the best coefficient list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using FFT.
    Returns a non‑negative array.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negative numbers caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute Σ_k p_k·a_{k‑i} for all i (gradient of Σ p_k·(a*a)[k]).
    Implements it with a single convolution: reverse p and convolve with a.
    """
    n = a.shape[0]
    p_rev = p[::-1]                     # reversed weight vector
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the slice corresponding to indices n‑1 … 2n‑2 (inclusive)
    seg = conv[n - 1 : 2 * n - 1]
    # reverse back to obtain the gradient vector
    g = seg[::-1]
    return g


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b

# Smoothed‑max (log‑sum‑exp) mirror descent
# ----------------------------------------------------------------------


def _smoothed_descent_new(a_start: np.ndarray,

    Mirror‑descent minimisation of a smooth surrogate of max(a*a).
    Returns the best vector (as np.ndarray) and its true evaluation.
    """
    # make sure entries are strictly positive before taking logs
    eps_nan = 1e-30
    a = np.maximum(a_start, eps_nan)
    w = np.log(a)                       # entropic dual variable
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent] start objective = {best_val:.6f}")

    # total number of inner iterations (adjusted on the fly)
    # we aim for a few thousand FFT steps per restart
    max_iters_per_eps = 2000

    for eps in eps_schedule:
        # initial step size – larger for large eps, smaller later
        eta = 0.6 / (1.0 + eps)

        for it in range(max_iters_per_eps):
            # time guard
            if time.time() >= deadline - 0.5:
                break

            # ----- convolution and soft‑max weights -----
            conv = _full_convolution(a)          # length 2n‑1
            shift = conv.max()                   # stability shift
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()        # Σp = 1, p≥0

def _hill_climb_refine(a: np.ndarray,
                       cur_val: float,
                       deadline: float,
                       n: int):
    """
    Light local search: random swaps or tiny mass transfers.
    Only strictly improving moves are accepted.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline - 0.5:
        # decide operation
        if rng.random() < 0.3:                     # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # transfer a small amount
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.05 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto the simplex (non‑negativity + sum=1)
        new_a = np.maximum(new_a, 0.0)
        total = new_a.sum()
        if total <= 0.0:
            continue
        new_a /= total

        val = evaluate_sequence(new_a.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] new best = {best_val:.6f}")

    return best_a, best_val

        s0 = a0.sum()
        if s0 > 0.0:
            a0 /= s0
        else:
            a0 = np.full(n, 1.0 / n)
    except Exception:
        a0 = None

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # schedule of smoothing parameters (large → small)
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    restart = 0
    max_restarts = 8

    # ------------------------------------------------------------------
    # Main optimisation loop (multiple restarts)
    # ------------------------------------------------------------------
    while time.time() < deadline - 10 and restart < max_restarts:
        restart += 1

        # ----- choose an initial vector -----
        if a0 is not None and restart == 1:
            a_start = a0.copy()
            init_name = "previous"
        else:
            strat = rng.choice(["uniform", "sin", "exp", "mix"])
            if strat == "uniform":
                a_start = np.full(n, 1.0 / n)
                init_name = "uniform"
            elif strat == "sin":
                idx = np.arange(n) + 0.5
                a_start = np.sin(np.pi * idx / n)
                a_start = np.maximum(a_start, 0.0)
                a_start = a_start / a_start.sum()
                init_name = "sinusoid"
            elif strat == "exp":
                a_start = rng.exponential(scale=1.0, size=n)
                a_start = a_start / a_start.sum()
                init_name = "exp"
            else:   # mixed spikes
                a_start = np.zeros(n)
                # place a few random spikes
                spike_pos = rng.integers(0, n, size=30)
                a_start[spike_pos] = rng.random(30) + 0.1
                # add a small uniform background
                a_start += 0.001 * rng.random(n)
                a_start = a_start / a_start.sum()
                init_name = "mix"

        # ----- run the smoothed‑max descent -----
        a_opt, val_opt = _smoothed_descent_new(a_start, eps_schedule, deadline)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[global restart {restart} ({init_name})] new global best = {best_val:.6f}")

        # ----- short local hill‑climb after each descent (if time permits) -----
        if time.time() < deadline - 5:
            a_ref, val_ref = _hill_climb_refine(best_seq, best_val, deadline, n)
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[global after hill] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing on the best vector found so far
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        best_seq, best_val = _hill_climb_refine(best_seq, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_seq.tolist()
