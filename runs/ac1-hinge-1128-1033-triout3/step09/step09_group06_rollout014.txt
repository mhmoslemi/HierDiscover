"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  The optimiser works in three phases:

1. **Diverse initialisation** – a handful of restarts with different
   simple constructions (uniform, random Dirichlet, edge‑exponential,
   V‑shaped power laws).  The initial vectors are placed on the simplex
   and anti‑symmetric‑reordered (largest values on the left, smallest on
   the right) – a transformation that never worsens the max‑convolution.

2. **Smoothed‑max descent** – minimises a log‑sum‑exp surrogate of the
   true objective using multiplicative (exponentiated‑gradient) updates.
   The smoothing parameter ε is decreased from 1.0 down to 0.001, with a
   fixed budget of FFT‑based steps (≈20 000 overall).  The current best
   true objective is evaluated periodically.

3. **Fine‑grained refinement** – a short sub‑gradient descent that
   targets the actual max‑convolution index, followed by a cheap stochastic
   hill‑climb (random swaps / tiny mass transfers).  The best vector found
   across all restarts is returned.

All operations respect the 1000 s time budget and the simplex constraint
(`sum a = 1`, `a_i ≥ 0`).  The algorithm is completely self‑contained and
relies only on `numpy` and the supplied `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (FFT‑based convolutions)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·(a*a)_k with respect to a.
    The weight vector p must have length 2·len(a)‑1 and sum to 1.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    p_rev = p[::-1]                         # reversed weight
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # relevant slice corresponds to indices n‑1 … 2n‑2 (inclusive)
    seg = conv[n - 1 : 2 * n - 1]
    # reverse again to obtain the gradient
    g = seg[::-1]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]            # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Arrange the sorted values so that the largest entries occupy the left
    side of the vector while the smallest occupy the right side.  This
    never increases the maximum of the self‑convolution.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Initialisers – a few parametric families
# ----------------------------------------------------------------------


def _uniform_init(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n)


def _random_dirichlet_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑like vector via independent exponentials."""
    a = rng.exponential(scale=1.0, size=n)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else _uniform_init(n)


def _vshape_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    V‑shaped power law: a_i ∝ ((i+1)^(-p) + (n-i)^(-p))^q,
    optionally mixed with a uniform component.
    """
    p = rng.uniform(0.2, 4.0)           # controls edge steepness
    q = rng.uniform(0.5, 2.5)           # accentuates extremes
    beta = rng.uniform(0.0, 1.0)        # mixing weight with uniform part
    i = np.arange(n)
    base = ((i + 1) ** -p + (n - i) ** -p) ** q
    a = (1 - beta) * np.ones_like(base) + beta * base
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else _uniform_init(n)


def _edge_exponential_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Edge‑exponential: a_i ∝ exp(-p * min(i, n‑1-i)),
    mixed with a uniform component.
    """
    p = rng.uniform(0.2, 4.0)
    beta = rng.uniform(0.0, 1.0)
    i = np.arange(n)
    dist = np.minimum(i, n - 1 - i)
    base = np.exp(-p * dist)
    a = (1 - beta) * np.ones_like(base) + beta * base
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else _uniform_init(n)


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) descent
# ----------------------------------------------------------------------


def _smoothed_descent(a_start: np.ndarray,
                      deadline: float,
                      max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the log‑sum‑exp surrogate of
    max‑convolution.  Returns the best vector observed and its true
    evaluation value.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent] start value = {best_val:.6f}")

    # epsilon schedule – from coarse to very fine
    eps_schedule = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    # allocate iterations proportionally (more for larger eps)
    total_weight = sum(1.0 / (e + 0.1) for e in eps_schedule)
    it_per_eps = [max(1, int(max_iters * (1.0 / (e + 0.1)) / total_weight))
                  for e in eps_schedule]

    for eps, it_budget in zip(eps_schedule, it_per_eps):
        eta = 0.7                       # base step size for this epsilon

        for it in range(it_budget):
            if time.time() >= deadline - 5.0:
                break

            # ---- convolution & softmax weights ----
            conv = _full_convolution(a)            # length 2n‑1
            max_conv = conv.max()
            shift = max_conv                       # for numerical stability
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()          # Σp = 1, p≥0

            # ---- gradient of the smooth surrogate ----
            grad = 2.0 * _gradient_weighted_peaks(a, p)   # shape (n,)

            # ---- multiplicative mirror‑descent step ----
            delta = -eta * grad
            delta = np.clip(delta, -20.0, 0.0)       # avoid overflow
            a = a * np.exp(delta)
            a = _project_simplex(a)                 # stay on simplex

            # occasional true evaluation
            if it % 50 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.3f} it {it}] new best = {best_val:.6f}")

            # reorder every 200 steps – never hurts the max
            if it % 200 == 0:
                a = _anti_symmetric_reorder(a)

            # slowly decay the step size (helps stability at small eps)
            eta *= 0.9995

        # final reorder for the current epsilon stage
        a = _anti_symmetric_reorder(a)

        # early‑exit if time is almost up
        if time.time() >= deadline - 5.0:
            break

    # final evaluation before returning
    final_val = evaluate_sequence(a.tolist())
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
        print(f"[descent] final best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement targeting the true max index
# ----------------------------------------------------------------------


def _subgradient_refine(a_start: np.ndarray,
                        deadline: float,
                        max_steps: int = 2500,
                        eta: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that always follows the *actual* max‑convolution
    entry.  The update is multiplicative and projected onto the simplex.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())

    step = 0
    while time.time() < deadline - 3.0 and step < max_steps:
        conv = _full_convolution(a)
        k = int(conv.argmax())

        # gradient for a single active index k
        grad = np.zeros_like(a)
        if k < n:
            # indices 0 .. k contribute
            grad[: k + 1] = 2.0 * a[k::-1]
        else:
            start = k - (n - 1)                 # first index that receives mass
            grad[start:] = 2.0 * a[: n - start][::-1]

        delta = -eta * grad
        delta = np.clip(delta, -20.0, 0.0)
        a = a * np.exp(delta)
        a = _project_simplex(a)

        if step % 10 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[subgrad] step {step} new best = {best_val:.6f}")

        step += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Light stochastic hill‑climbing (random swaps / tiny transfers)
# ----------------------------------------------------------------------


def _hill_climb_refine(a: np.ndarray,
                       cur_val: float,
                       deadline: float,
                       n: int,
                       max_trials: int = 10_000) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: random swaps or tiny mass transfers,
    kept only if they *strictly* improve the true objective.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    trials = 0
    while time.time() < deadline - 2.0 and trials < max_trials:
        if rng.random() < 0.25:                 # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                trials += 1
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                   # transfer a tiny amount
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                trials += 1
                continue
            delta = best_a[i] * 0.03 * rng.random()   # up to 3 % of mass at i
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto simplex (just to be safe against rounding)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            trials += 1
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] new best = {best_val:.6f}")

        trials += 1

    return best_a, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # problem size – 8 000 entries gave a good trade‑off in earlier experiments
    n = 8000

    # ------------------------------------------------------------------
    # Try to reuse a previously found construction, if available
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to desired length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        a0 = a0 / a0_sum if a0_sum > 0 else _uniform_init(n)
    except Exception:
        a0 = None

    best_a = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets its own descent + refinement
    # ------------------------------------------------------------------
    restart = 0
    max_restarts = 8
    while time.time() < deadline - 200 and restart < max_restarts:
        restart += 1

        # ----------------------------------------------------------------
        # Choose an initial vector (different strategy each restart)
        # ----------------------------------------------------------------
        if a0 is not None and restart == 1:
            a = a0.copy()
            init_name = "previous"
        else:
            strat = rng.choice(["uniform", "random", "vshape", "edge_exp"])
            if strat == "uniform":
                a = _uniform_init(n)
                init_name = "uniform"
            elif strat == "random":
                a = _random_dirichlet_init(n, rng)
                init_name = "random"
            elif strat == "vshape":
                a = _vshape_init(n, rng)
                init_name = "vshape"
            else:  # edge_exponential
                a = _edge_exponential_init(n, rng)
                init_name = "edge_exp"

        # enforce anti‑symmetric ordering (never hurts the objective)
        a = _anti_symmetric_reorder(a)

        # ----------------------------------------------------------------
        # Phase 1 – smoothed‑max descent
        # ----------------------------------------------------------------
        a_opt, val_opt = _smoothed_descent(a, deadline)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[global] restart {restart} ({init_name}) new best = {best_val:.6f}")

        # ----------------------------------------------------------------
        # Phase 2 – sub‑gradient refinement (focus on true max)
        # ----------------------------------------------------------------
        if time.time() < deadline - 30:
            a_sub, val_sub = _subgradient_refine(best_a, deadline, max_steps=2500, eta=0.30)
            if val_sub < best_val:
                best_val = val_sub
                best_a = a_sub.copy()
                print(f"[global] after sub‑grad new best = {best_val:.6f}")

        # ----------------------------------------------------------------
        # Phase 3 – cheap hill‑climb local search
        # ----------------------------------------------------------------
        if time.time() < deadline - 10:
            a_hill, val_hill = _hill_climb_refine(best_a, best_val, deadline, n)
            if val_hill < best_val:
                best_val = val_hill
                best_a = a_hill.copy()
                print(f"[global] after hill‑climb new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polish on the overall best vector
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5:
        best_a, best_val = _hill_climb_refine(best_a, best_val, deadline, n)
        print(f"[final] polished best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – uniform if everything failed (should never happen)
        best_a = _uniform_init(n)

    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return final_seq
