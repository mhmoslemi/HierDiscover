"""
Non‑negative step‑function optimiser.

Phase 1 – Fast “MSE‑gradient descent”:
    minimise the squared deviation of the full self‑convolution a*a from the
    perfectly flat target μ = 1/(2·n‑1).  The gradient is computed analytically
    via a single FFT‑based convolution, a simple additive update is taken,
    and the iterate is projected onto the probability simplex (Σa=1, a≥0).

Phase 2 – Refined simulated‑annealing hill‑climbing (the same strategy that
    was used before, but started from the much better MSE‑solution).

Both phases respect the 1000 s budget, keep the best‑ever candidate, and
finally call ``evaluate_sequence`` for the secure score.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# FFT based full linear convolution (generic – works for two different vectors)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of `x` and `y` using FFT (real‑valued, full length)."""
    len1 = x.shape[0]
    len2 = y.shape[0]
    out_len = len1 + len2 - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:out_len]
    return conv


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a * a."""
    return _full_convolution_fft(x, x)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Σv = 1, v_i ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector `v` onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # extreme case – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))   (identical to evaluate_sequence for Σa=1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# MSE‑gradient of ½·∑(conv(a,a)‑μ)²   where μ = 1/(2·n‑1)
# ----------------------------------------------------------------------
def _grad_mse(a: np.ndarray) -> np.ndarray:
    """Return gradient of the MSE loss w.r.t. `a`."""
    n = a.shape[0]
    # convolution a*a
    conv = _full_convolution(a)                     # length 2n‑1
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu                                 # (a*a)‑μ
    a_rev = a[::-1]
    # gradient = 2·(diff * a_rev) evaluated on the central slice
    tmp = _full_convolution_fft(diff, a_rev)        # length 3n‑2
    grad = 2.0 * tmp[n - 1 : n - 1 + n]              # slice of length n
    return grad


# ----------------------------------------------------------------------
# Phase‑1: plain gradient descent on the MSE loss
# ----------------------------------------------------------------------
def _phase1_mse_descent(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """
    Runs additive gradient descent on the MSE loss while projecting onto
    the simplex.  Keeps the best candidate seen so far.
    """
    a = a0.copy()
    cur_val = best_val
    prev_val = cur_val

    # step‑size schedule (large → small)
    eta_start, eta_end = 0.5, 5e-4
    stagnation_counter = 0
    STAGNATE_LIMIT = 8_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress ∈[0,1] for the whole phase‑1 interval
        prog = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - prog) + eta_end * prog

        # gradient of the MSE loss
        grad = _grad_mse(a)

        # additive update + simplex projection
        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        # compute objective value (fast, no extra clipping)
        b = _full_convolution(a_new)
        new_val = 2.0 * n * b.max()

        # accept if we made progress, otherwise keep the old iterate
        if new_val <= prev_val:
            a = a_new
            prev_val = new_val
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        # store global best
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[MSE] new global best = {best_val:.6f}")

        # occasional random perturbation on stagnation
        if stagnation_counter > STAGNATE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            prev_val = _objective(a)
            stagnation_counter = 0
            print("[MSE] stagnation – perturb")

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper – cheap Beta‑family sweep (kept for possible warm‑starts)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑2: intensive hill‑climbing (identical to the previous version)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from an index that
    contributes heavily to the current maximal convolution entry to an index
    that contributes little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule – deliberately stretched so the search lasts until the deadline
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each position to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – draw proportional to its contribution
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – biased towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # amount of mass to move
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # handling stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,

                     **kwargs) -> list[float]:
    """

    ``evaluate_sequence``.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start from a previously saved solution (if any)

    a0 = None

        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:

    try:

            print(f"[init] beta‑family improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # a few random Dirichlet draws for diversity
    for _ in range(3):
        rand_a = np.random.dirichlet(np.ones(n))
        val = _objective(rand_a)
        if val < best_val:
            best_a, best_val = rand_a, val
            print(f"[init] random Dirichlet improved value={best_val:.6f}")

    if best_a is None:

        best_a /= best_a.sum()
        best_val = _objective(best_a)

    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
