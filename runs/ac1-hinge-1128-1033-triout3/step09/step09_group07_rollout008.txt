"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence` within a 1000 s budget.

Algorithm
---------
1. **Initialisation** – draw a Dirichlet (exponential) vector, then apply an
   *anti‑symmetric reordering* that places the largest coefficients on the
   left side and the smallest on the right.  The resulting shape is a good
   heuristic for a low self‑convolution peak.

2. **Phase 1 – smoothed exponential‑mirror descent**  
   We minimise a smooth approximation of the ℓ∞‑norm
   `max(c)`, where `c = a * a` is the self‑convolution.
   The approximation uses soft‑max weights
   `w_k = exp(β·(c_k – max(c)))`.  
   The gradient of `∑ w_k·c_k` w.r.t. `a` can be written as a single
   convolution, so each iteration costs only a few FFTs.
   The smoothing parameter `β` is increased from 1 up to a few thousand,
   while a multiplicative mirror‑descent step with a slowly decaying step
   size `α` keeps the vector on the simplex.

3. **Phase 2 – max‑focused sub‑gradient descent**  
   After the smooth phase we concentrate on the *actual* maximal entry.
   Let `I = {k | (a*a)_k = max}`.  Using the indicator weights on these
   indices we obtain a true sub‑gradient of the ℓ∞‑objective
   (`g_i = 2·∑_{k∈I} a_{k-i}`) via another convolution and take a tiny
   multiplicative step.  This quickly reduces the true peak.

4. **Phase 3 – hill‑climbing refinement**  
   Random swaps and tiny mass transfers are accepted only if they strictly
   improve the objective.  This stochastic local search cleans up any
   remaining irregularities.

The three phases are run until the time budget is exhausted (a few seconds

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives; clamp to zero.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Re‑order a vector so that the largest values appear on the
    left side and the smallest on the right side (pairwise).
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. `a`.  Uses a single FFT.
    The output is g_i = 2·Σ_j a_j·w_{i+j} (the convolution of a with w_rev).
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1                 # index aligning with a
    seg = conv[start:start + n]                # length n
    return 2.0 * seg


def _max_convolution_info(a: np.ndarray):
    """
    Compute self‑convolution, its maximal value and all indices attaining it.
    Returns (c, max_c, idxs).
    """
    c = _full_convolution(a)
    max_c = c.max()
    idxs = np.nonzero(c == max_c)[0]
    return c, max_c, idxs
