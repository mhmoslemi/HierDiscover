"""
Projected‑gradient / continuation optimizer for the step‑function attack.

The algorithm works in three phases:

1️⃣  Warm‑start – try to reuse the previous best solution, otherwise
    scan a two‑parameter Beta family (a_i ∝ i^{p‑1}(n‑i+1)^{q‑1})
    on a dense grid and keep the best candidate.

2️⃣  Continuation gradient descent – minimise the Lp‑norm of the
    self‑convolution (p starts at 2 and is increased slowly to
    a large value, thereby turning the smooth Lp‑objective into the
    true max‑norm).  In each iteration we compute the gradient

        g_i = 2 * Σ_k  b_k^{p‑1} * a_{k‑i},

    where b = a * a is the full convolution.
    A step of size `step_frac / ||g||_1` (so the total moved L1‑mass
    equals `step_frac`) is taken, the vector is clipped to non‑negativity
    and re‑normalised (projection onto the simplex).  An annealed
    Metropolis rule permits occasional uphill moves.

3️⃣  Final evaluation – the best sequence found is returned as a plain
    Python list.  The routine respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_a_wrev(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Convolution of a (len=n) with w_rev (len=2n‑1).  Returns the full
    linear convolution (len = 3n‑2).
    """
    n = a.shape[0]
    m = w_rev.shape[0]               # = 2n‑1
    out_len = n + m - 1               # = 3n‑2
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:out_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised sum(a)=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Beta‑family sweep (dense grid)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf

    # a fairly dense grid, extending to fairly extreme shapes
    p_vals = np.concatenate([
        np.geomspace(0.2, 1.0, 7),                 # < 1
        np.linspace(1.0, 5.0, 9),                 # 1 … 5
        np.geomspace(5.0, 30.0, 6)[1:]            # 5 … 30
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        # vectorised over all q – avoids a double python loop overhead
        # (but we keep it simple and explicit for readability)
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient‑descent on a smoothed max‑norm (Lp‑norm continuation)
# ----------------------------------------------------------------------
def _gradient_lp(a: np.ndarray, p: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the gradient of the Lp‑norm of the self‑convolution of `a`
    (for p≥2) and return both the gradient and the convolution `b`.
    """
    n = a.shape[0]

    # 1) self‑convolution
    b = _full_convolution(a)                     # length 2n‑1

    # 2) weight w = b^{p‑1} (for p=2 we have w = b)
    if p == 2.0:
        w = b.copy()
    else:
        # values are in [0,1]; raising to a high power is safe
        w = np.power(b, p - 1.0)

    # 3) convolution a * w_rev  (length 3n‑2)
    w_rev = w[::-1]
    conv = _conv_a_wrev(a, w_rev)

    # 4) extract entries: g_i = 2 * conv[(len(w)-1) - i]
    idx = (w.shape[0] - 1) - np.arange(n)        # (2n‑2)‑i
    g = 2.0 * conv[idx]
    return g, b


def _gradient_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_time: float,
    best_a: np.ndarray,
    best_val: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Continuation gradient descent with a simple Metropolis acceptance rule.
    """
    a = a0.copy()
    # evaluate current state once
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # schedule parameters
    step_start, step_end = 0.2, 1e-6            # L1‑mass moved per iteration
    temp_start, temp_end = 1e-2, 1e-8          # temperature for Metropolis
    p_start, p_end = 2.0, 30.0                  # continuation from L2 to near‑max
    no_improve = 0
    IMPROVE_LIMIT = 100_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress ratio in [0,1]
        prog = (now - start_time) / (deadline - start_time)

        # continuation parameter p
        p = p_start + (p_end - p_start) * prog

        # temperature and step‑size schedules
        T = temp_start * (temp_end / temp_start) ** prog
        step_frac = step_start * (1.0 - prog) + step_end * prog

        # gradient of the Lp‑norm
        g, b = _gradient_lp(a, p)

        # determine a step that moves exactly `step_frac` total L1‑mass
        norm_g1 = np.linalg.norm(g, 1) + 1e-16
        lr = step_frac / norm_g1

        # tentative update
        a_new = a - lr * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:                     # degenerate case – keep old state
            a_new = a.copy()
            s = 1.0
        a_new /= s                         # projection onto the simplex

        # evaluate new objective
        b_new = _full_convolution(a_new)
        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Iter {iteration:07d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – perturb around the current best
        if no_improve > IMPROVE_LIMIT:
            print(f"[Iter {iteration:07d}] stagnation – perturbation")
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val
