    # security checks

"""
Improved step‑function optimizer.

The core idea is a *monotone pairwise exchange* algorithm with an
exact line‑search: at each iteration we identify the index k that
realises the current maximal convolution, find the coefficient a_i that
contributes the most to that peak and the coefficient a_j that contributes
the least, and move as much mass as possible from i to j while guaranteeing
that no other convolution entry exceeds the old maximum.  The update can be
performed by a cheap O(n) operation on the convolution vector, so millions
of iterations fit comfortably into the 1000 s budget.

The optimizer starts from a collection of heuristic candidates (uniform,
exponential families, power‑law × exponential, random Dirichlet draws and,
if available, the previous best solution `height_sequence_1`).  After a quick
phase of random exploration it runs the monotone exchange until the time
budget expires, occasionally jittering the vector to escape stagnation.
A tiny uniform mixing step is performed at the end, and the final sequence
is returned (it is normalised; the security wrapper `evaluate_sequence`
takes care of the remaining checks).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective for a normalised vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for a vector that
    already satisfies Σ a = 1.
    """
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric families (same as the previous state‑of‑the‑art code)
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power(n: int, p: float) -> np.ndarray:
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)          # stabilise
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Monotone pairwise exchange with exact line‑search
# ----------------------------------------------------------------------
def _monotone_improve(a: np.ndarray,
                      n: int,
                      deadline: float,
                      *,
                      max_iters: int = 10_000_000,
                      jitter_every: int = 5_000,
                      recompute_every: int = 2_000) -> tuple[np.ndarray, float]:
    """
    Starting from a feasible a (∑a = 1, a ≥ 0) repeatedly move mass from the
    coefficient that contributes most to the current maximal convolution
    entry to the coefficient that contributes the least, choosing the
    maximal admissible step size that does not increase any other entry.

    Returns the best vector seen and its objective value.
    """
    cur = a.copy()
    # initialise convolution once
    conv = _full_convolution(cur)
    best_val = 2.0 * n * conv.max()
    best_a = cur.copy()
    it = 0
    no_improve = 0

    while it < max_iters and time.time() < deadline:
        # current peak
        k0 = int(np.argmax(conv))
        max_val = conv[k0]

        # rev[i] = a_{k0-i} (zero outside range)
        lo = max(0, k0 - (n - 1))
        hi = min(n - 1, k0)
        rev = np.zeros(n, dtype=np.float64)
        rev[lo:hi + 1] = cur[k0 - np.arange(lo, hi + 1)]

        # choose donor i (largest rev) and recipient j (smallest rev)
        i = int(np.argmax(rev))
        # exclude the donor when looking for the smallest
        mask = np.arange(n) != i
        if not np.any(mask):
            break                     # nothing to move
        j = int(np.argmin(np.where(mask, rev, np.inf)))

        if cur[i] <= 1e-15:
            # donor empty – jitter and recompute
            noise = np.random.normal(scale=1e-5, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            conv = _full_convolution(cur)
            it += 1
            continue

        # build the two shifted copies of a (length 2n‑1)
        shift_i = np.zeros(2 * n - 1, dtype=np.float64)
        shift_i[i:i + n] = cur
        shift_j = np.zeros(2 * n - 1, dtype=np.float64)
        shift_j[j:j + n] = cur

        # slope for each convolution entry when moving δ from i to j
        s = 2.0 * (shift_j - shift_i)          # shape (2n‑1,)
        s0 = s[k0]                             # ≤ 0

        # admissible δ – we must keep every entry ≤ max_val
        over = s > s0
        if np.any(over):
            numer = max_val - conv[over]            # non‑negative
            denom = s[over] - s0                    # positive
            # protect against degenerate divisions
            with np.errstate(divide='ignore', invalid='ignore'):
                cand = numer / denom
            cand = cand[(cand > 0) & np.isfinite(cand)]
            if cand.size > 0:
                delta_bound = cand.min()
            else:
                delta_bound = cur[i]
        else:
            delta_bound = cur[i]                     # no competing constraint

        # we cannot move more than we have, keep a safety margin
        delta = min(delta_bound, cur[i]) * 0.99
        if delta <= 0:
            # stagnation → jitter
            noise = np.random.normal(scale=1e-5, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            conv = _full_convolution(cur)
            it += 1
            continue

        # apply the move
        cur[i] -= delta
        cur[j] += delta
        # linear update of the convolution vector (exact up to O(δ²) terms)
        conv += s * delta
        # renormalise to eliminate tiny drift
        cur /= cur.sum()

        # record best
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        # periodical full recompute to curb accumulated round‑off errors
        if (it + 1) % recompute_every == 0:
            conv = _full_convolution(cur)

        # occasional random jitter to escape tiny plateaus
        if (it + 1) % jitter_every == 0 and no_improve > 5000:
            noise = np.random.normal(scale=1e-5, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            conv = _full_convolution(cur)
            no_improve = 0

        it += 1

    return best_a, best_val

# Tiny uniform mixing (same as earlier)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (normalised to sum = 1) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – a balance between quality and speed
    n = 8000

    # ------------------------------------------------------------------
    # Build an initial pool of candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1. Uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2. Exponential families (dense sweep over r)
    for r in np.linspace(0.95, 0.99999, 120):
        candidates.append(_gen_exp(n, r))
        candidates.append(_gen_rev_exp(n, r))

    # 3. Mixtures of exponentials
    for _ in range(30):
        r1 = np.random.uniform(0.9, 0.99999)
        r2 = np.random.uniform(0.9, 0.99999)
        w = np.random.rand()
        candidates.append(_gen_mix_exp(n, r1, r2, w))

    # 4. Power‑law × exponential
    for _ in range(30):
        p = np.random.uniform(-4.0, 4.0)
        r = np.random.uniform(0.9, 0.99999)
        candidates.append(_gen_power_exp(n, p, r))

    # 5. Random Dirichlet draws (a few different concentration levels)
    for conc in (0.5, 1.0, 2.0, 5.0):
        for _ in range(10):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # 6. Previous best (if present)
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.ndim == 1 and old.size > 0:
            # resample/interpolate to current n
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, old)
            interp = np.maximum(interp, 0.0)
            interp /= interp.sum()
            candidates.append(interp)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Evaluate the pool and keep the best start point
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(candidates):
        cand = np.asarray(cand, dtype=np.float64)
        if cand.ndim != 1 or cand.shape[0] != n:
            continue
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0:
            continue
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Init] candidate {idx:3d} new best = {best_val:.6f}")

    if best_a is None:
        # fallback – uniform (should never happen)
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a, n)

    print(f"[Init] overall best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – quick random exploration (short window)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    random_deadline = start_time + 0.15 * (deadline - start_time)
    while time.time() < random_deadline:
        # tiny random perturbations of the current best
        noise = rng.normal(scale=0.005, size=n)
        trial = np.maximum(best_a + noise, 0.0)
        trial /= trial.sum()
        val = _objective(trial, n)
        if val < best_val:
            best_val, best_a = val, trial.copy()
            print(f"[Random] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – monotone pairwise improvement (the core)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.80 * (deadline - start_time)
    improved_a, improved_val = _monotone_improve(best_a, n, phase1_deadline,
                                                 jitter_every=4000,
                                                 recompute_every=3000)
    if improved_val < best_val:
        best_val, best_a = improved_val, improved_a.copy()
        print(f"[Monotone] improved best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final tiny uniform mixing (helps escape very flat spots)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification with the security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list (the wrapper expects that)
    return best_a.tolist()
