"""
Improved optimizer for the step‑function challenge.

The search proceeds in three stages:

1. **Initialisation** – cheap parametric families (Beta and power‑law) and a few
   Dirichlet draws are evaluated with the exact objective.
2. **Adam optimisation** – minimise a smooth approximation of the objective
   (`2·n·smooth_max(conv)`) plus a variance‑penalty that flattens the whole
   convolution.  Hyper‑parameters (learning‑rate, λ‑schedule, ε‑schedule) are
   tuned for faster convergence and a longer optimisation window.
3. **Hill‑climbing (simulated annealing)** – a fast O(n) local search moves a
   fraction of mass from indices that contribute heavily to the current maximal
   convolution entry toward indices that contribute little.  The schedule for
   the maximal transfer fraction and the temperature is more aggressive early
   on.
4. **Sub‑gradient polishing** – projected sub‑gradient descent on the exact
   `max(conv)` objective finishes the search.

The routine respects the 1000 s budget, stops early if the target
`1.503` is reached, and finally returns a non‑negative list whose sum is 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries that affect a
    grad_a = 4.0 * n * segment[::-1]             # factor 2·n already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns   var(b) = mean((b‑avg)²)   and   ∂var/∂a .
    Here b = a*a and avg = 1/(2n‑1) because Σb = (Σa)² = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v (see derivation)
    conv_ab = _conv_fft(a, v[::-1])              # length 3n‑2
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (smooth max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises
        L(a) = 2·n·smooth_max(a) + λ·Var(conv(a,a)),
    where λ is annealed from a large start value to 0.
    The simplex constraint Σa = 1 is enforced via a log‑softmax
    parametrisation.
    """
    # initialise logits (log‑softmax parametrisation)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.95, 0.999
    lr_start, lr_end = 0.20, 0.001
    eps_start, eps_end = 0.5, 1e-5

    # variance‑penalty schedule (much stronger early flattening)
    lam_start = 50.0 * n
    lam_end = 0.0

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 20000          # restart after many idle steps

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # decode logits to probabilities (simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedule parameters
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # smooth‑max and its gradient
        _, grad_smooth = _smooth_max_and_grad(a, eps)

        # variance penalty
        _, grad_var = _var_and_grad(a)

        grad_a = grad_smooth + lam * grad_var

        # chain rule through the soft‑max (gradient w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.015, size=n)

        # restart when stuck
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restart after {idle} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (enhanced schedule)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast local search that moves a fraction of mass from an index that
    heavily contributes to the current maximal convolution entry towards an
    index that contributes little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule
    max_frac_start, max_frac_end = 0.6, 0.001
    temp_start, temp_end = 0.05, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 120_000      # after this we re‑perturb

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # index of current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each a_i to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source: sample proportional to contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target: bias towards indices that contribute little
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform (excluding i)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – small random perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503,
                        max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a) = 2·n·max(conv(a,a))
    using the analytic sub‑gradient.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # sub‑gradient of max(conv) w.r.t. a
        b = _full_convolution(a)
        max_val = b.max()
        k_maxes = np.where(b == max_val)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step size – linear decay (more aggressive early)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.25 * (1.0 - progress) + 5e-6 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Subgrad {it:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break

        # occasional jitter to break flat plateaus
        if np.random.rand() < 0.003:
            a = np.maximum(a + np.random.normal(scale=0.015, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric candidates (Beta family and simple power‑law)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.2, 2.0, 9),
                             np.geomspace(2.0, 8.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _powerlaw_random_best(n: int, trials: int = 120) -> tuple[np.ndarray, float]:
    """Random power‑law shapes a_i ∝ (i+1)^p with p ∈ [‑2, 2]."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    rng = np.random.default_rng()
    for _ in range(trials):
        p = rng.uniform(-2.0, 2.0)
        a = (i ** p)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val
