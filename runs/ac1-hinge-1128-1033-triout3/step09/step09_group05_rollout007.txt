def _project_simplex(v):
    # v can be any real vector; project onto {x >= 0, sum=1}
    # algorithm from "Efficient Projections onto the l1-ball for

    # descent for decreasing l1-norm" etc.

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    state‑of‑the‑art) to polish the solution.

4️⃣  Tiny mixing with the uniform vector, which is known to improve the
    bound a little bit.

All operations respect the time budget (default 1000 s) and the
``evaluate_sequence`` security wrapper.  The returned list satisfies
Σa = 1, a_i ≥ 0.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolutions (FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _full_convolution_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary 1‑D arrays."""
    out_len = x.shape[0] + y.shape[0] - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a normalized vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Parametric families (used in the random‑exploration phase)
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power(n: int, p: float) -> np.ndarray:
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)                 # stabilise exponentials
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta density on a regular grid, shifted to the centre of each bin."""
    x = (np.arange(1, n + 1, dtype=np.float64) - 0.5) / n
    # Guard against division by zero for extreme parameters
    a = np.where(
        (x > 0) & (x < 1),
        np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0),
        0.0,
    )
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_sym_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetrised beta shape (mirrored around the centre)."""
    half = n // 2
    left = _gen_beta(half + (n % 2), alpha, beta)
    if n % 2 == 0:
        right = left[::-1]
    else:
        right = left[:-1][::-1]
    a = np.concatenate([left, right])
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Random exploration of parametric families
# ----------------------------------------------------------------------
def _random_exploration(n: int, deadline: float, rng: np.random.Generator,
                        best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """
    Draw many random parameter combinations from the families above.
    Updates and returns the best (a, value) pair found before ``deadline``.
    """
    while time.time() < deadline:
        choice = rng.integers(0, 8)          # eight families
        if choice == 0:                      # pure exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_exp(n, r)
        elif choice == 1:                    # reversed exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_rev_exp(n, r)
        elif choice == 2:                    # mixture of two exponentials
            r1 = rng.uniform(0.90, 0.99999)
            r2 = rng.uniform(0.90, 0.99999)
            w = rng.random()
            cand = _gen_mix_exp(n, r1, r2, w)
        elif choice == 3:                    # pure power‑law
            p = rng.uniform(-5.0, 5.0)
            cand = _gen_power(n, p)
        elif choice == 4:                    # power‑law × exponential
            p = rng.uniform(-5.0, 5.0)
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_power_exp(n, p, r)
        elif choice == 5:                    # beta distribution
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            cand = _gen_beta(n, a, b)
        elif choice == 6:                    # symmetric beta
            a = rng.uniform(0.2, 5.0)
            b = rng.uniform(0.2, 5.0)
            cand = _gen_sym_beta(n, a, b)
        else:                                # centre‑heavy exponential
            lam = rng.uniform(0.001, 0.02)
            i = np.arange(n, dtype=np.float64)
            centre = (n - 1) / 2.0
            a = np.exp(-lam * np.abs(i - centre))
            cand = a / a.sum()

        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Random] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Smoothed gradient descent on a soft‑max approximation of the max‑conv term
# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, n: int,
                            start: float, deadline: float,
                            best_a: np.ndarray, best_val: float,
                            target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Multiplicative update with a gradient that comes from the
    soft‑max of the convolution values.
    """
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 5000                     # after this many stagnating steps we jitter

    # initial temperature (fraction of the current spread)
    conv0 = _full_convolution(cur)
    spread0 = conv0.max() - conv0.min()
    tau_initial = max(spread0 * 0.25, 1e-8)
    tau_final = 1e-9

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        conv = _full_convolution(cur)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- temperature schedule (linear in elapsed fraction) -----
        progress = (now - start) / (deadline - start)
        tau = tau_initial * (1.0 - progress) + tau_final * progress
        # ----- soft‑max weights -----
        # shift by max to avoid overflow, all entries ≤ 0
        w = np.exp((conv - max_conv) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:            # numerical safeguard – fallback to pure sub‑gradient
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- gradient = 2 * (w * rev(a)) -----
        rev_cur = cur[::-1]
        conv_wr = _full_convolution_general(w, rev_cur)          # length 3n‑2
        grad = 2.0 * conv_wr[n - 1:n - 1 + n]                    # slice of length n

        # ----- learning‑rate schedule (fast → slow) -----
        eta = (1.0 - progress) * 0.48 + progress * 0.001

        # multiplicative (exponentiated) update, stays on the simplex
        cur = cur * np.exp(-eta * grad)
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional jitter to escape plateaus
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.004, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SGD {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climbing (same as in the previous state‑of‑the‑art)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            cur_val = new_val
        else:
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Final tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under the given time budget.
    Returns the best list of floats found (Σa = 1) which minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – leave a little breathing room for the wrapper
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size.  12 000 gives a nice trade‑off between resolution and speed.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise – use the previous construction if available,
    # otherwise fall back to a quick deterministic sweep.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        # a quick deterministic start (uniform) – it will be immediately refined.
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase‑0 – random exploration (≈ 20 % of the time budget)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    rand_deadline = start_time + 0.20 * (deadline - start_time)
    best_a, best_val = _random_exploration(n, rand_deadline, rng, best_a, best_val)

    # ------------------------------------------------------------------
    # Phase‑1 – smoothed gradient descent (≈ 55 % of the time budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.75 * (deadline - start_time)
    best_a, best_val = _smooth_gradient_descent(
        best_a, n,
        start=start_time,
        deadline=phase1_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase‑2 – deterministic greedy hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=phase1_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase‑3 – final uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification with the provided security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:                                   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list (the evaluator expects exactly that)
    return best_a.tolist()
