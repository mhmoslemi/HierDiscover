The algorithm combines:
- a fast Adam optimiser on the logarithmic parameters (multiplicative
  updates) that works on the top K active convolution peaks;
- a deterministic hill‑climb that moves mass from the biggest
  contributors to the smallest;
- a small set of convex mixtures with the uniform and the reversed
  vector to smooth the autocorrelation.
The whole routine respects a 1000 s wall‑clock budget and keeps the
best vector found across many cheap random restarts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution (full linear) using FFT – length 2·n‑1
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

# ----------------------------------------------------------------------
# Objective for a vector that is already normalised (sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)

    val = _objective(a, n)

    val = _objective(a, n)

    a /= a.sum()
    val = _objective(a, n)

    for _ in range(12):
        cand = rng.dirichlet(np.ones(n))
        val = _objective(cand, n)
        if val < best_val:
            best_a, best_val = cand, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the logarithm of the coefficients (multiplicative update)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                global_start: float,
                deadline: float,
                top_k: int = 3,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in log‑space.  In each iteration we compute the gradient
    of the (sub‑)gradient of the *top‑K* convolution peaks,
    update the log‑variables, and renormalise to Σa = 1.
    """
    cur = a0.copy()
    eps = 1e-14
    log_a = np.log(np.maximum(cur, eps))

    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 1

    best_a = cur.copy()
    best_val = _objective(cur, n)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # evaluate current objective
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            if best_val <= target_val:
                break

        # --- sub‑gradient from the K largest convolution entries ---
        if top_k == 1:
            ks = [int(np.argmax(b))]
        else:
            ks = np.argpartition(b, -top_k)[-top_k:]

        g = np.zeros_like(cur)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            # contribution a_{k-i}
            g[idx] += 2.0 * cur[k - idx]

        # Adam step – step‑size linearly decays from ≈0.3 to ≈5e‑4
        progress = (now - global_start) / (deadline - global_start)
        lr = (1.0 - progress) * 0.30 + progress * 0.0005

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # back to the primal domain and renormalise
        cur = np.exp(log_a)
        s = cur.sum()
        if s == 0.0:
            cur[:] = 1.0 / n
        else:
            cur /= s

        # keep log_a consistent with the renormalised cur
        log_a = np.log(np.maximum(cur, eps))

        t += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climb – move a fraction of the mass
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     start_frac: float = 0.30,
                     end_frac: float = 0.001,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Repeatedly transfer a fraction of the mass from the index that
    contributes most to the current peak to the index that contributes
    least.  The move is kept only if it strictly improves the objective.
    """
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 250_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing of the transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # locate the peak index
        k = int(np.argmax(b))

        # slice a_{k-i}
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)

        rev = np.zeros_like(cur)
        rev[idx] = cur[k - idx]          # contribution of each index to the peak

        src = int(np.argmax(rev))

        # target = index with smallest contribution (excluding src)
        rev_tmp = rev.copy()
        rev_tmp[src] = np.inf
        tgt = int(np.argmin(rev_tmp))

        if cur[src] <= 1e-15:
            # source depleted – occasional random shake
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # move accepted
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the reversed vector
# ----------------------------------------------------------------------
def _mix_with_reverse(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex mixes with the time‑reversed version of the current vector."""
    rev = a[::-1].copy()
    best_a, best_val = a, _objective(a, n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020, 0.050):
        cand = (1.0 - alpha) * a + alpha * rev
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Full linear convolution via FFT (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

    # Final verification via the supplied secure wrapper (optional)

        print(f"[final] evaluate_sequence reports = {reported:.6f}")

        print("[final] evaluate_sequence raised:", exc)

    # Return a plain Python list (sum ≈ 1)
