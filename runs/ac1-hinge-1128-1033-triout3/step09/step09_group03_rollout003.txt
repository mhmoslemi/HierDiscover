"""
Simple parametric search + projected sub‑gradient polishing.

The idea is to search a low‑dimensional family of symmetric “U‑shaped”
distributions (exponential, power‑law and inverse‑law) on a modest size
(n≈2000).  Random samples of the parameters are evaluated using a fast FFT
based convolution and the best setting is kept.  A small local refinement
then tweaks the parameters.  The resulting shape is up‑sampled to a larger
size (≈8000) and a few thousand iterations of exact projected sub‑gradient
descent are performed to polish the solution.

All vectors are kept on the probability simplex (non‑negative, sum = 1) –
the evaluation function is scale invariant, so this choice does not affect
the score.  The routine respects the 1000 s budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – O(n log n))
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length = 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (scale invariant – assumes Σa = 1)
# ----------------------------------------------------------------------
def obj_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families of symmetric “U‑shaped” distributions
# ----------------------------------------------------------------------
def shape_exp(r: float, n: int) -> np.ndarray:
    """a_i ∝ r**i + r**(n‑1‑i) , 0 < r < 1."""
    i = np.arange(n, dtype=float)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def shape_pow(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)**p + (n‑i)**p , p < 0."""
    i = np.arange(n, dtype=float)
    a = np.power(i + 1.0, p) + np.power(n - i, p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def shape_inv(lam: float, n: int) -> np.ndarray:
    """a_i ∝ 1/(i+lam) + 1/(n‑1‑i+lam) , lam > 0."""
    i = np.arange(n, dtype=float)
    a = 1.0 / (i + lam) + 1.0 / (n - 1 - i + lam)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_shape(family: str, params: tuple, n: int) -> np.ndarray:
    """Dispatch to the appropriate shape generator."""
    if family == "exp":
        return shape_exp(params[0], n)
    if family == "pow":
        return shape_pow(params[0], n)
    if family == "inv":
        return shape_inv(params[0], n)
    raise ValueError(f"unknown family {family}")


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Searches for a low‑max self‑convolution step function.
    Returns a list of non‑negative floats (already normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a tiny safety margin for the final evaluate_sequence call
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Phase 0 – cheap random sampling on a small problem size
    # ------------------------------------------------------------------
    n_small = 2000                      # cheap to evaluate
    best_val = float("inf")
    best_family = None
    best_params = None

    # allocate about 30 % of the budget for this stage
    phase0_deadline = start + 0.30 * (deadline - start)

    families = ["exp", "pow", "inv"]
    while time.time() < phase0_deadline:
        fam = rng.choice(families)
        if fam == "exp":
            # r in (0.5, 0.9999); more weight near edges for smaller r
            r = rng.uniform(0.5, 0.9999)
            params = (r,)
        elif fam == "pow":
            # p negative, typically between -5 and -0.1
            p = rng.uniform(-5.0, -0.05)
            params = (p,)
        else:  # inv
            lam = rng.uniform(0.05, 5.0)
            params = (lam,)

        a = generate_shape(fam, params, n_small)
        val = obj_value(a)
        if val < best_val:
            best_val = val
            best_family = fam
            best_params = params
            print(f"[Phase0] new best {best_val:.6f} fam={fam} params={params}")

    # ------------------------------------------------------------------
    # Phase 1 – local refinement of the best parametric family
    # ------------------------------------------------------------------
    # about 20 % of the budget
    phase1_deadline = start + 0.50 * (deadline - start)
    if best_family is not None:
        while time.time() < phase1_deadline:
            if best_family == "exp":
                r = best_params[0]
                # multiplicative log‑normal perturbation
                r_new = r * np.exp(rng.normal(scale=0.015))
                r_new = np.clip(r_new, 0.5, 0.999999)
                params_new = (r_new,)
            elif best_family == "pow":
                p = best_params[0]
                p_new = p + rng.normal(scale=0.05)
                p_new = np.clip(p_new, -8.0, -0.001)
                params_new = (p_new,)
            else:  # inv
                lam = best_params[0]
                lam_new = lam * np.exp(rng.normal(scale=0.07))
                lam_new = max(lam_new, 1e-3)
                params_new = (lam_new,)

            a_new = generate_shape(best_family, params_new, n_small)
            val_new = obj_value(a_new)
            if val_new < best_val - 1e-14:
                best_val = val_new
                best_params = params_new
                print(f"[Phase1] improved to {best_val:.6f} params={best_params}")

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to a larger size and polish with exact sub‑gradient
    # ------------------------------------------------------------------
    # Choose a larger n that still fits comfortably in the remaining budget.
    n_large = 8000

    # build the up‑sampled candidate
    a = generate_shape(best_family, best_params, n_large)
    a = a / a.sum()
    best_a = a.copy()
    best_val = obj_value(best_a)

    # Sub‑gradient descent (same as “Phase C” of the reference).
    # We run until time runs out; step size decays geometrically.
    eta0 = 0.5 / n_large
    eta = eta0
    decay = 0.9994
    iter_cnt = 0
    # allocate the remainder of the budget
    while time.time() < deadline:
        # convolution and current value
        b = full_convolution(a)
        cur_val = 2.0 * n_large * b.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            if iter_cnt % 500 == 0:
                print(f"[SubGrad] it {iter_cnt:07d} new best {best_val:.6f}")

        # sub‑gradient of the maximal entry
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n_large)
        mask = (idx >= 0) & (idx < n_large)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # gradient step + projection onto the simplex
        a = project_simplex(a - eta * grad)

        # step‑size schedule
        eta *= decay
        iter_cnt += 1

        # occasional “shake” to escape stagnation
        if iter_cnt % 20000 == 0:
            noise = rng.normal(scale=0.008, size=n_large)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            eta = eta0

    # ------------------------------------------------------------------
    # Phase 3 – tiny mixing with the uniform distribution (often helps)
    # ------------------------------------------------------------------
    uniform = np.full(n_large, 1.0 / n_large)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * best_a + alpha * uniform
        cand /= cand.sum()
        cand_val = obj_value(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_a = cand
            print(f"[Mix] alpha={alpha:.3f} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final reporting (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f} (obj ≈ {best_val:.6f})")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
