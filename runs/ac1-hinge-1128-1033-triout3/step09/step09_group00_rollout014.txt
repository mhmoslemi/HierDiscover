def _subgradient_descent(a0,n,deadline,start_time,best_a,best_val):

    a = a0.copy()
    cur_val = _objective(a)
    # step schedule parameters
    step_start = 0.5
    step_end = 1e-5

    while time.time() < deadline:
        progress = (time.time() - start_time) / (deadline - start_time)
        step = step_start * (1 - progress) + step_end * progress

        # compute b and gradient
        b = _full_convolution(a)
        max_b = b.max()
        k_maxes = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # update
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_a = a_new.copy()
                best_val = new_val
                print(f"[Subgrad] new best = {best_val:.6f}")
        else:
            # occasional jitter
            if np.random.rand() < 0.001:
                a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
                a = _project_simplex(a)
                cur_val = _objective(a)

    return best_a, best_val

def _hillclimb_sa(a0, n, deadline, start_time, best_a, best_val):

    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # parameters
    temp_start = 0.15
    temp_end = 1e-8
    frac_start = 0.9
    frac_end = 0.001

    no_improve = 0
    IMPROVE_LIMIT = 20000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        T = temp_start * (temp_end / temp_start) ** progress
        max_frac = frac_start * (1 - progress) + frac_end * progress

        # compute contributions for current max index
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        if total > 0:
            src_probs = contrib / total
            i = np.random.choice(n, p=src_probs)
        else:
            i = np.random.randint(n)

        # Target selection
        inv_contrib = np.where(contrib>0, 1.0/(contrib+1e-12), 0.0)
        inv_contrib[i] = 0.0  # can't be same
        wsum = inv_contrib.sum()
        if wsum > 0:
            j = np.random.choice(n, p=inv_contrib/wsum)
        else:
            # uniform among others
            choices = list(range(n))
            choices.remove(i)
            j = np.random.choice(choices)

        # delta
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0:
            continue

        # compute new candidate a and b incremental
        # Make copy of b for trial
        b_new = b.copy()
        # using a (original) for shift
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[2*j] += delta*delta
        b_new[2*i] += delta*delta
        b_new[i+j] -= 2.0*delta*delta

        # compute new max
        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            # ensure nonnegativity due to floating errors,
            # clip
            if a[i] < 0:
                a[i] = 0.0
            # renormalize not needed: sum unchanged
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # restart from best with small perturb
            a = best_a.copy()
            # add small gaussian noise then reproject onto simplex
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                a = best_a.copy()
                a_sum = a.sum()
            a /= a_sum
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {iteration:7d}] stagnation - perturb")
        iteration += 1

    return best_a, best_val

"""
Simulated‑annealing/greedy optimiser for the step‑function attack.

* Initialise a candidate by a cheap sweep of the 2‑parameter Beta
  family and a few Dirichlet draws (all on the simplex Σa = 1).
* Run a fast O(n) simulated‑annealing that repeatedly moves a small
  amount of mass from an index that contributes heavily to the current
  maximal convolution entry to an index that contributes little.
  The convolution is updated incrementally using the exact
  algebraic formula (including the ‑2 δ² correction).  Temperature
  and maximal transfer fraction are annealed.
* Whenever the current value improves, the best‑ever solution is saved.
* Simple “stagnation” handling: after many unrewarded steps the current
  state is re‑initialised to the best solution plus a tiny Gaussian
  perturbation.
* Finally the best sequence is passed to ``evaluate_sequence`` for the
  official score.

The implementation works for any length ``n``; here we use a moderate
size (≈ 2000) that gives a good trade‑off between expressivity and speed.
"""

import time

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (Σa = 1) and helper
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial candidate (Beta sweep + Dirichlet)
# ----------------------------------------------------------------------
def _beta_shape(p: float, q: float, n: int) -> np.ndarray:
    """Return a Beta‑shaped probability vector on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:

        s = a.sum()
    return a / s

# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (corrected incremental update)
# ----------------------------------------------------------------------
def _hillclimb_sa(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_a: np.ndarray,
                  best_val: float,
                  target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast O(n) SA that moves a small amount of mass from an index that
    heavily contributes to the current maximal convolution entry towards an
    index that contributes little.  The incremental convolution update
    contains the exact −2 δ² term.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # annealing schedules
    temp_start, temp_end = 0.15, 1e-8
    frac_start, frac_end = 0.9, 0.001

    no_improve = 0
    IMPROVE_LIMIT = 40000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        T = temp_start * (temp_end / temp_start) ** progress
        max_frac = frac_start * (1 - progress) + frac_end * progress

        # --- pick source index (probability ∝ contribution) ---
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = np.random.randint(n)

        # ---- pick target index (biased towards low contribution) ----
        inv_contrib = np.where(contrib > 0, 1.0 / (contrib + eps), 0.0)
        inv_contrib[i] = 0.0
        wsum = inv_contrib.sum()
        if wsum > 0:
            j = np.random.choice(n, p=inv_contrib / wsum)
        else:
            # uniform fallback (any index except i)
            j = (i + 1) % n

        # ---- amount of mass to move ----
        if a[i] <= eps:
            continue
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- propose new state using the exact incremental update ----
        # copy of current convolution
        b_new = b.copy()
        # linear part (uses the *old* a)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic correction
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- Metropolis acceptance ----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # apply the move
            a[i] -= delta
            a[j] += delta
            # protect against tiny negatives due to rounding
            if a[i] < 0.0:
                a[i] = 0.0
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[SA {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            # perturb the best solution slightly and restart from it
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                a = best_a.copy()
                a_sum = a.sum()
            a /= a_sum
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

            print("[init] dirichlet improved to {:.6f}".format(val))

    # Fallback – uniform random
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] fallback random, value={:.6f}".format(best_val))

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (corrected incremental update)
# ----------------------------------------------------------------------
def _hillclimb_sa(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  start_time: float,
                  best_a: np.ndarray,
                  best_val: float,
                  target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast O(n) SA that moves a small amount of mass from an index that
    heavily contributes to the current maximal convolution entry towards an
    index that contributes little.  The incremental convolution update
    contains the exact ‑2 δ² term.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # annealing schedules
    temp_start, temp_end = 0.15, 1e-8
    frac_start, frac_end = 0.9, 0.001

    no_improve = 0
    IMPROVE_LIMIT = 40000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        T = temp_start * (temp_end / temp_start) ** progress
        max_frac = frac_start * (1 - progress) + frac_end * progress

        # ----- source index (probability ∝ contribution to current max) -----
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = np.random.randint(n)

        # ----- target index (biased towards low contribution) -----
        inv_contrib = np.where(contrib > 0, 1.0 / (contrib + eps), 0.0)
        inv_contrib[i] = 0.0
        wsum = inv_contrib.sum()
        if wsum > 0:
            j = np.random.choice(n, p=inv_contrib / wsum)
        else:
            j = (i + 1) % n

        # ----- amount of mass to move -----
        if a[i] <= eps:
            continue
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- propose new state using the exact incremental update -----
        b_new = b.copy()
        # linear part (uses old a)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic correction
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # apply move
            a[i] -= delta
            a[j] += delta
            if a[i] < 0.0:
                a[i] = 0.0
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[SA {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                a = best_a.copy()
                a_sum = a.sum()
            a /= a_sum
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    The routine builds an initial candidate (β‑family + Dirichlet),
    then refines it with a fast simulated‑annealing hill‑climber that
    updates the convolution incrementally.
    The best sequence found is finally evaluated with ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    n = 2000                                 # compromise between size and speed

    # ------------------------------------------------------------------
    # Phase 0 – cheap initialisation
    # ------------------------------------------------------------------
    best_a, best_val = _initial_candidate(n)
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Simulated‑annealing hill‑climbing (majority of the budget)
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_sa(
        best_a, n,
        deadline=deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
