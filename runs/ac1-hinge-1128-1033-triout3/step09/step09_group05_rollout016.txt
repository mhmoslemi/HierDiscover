def _full_convolution(a):
    n = a.shape[0]
    out_len = 2*n -1
    size = 1 << (out_len-1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

"""
Propose a step‑function that beats the current best bound (≈1.509).
The optimizer works in three stages:

1. **Initialisation** – take the previous solution if available,
   otherwise start from a random Dirichlet vector.

2. **Multi‑peak exponentiated‑gradient descent** – at every iteration the
   K largest convolution entries are identified and their sub‑gradients
   are summed.  A multiplicative update (mirror–descent) is applied with
   a linearly decaying learning rate.  After a long stagnation period a
   small Gaussian jitter is added to escape plateaus.

3. **Deterministic hill‑climb** – using the current worst shift, mass is
   transferred from the index that contributes most to that shift towards
   the index that contributes least.  The step size shrinks with time.

Finally a tiny mixing with the uniform vector is performed and the best
sequence found (Σa = 1) is returned.  All steps respect the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via FFT (re‑used many times)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    Returns the best list of floats found (Σa = 1) which minimises
    evaluate_sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – 8000 works well for the time budget and quality
    n = 8000

    # --------------------------------------------------------------
    # 1️⃣  Initialisation
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 9999)
    try:

        a = init_vec / init_vec.sum()
        print("[init] reused previous sequence")
    except Exception:
        # fallback – simple Dirichlet draw (uniform on the simplex)
        a = rng.dirichlet(np.ones(n))
        a = a / a.sum()
        print("[init] random Dirichlet start")

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Multi‑peak exponentiated‑gradient descent
    # --------------------------------------------------------------
    K = 3                     # number of top convolution peaks used for the gradient
    stagnation_limit = 300_000
    it = 0
    no_improve = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # convolution and current objective
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()

        # record improvement
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[expgrad {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---- compute a summed sub‑gradient from the K largest peaks ----
        # indices of the K biggest entries (including the absolute maximum)
        if K == 1:
            ks = (np.argmax(conv),)
        else:
            ks = np.argpartition(conv, -K)[-K:]

        g = np.zeros(n, dtype=np.float64)          # aggregated sub‑gradient
        for k in ks:
            # the slice of a that overlaps with its reversed copy at shift k
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)

            rev = a[k - idx]
            g[idx] += 2.0 * rev                     # derivative of (a*a)_k

        a = a * np.exp(-eta * g)

        a[i] -= delta
        a[j] += delta
        a /= a.sum()
        it += 1

    # 4️⃣  Tiny mixing with the uniform vector (helps escape tiny local minima)
    # --------------------------------------------------------------

        cand /= cand.sum()

        print(f"[final] reported evaluation = {reported:.6f}")

        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
