"""
A two‑stage approach for the “min‑max self‑convolution” problem.

Stage 1 – cheap parametric sweep (n≈3000):
    * families: symmetric exponential, symmetric power‑law, stretched sine,
      random Dirichlet draws.
    * each candidate is normalised to sum = 1 and evaluated with a fast FFT
      convolution.  The best candidate (lowest exact objective) is kept.

Stage 2 – up‑sample to a larger problem (n≈12000) and run a short but
intensive local optimisation consisting of:
    – Adam on a smooth log‑sum‑exp surrogate of max‑convolution,
    – a simulated‑annealing hill‑climber that works with an incremental
      O(n) convolution update,
    – a final tiny uniform‑mixing step.

All work respects the 1000 s time budget, prints intermediate bounds
and finally returns a list of non‑negative floats that (approximately)
minimises ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (len = 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp surrogate of max(conv)."""
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules
        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient w.r.t. a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)           # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # tiny jitter
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.006, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Classic hill‑climbing with a temperature‑controlled acceptance
    probability for uphill moves.  Full convolution is updated incrementally.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    T0 = cur_val * 0.02
    max_frac_start, max_frac_end = 0.30, 0.0005
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        k_max = int(np.argmax(b))

        # source selection – indices that heavily contribute to the peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # target selection – favour indices with little contribution
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA  {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake up the current state
            print(f"[SA  {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: mixing a tiny amount of uniform mass
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α values."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function under the 1000 s time budget.
    Returns a list of non‑negative floats that (approximately) minimise
    `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Stage 1 – fast parametric sweep on a modest size (n0)
    # ------------------------------------------------------------------
    n0 = 3000
    best_a = np.full(n0, 1.0 / n0)
    best_val = _objective_value(best_a)

    def _try_candidate(a_vec):
        nonlocal best_a, best_val
        a = np.asarray(a_vec, dtype=np.float64)
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a /= s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Sweep] new best = {best_val:.6f}")

    # ----- uniform (already baseline)
    _try_candidate(np.ones(n0))

    # ----- symmetric exponential (U‑shaped)
    for c in np.linspace(0.5, 6.0, 120):
        r = np.exp(-c / n0)
        i = np.arange(n0, dtype=np.float64)
        a = r ** i + r ** (n0 - 1 - i)
        _try_candidate(a)

    # ----- symmetric power‑law (U‑shaped)
    for p in np.arange(0.2, 5.01, 0.2):
        idx = np.arange(1, n0 + 1, dtype=np.float64)
        a = idx ** (-p) + (n0 + 1 - idx) ** (-p)
        _try_candidate(a)

    # ----- stretched half‑wave sine
    for power in (0.5, 1.0, 1.5, 2.0, 3.0):
        a = np.sin(np.pi * (np.arange(1, n0 + 1) / (n0 + 1))) ** power
        _try_candidate(a)

    # ----- random Dirichlet samples
    for _ in range(200):
        a = rng.dirichlet(np.ones(n0))
        _try_candidate(a)

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample and local refinement (larger n)
    # ------------------------------------------------------------------
    n_target = 12000                       # more degrees of freedom → better bound
    old_x = np.linspace(0.0, 1.0, n0)
    new_x = np.linspace(0.0, 1.0, n_target)
    a_up = np.interp(new_x, old_x, best_a)
    a_up = np.maximum(a_up, 0.0)
    a_up /= a_up.sum()

    best_a = a_up
    best_val = _objective_value(best_a)
    print(f"[Upsample] objective = {best_val:.6f}")

    # allocate remaining time to local optimisation
    remaining = deadline - time.time()
    if remaining > 30.0:
        # Phase A – Adam
        phase_a_deadline = time.time() + 0.65 * remaining
        best_a, best_val = _adam_phase(
            best_a, n_target,
            start=start_time,
            deadline=phase_a_deadline,
            best_a=best_a,
            best_val=best_val,
            target_val=1.503
        )
        remaining = deadline - time.time()

    if remaining > 20.0:
        # Phase B – Simulated‑annealing
        phase_b_deadline = time.time() + 0.90 * remaining
        best_a, best_val = _simulated_annealing_phase(
            best_a, n_target,
            start=start_time,
            deadline=phase_b_deadline,
            best_a=best_a,
            best_val=best_val,
            rng=rng,
            target_val=1.503
        )

    # final tiny uniform mixing
    best_a, best_val = _mix_with_uniform(best_a, n_target)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
