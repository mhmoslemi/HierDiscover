"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation**
   * Use the previously discovered best solution (`height_sequence_1`) if
     available; otherwise fall back to a uniform distribution.
   * Normalise the vector to sum‑to‑one.

2. **Alternating linear programme (ALP)**
   * Let `a` be the current distribution, `M = max(a*a)`.
   * Solve the linear programme
        maximise   Σ g_j
        subject to g ≥ 0  and  a ⊛ g ≤ M                (⊛ = linear convolution)
     This LP is linear because `a` is constant.
   * Rescale the optimiser output `g` to a probability vector and mix it
     with `a` (convex combination).  The mixing factor is fixed to `t = 0.5`.
   * Repeat until the time budget is almost exhausted or no further
     improvement is observed.

   The LP is built once with CVXPY using the expression ``cp.conv(a_param,g)``,
   which internally represents the convolution as a linear operator – no
   dense matrix is ever materialised.

3. **Deterministic sub‑gradient polishing**
   * After the ALP phase perform a few thousand projected sub‑gradient
     steps on the exact objective
        f(a) = 2·n·max(a⊛a)   (with Σa = 1)
     The sub‑gradient of the max‑convolution is known analytically.
   * The step size is linearly annealed; after each step the iterate is
     projected onto the probability simplex.

4. **Final evaluation**
   * Call the provided ``evaluate_sequence`` on the best vector found and
     return it.

The whole routine respects the 1000 s budget, constantly checking the
remaining time and stopping before the deadline.  All numbers are kept
non‑negative and the returned sequence sums to 1 (the scaling is
irrelevant for the objective because ``evaluate_sequence`` is homogeneous).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a compromise between expressivity and speed
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise – try to reuse the previously best solution
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is injected by the surrounding harness.
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_len = old.shape[0]
        if old_len != n:
            # linear interpolation onto the new grid
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
        a = np.maximum(a, 0.0)
        a /= a.sum()
        print("[init] interpolated previous solution")
    except Exception:
        a = np.full(n, 1.0 / n, dtype=float)
        print("[init] fallback to uniform start")

    best_a = a.copy()
    best_val = _objective(best_a)
    print(f"[init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Alternating Linear Programme (ALP) phase
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
        has_cvx = True
    except Exception:
        has_cvx = False
        print("[ALP] cvxpy not available – skipping this phase")

    if has_cvx:
        # Build the LP once – parameters will be updated each iteration
        f_param = cp.Parameter(n)
        M_param = cp.Parameter()
        g = cp.Variable(n)
        constraints = [
            g >= 0,
            cp.conv(f_param, g) <= M_param   # element‑wise bound
        ]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        iteration = 0
        t_mix = 0.5               # mixing factor (convex combination weight)
        while time.time() < deadline - 5.0:
            # current bound
            M = _full_convolution(a).max()
            cur_obj = 2.0 * n * M
            if cur_obj <= 1.503:

            M_param.value = M

            if g.value is None:

            sum_g0 = g0.sum()
            if sum_g0 < 1e-12:
                print("[ALP] obtained zero mass – exiting")
                break

            g_norm = g0 / sum_g0                   # rescale to a probability vector

            a_new = np.maximum(a_new, 0.0)

            new_obj = _objective(a_new)

            # always move to the new point (the LP is designed to lower M)
            a = a_new
            iteration += 1

    # ------------------------------------------------------------------
    # 2️⃣  Deterministic sub‑gradient polishing (exact max)
    # ------------------------------------------------------------------
    def _subgradient_polish(a0: np.ndarray, max_iters: int = 20000) -> tuple[np.ndarray, float]:
        """Projected sub‑gradient descent on  f(a)=2·n·max(conv(a,a))."""
        a_loc = a0.copy()
        cur_val = _objective(a_loc)
        best_seq = a_loc.copy()
        best_val_loc = cur_val

        for it in range(max_iters):
            if time.time() >= deadline:
                break

            b = _full_convolution(a_loc)
            max_b = b.max()
            idxs = np.where(b == max_b)[0]

            grad = np.zeros_like(a_loc)
            for k in idxs:
                # contribution of a_i to b_k is a_i * a_{k-i}
                # sub‑gradient w.r.t. a_j is 2 * a_{k-j} (ordered pairs)
                j = k - np.arange(n)
                mask = (j >= 0) & (j < n)
                grad[mask] += 2.0 * a_loc[j[mask]]

            # linear step‑size schedule (aggressive → tiny)
            prog = (time.time() - start_time) / (deadline - start_time)
            step = 0.35 * (1.0 - prog) + 1e-5 * prog

            a_next = a_loc - step * grad
            a_next = _project_simplex(a_next)

            val_next = _objective(a_next)

            if val_next < cur_val - 1e-12:
                a_loc = a_next
                cur_val = val_next
                if val_next < best_val_loc:
                    best_val_loc = val_next
                    best_seq = a_loc.copy()
                    print(f"[Subgrad {it:06d}] new best = {best_val_loc:.6f}")

            # occasional tiny jitter to escape plateaus
            if np.random.rand() < 0.001:
                a_loc = np.maximum(a_loc + np.random.normal(scale=0.02, size=n), 0.0)
                a_loc = _project_simplex(a_loc)

        return best_seq, best_val_loc

    # run polishing (time permitting)
    best_a, best_val = _subgradient_polish(best_a, max_iters=30000)

    # ------------------------------------------------------------------
    # 3️⃣  Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
