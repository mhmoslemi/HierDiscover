"""
Optimiser for the step‑function evaluation problem.

Approach
--------
1. Initialise a moderate‑size probability vector (n≈500) by random Dirichlet draws
   and a few handcrafted families.
2. Iteratively improve it with a linear‑programming (LP) step:
   * Let f be the current vector and M = max(conv(f,f)).
   * Solve max Σ g_j subject to g_j ≥ 0 and (conv(f,g))_k ≤ M for all k.
   * Rescale the LP solution to Σg = 1 and replace f ← (1‑t)·f + t·g
     (t decays during the run).
   This is the “LP‑boost” described in the original paper.
3. Upsample the refined shape to a large size (n≈8000) by linear interpolation.
4. Perform a deterministic, optimal‑epsilon hill‑climb on the up‑sampled vector:
   at the index k where the convolution peak occurs we move mass from the
   entry contributing most to that peak to the entry contributing least,
   choosing the largest step that brings the peak down to the second‑largest
   convolution value.
5. Return the best vector found (normalised, non‑negative).

The routine respects the 1000 s budget, prints progress, and finally
verifies the result with the provided `evaluate_sequence` wrapper.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution (FFT based, works for any length)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()            # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce very small negative values; clip them.
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Helper: build the LP constraint matrix for a given f
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return a (2n‑1) × n matrix A such that (A @ g)[k] == (conv(f,g))[k].
    All rows correspond to the linear constraints (conv(f,g))_k ≤ M.
    """
    n = f.shape[0]
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    # Fill row by row (still cheap for n ≲ 1000)
    for k in range(K):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # indices j = lo..hi, coefficient = f[k-j]
        idx = np.arange(lo, hi + 1)
        A[k, lo:hi + 1] = f[k - idx]
    return A


# ----------------------------------------------------------------------
# Phase 1 – LP‑boost optimisation on a medium sized vector
# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               deadline: float,
               max_iter: int = 200,
               start_t: float = 0.5) -> np.ndarray:
    """
    Perform a sequence of LP refinements.
    Returns the best vector obtained before the deadline.
    """
    import scipy.optimize  # imported lazily – SciPy may be unavailable

    n = f.shape[0]
    best = f.copy()
    best_val = _objective(best, n)
    t = start_t
    iteration = 0

    while iteration < max_iter and time.time() < deadline:
        # 1. current max convolution value
        b = _full_convolution(best)
        M = b.max()

        # 2. build linear constraints  (conv(best, g) ≤ M)
        A = _build_constraint_matrix(best)
        K = A.shape[0]

        # 3. solve LP: maximise Σ g_j  <=> minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        linprog_opts = {
            "presolve": True,
            "time_limit": max(0.0, deadline - time.time())
        }
        try:
            res = scipy.optimize.linprog(
                c,
                A_ub=A,
                b_ub=np.full(K, M, dtype=np.float64),
                bounds=bounds,
                method="highs",
                options=linprog_opts,
            )
        except Exception as exc:
            print("[LP] exception:", exc)
            break

        if not res.success:
            print("[LP] failed:", res.message)
            break

        g = res.x
        if g.sum() == 0.0:
            break
        g /= g.sum()                     # normalise Σg = 1

        # 4. convex combination
        best = (1.0 - t) * best + t * g
        best /= best.sum()               # re‑normalise (numerical safety)

        # 5. record improvement
        cur_val = _objective(best, n)
        if cur_val < best_val - 1e-15:
            best_val = cur_val
            print(f"[LP] iter {iteration:03d} – new best = {best_val:.6f}")

        # 6. slowly reduce the step size
        iteration += 1
        if iteration % 5 == 0:
            t = max(0.05, t * 0.9)

    return best


# ----------------------------------------------------------------------
# Phase 2 – Optimal‑epsilon deterministic hill climbing
# ----------------------------------------------------------------------
def _optimal_hillclimb(a: np.ndarray,
                       deadline: float,
                       max_no_improve: int = 2000) -> np.ndarray:
    """
    Reduce the convolution peak by moving mass from the index that contributes
    most to the current peak to the index that contributes least.
    The step size is chosen analytically (epsilon that would bring the peak
    down to the second‑largest convolution value).
    """
    n = a.shape[0]
    cur = a.copy()
    best = cur.copy()
    best_val = _objective(best, n)

    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        b = _full_convolution(cur)
        k = int(np.argmax(b))
        max_b = b[k]

        # second largest convolution value (ignoring the current maximum)
        mask = np.ones_like(b, dtype=bool)
        mask[k] = False
        second_max = b[mask].max() if np.any(mask) else max_b

        # contributions of each a_j to b_k (rev vector)
        rev = np.zeros(n, dtype=np.float64)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))               # index with largest contribution
        # index with smallest contribution (different from i)
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        delta_rev = rev[i] - rev[j]
        if delta_rev <= 0:
            # No useful direction – break out
            break

        # Maximum epsilon that would lower the peak to the second‑largest value
        eps_opt = (max_b - second_max) / (2.0 * delta_rev)
        # Guard against numerical issues
        eps_opt = max(0.0, min(eps_opt, cur[i]))

        # Take a slightly smaller step to stay safely on the feasible side
        eps = eps_opt * 0.97
        if eps <= 1e-15:
            no_improve += 1
            continue

        # Perform the move
        cur[i] -= eps
        cur[j] += eps

        # Re‑normalise (should stay normalized, but floating‑point drift may appear)
        cur = np.maximum(cur, 0.0)
        cur_sum = cur.sum()
        if cur_sum == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= cur_sum

        cur_val = _objective(cur, n)
        if cur_val < best_val - 1e-15:
            best_val, best = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb] new best = {best_val:.6f}")
        else:
            no_improve += 1

    return best

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative sequence that minimises `evaluate_sequence`.
    The function respects the given time budget (default 1000 s) and prints
    intermediate progress.  The final result is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 0. Choose a modest problem size for the LP phase.
    # ------------------------------------------------------------------
    n_mid = 500        # medium size – fast LP solves
    rng = np.random.default_rng(seed + 999)

    # ------------------------------------------------------------------
    # 1. Initialise a reasonable candidate.
    # ------------------------------------------------------------------
    # Try to reuse a previously found construction if available.
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n_mid)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        # fall back to a random Dirichlet plus a few deterministic families
        a0 = np.full(n_mid, 1.0 / n_mid)
        best_val = _objective(a0, n_mid)

        # deterministic families (U‑shapes, exponentials, power‑laws)
        def try_update(vec):
            nonlocal a0, best_val
            vec = np.asarray(vec, dtype=np.float64)
            if np.any(vec < 0):
                return
            s = vec.sum()
            if s <= 0:
                return
            vec = vec / s
            v = _objective(vec, n_mid)
            if v < best_val:
                a0, best_val = vec, v

        idx = np.arange(n_mid)
        try_update(np.arange(n_mid, 0, -1, dtype=float))                # decreasing
        try_update(np.maximum(idx + 1, n_mid - idx))                   # linear U‑shape
        try_update(np.sin(np.pi * (np.arange(1, n_mid + 1) / (n_mid + 1))))   # half‑wave sine
        for p in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0):
            i = np.arange(1, n_mid + 1, dtype=float)
            a = i ** (-p) + (n_mid - i + 1) ** (-p)
            try_update(a)

        # a few random Dirichlet draws
        for _ in range(15):
            try_update(rng.dirichlet(np.ones(n_mid)))

    # Ensure normalisation
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    best_a = a0.copy()
    best_val = _objective(best_a, n_mid)
    print(f"[Init] objective (n={n_mid}) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑boost phase (medium size)
    # ------------------------------------------------------------------
    lp_deadline = start_time + 0.45 * (deadline - start_time)
    best_a = _lp_refine(best_a, lp_deadline, max_iter=300, start_t=0.6)
    best_val = _objective(best_a, n_mid)
    print(f"[LP‑phase] best after LP (n={n_mid}) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample to a large vector (for final reporting)
    # ------------------------------------------------------------------
    n_final = 8000
    if n_final != n_mid:
        x_old = np.linspace(0.0, 1.0, n_mid)
        x_new = np.linspace(0.0, 1.0, n_final)
        upsampled = np.interp(x_new, x_old, best_a)
        upsampled = np.maximum(upsampled, 0.0)
        upsampled /= upsampled.sum()
        best_a = upsampled
        best_val = _objective(best_a, n_final)
        print(f"[Upsample] objective (n={n_final}) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Deterministic optimal‑epsilon hill‑climb (large size)
    # ------------------------------------------------------------------
    hill_deadline = deadline - 0.2               # keep a little margin for final check
    best_a = _optimal_hillclimb(best_a, hill_deadline, max_no_improve=3000)
    best_val = _objective(best_a, n_final)
    print(f"[HillClimb] final objective (n={n_final}) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final verification with the provided security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
