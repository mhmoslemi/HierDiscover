    a = softmax(y)
    b = full_convolution(a)
    # compute w
    max_b = b.max()
    # shift for numerical stability
    shifted = b - max_b
    expb = np.exp(shifted / epsilon)
    w = expb / expb.sum()
    # compute grad_a
    a_rev = a[::-1]
    # compute convolution a_rev and w
    size = 1 << ((len(a_rev) + len(w) - 1) - 1).bit_length()
    Fa = np.fft.rfft(a_rev, size)
    Fw = np.fft.rfft(w, size)

def _anneal_hill_climb(a0, n, deadline, best_a, best_val, target_val=1.503):
    a = a0.copy()
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max

    eps = 1e-12
    no_improve = 0
    IMPROVE_LIMIT = 20000

    while time.time() < deadline:
        prog = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # find k_max
        k_max = int(np.argmax(b))

        # compute contrib for each index for that k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros(n)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target
        tgt_weights = np.zeros(n)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = (i_src + np.random.randint(1, n)) % n

        # delta
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental update
        b_new = b.copy()
        # update slices
        b_new[j_tgt:j_tgt+n] += 2.0 * delta * a
        b_new[i_src:i_src+n] -= 2.0 * delta * a
        # add delta^2 terms
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept
        if new_val < cur_val - 1e-15 or np.random.rand() < math.exp(-(new_val - cur_val)/T):
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb restart: add noise
            noise = np.random.normal(scale=0.02, size=n)
            a = best_a.copy()
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print("[Hill] stagnation – perturb")

    return best_a, best_val

"""
Non‑negative step‑function optimizer with extended search.

The optimizer combines three phases:

1️⃣  **Parameter sweep** – a dense search over the two‑parameter Beta family
    a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} (p,q∈[0.1,6]) and a simple power‑law family.
    The best sequence from the sweep is kept.

2️⃣  **Multi‑restart annealed hill‑climbing** – starting from the best
    sequence, a simulated‑annealing move transfers a small amount of mass
    from an index that heavily contributes to the current maximal convolution
    entry to an index that contributes little.  Several short restarts
    (perturbing the current best) are performed while the global time budget
    lasts.

3️⃣  **Final return** – the best sequence found (already normalised to sum 1)
    is returned as a plain Python list.  The external ``evaluate_sequence``
    function will apply the final security checks.

All helper functions are defined at the top level; no closures are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1).  Returns 2·n·max(conv(a,a)).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Beta‑family dense sweep
# ----------------------------------------------------------------------
def _beta_dense_search(n: int) -> tuple[np.ndarray, float]:
    """Exhaustive grid search over the Beta family (p,q in [0.1,6] with step 0.05)."""
    best_a = None
    best_val = np.inf
    # grid step 0.05 gives ~120 values per dimension
    p_vals = np.arange(0.1, 6.001, 0.05)
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        # pre‑compute i^{p‑1}
        ip = i ** (p - 1.0)
        for q in q_vals:
            a = ip * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple power‑law family sweep
# ----------------------------------------------------------------------
def _powerlaw_search(n: int) -> tuple[np.ndarray, float]:
    """Search a one‑parameter family a_i ∝ (i+1)^{-α} + (n-i)^{-α}."""
    best_a = None
    best_val = np.inf
    alphas = np.arange(0.1, 5.01, 0.1)
    i = np.arange(n, dtype=np.float64) + 1.0                # 1 … n
    rev = n - i + 1.0

    for alp in alphas:
        a = (i ** -alp) + (rev ** -alp)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (single run)
# ----------------------------------------------------------------------
def _anneal_hillclimb(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly moves a small amount of mass from a heavy
    contributor to a light one.  Temperature and move‑size decay slowly.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule (more aggressive than the original version)
    max_frac_start, max_frac_end = 0.9, 0.001
    temp_start, temp_end = 0.5, 1e-6

    no_improve = 0
    IMPROVE_LIMIT = 20000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)      # ∈[0,1]
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source – usually proportional to its contribution
        if total_contrib > 0.0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target – biased towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (different from source)
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # amount of mass to move
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturbation to escape stagnation
            noise = np.random.normal(scale=0.02, size=n)
            a = best_a.copy()
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val
