def refine_with_lp(f, n, max_iter, t, deadline):
    best_f = f.copy()
    best_M = np.max(_full_convolution(f))
    for it in range(max_iter):
        if time.time() >= deadline: break
        # compute M
        conv_f = _full_convolution(f)
        M = conv_f.max()
        # Build A_ub matrix lazily maybe use csc_matrix
        K = len(conv_f)
        rows = []
        cols = []
        data = []
        for k in range(K):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # vector of i indices
            i_range = np.arange(lo, hi+1)
            rows.extend([k] * len(i_range))
            cols.extend(i_range.tolist())
            # f[k - i] corresponds to idx = k - i_range
            f_vals = f[k - i_range]
            data.extend(f_vals.tolist())
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))
        # LP: maximize sum(g) => minimize -ones @ g
        c = -np.ones(n)
        b_ub = np.full(K, M)
        bounds = [(0, None) for _ in range(n)]
        # Solve
        result = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not result.success:
            # cannot solve; break
            break
        g = result.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g
        # mix
        f_new = (1 - t) * f + t * g_norm
        # re-normalize sum to 1
        f_new = f_new / f_new.sum()
        # compute new objective
        M_new = _full_convolution(f_new).max()
        if M_new < best_M - 1e-12:
            best_f = f_new.copy()
            best_M = M_new
        f = f_new

    return best_f

"""
Linear‑programming driven refinement of a step function.

The algorithm works with a modest dimension (≈500), where a single LP
solves the sub‑problem

    minimise   M
    subject to f * g ≤ M ,   Σ g = 1 , g ≥ 0

for the current vector `f`.  The optimum `g` yields a strictly smaller
peak of the convolution (the LP is the exact formulation of the
“maximise sum(g) subject to f·g≤M” scheme described in the statement).
Iterating this LP quickly drives the peak toward its theoretical
lower bound ≈ 1.  The final low‑dimensional shape is interpolated to the
desired length (8000) and normalised – a tiny uniform mixing is applied
as a last polishing step.

All heavy work is done with SciPy's sparse matrices and the HiGHS
solver, which comfortably fits the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional imports – fall back to a very simple method if SciPy is absent
# ----------------------------------------------------------------------
try:
    import scipy.sparse as sp
    import scipy.optimize as opt
    SCIPY_OK = True
except Exception:          # pragma: no cover
    SCIPY_OK = False


# ----------------------------------------------------------------------
# Helper: full linear convolution (naïve – fine for n ≤ 2000)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)-1)."""
    return np.convolve(a, a)


# ----------------------------------------------------------------------
# Objective used inside the search (same scaling as evaluate_sequence)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n_full: int) -> float:
    """Return 2·n·max(conv) assuming Σa = 1."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n_full * maxb)


# ----------------------------------------------------------------------
# Build the LP matrix for a given vector f.
#   Variables: g_0 … g_{n-1}, M  (total n+1 variables)
#   Constraints: (f * g)_k ≤ M   for all k = 0 … 2n‑2
#   Equality:    Σ g_j = 1
# ----------------------------------------------------------------------
def _build_lp_matrix(f: np.ndarray):
    """Return (A_ub, b_ub, A_eq, b_eq, c, bounds) for the LP described above."""
    n = f.shape[0]
    K = 2 * n - 1                     # number of convolution positions

    rows = []
    cols = []
    data = []

    for k in range(K):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # indices of g that appear in (f*g)_k
        js = np.arange(lo, hi + 1, dtype=int)
        # corresponding coefficients from f
        coeffs = f[k - js]

        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())

        # the -M term (coefficient -1 on the last variable)
        rows.append(k)
        cols.append(n)               # last column = M
        data.append(-1.0)

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(K, n + 1))
    b_ub = np.zeros(K, dtype=float)

    # Equality Σ g_j = 1   (M does not appear)
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=float)

    # Objective: minimise M (last variable)
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    bounds = [(0.0, None)] * n + [(0.0, None)]  # g ≥ 0, M ≥ 0

    return A_ub, b_ub, A_eq, b_eq, c, bounds


# ----------------------------------------------------------------------
# One LP iteration: given f, solve for the optimal g (with a new bound M)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray):
    """Solve the LP for the current f and return (g, M_opt)."""
    A_ub, b_ub, A_eq, b_eq, c, bounds = _build_lp_matrix(f)
    # HiGHS is fast for the sizes we use (n≈500 → ~2 · 10⁵ non‑zeros)
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None, None
    g = res.x[:f.shape[0]]
    M_opt = res.x[-1]
    return g, M_opt


# ----------------------------------------------------------------------
# Iterative LP refinement – stop when the self‑convolution peak stops
# improving or when time runs out.
# ----------------------------------------------------------------------
def _refine_lp(initial_f: np.ndarray,
               n_full: int,
               max_iter: int,
               deadline: float):
    """Iteratively replace f by the LP‑optimal g."""
    f = initial_f.copy()
    best_f = f.copy()
    best_val = _objective(f, n_full)

    for it in range(max_iter):
        now = time.time()
        if now >= deadline:
            break

        g, M_bound = _lp_step(f)
        if g is None:
            break

        # Normalise (the LP already enforces Σg = 1, but numerical drift may occur)
        s = g.sum()
        if s == 0.0:
            g = np.full_like(g, 1.0 / g.shape[0])
        else:
            g = g / s

        # Self‑convolution peak of the candidate
        val = _objective(g, n_full)

        if val < best_val - 1e-12:
            best_f, best_val = g.copy(), val
            print(f"[LP‑refine {it:02d}] new best = {best_val:.6f}")

        # Simple termination test – stop if no clear improvement
        if val >= best_val - 1e-14:
            # stagnation
            break

        f = g

    return best_f, best_val


# ----------------------------------------------------------------------
# Upsample a small vector to the final length (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(vec_small: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation from len(vec_small) → n_target, keep non‑negatives."""
    x_small = np.linspace(0.0, 1.0, vec_small.shape[0])
    x_target = np.linspace(0.0, 1.0, n_target)
    up = np.interp(x_target, x_small, vec_small)
    up = np.maximum(up, 0.0)
    total = up.sum()
    if total == 0.0:
        up = np.full(n_target, 1.0 / n_target)
    else:
        up /= total
    return up


# ----------------------------------------------------------------------
# Tiny uniform mixing – a last chance to shave off a few percent
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> np.ndarray:
    """Blend a with the uniform vector; keep the better of a and the mix."""
    best = a.copy()
    best_val = _objective(best, n)
    uniform = np.full(n, 1.0 / n)

    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best, best_val = cand, val
    return best


# ----------------------------------------------------------------------
# Main entry point – orchestrates several random restarts
# ----------------------------------------------------------------------

                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Uses a small‑scale linear‑programming refinement loop (see the module
    docstring) and a handful of random restarts.  The final vector is
    interpolated to 8000 entries and returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # tiny safety margin
    n_target = 8000                         # length of the final sequence

    # ------------------------------------------------------------------
    # Small optimisation dimension – a trade‑off between LP size and
    # expressive power.  600 gives ≈ 7·10⁵ non‑zero matrix entries,
    # solved comfortably in <0.2 s on a single CPU.
    # ------------------------------------------------------------------
    n_small = 600

    # ------------------------------------------------------------------
    # Helper to obtain a starting vector of length n_small
    # ------------------------------------------------------------------
    def _initial_vector():
        # try to reuse the previous best if available

        except Exception:

        # Ensure a proper probability simplex
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            vec = np.full(n_small, 1.0 / n_small)
        else:
            vec = vec / s
        return vec

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_vec_full = None
    best_val_full = float('inf')
    restart = 0

    # ------------------------------------------------------------------
    # Main restart loop – we allocate roughly equal time slices to each
    # restart (≈2 % of the remaining budget, but at least 1 s).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        slice_budget = max(1.0, 0.02 * remaining)

        restart_start = time.time()
        restart_deadline = restart_start + slice_budget

                f_small = np.maximum(f_small, 0.0)
                f_small /= f_small.sum()
        # ------------------- upsample to the final length ---------------
        f_full = _upsample(f_small, n_target)

        # --------------------- a final tiny uniform mix ------------------
        f_full = _mix_with_uniform(f_full, n_target)

        # --------------------- evaluate and keep the best ---------------

        print(f"[Restart {restart:02d}] candidate value = {val:.6f}")

        if val < best_val_full:
