"""
Non‑negative step‑function optimizer – pairwise mass‑transfer annealing.

The optimizer works in three phases:

1️⃣  **Warm‑start** – take the best known solution (global variable
    ``height_sequence_1``) if it exists, otherwise fall back to a short
    sweep of the two‑parameter Beta family and a few Dirichlet draws.

2️⃣  **Pairwise annealing** – repeatedly selects the convolution entry
    `k_max` with the biggest value.  The contribution of each index
    `i` to that entry is `c_i = a_i * a_{k_max‑i}`.  The algorithm picks
    the pair `(i_s, j_s)` with the largest contribution and moves a small
    amount `δ` from *both* of those indices to a low‑contribution pair
    `(i_t, j_t)`.  The move is accepted if the objective improves,
    otherwise it is accepted with a Metropolis probability that decays
    slowly.  The schedule (`max_frac` and temperature `T`) is stretched
    over the whole remaining budget, allowing a long exploratory phase.

3️⃣  **Final evaluation** – the best candidate found is returned as a
    plain list of floats.  The routine respects the 1000 s time budget.

The pairwise move reduces the dominant convolution term much faster
than moving a single index, which leads to a lower maximal auto‑convolution
and therefore a smaller evaluation score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Beta‑family sweep (cheap parametric search)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # a finer grid than the previous version
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 8.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Pairwise annealing hill‑climbing
# ----------------------------------------------------------------------
def _pairwise_anneal(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from the two
    indices that most contribute to the current maximal convolution
    entry to a pair of indices that contribute little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-14

    # schedule – stretched over the whole remaining budget
    max_frac_start, max_frac_end = 0.70, 0.001
    temp_start, temp_end = 0.02, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear progress in [0,1]
        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        idx = np.arange(n)
        mir = k_max - idx
        mask = (mir >= 0) & (mir < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[mir[mask]]

        # source pair – index with largest contribution
        i_src = int(np.argmax(contrib))
        j_src = k_max - i_src
        if not (0 <= j_src < n):
            # should not happen, but guard
            i_src = np.random.randint(n)
            j_src = k_max - i_src

        # target pair – bias towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        tgt_weights[j_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            i_tgt = int(np.random.choice(n, p=tgt_weights / wsum))
            j_tgt = k_max - i_tgt
            if not (0 <= j_tgt < n):
                # fall back to a random feasible partner
                i_tgt = np.random.randint(n)
                j_tgt = k_max - i_tgt
        else:
            # uniform fallback
            i_tgt = np.random.randint(n)
            while i_tgt in (i_src, j_src):
                i_tgt = np.random.randint(n)
            j_tgt = k_max - i_tgt
            if not (0 <= j_tgt < n):
                j_tgt = i_tgt

        # amount of mass to move (same from both source indices)
        if i_src == j_src:
            delta_max = a[i_src]
        else:
            delta_max = min(a[i_src], a[j_src])

        delta = delta_max * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # apply the pairwise move
        a_new = a.copy()
        if i_src == j_src:
            # single‑index case – treat like the old algorithm
            a_new[i_src] -= delta
            a_new[i_tgt] += delta
        else:
            a_new[i_src] -= delta
            a_new[j_src] -= delta
            a_new[i_tgt] += delta
            a_new[j_tgt] += delta

        # protect against negatives caused by rounding
        if a_new[i_src] < -eps or (i_src != j_src and a_new[j_src] < -eps):
            continue
        a_new = np.maximum(a_new, 0.0)

        # renormalise (does not change the objective but avoids drift)
        s = a_new.sum()
        if s == 0.0:
            continue
        a_new /= s

        # evaluate new candidate
        b_new = _full_convolution(a_new)
        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Iter {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Iter {iteration:7d}] stagnated – perturb")
        iteration += 1

    return best_a, best_val
