def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _conv_full(x, y):
    # full linear convolution via FFT
    out_len = len(x) + len(y) - 1

    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

    return _conv_full(a, a)

    b = _full_convolution(a)

        # shift for stability
        shift = max_b
        w_unnorm = np.exp(epsilon * (b - shift))
        w_sum = w_unnorm.sum()
        if w_sum == 0:
            w = np.full_like(w_unnorm, 1.0 / w_unnorm.size)
        else:
            w = w_unnorm / w_sum

        # --- gradient of the smoothed objective (mirror of sub‑gradient) ---
        # grad_i = 2 * sum_k w_k * a_{k-i}
        w_rev = w[::-1]
        conv = _conv_full(cur, w_rev)          # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)       # positions that give sum_{k} w_k a_{k-i}
        g = 2.0 * conv[idx]

        # --- exponentiated update (mirror descent on simplex) -----
        cur = cur * np.exp(-eta * g)

        # --- renormalise to the probability simplex -------------
        tot = cur.sum()
        if tot == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= tot

        # --- occasional jitter to escape plateaus ---------------
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=1e-3, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SmoothExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val

def _hillclimb_phase(a_start, n, start_time, deadline, best_a, best_val,
                     start_frac=0.30, end_frac=0.001, target_val=1.503):
    """Deterministic hill‑climb that shifts a fraction of mass from the biggest
    contributor to the smallest until no improvement is seen."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_a = cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # fraction annealing
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # pick one of the maximal convolution entries (if several, any)
        k = int(np.argmax(b))

        # contribution of each index i to that entry:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        contrib = np.zeros_like(cur)
        contrib[idx] = cur[k - idx]          # a_{k-i}
        # source = index with largest contribution
        src = int(np.argmax(contrib))
        # target = index with smallest contribution (excluding source)
        tmp = contrib.copy()
        tmp[src] = np.inf
        tgt = int(np.argmin(tmp))

        if cur[src] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            it += 1
            continue

        cur[src] -= delta
        cur[tgt] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # keep change
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val

def _mix_with_uniform(a, n):
    """Test a few tiny convex mixtures with the uniform vector."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_a, best_val = cand, val
    return best_a, best_val

Search for a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm combines three phases:

1. **Smooth exponentiated‑gradient descent** – a mirror‑descent on the
   probability simplex where the non‑smooth max‑operator is replaced by a
   soft‑max (log‑sum‑exp) with an annealed temperature.  This yields a
   well‑behaved gradient that gradually focuses on the worst convolution
   peaks.

2. **Deterministic hill‑climb** – after the gradient phase we move a
   fraction of the mass from the index that contributes most to the active
   peak to the index that contributes the least.  The transfer fraction
   anneals from 30 % down to 0.1 % and a move is kept only if it improves
   the objective.

3. **Tiny mixing with the uniform vector** – finally a few convex
   combinations with the uniform distribution are tested; occasionally this
   lowers the peak a bit.

A long warm‑up (≈30 % of the total budget) runs phase 1+2+mix on the
initial vector (or the previously best known vector).  The remaining time
is spent on many cheap random‑restart runs, each consisting of a short
gradient phase, a brief hill‑climb and the final mixing.  The best vector

# ----------------------------------------------------------------------
# Helper: FFT‑based full convolution
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of *x* and *y* (real FFT)."""
    out_len = x.shape[0] + y.shape[0] - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """a * a (full linear convolution)."""
    return _conv_full(a, a)


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)   # denominator = 1 because Σa = 1

    max‑convolution.  The temperature ε grows from 0.1 to ≈50 while the
    step‑size η decays from 0.12 to 5·10⁻⁴.
