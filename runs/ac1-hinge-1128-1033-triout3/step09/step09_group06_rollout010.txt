""" 
Search for a non‑negative step‑function minimizing `evaluate_sequence`.

The algorithm combines:
* a long mirror‑descent on a smooth log‑sum‑exp surrogate of the true
  max‑convolution objective (multiple eps‑stages, momentum and adaptive
  step‑size);
* an anti‑symmetric re‑ordering after each stage (guaranteed not to
  increase the true objective);
* an aggressive hill‑climbing phase that performs random swaps and
  small mass‐transfers while evaluating the true objective only at the
  end.
It works with up to 20000 coefficients and respects the 1000 s budget.
"""

import time, math, numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # avoid tiny negative numerical artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv

def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """∇_a Σ_k p_k (a*a)[k] ."""
    n = a.shape[0]
    # reverse p because convolution with reversed p gives the required cross‑correlation
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # indices n‑1 … 2n‑2 contain the needed values
    seg = conv[n - 1 : 2 * n - 1]
    return seg[::-1]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place biggest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)               # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]         # large at the left side
        b[-1 - i] = s[i]         # small at the right side
    if n % 2:
        b[half] = s[half]
    return b

def _internal_objective(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a,a))/sum(a)^2 – the same quantity that
    `evaluate_sequence` returns after the internal clipping."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

def _smoothed_descent(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector (as np.ndarray) seen during the run and its
    internal objective value.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = _internal_objective(a, n)

    # momentum buffer
    m = np.zeros_like(a)
    beta = 0.9

    # smoothing schedule – from very smooth to almost hard max
    eps_schedule = [0.5, 0.2, 0.1, 0.07, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]

    # total number of gradient steps we would like to execute
    total_iters = 15000
    it_per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        eta = 0.7                      # base step size for this stage
        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:
                break

            # ----- compute convolution and soft‑max weights -----
            conv = _full_convolution(a)
            max_c = conv.max()
            # soft‑max (stable version)
            shift = max_c
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()           # Σp = 1, p ≥ 0

            # ----- gradient of the smooth surrogate -----
            grad = 2.0 * _gradient_weighted_peaks(a, p)   # shape (n,)

            # ----- momentum update -----
            m = beta * m + (1.0 - beta) * grad

            # ----- mirror‑descent multiplicative update -----
            delta = -eta * m
            # keep the exponent in a numerically safe range
            delta = np.clip(delta, -10.0, 10.0)
            a = a * np.exp(delta)

            # ----- project back onto the simplex -----
            a = _project_simplex(a)

            # ----- keep track of the best internal value -----
            cur_val = 2.0 * n * max_c / (a.sum() ** 2)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                # occasional logging
                print(f"[descent eps={eps:.3g} it={it}] new best internal = {best_val:.6f}")

            # ----- decay the step size slowly (helps stability for small eps) -----
            eta *= 0.9995

        # a simple anti‑symmetric re‑ordering after each ε‑stage
        a = _anti_symmetric_reorder(a)

        # recompute internal objective after the re‑ordering (it never increases)
        cur_val = _internal_objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[reorder after eps={eps:.3g}] improved internal = {best_val:.6f}")

    return best_a, best_val

def _hill_climb_refine(a: np.ndarray,
                       cur_val: float,
                       n: int,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Light local search that only uses the cheap internal objective.
    Random swaps and tiny mass transfers are accepted when they strictly
    improve the objective.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        # decide which elementary move to try
        if rng.random() < 0.3:
            # swap two positions
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer a small proportion from i to j
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-16:
                continue
            # transfer at most 5 % of the source mass
            delta = best_a[i] * 0.05 * rng.random()
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta

        # projection onto simplex (keeps sum = 1 and non‑negativity)
        cand = _project_simplex(cand)

        # cheap internal objective (no call to evaluate_sequence)
        cand_val = _internal_objective(cand, n)

        if cand_val + 1e-12 < best_val:      # strict improvement
            best_val = cand_val
            best_a = cand
            print(f"[hill] improved internal = {best_val:.6f}")

    return best_a, best_val

"""search for a non‑negative step‑function minimizing `evaluate_sequence`

The algorithm consists of three phases

1. **Smoothed‑max descent** – mirror‑descent on a log‑sum‑exp
   surrogate of the max‑convolution.  A sequence of decreasing smoothing
   parameters (`eps`) drives the optimisation from a smooth landscape to
   the true max.  Momentum and a modest step‑size schedule improve
   convergence.  After every `eps`‑stage the vector is reordered by the
   anti‑symmetric rule (largest opposite smallest) – this operation can
   never increase the true objective.

2. **Hill‑climbing refinement** – a fast local search that repeatedly
   performs either a random swap or a tiny mass transfer between two
   positions.  The internal objective
   `2·n·max(conv(a,a))/sum(a)^2` is used for cheap evaluation; an
   improvement is kept and the vector is projected back onto the simplex.

3. **Final evaluation** – the best vector found is handed to the
   supplied `evaluate_sequence` function.

The code is written to respect the 1000 s time budget, works with up to
20 000 coefficients and is fully self‑contained (no filesystem or network
access).  All helper functions are top‑level and `evaluate_sequence` is
invoked only at the very end (and occasionally for logging)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padded FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # wipe tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·(a*a)[k] w.r.t. a.

    Result g_i = Σ_k p_k·a_{k‑i}  (out‑of‑range indices are treated as zero).
    It is computed with a single FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]                         # reverse the weight vector
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv[n - 1 : 2 * n - 1]           # keep the central part
    return seg[::-1]                         # reverse to obtain the gradient


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                    # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)      # fallback: uniform
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                          # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]                     # big → left side
        b[-1 - i] = s[i]                     # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _internal_objective(a: np.ndarray, n: int) -> float:
    """Cheap counterpart of `evaluate_sequence`: 2·n·max(conv)/sum(a)^2."""
    conv = _full_convolution(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
# Smoothed‑max descent (mirror‑descent with momentum)
# ----------------------------------------------------------------------


def _smoothed_descent(a_start: np.ndarray, n: int, deadline: float):
    """
    Perform a long mirror‑descent on the log‑sum‑exp surrogate of the
    max‑convolution.  Returns the best vector seen and its internal
    objective value.
    """
    a = a_start.copy()
    best_a = a.copy()

    # momentum buffer
    m = np.zeros_like(a)
    beta = 0.9

                    0.02, 0.01, 0.005, 0.002, 0.001]

    total_iters = 15000                     # total gradient steps we aim for
    it_per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        eta = 0.7                              # base step size for this stage
        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:

            shift = max_c                      # for numerical stability

# ----------------------------------------------------------------------
# Hill‑climbing refinement (cheap internal objective)
# ----------------------------------------------------------------------


def _hill_climb_refine(a: np.ndarray,
                       cur_val: float,

    Very light local search that uses random swaps or tiny mass transfers.
    A move is kept only if it strictly improves the cheap internal
    objective.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

            cand[i], cand[j] = cand[j], cand[i]
        else:                                      # tiny transfer
