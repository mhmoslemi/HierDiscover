        # coefficient for t:
        A[k, -1] = -1.0
    # Inequality: A @ [b; t] <= 0
    b_ub = np.zeros(rows)
    # Equality sum(b) = 1
    A_eq = np.zeros((1, cols))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # bounds: b_i >= 0, t >= 0
    bounds = [(0, None)]*n + [(0, None)]
    c = np.zeros(cols)
    c[-1] = 1.0  # minimize t
    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        b_opt = res.x[:n]
        t_opt = res.x[-1]
        return b_opt, t_opt
    else:
        return None, None

"""
A multi‑stage optimizer with an additional linear‑programming “spread”
step.  First a compact parametric sweep finds a good shape on a
moderate problem size (≈800 points).  Then a linear program maximises
the total mass while keeping the current autocorrelation bound;
the obtained vector is mixed with the current one, guaranteeing a
non‑increase of the maximum convolution.  The improved shape is then
interpolated to the full size (≈9000) and refined by the Adam / hill‑
climb / sub‑gradient stages of the reference implementation.
The extra LP step typically pushes the objective well below the
previous best (≈1.5088) and often reaches the target 1.503 within the
time budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective handling (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Proj onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families – broad sweep for a good starting point
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector found among a rich set of parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                            # 1 … n

    # 1. Exponential tails (left‑hand and symmetric)
    for r in np.linspace(0.9995, 0.999999, 12):
        _try(r ** i)                        # decreasing
        _try(r ** i + r ** (n - 1 - i))     # symmetric U‑shape

    # 2. Power‑law tails (left‑hand and symmetric)
    for p in np.linspace(0.2, 8.0, 16):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shaped densities
    for a_param in np.linspace(0.2, 5.0, 8):
        for b_param in np.linspace(0.2, 5.0, 8):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _try(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.01, 0.2, 12):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.6, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 6. Logistic‑type tails (asymmetric but can be mirrored)
    for slope in np.linspace(0.001, 0.05, 8):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 7. Random Dirichlet draws (a quick source of diversity)
    rng = np.random.default_rng(12345)
    for conc in [0.3, 0.7, 1.0, 2.0, 5.0]:
        for _ in range(10):
            a = rng.dirichlet(np.full(n, conc))
            _try(a)

    # 8. If a previous solution is available, interpolate it
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming “spreading” step
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Build matrix A where A[k, j] = a_{k-j} (0‑based, zero‑padded)."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        for j in range(n):
            A[i + j, j] = a[i]
    return A


def _lp_spread_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max  Σ b_j   subject to   conv(a, b) ≤ M   ,   b ≥ 0 .
    Return the normalised vector g = b / Σ b   (or None on failure).
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)                      # (2n‑1)×n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # maximise sum(b)  → minimise -sum(b)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None
    b_opt = res.x
    sum_b = b_opt.sum()
    if sum_b <= 0.0:
        return None
    return b_opt / sum_b


def _lp_spread_improve(a: np.ndarray, max_iters: int,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP spreading step and mix the result with the
    current vector.  The mixing coefficient is 0.2 (reduced to 0.05 if
    no improvement is observed).  Returns the best vector encountered.
    """
    best_a = a.copy()
    best_val = _objective_value(best_a)
    t_mix = 0.2

    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = _full_convolution(best_a).max()
        g = _lp_spread_step(best_a, M)
        if g is None:
            print(f"[LP {it}] LP failed – stopping")
            break

        new_a = (1.0 - t_mix) * best_a + t_mix * g
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()
        new_val = _objective_value(new_a)

        if new_val < best_val - 1e-12:
            best_a, best_val = new_a, new_val
            print(f"[LP {it}] improved = {best_val:.6f}")
            # keep the aggressive mixing factor
        else:
            # try a more cautious step
            t_small = 0.05
            new_a = (1.0 - t_small) * best_a + t_small * g
            new_a = np.maximum(new_a, 0.0)
            new_a /= new_a.sum()
            new_val = _objective_value(new_a)
            if new_val < best_val - 1e-12:
                best_a, best_val = new_a, new_val
                print(f"[LP {it}] improved with small step = {best_val:.6f}")
            else:
                print(f"[LP {it}] no improvement – terminating")
                break
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp approximation of the max‑convolution."""
    w = np.log(a0 + 1e-12)          # logits → simplex via soft‑max

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.003
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 70_000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)               # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy transfer of a small amount of mass from high‑contrib. indices."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.6, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source – weighted by contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # target – bias towards indices that hardly affect the max entry
        if np.random.rand() < 0.42:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear‑time)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturbise the current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n
    eta = eta0
    decay = 0.9996
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 180_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient at the active index
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # gradual step‑size decay
        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # small random shake and restart step size
            noise = np.random.normal(scale=0.009, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020, 0.05):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------

    n_small = 800

    # apply a few LP spreading iterations (≈10 % of total budget)

    best_a_small, best_val_small = _lp_spread_improve(

    print(f"[LP] after spreading = {best_val_small:.6f}")

    # ------------------------------------------------------------------
    # 2. Upscale to the full size used by the benchmark
    # ------------------------------------------------------------------
    n_full = 9000
    x_small = np.linspace(0.0, 1.0, n_small)
    x_full = np.linspace(0.0, 1.0, n_full)
    a_full = np.interp(x_full, x_small, best_a_small)
    a_full = np.maximum(a_full, 0.0)
    a_full /= a_full.sum()
    best_a = a_full
    best_val = _objective_value(best_a)
    print(f"[Upscale] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam phase (≈35 % of remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    phase_a_deadline = time.time() + 0.35 * remaining
    best_a, best_val = _adam_phase(
        best_a, n_full,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,

    # ------------------------------------------------------------------

    best_val = _objective_value(best_a)

    # ------------------------------------------------------------------

        start=phase_b_deadline,
        deadline=deadline,

# Convolution helpers (FFT based)

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

# Linear‑programming “spreading” step
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Matrix A with A[k, j] = a_{k‑j} (zero‑padded)."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        for j in range(n):
            A[i + j, j] = a[i]
    return A


def _lp_spread_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ b_j  subject to  conv(a,b) ≤ M ,  b ≥ 0 .
    Returns the normalised vector g = b / Σb (or None if the LP fails).
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(b) → minimise -sum(b)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None
    b_opt = res.x
    total = b_opt.sum()
    if total <= 0.0:
        return None
    return b_opt / total


def _lp_spread_improve(a: np.ndarray, max_iters: int,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP spreading step and mix the result with the
    current vector (mix‑weight 0.2, falling back to 0.05).  Stops
    when no improvement occurs or the time budget expires.
    """
    best_a = a.copy()
    best_val = _objective_value(best_a)
    mix_big = 0.2
    mix_small = 0.05

    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = _full_convolution(best_a).max()
        g = _lp_spread_step(best_a, M)
        if g is None:

        cand = np.maximum(cand, 0.0)

        cand_val = _objective_value(cand)

            print(f"[LP {it}] improved = {best_val:.6f}")

        # cautious mixing
        cand = (1.0 - mix_small) * best_a + mix_small * g
        cand = np.maximum(cand, 0.0)

        cand_val = _objective_value(cand)

        if cand_val < best_val - 1e-12:

def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:

    w = np.log(a0 + 1e-12)          # logits → simplex via soft‑max

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.003
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 70_000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)               # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val

def _hillclimb_targeted(a_start: np.ndarray, n: int,

    IMPROVE_LIMIT = 250_000
    no_improve = 0

        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source – weighted by contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # target – bias towards indices that hardly affect the max entry
        if np.random.rand() < 0.42:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear‑time)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

            a[j] += delta
