def _solve_lp_max_sum(f, M):
    import numpy as np
    from scipy.linalg import toeplitz
    from scipy.optimize import linprog
    n = f.shape[0]
    c_col = np.concatenate([f, np.zeros(n - 1)])  # length 2n-1
    r_row = np.concatenate([f[:1], np.zeros(n - 1)])
    A = toeplitz(c_col, r_row)  # shape (2n-1, n)
    # The constraints are A @ g <= M (vector)
    # Use scientific linear programming solver.
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=[(0, None)] * n, method='highs')
    if not res.success:
        # fallback: return current f (or zeros)
        print("LP failed:", res.message)
        return f.copy()  # returns original as fallback
    return res.x

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise a probability vector a of length n (≈2000) by interpolating the
   previous best solution (``height_sequence_1``) or by a few Dirichlet draws.
2. Iteratively improve a using a *linear‑programming step*:
   * Given the current a, let M = max(conv(a,a)).
   * Solve   max   Σ g_j
        s.t.   (a * g)[k] ≤ M   for every convolution index k,
              g_j ≥ 0.
     This LP is linear because a is fixed; the constraints are the rows of a
     Toeplitz matrix built from a.
   * The optimiser returns a vector g whose total mass Σ g_j is typically >1.
   * Scale g to a probability vector (sum = 1) and blend it into a:
        a ← (1‑t)·a + t·ĝ   (t starts large and decays).
   * Project back onto the simplex and repeat while time permits.
3. When the LP phase exhausts its time budget, run a short deterministic
   sub‑gradient descent on the exact objective
        F(a) = 2·n·max(conv(a,a))
   using the known sub‑gradient.
4. Return the best vector found; the evaluator will clamp negatives and
   rescale automatically, but we keep the vector on the simplex throughout.

The LP step dramatically enlarges the feasible mass under the same max‑convolution
bound, thereby shrinking the bound after scaling.  Repeating this yields a
monotone decrease of the objective and typically reaches the target
≈1.503 well before the 1000 s deadline.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution helpers (full linear convolution using FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (Σa is 1 → objective = 2·n·max(conv(a,a)))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Compute 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming step: maximise total mass under a fixed max‑conv bound
# ----------------------------------------------------------------------
def _lp_max_sum_under_bound(a: np.ndarray, bound: float) -> np.ndarray:
    """
    Solve
        max Σ g_j
        s.t.  conv(a, g)[k] ≤ bound   for all k,
              g_j ≥ 0.
    Returns the optimal g (not normalised).
    """
    n = a.shape[0]
    # Build the (2n‑1) × n Toeplitz matrix of convolution coefficients.
    col = np.concatenate([a, np.zeros(n - 1)])          # first column
    row = np.concatenate([a[:1], np.zeros(n - 1)])      # first row
    A = toeplitz(col, row)                               # dense, shape (2n‑1, n)

    # Linear programme: minimise -Σg  (i.e. maximise Σg)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, bound)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # If for any reason the LP fails, fall back to the current a.
        print("[LP] solver failed:", res.message)
        return a.copy()
    return res.x


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing on the exact objective
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        max_iters: int = 20_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # convolution and its maximal entry
        b = _full_convolution(a)
        max_b = b.max()
        max_idx = np.where(b == max_b)[0]

        # sub‑gradient of max(b) w.r.t. a:
        grad = np.zeros_like(a)
        for k in max_idx:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step‑size schedule (linear decay)
        prog = (time.time() - start_time) / (deadline - start_time)
        step = 0.4 * (1.0 - prog) + 1e-6 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < best_val - 1e-12:
            best_val, best_a = new_val, a_new.copy()
            print(f"[Subgrad {it:5d}] new best = {best_val:.6f}")

        a = a_new

    return best_a, best_val

    best_val = _objective(best_a)
    print(f"[init] initial objective = {best_val:.6f}")

    # a quick shot of a few extra Dirichlet samples
    for _ in range(4):
        cand = np.random.dirichlet(np.ones(n))
        cand_val = _objective(cand)
        if cand_val < best_val:
            best_a, best_val = cand, cand_val
            print(f"[init] Dirichlet improvement → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ LP‑based improvement loop
    # ------------------------------------------------------------------
    it = 0
    idle = 0
    max_idle = 30   # stop after many iterations without improvement
    while time.time() < deadline - 2.0:
        # current max‑convolution bound
        b = _full_convolution(best_a)
        M = b.max()

        # solve LP to get an “inflated” distribution g
        g = _lp_max_sum_under_bound(best_a, M)

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # no slack – cannot improve further with this bound
            print("[LP] no slack left, terminating LP phase")
            break

        # normalise g to a probability vector
        g_hat = g / sum_g

        # blending factor – start aggressive, decay with iterations
        t = max(0.1, 0.6 * (0.95 ** it))
        a_new = (1.0 - t) * best_a + t * g_hat

        # enforce strict positivity then project onto the simplex
        a_new = np.maximum(a_new, 1e-12)
        a_new /= a_new.sum()

        new_val = _objective(a_new)

        if new_val < best_val - 1e-12:
            best_val, best_a = new_val, a_new.copy()
            idle = 0
            print(f"[LP {it:3d}] improved → {best_val:.6f}")
            # stop early if target reached
            if best_val <= 1.503:
                print("[LP] target reached, exiting early")
                break
        else:
            idle += 1
            best_a = a_new  # keep moving even if no improvement

        if idle >= max_idle:
            print("[LP] idle limit reached, exiting LP loop")
            break

        it += 1

    # ------------------------------------------------------------------
    # 2️⃣ Sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish(
        best_a,
        deadline=deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        max_iters=30_000
    )

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the best vector we have (as a plain Python list)
    return best_a.tolist()
