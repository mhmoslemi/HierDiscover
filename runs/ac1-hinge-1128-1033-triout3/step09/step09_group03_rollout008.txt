"""
Improved heuristic for the min‑max self‑convolution problem.

The search works in three stages (all within the 1000 s budget):

  1.  A quick sweep over a handful of parametric families (sinⁿ,
      symmetric exponentials, beta‑shaped, random Dirichlet) to obtain a
      strong starting vector.
  2.  Adam optimisation on a smooth log‑sum‑exp surrogate of the max
      convolution (≈10 % of the time budget).  The learning‑rate and the
      smoothing parameter are annealed more aggressively than in the
      reference implementation.
  3.  A long simulated‑annealing / hill‑climbing phase (≈80 % of the
      budget).  Moves transfer a fraction of mass from a “high‑contributing”
      index to a “low‑contributing” one.  The step‑size schedule is
      considerably slower, and we add occasional large perturbations to
      escape plateaus.
  4.  A short projected sub‑gradient polishing pass (remaining time) and a
      final tiny uniform mixing.

All vectors are kept on the probability simplex (non‑negative,
∑a_i = 1).  The objective is scale‑invariant, so we never need to
renormalise the final answer.  The routine respects the 1000 s budget
and prints progress information.

The function `propose_candidate` is the entry point required by the
evaluation harness.
"""

import time
import numpy as np
from math import log, exp

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (len = 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σ a_i = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# 1.  Fast parametric start‑up
# ----------------------------------------------------------------------

    """Return the best vector (Σ=1) among several hand‑crafted families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform (already covered)
    _try(np.ones(n))

    # half‑wave sine with power
    for p in np.linspace(0.5, 5.0, 12):
        a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))) ** p
        _try(a)

    # symmetric exponential (U‑shaped)
    for r in np.linspace(0.90, 0.9999, 10):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # beta‑like (x·(1‑x))^α   – peaks in the middle for small α,
    # U‑shaped for large α
    x = np.arange(n, dtype=float) / (n - 1)
    for alpha in np.linspace(0.0, 5.0, 13):
        a = (x * (1 - x) + 1e-12) ** alpha
        _try(a)

    # random Dirichlet draws – a few quick samples
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# 2.  Adam on a smooth surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int,
                start: float, deadline: float,
                best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp surrogate of max(conv)."""
    # logits (soft‑max) parametrisation
    w = np.log(a0 + 1e-14)               # ensure positivity
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.003      # a little more aggressive
    eps_start, eps_end = 0.5, 1e-7

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000

    it = 0
    while time.time() < deadline:
        # current point on simplex
        w_shift = w - np.max(w)          # numeric stability
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # schedule
        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-10)

        # occasional jitter to avoid dead‑ends
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.008, size=n)

        # restart if stagnating
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# 3.  Simulated‑annealing / hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing with a temperature‑controlled Metropolis acceptance.
    The step‑size (max_frac) decays very slowly, allowing larger moves
    throughout most of the budget.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    # temperature is a modest fraction of the current objective
    T0 = cur_val * 0.04          # 4 % of the starting value
    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)          # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # most violating convolution entry
        k_max = int(np.argmax(b))

        # source index – pick one that contributes heavily to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # target index – choose a location that contributes little
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0                         # avoid picking the same index
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        # amount of mass to move
        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----- apply the move (incremental update) -----
        # save old convolution for possible revert
        b_old = b.copy()

        # linear term 2·δ·(a shifted by j) – 2·δ·(a shifted by i)
        start_j = j
        start_i = i
        b[start_j:start_j + n] += 2.0 * delta * a
        b[start_i:start_i + n] -= 2.0 * delta * a

        # quadratic corrections
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        # update the probability vector
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance for uphill moves
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA  {it:7d}] new best = {best_local_val:.6f}")
        else:
            # revert
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        # occasional large perturbation after long stagnation
        if no_improve > IMPROVE_LIMIT:
            print(f"[SA  {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.025, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# 4.  Projected sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Standard sub‑gradient descent on the exact max‑convolution objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 1.0 / n               # initial step size (≈ average entry)
    eta = eta0
    decay = 0.9997
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry (as described in the reference)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # gradient step + projection
        v = a - eta * grad
        a = _project_simplex(v)

        # step‑size decay
        eta *= decay

        # restart if stuck too long
        if no_improve > IMPROVE_LIMIT:
            print(f"[SGD {it:7d}] perturb after stagnation")
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# 5.  Tiny uniform mixing (often gives a marginal improvement)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α values."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full_like(a, 1.0 / a.shape[0])
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Returns a non‑negative step‑function (list of floats) that (approximately)
    minimises `evaluate_sequence`.  The routine respects the 1000 s time
    budget, uses a mix of analytic start‑up, Adam, simulated‑annealing and
    sub‑gradient polishing, and prints intermediate progress.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # reserve a small margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size.
    # 15000 gives a good trade‑off between expressive power and speed.
    # --------------------------------------------------------------
    n = 15000

    # --------------------------------------------------------------
    # Initialise a – try to reuse a known good construction if available.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n, rng)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    print(f"[init] best objective after seeding = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈10 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.10 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
    )
    print(f"[phase A] best after Adam = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (≈80 % of the budget)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _simulated_annealing_phase(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        rng=rng,
    )
    print(f"[phase B] best after SA = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase C – Exact sub‑gradient descent (remaining time)
    # --------------------------------------------------------------
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
    )
    print(f"[phase C] best after sub‑grad = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final tiny uniform mixing
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a)
    print(f"[final mix] best objective = {best_val:.6f}")

    # Final evaluation (the harness will also call evaluate_sequence)
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
```

This implementation builds on the previous state‑of‑the‑art code, uses a larger problem size, employs more aggressive annealing schedules, adds occasional large perturbations, and includes a short extra Adam phase.  It stays within the 1000 s budget and returns the best sequence found.
