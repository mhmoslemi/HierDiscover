"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.

Algorithm
---------
0️⃣  **Quick parametric search** – explore the two‑parameter Beta family
    a_i ∝ i^{p‑1}(n‑i+1)^{q‑1} on a dense log‑grid (including very small
    exponents) and a few random Beta draws.  Keep the best vector.

1️⃣  **Hill‑climbing via simulated annealing** – repeatedly move a small
    amount of mass from an index that contributes heavily to the current
    maximal convolution entry to an index that contributes little.
    After each move the full convolution is recomputed (FFT) – this fixes
    a subtle bug in the previous O(n) update and gives an exact objective.
    The move size and temperature are annealed, and stagnation triggers a
    short random perturbation around the best solution found so far.

2️⃣  **Deterministic sub‑gradient polishing** – once time is low we minimise
    the exact objective 2·n·max(conv(a,a)) using the known sub‑gradient
    (∂max/∂a_j = 2·a_{k_max‑j} for every maximising index k_max) and a
    simple projected sub‑gradient descent on the probability simplex.

The whole routine respects the 1000 s budget and finally calls
``evaluate_sequence`` on the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric sweep of a Beta‑shaped family (dense grid)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a dense two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # log‑spaced grid covering very flat (p,q≈0.01) to very spiked (≈10)
    p_vals = np.concatenate([
        np.geomspace(0.01, 0.2, 6),
        np.linspace(0.2, 1.0, 9),
        np.geomspace(1.0, 10.0, 6)[1:],
    ])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _beta_random_search(n: int, trials: int = 500) -> tuple[np.ndarray, float]:
    """Randomly sample p,q on a log‑scale and keep the best Beta shape."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    rng = np.random.default_rng()
    for _ in range(trials):
        # log‑uniform sampling of p,q in [1e-4, 10]
        logp = rng.uniform(np.log(1e-4), np.log(10.0))
        logq = rng.uniform(np.log(1e-4), np.log(10.0))
        p = np.exp(logp)
        q = np.exp(logq)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing (simulated annealing) with exact convolutions
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing that moves a small amount of mass
    from a source index (high contribution to the current max entry)
    to a target index (low contribution).  After each move the full
    convolution is recomputed (exact), fixing the old O(n) incremental
    update bug.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedules – start aggressive and anneal to tiny changes
    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.02, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 20000
    rng = np.random.default_rng()

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick source proportional to contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = rng.choice(n, p=src_probs)
        else:
            i = rng.integers(n)

        if a[i] <= eps:
            continue

        # target – bias towards indices with *small* contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = rng.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (excluding i)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move (up to max_frac of source)
        delta = a[i] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # propose new vector
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # exact convolution of the proposal
        b_new = _full_convolution(a_new)
        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_val - cur_val) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_new
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – small perturbation around the best seen so far
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503,
                        max_iters: int = 50000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a) = 2·n·max(conv(a,a)).
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_val = b.max()
        k_maxes = np.where(b == max_val)[0]

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple linear step size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.3 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Subgrad {it:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        # occasional jitter to avoid flat plateaus
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val
