def _project_simplex(v):

    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v

    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    # gradient via cross‑correlation (computed as convolution with reversed p)
    grad_full = _conv_fft(a, p[::-1], size_ap)      # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)                # map to the needed entries
    grad = 2.0 * grad_full[idx]                     # shape (n,)

    # we return the genuine objective (2·n·max_b) for consistency with the
    # rest of the code base.
    true_val = float(2.0 * n * max_b)
    return true_val, grad, b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a non‑negative step function that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds (≈0.5 s are reserved for
    the final evaluation call).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # reserve a tiny margin

    # problem size – 8000 gives a good trade‑off between expressiveness
    # and speed for the FFT based steps.
    n = 8000

    # --------------------------------------------------------------
    # 0️⃣ Warm‑start from any previously known solution
    # --------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Build a portfolio of cheap candidates and keep the best
    # --------------------------------------------------------------
    best_a, best_val = None, np.inf
    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] previous best value = {best_val:.6f}")

    try:
        beta_a, beta_val = _beta_family_best(n)
        if beta_val < best_val:
            best_a, best_val = beta_a, beta_val
            print(f"[init] beta‑family best value = {best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family sweep failed:", exc)

    # a few random Dirichlet seeds for diversity
    for _ in range(4):
        r = np.random.dirichlet(np.ones(n))
        v = _objective(r)
        if v < best_val:
            best_a, best_val = r, v
            print(f"[init] Dirichlet seed improved to {best_val:.6f}")

    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] fallback random start, value =", best_val)

    # --------------------------------------------------------------
    # 2️⃣  Smoothed convex optimisation (log‑sum‑exp surrogate)
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    best_seq = a.copy()
    best_seq_val = cur_val

    size_a = 1 << ((2 * n - 2).bit_length())      # FFT size for a∗a
    size_ap = 1 << ((3 * n - 2).bit_length())     # FFT size for a⋆p

    stagnation = 0
    IMP_LIMIT = 5000    # after this many non‑improving steps we perturb

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)   # ∈[0,1]

        # step size decays geometrically, β grows geometrically
        step = 0.2 * (0.01 / 0.2) ** prog      # 0.2 → 0.01
        beta = 5.0 * (200.0 / 5.0) ** prog      # 5 → 200

        # smooth objective, its gradient and the raw a∗a convolution
        true_val, grad, _ = _smooth_objective_and_grad(a, beta, size_a, size_ap)

        # projected gradient descent step on the simplex
        a_new = _project_simplex(a - step * grad)

        # evaluate the *true* objective on the new point
        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            # improvement – keep the new point
            cur_val = new_val
            a = a_new
            stagnation = 0
            if new_val < best_seq_val - 1e-12:
                best_seq_val = new_val
                best_seq = a.copy()
                print(f"[iter {iteration:06d}] NEW BEST = {best_seq_val:.6f}")
        else:
            # occasional random walk to avoid dead‑ends
            if np.random.rand() < 0.02:
                a = a_new
            stagnation += 1

        if stagnation > IMP_LIMIT:
            # escape a flat region via a small Gaussian perturbation
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            cur_val = _objective(a)
            stagnation = 0
            print(f"[iter {iteration:06d}] stagnation – perturb, val={cur_val:.6f}")

        if iteration % 2500 == 0 and iteration > 0:
            print(f"[iter {iteration:06d}] prog={prog:.3f} step={step:.5f} β={beta:.1f} cur={cur_val:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 3️⃣  Final evaluation via the provided harness
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = float('inf')

    return best_seq.tolist()
