"""
Optimiser for the step‑function evaluation problem.

The algorithm consists of three stages:
1️⃣  A broad parametric sweep (exponential, power‑law, beta, sine‑power,
    piecewise‑linear, normal‑pdf, random Dirichlet) – the best candidate is kept.
2️⃣  A long simulated‑annealing / hill‑climbing phase that moves tiny amounts of
    mass between indices while maintaining the simplex constraint.  The move is
    accepted if it improves the objective or with a Metropolis probability that
    slowly cools down.
3️⃣  A final projected sub‑gradient descent (with simple momentum) that polishes
    the solution, followed by a tiny mixture with the uniform distribution.

All arithmetic is performed on arrays that sum to 1, because the wrapper
`evaluate_sequence` rescales internally – the objective of interest is
    2 · n · max(conv(a,a)).
The code respects the 1000 s time budget, prints progress and finally returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, exact up to rounding)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (real signals)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (sum of a is 1 ⇒ obj = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {w≥0, Σw=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the (closed) probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families – a richer sweep for a good starting point
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Sweep a diverse set of parametric families and keep the best."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                           # 1 … n

    # 1. Exponential tails (decreasing & symmetric)
    for r in np.linspace(0.9995, 0.99999, 12):
        _try(r ** i)                       # left‑hand
        _try(r ** i + r ** (n - 1 - i))    # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 5.0, 15):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑shaped densities
    for a_param in np.linspace(0.2, 5.0, 8):
        for b_param in np.linspace(0.2, 5.0, 8):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 10):
        _try(base_sin ** p)

    # 5. Piecewise‑linear “tent” shapes
    for peak_frac in np.linspace(0.01, 0.4, 8):
        peak_idx = int(n * 0.5)
        left = np.linspace(0, 1.0, peak_idx + 1)
        right = np.linspace(1.0, 0, n - peak_idx)
        a = np.concatenate([left, right])
        a = a * peak_frac
        _try(a)

    # 6. Normal‑pdf shaped hills
    xs = np.linspace(-3, 3, n)
    for sigma in np.linspace(0.3, 1.5, 6):
        a = np.exp(-0.5 * (xs / sigma) ** 2)
        _try(a)

    # 7. Logistic tails (asymmetric & mirrored)
    for slope in np.linspace(0.005, 0.05, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 8. Random Dirichlet draws (more extensive)
    rng = np.random.default_rng(12345)
    for _ in range(300):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 9. Interpolate a previously found solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climbing phase (exact objective)
# ----------------------------------------------------------------------
def _annealing_phase(a0: np.ndarray, n: int,
                     start: float, deadline: float,
                     best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """
    Targeted mass‑transfer SA.
    Moves a small amount of probability mass from a source index (chosen
    proportionally to its contribution to the current maximal convolution entry)
    to a target index (chosen away from the active pair).  Accepts worsening
    moves with a temperature that decays linearly.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # temperature schedule
    T_start = 5e-3          # a few thousandths of the objective scale
    T_end   = 1e-7

    # fraction of transferred mass – slowly shrinks
    max_frac_start, max_frac_end = 0.35, 5e-4

    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # --- active convolution index -------------------------------------------------
        k_max = int(np.argmax(b))

        # contribution of each index to the active entry
        idx = k_max - np.arange(n)                # partner index for each i
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source selection – weighted by contribution (more likely to be “dangerous”)
        if total_contrib > 0.0:
            probs_src = contrib / total_contrib
            i = int(np.random.choice(n, p=probs_src))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-18:
            # nothing to move from this index – skip
            it += 1
            continue

        # target selection – we would like a partner that does *not* affect the active entry
        # use the inverse of the partner weights; add a tiny epsilon to avoid div‑by‑0.
        partner = k_max - i
        w = np.ones(n)
        if 0 <= partner < n:
            w[partner] = 0.0
        # also down‑weight indices with large a‑partner product
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0                           # we already took mass from i
        if w.sum() > 0.0:
            probs_tgt = w / w.sum()
            j = int(np.random.choice(n, p=probs_tgt))
        else:
            # fall back to a random different index
            j = (i + np.random.randint(1, n)) % n

        # amount to move – proportional to current mass at i
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            it += 1
            continue

        # ----- incremental convolution update (exact) ---------------------------------
        # store a copy of a for the update (the formulas assume the *old* a)
        a_old = a.copy()
        # update convolution (vectorised)
        b[j:j + n] += 2.0 * delta * a_old
        b[i:i + n] -= 2.0 * delta * a_old
        b[2 * j] += delta * delta
        b[2 * i] += delta * delta
        b[i + j] -= delta * delta

        # keep numerical noise non‑negative
        if np.min(b) < 0.0:
            b = np.maximum(b, 0.0)

        # evaluate new objective
        new_val = 2.0 * n * b.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp((cur_val - new_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            # commit the mass transfer
            a[i] = a_old[i] - delta
            a[j] = a_old[j] + delta
            cur_val = new_val

            # update best if improved
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")

        else:
            # revert the convolution change (undo the in‑place updates)
            b[j:j + n] -= 2.0 * delta * a_old
            b[i:i + n] += 2.0 * delta * a_old
            b[2 * j] -= delta * delta
            b[2 * i] -= delta * delta
            b[i + j] += delta * delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑conv objective)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Simple sub‑gradient descent with momentum and occasional perturbations."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.6 / n               # initial step size
    eta = eta0
    decay = 0.9996
    momentum = np.zeros_like(a)
    beta = 0.9

    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD   {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= 1.503:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of max entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        v = a - eta * momentum
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # perturb current best slightly and restart
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            momentum.fill(0.0)
            no_improve = 0
            print(f"[SGD   {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex mix with a small amount of uniform mass (polishing)."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list whose sum is 1 (the wrapper rescales internally).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Choose problem size – 8 000 works well for the time budget.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 1️⃣  Parametric sweep – quick exhaustive pass.
    # --------------------------------------------------------------
    print("[Init] start parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Simulated‑annealing hill‑climbing (≈45 % of remaining budget).
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * (deadline - start_time)
    best_a, best_val = _annealing_phase(best_a, n,
                                        start=start_time,
                                        deadline=phase_a_deadline,
                                        best_a=best_a,
                                        best_val=best_val)
    print(f"[Anneal] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Sub‑gradient descent (use the rest of the time).
    # --------------------------------------------------------------
    best_a, best_val = _subgradient_phase(best_a, n,
                                          start=phase_a_deadline,
                                          deadline=deadline,
                                          best_a=best_a,
                                          best_val=best_val)
    print(f"[SGD] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final tiny uniform blend (polishing).
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation via the secure wrapper (optional, for logging).
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
