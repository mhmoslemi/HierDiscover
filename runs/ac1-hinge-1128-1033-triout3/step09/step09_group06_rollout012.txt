"""
Optimization of non‑negative step function via p‑norm surrogate descent.

The algorithm proceeds as follows:
1. Initialise the sequence (re‑use the previous best if available,
   otherwise uniform) and normalise to unit sum.
2. Repeatedly minimise the smooth p‑norm
        ‖a∗a‖_p   (p grows from 2 to 256)
   using multiplicative mirror‑descent on the probability simplex.
   The gradient of the p‑norm can be written as a single FFT‑based
   convolution, so each iteration costs only O(n log n).
3. After each p‑stage the vector is reordered by the classic
   anti‑symmetric rule (largest ↔ smallest) which never increases the
   maximal convolution value.
4. Finally a cheap simulated‑annealing hill‑climber makes tiny mass
   transfers to polish the result.
The whole search respects a 1000‑second budget and returns the best
sequence found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·(a∗a)_k with respect to a.
    The result is 2·(a ∗ p_rev) restricted to the n entries that
    affect the convolution.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv[n - 1 : 2 * n - 1]
    g = seg[::-1]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σ x = 1}.
    """
    n = v.shape[0]
    # Sorting does the heavy lifting
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _objective_val(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) (sum of a is 1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()
