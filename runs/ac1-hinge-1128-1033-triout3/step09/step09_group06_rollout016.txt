def targeted_hill_climb(a, deadline):
    n = len(a)
    best = a.copy()
    best_val = compute_obj(best)
    rng = np.random.default_rng()
    while time.time() < deadline - 0.5:
        # compute conv and kmax once
        conv = _full_convolution_fft(best)
        kmax = conv.argmax()
        # contributions
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        i_range = np.arange(i_start, i_end + 1)
        j_range = kmax - i_range
        contrib = best[i_range] * best[j_range]
        total_contrib = contrib.sum()
        if total_contrib == 0:
            break
        probs = contrib / total_contrib
        # choose i weighted by contribution
        i = rng.choice(i_range, p=probs)
        # choose j random from all indices
        j = rng.integers(0, n)
        if i == j:
            continue
        # transfer
        delta = best[i] * rng.uniform(0.01, 0.07)
        if delta <= 0:
            continue
        a_new = best.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new = _project_simplex(a_new)
        val = compute_obj(a_new)
        if val < best_val:
            best_val = val
            best = a_new
            print(f"[target hill] new best = {best_val:.6f}")
        # also attempt swap moves occasionally
        if rng.random() < 0.2:
            i2 = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if i2 == j2:

            a_swap = best.copy()
            a_swap[i2], a_swap[j2] = a_swap[j2], a_swap[i2]

"""
Propose a non‑negative step function that minimises the
`evaluate_sequence` objective.

Algorithm
---------
* **Initial pool** – generate a handful of diversified seeds:
  uniform, exponential, several Beta‑shaped PDFs and (if present)
  the historic `height_sequence_1`.  Every seed is re‑ordered by the
  anti‑symmetric scheme (largest <-> smallest) and projected onto
  the probability simplex (∑a = 1).

* **Sub‑gradient descent on the true max** – for a fixed vector `a`
  the value `max_k (a*a)[k]` is a convex piecewise‑linear function.
  Its (sub‑)gradient at the active index `k*` is simply
      g_i = 2·a_{k*‑i}   (zero when the index is out of range).
  We perform many additive sub‑gradient steps with a diminishing
  step size η_t = η₀ / √t, each time projecting back onto the simplex.
  Every 200 iterations the anti‑symmetric ordering is reapplied; it
  never increases the convolution peak.

* **Targeted hill‑climbing** – after the descent we know the worst
  convolution index `k*`.  Contributions `a_i·a_{k*‑i}` are used as a
  probability distribution to pick a “heavy” index `i`.  A small
  amount of mass is moved from `i` to a random location `j`;
  the move is kept only if the true objective improves.
  Occasionally a random swap is tried.  This focuses the search
  exactly where the current maximum lives.

* **Time budget** – the whole pipeline runs until the 1000 s budget
  (minus a safety margin) expires, constantly keeping the best
  vector seen.  The final vector is checked with the official
  `evaluate_sequence` before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (all top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a zero‑padded FFT (length 2·n‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # eliminate tiny negative values caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest ones.
    For a sorted (ascending) array `s` the result is
    [s_max, s_{max‑1}, …, s_mid, s_mid‑1, …, s_min].
    """
    n = a.shape[0]
    s = np.sort(a)                # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]          # large → left side
        b[-1 - i] = s[i]          # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Objective (fast replica of `evaluate_sequence`)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Compute 2·n·max(a*a) / (sum a)^2.
    The vector is assumed to be non‑negative; we enforce the simplex
    constraint internally for numerical stability.
    """
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = _full_convolution_fft(a_norm)
    max_conv = conv.max()
    n = a.shape[0]
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------
def _generate_initial_candidates(n: int, rng: np.random.Generator):
    """
    Produce a short list of diverse starting vectors (all already
    simplex‑projected and anti‑symmetric reordered).
    """
    candidates = []

    # 1. uniform
    a = np.full(n, 1.0 / n)
    candidates.append(("uniform", a))

    # 2. exponential (sparse‑ish) distribution
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    candidates.append(("exp", a))

    # 3. sinusoidal (half‑wave)
    idx = np.arange(n) + 0.5
    a = np.sin(np.pi * idx / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    candidates.append(("sin", a))

    # 4. a few Beta‑shaped PDFs (α,β ∈ {0.5,1,2,3})
    x = np.linspace(0.0, 1.0, n)
    for α in (0.5, 1.0, 2.0, 3.0):
        for β in (0.5, 1.0, 2.0, 3.0):
            a = (np.power(x, α - 1.0) *
                 np.power(1.0 - x, β - 1.0))
            # replace NaNs (possible at the very ends) with a tiny constant
            a = np.nan_to_num(a, nan=0.0)
            a += 1e-12
            a /= a.sum()
            name = f"beta_{α}_{β}"
            candidates.append((name, a))

    # 5. historic construction, if it exists in the global scope
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # simple linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            old = np.interp(new_x, old_x, old)
        old = np.maximum(old, 0.0)
        if old.sum() > 0:
            old /= old.sum()
            candidates.append(("previous", old))
    except Exception:
        pass

    # post‑process: anti‑symmetric re‑ordering + simplex projection
    processed = []
    for name, vec in candidates:
        vec = _anti_symmetric_reorder(vec)
        vec = _project_simplex(vec)
        processed.append((name, vec))
    return processed


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a_start: np.ndarray,
                         deadline: float,
                         max_iters: int = 12000,
                         eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Minimise the true `max(a*a)` using a simple (additive) sub‑gradient
    method.  The algorithm keeps the vector on the simplex and
    periodically re‑applies the anti‑symmetric ordering.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(best_a)

    n = a.shape[0]
    it = 0
    while time.time() < deadline - 1.0 and it < max_iters:
        it += 1

        # current convolution and its worst index
        conv = _full_convolution_fft(a)
        max_conv = conv.max()
        kmax = int(conv.argmax())

        # sub‑gradient of the active max (g_i = 2·a_{kmax‑i})
        g = np.zeros_like(a)
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = kmax - i_range
        g[i_range] = 2.0 * a[j_range]

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # additive update + projection
        a_new = a - eta * g
        a_new = np.clip(a_new, 0.0, None)
        a_new = _project_simplex(a_new)
        a = a_new

        # occasional bookkeeping
        if it % 20 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[subgrad it {it}] new best = {best_val:.6f}")

        # every 200 steps apply the anti‑symmetric ordering
        if it % 200 == 0:
            a = _anti_symmetric_reorder(a)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climbing (focus on the current worst convolution index)
# ----------------------------------------------------------------------
def _targeted_hill_climb(a_start: np.ndarray,
                         deadline: float,
                         max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """
    Very light local search that repeatedly moves a small fraction of mass
    from a “heavy” contributor to a random location.  The heavy contributor
    is sampled proportionally to its contribution a_i·a_{k*‑i} to the current
    maximal convolution entry `k*`.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(best_a)

    n = a.shape[0]
    rng = np.random.default_rng()
    no_improve = 0

    while time.time() < deadline - 0.5 and no_improve < max_no_improve:
        # compute the worst convolution index once per outer loop
        conv = _full_convolution_fft(a)
        kmax = int(conv.argmax())

        # contributions to that index
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = kmax - i_idx
        contrib = a[i_idx] * a[j_idx]
        total = contrib.sum()
        if total == 0.0:
            break

        probs = contrib / total
        i = rng.choice(i_idx, p=probs)

        # pick a random destination
        j = rng.integers(0, n)
        if i == j or a[i] <= 1e-15:
            continue

        # transfer a small fraction (1‑7 %) of a[i] to j
        frac = rng.uniform(0.01, 0.07)
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.clip(a_new, 0.0, None)
        a_new = _project_simplex(a_new)

        val = _objective(a_new)
        if val < best_val - 1e-12:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            print(f"[hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional random swap (kept only if it improves)
        if rng.random() < 0.15:
            i2 = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if i2 == j2:
                continue
            a_swap = a.copy()
            a_swap[i2], a_swap[j2] = a_swap[j2], a_swap[i2]
            a_swap = _project_simplex(a_swap)
            val_swap = _objective(a_swap)
            if val_swap < best_val - 1e-12:
                best_val = val_swap
                best_a = a_swap.copy()
                a = a_swap
                no_improve = 0
                print(f"[swap] new best = {best_val:.6f}")

    return best_a, best_val

    Search for a non‑negative step‑function (list of floats) that minimises

    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a few seconds safety margin

    # problem size – 8000 gives a good trade‑off between quality and speed
    n = 8000

    # evaluate all candidates quickly (our fast replica) and keep the best few
    scored = []

        val = _objective(vec)
        scored.append((val, vec, name))

    top_seeds = scored[:4]                     # we keep at most four starters

    # ------------------------------------------------------------------
    # 2️⃣  Run the sub‑gradient descent + hill‑climbing on each starter
    # ------------------------------------------------------------------
    best_vec = None

    for rank, (init_val, vec, name) in enumerate(top_seeds, start=1):
        if time.time() >= deadline - 10.0:

        # sub‑gradient descent on the true max‑convolution

                                              max_iters=12000, eta0=0.6)

    # ------------------------------------------------------------------
    try:

        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_vec.tolist()
