"""
Propose a step‑function (non‑negative sequence) that minimises the provided
`evaluate_sequence` objective.

The algorithm consists of four stages:

1. **Broad parametric sweep** – generate candidates from many
   analytically‑defined families (exponential, power‑law, Beta‑shaped,
   sinusoidal, logistic, random Dirichlet).  The best among them is kept.

2. **Focused random search** – repeatedly sample a two‑parameter family
   `a_i ∝ ((i+1)*(n-i))^{-e}` and optionally mix it with the uniform
   distribution.  This quickly explores the region around the optimum.

3. **Deterministic hill‑climbing** – repeatedly move a small amount of
   mass from an index that heavily contributes to the current maximal
   convolution entry to an index that contributes little.  The convolution
   is updated incrementally (O(n) per move) and the move is accepted only
   if the objective improves.  Stagnation triggers a small random
   perturbation of the best solution found so far.

4. **Tiny uniform mixing** – a final convex combination with a few
   percent of the uniform distribution often smoothes the remaining peak.

All stages respect the simplex constraint (non‑negative entries that sum
to one) and are time‑controlled so that the function returns well before
the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector on the probability simplex."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Small mixture with uniform distribution (final polishing)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a tiny amount of the uniform distribution."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        # sum stays 1, no need to renormalise
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hill_climb(a_start: np.ndarray, start: float, deadline: float) -> tuple[np.ndarray, float]:
    """Greedy transfer of a small amount of mass from high‑contrib. to low‑contrib. indices."""
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a, best_val = a.copy(), cur_val

    no_improve = 0
    IMPROVE_LIMIT = 200_000  # after which we perturb the current best

    while time.time() < deadline:
        # progress‑dependent step size
        prog = (time.time() - start) / (deadline - start)
        max_frac = 0.6 * (1.0 - prog) + 0.001 * prog

        k_max = int(np.argmax(b))

        # contributions of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        # source = index with largest contribution
        i = int(np.argmax(contrib))

        # target = index with a small contribution (10‑th percentile)
        thresh = np.percentile(contrib, 10)
        low_idxs = np.where(contrib <= thresh)[0]
        if low_idxs.size > 0:
            j = int(np.random.choice(low_idxs))
        else:
            j = (i + np.random.randint(1, n)) % n

        if i == j or a[i] <= 1e-15:
            # nothing to move – try again
            continue

        # amount of mass to transfer
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        s = i + j
        if s < b_new.size:
            b_new[s] -= 2.0 * delta * delta
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            # accept move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb the current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print("[Hill] perturbing after stagnation")

    return best_a, best_val
