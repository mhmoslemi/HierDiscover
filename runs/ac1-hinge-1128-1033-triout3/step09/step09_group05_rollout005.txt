"""
Step‑function optimiser – softmax annealing + multi‑restart L‑BFGS.

We work with a probability vector a (a_i ≥ 0, Σa = 1).  The exact
objective is

    F(a) = 2·n·max_k (a * a)_k                (evaluate_sequence with Σa = 1)

Directly minimising the max is hard because it is non‑smooth.
We therefore minimise a smooth approximation

    M_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)   →  max_k (a*a)_k  as β→∞

and use the gradient of M_β.  The optimisation variable is x∈ℝⁿ with
    a_i = softmax(x)_i = e^{x_i}/Σ_j e^{x_j}
which automatically satisfies the simplex constraints.
For a fixed β we minimise 2·n·M_β(a) with L‑BFGS‑B, then increase β
(annealing) and repeat, using the previous solution as a warm start.
Multiple random restarts are performed until the 1000 s budget expires.
At the end a tiny uniform mixing step is applied (as in the reference
code) and the best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Fast full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _exact_objective(a: np.ndarray, n: int) -> float:
    """2·n·max(conv(a,a)).  Used for the final verification."""
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its gradient w.r.t. the softmax parameters
# ----------------------------------------------------------------------
def _softmax_obj_and_grad(x: np.ndarray, n: int, beta: float):
    """
    Returns (f, g) where
        f = 2·n·M_β(softmax(x))
    and g = ∇_x f .
    """
    # ---- a = softmax(x) ------------------------------------------------
    x_shift = x - np.max(x)          # avoid overflow
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()          # probability vector, Σa = 1

    # ---- convolution ---------------------------------------------------
    conv = np.convolve(a, a)         # length 2n‑1

    # ---- smooth max (log‑sum‑exp) --------------------------------------
    cmax = conv.max()
    shifted = beta * (conv - cmax)   # all ≤ 0  → safe exponentiation
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    M_beta = cmax + np.log(sum_exp) / beta

    # ---- softmax weights (∂M/∂conv) ------------------------------------
    w = exp_shifted / sum_exp        # shape (2n‑1,)

    # ---- gradient w.r.t. a  (∂M/∂a = 2· Σ_k w_k a_{k‑i}) ---------------
    #   Using correlation:  g_a[i] = 2 * Σ_j w[i+j] * a[j]
    #   This is exactly np.correlate(w, a, mode='valid')
    g_a = 2.0 * np.correlate(w, a, mode='valid')

    # ---- gradient w.r.t. x  (chain rule through softmax) -------------
    #   ∂a_i/∂x_j = a_i (δ_{ij} - a_j)
    #   Hence  g_x = a * (g_a - ⟨g_a, a⟩)
    dot = np.dot(g_a, a)
    g_x = a * (g_a - dot)

    # ---- final objective ------------------------------------------------
    f = 2.0 * n * M_beta
    return f, g_x


# ----------------------------------------------------------------------
# Simple deterministic families – cheap initial sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int, rng: np.random.Generator):
    """Return a decent start vector (Σa = 1) and its exact objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _exact_objective(best_a, n)

    def try_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _exact_objective(vec, n)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    idx = np.arange(n)

    # handful of classic shapes
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # U‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # a few random Dirichlet draws
    for _ in range(15):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Softmax annealing optimisation (multiple restarts)
# ----------------------------------------------------------------------
def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     deadline: float,
                     start_time: float,
                     init_beta: float = 5.0,
                     final_beta: float = 800.0,
                     beta_factor: float = 3.0,
                     max_iter_per_beta: int = 80) -> np.ndarray:
    """
    Optimise a starting vector a0 using a sequence of L‑BFGS runs on the
    smoothed objective with increasing β.  Returns the best a found before the
    deadline.
    """
    # transform to the unconstrained space
    x = np.log(a0 + 1e-15)                # avoid log(0)

    beta = init_beta
    best_a = a0.copy()
    best_val = _exact_objective(best_a, n)

    while time.time() < deadline and beta <= final_beta:
        # L‑BFGS for the current β
        res = minimize(fun=lambda xv: _softmax_obj_and_grad(xv, n, beta),
                       x0=x,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter_per_beta,
                                'ftol': 1e-12,
                                'disp': False})
        x = res.x

        # obtain the probability vector
        a = np.exp(x - np.max(x))
        a /= a.sum()

        # exact evaluation (the real target metric)
        val = _exact_objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Anneal β={beta:.0f}] new best = {best_val:.6f}")

        beta = min(beta * beta_factor, final_beta)

    return best_a


# ----------------------------------------------------------------------
# Final tiny uniform mixing (same spirit as the reference code)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> np.ndarray:
    """Blend a little uniform noise; keep the best."""
    best_a, best_val = a, _exact_objective(a, n)
    uniform = np.full(n, 1.0 / n)

    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _exact_objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a
