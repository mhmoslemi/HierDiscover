"""
Propose a step‑function minimizing `evaluate_sequence`.

Algorithm:
1. Initialise a non‑negative vector of length n (default 12000) from an
   exponential Dirichlet and normalise its sum to sqrt(2n) (the scale at
   which `evaluate_sequence` reduces to the peak of the self‑convolution).
2. Run a smoothed exponential‑mirror‑descent phase similar to the
   reference implementation, but with a slower step‑size decay and a
   higher final smoothing parameter β.  Periodically the current best
   vector is re‑ordered by the anti‑symmetric scheme.
3. When the descent stalls, switch to a *targeted hill‑climbing* phase.
   In each iteration the shift k that attains the current maximal
   convolution is identified.  Mass is transferred from the indices
   (i, k‑i) that contribute most to that peak towards indices that
   hardly affect it.  The move is accepted if it strictly reduces the
   objective, otherwise it may be accepted with a small temperature‑
   dependent probability (simulated‑annealing).
4. The best vector seen during the whole run is returned (as a plain
   Python list).  Throughout the run the true `evaluate_sequence`
   value of the best candidate is printed for logging.
"""

import time
import numpy as np

# ---------- helper FFT utilities ----------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear self‑convolution using an FFT of size power‑of‑two."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may create tiny negatives – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv

def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a*a)_k with respect to a.
    Input:
        a       – current vector (length n)
        w_rev   – reversed weight vector w (length 2n‑1)
    Returns:
        g_i = 2· Σ_j a_j·w_{i+j}
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1  # = 3n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1          # index aligning the centre (2n‑2)
    seg = conv[start:start + n]         # n‑length slice
    return 2.0 * seg

def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]       # large → left side
        b[-1 - i] = s[i]       # small → right side
    if n % 2:
        b[half] = s[half]
    return b

# ---------- objective utilities ----------
def _objective_from_conv(conv: np.ndarray, sum_a: float, n: int) -> float:
    """Return the value of evaluate_sequence for a given convolution."""
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _objective(a: np.ndarray) -> float:
    """Directly compute the evaluation without calling evaluate_sequence."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.sum(), a.shape[0])

# ---------- targeted hill‑climbing ----------
def _targeted_hill_climb(a: np.ndarray,
                         time_limit: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce the peak of the self‑convolution by repeatedly moving
    mass away from the indices that contribute most to the current
    maximum shift.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(best_a)
    start = time.time()
    it = 0

    while time.time() - start < time_limit:
        conv = _full_convolution(a)
        max_val = conv.max()
        # all shifts attaining the maximum
        ks = np.flatnonzero(conv == max_val)
        k = int(rng.choice(ks))

        # contributions of each index to this shift
        i_vec = np.arange(n)
        j_vec = k - i_vec
        mask = (j_vec >= 0) & (j_vec < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = a[i_vec[mask]] * a[j_vec[mask]]
        total = contrib.sum()

        if total > 0:
            probs = contrib / total
            i_donor = int(rng.choice(n, p=probs))
        else:
            i_donor = int(rng.integers(0, n))

        j_partner = k - i_donor
        donors = [i_donor]
        if 0 <= j_partner < n and j_partner != i_donor:
            donors.append(j_partner)

        # pick distinct recipients that are not donors
        recipients = []
        while len(recipients) < len(donors):
            r = int(rng.integers(0, n))
            if r not in donors and r not in recipients:
                recipients.append(r)

        # amount to move – a random fraction of the donor's mass
        a_new = a.copy()
        for donor, rec in zip(donors, recipients):
            if a_new[donor] <= 0:
                continue
            # fraction between 1 % and 10 %
            frac = float(rng.uniform(0.01, 0.10))
            delta = a_new[donor] * frac
            a_new[donor] -= delta
            a_new[rec] += delta

        # keep the total mass unchanged (numerical drift may appear)
        # we therefore renormalise to the original sum
        s = a_new.sum()
        if s <= 0:
            continue
        a_new *= a.sum() / s

        # verify bounds – nothing should be larger than 1000 (practically impossible)
        if np.any(a_new < 0):
            continue

        new_val = _objective(a_new)

        # accept only strict improvements, otherwise use a tiny SA probability
        if new_val < best_val - 1e-12:
            a = a_new
            best_a = a_new.copy()

"""
Propose a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. Initialise a non‑negative vector of length *n* (default 12000) from a
   Dirichlet distribution and renormalise its sum to √(2 n).  This scale
   makes the evaluation equal to the peak of the self‑convolution.
2. **Smoothed exponential‑mirror descent** – an accelerated version of
   the reference routine.  The smoothing parameter β grows to a larger
   value (≈600) and the step‑size decays slowly.  Every few thousand
   iterations the vector is anti‑symmetrically reordered (large values
   opposite small ones) to break residual structure.
3. **Targeted hill‑climbing** – when the descent stalls we switch to a
   specialised local search.  The shift *k* that attains the current
   maximal convolution is identified; the indices that contribute most
   to this peak are selected as donors and their mass is moved towards
   indices that hardly affect the peak.  Moves are accepted if they
   strictly improve the objective, otherwise a tiny simulated‑annealing
   probability is used.
4. The best vector seen during the whole run is returned as a plain list.
   The true `evaluate_sequence` value of the final candidate is printed
   for logging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear self‑convolution (length 2·len‑1) using an FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)   # eliminate tiny negatives
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. `a`.
    g_i = 2· Σ_j a_j·w_{i+j}.
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1                 # centre index (2n‑2)
    seg = conv[start:start + n]                # length n
    return 2.0 * seg


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Objective utilities (no side effects)
# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, sum_a: float, n: int) -> float:
    """Exact value of `evaluate_sequence` for a given convolution."""
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _objective(a: np.ndarray) -> float:
    """Fast evaluation of the objective (avoids the security wrapper)."""
    conv = _full_convolution(a)
    return _objective_from_conv(conv, a.sum(), a.shape[0])


# ----------------------------------------------------------------------
# Targeted hill‑climbing (simulated annealing flavour)
# ----------------------------------------------------------------------
def _targeted_hill_climb(a: np.ndarray,
                         time_limit: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce the peak of the self‑convolution by moving mass away from the
    indices that contribute most to the current maximal shift.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(best_a)
    start = time.time()
    it = 0

    while time.time() - start < time_limit:
        conv = _full_convolution(a)
        max_val = conv.max()
        ks = np.flatnonzero(conv == max_val)
        k = int(rng.choice(ks))                     # a shift attaining the maximum

        # contributions of each index to this shift
        i_vec = np.arange(n)
        j_vec = k - i_vec
        mask = (j_vec >= 0) & (j_vec < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = a[i_vec[mask]] * a[j_vec[mask]]
        total = contrib.sum()

        # pick a donor proportional to its contribution
        if total > 0:
            probs = contrib / total
            i_donor = int(rng.choice(n, p=probs))
        else:
            i_donor = int(rng.integers(0, n))

        j_partner = k - i_donor
        donors = [i_donor]
        if 0 <= j_partner < n and j_partner != i_donor:
            donors.append(j_partner)

        # pick distinct recipients that are not donors
        recipients = []
        while len(recipients) < len(donors):
            r = int(rng.integers(0, n))
            if r not in donors and r not in recipients:
                recipients.append(r)

        # move a random fraction of each donor’s mass
        a_new = a.copy()
        for donor, rec in zip(donors, recipients):
            if a_new[donor] <= 0:
                continue
            frac = float(rng.uniform(0.01, 0.10))   # 1 %–10 %
            delta = a_new[donor] * frac
            a_new[donor] -= delta
            a_new[rec] += delta

        # renormalise to the original total (tiny numerical drift)
        s = a_new.sum()
        if s <= 0:
            continue
        a_new *= a.sum() / s

        # compute new objective
        new_val = _objective(a_new)

        # strict improvement -> accept
        if new_val < best_val - 1e-12:
            a = a_new
            best_a = a_new.copy()
            best_val = new_val
            print(f"[hill {n} it {it}] new best = {best_val:.6f}")
        else:
            # occasional uphill move (simulated‑annealing)
            T = max(0.001, (time_limit - (time.time() - start)) / time_limit)
            if rng.random() < np.exp(-(new_val - best_val) / (T * best_val)):
                a = a_new

        it += 1
        if it % 2000 == 0:
            print(f"[hill {n} it {it}] current best = {best_val:.6f}")

    return best_a, best_val
