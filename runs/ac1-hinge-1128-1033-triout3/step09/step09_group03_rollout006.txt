At each iteration we:

1. Compute M = max(conv(f, f)).
2. Solve the linear program
        maximise   Σ g_j
        subject to conv(f, g) ≤ M   (element‑wise) ,   g ≥ 0 .
   The LP is sparse because the convolution constraints have a
   “consecutive‑ones” structure.
3. Rescale the LP solution to have Σ g = 1 and take a convex
   combination f ← (1‑t)·f + t·g (t chosen from a small list).
4. Keep the best vector seen so far.

A few sub‑gradient polishing steps are performed at the end.
The whole routine respects the 1000 s budget and returns the
best sequence it discovered.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (len = 2·len(a)‑1)."""
    return np.convolve(a, a, mode="full")


def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector (sum=1)."""
    n = a.shape[0]
    return 2.0 * n * _full_convolution(a).max()


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a quick hand‑crafted start vector of length n.
    Uses a half‑sine shape (known to be a decent starter) and
    adds a few random Dirichlet draws, keeping the best.
    """
    # half‑sine
    a = np.sin(np.pi * np.arange(1, n + 1) / (n + 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    best = a.copy()
    best_val = _objective_value(best)

    for _ in range(12):
        cand = rng.dirichlet(np.ones(n))
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best = val, cand.copy()
    return best


def _build_convolution_matrix(f: np.ndarray, row_idx, col_idx, i_idx):
    """
    Build the sparse matrix representing conv(f,·) ≤ M.
    The matrix has shape (2*n-1, n) and each row k contains the
    coefficients f_i for i that satisfy 0 ≤ k-i < n.
    """
    data = f[i_idx]                     # length = nnz
    n = f.shape[0]
    return scipy.sparse.csr_matrix(
        (data, (row_idx, col_idx)), shape=(2 * n - 1, n)
    )


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      row_idx, col_idx, i_idx) -> np.ndarray:
    """
    Solve the LP   maximise Σ g_j   s.t.  conv(f,g) ≤ M , g ≥ 0.
    Returns the optimal vector g (may contain small numerical negatives –
    they are clipped to zero by the caller).
    """
    n = f.shape[0]
    A_ub = _build_convolution_matrix(f, row_idx, col_idx, i_idx)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g   <=>   minimise -Σ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # use HiGHS (sparse‑aware) – it is fast for the sizes we use
    res = scipy.optimize.linprog(
        c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
        method="highs", options={"presolve": True}
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def _subgradient_polish(a: np.ndarray, deadline: float) -> np.ndarray:
    """A few iterations of classic sub‑gradient descent (like Phase C)."""
    n = a.shape[0]
    eta0 = 0.5 / n
    eta = eta0
    decay = 0.9995
    no_improve = 0
    IMPROVE_LIMIT = 100_000

    best = a.copy()
    best_val = _objective_value(best)

    while time.time() < deadline:
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val + 1e-14 < best_val:
            best_val, best = cur_val, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        k_max = int(np.argmax(conv))
        # gradient of the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # projected sub‑gradient step onto the simplex
        v = a - eta * grad
        # Euclidean projection onto simplex
        if np.abs(v.sum() - 1.0) > 1e-12:
            u = np.sort(v)[::-1]
            cssv = np.cumsum(u) - 1
            rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
            theta = cssv[rho] / (rho + 1)
            a = np.maximum(v - theta, 0.0)
        else:
            a = np.maximum(v, 0.0)
        a /= a.sum()

        eta *= decay
        if no_improve > IMPROVE_LIMIT:
            # small random perturbation to escape stagnation
            a = best + np.random.normal(scale=0.008, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
    return best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list of floats whose evaluation (via ``evaluate_sequence``)
    is expected to be lower than the previously reported 1.508809.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate n keeps the LP tractable while still
    # offering enough degrees of freedom.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Initialise f – use the globally stored construction if available,
    # otherwise fall back to a half‑sine shape with a few random tweaks.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        f = np.interp(new_x, old_x, old)
        f = np.maximum(f, 1e-12)
    except Exception:
        f = _initial_candidate(n, rng)

    # Normalise to a probability vector (sum = 1)
    f = np.maximum(f, 1e-12)
    f /= f.sum()

    best_f = f.copy()
    best_val = _objective_value(best_f)

    # ------------------------------------------------------------------
    # Pre‑compute the index pattern of the convolution matrix.
    # This depends only on n and will be reused for every LP solve.
    # ------------------------------------------------------------------
    row_idx, col_idx, i_idx = [], [], []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            row_idx.append(k)
            col_idx.append(k - i)
            i_idx.append(i)
    row_idx = np.array(row_idx, dtype=np.int32)
    col_idx = np.array(col_idx, dtype=np.int32)
    i_idx = np.array(i_idx, dtype=np.int32)

    # ------------------------------------------------------------------
    # Main LP‑based improvement loop.
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline and iteration < 30:
        # Current max of the self‑convolution
        conv_ff = _full_convolution(f)
        M = conv_ff.max()
        cur_val = 2.0 * n * M                            # sum(f)=1 ⇒ objective
        if cur_val < best_val - 1e-14:
            best_val, best_f = cur_val, f.copy()
            print(f"[LP   {iteration:3d}] new best = {best_val:.6f}")

        # Solve the LP to obtain a direction that maximises mass
        try:
            g0 = _solve_lp_max_sum(f, M, row_idx, col_idx, i_idx)
        except Exception as exc:
            print(f"[LP   {iteration:3d}] LP error: {exc}")
            break

        # Normalise the LP solution to a probability vector
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print(f"[LP   {iteration:3d}] LP returned zero sum – abort")
            break
        g = g0 / sum_g0

        # ------------------------------------------------------------------
        # Try several mixing coefficients and keep the best improvement.
        # ------------------------------------------------------------------
        best_local_f = f
        best_local_val = cur_val
        for t in (0.5, 0.3, 0.1, 0.05):
            cand = (1.0 - t) * f + t * g
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum()
            val = _objective_value(cand)
            if val < best_local_val - 1e-14:
                best_local_val, best_local_f = val, cand.copy()
        # Update f with the best candidate found this round
        f = best_local_f

        iteration += 1

    # ------------------------------------------------------------------
    # Polishing phase – a short sub‑gradient descent (Phase C).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        f = _subgradient_polish(f, deadline)

    # ------------------------------------------------------------------
    # Final sanity check – evaluate with the provided function.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(f.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return f.tolist()
