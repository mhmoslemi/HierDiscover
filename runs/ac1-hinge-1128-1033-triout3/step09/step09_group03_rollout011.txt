def _lp_refine(a, n, start, deadline, best_val):
    while time.time()<deadline:
        # compute conv
        b = _full_convolution(a)
        max_b = b.max()
        if max_b <= best_val/(2*n) + 1e-12: # No improve?
            break
        # active indices where b >= (1 - rel_tol)*max_b
        rel_tol = 0.01
        active = np.where(b >= max_b * (1 - rel_tol))[0]  # approx top 1%
        if len(active)==0:
            break
        # Build A_ub matrix: shape (len(active), n+1)
        # We'll use scipy.sparse to reduce memory.
        from scipy.sparse import lil_matrix
        A_ub = lil_matrix((len(active), n+1))
        for row_idx, k in enumerate(active):
            # For each i that contribute to k: i in [max(0,k-(n-1)), min(n-1,k)]
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            # For i in range(i_start,i_end+1):
            # coeff = 2 * a[k-i]
            # Note that k-i is in range too.
            # We'll vectorize:
            i_range = np.arange(i_start, i_end+1)
            coeffs = 2.0 * a[k - i_range]  # a_{j}
            A_ub[row_idx, i_range] = coeffs
            A_ub[row_idx, n] = -1.0  # M coefficient
        # b_ub = -b[active]
        b_ub = -b[active]

        # Equality constraint sum_i Δa_i = 0
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # Bounds for Δa_i
        lower_bounds = -a  # each Δa_i >= -a_i
        upper_bounds = [None]*n
        bounds = [(lb, None) for lb in lower_bounds] + [(None, None)] # for M

        # Solve
        res = linprog(c=np.concatenate([np.zeros(n), [1.0]]),  # objective minimize M
                      A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            break
        delta = res.x[:n]
        M_opt = res.x[-1]
        # Compute new objective = 2*n*M_opt
        new_val = 2.0 * n * M_opt
        if new_val < best_val - 1e-14:
            a = a + delta
            best_val = new_val
            print(f"[LP  {int(time.time() - start)}] new best = {best_val:.6f}")
        else:
            break
    return a, best_val

def _lp_refine_phase(a, n, start, deadline, best_val, rng):
    while time.time() < deadline:
        # compute conv, max
        b = _full_convolution(a)
        max_b = b.max()
        if 2.0*n*max_b >= best_val - 1e-12:
            # no improvement possible? break?
            # we can still try to refine anyway perhaps using tolerance.
            pass
        # active indices: top fraction
        # use tolerance = 0.01
        rel = 0.01
        active = np.where(b >= max_b * (1 - rel))[0]
        if len(active) == 0:
            break
        # Build A_ub
        from scipy.sparse import lil_matrix
        A_ub = lil_matrix((len(active), n+1))
        for row_idx, k in enumerate(active):
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i_range = np.arange(i_start, i_end+1)

                          bounds=bounds,

"""
min‑max self‑convolution optimizer

A three‑stage heuristic:
  1️⃣  Quick warm‑start on the simplex using a few hand‑crafted families.
  2️⃣  Long simulated‑annealing hill‑climbing (higher temperature & more
      perturbations than the reference implementation).
  3️⃣  Linear‑programming refinement on the active constraints of the
      max‑convolution (requires SciPy).  Followed by a short projected‑subgradient
      polishing pass and a tiny convex mixing with several analytic families.
The whole pipeline respects a 1000 s budget and returns the best
non‑negative step‑function it could find.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length = 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small library of parametric families – used for a quick start.
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector (Σa=1) among a few handcrafted families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # uniform
    _try(np.ones(n))

    # symmetric exponential (U‑shaped)
    for r in (0.9999, 0.9995, 0.998, 0.995):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _try(a)

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _try(a)

    # power‑law (slightly heavy tails)
    for p in (0.3, 0.5, 0.8):
        i = np.arange(1, n + 1, dtype=float)
        a = i ** (-p) + (n - i + 1) ** (-p)
        _try(a)

    # random Dirichlet draws – a few quick samples
    rng = np.random.default_rng(12345)
    for _ in range(20):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth (log‑sum‑exp) surrogate of the max‑convolution.
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp surrogate of max(conv)."""
    w = np.log(a0 + 1e-12)                  # logits → simplex via soft‑max
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.004
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 50000

    it = 0
    while time.time() < deadline:
        # Current point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # tiny jitter to escape flat regions
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.005, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val

# Phase B – Simulated‑annealing hill‑climbing (enhanced).
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator,
                               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Hill‑climbing with a higher initial temperature and aggressive moves."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = a.copy()
    best_local_val = cur_val

    # temperature as a fraction of the current objective, larger than before
    T0 = cur_val * 0.05                     # 5 % of the starting value
    max_frac_start, max_frac_end = 0.40, 0.0005
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)     # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # location of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # --- source selection (indices that heavily contribute to b[k_max]) ---
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # --- target selection (prefer indices that *do not* contribute much) ---
        inv = 1.0 / (contrib + 1e-14)
        inv[i] = 0.0                      # avoid picking the source again
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----- apply the move (incremental convolution update) -----
        b_old = b.copy()                 # cheap O(n) copy for easy revert
        # linear part 2·δ·(a shifted by j) – 2·δ·(a shifted by i)
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        # quadratic part  δ² at 2j, 2i and –2δ² at i+j
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        # update the probability vector
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # uphill move – accept with Metropolis probability
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA  {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            # revert everything
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the current state a bit and continue
            print(f"[SA  {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val

# Phase C – LP refinement on the active constraints (requires SciPy)
# ----------------------------------------------------------------------
def _lp_refine_phase(a: np.ndarray, n: int,
                     start: float, deadline: float,
                     best_val: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Linear‑programming refinement on the most active convolution entries."""
    # Try to import SciPy lazily; if unavailable we simply skip this phase.
    try:
        import scipy.optimize as opt
        import scipy.sparse as sparse
    except Exception:
        print("[LP] SciPy not available – skip refinement")
        return a, best_val

    # active‑set tolerance (percentage of the current maximum)
    rel_tol = 0.01          # 1 % of the peak
    IMPROVE_LIMIT = 5       # stop after a few successful refinements

    improvements = 0
    while time.time() < deadline and improvements < IMPROVE_LIMIT:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            print(f"[LP] current best = {best_val:.6f}")

        # select indices whose convolution value is close to the maximum
        active = np.where(b >= max_b * (1 - rel_tol))[0]
        if len(active) == 0:
            break

        # build sparse constraint matrix A_ub·x ≤ b_ub.
        # variables: Δa[0…n‑1] and M (last variable)
        rows = len(active)
        A_ub = sparse.lil_matrix((rows, n + 1), dtype=np.float64)
        for r, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
            coeffs = 2.0 * a[k - i_range]          # gradient 2·a_{k‑i}
            A_ub[r, i_range] = coeffs
            A_ub[r, n] = -1.0                       # -M
        A_ub = A_ub.tocsr()
        b_ub = -b[active]                         # because we wrote -M + Σ 2 a_{k-i} Δa_i ≤ -c_k

        # equality: Σ Δa_i = 0 (mass conservation)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # bounds: Δa_i ≥ -a_i, M unrestricted
        bounds = [(-a_i, None) for a_i in a] + [(None, None)]

        # objective: minimise M
        c = np.zeros(n + 1, dtype=np.float64)
        c[-1] = 1.0

        # call the solver
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})
        if not res.success:
            # if the LP fails we abort this phase
            print("[LP] linprog failed – abort refinement")
            break

        delta = res.x[:n]
        a_new = a + delta
        # numerical safety – enforce non‑negativity and re‑normalise
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        # evaluate the true objective
        new_val = _objective_value(a_new)
        if new_val < best_val - 1e-14:
            a = a_new
            best_val = new_val
            improvements += 1
            print(f"[LP] improvement #{improvements}, new best = {best_val:.6f}")
        else:
            # no improvement – stop trying further refinements
            break

    return a, best_val

# Phase D – Exact projected sub‑gradient descent (polishing)
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n            # initial step size (≈ half the average entry)
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 300_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient of the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step & projection onto the simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # decay step size
        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # small perturbation and reset step size
            noise = np.random.normal(scale=0.010, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturb after stagnation")

        it += 1

    return best_local_a, best_local_val
