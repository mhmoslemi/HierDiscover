"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – use a moderately large length (n≈10 000) to give the
   optimiser many degrees of freedom while remaining fast.

2. **Initial candidates** – the best among
   * interpolation of a previously found solution (if available),
   * a dense sweep of the two‑parameter Beta family,
   * random Beta draws (log‑uniform p,q),
   * a simple exponential (geometric) family a_i ∝ r^i,
   * a few Dirichlet samples.

3. **Hill‑climbing with exact convolution updates** – a fast O(n) stochastic
   local search that repeatedly moves a small amount of mass from an index
   that contributes heavily to the current maximal convolution entry to an
   index that contributes little.  The update of the full convolution is
   performed analytically (no FFT per move) and the quadratic term
   ``‑2·Δ²`` is handled correctly (the previous implementation missed the
   factor 2).  Simulated‑annealing acceptance with a temperature schedule
   allows climbing out of shallow plateaus.

4. **Deterministic sub‑gradient polishing** – a few thousand projected
   sub‑gradient steps on the exact objective improve the final solution
   to the last decimal places.

5. **Final evaluation** – call the provided ``evaluate_sequence`` on the
   best sequence before the time limit expires.

All heavy work (FFT) is done only during the cheap initial sweep; the
local search works with O(n) updates, so hundreds of thousands of moves
fit well inside the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple parametric families
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Exhaustive grid over a Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _beta_random_search(n: int, trials: int = 300) -> tuple[np.ndarray, float]:
    """Randomly sample (p,q) on a log‑scale for the Beta family."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    rng = np.random.default_rng()
    for _ in range(trials):
        logp = rng.uniform(-6.9, 2.3)   # ln(0.001) .. ln(10)
        logq = rng.uniform(-6.9, 2.3)
        p = math.exp(logp)
        q = math.exp(logq)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


def _exp_family_best(n: int, trials: int = 250) -> tuple[np.ndarray, float]:
    """Search a simple exponential family a_i ∝ r**i (0<r<1)."""
    best_a = None
    best_val = np.inf
    rng = np.random.default_rng()
    # sample α = -log(r) uniformly in log‑space
    for _ in range(trials):
        # α in [1e-6, 0.05] roughly; use log‑uniform sampling
        log_alpha = rng.uniform(math.log(1e-6), math.log(0.05))
        alpha = math.exp(log_alpha)
        r = math.exp(-alpha)
        i = np.arange(n, dtype=np.float64)
        a = r ** i
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing with exact convolution updates (correct Δ² term)
# ----------------------------------------------------------------------
def _hillclimb_exact(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Stochastic local search that moves a tiny amount of mass each step."""
    n = a0.shape[0]
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-14

    # schedule parameters
    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.02, 1e-12

    no_improve = 0
    IMPROVE_LIMIT = 100_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # compute contribution of each index to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – sample proportionally to contribution
        if total_contrib > 0.0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – bias towards low contribution
        tgt_weights = np.zeros_like(a)
        # encourage moves to positions that hardly affect the current max
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0  # never move mass back to the source
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback: uniform among all other indices
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # --- exact O(n) convolution update (using the old a values) ---
        b_new = b.copy()
        # add/subtract the linear term 2·Δ·a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic term (‑2·Δ² at the cross index, +Δ² at the two self‑indices)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        elif T > 0.0:
            prob = math.exp(-(new_val - cur_val) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # periodic re‑evaluation of the full convolution to avoid drift
        if iteration % 2000 == 0:
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        # stagnation handling – random jitter
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.03, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503,
                        max_iters: int = 30_000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact objective."""
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(a)
        max_val = b.max()
        ks = np.where(b == max_val)[0]

        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # linear decay of step size (aggressive early, tiny later)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.3 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Subgrad {it:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break

        # occasional jitter to shake off flat plateaus
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val

    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to ``evaluate_sequence``
    deadline = start_time + budget_s - 0.5

    # Problem size – a good trade‑off between expressivity and speed
    n = 10000

    # --------------------------------------------------------------
    # 0️⃣  Initialise a – try to interpolate a previously known solution
    # --------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 0 – cheap exploratory candidates
    # --------------------------------------------------------------
    best_a, best_val = None, np.inf

    if a0 is not None:
        val = _objective(a0)
        best_a, best_val = a0.copy(), val
        print(f"[init] start from interpolated solution, value={best_val:.6f}")

    # beta family grid
    try:
        shape_a, shape_val = _beta_family_best(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] beta‑family grid improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta‑family grid failed:", exc)

    # random beta draws
    try:
        rand_a, rand_val = _beta_random_search(n, trials=500)
        if rand_val < best_val:
            best_a, best_val = rand_a, rand_val
            print(f"[init] beta‑family random improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] beta random search failed:", exc)

    # exponential family search
    try:
        exp_a, exp_val = _exp_family_best(n, trials=400)
        if exp_val < best_val:
            best_a, best_val = exp_a, exp_val
            print(f"[init] exponential family improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] exponential family failed:", exc)

    # a few Dirichlet draws
    for _ in range(5):
        cand = np.random.dirichlet(np.ones(n))
        cand_val = _objective(cand)
        if cand_val < best_val:
            best_a, best_val = cand, cand_val
            print(f"[init] Dirichlet seed improved value={best_val:.6f}")

    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[init] fallback random start, value=", best_val)

    # --------------------------------------------------------------
    # Phase 1 – Hill‑climbing with exact updates
    # --------------------------------------------------------------
    hill_deadline = start_time + 0.85 * budget_s  # leave ~15 % for polishing
    best_a, best_val = _hillclimb_exact(
        best_a, deadline=hill_deadline,
        start_time=start_time,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503
    )
    print(f"[phase 1] best after hill‑climb = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Deterministic sub‑gradient polishing (remaining time)
    # --------------------------------------------------------------
    best_a, best_val = _subgradient_polish(
        best_a,
        deadline=deadline,
        start_time=hill_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
        max_iters=40_000
    )
    print(f"[phase 2] final polished value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
