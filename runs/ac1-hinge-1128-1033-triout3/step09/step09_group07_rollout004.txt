def project_simplex(v):
    # ensure nonnegative and sum=1 after projection onto simplex.
    v = np.maximum(v, 0)
    if v.sum() == 0:
        return np.full_like(v, 1.0 / len(v))
    # efficiently project onto simplex using sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of a real vector `x` (length 2·len‑1).
    Uses an FFT of length at least the next power of two for speed.
    Tiny negative values caused by round‑off are clipped to zero.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)      # eliminate numerical negatives
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_k w_k·(a*a)_k w.r.t. `a`.
    The result is g_i = 2·Σ_j a_j·w_{i+j}.
    Implemented as a single FFT‑based convolution.
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1                 # 2n‑2
    seg = conv[start:start + n]                # length n
    return 2.0 * seg


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    # make sure we are dealing with a 1‑D array
    v = np.asarray(v, dtype=float)
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    # algorithm from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # safeguard against numerical drift
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / len(v))
    return w / w_sum


def _objective(a: np.ndarray, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(conv(a,a)).
    The input `a` is assumed to be non‑negative and sum to 1 (normalised).
    """
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()

# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    `evaluate_sequence` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds free for the final evaluation & printing
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Problem size – 20 000 coefficients give a good trade‑off between
    # expressive power and the speed of the FFT‑based gradient.
    # ------------------------------------------------------------------
    n = 20000

    # ------------------------------------------------------------------
    # Initialise from the historic construction if possible,
    # otherwise start from a random Dirichlet (exponential) vector.
    # ------------------------------------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if a.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, a)
        a = np.maximum(a, 0.0)
        s = a.sum()
        a = a / s if s > 0 else rng.exponential(scale=1.0, size=n)
    except Exception:
        a = rng.exponential(scale=1.0, size=n)

    a = np.maximum(a, 0.0)
    a /= a.sum()                     # normalise onto the simplex

    # ------------------------------------------------------------------
    # Keep the best solution found so far.
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_obj = _objective(best_a, n)
    print(f"[init] n={n}, objective = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a few random restarts to avoid a poor start.
    # ------------------------------------------------------------------
    for trial in range(3):
        if time.time() >= deadline - 300:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = _objective(a_rand, n)
        if val < best_obj:
            best_obj = val
            best_a = a_rand.copy()
            a = a_rand.copy()
            print(f"[rand init {trial}] new best = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam‑style mirror descent on a smoothed maximum.
    # ------------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v_vec = np.zeros_like(a)      # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                      # base learning rate

    it = 0
    max_iter = 400000
    beta_start, beta_end = 1.0, 2000.0
    beta_ramp = 200000            # iterations over which β grows
    last_improve = 0

    while time.time() < deadline - 30 and it < max_iter:
        # ---- compute self‑convolution and its maximum ----
        conv = _full_convolution(a)
        max_conv = conv.max()
        obj = 2.0 * n * max_conv

        # ---- keep the best observed solution ----
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            last_improve = it
            print(f"[gd iter {it}] new best = {best_obj:.6f}")

        # ---- smooth approximation of the max (soft‑max) ----
        delta = conv - max_conv                     # ≤ 0
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp)
        w = np.exp(beta * delta)                    # w[max] = 1, others ≤ 1
        w_rev = w[::-1]

        # ---- gradient of Σ w_k·c_k ----
        g = _gradient_weighted(a, w_rev)

        # ---- Adam update (additive) ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_vec / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # ---- re‑project onto the simplex ----
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total <= 0.0:
            # pathological case – re‑initialise randomly
            a = rng.exponential(scale=1.0, size=n)
            a /= a.sum()
            m.fill(0.0)
            v_vec.fill(0.0)
        else:
            a /= total

        # ---- stagnation handling – occasional random perturbation ----
        if (it - last_improve) > 50000:
            # blend current vector with fresh noise
            noise = rng.exponential(scale=1.0, size=n)
            noise /= noise.sum()
            a = 0.65 * a + 0.35 * noise
            a = _project_onto_simplex(a)
            m.fill(0.0)
            v_vec.fill(0.0)
            last_improve = it
            print(f"[gd perturb at iter {it}]")

    # ------------------------------------------------------------------
    # Phase 2 – cheap hill‑climbing refinement (local stochastic search).
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_obj = best_obj

    while time.time() < deadline:
        if rng.random() < 0.22:
            # pure swap of two coefficients
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # tiny mass transfer from i to j
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex (fast)
        new_a = np.maximum(new_a, 0.0)
        sm = new_a.sum()
        if sm <= 0.0:
            continue
        new_a /= sm

        new_obj = _objective(new_a, n)
        if new_obj < cur_obj - 1e-12:
            cur_a = new_a
            cur_obj = new_obj
            best_a = new_a.copy()
            best_obj = new_obj
            print(f"[refine] new best = {cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure function.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] official evaluation = {final_score:.6f}")
    return best_a.tolist()
