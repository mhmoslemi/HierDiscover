def _deterministic_greedy(a0, n, deadline, start_time, best_val):
    a = a0.copy()
    b = _full_convolution(a)  # compute initial conv
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val_local = cur_val

    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 20000
    while True:
        now = time.time()
        if now >= deadline:
            break

        # schedule gamma from 0.5 to 0.01
        progress = (now - start_time) / (deadline - start_time)
        gamma = 0.5 * (1 - progress) + 0.01 * progress

        k = int(np.argmax(b))
        # contribution slice indices
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)  # inclusive
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k - idx]

        # source: max contribution
        src_rel = np.argmax(contrib)
        i_src = idx[src_rel]

        # target: minimal a (or minimal contribution)
        # Choose index with smallest a (avoid src)
        # Can also consider a target with minimal contribution but not source
        # simple: j = argmin a not equal src
        # but ensure we don't pick an index where a[j] is zero and delta cannot be moved (but that's fine)
        # compute a copy to find min
        # Exclude src
        a_non_src = a.copy()
        a_non_src[i_src] = np.inf
        j = int(np.argmin(a_non_src))

        # delta
        delta = a[i_src] * gamma
        if delta < 1e-12:
            # skip small moves
            continue

        b_new = b.copy()
        # update as per earlier formula
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a

        b_new[2 * i_src] += delta * delta
        b_new[i_src + j] -= delta * delta

            a[i_src] -= delta
            a[j] += delta

"""
Non‑negative step‑function optimizer for the evaluate_sequence objective.

The algorithm consists of four stages:
1️⃣   A cheap multi‑start exploration (uniform, Dirichlet draws,
     Beta‑family, Gaussian‑like, triangular and geometric families).
2️⃣   A deterministic greedy hill‑climbing that repeatedly moves a
     fraction of the most “responsible’’ mass (the one that contributes
     most to the current maximal convolution entry) toward a barely‑used
     index.  The convolution is updated in O(n) time using an exact
     incremental formula.
3️⃣   A short projected sub‑gradient polishing phase that works on the
     exact max‑convolution objective.
4️⃣   The final secure call to evaluate_sequence.

All stages respect the 1000 s time budget; intermediate progress and the
best objective value are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution (FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a already normalised."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climbing
# ----------------------------------------------------------------------
def _deterministic_greedy(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Move a fraction of the mass that contributes most to the current
    maximal convolution entry to a poorly contributing index.  The loop
    runs until the time budget is exhausted.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_obj = cur_val

    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 15000  # trigger jitter after many rejections

    while True:
        now = time.time()
        if now >= deadline:
            break

        # gradually shrink the transferred fraction (0.5 → 0.01)
        progress = (now - start_time) / (deadline - start_time)
        frac = 0.5 * (1.0 - progress) + 0.01 * progress

        # index where convolution attains its maximum
        k = int(np.argmax(b))

        # contributions of each a_i to b_k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        contrib = a[idx] * a[k - idx]

        # source = index with the largest contribution (and positive mass)
        src_rel = np.argmax(contrib)
        i_src = int(idx[src_rel])

        # target = index with the smallest a (excluding the source)
        a_tmp = a.copy()
        a_tmp[i_src] = np.inf
        j = int(np.argmin(a_tmp))

        # amount of mass to move
        delta = a[i_src] * frac
        if delta < 1e-14:
            # nothing to move – abort this iteration
            iteration += 1
            continue

        # ---- incremental convolution update (exact) ----
        # we use the *old* a for the linear part, as derived in the
        # original hill‑climb implementation.
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j] -= delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept only if we strictly improve the objective
        if new_val < cur_val - 1e-15:
            a[i_src] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_obj - 1e-12:
                best_obj, best_a = new_val, a.copy()
                print(f"[Greedy {iteration:7d}] new best = {best_obj:.6f}")
        else:
            no_improve += 1

        # stagnation handling – small random perturbation
        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Greedy {iteration:7d}] jitter after stagnation")
        iteration += 1

    return best_a, best_obj


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503,
                        max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(conv(a,a)).
    Simple stopping criteria: time budget or target achieved.
    """
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # compute sub‑gradient of max(b) w.r.t. a
        b = _full_convolution(a)
        max_val = b.max()
        k_maxes = np.where(b == max_val)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            grad[idx] += 2.0 * a[k - idx]          # ∂(a_i a_{k-i})/∂a_i = a_{k-i}
        # note: each max entry contributes twice (i ↔ k-i); the loop above
        # already accounts for both directions.

        # linear step size schedule (aggressive early, tiny later)
        progress = (now - start_time) / (deadline - start_time)
        step = 0.3 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Subgrad {it:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        # occasional jitter to escape flat regions
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val

    """
    Return a non‑negative sequence (sum≈1) that attempts to minimise
    evaluate_sequence within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # problem size – a balance between expressiveness and speed
    n = 8000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    candidates = []

    candidates.append(np.full(n, 1.0 / n))

    # 0.2 a few Dirichlet draws
    for _ in range(12):

    # 0.3 Beta‑family grid (coarse)
    i = np.arange(1, n + 1, dtype=np.float64)
    p_vals = np.linspace(0.2, 5.0, 7)
    q_vals = np.linspace(0.2, 5.0, 7)
    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s > 0:
                candidates.append(a / s)

    # 0.4 random Beta (log‑uniform sampling of parameters)
    rng = np.random.default_rng()
    for _ in range(120):
        logp = rng.uniform(-5.0, 2.0)   # ln(p) in [~0.0067, 7.4]
        logq = rng.uniform(-5.0, 2.0)
        p = np.exp(logp)
        q = np.exp(logq)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        s = a.sum()
        if s > 0:
            candidates.append(a / s)

    # 0.5 simple triangular shape (min(i+1, n-i))
    tri = np.minimum(i, n - i + 1).astype(np.float64)
    candidates.append(tri / tri.sum())

    # 0.6 geometric (r^i) families

    # 0.7 discrete Gaussian‑like shapes (σ as a fraction of n)

    for sigma_factor in [0.25, 0.5, 0.75, 1.0, 1.5]:
        sigma = sigma_factor * n
        gau = np.exp(-0.5 * (x / sigma) ** 2)
        candidates.append(gau / gau.sum())

    # evaluate all candidates (fast exact objective)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    print(f"[Init] best cheap candidate value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Deterministic greedy hill‑climbing (major work)
    # ------------------------------------------------------------------
    best_a, best_val = _deterministic_greedy(
        best_a, n, deadline, start_time, best_val
    )
    print(f"[Greedy] finished with best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient polishing (final refinement)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish(
        best_a, n,
        deadline, start_time,
        best_a, best_val,
        target_val=1.503,
        max_iters=30000
    )
    print(f"[Polish] final polished value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Secure final evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    # we return the raw list – the evaluation framework will compute the
    # official score via evaluate_sequence.
    return best_a.tolist()
