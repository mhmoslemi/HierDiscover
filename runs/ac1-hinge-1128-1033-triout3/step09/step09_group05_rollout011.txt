"""
Optimiser for the step‑function evaluation problem.

Main ideas
----------
* Initialise from the previous best (`height_sequence_1`) if available,
  otherwise use a quick scan of simple parametric families.
* Keep the vector normalised (sum = 1) – the objective 2·n·max(conv) is
  scaling‑invariant.
* Perform a *deterministic* mass‑transfer hill‑climb:
      – locate the current peak of the full convolution,
      – compute the contribution of each coefficient to that peak,
      – pick a few indices with the largest and smallest contributions,
      – for each candidate pair (i_from, i_to) solve a 1‑D optimisation
        problem for the optimal amount of mass δ to move (exact update of
        the convolution is used, including the quadratic Δa*Δa term).
      – apply the best improving move.
* Occasionally add a tiny Gaussian jitter when the search stagnates.
* Occasionally recompute the full convolution from scratch to avoid
  floating‑point drift.
* After the main loop finish, mix a tiny amount of the uniform vector
  (as in the reference code) and return the best sequence found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Fast full linear convolution (a*a) using FFT.
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes sum(a) = 1)
# ----------------------------------------------------------------------
def objective_from_conv(b: np.ndarray, n: int) -> float:
    """Exact value returned by evaluate_sequence for a normalised vector."""
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Helpers for the deterministic move.
# ----------------------------------------------------------------------
def rev_at_peak(a: np.ndarray, k: int):
    """Return the vector of contributions of each a_i to b[k] (size n)."""
    n = a.shape[0]
    rev = np.zeros(n, dtype=np.float64)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    rev[idx] = a[k - idx]
    return rev, lo, hi


def shift_contrib(a: np.ndarray, idx: int) -> np.ndarray:
    """Full‑length contribution of coefficient a[idx] to a*a.
    Returns an array of length 2·n‑1 with a placed at positions idx … idx+n‑1."""
    n = a.shape[0]
    size = 2 * n - 1
    out = np.zeros(size, dtype=np.float64)
    out[idx:idx + n] = a
    return out


def diff_and_quad_term(a: np.ndarray, i_from: int, i_to: int):
    """
    For a move δ from i_from → i_to compute:
        diff  = rev_i_to – rev_i_from    (linear term coefficient)
        quad  = vector c for the δ² term (non‑zero at 2*i_from, 2*i_to, i_from+i_to)
    Both vectors have length 2·n‑1.
    """
    n = a.shape[0]
    size = 2 * n - 1
    rev_from = np.zeros(size, dtype=np.float64)
    rev_to   = np.zeros(size, dtype=np.float64)

    rev_from[i_from:i_from + n] = a
    rev_to[i_to:i_to + n] = a

    diff = rev_to - rev_from

    quad = np.zeros(size, dtype=np.float64)
    quad[2 * i_from] += 1.0
    quad[2 * i_to]   += 1.0
    s = i_from + i_to
    if s < size:
        quad[s] -= 2.0
    return diff, quad


def optimal_delta(b: np.ndarray, diff: np.ndarray, quad: np.ndarray,
                  delta_max: float) -> (float, float):
    """
    Minimise max(b + 2·δ·diff + δ²·quad) for δ∈[0, delta_max].
    Returns (δ_opt, new_max) using a ternary search (the function is unimodal
    enough for the tiny intervals we deal with).
    """
    lo, hi = 0.0, delta_max
    # quick check of the ends
    f_lo = (b + 2 * lo * diff + (lo * lo) * quad).max()
    f_hi = (b + 2 * hi * diff + (hi * hi) * quad).max()
    # if the value at the right bound is already the best, skip the search
    if f_hi <= f_lo:
        return hi, f_hi

    for _ in range(25):           # 25 iterations give sub‑machine‑epsilon accuracy
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = (b + 2 * m1 * diff + (m1 * m1) * quad).max()
        f2 = (b + 2 * m2 * diff + (m2 * m2) * quad).max()
        if f1 > f2:
            lo = m1
        else:
            hi = m2
    delta_opt = (lo + hi) / 2.0
    new_max = (b + 2 * delta_opt * diff + (delta_opt * delta_opt) * quad).max()
    return delta_opt, new_max


# ----------------------------------------------------------------------
# Initial candidate generator (simple deterministic families + a few Dirichlet draws)
# ----------------------------------------------------------------------
def initial_candidate(n: int):
    """Return a decent starting vector (normalised) using cheap families."""
    rng = np.random.default_rng(12345)
    best_a = np.full(n, 1.0 / n)
    best_obj = objective_from_conv(full_convolution(best_a), n)

    def maybe_update(v):
        nonlocal best_a, best_obj
        v = np.asarray(v, dtype=np.float64)
        if v.ndim != 1 or v.shape[0] != n:
            return
        if np.any(v < 0):
            return
        s = v.sum()
        if s <= 0:
            return
        v = v / s
        b = full_convolution(v)
        val = objective_from_conv(b, n)
        if val < best_obj:
            best_obj, best_a = val, v.copy()

    # deterministic families (mirroring the previous code)
    idx = np.arange(n)
    maybe_update(np.arange(n, 0, -1, dtype=float))
    maybe_update(np.maximum(idx + 1, n - idx))
    maybe_update((np.arange(n, 0, -1, dtype=float) ** 2))
    maybe_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))
    maybe_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))

    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        maybe_update(a)

    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        maybe_update(a)

    for _ in range(12):
        maybe_update(rng.dirichlet(np.ones(n)))

    return best_a, best_obj

    # --------------------------------------------------------------

                if delta_max <= 1e-15:
                    continue
                diff_vec, quad_vec = diff_and_quad_term(a, i_from, i_to)
                delta_opt, new_max = optimal_delta(b, diff_vec, quad_vec, delta_max)
                new_obj = 2.0 * n * new_max
                if new_obj < best_local_obj - 1e-14:
                    best_local_obj = new_obj
                    best_pair = (i_from, i_to)
                    best_delta = delta_opt
                    best_diff = diff_vec
                    best_quad = quad_vec

        # ----- Apply the best improving move (if any)
        if best_pair is not None:
            i_from, i_to = best_pair
            a[i_from] -= best_delta
            a[i_to] += best_delta
            # exact update of the full convolution
            b = b + 2.0 * best_delta * best_diff + (best_delta * best_delta) * best_quad
            best_obj = best_local_obj
            best_a = a.copy()
            stagnation = 0
            print(f"[Move {iteration:7d}] new best = {best_obj:.6f}")
            # periodic recomputation to curb numerical drift
            if iteration % RECOMP_INTERVAL == 0:
                b = full_convolution(a)
            continue

        # ----- No improvement found – increase stagnation counter
        stagnation += 1

        # ----- Occasionally try a random Dirichlet perturbation
        if stagnation % 500 == 0:
            cand = np.random.default_rng().dirichlet(np.ones(n))
            cand_b = full_convolution(cand)
            cand_obj = objective_from_conv(cand_b, n)
            if cand_obj < best_obj:
                a = cand
                b = cand_b
                best_obj = cand_obj
                best_a = cand.copy()
                stagnation = 0
                print(f"[Random {iteration:7d}] new best = {best_obj:.6f}")
                continue

        # ----- When stuck, add a tiny Gaussian jitter and re‑normalise
        if stagnation >= STAGNATE_LIMIT:
            sigma = 0.003 * (1.0 / n)  # scale with typical coefficient size
            noise = np.random.normal(scale=sigma, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)  # recompute from scratch
            stagnation = 0
            print(f"[Jitter {iteration:7d}] applied small Gaussian noise")

    # --------------------------------------------------------------
    # Final tiny uniform mixing (helps escape shallow local minima)
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        mixed = (1.0 - alpha) * best_a + alpha * uniform
        mixed /= mixed.sum()
        mixed_b = full_convolution(mixed)
        mixed_obj = objective_from_conv(mixed_b, n)
        if mixed_obj < best_obj:
            best_obj = mixed_obj
            best_a = mixed.copy()
            print(f"[Mix] alpha={alpha:.3f} improved to {best_obj:.6f}")

    # --------------------------------------------------------------
    # Final verification with the provided security wrapper
    # --------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
