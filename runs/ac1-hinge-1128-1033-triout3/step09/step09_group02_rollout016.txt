"""
search_step_function.py

A lightweight multi‑stage optimiser for the step‑function evaluation task.
The optimiser proceeds as follows:

1. **Broad parametric sweep** – generate a diverse set of candidate
   vectors from many families (uniform, exponential, power‑law,
   beta, arcsine, logistic, piecewise‑constant, random Dirichlet).
   The best candidate from this sweep is kept.

2. **Simulated‑annealing hill‑climber** – iteratively move a small
   amount of mass from indices that contribute most to the current
   maximal convolution entry toward indices that contribute the least.
   Moves are accepted if they improve the objective, otherwise with
   probability `exp(-Δ/T)`.  The temperature and maximal transfer
   fraction decay with time.

3. **Targeted interior mixing** – after the stochastic search we try
   a few deterministic convex combinations with specially crafted
   “interior‑uniform’’ vectors (mass placed only on the low‑mass
   region of the current best vector).  This often reduces the peak
   of the convolution without changing the total sum, yielding a final
   improvement.

The routine returns a list of non‑negative floats ready for
`evaluate_sequence`.  All work respects the 1000‑second time budget.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (sum is kept = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Parameteric families – larger and more exhaustive sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Search a wide library of parametric families for a good start."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(arr):
        nonlocal best_a, best_val
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                           # 1 … n

    # 1. Exponential (both sides)
    for r in np.linspace(0.9990, 0.99999, 25):
        _try(r ** i)                       # decreasing tail
        _try(r ** i + r ** (n - 1 - i))    # symmetric U‑shape

    # 2. Power‑law (extended exponent range)
    for p in np.linspace(0.1, 10.0, 30):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution (including very small parameters)
    for a_param in np.linspace(0.05, 5.0, 20):
        for b_param in np.linspace(0.05, 5.0, 20):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Arcsine (Beta(0.5,0.5)) – classic U‑shape
    arc = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    _try(arc)

    # 5. Sine‑power family (half‑wave sinusoid)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 15):
        _try(base_sin ** p)

    # 6. Logistic‑type tails (asymmetric and mirrored)
    for slope in np.linspace(0.002, 0.06, 12):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 7. Piecewise‑constant “high‑at‑edges’’ shapes
    for k_frac in np.linspace(0.005, 0.15, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.6, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 0.9:               # avoid trivial near‑uniform
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 8. Linear ramps (edges taper to a low interior)
    for w_frac in np.linspace(0.01, 0.2, 10):
        w = max(1, int(round(w_frac * n)))   # taper width
        for high in np.linspace(0.2, 0.8, 8):
            low = (1.0 - 2 * w * (high + low := 0))   # placeholder – will be recomputed
        # (skip – already covered by previous family)

    # 9. Random Dirichlet draws (more diversity)
    rng = np.random.default_rng(12345)
    for _ in range(60):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 10. Interpolate previous best if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_sa(a_start: np.ndarray, n: int,
                  start: float, deadline: float,
                  best_a: np.ndarray, best_val: float,
                  target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA optimiser (exact max‑convolution)."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max

    # temperature schedule – start relatively high, cool down exponentially
    T0 = 0.4 * cur_val          # generous early acceptance
    T_end = 1e-6
    # maximal transfer fraction schedule
    max_frac_start, max_frac_end = 0.5, 0.001

    it = 0
    while time.time() < deadline:
        # progress based schedules
        prog = (time.time() - start) / (deadline - start)
        T = T0 * (T_end / T0) ** prog
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # ----- pick source (i) weighted by contribution to the current max -----
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))   # fallback

        if a[i] <= 1e-16:
            continue

        # ----- pick target (j) – favour indices that hardly affect the max entry -----
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0.0:
            probs_j = w / w.sum()
            j = int(np.random.choice(n, p=probs_j))
        else:
            j = (i + np.random.randint(1, n)) % n

        # ----- propose a transfer -----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Incremental convolution update (exact)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- acceptance test -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probabilistic uphill move
            if T > 0.0:
                prob = math.exp(-(new_val - cur_val) / T)
                if random.random() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[SA {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Final deterministic mixing with interior‑uniform vectors
# ----------------------------------------------------------------------
def _final_mixing(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Blend with interior‑only uniform mass; also try arcsine mix."""
    best_a, best_val = a, _objective_value(a)

    # interior region – pick the ~70 % least massive indices
    threshold = np.percentile(a, 70)
    interior_mask = a <= threshold
    if interior_mask.sum() >= 2:
        interior_u = np.zeros_like(a)
        interior_u[interior_mask] = 1.0 / interior_mask.sum()
        for alpha in (0.001, 0.005, 0.010, 0.020, 0.050):
            cand = (1.0 - alpha) * a + alpha * interior_u
            cand /= cand.sum()
            val = _objective_value(cand)
            if val < best_val:
                best_val, best_a = val, cand

    # mix with the arcsine shape (Beta(0.5,0.5))
    arc = 1.0 / np.sqrt((np.arange(n) + 0.5) * (n - 0.5 - np.arange(n)))
    arc /= arc.sum()
    for alpha in (0.001, 0.003, 0.007, 0.015):
        cand = (1.0 - alpha) * a + alpha * arc
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand

    return best_a, best_val


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a non‑negative step‑function within the allotted time.
    Returns a list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – a few thousand entries balances quality & speed
    n = 9000

    print("[Init] start parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Simulated‑annealing hill‑climber (≈55 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.55 * (deadline - start_time)
    best_a, best_val = _hillclimb_sa(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase A] best after SA = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Deterministic interior‑mixing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _final_mixing(best_a, n)
    print(f"[Phase B] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the supplied secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
