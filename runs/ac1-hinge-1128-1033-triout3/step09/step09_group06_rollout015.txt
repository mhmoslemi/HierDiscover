    eval_interval = 100
    reorder_interval = 500

            beta2_pow *= beta2
            m_hat = m / (1 - beta1_pow)
            v_hat = v / (1 - beta2_pow)
            step = -base_lr * (m_hat / (np.sqrt(v_hat) + eps_adam))
            # clip for stability
            step = np.clip(step, -20.0, 0.0)
            a = a * np.exp(step)
            # project onto simplex
            a = _project_simplex(a)

            # occasional evaluation
            if it % eval_interval == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.4f} it {it}] new best = {best_val:.6f} (max_conv {max_conv:.6e})")
                # reorder check
                if it % reorder_interval == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    val_ord = evaluate_sequence(a_ord.tolist())
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        # reset optimizer state after reorder
                        m = np.zeros_like(a)
                        v = np.zeros_like(a)
                        beta1_pow = 1.0
                        beta2_pow = 1.0
                        print(f"[reorder eps {eps:.4f} it {it}] improved to {best_val:.6f}")

        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow = 1.0
        beta2_pow = 1.0

    return best_a, best_val

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.

Algorithm
---------
1. **Multiple restarts** – different lengths (2000 … 16000) and several
   simple seeds (uniform, sinusoid, exponential, power‑law, previous
   construction).

2. **Smoothed‑max descent with Adam** – minimise a log‑sum‑exp surrogate
   of the convolution peak:
        f_ε(a) = (1/ε)·log∑_k exp(ε·(a*a)[k]).
   The gradient of the surrogate is obtained with two FFTs
   (full convolution + weighted‑peak convolution).  An Adam‑style
   exponentiated‑gradient (mirror‑descent on the simplex) updates the
   vector, keeping it non‑negative and of unit sum.  A schedule of
   decreasing ε (1 → 0.001) moves from a smooth landscape to the true
   max.

3. **Anti‑symmetric re‑ordering** – after a fixed number of iterations
   the current vector is reordered by pairing the largest and smallest
   entries (largest at the ends, smallest in the centre).  This never
   increases the maximal convolution value and often improves it.

4. **Hill‑climbing refinement** – cheap local search (random swaps,
   small mass transfers, and targeted transfers from the largest to the
   smallest entries).  Only strictly improving moves are kept.

All heavy work (FFT‑based gradient steps) is performed without calling
`evaluate_sequence`; the true objective is queried only occasionally.
The routine respects the 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)  # eliminate tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·b_k w.r.t. a,
    where b_k = (a*a)[k] and p is a (2n‑1) weight vector (Σp = 1, p≥0).

    Resulting gradient g_i = Σ_k p_k·a_{k‑i}
    (out‑of‑range indices are taken as 0).  Computed with one FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # indices n‑1 … 2n‑2 (inclusive) correspond to the required slice
    seg = conv[n - 1 : 2 * n - 1]
    # reverse back to obtain the gradient
    return seg[::-1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]            # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest ones (pairwise).
    The resulting arrangement never increases the maximal convolution
    value (rearrangement inequality).
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Smoothed‑max descent using Adam (exponentiated mirror‑descent)
# ----------------------------------------------------------------------


def _smoothed_descent_adam(a_start: np.ndarray, deadline: float):
    """
    Minimise the log‑sum‑exp surrogate of max‑convolution.
    Returns the best vector encountered and its true evaluation.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent init] objective = {best_val:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.7

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # ε‑schedule (large → small)
    eps_schedule = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]

    eval_interval = 100   # call evaluate_sequence only every 100 steps
    reorder_interval = 500

    for eps in eps_schedule:
        it = 0
        # iterate until time runs out (leaving a few seconds margin)
        while time.time() < deadline - 5:
            # -------- convolution and soft‑max weights ----------
            conv = _full_convolution(a)            # length 2n‑1
            max_conv = conv.max()
            shift = max_conv
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()          # Σp = 1, p≥0

            # -------- gradient of the smooth surrogate ----------
            grad = 2.0 * _gradient_weighted_peaks(a, p)   # shape (n,)

            # -------- Adam update (exponentiated) -------------
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1 - beta1_pow)
            v_hat = v / (1 - beta2_pow)
            step = -base_lr * (m_hat / (np.sqrt(v_hat) + eps_adam))
            step = np.clip(step, -20.0, 0.0)       # avoid overflow/underflow
            a = a * np.exp(step)                  # multiplicative update
            a = _project_simplex(a)                # keep Σa = 1, a≥0

            # -------- occasional true‑objective check ----------
            if it % eval_interval == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.4f} it {it}] new best = {best_val:.6f}"
                          f" (max conv {max_conv:.6e})")

                if it % reorder_interval == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    val_ord = evaluate_sequence(a_ord.tolist())
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        # reset Adam state after a drastic re‑ordering
                        m = np.zeros_like(a)
                        v = np.zeros_like(a)
                        beta1_pow = 1.0
                        beta2_pow = 1.0
                        print(f"[reorder eps {eps:.4f} it {it}] improved to"
                              f" {best_val:.6f}")

            it += 1

        # final deterministic reordering for the current ε‑stage
        a = _anti_symmetric_reorder(a)
        # reset Adam statistics before the next ε‑stage
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow = 1.0
        beta2_pow = 1.0

    return best_a, best_val

def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float, n: int):
    """
    Random swaps / tiny mass transfers.
    Each move is kept only if it **strictly** improves the true objective.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    # pre‑computed helper arrays for targeted moves
    top_fraction = max(1, n // 20)  # 5 % of indices

    while time.time() < deadline - 2:
        r = rng.random()
        if r < 0.15:
            # -------- targeted transfer from a large to a tiny entry --------
            large_idxs = np.argpartition(-best_a, top_fraction)[:top_fraction]
            i = int(rng.choice(large_idxs))
            small_idxs = np.argpartition(best_a, top_fraction)[:top_fraction]
            j = int(rng.choice(small_idxs))
            if best_a[i] <= 0:
                continue
            delta = best_a[i] * 0.1 * rng.random()

            new_a[i] -= delta
            new_a[j] += delta

            if i == j:

            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
