"""
Propose a step‑function (list of non‑negative floats) that minimises
`evaluate_sequence`.  

The algorithm consists of three parts

1. **Pre‑processing** – try to reuse a historic construction (height_sequence_1),
   otherwise start from a random Dirichlet vector.  The vector is normalised to
   unit L1‑norm, because the objective is homogeneous.

2. **Smoothed exponential‑mirror descent** – we repeatedly
   * compute the self‑convolution,
   * build a smooth approximation of the maximum (`soft‑max`),
   * obtain the gradient of the weighted sum via a second FFT,
   * take a multiplicative (exponential‑mirror) step and re‑project onto the
     simplex.
   The smoothing parameter β grows from 1 to a few thousand, so the descent
   gradually focuses on the true maximum.  A simple stagnation detector injects
   random noise when progress stalls.

3. **Fine‑grained local search** – after the main descent we prune leading /
   trailing entries that are essentially zero (this reduces the factor *n* in
   the objective) and perform a cheap hill‑climbing phase with random swaps
   and tiny mass transfers, accepting only *strict* improvements.

The whole routine respects the 1000 s time budget, prints occasional progress
(and stops early if the target bound `1.503` is reached), and finally returns
the best list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (required for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a real vector (length 2·len‑1) via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can produce tiny negatives – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. `a`.  Implements the formula
    g_i = 2·∑_j a_j·w_{i+j}  (where w is reversed for convenient convolution).
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3 n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]

    # Index of the first element that corresponds to i = 0.
    start = w_rev.shape[0] - 1                 # = 2 n‑2
    seg = conv[start:start + n]                # length n
    return 2.0 * seg


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Order the vector so that the largest values sit opposite the smallest ones."""
    s = np.sort(a)                     # ascending
    n = s.shape[0]
    b = np.empty_like(s)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _trim_zeros(v: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Remove leading and trailing entries that are ≤ eps.
    The function is used after the descent to lower the factor `n`
    in the objective without changing the convolution peak much.
    """
    idx = np.where(v > eps)[0]
    if idx.size == 0:
        # all entries are negligible – fall back to a tiny uniform vector
        return np.full(1, 1.0)
    return v[idx[0] : idx[-1] + 1]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # keep a few seconds for finalisation

    # --------------------------------------------------------------
    # Problem size – 4000 coefficients are a good compromise:
    #   * FFTs are fast enough for many iterations.
    #   * The search space is still rich enough to beat the existing bound.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 0. Initialise from historic construction if available,
    #    otherwise fall back to a random Dirichlet sample.
    # --------------------------------------------------------------
    try:
        # height_sequence_1 may be defined globally by the runner.
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, a)
        a = np.maximum(a, 0.0)
        total = a.sum()
        a = a / total if total > 0 else np.full(n, 1.0 / n)
    except Exception:
        a = rng.exponential(scale=1.0, size=n)
        a = np.maximum(a, 0.0)
        a /= a.sum()

    # --------------------------------------------------------------
    # Keep the best solution seen so far.
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Small random‑restart stage – a few fresh starts in the early budget.
    # --------------------------------------------------------------
    for tr in range(4):
        if time.time() >= deadline - 150:
            break
        a_rand = rng.exponential(scale=1.0, size=n)
        a_rand /= a_rand.sum()
        val = evaluate_sequence(a_rand.tolist())
        if val < best_val:
            best_val = val
            best_a = a_rand.copy()
            print(f"[rand init {tr}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. Smoothed exponential‑mirror descent.
    # --------------------------------------------------------------
    a = best_a.copy()
    it = 0
    max_iter = 250_000                     # upper bound – will be cut by time.
    # step‑size schedule (α decays slowly)
    alpha0 = 0.70
    # smoothing schedule for β – start soft, finish sharp.
    beta_start = 1.0
    beta_end = 1500.0
    beta_ramp_iters = 120_000               # linear ramp length

    last_improve_it = 0
    while time.time() < deadline - 30 and it < max_iter:
        # ----- convolution and smooth max weights -----
        conv = _full_convolution(a)                 # length 2n‑1
        max_conv = conv.max()

        # smooth weights approximating the max (they lie in (0,1])
        delta = conv - max_conv                     # ≤ 0
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)
        # avoid overflow for very large β – clip the exponent range.
        w = np.exp(np.clip(beta * delta, -40.0, 0.0))   # w[max]=1, others ≤1

        # ----- gradient of the weighted sum (FFT‑based) -----
        g = _gradient_weighted(a, w[::-1])           # shape (n,)

        # ----- step‑size (slow decay) -----
        # a practical decay: α = α0 / (1 + it/5000)
        alpha = alpha0 / (1.0 + it / 5000.0)

        # ----- mirror‑descent multiplicative update -----
        delta_exp = -alpha * g
        delta_exp = np.clip(delta_exp, -25.0, 25.0)   # exp(±25) ≈ 7e10
        a = a * np.exp(delta_exp)

        # keep positivity, re‑project onto the simplex
        a = np.maximum(a, 1e-14)
        a /= a.sum()

        # ----- periodic logging & possible re‑ordering -----
        if it % 400 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                last_improve_it = it
                print(f"[gd iter {it}] new best = {best_val:.6f}")

            # try anti‑symmetric re‑ordering every 4000 it
            if it % 4000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    last_improve_it = it
                    print(f"[gd reorder {it}] improved to {best_val:.6f}")

        # ----- stagnation handling – occasional random perturbation -----
        if (it - last_improve_it) > 20_000:
            # blend the current vector with a fresh random Dirichlet sample
            noise = rng.exponential(scale=1.0, size=n)
            noise /= noise.sum()
            a = 0.65 * a + 0.35 * noise
            a = np.maximum(a, 1e-14)
            a /= a.sum()
            last_improve_it = it
            print(f"[gd perturbed at {it}]")

        it += 1

    # --------------------------------------------------------------
    # 2. Trim leading/trailing near‑zero entries (reduces factor “n”).
    # --------------------------------------------------------------
    trimmed = _trim_zeros(best_a, eps=1e-12)
    # Re‑normalise to keep the objective homogeneous (not strictly necessary,
    # but avoids pathological zero‑sum if everything was trimmed away).
    trimmed = np.maximum(trimmed, 0.0)
    if trimmed.sum() == 0.0:
        trimmed = np.full(1, 1.0)
    trimmed /= trimmed.sum()
    trim_val = evaluate_sequence(trimmed.tolist())
    if trim_val < best_val:
        best_val = trim_val
        best_a = trimmed.copy()
        print(f"[trim] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Final hill‑climbing refinement (random swaps & mass transfers).
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    while time.time() < deadline:
        # pick operation
        if rng.random() < 0.2:                     # pure swap (20 %)
            i, j = rng.integers(0, cur_a.shape[0], size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # tiny mass transfer (80 %)
            i, j = rng.integers(0, cur_a.shape[0], size=2)
            if i == j or cur_a[i] < 1e-14:
                continue
            # transfer a random fraction (up to 4 %) from i to j
            delta = cur_a[i] * (0.04 * rng.random())
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto the simplex (non‑negativity + normalisation)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:          # strict improvement only
            cur_a = new_a
            cur_val = val
            best_a = new_a.copy()
            best_val = val
            print(f"[refine] new best = {cur_val:.6f}")

            # if we have already beaten the target, stop early
            if best_val < 1.503:
                print("[target reached]")
                break

    # --------------------------------------------------------------
    # Final evaluation & return
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_a.tolist()
