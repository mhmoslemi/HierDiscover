def simulated_annealing(a, remaining_time, rng):
    n = a.shape[0]
    b = full_convolution(a)
    cur_val = 2*n*b.max()
    best_val = cur_val
    best_a = a.copy()
    start = time.time()
    # set schedule parameters:
    T0 = cur_val * 0.05
    Tmin = cur_val * 1e-5
    start_frac = 0.5
    end_frac = 1e-3
    max_no_improve = 200000
    no_improve = 0
    iter = 0
    while time.time() - start < remaining_time:
        prog = (time.time() - start) / remaining_time
        T = T0 * (1 - prog) + Tmin * prog
        max_frac = start_frac * (1 - prog) + end_frac * prog
        k_max = int(np.argmax(b))
        # compute contributions:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        # source selection:
        if contrib.sum() > 0 and rng.random() < 0.9:
            probs = contrib / contrib.sum()
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(0, n)
        # target selection:
        # weight by (max_contrib - contrib) to choose low contributions
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_t = inv / inv.sum()
        j = rng.choice(n, p=probs_t)
        if i == j:
            continue
        delta = a[i] * max_frac * rng.random()
        if delta <= 0:
            continue
        # incremental update:
        b_old = b.copy()
        # partner arrays: we need copy of a for later revert if reject
        a_i = a[i]
        # update b:
        b[i:i+n] -= 2.0 * delta * a
        b[j:j+n] += 2.0 * delta * a
        two_i = 2*i
        two_j = 2*j
        ij = i + j
        b[two_i] += delta*delta
        b[two_j] += delta*delta
        b[ij] -= 2.0 * delta*delta
        # update a:
        a[i] -= delta
        a[j] += delta
        # compute new val
        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1
        if no_improve > max_no_improve:
            # perturb:
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
        iter += 1
    return best_a, best_val

def subgradient_descent(a, time_remaining):
    n = len(a)
    start = time.time()
    best_a = a.copy()
    cur_val = 2 * n * np.max(full_convolution(a))
    best_val = cur_val
    eta = 1.0  # large
    decay = 0.9999
    while time.time() - start < time_remaining:
        b = full_convolution(a)
        cur_val = 2 * n * b.max()
        # update best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # Identify k_max
        k_max = int(np.argmax(b))
        # gradient:
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]
        # Step:
        v = a - eta * grad
        a = project_simplex(v)
        eta *= decay
        # occasionally random noise
        if np.random.rand() < 0.01:
            a += np.random.normal(scale=0.001, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
    return best_a, best_val

"""
Propose a step‐function (non‑negative list of floats) that minimises
`evaluate_sequence`.  The optimiser combines several ideas:

*   A simple parametric initialisation (uniform, sine, exponential,
    Dirichlet or a known previous construction).
*   A long simulated‑annealing phase that works directly on the exact
    objective.  The move consists of transferring a small amount of mass
    `δ` from a source index to a target index while updating the full
    self‑convolution **incrementally** (O(n) per trial).  The schedule for
    temperature and for the maximal transferred fraction gradually cools.
*   A fast projected sub‑gradient polishing phase with a larger initial
    step size than the reference implementation.
*   Several independent restarts (as many as the 1000 s budget permits)
    and a tiny uniform‑mixing post‑processing step.

The code is entirely self‑contained, uses only NumPy and the supplied
`evaluate_sequence`, and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of a with itself (len = 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * full_convolution(a).max())


# ----------------------------------------------------------------------
# Tiny post‑processing: convex mix with the uniform vector
# ----------------------------------------------------------------------
def mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Try a few convex combinations (1‑α)·a + α·uniform, keep the best."""
    n = a.shape[0]
    best_a = a
    best_val = objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing phase (exact objective, incremental updates)
# ----------------------------------------------------------------------
def simulated_annealing(a0: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Classical SA that moves a small amount of mass from a source index i
    (chosen among those that contribute heavily to the current maximal
    convolution entry) to a target index j (chosen among low‑contributing
    indices).  The convolution is updated incrementally in O(n).
    """
    a = a0.copy()
    n = a.shape[0]
    b = full_convolution(a)                       # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    best_a = a.copy()
    best_val = cur_val

    # temperature is a fraction of the starting value and decays linearly
    T_start = cur_val * 0.05                       # a bit hotter than the reference
    T_end = cur_val * 1e-4
    max_frac_start, max_frac_end = 0.5, 0.001    # allow large moves early
    max_no_improve = 150_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))
        # the above line intentionally keeps prog in [0,1] without storing start
        prog = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))  # dummy to avoid unused‑var warning
        # proper prog:
        prog = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))  # keep placeholder
        # ---------------------------------------------------------
        # NOTE: we recompute a clean progress at each loop using a local start
        # ---------------------------------------------------------
        # (the above three lines are harmless; the actual prog is recomputed below)
        # ---------------------------------------------------------
        # now compute real prog:
        start = deadline - (deadline - time.time())       # dummy, will be overwritten
        # The above is just to silence linter – we compute prog correctly:
        prog = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))
        # ---------------------------------------------------------
        # Actually we maintain a dedicated start time:
        if it == 0:
            start = time.time()
        prog = (time.time() - start) / (deadline - start)

        temperature = T_start * (1.0 - prog) + T_end * prog
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # ---------------------------------------------------------
        # Choose source i (heavy contributor to the maximal entry)
        # ---------------------------------------------------------
        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and rng.random() < 0.9:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(rng.integers(0, n))

        # ---------------------------------------------------------
        # Choose target j (prefers low contributors)
        # ---------------------------------------------------------
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        # -----------------------------------------------------------------
        # Decide how much mass to move
        # -----------------------------------------------------------------
        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # -----------------------------------------------------------------
        # Incremental convolution update
        # -----------------------------------------------------------------
        b_old = b.copy()                     # cheap O(n) copy for possible revert
        # linear part 2·δ·a (both directions)
        b[i:i + n] -= 2.0 * delta * a
        b[j:j + n] += 2.0 * delta * a
        # quadratic correction
        two_i = 2 * i
        two_j = 2 * j
        ij = i + j
        b[two_i] += delta * delta
        b[two_j] += delta * delta
        b[ij] -= 2.0 * delta * delta

        # update the probability vector (sum stays 1)
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0.0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-14:
                best_val, best_a = new_val, a.copy()
                print(f"[SA   {it:7d}] new best = {best_val:.6f}")
        else:
            # revert
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > max_no_improve:
            # random shake around the current best
            print(f"[SA   {it:7d}] perturb after stagnation")
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (polishing phase)
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses a larger initial step size than the reference code and a slower
    decay schedule to allow more aggressive moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = objective_value(a)

    eta = 0.5               # larger than 0.5/n
    decay = 0.9999
    max_no_improve = 120_000
    no_improve = 0
    it = 0
    start = time.time()

    while time.time() < deadline:
        b = full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD  {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step & projection onto simplex
        v = a - eta * grad
        a = project_simplex(v)

        eta *= decay

        if no_improve > max_no_improve:
            # small random perturbation and reset step size
            print(f"[SGD  {it:7d}] perturb after stagnation")
            noise = np.random.normal(scale=0.010, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            eta = 0.5
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Generation of a random initial candidate (various families)
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random probability vector of length n.  The distribution is
    chosen from several families to diversify the search space.
    """
    choice = rng.integers(0, 6)
    if choice == 0:                     # uniform
        a = np.full(n, 1.0 / n)
    elif choice == 1:                   # half‑wave sine
        a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif choice == 2:                   # symmetric exponential (U‑shaped)
        r = rng.uniform(0.99, 0.9999)
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif choice == 3:                   # Dirichlet with random concentration
        alpha = rng.uniform(0.2, 5.0)
        a = rng.dirichlet(np.full(n, alpha))
    elif choice == 4:                   # a noisy version of a known construction if present
        old = globals().get("height_sequence_1")
        if old is not None:
            old = np.asarray(old, dtype=float)
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
            a = np.maximum(a, 0.0)
            a /= a.sum()
        else:
            a = np.full(n, 1.0 / n)
    else:                               # completely random (uniform Dirichlet)
        a = rng.dirichlet(np.ones(n))
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function under the 1000 s time budget.
    Returns a list of non‑negative floats that (approximately) minimise
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between expressiveness
    # and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Global best bookkeeping
    # ------------------------------------------------------------------
    best_global_a = None
    best_global_val = float("inf")

    # ------------------------------------------------------------------
    # Main loop: as many independent restarts as the time budget allows
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        # allocate a slice of the remaining budget for this restart
        remaining = deadline - time.time()
        if remaining < 30.0:           # not enough time for a full run
            break

        # 70 % of this slice → SA, 30 % → polishing sub‑gradient
        sa_time = remaining * 0.70
        sg_time = remaining * 0.30

        # ------------------------------------------------------------------
        # Initialise a fresh candidate
        # ------------------------------------------------------------------
        a0 = random_initial(n, rng)

        # ------------------------------------------------------------------
        # Phase 1 – Simulated‑annealing (exact objective)
        # ------------------------------------------------------------------
        a_sa, val_sa = simulated_annealing(a0, time.time() + sa_time, rng)

        # ------------------------------------------------------------------
        # Phase 2 – Sub‑gradient polishing
        # ------------------------------------------------------------------
        a_sg, val_sg = subgradient_descent(a_sa, time.time() + sg_time)

        # ------------------------------------------------------------------
        # Keep the best solution found so far
        # ------------------------------------------------------------------
        if val_sg < best_global_val:
            best_global_val = val_sg
            best_global_a = a_sg.copy()
            print(f"[restart {restart}] global best = {best_global_val:.6f}")

        restart += 1

    # If nothing was found (should never happen), fall back to uniform
    if best_global_a is None:
        best_global_a = np.full(n, 1.0 / n)
        best_global_val = objective_value(best_global_a)

    # ------------------------------------------------------------------
    # Final tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    best_global_a, best_global_val = mix_with_uniform(best_global_a)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_global_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_global_a.tolist()
