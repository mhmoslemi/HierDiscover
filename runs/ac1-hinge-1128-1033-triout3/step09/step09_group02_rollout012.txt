def _lp_refine(a: np.ndarray, start: float, deadline: float, max_iter: int = 10):
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective_value(best_a)
    for it in range(max_iter):
        current_time = time.time()
        if current_time >= deadline:
            break
        # compute M
        conv_a = _full_convolution(best_a)
        M = conv_a.max()
        # build A using scipy.sparse.diags
        import scipy.sparse as sp
        # generate list of same length diag vectors
        diags = [np.full(n, best_a[i]) for i in range(n)]
        offsets = np.arange(n)
        A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None) for _ in range(n)]
        # We'll use hiGHS' method 'highs' for speed
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP {it}] linprog failed: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            print(f"[LP {it}] sum_g too small")
            break
        # Scale g to have sum=1 (since best_a sum = 1)
        g_scaled = g / sum_g
        # Mix best_a and g_scaled using various t
        best_new_val = best_val
        best_new_a = best_a
        # try t in list
        for t in (0.1, 0.3, 0.5, 0.7, 0.9, 1.0):
            cand = (1 - t) * best_a + t * g_scaled
            cand = cand / cand.sum()  # enforce sum =1
            val = _objective_value(cand)
            if val < best_new_val:
                best_new_val, best_new_a = val, cand.copy()
        # if improvement, update
        if best_new_val < best_val - 1e-12:
            best_a, best_val = best_new_a, best_new_val
            print(f"[LP {it}] improvement: val = {best_val:.6f}")
        else:
            print(f"[LP {it}] no improvement")
            break
    return best_a, best_val

"""
Simple yet powerful 3‑stage optimiser for the step‑function bound.

Stage 1 – Fast parametric sweep (exponential, power‑law, beta, sine‑power,
piecewise‑constant) on a modest size (≈1 k) to obtain a “low‑max”
seed.

Stage 2 – Linear‑programming refinement: solve
    maximise Σ g   s.t.   f * g ≤ M  ,  g ≥ 0
where M = max(f*f).  The LP enlarges the total mass under the same
peak‑convolution bound; mixing the result back into f strictly lowers M.
Only a few LP iterations are needed (the matrix is a Toeplitz convolution
matrix, built once per iteration).

Stage 3 – Simulated‑annealing hill‑climb on the full objective for a
large vector (≈8 k).  Moves transfer a small amount of mass from indices
that heavily contribute to the current maximal convolution entry to
indices that hardly affect it; occasional uphill moves are accepted with
a temperature schedule, allowing the search to escape local minima.

The code respects the 1000 s budget, prints progress and finally returns
the best sequence found (non‑negative, Σ = 1).  The evaluation harness
will internally rescale the sum, so the returned list can be fed directly
to ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.optimize as _opt
import scipy.sparse as _sp

# ----------------------------------------------------------------------
# Convolution helpers (FFT‑based, exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (sum of a already normalised to 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact 2·n·max(conv(a,a)) for a normalised vector."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex {w ≥0, Σw = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    if abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(v.shape[0]) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families – a cheap exhaustive sweep (≈1 k points)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector found among a wide set of low‑dimensional families."""
    # start from uniform (guaranteed feasible)
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                            # 1 … n

    # 1. Exponential tails
    for r in np.linspace(0.9995, 0.99999, 12):
        _try(r ** i)                        # decreasing
        _try(r ** i + r ** (n - 1 - i))     # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 5.0, 15):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑like shapes
    for a_param in np.linspace(0.2, 5.0, 8):
        for b_param in np.linspace(0.2, 5.0, 8):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power (half‑wave sinusoid)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 10):
        _try(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.01, 0.2, 10):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.5, 10):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a3 = np.full(n, low)
            a3[:k] = high_frac
            a3[-k:] = high_frac
            _try(a3)

    # 6. Logistic‑type tails (asymmetric)
    for slope in np.linspace(0.005, 0.05, 6):
        a4 = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a4)
        _try(np.flip(a4))

    # 7. Random Dirichlet draws – quick source of diversity
    rng = np.random.default_rng(12345)
    for _ in range(30):
        _try(rng.dirichlet(np.ones(n)))

    # 8. If the runner provides a previous solution, interpolate it
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 2 – Linear‑programming refinement of a given seed vector
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, start: float, deadline: float,
               max_iter: int = 8) -> tuple[np.ndarray, float]:
    """
    Repeatedly:
      * compute M = max(f*f),
      * solve  maximise Σ g  s.t. f * g ≤ M , g ≥ 0,
      * mix the optimum back into f (line‑search on mixing weight).
    Stops when no improvement or time runs out.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective_value(best_a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current peak of the autocorrelation
        conv_f = _full_convolution(best_a)
        M = conv_f.max()

        # Build the (2n‑1) × n Toeplitz matrix for convolution f * g
        # Row k contains f_{k‑j} for column j (if 0 ≤ k‑j < n).
        # Using scipy.sparse.diags is much faster than explicit loops.
        diags = [np.full(n, best_a[i]) for i in range(n)]
        offsets = np.arange(n)                     # diagonal offsets k‑j = i
        A = _sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')

        # Linear programme: maximise sum(g)  →  minimise –sum(g)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        # HiGHS is fast and reliable
        res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                           method='highs', options={"presolve": True})

        if not res.success:
            print(f"[LP {it}] linprog failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            print(f"[LP {it}] sum(g) too small – abort")
            break

        # Scale g to have the same total mass as best_a (which is 1)
        g_scaled = g / sum_g

        # Try several mixing ratios and keep the best reduction of the objective.
        improved = False
        for t in (0.1, 0.3, 0.5, 0.7, 0.9, 1.0):
            cand = (1.0 - t) * best_a + t * g_scaled
            cand /= cand.sum()
            val = _objective_value(cand)
            if val < best_val - 1e-12:
                best_val, best_a = val, cand.copy()
                improved = True
                print(f"[LP {it}] mix t={t:.2f} → val = {best_val:.6f}")
        if not improved:
            print(f"[LP {it}] no improvement – stop")
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – Simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_anneal(a_start: np.ndarray, start: float, deadline: float,
                      best_val: float) -> tuple[np.ndarray, float]:
    """
    Transfer a (small) amount of mass from indices that heavily contribute
    to the current maximal convolution entry to indices that hardly affect it.
    Uphill moves are occasionally accepted according to a linear temperature
    schedule, allowing escape from local plateaus.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.8, 0.01   # fraction of source mass moved
    T_start, T_end = cur_val * 0.01, 1e-6    # temperature schedule
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = T_start * (1.0 - progress) + T_end * progress

        # --- pick source index (high contribution to the active entry) ---
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # --- pick destination index (preferably invisible to the active entry) ---
        if np.random.rand() < 0.5:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # --- incremental update of the convolution (O(n)) ---
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # decide whether to accept the move
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # uphill move – accept with probability exp(-Δ/T)
            if T > 0 and np.random.rand() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[SA {it:6d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # --- stagnation handling ---
        if no_improve > IMPROVE_LIMIT:
            # perturb around the current best and resume
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {it:6d}] perturb after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Final polishing – tiny uniform blend (often helps the last decimal)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective_value(a)
    uniform = np.full_like(a, 1.0 / a.shape[0])
    for alpha in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the prescribed time budget.
    Returns a list of non‑negative floats (automatically normalised to sum = 1)
    that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Stage 1 – cheap parametric sweep on a modest size (≈1200)
    # --------------------------------------------------------------
    n_small = 1200
    print("[Stage 1] Parametric sweep (n =", n_small, ")")
    best_a, best_val = _initial_candidate(n_small)
    print(f"[Stage 1] best val = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – LP refinement (a few iterations, fast)
    # --------------------------------------------------------------
    print("[Stage 2] Linear‑programming refinement")
    best_a, best_val = _lp_refine(best_a, start=start_time,
                                  deadline=deadline, max_iter=8)
    print(f"[Stage 2] best val = {best_val:.6f}")

    # --------------------------------------------------------------
    # Upsample to a larger dimension for fine‑grained tuning
    # --------------------------------------------------------------
    n_big = 8000
    old_x = np.linspace(0.0, 1.0, n_small)
    new_x = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(new_x, old_x, best_a)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    best_val = _objective_value(a_big)
    best_a = a_big
    print(f"[Upsample] n = {n_big}, val = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Simulated‑annealing hill‑climb on the full objective
    # --------------------------------------------------------------
    print("[Stage 3] Simulated‑annealing hill‑climb")
    best_a, best_val = _hillclimb_anneal(best_a,
                                         start=start_time,
                                         deadline=deadline,
                                         best_val=best_val)
    print(f"[Stage 3] best val = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final polishing – tiny uniform mixture
    # --------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # --------------------------------------------------------------
    # Report to the external evaluator (optional, for debugging)
    # --------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[report] evaluate_sequence → {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[report] evaluate_sequence raised", exc)

    return best_a.tolist()
