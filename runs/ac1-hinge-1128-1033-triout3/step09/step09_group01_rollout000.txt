def _subgradient_descent(a0, n, deadline, start_time, best_a, best_val, target_val=1.503):

    a = a0.copy()
    cur_val = _objective(a)
    iteration=0
    while True:
        now=time.time()
        if now>=deadline:
            break
        # compute convolution and find max
        b = _full_convolution(a)
        bmax = b.max()
        k = int(np.argmax(b))
        # gradient
        idx = k - np.arange(n)
        mask = (idx>=0)&(idx<n)
        g = np.zeros_like(a)
        g[mask] = 2 * a[idx[mask]]
        # step size
        eta = 0.5 / np.sqrt(iteration+1)  # may adjust
        # update
        a_new = a - eta * g
        # projection onto simplex
        a_new = _project_simplex(a_new)
        new_val = _objective(a_new)
        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Descent {iteration:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        iteration+=1
    return best_a,best_val

"""
Non‑negative step‑function optimizer (improved).

Approach
--------
1. Initialise a candidate from the previously best solution (if available),
   otherwise try a few parametric families:
   * Two‑parameter Beta family (already used in the baseline).
   * Simple exponential family a_i ∝ exp(-λ·i)   (λ swept on a log‑scale).
   * A handful of random Dirichlet draws.

2. Hill‑climbing with simulated‑annealing style moves (the same incremental
   convolution update as the baseline) consumes roughly 70 % of the time
   budget.  The move transfers a fraction of the mass from a source index
   that contributes heavily to the current maximal convolution entry to a
   sink index with little contribution.

3. The remaining time is spent on a projected sub‑gradient descent.
   The objective is 2·n·max(conv(a,a)) under the simplex constraint Σa=1.
   When the maximal convolution index is k*, the (sub‑)gradient is
   g_i = 2·a_{k*−i} (zero outside the support).  We step opposite to
   the gradient with a diminishing stepsize and project back onto the
   probability simplex.

Both stages keep track of the best sequence found so far.  After the search
the sequence is evaluated with the provided ``evaluate_sequence`` function,
stored globally for a possible warm‑start in the next run, and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Beta‑family sweep (cheap parametric search)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple exponential family sweep: a_i ∝ exp(-λ·i)
# ----------------------------------------------------------------------
def _exp_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search an exponential decay family for the best λ."""
    best_a = None
    best_val = np.inf
    i = np.arange(n, dtype=np.float64)
    # linear grid in the low‑λ region and a log‑grid for larger λ
    lambdas = np.concatenate([
        np.linspace(1e-4, 1e-2, 30),
        np.geomspace(1e-2, 0.1, 20)
    ])
    for lam in lambdas:
        a = np.exp(-lam * i)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing (identical in spirit to the baseline)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers mass from a heavy‑contributing index to a light one.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule – stretched to fill the remaining time
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # compute contribution of each index to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – usually proportional to its contribution
        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        # target index – biased towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # uniform fallback (different from source)
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        # amount of mass to move
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb restart")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (fine‑tuning after hill‑climb)
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         start_time: float,
                         best_a: np.ndarray,
                         best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on 2·n·max(conv(a,a)).
    """
    a = a0.copy()
    cur_val = _objective(a)
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # convolution and maximal index
        b = _full_convolution(a)
        bmax = b.max()
        k = int(np.argmax(b))

        # sub‑gradient of the max entry
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # diminishing stepsize
        eta = 0.5 / math.sqrt(iteration + 1)

        # gradient step and projection onto simplex
        a_new = a - eta * g
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Descent {iteration:6d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        iteration += 1

    return best_a, best_val
