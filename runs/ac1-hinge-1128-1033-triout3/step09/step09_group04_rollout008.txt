assistantfinalBelow is a self‑contained implementation of a new search strategy.  
It combines a **smooth‑max Adam optimizer** (the objective is approximated
by a soft‑max of the convolution entries) with a short **deterministic hill‑climb**
that moves mass from the worst contributing index to the best one.
The optimizer works on the logits of a soft‑max parametrisation,
so the non‑negativity and sum‑to‑one constraints are always satisfied.
Random restarts are performed after the main optimisation phase to
escape possible local minima.

```python
"""
Propose a sequence of non‑negative heights that minimises
`evaluate_sequence`.  The algorithm consists of

1.  a fast Adam optimiser on a smooth approximation of the max‑convolution
    (soft‑max temperature β is increased from 5 to 200);
2.  a deterministic hill‑climb that transfers a small fraction of mass
    from the index that contributes most to the current peak to the index
    that contributes least;
3.  many cheap random‑restart Adam runs in the remaining time budget.

All helpers are top‑level functions; no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a) evaluated at lags 0 … len(a)-1.
    Both inputs are real and non‑negative.
    """
    n = a.shape[0]
    w_len = w.shape[0]               # = 2·n‑1
    out_len = w_len + n - 1          # = 3·n‑2
    size = 1 << (out_len - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a[::-1], size)
    corr = np.fft.irfft(fw * fa_rev, size)[:out_len]
    # g_i = corr[n‑1‑i] for i=0..n‑1
    g = corr[n - 1::-1][:n]
    return g


# ----------------------------------------------------------------------
# Exact objective (the same as evaluate_sequence for a normalised vector)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a) for a vector a with sum(a)=1."""
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of max(conv(a))
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Optimise a using Adam on the soft‑max of the convolution.
    The temperature β grows linearly from 5 to 200.
    """
    # initialise logits (log‑softmax parameters)
    eps = 1e-15
    p = np.log(a0 + eps)               # shape (n,)
    m = np.zeros_like(p)
    v = np.zeros_like(p)

    beta1, beta2 = 0.9, 0.999
    it = 0
    no_improve = 0
    IMPATIENT = 30_000                 # after this many stagnant steps we jitter

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---------- a = softmax(p) ----------
        p_shift = p - np.max(p)                     # numerical stability
        a = np.exp(p_shift)
        a /= a.sum()

        # ---------- convolution ----------
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()

        # keep global best
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---------- temperature schedule ----------
        progress = (now - start_time) / (deadline - start_time)
        beta = 5.0 + 195.0 * progress   # 5 → 200

        # ---------- soft‑max weights ----------
        tmp = beta * conv
        tmp_shift = tmp - np.max(tmp)   # avoid overflow
        w = np.exp(tmp_shift)
        w_sum = w.sum()
        w_norm = w / w_sum

        # ---------- gradient w.r.t. a ----------
        grad_a = 2.0 * _cross_corr(w_norm, a)       # shape (n,)

        # ---------- gradient w.r.t. logits p ----------
        avg = np.dot(a, grad_a)
        grad_p = a * (grad_a - avg)                # (∂L/∂p)

        # ---------- Adam update ----------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p * grad_p)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = 0.12 * (1.0 - progress) + 0.001 * progress   # anneal learning rate
        p -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # ---------- jitter if stuck ----------
        if no_improve > IMPATIENT:
            p += np.random.normal(scale=0.15, size=n)
            no_improve = 0
            print(f"[Adam {it}] jitter after stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple deterministic hill‑climb
# ----------------------------------------------------------------------
def _hillclimb(
    a_start: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of the mass from the index that
    contributes most to the current maximum convolution entry to the index
    that contributes the least.  Moves are kept only if they improve the
    objective.
    """
    a = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Hill {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- locate peak and contributions -----
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)                # positions i that appear in b_k
        contrib = a[k - idx]                       # a_i * a_{k-i} contribution weight
        src = int(idx[np.argmax(contrib)])         # source index (largest contribution)

        # target = index with smallest contribution (ignore source)
        contrib_tmp = contrib.copy()
        contrib_tmp[np.argmax(contrib)] = np.inf
        tgt = int(idx[np.argmin(contrib_tmp)])

        if a[src] <= 1e-15:
            # source depleted – occasional random shake
            if no_improve > IMPATIENT:
                a += np.random.normal(scale=0.006, size=n)
                a = np.clip(a, 0, None)
                a /= a.sum()
                no_improve = 0
                print(f"[Hill {it}] jitter after stagnation")
            continue

        # annealing fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = 0.10 * (1.0 - progress) + 0.001 * progress
        delta = a[src] * frac
        if delta <= 0:
            continue

        # trial move
        a[src] -= delta
        a[tgt] += delta

        new_val = 2.0 * n * _full_convolution(a).max()
        if new_val < cur_val - 1e-15:
            # keep move
            pass
        else:
            # revert
            a[src] += delta
            a[tgt] -= delta

        it += 1

    return best_a, best_val

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations with the uniform vector."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The method consists of a long Adam optimisation,
    followed by a deterministic hill‑climb and a few cheap random restarts
    within the remaining time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.45   # safety margin

    # ------------------------------------------------------------------
    # Length of the candidate sequence.
    # Larger n gives more flexibility but costs a bit more time.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available,
    # otherwise use a smooth “arcsine” shape.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.clip(a0, 0.0, None)
        a0 /= a0.sum()
    except Exception:
        # arcsine‑like start (high at the ends, low in the middle)
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
        a0 = np.clip(a0, 0.0, None)
        a0 /= a0.sum()

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – a long Adam run (≈30 % of the total budget)
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.30 * (deadline - start_time)
    best_a, best_val = _adam_optimize(
        best_a, n, start_time, phase0_deadline, best_a, best_val, target_val=1.503
    )
    best_a, best_val = _hillclimb(
        best_a, n, phase0_deadline, deadline, best_a, best_val, target_val=1.503
    )
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Phase 1 – cheap random restarts (the remaining time)
    # ------------------------------------------------------------------
    restart_counter = 0
    rng = np.random.default_rng(seed + 123)
    while time.time() < deadline:
        remaining = deadline - time.time()
        # each restart gets at most ~0.7 % of the remaining budget,
        # but not less than 0.5 s.
        slice_budget = max(0.5, 0.007 * remaining)

        r_start = time.time()
        r_deadline = r_start + slice_budget

        # ---- random Dirichlet seed (log‑uniform concentration) ----
        log_alpha = rng.uniform(-2, 2)          # α ∈ [10⁻², 10²]
        alpha = 10.0 ** log_alpha
        a = rng.dirichlet(alpha * np.ones(n))

        # ---- short Adam optimisation (no early‑exit) ----
        a, _ = _adam_optimize(
            a, n, r_start, r_deadline, best_a, best_val, target_val=1.503
        )

        # ---- brief hill‑climb ----
        a, _ = _hillclimb(
            a, n, r_start, r_deadline, best_a, best_val, target_val=1.503
        )

        a, val = _mix_with_uniform(a, n)

        # ---- update global best ----
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Restart {restart_counter:3d}] new global best = {best_val:.6f}")

        if best_val <= 1.503:
            print("[Target reached] exiting early.")
            break

        restart_counter += 1

    # ------------------------------------------------------------------
    # Final verification (optional)
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return plain Python list (the sum is ≈1, which is acceptable)
    return best_a.tolist()
```
