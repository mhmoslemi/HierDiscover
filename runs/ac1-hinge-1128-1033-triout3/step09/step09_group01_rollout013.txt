"""
Gradient‑softmax + simulated‑annealing optimizer for the step‑function problem.

The routine combines three phases:

1️⃣ Warm‑start: use the previously best solution if it exists,
   otherwise start from a random Dirichlet distribution.

2️⃣ Gradient descent on a smooth approximation of the L∞‑norm
   (soft‑max of the self‑convolution).  The gradient is obtained by
   a single FFT‑based convolution, the step is projected onto the
   probability simplex and accepted with a simple annealing rule.

3️⃣ Final refinement with the (previously used) hill‑climbing annealer.
   This adds a few local “mass‑transfer’’ moves that can still improve
   the best solution found by the gradient phase.

The whole search runs inside the allotted time budget (default 1000 s)
and returns the best sequence encountered.  All numbers are non‑negative,
sum‐to‑one, and therefore satisfy the constraints of ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution (full linear convolution a*a)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (Σ=1, x≥0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Hill‑climbing annealer (kept as a fallback for fine tuning)
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass from an index that
    contributes heavily to the current maximal convolution entry to an
    index that contributes little.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # very slow temperature / step schedule – runs until the deadline
    max_frac_start, max_frac_end = 0.75, 0.001
    temp_start, temp_end = 0.04, 1e-9

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = temp_start * (temp_end / temp_start) ** prog

        k_max = int(np.argmax(b))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0 and np.random.rand() < 0.9:
            src_probs = contrib / total_contrib
            i_src = np.random.choice(n, p=src_probs)
        else:
            i_src = int(np.argmax(contrib))

        if a[i_src] <= eps:
            continue

        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i_src] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j_tgt = np.random.choice(n, p=tgt_weights / wsum)
        else:
            j_tgt = np.random.randint(n - 1)
            if j_tgt >= i_src:
                j_tgt += 1

        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= eps:
            continue

        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient‑softmax optimisation
# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max weights that approximate the L∞‑norm of `b`."""
    delta = b - b.max()                 # ≤ 0, avoids overflow
    exp_vals = np.exp(gamma * delta)
    w = exp_vals / exp_vals.sum()
    return w


def _gradient_step(a: np.ndarray, gamma: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the soft‑max‑approximated max‑convolution
    and return (gradient, current max_b).
    """
    n = a.shape[0]
    # 1) self‑convolution (full)
    b = _full_convolution(a)
    maxb = b.max()

    # 2) soft‑max weights on the convolution entries
    w = _softmax_weights(b, gamma)

    # 3) gradient = 2 * (a * w_rev) sliced appropriately
    w_rev = w[::-1]
    L = 1 << ((n + len(w_rev) - 1 - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, L)
    Fw = np.fft.rfft(w_rev, L)
    conv = np.fft.irfft(Fa * Fw, n=L)
    # slice conv[n-1 : 2*n-1] and reverse to obtain gradient
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]

    return grad, maxb
