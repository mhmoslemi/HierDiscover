def interleave_order(a):
    s = np.sort(a)[::-1]  # descending
    n = len(a)
    b = np.empty_like(a)
    left = 0
    right = n-1
    toggle = True
    for val in s:
        if toggle:
            b[left] = val
            left += 1
        else:
            b[right] = val
            right -= 1
        toggle = not toggle
    return b

This places largest at leftmost, second at rightmost, third at left+1, fourth at right-1, etc. This is similar to anti_symmetric but interleaved across entire array.

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  

The algorithm consists of three phases

1. **Good deterministic start** – a discretised arcsine density
   (large values at the two ends, small in the middle) is interleaved
   so that the largest coefficients are placed far apart.  The vector
   is projected onto the probability simplex (∑a_i = 1) – this makes
   the objective scale‑invariant.

2. **Smoothing‑max mirror descent** – minimise the smooth surrogate
   ``f_ε(a) = (1/ε)·log∑_k exp(ε·b_k)`` where ``b = a*a``.  The
   gradient is obtained with a single FFT; a multiplicative
   (exponentiated) mirror‑descent step keeps the coefficients non‑negative
   and on the simplex.  The smoothing parameter ε is decreased from
   0.5 down to 0.001, while the step‑size slowly decays.

3. **Hard‑max and greedy refinement** – after the smooth stage we
   target the *actual* maximum of the convolution.  A sub‑gradient of
   the true max is used for a few more exponentiated‑gradient steps.
   Finally a greedy “mass transfer” routine repeatedly moves a small
   amount of probability from the pair that contributes most to the
   current peak to a pair that contributes least, accepting only
   improving moves.  All heavy work uses FFT‑based convolutions;
   the true objective is evaluated only occasionally.

The whole search respects a 1000 s wall‑clock budget and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution / optimisation utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k p_k·b_k   wrt a,
    where   b = a*a   (full linear convolution)   and   p ≥ 0, Σp = 1.
    The result is   g_i = Σ_k p_k·a_{k‑i}   (out‑of‑range indices are zero).
    Computed with one FFT.
    """
    n = a.shape[0]
    # reverse the weight vector – convolution with reversed weights implements the required sum
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # the slice that corresponds to indices n‑1 … 2n‑2
    seg = conv[n - 1: 2 * n - 1]
    # reverse again to obtain the gradient
    g = seg[::-1]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σx = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _interleave_high_low(a: np.ndarray) -> np.ndarray:
    """
    Order the coefficients by alternately placing the largest remaining
    value at the leftmost free position and the smallest remaining value
    at the rightmost free position.
    This produces a “high‑low‑high‑low …” layout that keeps large entries
    far apart.
    """
    s = np.sort(a)[::-1]           # descending
    n = len(a)
    b = np.empty_like(a)
    lo, hi = 0, n - 1
    toggle = True
    for val in s:
        if toggle:
            b[lo] = val
            lo += 1
        else:
            b[hi] = val
            hi -= 1
        toggle = not toggle
    return b


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective value 2·n·max(conv) (sum is 1)."""
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _arcsine_initial(n: int) -> np.ndarray:
    """
    Discrete approximation of the arcsine density on {0,…,n‑1}:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    The vector is normalised to have unit L1‑norm.
    """
    i = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n - i))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Phase 1 – smoothed‑max mirror descent
# ----------------------------------------------------------------------
def _smoothed_descent(a_start: np.ndarray, deadline: float):
    """
    Minimise a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector encountered and its true objective.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = _objective_from_conv(_full_convolution(a), n)

    print(f"[smoothed] start objective = {best_val:.6f}")

    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    total_iters = 5000
    it_per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        eta = 0.7                      # initial step size for this ε‑stage
        for it in range(it_per_eps):
            if time.time() >= deadline - 5.0:
                break

            conv = _full_convolution(a)
            max_c = conv.max()

            # soft‑max weights (stable)
            shift = max_c
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()          # Σp = 1, p ≥ 0

            # gradient of the smooth surrogate (factor 2 from d(b_k)/da_i)
            grad = 2.0 * _gradient_weighted_peaks(a, p)

            # exponentiated‑gradient step (mirror descent on the simplex)
            delta = -eta * grad
            # keep exponentials numerically safe
            delta = np.clip(delta, -20.0, 0.0)
            a = a * np.exp(delta)
            a = _project_simplex(a)

            # occasional true‑objective check
            if it % 30 == 0:
                cur_val = _objective_from_conv(_full_convolution(a), n)
                if cur_val < best_val - 1e-12:
                    best_val, best_a = cur_val, a.copy()
                    print(f"[smoothed eps {eps:.3f} it {it}] new best = {best_val:.6f}")

                # a cheap anti‑symmetric re‑ordering every 300 steps
                if it % 300 == 0:
                    a_ord = _interleave_high_low(a)
                    val_ord = _objective_from_conv(_full_convolution(a_ord), n)
                    if val_ord < best_val:
                        best_val, best_a = val_ord, a_ord.copy()
                        a = a_ord.copy()
                        print(f"[reorder eps {eps:.3f} it {it}] improved to {best_val:.6f}")

            # slowly decay step size
            eta *= 0.9995

        # final re‑ordering for the ε‑stage (cannot increase the max)
        a = _interleave_high_low(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – hard‑max sub‑gradient mirror descent
# ----------------------------------------------------------------------
def _hard_max_descent(a_start: np.ndarray, deadline: float):
    """
    Mirror‑descent on the *true* max‑convolution (no smoothing).
    Uses the sub‑gradient coming from the current maximal index.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective_from_conv(_full_convolution(a), n)

    print(f"[hard‑max] start objective = {best_val:.6f}")

    eta = 0.4
    it = 0
    while time.time() < deadline - 5.0 and it < 3000:
        it += 1
        conv = _full_convolution(a)
        k = int(conv.argmax())                          # index of the current peak
        # sub‑gradient for this k
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        delta = -eta * grad
        delta = np.clip(delta, -20.0, 0.0)
        a = a * np.exp(delta)
        a = _project_simplex(a)

        # periodic evaluation
        if it % 20 == 0:
            cur_val = _objective_from_conv(_full_convolution(a), n)
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[hard‑max it {it}] new best = {best_val:.6f}")
            # optional slight re‑ordering to keep massive entries apart
            if it % 200 == 0:
                a_ord = _interleave_high_low(a)
                val_ord = _objective_from_conv(_full_convolution(a_ord), n)
                if val_ord < best_val:
                    best_val, best_a = val_ord, a_ord.copy()
                    a = a_ord.copy()
                    print(f"[reorder hard‑max it {it}] improved to {best_val:.6f}")

        # decay step size a little
        eta *= 0.9997

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – greedy max‑reduction moves
# ----------------------------------------------------------------------
def _max_reduction_greedy(a_start: np.ndarray, deadline: float):
    """
    Repeatedly move a small amount of probability from the pair that
    contributes most to the current peak to a pair that contributes the
    least.  Accept only improving moves.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective_from_conv(_full_convolution(a), n)

    print(f"[greedy] start objective = {best_val:.6f}")

    rng = np.random.default_rng()
    attempts = 0
    while time.time() < deadline - 5.0 and attempts < 5000:
        attempts += 1

        conv = _full_convolution(a)
        k = int(conv.argmax())
        # product contributions for this k
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        prod = np.zeros_like(a)
        prod[mask] = a[mask] * a[idx[mask]]

        # choose source (largest product) and destination (smallest product)
        src = int(prod.argmax())
        dst_candidates = np.where(prod <= prod.min())[0]
        if len(dst_candidates) == 0:
            continue
        dst = int(rng.choice(dst_candidates))

        if src == dst or a[src] <= 1e-15:
            continue

        # transfer a modest fraction of the source mass
        delta = a[src] * 0.07 * rng.random()   # ≤7 % of src
        delta = min(delta, a[src])            # safety

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        # stay on the simplex exactly (sum unchanged, non‑negativity already enforced)
        new_a = np.maximum(new_a, 0.0)

        new_val = _objective_from_conv(_full_convolution(new_a), n)
        if new_val < best_val - 1e-12:
            best_val, best_a = new_val, new_a.copy()
            a = new_a
            # occasional log
            if attempts % 200 == 0:
                print(f"[greedy] iter {attempts} improved to {best_val:.6f}")
        else:
            # rarely accept worsening moves to escape plateaus (simulated‑annealing style)
            if rng.random() < 0.0005:
                a = new_a

    return best_a, best_val


# ----------------------------------------------------------------------
# Light hill‑climbing refinement (same as in the previous baseline)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float):
    """
    Random swaps / tiny transfers, kept only if they strictly improve the true objective.
    """
    best_a = a.copy()
    best_val = cur_val
    n = a.shape[0]
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        if rng.random() < 0.25:                 # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                   # transfer a small amount
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.04 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑normalise onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        val = _objective_from_conv(_full_convolution(new_a), n)
        if val < best_val - 1e-12:
            best_val, best_a = val, new_a
            print(f"[hill‑climb] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimizes
    `evaluate_sequence` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # problem size – a few‑ten‑thousands give a good trade‑off
    n = 10000

    # ------------------------------------------------------------------
    # 1) Initialise with a discretised arcsine density and spread it out
    # ------------------------------------------------------------------
    a0 = _arcsine_initial(n)
    a0 = _interleave_high_low(a0)
    a0 = _project_simplex(a0)               # ensure exact sum‑1

    # ------------------------------------------------------------------
    # 2) Smoothed‑max descent
    # ------------------------------------------------------------------
    a_best, val_best = _smoothed_descent(a0, deadline)
    print(f"[global] after smoothed descent best = {val_best:.6f}")

    # ------------------------------------------------------------------
    # 3) Hard‑max mirror descent
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_hard, val_hard = _hard_max_descent(a_best, deadline)
        if val_hard < val_best:
            a_best, val_best = a_hard, val_hard
            print(f"[global] after hard‑max best = {val_best:.6f}")

    # ------------------------------------------------------------------
    # 4) Greedy max‑reduction moves
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_greedy, val_greedy = _max_reduction_greedy(a_best, deadline)
        if val_greedy < val_best:
            a_best, val_best = a_greedy, val_greedy
            print(f"[global] after greedy best = {val_best:.6f}")

    # ------------------------------------------------------------------
    # 5) Final local hill‑climbing polish
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_final, val_final = _hill_climb_refine(a_best, val_best, deadline)
        if val_final < val_best:
            a_best, val_best = a_final, val_final
            print(f"[global] after hill‑climb best = {val_best:.6f}")

    # ------------------------------------------------------------------
    # 6) Final verification with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_best.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    return a_best.tolist()
