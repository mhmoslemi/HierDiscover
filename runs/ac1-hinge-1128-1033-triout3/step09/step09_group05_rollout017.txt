    a = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 100_000
    max_K = 8
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[ExpGradMulti {it}] new best = {best_val:.6f}")
            if cur_val <= target_val:
                break
        else:
            no_improve += 1

        # compute top K indices
        K = max_K
        # if few indices have same max, treat
        idxs = np.argpartition(-b, K-1)[:K]
        # compute gradient vector g (nonnegative)
        g = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # indices where rev_i = a[k - i]
            # we can compute slice:
            i_range = np.arange(lo, hi+1)
            rev_slice = a[k - i_range]
            g[i_range] += rev_slice

        # optionally weight each rev by b[k] / max_b
        weights = (b[idxs] / max_b)[:, np.newaxis]
        # could multiply each rev by weight but we summed equally: we can compute weighted sum by scaling contributions:
        # Do as loop with weight:
        # We'll recompute with weighting:
        g = np.zeros_like(a)
        for idx, k in enumerate(idxs):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i_range = np.arange(lo, hi+1)
            rev_slice = a[k - i_range]
            weight = b[k] / max_b
            g[i_range] += weight * rev_slice

        # update with step size schedule
        progress = (time.time() - start) / (deadline - start)
        lr = (1 - progress) * 0.5 + progress * 0.001

        a = a * np.exp(-lr * g)
        # renormalize onto simplex
        total = a.sum()
        if total <= 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / total

        if no_improve > IMPATIENT:

            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            no_improve = 0
            print(f"[ExpGradMulti {it}] jitter after stagnation")
        it += 1

"""
step‑function optimizer – 1000 s run‑time budget

The algorithm consists of four stages

    0.  initialise from the previous solution (if any) or from a few cheap
        deterministic families.
    1.  a very wide random sweep over many parametric families
        (exponential, reversed exponential, mixtures, power‑law × exponential,
        beta‑distribution, sigmoid and a two‑phase exponential). The best
        candidate found is kept.
    2.  a long exponentiated‑gradient descent.  The sub‑gradient of the
        objective “max(conv(a,a))’’ is the reversed slice of ``a`` that
        corresponds to the largest convolution entry.  Instead of using only
        the single peak we take the top‑K peaks (K≈10) and move mass away
        from the positions that appear in many of those peaks – this gives a
        smoother descent and lets us escape shallow local minima.
        A simple momentum term and a linearly decaying learning rate are used.
    3.  a tiny uniform mixing step (as in the reference implementation) to
        smooth out possible spikes.

All vectors are kept on the probability simplex (non‑negative, sum = 1)
by a multiplicative update followed by normalisation.
The routine respects the 1000 s limit by checking the deadline frequently.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via FFT (single‑precision is enough for the optimisation)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a (size 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (sum is assumed to be 1, i.e. we work on the simplex)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a normalized vector."""
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    # reference: https://arxiv.org/abs/1101.6081 (Algorithm 1)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i   (0<r<1)."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**(n‑1‑i)."""
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    """Mixture w·r1**i + (1‑w)·r2**(n‑1‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    """a_i ∝ (i+1)**p · r**i."""
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_beta(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n          # centre points
    with np.errstate(divide='ignore', invalid='ignore'):
        a_vec = np.power(x, a - 1) * np.power(1 - x, b - 1)
    a_vec[np.isnan(a_vec)] = 0.0
    s = a_vec.sum()
    return a_vec / s if s != 0 else np.full(n, 1.0 / n)


def _gen_sigmoid(n: int, slope: float, centre: float) -> np.ndarray:
    """Monotonically decreasing sigmoid: 1/(1+exp(slope·(i-centre)))"""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(slope * (i - centre)))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_two_phase_exp(n: int, m: int, r1: float, r2: float) -> np.ndarray:
    """Two‑phase exponential: r1**i for i<m, r2**(i‑m) thereafter."""
    a = np.empty(n, dtype=np.float64)
    a[:m] = np.power(r1, np.arange(m, dtype=np.float64))
    a[m:] = np.power(r2, np.arange(n - m, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent with a top‑K sub‑gradient
# ----------------------------------------------------------------------
def _expgrad_topk(a0: np.ndarray,
                  n: int,
                  start: float,
                  deadline: float,
                  best_a: np.ndarray,
                  best_val: float,
                  target_val: float = 1.503,
                  top_k: int = 12) -> tuple[np.ndarray, float]:
    """Multiplicative updates using the average of the ``top_k`` rev‑vectors."""
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 250_000               # after this many non‑improving steps we jitter
    rng = np.random.default_rng(42)

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if cur_val <= target_val:
                break
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # sub‑gradient: weighted sum of the ``top_k`` reversed slices
        # ------------------------------------------------------------------
        # Get indices of the K largest convolution entries
        k_idxs = np.argpartition(-b, top_k - 1)[:top_k]
        g = np.zeros_like(cur)

        # accumulate the rev‑vectors (optionally weight by the corresponding
        # convolution value – this gives more importance to the true maximum)
        weights = b[k_idxs] / max_b          # in [0,1]
        for w, k in zip(weights, k_idxs):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            rev_slice = cur[k - idx]          # a_{k‑i}
            g[idx] += w * rev_slice

        # ------------------------------------------------------------------
        # learning‑rate schedule (linear decay)
        # ------------------------------------------------------------------
        progress = (now - start) / (deadline - start)
        lr = (1.0 - progress) * 0.30 + progress * 0.001   # start big, end tiny

        # multiplicative (exponentiated) update, stay on the simplex
        cur = cur * np.exp(-lr * g)
        cur = _project_simplex(cur)

        # stochastic jitter when we appear stuck
        if no_improve > IMPATIENT:
            sigma = 0.004 * (1.0 - progress) + 0.0005 * progress
            cur = np.maximum(cur + rng.normal(scale=sigma, size=n), 0.0)
            cur = _project_simplex(cur)
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation (σ={sigma:.6f})")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (tiny local moves)
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               n: int,
               start: float,
               deadline: float,
               best_a: np.ndarray,
               best_val: float,
               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    rng = np.random.default_rng(999)

    while time.time() < deadline:
        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if cur_val <= target_val:
                break
        else:
            no_improve += 1

        # pick a random index with sizeable mass and a random destination
        i = rng.choice(n, p=cur)                     # heavy index
        j = rng.integers(0, n)                       # any other index
        if i == j or cur[i] < 1e-12:
            continue

        max_frac = 0.4 * (1.0 - (time.time() - start) / (deadline - start)) + 0.02
        delta = cur[i] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)                 # keep exact sum = 1

        cand_val = 2.0 * n * _full_convolution(cand).max()
        if cand_val < cur_val - 1e-15:
            cur = cand
            cur_val = cand_val
            # keep the funnel of best values also for this local search
            if cand_val < best_val - 1e-15:
                best_val, best_a = cand_val, cand.copy()
                print(f"[HillClimb {it:7d}] improved via local move = {best_val:.6f}")

        # occasional jitter to escape plateaus
        if no_improve > IMPATIENT:
            sigma = 0.003
            cur = np.maximum(cur + rng.normal(scale=sigma, size=n), 0.0)
            cur = _project_simplex(cur)
            no_improve = 0
            print(f"[HillClimb {it:7d}] jitter after stagnation (σ={sigma:.6f})")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing (helps to smooth tiny spikes)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under a 1000 s budget.
    Returns a list of floats (Σa = 1) minimising ``evaluate_sequence``.
    The routine follows the four‑stage plan described in the module docstring.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – 8000 gives a good quality/ speed trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # 0. initialise from the previous run if possible
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 = _project_simplex(a0)
    except Exception:
        a0 = np.full(n, 1.0 / n)                # uniform fallback

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # 1. random exploration of rich parametric families (≈20 % of time)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 777)
    phase1_deadline = start_time + 0.20 * (deadline - start_time)

    while time.time() < phase1_deadline:
        family = rng.integers(0, 7)   # seven families
        if family == 0:                       # pure exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_exp(n, r)
        elif family == 1:                     # reversed exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_rev_exp(n, r)
        elif family == 2:                     # mixture of two exponentials
            r1 = rng.uniform(0.90, 0.99999)
            r2 = rng.uniform(0.90, 0.99999)
            w = rng.random()
            cand = _gen_mix_exp(n, r1, r2, w)
        elif family == 3:                     # power‑law × exponential
            p = rng.uniform(-6.0, 6.0)
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_power_exp(n, p, r)
        elif family == 4:                     # beta‑distribution shape
            a_param = rng.exponential(scale=2.0) + 0.1   # >0
            b_param = rng.exponential(scale=2.0) + 0.1
            cand = _gen_beta(n, a_param, b_param)
        elif family == 5:                     # sigmoid (monotone)
            slope = rng.uniform(0.01, 0.5) * (1 if rng.random() < 0.5 else -1)
            centre = rng.uniform(0, n)
            cand = _gen_sigmoid(n, slope, centre)
        else:                                 # two‑phase exponential
            m = rng.integers(int(0.2 * n), int(0.8 * n))
            r1 = rng.uniform(0.90, 0.99999)
            r2 = rng.uniform(0.90, 0.99999)
            cand = _gen_two_phase_exp(n, m, r1, r2)

        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Random] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. exponentiated‑gradient descent (≈55 % of time)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.75 * (deadline - start_time)
    best_a, best_val = _expgrad_topk(
        best_a, n,
        start=start_time,
        deadline=phase2_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
        top_k=10
    )

    # ------------------------------------------------------------------
    # 3. deterministic hill‑climbing (≈20 % of time)
    # ------------------------------------------------------------------
    phase3_deadline = deadline
    best_a, best_val = _hillclimb(
        best_a, n,
        start=phase2_deadline,
        deadline=phase3_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET
    )

    # ------------------------------------------------------------------
    # 4. tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # final verification using the provided security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:                                   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
