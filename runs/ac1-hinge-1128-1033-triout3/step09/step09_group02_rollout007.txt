def upsample(a_small, n_large):
    m = len(a_small)
    x_small = np.linspace(0, 1, m)
    x_large = np.linspace(0, 1, n_large)
    a_large = np.interp(x_large, x_small, a_small, left=0, right=0)
    a_large = np.maximum(a_large, 0)
    a_large /= a_large.sum()
    return a_large
```

def _initial_candidate(n, rng):
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def test_and_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective_value(vec)
        if val < best_val:
            best_val = val
            best_a = vec.copy()
    i = np.arange(n, dtype=np.float64)
    i1 = i + 1.0

    # families...
    for r in np.linspace(0.995, 0.9999, 10):
        test_and_update(r ** i) # decreasing
        test_and_update(r ** i + r ** (n - 1 - i)) # symmetric

    for p in np.linspace(0.1, 4.0, 12):
        test_and_update(i1 ** (-p))
        test_and_update(i1 ** (-p) + (n - i) ** (-p))

    # beta family
    for a_param in np.linspace(0.2, 5.0, 6):
        for b_param in np.linspace(0.2, 5.0, 6):
            vec = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            test_and_update(vec)

    # sine power
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 8):
        test_and_update(base_sin ** p)

    # logistic
    for slope in np.linspace(0.005, 0.04, 5):
        vec = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        test_and_update(vec)
        test_and_update(np.flip(vec))

    # Gaussian
    for sigma in np.linspace(n/10, n/2, 5):
        vec = np.exp(-0.5 * ((i - (n-1)/2) / sigma) ** 2)
        test_and_update(vec)

    # Random Dirichlet draws

"""
Advanced optimizer for the step‑function evaluation task.

The algorithm works in three scales:
1. Search a good shape on a modest size (n≈2000) using a diverse
   parametric sweep, Adam on a smooth log‑sum‑exp surrogate of
   max‑convolution, a short linear‑program “oracle” step (Frank‑Wolfe
   style) and a targeted hill‑climber.
2. Upsample the best small‑scale shape to the required length (n=9000)
   by linear interpolation and renormalisation.
3. Polish the large‑scale vector with a final hill‑climber and a tiny
   uniform‑mix tweak.

All phases respect the time budget (default 1000 s) and keep the
vector non‑negative with Σa=1, which makes the objective
`2·n·max(conv(a,a))` scale‑invariant.
"""

import math
import numpy as np
from scipy import sparse
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum a is assumed to be 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Projection onto the probability simplex {w ≥ 0, Σ w = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric sweep – builds a solid starting point
# ----------------------------------------------------------------------
def _initial_candidate(n: int, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Return the best vector found among a rich set of parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective_value(vec)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                           # 1 … n

    # 1. Exponential / symmetric tails
    for r in np.linspace(0.995, 0.9999, 12):
        _try(r ** i)                        # monotone decreasing
        _try(r ** i + r ** (n - 1 - i))     # U‑shaped

    # 2. Power‑law tails
    for p in np.linspace(0.2, 5.0, 14):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.2, 5.0, 6):
        for b_param in np.linspace(0.2, 5.0, 6):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Half‑wave sinusoid raised to a power
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 9):
        _try(base_sin ** p)

    # 5. Logistic‑type tails
    for slope in np.linspace(0.005, 0.04, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 6. Gaussian bells (symmetric)
    for sigma in np.linspace(n / 10.0, n / 2.0, 5):
        a = np.exp(-0.5 * ((i - (n - 1) / 2.0) / sigma) ** 2)
        _try(a)

    # 7. Random Dirichlet draws
    for _ in range(40):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 8. Interpolation of a previous solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])

        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:

# ----------------------------------------------------------------------
# Phase A – Adam on a smooth log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp approximation of the peak convolution.
    """
    # Initialise logits (soft‑max parametrisation)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.24, 0.006
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80_000
    it = 0

    while time.time() < deadline:
        # Map logits to simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # Schedules
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # Smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted - np.max(shifted))   # avoid overflow
        p = weights / weights.sum()                  # soft‑max over positions

        # Gradient of the surrogate w.r.t. a (see derivation in analysis)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)               # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]            # shape (n,)

        # Back‑prop through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Occasional jitter to escape plateaux
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.01, size=n)

        # Restart if stagnating
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.3
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy mass‑transfer hill‑climber using the exact max‑convolution."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.55, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (draw proportionally to contribution)
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # target index – favour locations that hardly affect the current peak
        if np.random.rand() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of the full convolution (linear‑time)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.018, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a

# Phase C – Small final polishing (mix with uniform)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a tiny share of the uniform distribution."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):

        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program oracle (Frank‑Wolfe style step)
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2n‑1) × n matrix C where C @ x = conv(a, x).
    Uses a Toeplitz construction; result is sparse.
    """
    n = a.shape[0]
    col = np.concatenate([a, np.zeros(n - 1)])
    row = np.concatenate([a[:1], np.zeros(n - 1)])
    C_dense = toeplitz(col, row)[: 2 * n - 1, :]
    return sparse.csr_matrix(C_dense)


def _lp_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ x_i   subject to   conv(a, x) ≤ M,   x ≥ 0.
    Returns the optimal x (may be scaled arbitrarily) or None on failure.
    """
    A = _build_conv_matrix(a)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(len(a), dtype=np.float64)   # maximise Σx  ⇔ minimise -Σx
    bounds = [(0, None)] * len(a)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------
def _upsample(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from a small vector to a larger one, re‑normalised."""
    n_small = len(a_small)
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(x_large, x_small, a_small, left=0.0, right=0.0)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    return a_large

        best_val=_objective_value(a_large),
        target_val=1.503,
    )
    best_val_large = _objective_value(a_large)
    print(f"[Phase D] best after large hillclimb = {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    a_large, best_val_large = _mix_with_uniform(a_large, n_large)
    print(f"[Final] best after mixing = {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_large.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_large.tolist()
