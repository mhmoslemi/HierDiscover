"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm Overview
-------------------
1. **Multiple restarts** – start from a few simple seeds (uniform,
   sinusoid, exponential).  Each vector is normalised to sum = 1
   (the objective is scale‐invariant).

2. **Smoothed‑max descent** – minimise a log‑sum‑exp surrogate of the
   maximal convolution peak.  At each iteration we compute the full
   self‑convolution via a single FFT, obtain soft‑max weights,
   compute the gradient by one further FFT and perform a multiplicative
   mirror‑descent update.  The smoothing parameter ε is decreased
   gradually (0.5 → … → 0.001) and the step‑size slowly decays.

3. **Exact‑max descent** – after the smoothing schedule we switch to a
   true sub‑gradient (weights concentrate on the current maximal peak)
   and continue with a finer learning‑rate.  This tightens the bound
   around the true objective.

4. **Structure‑preserving re‑ordering** – after every few hundred
   iterations the current vector is reordered so that the largest
   entries sit at both ends of the array (a symmetric “large‑at‑ends”
   permutation).  The rearrangement inequality guarantees that this
   never increases the maximal convolution value; we keep the better
   ordering (the previous anti‑symmetric reorder is also tried).

5. **Hill‑climbing refinement** – a lightweight local search that
   performs random swaps or tiny mass transfers, accepting only
   improvements.  The objective is evaluated analytically
   (`2·n·max(a*a)`) which is much cheaper than calling
   `evaluate_sequence`.

All heavy work (FFT‑based steps) is performed without invoking the
expensive `evaluate_sequence`; the latter is called only for the final
report and occasional sanity checks.  The routine respects the 1000 s
budget and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using a single FFT.
    Returns a non‑negative array (tiny negatives are clipped to zero).
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·(a*a)[k] w.r.t. a.
    The weight vector p is of length 2·n‑1 and satisfies Σp = 1, p ≥ 0.
    Resulting gradient has shape (n,).
    """
    n = a.shape[0]
    # reverse the weight vector (convolution with reversed kernel)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # relevant slice corresponds to indices n‑1 … 2n‑2 (inclusive)
    seg = conv[n - 1 : 2 * n - 1]
    # reverse back to obtain the gradient
    g = seg[::-1]
    return g


def _normalize_simplex(v: np.ndarray) -> np.ndarray:
    """
    Simple renormalisation onto the probability simplex {x ≥ 0, Σx = 1}.
    Because the multiplicative update never creates negative entries,
    this is sufficient and far cheaper than a full Euclidean projection.
    """
    s = v.sum()
    if s <= 0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def _symmetric_end_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values at both ends of the array and monotonically
    decrease towards the centre (large‑at‑ends permutation).
    """
    s = np.sort(a)[::-1]                   # descending
    n = a.shape[0]
    b = np.empty_like(a)
    left, right = 0, n - 1
    idx = 0
    while left <= right:
        b[left] = s[idx]
        idx += 1
        left += 1
        if left <= right:
            b[right] = s[idx]
            idx += 1
            right -= 1
    return b


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Legacy ordering (largest ↔ smallest). Kept for occasional checks."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact value of `evaluate_sequence` for vectors that already satisfy
    the internal security checks (non‑negative, sum > 0.01, values ≤ 1000).
    Since we keep sum(a) = 1, the objective reduces to 2·n·max(a*a).
    """
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed‑max descent (multiplicative mirror‑descent)
# ----------------------------------------------------------------------
def _smoothed_descent(a_start: np.ndarray, n: int, deadline: float):
    """
    Minimise a log‑sum‑exp surrogate of the maximal convolution.
    Returns the best vector found during this phase and its objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a, n)
    print(f"[smoothed start] val = {best_val:.6f}")

    # smoothing schedule – from very smooth to almost exact
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    # rough total budget for this stage (iterations per ε)
    it_per_eps = 1200

    for eps in eps_schedule:
        eta = 0.7                     # initial step size for this ε
        it = 0
        while it < it_per_eps and time.time() < deadline - 4.0:
            # ----- convolution and soft‑max weights -----
            conv = _full_convolution(a)                     # length 2n‑1
            max_conv = conv.max()
            shift = max_conv                                 # for numerical stability
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()                   # Σp = 1, p ≥ 0

            # ----- gradient of the smooth surrogate -----
            grad = 2.0 * _gradient_weighted_peaks(a, p)    # shape (n,)

            # ----- multiplicative mirror‑descent update -----
            delta = -eta * grad
            delta = np.clip(delta, -20.0, 0.0)               # avoid overflow
            a = a * np.exp(delta)
            a = _normalize_simplex(a)

            # ----- occasional objective tracking -----
            if (it % 30) == 0:
                cur_val = _objective(a, n)
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.3f} it {it:4d}] new best = {best_val:.6f}")

                # try the two re‑orderings every ~200 steps
                if (it % 200) == 0:
                    a_sym = _symmetric_end_reorder(a)
                    val_sym = _objective(a_sym, n)
                    if val_sym < best_val:
                        best_val = val_sym
                        best_a = a_sym.copy()
                        a = a_sym.copy()
                        print(f"[reorder sym eps {eps:.3f} it {it}] improved to {best_val:.6f}")

                    a_anti = _anti_symmetric_reorder(a)
                    val_anti = _objective(a_anti, n)
                    if val_anti < best_val:
                        best_val = val_anti
                        best_a = a_anti.copy()
                        a = a_anti.copy()
                        print(f"[reorder anti eps {eps:.3f} it {it}] improved to {best_val:.6f}")

            # slowly decay the learning rate – helps stability for tiny ε
            eta *= 0.9995
            it += 1

        # final re‑order for this ε (cannot increase the max)
        a = _symmetric_end_reorder(a)

    return a, best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max (sub‑gradient) descent – focuses on the true peak
# ----------------------------------------------------------------------
def _exact_max_descent(a_start: np.ndarray, n: int, deadline: float):
    """
    Continue optimisation using the true sub‑gradient (weights concentrate
    on the current maximal convolution entries).  Returns the best vector
    discovered and its objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = _objective(a, n)
    print(f"[exact start] val = {best_val:.6f}")

    max_iters = 2500
    eta = 0.25                # smaller step size for fine tuning
    it = 0
    while it < max_iters and time.time() < deadline - 3.0:
        conv = _full_convolution(a)
        max_conv = conv.max()
        # indices attaining the current maximum (within a tiny tolerance)
        max_inds = np.where(conv >= max_conv - 1e-13)[0]
        # uniform weight on the active indices
        p = np.zeros_like(conv)
        p[max_inds] = 1.0 / max_inds.size

        # gradient w.r.t the exact‑max surrogate
        grad = 2.0 * _gradient_weighted_peaks(a, p)

        # multiplicative update
        delta = -eta * grad
        delta = np.clip(delta, -20.0, 0.0)
        a = a * np.exp(delta)
        a = _normalize_simplex(a)

        # occasional tracking
        if (it % 10) == 0:
            cur_val = _objective(a, n)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[exact it {it:4d}] new best = {best_val:.6f}")

        # occasional structured re‑ordering
        if (it % 200) == 0:
            a_sym = _symmetric_end_reorder(a)
            val_sym = _objective(a_sym, n)
            if val_sym < best_val:
                best_val = val_sym
                best_a = a_sym.copy()
                a = a_sym.copy()
                print(f"[reorder sym exact it {it}] improved to {best_val:.6f}")

        # decay η very slowly
        eta *= 0.9998
        it += 1

    # final tidy‑up re‑order
    a = _symmetric_end_reorder(a)
    return a, best_a, best_val


# ----------------------------------------------------------------------
# Light hill‑climbing refinement (mass swaps / transfers)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float, n: int):
    """
    Simple local search: random swaps or tiny mass transfers, accepted
    only if they strictly improve the true objective.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        if rng.random() < 0.25:                 # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                   # transfer a small amount
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] <= 1e-15:
                continue
            delta = best_a[i] * 0.05 * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # renormalise (the sum is conserved for swaps; for transfers we
        # keep the sum unchanged, but floating point errors may appear)
        new_a = np.maximum(new_a, 0.0)
        new_a = _normalize_simplex(new_a)

        val = _objective(new_a, n)
        if val < best_val - 1e-12:
            best_val = val
            best_a = new_a
            print(f"[hill] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` under the given time budget.  The routine combines
    smoothed‑max descent, exact‑max sub‑gradient steps, structure‑preserving
    re‑ordering and a final hill‑climbing refinement.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin (the harness terminates after the budget)
    deadline = start_time + budget_s - 4.0

    # problem size – a few thousand coefficients give a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # Load a previous construction if available (optional)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 = _normalize_simplex(a0)
    except Exception:
        a0 = None

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # A few restarts with diverse initialisations
    # ------------------------------------------------------------------
    init_strategies = ["uniform", "sin", "rand"]
    restart = 0
    max_restarts = 6
    while time.time() < deadline - 150 and restart < max_restarts:
        restart += 1

        # ----------------------------------------------------------------
        # Choose initial vector
        # ----------------------------------------------------------------
        if a0 is not None and restart == 1:
            a = a0.copy()
            init_name = "from_previous"
        else:
            strat = rng.choice(init_strategies)
            if strat == "uniform":
                a = np.full(n, 1.0 / n)
                init_name = "uniform"
            elif strat == "sin":
                idx = np.arange(n) + 0.5
                a = np.sin(np.pi * idx / n)
                a = np.maximum(a, 0.0)
                a = _normalize_simplex(a)
                init_name = "sinusoid"
            else:   # exponential random
                a = rng.exponential(scale=1.0, size=n)
                a = _normalize_simplex(a)
                init_name = "random"

        print(f"\n[restart {restart}] init = {init_name}")

        # ----------------------------------------------------------------
        # Smoothed‑max descent phase
        # ----------------------------------------------------------------
        a, local_best_a, local_best_val = _smoothed_descent(a, n, deadline)

        if local_best_val < best_val:
            best_val = local_best_val
            best_a = local_best_a.copy()
            print(f"[global] new best after smoothed = {best_val:.6f}")

        # ----------------------------------------------------------------
        # Exact‑max (sub‑gradient) phase
        # ----------------------------------------------------------------
        a, local_best_a2, local_best_val2 = _exact_max_descent(a, n, deadline)

        if local_best_val2 < best_val:
            best_val = local_best_val2
            best_a = local_best_a2.copy()
            print(f"[global] new best after exact = {best_val:.6f}")

        # ----------------------------------------------------------------
        # Light hill‑climb if time permits
        # ----------------------------------------------------------------
        if time.time() < deadline - 30:
            a_hc, val_hc = _hill_climb_refine(a, _objective(a, n),
                                             deadline, n)
            if val_hc < best_val:
                best_val = val_hc
                best_a = a_hc.copy()
                print(f"[global] post‑hill improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polish on the overall best vector
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5:
        best_a, best_val = _hill_climb_refine(best_a,
                                               _objective(best_a, n),
                                               deadline, n)

    # ------------------------------------------------------------------
    # Final evaluation via the provided black‑box (for reporting purposes)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
