"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm (high‑level)
----------------------
* Work on a probability vector (∑a = 1).  The objective is scale‑invariant.
* Initialise several restarts (uniform, Dirichlet, exponential) with length
  `n` (default 6000 – a good trade‑off between quality and speed).
* For each restart run an *inner optimisation* that minimises a smooth
  surrogate of the max‑convolution:
        F_ε(a) = (1/ε)·log(∑_k exp(ε·(a*a)[k]))               (log‑sum‑exp)
  The gradient of F_ε is obtained with a single FFT; a multiplicative
  mirror‑descent step keeps the vector non‑negative.
* A momentum term (`m`) is added to the gradient to accelerate convergence.
* The smoothing parameter ε is decreased in many stages (1.0 → 0.001); at
  the end the surrogate coincides with the true max.
* After each ε‑stage we apply the *anti‑symmetric re‑ordering* (largest ↔
  smallest, second largest ↔ second smallest, …).  By the rearrangement
  inequality this never increases the true convolution peak.
* Periodically the true objective is evaluated; the best vector seen across
  all restarts is kept.
* A final *hill‑climbing* phase performs cheap random mass‑transfers and
  swaps (accepted only if they improve the true objective) until the time

The whole routine respects the 1000 s budget, prints progress information,
and returns the best list of floats discovered.

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _linear_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Equivalent to `np.convolve(a, a)`.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k p_k·(a*a)[k]  w.r.t. a.
    Here p is a probability vector (Σp = 1, p ≥ 0).  The result is:
          g_i = Σ_k p_k·a_{k‑i}
    Computed with a single FFT.
    """
    n = a.shape[0]
    # reversed weight vector for convolution
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # The needed slice corresponds to indices n‑1 … 2n‑2 (inclusive)
    seg = conv[n - 1 : 2 * n - 1]
    # reverse to obtain the gradient in the original order
    g = seg[::-1]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0 , Σx = 1}.  Implementation follows the classic O(n) algorithm.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]            # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest ones (pairwise).
    For a sorted ascending array s the result is
        [s_max, s_2nd_max, …, s_mid, s_mid+1, …, s_min]
    This never worsens the max‑convolution (rearrangement inequality).
    """
    s = np.sort(a)                 # ascending
    n = s.shape[0]
    out = np.empty_like(s)
    half = n // 2
    for i in range(half):
        out[i] = s[-1 - i]         # large → left side
        out[-1 - i] = s[i]         # small → right side
    if n % 2:
        out[half] = s[half]
    return out

def _hill_climb(a: np.ndarray, cur_val: float,
