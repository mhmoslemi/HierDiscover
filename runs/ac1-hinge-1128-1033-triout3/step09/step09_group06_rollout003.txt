"""
Propose a high‑quality step function for `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – generate several diverse starts:
   • the previous best (`height_sequence_1`) if available,
   • uniform,
   • truncated exponential with several decay rates,
   • random Dirichlet (flat‑simplex) draws.

2. **Smoothed‑max descent** – minimise a log‑sum‑exp surrogate of the
   maximum convolution value.
   * The surrogate is
        f_ε(a) = (1/ε)·log Σ_k exp(ε·c_k) ,  c = a∗a.
   * Gradient is obtained with a single FFT‑based convolution.
   * An exponentiated‑gradient (mirror‑descent) update keeps the vector
     non‑negative and on the probability simplex.
   * ε is decreased in many stages (0.5 → 0.0005); the last stages are
     allocated many more iterations so that the optimiser focuses on the
     true L∞‑norm.
   * After each stage the vector is **re‑ordered** with two different
     deterministic schemes:
        – anti‑symmetric (largest ↔ smallest, proven not to increase the max),
        – alternating (largest, smallest, second‑largest, second‑smallest …)
     and the better ordering is kept.

3. **Intensive hill‑climbing** – a simple simulated‑annealing style local
   search that repeatedly transfers a tiny amount of mass between two
   positions or swaps two entries.  Only improving moves are kept.
   The search continues until the time budget is exhausted.

All heavy work (FFT, gradient) is performed without calling
`evaluate_sequence`; the true objective is queried only occasionally to
track progress.  The routine respects the 1000 s budget and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)            # remove tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·c_k w.r.t. a, where c = a*a and p is a weight vector
    (Σp = 1, p ≥ 0).  Returns a vector of length n.
    """
    n = a.shape[0]
    # The convolution kernel for the gradient is the reverse of p.
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # The segment that corresponds to the gradient is the slice
    # indices n‑1 … 2n‑2 (inclusive) of the linear convolution.
    seg = conv[n - 1 : 2 * n - 1]
    # Reverse back to obtain the gradient w.r.t. a.
    return seg[::-1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Largest values on the left, smallest on the right (monotone decreasing)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Interleave largest and smallest values:
       large, small, 2nd‑large, 2nd‑small, …
    This yields a symmetric “U‑shaped” profile.
    """
    n = a.shape[0]
    s = np.sort(a)[::-1]                # descending
    b = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    idx = 0
    while left <= right:
        if toggle:
            b[left] = s[idx]
            left += 1
        else:
            b[right] = s[idx]
            right -= 1
        toggle = not toggle
        idx += 1
    return b

# Smoothed‑max descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _smoothed_descent(a_start: np.ndarray, deadline: float):
    """
    Minimise a log‑sum‑exp surrogate of max(convolution).
    Returns the best vector and its true evaluation (via evaluate_sequence).
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[descent init] objective = {best_val:.6f}")

    # ε schedule – we finish at ε = 5e‑4.
    eps_schedule = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01,
                    0.005, 0.002, 0.001, 0.0005]

    # Number of iterations per ε (more iterations for the finest eps).
    iters_per_eps = {
        0.5:   1500,
        0.2:   1500,
        0.1:   1500,
        0.05:  1500,
        0.02:  2000,
        0.01:  2500,
        0.005: 3000,
        0.002: 3500,
        0.001: 4000,
        0.0005:5000,
    }

    # Initial (multiplicative) step size.
    eta = 1.0

    for eps in eps_schedule:
        it_target = iters_per_eps[eps]
        for it in range(it_target):
            if time.time() >= deadline - 5.0:
                break

            # ----- convolution and soft‑max weights (smooth max) -----
            conv = _full_convolution(a)          # length 2n‑1
            max_conv = conv.max()

            # stable softmax
            shift = max_conv
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()        # Σp = 1, p ≥ 0

            # ----- gradient of the smooth surrogate -----
            grad = 2.0 * _gradient_weighted_peaks(a, p)   # shape (n,)

            # ----- exponentiated‑gradient (mirror descent) step -----
            # Clip to avoid overflow/underflow in the exponent.
            delta = -eta * grad
            delta = np.clip(delta, -20.0, 0.0)    # only decreasing updates
            a = a * np.exp(delta)
            a = _project_simplex(a)               # keep on the simplex

            # ----- occasional true‑objective check -----
            if it % 100 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[eps {eps:.4f} it {it}] new best = {best_val:.6f} (max conv {max_conv:.6e})")

                # try deterministic reorderings every 500 steps
                if it % 500 == 0:
                    a1 = _anti_symmetric_reorder(a)
                    val1 = evaluate_sequence(a1.tolist())
                    a2 = _alternating_reorder(a)
                    val2 = evaluate_sequence(a2.tolist())

                    # keep the best ordering
                    if val1 < best_val:
                        best_val, best_a, a = val1, a1.copy(), a1.copy()
                        print(f"[reorder anti eps {eps:.4f} it {it}] improved to {best_val:.6f}")
                    if val2 < best_val:
                        best_val, best_a, a = val2, a2.copy(), a2.copy()
                        print(f"[reorder alt  eps {eps:.4f} it {it}] improved to {best_val:.6f}")

            # slowly decay the step size for stability, especially at tiny eps.
            eta *= 0.9998

        # A final re‑ordering after each ε‑stage (cannot increase the max)
        a = _anti_symmetric_reorder(a)

    return best_a, best_val

# Hill‑climbing refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def _hill_climb_refine(a: np.ndarray, cur_val: float,
                       deadline: float, n: int):
    """
    Local search that randomly transfers a small amount of mass or swaps two
    entries.  Only moves that strictly improve the true objective are kept.
    """
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    # annealing schedule: start with larger transfers, gradually shrink.
    max_transfer = 0.05
    while time.time() < deadline - 2.0:
        # decide move type
        if rng.random() < 0.3:                     # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = best_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # transfer mass
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] < 1e-15:
                continue
            # amount shrinks as we get closer to the deadline
            time_frac = (deadline - time.time()) / (deadline - (deadline - 1000))
            cur_transfer = max_transfer * time_frac
            delta = best_a[i] * cur_transfer * rng.random()
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # enforce simplex and non‑negativity
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0.0:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < best_val - 1e-13:
            best_val = val
            best_a = new_a
            print(f"[hill] new best = {best_val:.6f}")

    return best_a, best_val
