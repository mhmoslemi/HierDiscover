def _adam_phase(a0, n, start_time, deadline, best_a, best_val, target_val=1.503, top_k=3):
    cur = a0.copy()
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute convolution and max
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb
        # update best
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            print(f"[Adam {t}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        # compute gradient from top K
        if top_k == 1:
            ks = [int(np.argmax(b))]
        else:
            ks = np.argpartition(b, -top_k)[-top_k:]
        g = np.zeros_like(cur)
        for k in ks:
            lo = max(0, k-(n-1))
            hi = min(k, n-1)
            idx = np.arange(lo,hi+1)
            g[idx] += 2.0 * cur[k - idx]
        # average if needed: Maybe divide by top_k.
        g = g / len(ks)

        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # step size schedule
        progress = (now - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * 0.5 + progress * 0.001 # start bigger
        # update
        cur = cur - eta * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        cur = _project_simplex(cur)
        # maybe jitter if stagnated? Could add small noise occasionally.
    return best_a, best_val

def _project_simplex(v):
    v = np.asarray(v)
    if v.ndim != 1:
        raise ValueError
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _lp_refine_phase(a, n, start_time, deadline, best_a, best_val, max_iter=5):
    import scipy.optimize as opt
    cur = a.copy()
    for i in range(max_iter):
        now = time.time()
        if now >= deadline:
            break
        b = _full_convolution(cur)
        M = b.max()
        cur_val = 2.0 * n * M
        # update best
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            print(f"[LPRefine {i}] new best = {best_val:.6f}")
        # find active indices within tolerance
        tol = 1e-12
        act = np.flatnonzero(b >= M - tol)
        if act.size == 0:
            break
        # construct A_ub
        rows = []
        for k in act:
            row = np.zeros(n + 1)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            row[idx] = 2.0 * cur[k - idx]
            row[n] = 1.0  # coefficient for eps
            rows.append(row)
        A_ub = np.vstack(rows)
        b_ub = np.zeros(len(rows))
        # equality sum(d)=0
        A_eq = np.concatenate([np.ones(n), [0.0]])[np.newaxis, :]
        b_eq = np.array([0.0])
        # objective: maximize eps => minimize -eps
        c = np.zeros(n + 1)
        c[-1] = -1.0
        # bounds
        bounds = [(-cur[i], None) for i in range(n)] + [(0, None)]
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})
        if not res.success:
            print(f"[LPRefine {i}] linprog failed: {res.message}")
            break
        eps = res.x[-1]
        d = res.x[:n]
        if eps < 1e-12:
            # no improvement
            break
        # line search
        step = 1.0
        for _ in range(10):
            cand = cur + step * d
            # enforce non-negativity and sum
            cand = np.maximum(cand, 0)
            cand_sum = cand.sum()
            if cand_sum == 0:
                cand = np.full(n, 1.0 / n)

                cand /= cand_sum
            cand_val = _objective(cand, n)
            if cand_val < cur_val:
                cur = cand
                cur_val = cand_val
                break
            step *= 0.5
        else:
            # no improvement found
            break
    return best_a, best_val

"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Initial deterministic search** – test several simple parametric families
   (sin^p, (1+cos)^p, Gaussian) on the full length (default 8000) and keep the
   best one.

2. **Random restarts** – each restart draws a Dirichlet vector with a
   randomly‑chosen concentration and improves it with three phases:
   * **Adam‑style projected gradient descent** – the sub‑gradient of the
     peak(s) of the auto‑convolution is computed, an Adam momentum update is
     applied, and the vector is projected onto the probability simplex.
   * **Linear‑program refinement** – at the current point the active peak(s)
     are linearised; a small linear program (maximising a feasible reduction
     ``ε``) yields a direction that provably reduces the maximal convolution
     entry.  A simple back‑tracking line search applies the direction.
   * **Deterministic hill‑climb** – move a fraction of the mass from the
     index contributing most to the worst peak to the index contributing the
     least, accepting only improvements.

   The best vector across all restarts is kept globally.

3. **Final polishing** – a short LP‑refinement and a tiny mixing with the
   uniform vector are performed on the overall best candidate.

All heavy work respects a global time budget (≈ 1000 s).  The code uses only
``numpy`` and ``scipy.optimize.linprog``; ``evaluate_sequence`` is called only
once at the very end to obtain the official score.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)   # denominator = 1 because Σa = 1


# ----------------------------------------------------------------------
# Simple parametric families – fast deterministic sweep
# ----------------------------------------------------------------------
def _parametric_initial_candidate(n: int):
    """Return a decent starting vector found by scanning a few families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    # sine power
    xs = np.linspace(0, np.pi, n)
    for p in np.linspace(0.2, 5.0, 25):
        a = np.sin(xs) ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # (1+cos) power – a raised‑cosine “bump”
    xs = np.linspace(0, np.pi, n)
    for p in np.linspace(0.2, 5.0, 25):
        a = (1.0 + np.cos(xs)) ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # Gaussian‑like bumps
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    for sigma_frac in np.linspace(0.05, 0.30, 10):
        sigma = sigma_frac * n
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a /= a.sum()
        val = _objective(a, n)
        if val < best_val:
            best_val, best_a = val, a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent (uses sub‑gradient of top‑K peaks)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start_time: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                top_k: int = 3,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected Adam optimisation of the max‑convolution objective."""
    cur = a0.copy()
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # Active set – indices whose convolution value is within a tiny
        # absolute tolerance of the current maximum.
        tol_abs = 1e-12
        active = np.flatnonzero(b >= maxb - tol_abs)
        # keep at most ``top_k`` entries for efficiency
        if active.shape[0] > top_k:
            idxs = np.argpartition(b[active], -top_k)[-top_k:]
            ks = active[idxs]
        else:
            ks = active

        # sub‑gradient (averaged over the active indices)
        g = np.zeros_like(cur)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * cur[k - idx]
        g /= max(1, len(ks))

        it += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # linear decay of the learning rate (0.5 → 0.001)
        progress = (now - start_time) / (deadline - start_time)
        eta = (1 - progress) * 0.5 + progress * 0.001

        cur = cur - eta * m_hat / (np.sqrt(v_hat) + eps)
        cur = _project_simplex(cur)

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program refinement of the active peak(s)
# ----------------------------------------------------------------------
def _lp_refine_phase(a0: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     max_iter: int = 5) -> tuple[np.ndarray, float]:
    """Solve a tiny LP that guarantees a first‑order reduction of the peak."""
    cur = a0.copy()
    for it in range(max_iter):
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        M = b.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            print(f"[LPRefine {it:2d}] new best = {best_val:.6f}")

        # active peak indices (absolute tolerance)
        active = np.flatnonzero(b >= M - 1e-12)
        if active.size == 0:
            break

        # Build LP matrices.
        # Variables: d_0 … d_{n‑1} (direction for a) and eps ≥ 0 (reduction).
        rows = []
        for k in active:
            row = np.zeros(n + 1)
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            row[idx] = 2.0 * cur[k - idx]          # gradient of (a*a)[k]
            row[n] = 1.0                           # coeff. of eps
            rows.append(row)
        A_ub = np.vstack(rows)
        b_ub = np.zeros(len(rows))

        # Equality: Σ d_i = 0 (keeps total mass unchanged)
        A_eq = np.concatenate([np.ones(n), [0.0]])[np.newaxis, :]
        b_eq = np.array([0.0])

        # Objective: maximise eps  → minimise -eps
        c = np.zeros(n + 1)
        c[-1] = -1.0

        # Bounds: d_i ≥ –a_i (so a_i + d_i stays non‑negative), eps ≥ 0
        bounds = [(-cur[i], None) for i in range(n)] + [(0.0, None)]

        res = linprog(c,
                      A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # LP failed – give up refinement
            print(f"[LPRefine {it}] linprog failed: {res.message}")
            break

        eps = res.x[-1]
        d = res.x[:n]

        if eps < 1e-12:
            # No meaningful reduction found
            break

        # Back‑tracking line search (full step first)
        step = 1.0
        improved = False
        for _ in range(10):
            cand = cur + step * d
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                cand = np.full(n, 1.0 / n)
            else:
                cand /= s
            cand_val = _objective(cand, n)
            if cand_val < cur_val - 1e-14:
                cur = cand
                cur_val = cand_val
                improved = True
                break
            step *= 0.5
        if not improved:
            # No improvement even with small step → stop refining
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb – deterministic mass transfer
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     start_frac: float = 0.30,
                     end_frac: float = 0.001,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move mass from the worst contributor to the best; accept only improvements."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        k = int(np.argmax(b))                     # current worst peak index
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)

        # contribution of each a_i to the peak k
        contrib = np.zeros_like(cur)
        contrib[idx] = cur[k - idx]

        src = int(np.argmax(contrib))
        # source may be depleted – skip in that case
        if cur[src] <= 1e-15:
            if no_improve > IMPATIENT:
                # occasional jitter to escape stagnation
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur = _project_simplex(cur)
                no_improve = 0
                print(f"[HillClimb {it}] jitter after stagnation")
            it += 1
            continue

        # target = index with minimal contribution (excluding source)
        tmp = contrib.copy()
        tmp[src] = np.inf
        tgt = int(np.argmin(tmp))

        delta = cur[src] * frac
        if delta <= 0.0:
            it += 1
            continue

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        new_val = _objective(cur, n)
        if new_val < cur_val - 1e-15:
            # keep the move
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combinations (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand = _project_simplex(cand)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val
