def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5
    n = 9000
    # define compute objective
    def objective_from_indices(indices):
        # indices is numpy array of ints, size m
        # build indicator array
        ind = np.zeros(n, dtype=np.float64)
        ind[indices] = 1.0/len(indices)
        # FFT convolution
        L = 2*n - 1
        size = 1 << (L - 1).bit_length()
        af = np.fft.rfft(ind, size)
        conv = np.fft.irfft(af * af, size)[:L]
        max_b = conv.max()
        return 2.0 * n * max_b
    # parameters for search
    target_val=1.503  # for early exit
    best_val = float('inf')
    best_indices = None
    # define list of m candidates
    ms = [220, 250, 280, 300, 340, 380] # we can choose few
    for m in ms:
        # start with random support
        indices = rng.choice(n, size=m, replace=False)
        indices_set = set(indices.tolist())
        cur_val = objective_from_indices(indices)
        # keep best
        if cur_val < best_val:
            best_val = cur_val
            best_indices = indices.copy()
            print(f"Init m={m} val={cur_val:.6f}")
        # hill climbing loop
        # schedule similar to simulated annealing with temperature decreasing

        iterations = 0
        while time.time() < deadline:
            # temperature schedule linear
            elapsed = time.time() - start
            total_time = deadline - start
            temperature = T0 * max(0.0, 1.0 - elapsed/total_time)
            # propose swap
            # choose i from current support
            i_idx = rng.integers(m)   # index into array
            i = indices[i_idx]
            # choose new index not in support
            # sample until not in set
            while True:

                if new not in indices_set:

            cand_indices = indices.copy()
            cand_indices[i_idx] = new

            cand_val = objective_from_indices(cand_indices)
            # acceptance
            accept = False
            if cand_val < cur_val:

                indices_set.add(new)

"""
Sparse‑support simulated‑annealing for the min‑max self‑convolution problem.

The idea:
* The objective is 2·n·max(conv(a,a)) / (sum(a)^2).  Because the evaluation is
  scale‑invariant we may enforce sum(a)=1.
* If we restrict a to have only *m* non‑zero entries (all equal to 1/m)
  then max(conv(a,a)) = max_count / m² where *max_count* is the maximal
  number of ordered pairs (i,j) from the support that share the same sum
  i+j.  The problem reduces to:  minimise the maximal collision count of
  the sum‑set of a size‑m subset of {0,…,n‑1}.
* A random subset gives a moderate collision count; by repeatedly swapping
  one element of the subset with a new index (simulated‑annealing) we can
  drive the maximal collision count down dramatically.
* After a good subset is found we return the corresponding step‑function:
  all selected positions receive weight 1/m, all others 0.
* The algorithm stays well within the 1000 s budget, uses only NumPy and
  the provided `evaluate_sequence`, and does not rely on any external
  optimisation libraries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: objective from a set of indices (equal weights)
# ----------------------------------------------------------------------
def _objective_from_indices(indices: np.ndarray, n: int) -> tuple[float, int, np.ndarray]:
    """
    Compute the evaluation value for a vector that has weight 1/|indices|
    on the given positions and zero elsewhere.

    Returns
    -------
    value : float
        2·n·max(conv) for the corresponding vector.
    max_counts : int
        maximal number of ordered pairs (i,j) from the support having the
        same sum i+j.
    counts : ndarray
        full histogram of collision counts (length 2·n‑1).
    """
    m = indices.shape[0]
    # all ordered sums (i+j) as a flat array
    sums = indices[:, None] + indices[None, :]          # shape (m,m)
    # histogram of how many ordered pairs hit each possible sum
    counts = np.bincount(sums.ravel(), minlength=2 * n - 1)
    max_counts = int(counts.max())
    # max(conv) = max_counts / m²  (because each pair contributes 1/m²)
    value = 2.0 * n * max_counts / (m * m)
    return value, max_counts, counts


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) with a low value of
    `evaluate_sequence`.  A sparse support is optimised by simulated annealing.
    The function respects the 1000 s time budget and returns the best sequence
    found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    n = 9000                     # length of the vector (a good trade‑off)
    TARGET = 1.503               # stop early if we reach this bound

    best_val = float("inf")
    best_indices: np.ndarray | None = None

    # ------------------------------------------------------------------
    # Try a few different support sizes.  The theoretical estimate
    # suggests m ≈ 2·√(2·n) ≈ 260 gives a good chance of beating 1.5.
    # ------------------------------------------------------------------
    for m in (220, 250, 280, 300, 340):
        if time.time() >= deadline:
            break

        # ---- initialise a random support of size m -------------------
        indices = np.sort(rng.choice(n, size=m, replace=False))
        cur_val, cur_maxcnt, cur_counts = _objective_from_indices(indices, n)
        cur_set = set(indices.tolist())

        if cur_val < best_val:
            best_val = cur_val
            best_indices = indices.copy()
            print(f"[init] m={m}  value={cur_val:.6f}")

        # ---- simulated‑annealing loop --------------------------------
        T0 = cur_val * 0.02                # initial temperature (2 % of value)
        it = 0
        while time.time() < deadline:
            # linear cooling schedule (fraction of total remaining time)
            elapsed = time.time() - start_time
            total = deadline - start_time
            temperature = T0 * max(0.0, 1.0 - elapsed / total)

            # propose a single swap i → j
            pos_i = rng.integers(m)                 # which element to replace
            old = int(indices[pos_i])

            # draw a new index not already in the support
            # (few attempts are enough because m << n)
            for _ in range(5):
                new = int(rng.integers(n))
                if new not in cur_set:
                    break
            else:
                # give up this iteration if we couldn't find a free slot
                continue

            cand_indices = indices.copy()
            cand_indices[pos_i] = new

            cand_val, cand_maxcnt, _ = _objective_from_indices(cand_indices, n)

            # acceptance test (Metropolis criterion)
            accept = False
            if cand_val < cur_val:
                accept = True
            elif temperature > 0:
                prob = np.exp(-(cand_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

            if accept:
                # update current state
                cur_set.remove(old)
                cur_set.add(new)
                indices = cand_indices
                cur_val = cand_val
                cur_maxcnt = cand_maxcnt

                # record global best
                if cur_val < best_val:
                    best_val = cur_val
                    best_indices = indices.copy()
                    print(f"[m={m}] new best {best_val:.6f} after {it:,} iterations")
                    if best_val <= TARGET:
                        break

            it += 1
            # a soft limit to avoid endless loops for a given m
            if it > 150_000:
                break

        if best_val <= TARGET:
            break

    # ------------------------------------------------------------------
    # Build the final sequence (all selected positions get equal mass)
    # ------------------------------------------------------------------
    sequence = np.zeros(n, dtype=float)
    if best_indices is not None:
        m_best = best_indices.shape[0]
        sequence[best_indices] = 1.0 / m_best
    else:
        # fallback – uniform distribution
        sequence[:] = 1.0 / n

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(sequence.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return sequence.tolist()
