"""
Propose a step‑function sequence that minimizes the security‑wrapped
objective:

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in four stages within the given time budget:

1. **Initialisation** – try a few deterministic families and, if available,
   interpolate a previously found solution (`height_sequence_1`).

2. **Broad random exploration** – sample thousands of candidates from a
   rich collection of parametric families (exponential, power‑law,
   beta‑like, Gaussian and mixtures).  Every candidate is normalised to
   `sum = 1` and its objective (the exact value returned by
   `evaluate_sequence` for a normalised vector) is computed with an FFT‑based
   convolution.  The best candidate so far is kept.

3. **Adaptive exponentiated‑gradient descent** – a sub‑gradient of
   `max(conv(a,a))` is obtained from the index where the convolution attains
   its maximum.  The update uses an Adam‑style momentum scheme and an
   exponentially decreasing learning‑rate.  Periodic jitter escapes plateaus.

4. **Deterministic greedy hill‑climbing** – repeatedly move a small
   fraction of mass from the index that contributes most to the current
   peak to the index that contributes least.  The step size shrinks
   linearly with time and jitter is added after long periods without
   improvement.

5. **Final polishing** – blend the best vector with a tiny amount of the
   uniform distribution (and optionally re‑evaluate a few tiny random
   perturbations) before returning it.

All steps respect the non‑negativity and normalisation constraints, and
the total runtime never exceeds ``budget_s`` seconds (a small safety margin
is kept).  The function prints intermediate objective values so that the
progress can be monitored.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via FFT (real‑valued)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective for a normalised vector (sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a unit‑sum vector."""
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i   (0 < r < 1)."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**(n‑1‑i)."""
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    """mixture w·r1**i + (1‑w)·r2**(n‑1‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)**p."""
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    """a_i ∝ (i+1)**p · r**i."""
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)               # stabilise
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_beta_like(n: int, a: float, b: float) -> np.ndarray:
    """U‑shaped beta‑like density:
       a_i ∝ (i+1)^{a-1} * (n-i)^{b-1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    # add tiny epsilon to avoid zero exponents on the boundary
    eps = 1e-12
    left = np.power(i + eps, a - 1.0)
    right = np.power(n - i + 1 + eps, b - 1.0)
    a_vec = left * right
    s = a_vec.sum()
    return a_vec / s if s != 0 else np.full(n, 1.0 / n)


def _gen_gaussian(n: int, mu: float, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-0.5*((i‑mu)/σ)^2)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_gauss_mix(n: int, mu1: float, mu2: float, sigma: float, w: float) -> np.ndarray:
    """mixture of two Gaussians."""
    i = np.arange(n, dtype=np.float64)
    g1 = np.exp(-0.5 * ((i - mu1) / sigma) ** 2)
    g2 = np.exp(-0.5 * ((i - mu2) / sigma) ** 2)
    a = w * g1 + (1.0 - w) * g2
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)

def _random_search(n: int, deadline: float, best_a: np.ndarray,

        elif choice == 1:                          # reversed exponential

            cand = _gen_rev_exp(n, r)
        elif choice == 2:                          # mixture of two exponentials
            r1 = rng.uniform(0.998, 0.9999995)
            r2 = rng.uniform(0.998, 0.9999995)
            w = rng.random()
            cand = _gen_mix_exp(n, r1, r2, w)
        elif choice == 3:                          # pure power‑law
            p = rng.uniform(-5.0, 5.0)
            cand = _gen_power(n, p)
        elif choice == 4:                          # power‑law × exponential
            p = rng.uniform(-5.0, 5.0)
            r = rng.uniform(0.998, 0.9999995)
            cand = _gen_power_exp(n, p, r)
        elif choice == 5:                          # beta‑like U‑shape

            sigma = rng.uniform(0.07 * n, 0.2 * n)
            w = rng.random()
            cand = _gen_gauss_mix(n, mu1, mu2, sigma, w)

        val = _objective(cand, n)
        if val < best_val:

# Phase‑A : Adam‑style exponentiated‑gradient descent -------------------
def _adam_expgrad(a0: np.ndarray, n: int, start: float,
                  deadline: float, best_a: np.ndarray,
                  best_val: float, target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Multiplicative Adam‑like updates on the sub‑gradient of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0
    no_improve = 0
    IMPATIENT = 180_000

    while time.time() < deadline:
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[AdamGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # sub‑gradient at the current maximum index
        k = int(np.argmax(conv))
        rev = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = a[k - idx]                 # a_{k‑i}
        g = 2.0 * rev                         # ∇ max_b

        # Adam moments
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # schedule: start relatively aggressive, end gentle
        progress = (time.time() - start) / (deadline - start)
        lr = (1.0 - progress) * 0.18 + progress * 0.001

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # multiplicative update keeps positivity automatically
        a = a * np.exp(-step)

        # renormalise to sum == 1
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        if no_improve > IMPATIENT:
            # jitter to escape possible stagnation
            noise = np.random.normal(scale=0.009, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            no_improve = 0
            print(f"[AdamGrad {it:7d}] jitter")

    return best_a, best_val

# Phase‑B : deterministic greedy hill‑climbing -------------------------
def _hill_climb(a_start: np.ndarray, n: int, start: float,
                deadline: float, best_a: np.ndarray,
                best_val: float, target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass from a high‑contributor to a low‑contributor."""
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 250_000
    max_frac_start, max_frac_end = 0.55, 0.001      # shrink to tiny moves

    rng = np.random.default_rng(123456 + it)       # local RNG for reproducibility

    while time.time() < deadline:
        conv = _full_convolution(cur)
        cur_val = 2.0 * n * conv.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if cur_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index of the offending peak
        k = int(np.argmax(conv))
        rev = np.zeros_like(cur)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))                 # donor (largest contributor)
        # choose a distinct receiver with smallest contribution
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))              # receiver

        if cur[i] <= 1e-15:
            # donor empty – apply jitter if we have been stuck too long
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.007, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        # amount to transfer
        delta = cur[i] * max_frac * rng.random()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        # evaluate the tentative move
        new_conv = _full_convolution(cur)
        new_val = 2.0 * n * new_conv.max()
        if new_val < cur_val - 1e-15:
            cur_val = new_val                     # keep the change
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑C : tiny mixing with the uniform vector ------------------------
def _final_mix(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Blend a tiny amount of the uniform distribution; try a few alphas."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)

    for alpha in (0.001, 0.003, 0.006, 0.010, 0.018):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a small
    ``evaluate_sequence`` value.  The routine follows the stages described
    in the module docstring and respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.4

    # ------------------------------------------------------------------
    # Problem size – a power of two gives fast FFTs.
    # ------------------------------------------------------------------
    n = 8192

    # ------------------------------------------------------------------
    # Initialise: try to reuse a previously found vector, otherwise fall back
    # to a few deterministic shapes.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        # simple deterministic starter (uniform)
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑0 : random exploration of many parametric families
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    random_deadline = start_time + 0.22 * (deadline - start_time)
    best_a, best_val = _random_search(n, random_deadline, best_a, best_val, rng)

    # ------------------------------------------------------------------
    # Phase‑A : Adam‑style exponentiated‑gradient descent
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.57 * (deadline - start_time)
    best_a, best_val = _adam_expgrad(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
    )

    # ------------------------------------------------------------------
    # Phase‑B : deterministic greedy hill‑climbing
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.90 * (deadline - start_time)
    best_a, best_val = _hill_climb(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
    )

    # ------------------------------------------------------------------
    # Phase‑C : final tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _final_mix(best_a, n)

    # ------------------------------------------------------------------
    # Verify with the security‑wrapped evaluator
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:                       # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
