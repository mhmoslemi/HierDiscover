def _project_simplex(v):
    """Projection of v onto the probability simplex."""
    # based on Duchi et al. 2008

    u = np.sort(v)[::-1]

"""
Propose a non‑negative step function (list of floats) that minimises the
`evaluate_sequence` objective within a 1000 s budget.

The algorithm proceeds in three stages:

1. **Initialisation & random restarts** – start from the historic
   construction if available, otherwise from a Dirichlet (exponential)
   draw, normalised to a probability vector.  A few additional random
   vectors are evaluated to avoid a bad start.

2. **Smoothed log‑sum‑exp mirror descent with Adam** – we minimise a smooth
   approximation of the maximal self‑convolution
   `max(a*a)`.  For a given vector `a` (sum = 1) we compute the full linear
   convolution `c = a*a` using an FFT.  The surrogate loss
   `L = (1/β)·log(∑ exp(β·c))` has gradient
   `∇L = 2·(a * p_rev)` where `p = softmax(β·c)`.  An Adam–style update
   is performed on the probability simplex (multiplicative update
   replaced by a projected additive update).  The smoothing parameter `β`
   is increased from 1 to a large value (≈ 500) during the run, gradually
   turning the surrogate into the true maximum.

3. **Local refinement** – a cheap stochastic hill‑climbing phase that
   tries random swaps and tiny mass transfers, accepting only strict
   improvements.  The convolution‑based objective is evaluated directly
   (no call to `evaluate_sequence`) to keep the loop fast.

The best vector found is finally evaluated with `evaluate_sequence` and
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a∗a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negative rounding artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Implements g = 2·(a * w_rev) using a single FFT convolution.
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1            # = 3·n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    # the relevant slice starts at offset (len(w_rev)-1)
    start = w_rev.shape[0] - 1
    seg = conv[start:start + n]
    return 2.0 * seg


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values opposite the smallest ones (pairwise).
    This heuristic often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b

# Main optimisation routine
# ----------------------------------------------------------------------

    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` within the given time budget.
    """

    rng = np.random.default_rng(seed)

    # keep a few seconds free for the final evaluation
    deadline = start_time + budget_s - 5.0

    # Problem size – a larger N gives more freedom for flattening the
    # convolution while still keeping each iteration cheap.
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            # linear interpolation to the target length
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        a = a0 / s0 if s0 > 0 else np.full(n, 1.0 / n)
    except Exception:

        a = rng.exponential(scale=1.0, size=n)

        a /= a.sum()
