    n = len(a0)
    a = a0.copy()

    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max
    no_improve = 0

    while time.time() < deadline:
        progress = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # find k_max
        k_max = int(b.argmax())
        # contributions
        i_indices = np.arange(n)
        pos = k_max - i_indices
        mask = (pos >= 0) & (pos < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[pos[mask]]
        # source index: max contribution
        if mask.sum() == 0:
            continue
        i = np.argmax(contrib)

        if a[i] < 1e-12:
            # maybe choose other index
            i = int(np.argmax(a))
        # target selection weights
        # we want indices where contrib is low.
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + 1e-12)
        tgt_weights[i] = 0.0

        if wsum > 0:

        else:

            j = (i + np.random.randint(1, n)) % n

"""
A fast three‑phase optimiser for the step‑function problem.

Phase 0 – cheap parametric sweep (Beta family + Dirichlet) to obtain a
reasonable start point (sum of the vector is forced to 1).

Phase 1 – aggressive simulated‑annealing hill‑climbing.
    * The convolution is kept incrementally, so each move costs O(n) only.
    * The update of the convolution has been corrected (the quadratic
      term at index i+j uses –2·δ², the previous code used –δ²).
    * Source of mass = index that contributes most to the current maximal
      convolution entry; target = an index with a small contribution.
    * A temperature and a maximal transfer fraction decay linearly,
      providing coarse moves early and fine polishing later.
    * Stagnation triggers a small random perturbation around the best
      solution seen so far.

Phase 2 – deterministic projected sub‑gradient descent on the exact
objective, final polishing.

The whole routine respects a 1000 s wall‑time budget, leaves a short
margin for the final secure call to ``evaluate_sequence`` and returns
the best sequence found.
"""

from __future__ import annotations
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (ℓ₁‑ball with sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = 1, x ≥ 0."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[mask][-1] - 1
    theta = cssv[mask][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Parametric families for a quick initial guess
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray | None, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    # a modest grid that still covers a wide range
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def _beta_random_search(n: int, trials: int = 200) -> tuple[np.ndarray | None, float]:
    """Randomly sample (p,q) on a log‑scale and keep the best Beta shape."""
    best_a = None
    best_val = np.inf
    i = np.arange(1, n + 1, dtype=np.float64)
    rng = np.random.default_rng()
    for _ in range(trials):
        logp = rng.uniform(-6.9, 2.3)   # ln(0.001) … ln(10)
        logq = rng.uniform(-6.9, 2.3)
        p = np.exp(logp)
        q = np.exp(logq)
        a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Hill‑climbing with corrected convolution update
# ----------------------------------------------------------------------
def _hillclimb(
    a_start: np.ndarray,
    start_time: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing.
    Returns (best_a, best_max_conv) where best_max_conv = max(conv(a,a)).
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)          # length 2n‑1
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # schedule parameters
    max_frac_start, max_frac_end = 0.8, 0.001
    temp_start, temp_end = 0.02, 1e-12
    no_improve = 0
    IMPROVE_LIMIT = 60_000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # ---- locate the current maximal convolution entry ----------
        k_max = int(b.argmax())

        # ---- contributions of each index to that entry -------------
        idx = np.arange(n)
        pos = k_max - idx
        mask = (pos >= 0) & (pos < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[pos[mask]]

        # ---- source: index that contributes the most ---------------
        if mask.sum() == 0:
            # fallback – pick the globally largest coefficient
            i = int(np.argmax(a))
        else:
            i = int(np.argmax(contrib))

        if a[i] < 1e-14:
            # extremely small – pick a random non‑zero index
            nz = np.where(a > 1e-14)[0]
            if nz.size == 0:
                break
            i = int(np.random.choice(nz))

        # ---- target: bias towards indices with low contribution ----
        tgt_w = np.zeros_like(a)
        tgt_w[mask] = 1.0 / (contrib[mask] + 1e-14)
        tgt_w[i] = 0.0
        total_w = tgt_w.sum()
        if total_w > 0:
            j = int(np.random.choice(n, p=tgt_w / total_w))
        else:
            # uniform fallback (excluding i)
            j = int(np.random.randint(n - 1))
            if j >= i:
                j += 1

        # ---- amount of mass to move --------------------------------
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the full convolution -----------
        b_new = b.copy()
        # linear part
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic part (corrected!)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()

        # ---- Metropolis acceptance ---------------------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(new_max - cur_max) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_max = new_max
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()
                print(f"[Hillclimb] t={now - start_time:5.1f}s iter={it} new best max={best_max:.6f}")
                if 2.0 * n * best_max <= 1.503:      # early exit if target already reached
                    break
        else:
            no_improve += 1

        # ---- stagnation handling ----------------------------------
        if no_improve > IMPROVE_LIMIT:
            # Small perturbation around the best point seen so far
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_max = b.max()
            no_improve = 0
            print(f"[Hillclimb] t={now - start_time:5.1f}s stagnation – re‑perturb")
        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Deterministic projected sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(
    a_start: np.ndarray,
    start_time: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a) = 2·n·max(conv(a,a))
    Returns the best (a, max_conv) found before the deadline.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_max = b.max()
    best_a = a.copy()
    best_max = cur_max

    # step schedule (linear decay)
    step_start, step_end = 0.5, 1e-5

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        step = step_start * (1.0 - progress) + step_end * progress

        # sub‑gradient of max(b) w.r.t. a
        b = _full_convolution(a)
        max_val = b.max()
        k_maxes = np.where(b == max_val)[0]

        grad = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        b_new = _full_convolution(a_new)
        new_max = b_new.max()

        if new_max < cur_max - 1e-12:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                print(f"[Subgrad] t={now - start_time:5.1f}s iter={it} new best max={best_max:.6f}")
                if 2.0 * n * best_max <= 1.503:
                    break
        # occasional jitter to escape flat regions
        if np.random.rand() < 0.001:
            a = np.maximum(a + np.random.normal(scale=0.01, size=n), 0.0)
            a = _project_simplex(a)

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimises a non‑negative step function under a 1000 s wall‑time budget.
    Returns a list whose sum is 1 and that (approximately) minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a trade‑off between expressivity and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric initialisation
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf

    # 0.1 s – try to reuse a previous solution if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val0 = _objective(a0)
        best_a, best_val = a0, val0
        print(f"[Init] interpolated previous solution, value={val0:.6f}")
    except Exception:
        pass

    # 0.3 s – Beta‑family grid search
    shape_a, shape_val = _beta_family_best(n)
    if shape_a is not None and shape_val < best_val:
        best_a, best_val = shape_a, shape_val
        print(f"[Init] beta‑grid improved value={best_val:.6f}")

    # 0.3 s – random Beta draws
    rand_a, rand_val = _beta_random_search(n, trials=250)
    if rand_a is not None and rand_val < best_val:
        best_a, best_val = rand_a, rand_val
        print(f"[Init] beta‑random improved value={best_val:.6f}")

    # a few Dirichlet draws for extra diversity
    for _ in range(3):
        cand = np.random.dirichlet(np.ones(n))
        cand_val = _objective(cand)
        if cand_val < best_val:
            best_a, best_val = cand, cand_val
            print(f"[Init] Dirichlet seed improved value={best_val:.6f}")

    # fallback – uniform random
    if best_a is None:
        best_a = np.random.rand(n)
        best_a /= best_a.sum()
        best_val = _objective(best_a)
        print("[Init] fallback uniform random, value=", best_val)

    # --------------------------------------------------------------
    # Phase 1 – hill‑climbing (≈80 % of the remaining budget)
    # --------------------------------------------------------------
    hill_deadline = start_time + 0.80 * budget_s
    best_a, best_max = _hillclimb(best_a, start_time, hill_deadline)
    best_val = 2.0 * n * best_max
    print(f"[Phase 1] best value after hillclimb = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deterministic sub‑gradient polishing (remaining time)
    # --------------------------------------------------------------
    polish_deadline = deadline
    best_a, best_max = _subgradient_polish(best_a, hill_deadline, polish_deadline)
    best_val = 2.0 * n * best_max
    print(f"[Phase 2] final polished value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final secure evaluation
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[Final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[Final] evaluate_sequence raised:", exc)

    return best_a.tolist()
