"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence.  The algorithm is a hybrid search:

1.  Initialise with a deterministic family or a previously found vector.
2.  A short random Dirichlet sweep to catch easy improvements.
3.  **Soft‑max mirror descent**:
    –  minimise an annealed “soft‑max’’ of the self‑convolution,
       i.e.  Mβ(a)= (1/β)·log Σ_k exp(β·(a*a)_k).
    –  The gradient of Mβ can be written as
          ∇Mβ(a)= 2·(a ⋆ w)   where  w_k ∝ exp(β·(a*a)_k)
       (⋆ = cross‑correlation).  Both the convolution and the
       correlation are evaluated by FFT, so one iteration costs
       O(n log n) and is very fast for n≈8000.
    –  β is annealed from 1 to ~128, turning the smooth objective into
       the true max‑convolution near the end of the run.
    –  Updates are performed with exponentiated‑gradient (mirror descent)
       to keep the variables non‑negative and automatically normalised.
    –  Periodic jitter (Gaussian noise) helps escape shallow local minima.
4.  A final deterministic greedy hill‑climb (the same routine used in the
    reference solution) refines the result.
5.  A tiny mixture with the uniform vector is tried; the best of the two
    is returned.
The whole routine respects the 1000 s time budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolutions (FFT based)
# ----------------------------------------------------------------------
def _full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power‑of‑two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (length len(x)+len(y)‑1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by evaluate_sequence for a normalised vector."""
    max_b = _full_convolution_self(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Deterministic families – cheap initial sweep
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent start vector and its objective."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = _objective(best_a, n)

    def try_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective(vec, n)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    rng = np.random.default_rng(12345)

    # some simple deterministic shapes
    idx = np.arange(n)
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # U‑shape linear
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # decreasing quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996,
              0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # a few random Dirichlet draws
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic greedy hill‑climbing (identical to reference)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution_self(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if cur_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        new_val = 2.0 * n * _full_convolution_self(cur).max()
        if new_val < cur_val - 1e-15:
            cur_val = new_val
        else:
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny uniform mixing – final polishing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  A hybrid of random seeding, soft‑max mirror descent and a
    final greedy hill‑climb is employed within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a short safety margin
    deadline = start_time + budget_s - 0.5

    # problem size – 8000 offers a good speed / quality trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # Initial guess (use previous solution if available)
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a0 = np.maximum(a0, 0.0)
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – quick random Dirichlet exploration (≈5 % of budget)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    phase0_deadline = start_time + 0.05 * (deadline - start_time)
    while time.time() < phase0_deadline:
        cand = rng.dirichlet(np.ones(n))
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Rand0] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – annealed soft‑max mirror descent
    # ------------------------------------------------------------------
    cur = best_a.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 200_000
    TARGET = 1.503

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- objective & soft‑max weights ---------------------------------
        conv = _full_convolution_self(cur)               # length 2n‑1
        max_c = conv.max()

        # progress ∈ [0,1]
        progress = (now - start_time) / (deadline - start_time)

        # learning‑rate schedule (fast start, slow finish)
        eta = (1.0 - progress) * 0.4 + progress * 0.0008

        # annealing of β (soft‑max sharpness)
        beta = 1.0 + 127.0 * progress                 # 1 → 128

        # soft‑max weights (stable – shift by max_c)
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:          # numerical safety
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ----- gradient = 2·(a ⋆ w) ----------------------------------------
        rev_w = w[::-1]                                  # length 2n‑1
        g_full = _full_convolution_two(cur, rev_w)       # length 3n‑2
        start = len(w) - 1                               # = 2n‑2
        grad = 2.0 * g_full[start:start + n]             # size n

        # ----- exponentiated‑gradient update -------------------------------
        cur = cur * np.exp(-eta * grad)
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # ----- evaluate ----------------------------------------------------
        cur_val = _objective(cur, n)

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_val:.6f}")
            if best_val <= TARGET:
                break
        else:
            no_improve += 1

        # ----- occasional jitter to escape stagnation ---------------------
        if no_improve > IMPATIENT:
            noise = rng.normal(scale=0.006, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur_sum = cur.sum()
            if cur_sum == 0.0:
                cur = np.full(n, 1.0 / n)
            else:
                cur /= cur_sum
            no_improve = 0
            print(f"[SGD {it:7d}] jitter after stagnation")

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – deterministic greedy hill‑climb (≈15 % of remaining time)
    # ------------------------------------------------------------------
    hill_start = time.time()
    hill_deadline = deadline - 0.1 * (deadline - start_time)   # leave a little for final mix
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=hill_start,
        deadline=hill_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase 3 – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final verification with the provided security wrapper
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:        # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
```

This implementation builds on the previous approach but replaces the exponentiated‑gradient / hill‑climbing pipeline with a **soft‑max mirror‑descent** phase that gradually sharpens the objective, followed by a short deterministic hill‑climb and a final uniform mixing.  All operations are O(n log n) per iteration, allowing hundreds of thousands of updates inside the 1000 s budget and giving a strong chance of beating the 1.503 target.
