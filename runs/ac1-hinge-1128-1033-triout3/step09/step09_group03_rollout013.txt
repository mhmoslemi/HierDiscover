"""
Heuristic optimizer for min‑max self‑convolution.

The optimizer mixes:
  - a parametric sweep (Beta‑shaped densities) to get a good starting point,
  - Adam on a smooth log‑sum‑exp surrogate,
  - simulated‑annealing hill‑climbing that moves mass between indices,
  - exponentiated‑gradient descent with a decreasing smoothing parameter,
  - a final uniform‑mix polishing step.

All phases respect the 1000 s time budget.
"""

import time, numpy as np

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _objective_value(a):
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())

def _project_simplex(v):
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

"""
Heuristic optimizer for the “min‑max self‑convolution” problem.

The routine combines several complementary strategies under the
1000 s time budget:

  * a quick parametric sweep over Beta‑shaped densities (U‑shaped,
    bell‑shaped, etc.) to obtain a solid starting point,
  * Adam optimisation on a smooth log‑sum‑exp surrogate of the
    max‑convolution,
  * a classic simulated‑annealing hill‑climber that moves mass between
    indices that contribute heavily / lightly to the current peak,
  * an exponentiated‑gradient fine‑tuning phase (multiplicative updates
    on the simplex) with a decreasing smoothing parameter,
  * a final mixing with a tiny amount of uniform mass.

All vectors stay on the simplex (non‑negative, sum = 1), which makes the
objective scale‑invariant.  The best sequence found is returned as a
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (sum of a is forced to 1 → objective = 2·n·max(conv(a,a)))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project an arbitrary vector onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small library of hand‑crafted / random families – quick start.
# ----------------------------------------------------------------------
def _initial_candidate(n: int, rng: np.random.Generator):
    """Return a decent seed vector (sum = 1) among a few families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _consider(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a.copy(), val

    # uniform
    _consider(np.ones(n))

    # symmetric exponentials (U‑shaped)
    for r in (0.995, 0.98, 0.95, 0.9):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        _consider(a)

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _consider(a)

    # random Dirichlet
    for _ in range(10):
        a = rng.dirichlet(np.ones(n))
        _consider(a)

    # random Beta‑shaped densities
    xs = (np.arange(n) + 0.5) / n          # mid‑point positions in (0,1)
    for _ in range(30):
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        # log‑pdf ∝ (α‑1)·log(x) + (β‑1)·log(1‑x)
        log_a = (alpha - 1) * np.log(xs) + (beta - 1) * np.log(1 - xs)
        a = np.exp(log_a)
        _consider(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smoothed max‑convolution objective."""
    w = np.log(a0 + 1e-12)          # logits → soft‑max → simplex
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 0.5, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 60000
    it = 0

    while time.time() < deadline:
        # current point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # schedules
        prog = (time.time() - start) / (deadline - start) if deadline > start else 0.0
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        shifted = np.clip(shifted, -700, 700)        # avoid overflow
        weights = np.exp(shifted)
        p = weights / weights.sum()                 # soft‑max over positions (len = 2n‑1)

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)               # length = 3n‑2
        segment = conv_ab[n - 1: 2 * n - 1]         # size n, corresponds to ∂/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the log‑softmax (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # tiny jitter to escape plateaus
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.006, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float,
                               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Classic SA that moves a tiny amount of mass from a “bad” index to a “good” one."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_local_a = a.copy()
    best_local_val = cur_val

    # temperature is a fraction of the current objective and decays linearly
    T0 = cur_val * 0.02                 # 2 % of the starting value
    max_frac_start, max_frac_end = 0.40, 0.001
    IMPROVE_LIMIT = 400_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)          # linear cooling
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # position of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # source: indices that strongly contribute to the peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = rng.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        # target: indices with little contribution
        inv = 1.0 / (contrib + 1e-12)
        inv[i] = 0.0
        probs_tgt = inv / inv.sum()
        j = rng.choice(n, p=probs_tgt)

        if i == j:
            continue

        a_i = a[i]
        delta = a_i * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ---- incremental convolution update (O(n)) ----
        b_old = b.copy()
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij] -= 2.0 * delta * delta

        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[SA  {it:7d}] new best = {best_local_val:.6f}")
        else:
            # revert
            b = b_old
            a[i] = a_i
            a[j] -= delta
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and continue
            print(f"[SA  {it:7d}] perturb after {no_improve} idle steps")
            noise = rng.normal(scale=0.015, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase C – Exponentiated‑gradient descent on a smooth surrogate
# ----------------------------------------------------------------------
def _exp_gradient_phase(a0: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Multiplicative updates (exponentiated gradient) with a decreasing epsilon."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.5 / n        # initial step size (≈ half the average entry)
    eta = eta0
    decay = 0.9999

    eps_start, eps_end = 0.5, 1e-12
    it = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            print(f"[ExpG {it:7d}] new best = {best_local_val:.6f}")

        prog = (time.time() - start) / (deadline - start) if deadline > start else 0.0
        eps = eps_start * (eps_end / eps_start) ** prog

        max_b = b.max()
        shifted = (b - max_b) / eps
        shifted = np.clip(shifted, -700, 700)
        w = np.exp(shifted)
        p = w / w.sum()                     # soft‑max over convolution entries

        # gradient of the smoothed objective (same expression as in Adam)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # multiplicative update then simplex projection
        a *= np.exp(-eta * grad_a)
        a = _project_simplex(a)

        eta *= decay
        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small post‑processing: convex mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try convex combinations (1‑α)·a + α·uniform for a few α."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function under the 1000 s time budget.
    Returns a list of non‑negative floats that (approximately) minimise
    `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a problem size – 9000 is a good trade‑off between expressivity
    # and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # Initialise a – reuse a known construction if available,
    # otherwise fall back to the parametric sweep.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)        # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n, rng)

    best_a = a0.copy()
    best_val = _objective_value(best_a)

    # ------------------------------------------------------------------
    # Phase 0 – quick Beta‑family sweep (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.10 * (deadline - start_time)
    xs = (np.arange(n) + 0.5) / n
    while time.time() < phase0_deadline:
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        log_a = (alpha - 1) * np.log(xs) + (beta - 1) * np.log(1 - xs)
        a = np.exp(log_a)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            continue
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Phase0] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈25 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.25 * (deadline - start_time)
    best_a, best_val = _adam_phase(best_a, n, start_time, phase_a_deadline,
                                   best_a, best_val)

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing hill‑climbing (≈65 % of the budget)
    # ------------------------------------------------------------------
    phase_b_deadline = deadline
    best_a, best_val = _simulated_annealing_phase(
        best_a, n,
        start=start_time,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        rng=rng
    )

    # ------------------------------------------------------------------
    # Phase C – Exponentiated‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _exp_gradient_phase(
        best_a, n,
        start=start_time,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val
    )

    # ------------------------------------------------------------------
    # Final uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
