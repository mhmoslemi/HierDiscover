    start_time = time.time()
    deadline = start_time + budget_s - 0.1

    n = 8000

def active_set_opt(a, n, deadline, target_val=1.503, max_active=30, top_k=3):
    import numpy as np, time
    # ensure normalization
    a = np.asarray(a, dtype=np.float64)
    if a.sum() <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    # compute initial objective
    conv = full_convolution_fft(a)
    maxb = conv.max()
    cur_obj = 2.0 * n * maxb

    # early exit if already good
    if cur_obj <= target_val:
        return a, cur_obj

    # active set: indices of top_k peaks
    active_set = set(np.argpartition(conv, -top_k)[-top_k:].tolist())

    # main loop
    while True:
        now = time.time()
        if now >= deadline:
            break

        # Build constraint matrix for active set
        rows = []
        for k in active_set:
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            idx = np.arange(start, end+1)
            row = np.zeros(n, dtype=np.float64)
            row[idx] = a[k - idx]
            rows.append(row)
        A_ub = np.vstack(rows)
        b_ub = np.full(len(active_set), maxb, dtype=np.float64)

        # Solve LP maximize sum(g)
        c = -np.ones(n, dtype=np.float64)  # minimize -sum(g)
        # If linprog missing, break
        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        except Exception as e:
            print("[LP] failed or not available:", e)
            break

        if not res.success:
            print("[LP] not successful:", res.message)
            break

        g0 = res.x
        if g0.sum() <= 0:
            break
        g = g0 / g0.sum()

        # Check full constraints
        conv_fg = full_convolution_fft(a, g)   # need function that returns convolution of two vectors
        max_fg = conv_fg.max()

        if max_fg <= maxb + 1e-12:
            # feasible globally
            # Evaluate improvement with g alone
            conv_gg = full_convolution_fft(g)
            obj_g = 2.0 * n * conv_gg.max()

            if obj_g < cur_obj - 1e-12:
                # adopt g
                a = g
                maxb = conv_gg.max()
                cur_obj = obj_g
                # reset active set to peaks of new self-convolution
                conv = conv_gg
                active_set = set(np.argpartition(conv, -top_k)[-top_k:].tolist())
                print(f"[ActiveSet] adopt g, obj={cur_obj:.6f}")
                continue

            # try mixing
            improved = False
            for t in (0.5, 0.3, 0.1, 0.03, 0.01):
                a_new = (1.0 - t) * a + t * g
                conv_new = full_convolution_fft(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < cur_obj - 1e-12:
                    a = a_new
                    maxb = conv_new.max()
                    cur_obj = obj_new
                    # update active set based on new conv
                    conv = conv_new
                    active_set = set(np.argpartition(conv, -top_k)[-top_k:].tolist())
                    print(f"[ActiveSet] mixed t={t:.3f}, obj={cur_obj:.6f}")
                    improved = True
                    break
            if not improved:
                # No progress; break
                break
        else:
            # add violating constraint
            violating_k = int(np.argmax(conv_fg))
            if violating_k not in active_set:
                active_set.add(violating_k)
                if len(active_set) > max_active:
                    print("[ActiveSet] active set too large")
                    break
            else:
                # already there, cannot fix
                break

        # check if target reached
        if cur_obj <= target_val:
            break

    return a, cur_obj

    n = len(x)
    if y is None:
        y = x
    out_len = len(x) + len(y) - 1

    conv = np.fft.irfft(fx * fy, size)[:out_len]

"""
propose_candidate – a hybrid optimiser for the “step‑function” challenge.

The search proceeds in three stages, all respecting the 1000 s budget.

1. **Initialisation** – start from the previously best sequence if it is
   available, otherwise from a uniform distribution (or a random Dirichlet).

2. **Active‑set linear‑programming refinement** – for a current distribution
   *f* we compute M = max_k (f*f)_k.  We then solve the linear program

        maximise   Σ_j g_j
        subject to  (f * g)_k ≤ M   for k in a small “active set”
                    g_j ≥ 0

   (the convolution (f*g)_k is linear in g).  The LP is solved with
   ``scipy.optimize.linprog`` – the active set starts from the K largest
   autocorrelation peaks and grows only when a constraint is violated.
   The optimal *g* is normalised to a probability vector and mixed with *f*
   (convex combination, several mixing factors are tried).  If the mixture
   reduces the objective we accept it and recompute the active set.
   This step can dramatically lower the maximal autocorrelation while keeping
   the sum of the coefficients fixed.

3. **Fine‑grained hill‑climb** – after the LP phase we apply a short
   exponentiated‑gradient / hill‑climb routine (similar to the previous
   solution) to polish the result.

All helper functions are top‑level, no closures are used, and the code
uses only ``numpy`` and ``scipy``.  Progress and intermediate objectives are
printed.  The returned list is a non‑negative step function; ``evaluate_sequence``
will return the final (minimal) upper bound.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, b: np.ndarray | None = None) -> np.ndarray:
    """Return the full linear convolution a*b (length len(a)+len(b)-1)."""
    if b is None:
        b = a
    n = a.shape[0]
    m = b.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    n = a.shape[0]
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Exponentiated‑gradient phase (identical to the previous code)
# ----------------------------------------------------------------------
def _expgrad_phase(a0: np.ndarray,
                   n: int,
                   start_time: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   top_k: int = 3,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 250_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # --- aggregated sub‑gradient from the K largest convolution entries ---
        if top_k == 1:
            ks = [int(np.argmax(b))]
        else:
            ks = np.argpartition(b, -top_k)[-top_k:]

        g = np.zeros_like(cur)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * cur[k - idx]

        # linear step‑size schedule
        progress = (now - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005

        cur = cur * np.exp(-eta * g)
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.009, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb (deterministic)
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start_time: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     start_frac: float = 0.30,
                     end_frac: float = 0.001,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # anneal transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        k = int(np.argmax(b))
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev = np.zeros_like(cur)
        rev[idx] = cur[k - idx]

        src = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[src] = np.inf
        tgt = int(np.argmin(rev_tmp))

        if cur[src] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            continue

        cur[src] -= delta
        cur[tgt] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            pass
        else:
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a)
    uniform = np.full_like(a, 1.0 / a.shape[0])
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Active‑set linear‑programming refinement
# ----------------------------------------------------------------------
def _active_set_opt(a: np.ndarray,
                    n: int,
                    deadline: float,
                    target_val: float = 1.503,
                    max_active: int = 30,
                    top_k: int = 3) -> tuple[np.ndarray, float]:
    """Improve ``a`` by solving successive LPs with a small active set."""
    a = np.asarray(a, dtype=np.float64)
    if a.sum() <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    cur_conv = _full_convolution(a)
    maxb = cur_conv.max()
    cur_obj = 2.0 * n * maxb
    if cur_obj <= target_val:
        return a, cur_obj

    # initialise active set with the K largest peaks
    active_set = set(int(k) for k in np.argpartition(cur_conv, -top_k)[-top_k:])

    while True:
        if time.time() >= deadline:
            break

        # --- build constraint matrix for the current active set -----------------
        rows = []
        for k in active_set:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            row = np.zeros(n, dtype=np.float64)
            row[idx] = a[k - idx]      # coefficient = a_{k-i}
            rows.append(row)
        A_ub = np.vstack(rows)                       # shape (|active|, n)
        b_ub = np.full(len(active_set), maxb, dtype=np.float64)

        # --- linear program: maximise Σ g_j  (i.e. minimise -Σ g_j) ------------
        c = -np.ones(n, dtype=np.float64)
        with np.errstate(all='ignore'):
            lp_res = linprog(c,
                             A_ub=A_ub, b_ub=b_ub,
                             bounds=[(0, None)] * n,
                             method='highs',
                             options={'presolve': True})
        if not lp_res.success:
            print("[ActiveSet] LP failure:", lp_res.message)
            break

        g0 = lp_res.x
        if g0.sum() <= 0:
            break

        g = g0 / g0.sum()          # normalise to a probability vector

        # --- check *all* convolution constraints ---------------------------------
        conv_fg = _full_convolution(a, g)
        max_fg = conv_fg.max()

        if max_fg <= maxb + 1e-12:
            # feasible globally – try to replace / mix
            conv_gg = _full_convolution(g)
            obj_g = 2.0 * n * conv_gg.max()

            if obj_g < cur_obj - 1e-12:
                a = g
                maxb = conv_gg.max()
                cur_obj = obj_g
                cur_conv = conv_gg
                active_set = set(int(k) for k in np.argpartition(cur_conv, -top_k)[-top_k:])
                print(f"[ActiveSet] adopt g, obj={cur_obj:.6f}")
                continue

            # try a few convex combinations
            improved = False
            for t in (0.5, 0.3, 0.1, 0.03, 0.01):
                a_new = (1.0 - t) * a + t * g
                a_new /= a_new.sum()
                conv_new = _full_convolution(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < cur_obj - 1e-12:
                    a = a_new
                    maxb = conv_new.max()
                    cur_obj = obj_new
                    cur_conv = conv_new
                    active_set = set(int(k) for k in np.argpartition(cur_conv, -top_k)[-top_k:])
                    print(f"[ActiveSet] mixed (t={t:.3f}), obj={cur_obj:.6f}")
                    improved = True
                    break
            if not improved:
                # no better mixture – we are stuck
                break
        else:
            # constraint violated – add the most offending index
            viol_k = int(np.argmax(conv_fg))
            if viol_k not in active_set:
                active_set.add(viol_k)
                if len(active_set) > max_active:
                    print("[ActiveSet] active set grew too large, stopping.")
                    break
            else:
                # already present – cannot resolve
                break

        if cur_obj <= target_val:
            break

    return a, cur_obj


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising `evaluate_sequence`.
    The method combines (i) a fast active‑set linear‑programming phase,
    (ii) a short exponentiated‑gradient + hill‑climb polishing, and
    (iii) random restarts if time permits.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin
    n = 8000

    # --------------------------------------------------------------
    # 1️⃣   Initial vector (reuse previous construction if present)
    # --------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
    except Exception:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_val = _objective(best_a)
    TARGET = 1.503
    print(f"[Init] objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣   Active‑set LP refinement (takes about 60 % of the budget)
    # --------------------------------------------------------------
    phase0_deadline = start_time + 0.60 * (deadline - start_time)
    best_a, best_val = _active_set_opt(best_a, n, phase0_deadline, target_val=TARGET)
    print(f"[After LP] best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣   Fine‑grained exponentiated‑gradient + hill‑climb
    # --------------------------------------------------------------
    # a short run similar to the previous solver – helps to polish the LP result
    best_a, best_val = _expgrad_phase(
        best_a, n,
        start_time, phase0_deadline,
        best_a, best_val,
        top_k=3, target_val=TARGET)
    best_a, best_val = _hillclimb_phase(
        best_a, n,
        phase0_deadline, deadline,
        best_a, best_val,
        start_frac=0.30, end_frac=0.001,
        target_val=TARGET)
    best_a, best_val = _mix_with_uniform(best_a)

    # --------------------------------------------------------------
    # 4️⃣   Random‑restart loop (uses remaining time)
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    restart_counter = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        slice_budget = max(0.6, 0.008 * remaining)   # at least ~0.6 s per restart
        rs = time.time()
        r_dead = rs + slice_budget

        # random Dirichlet start (log‑uniform concentration)
        log_alpha = rng.uniform(-2, 2)
        alpha = 10.0 ** log_alpha
        a = rng.dirichlet(alpha * np.ones(n))

        # a quick active‑set optimisation (few constraints, very fast)
        a, val = _active_set_opt(a, n, r_dead, target_val=TARGET)

        # optional tiny polishing
        a, val = _mix_with_uniform(a)

        if val < best_val - 1e-12:
            best_a, best_val = a, val
            print(f"[Restart {restart_counter:3d}] new global best = {best_val:.6f}")

        if best_val <= TARGET:
            print("[Target reached] early exit.")
            break

        restart_counter += 1

    # --------------------------------------------------------------
    # Final verification via the supplied secure wrapper
    # --------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
