"""
Propose a step‑function that minimizes the evaluation metric.

The algorithm works in two stages:

1. **Coarse search on a modest size (≈ 4 k).**
   - A sweep of simple parametric families produces a good seed.
   - A short Adam optimisation (smooth log‑sum‑exp surrogate of the max‑convolution)
     refines the seed.

2. **Fine optimisation on a larger size (≈ 15‑20 k).**
   - The coarse solution is interpolated to the larger grid.
   - A longer Adam run (still using the smooth surrogate) drives the objective
     towards the theoretical limit ≈ 1.5.
   - Finally a tiny mixing with the uniform distribution is performed.

All operations respect the simplex (non‑negative, sum = 1) and use FFT‑based
convolutions for speed.  The routine adapts the problem size to the total
time budget (default 1000 s) and returns the best sequence found before the
deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (FFT, real inputs)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (sum of a is 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Parametric families – quick sweep for a good seed
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector found among a handful of analytic families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0

    # 1. Exponential (both one‑sided and symmetric)
    for r in np.linspace(0.98, 0.9999, 12):
        _try(r ** i)                       # decreasing
        _try(r ** i + r ** (n - 1 - i))    # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 5.0, 12):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑shaped densities
    for a_param in np.linspace(0.3, 5.0, 6):
        for b_param in np.linspace(0.3, 5.0, 6):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power (half‑wave)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 8):
        _try(base_sin ** p)

    # 5. Linear “V” shape (higher at the edges)
    a = np.abs(i - (n - 1) / 2.0) + 0.5
    _try(a)

    # 6. Piecewise constant high‑edge shape
    for frac in (0.02, 0.05, 0.1):
        k = int(round(frac * n))
        if k == 0 or 2 * k >= n:
            continue
        high = 0.3
        low = (1.0 - 2 * k * high) / (n - 2 * k)
        a = np.full(n, low)
        a[:k] = high
        a[-k:] = high
        _try(a)

    # 7. Random Dirichlet draws (quick diversity)
    rng = np.random.default_rng(12345)
    for _ in range(20):
        _try(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimisation on a smooth log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, n: int, start: float, deadline: float,
              best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp approximation of the max‑convolution."""
    # initialise logits (soft‑max parametrisation)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    # learning‑rate and smoothing schedules (exponential in time)
    lr_start, lr_end = 0.5, 0.001
    eps_start, eps_end = 1.0, 1e-6

    it = 0
    while time.time() < deadline:
        # point on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            # occasional progress report
            if it % 2000 == 0:
                print(f"[Adam {it:6d}] new best = {best_val:.6f}")

        # schedule (progress ∈ [0,1])
        prog = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        # compute soft‑max weights over convolution positions
        shifted = (b_full - max_b) / eps
        # stabilise exponentials
        shifted -= shifted.max()
        weights = np.exp(shifted)
        p = weights / weights.sum()            # ∑ p = 1

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a 2·n·∑ p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 2n‑1
        segment = conv_ab[n - 1: 2 * n - 1]    # the central n entries
        grad_a = 4.0 * n * segment[::-1]

        # back‑prop through the soft‑max (logits → probabilities)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaux
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.012, size=n)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Very light uniform mixing (final polishing)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with a tiny amount of the uniform distribution."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    # Small stage (fast exploration) ≈ 25 % of time.
    # Large stage (fine optimisation) uses the remaining time.
    # --------------------------------------------------------------
    total_time = deadline - start_time
    stage1_deadline = start_time + 0.25 * total_time

    # small size – fast enough for many Adam iterations
    n_small = 4000 if total_time > 600 else 2500
    print("[Stage 1] small problem size =", n_small)

    # ---- Stage 1: coarse search -----------------------------------
    a_small, _ = _initial_candidate(n_small)
    best_a_small, best_val_small = _adam_opt(
        a_small, n_small,
        start=start_time,
        deadline=stage1_deadline,
        best_a=a_small,
        best_val=_objective_value(a_small),
    )
    print(f"[Stage 1] best objective = {best_val_small:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – up‑sample to a larger grid and continue optimisation.
    # --------------------------------------------------------------
    # Pick a larger n according to remaining budget.
    remaining = deadline - time.time()
    if remaining > 400:
        n_large = 18000
    elif remaining > 250:
        n_large = 15000
    else:
        n_large = 12000
    print("[Stage 2] large problem size =", n_large)

    # Interpolate the small‑scale solution onto the larger grid.
    x_small = np.linspace(0.0, 1.0, len(best_a_small))
    x_large = np.linspace(0.0, 1.0, n_large)
    a_large0 = np.interp(x_large, x_small, best_a_small)
    a_large0 = np.maximum(a_large0, 0.0)
    a_large0 /= a_large0.sum()

    # Continue Adam optimisation on the large grid.
    best_a_large, best_val_large = _adam_opt(
        a_large0, n_large,
        start=time.time(),
        deadline=deadline,
        best_a=a_large0,
        best_val=_objective_value(a_large0),
    )
    print(f"[Stage 2] best objective before mixing = {best_val_large:.6f}")

    # Final tiny uniform mixing (often gives a small extra drop).
    best_a_large, best_val_large = _mix_with_uniform(best_a_large, n_large)
    print(f"[Final] objective after mixing = {best_val_large:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure wrapper (for logging only)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a_large.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a_large.tolist()
