Algorithm
---------
1.  Work on a vector `a` of length `n = 8000`, kept on the probability
    simplex (non‑negative, sum = 1).  The objective for a normalized
    vector reduces to `2·n·max(conv(a,a))`, i.e. minimise the
    peak of the auto‑convolution.

2.  **Phase 0 – deep optimisation**

      *exactly* from *all* convolution entries that are within a tiny
      tolerance of the current maximum.  The gradient is obtained by a
      single FFT‑based convolution (no explicit loops over the active
      set).

      `η≈0.15` down to `5·10⁻⁵`.  After long stretches without improvement
      a small Gaussian jitter is added.

      fraction of the mass from the index that contributes most to the
      active peak(s) to the index that contributes least.  The fraction
      anneals from 30 % to 0.1 % and a move is kept only if the objective
      strictly improves.

    * Repeatedly draw a fresh Dirichlet start (log‑uniform concentration)

      hill‑climb moves.

      0.003, …, 0.02}) – this sometimes reduces the peak a little.

All phases respect the overall time budget (≈ 1000 s).  The code never

# Convolution helpers (FFT based, full linear convolution)

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _gradient_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) when `active` is a boolean mask
    indicating the (near‑maximal) convolution entries.
    Returns a vector of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    # convolution of `active` (0/1) with `a_rev`
    out_len = 3 * n - 2
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(active.astype(np.float64), size)
    fr = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fa * fr, size)[:out_len]
    # the relevant slice is indices (n‑1) … (n‑1+n‑1)
    g = conv[(n - 1):(n - 1 + n)]
    return 2.0 * g                      # factor 2 from the derivative of a*a

# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)        # denominator = 1 because Σa = 1


# ----------------------------------------------------------------------
# Simple candidate families (used when no previous solution is available)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent start (deterministic shapes + a few Dirichlet draws)."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(cand):
        nonlocal best_a, best_val
        cand = np.asarray(cand, dtype=np.float64)
        if cand.ndim != 1 or cand.shape[0] != n:
            return
        if np.any(cand < 0):
            return
        s = cand.sum()
        if s <= 0:
            return
        cand = cand / s
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()

    rng = np.random.default_rng(0)
    idx = np.arange(n, dtype=float)

    # a few handcrafted shapes
    try_update(np.arange(n, 0, -1, dtype=float))               # decreasing
    try_update(np.maximum(idx + 1, n - idx))                  # V‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))       # quadratic decay
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))# half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # a few random Dirichlet draws
    for _ in range(8):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑0 – exponentiated‑gradient with *full* active set
# ----------------------------------------------------------------------
def _expgrad_full_active(a0: np.ndarray,
                         n: int,
                         start_time: float,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) sub‑gradient descent.
    The sub‑gradient is taken from *all* entries that are within a tiny
    tolerance of the current maximum.
    """
    cur = a0.copy()
    rng = np.random.default_rng(12345)
    it = 0
    no_improve = 0
    IMPATIENT = 250_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        # record best
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGradFull {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- active set (all entries within a relative tolerance) ----
        eps = maxb * 1e-8 + 1e-12
        active = b >= (maxb - eps)

        # gradient (sub‑gradient of max_b)
        g = _gradient_active(cur, active)

        # step‑size schedule (linear decay)
        progress = (now - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * 0.15 + progress * 5e-5   # 0.15 → 5e‑5

        # exponentiated update (keeps non‑negativity)
        cur = cur * np.exp(-eta * g)

        # renormalise to Σa = 1
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        # occasional jitter if we seem stuck
        if no_improve > IMPATIENT:
            noise = rng.normal(scale=0.006, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGradFull {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑0 – deterministic hill‑climb (using the full active gradient)
# ----------------------------------------------------------------------
def _hillclimb_full_active(a_start: np.ndarray,
                           n: int,
                           start_time: float,
                           deadline: float,
                           best_a: np.ndarray,
                           best_val: float,
                           start_frac: float = 0.30,
                           end_frac: float = 0.001,
                           target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Move a fraction of the mass from the index contributing most to the
    (near‑maximal) convolution peak(s) to the index contributing least.
    Accept only strictly improving moves.
    """
    cur = a_start.copy()
    rng = np.random.default_rng(9876)
    it = 0
    no_improve = 0
    IMPATIENT = 300_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimbFull {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing of the transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # active set (same tolerance as in the gradient phase)
        eps = maxb * 1e-8 + 1e-12
        active = b >= (maxb - eps)

        # contributions of each coefficient to the *active* peak(s)
        rev = _gradient_active(cur, active) * 0.5   # because grad = 2·rev

        src = int(np.argmax(rev))
        # avoid moving mass from a position that already has (near) zero mass
        if cur[src] <= 1e-15:
            if no_improve > IMPATIENT:
                # jitter to free a stuck source
                noise = rng.normal(scale=0.005, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimbFull {it:7d}] jitter after stagnation")
            it += 1
            continue

        tgt = int(np.argmin(rev))

        delta = cur[src] * frac
        if delta <= 0.0:
            it += 1
            continue

        # trial move
        cur[src] -= delta
        cur[tgt] += delta

        new_val = _objective(cur, n)

        if new_val < cur_val - 1e-15:
            # accept
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny mixing with the uniform vector (helps a little at the end)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Test a few convex combos (1‑α)·a + α·uniform."""
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Fast exponentiated‑gradient (top‑K) used in cheap restarts
# ----------------------------------------------------------------------
def _expgrad_topk(a0: np.ndarray,
                  n: int,
                  start_time: float,
                  deadline: float,
                  best_a: np.ndarray,
                  best_val: float,
                  top_k: int = 3,
                  target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Multiplicative update that only looks at the `top_k` largest peaks."""
    cur = a0.copy()
    rng = np.random.default_rng(5555)
    it = 0
    no_improve = 0
    IMPATIENT = 150_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            if cur_val <= target_val:
                break

        else:
            no_improve += 1

        # pick the indices of the K largest convolution entries
        if top_k == 1:
            ks = [int(np.argmax(b))]
        else:
            ks = np.argpartition(b, -top_k)[-top_k:]

        g = np.zeros_like(cur)
        for k in ks:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * cur[k - idx]

        # step‑size schedule (linear decay)
        progress = (now - start_time) / (deadline - start_time)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005

        cur = cur * np.exp(-eta * g)
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        if no_improve > IMPATIENT:
            noise = rng.normal(scale=0.008, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap hill‑climb (same logic as the full version but with the top‑K gradient)
# ----------------------------------------------------------------------
def _hillclimb_simple(a_start: np.ndarray,
                      n: int,
                      start_time: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      start_frac: float = 0.25,
                      end_frac: float = 0.001,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    rng = np.random.default_rng(7777)
    it = 0
    no_improve = 0
    IMPATIENT = 200_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        maxb = b.max()
        cur_val = 2.0 * n * maxb

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            if cur_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing of the transfer fraction
        progress = (now - start_time) / (deadline - start_time)
        frac = start_frac * (1.0 - progress) + end_frac * progress

        # active index (single peak is enough for the cheap version)
        k = int(np.argmax(b))

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev = np.zeros_like(cur)
        rev[idx] = cur[k - idx]

        src = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[src] = np.inf
        tgt = int(np.argmin(rev_tmp))

        if cur[src] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = rng.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
            it += 1
            continue

        delta = cur[src] * frac
        if delta <= 0.0:
            it += 1
            continue

        cur[src] -= delta
        cur[tgt] += delta

        new_val = _objective(cur, n)

        if new_val < cur_val - 1e-15:
            # keep the move
            pass
        else:
            # revert
            cur[src] += delta
            cur[tgt] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine uses one long, high‑quality
    optimisation phase followed by many cheap random restarts.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 0.4

    n = 8000                      # length of the vector

    # ------------------------------------------------------------------
    # Initialise with the previous best if it exists; otherwise fall back.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    TARGET = 1.503
    print(f"[Init] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑0 : deep optimisation (full active‑set gradient)
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.35 * (deadline - start_time)

    best_a, best_val = _expgrad_full_active(
        best_a, n,
        start_time,
        phase0_deadline,
        best_a,
        best_val,
        target_val=TARGET
    )
    best_a, best_val = _hillclimb_full_active(
        best_a, n,
        phase0_deadline,
        deadline,
        best_a,
        best_val,
        start_frac=0.30,
        end_frac=0.001,
        target_val=TARGET
    )
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Phase‑0] best after deep optimisation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑1 : many cheap random restarts
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    restart_counter = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        # each restart gets at most ~0.8 % of the remaining budget,
        # but never less than ~0.6 s (so it actually runs)
        slice_budget = max(0.6, 0.008 * remaining)

        rs_start = time.time()
        rs_deadline = rs_start + slice_budget

        # random Dirichlet start (log‑uniform concentration)
        log_alpha = rng.uniform(-2, 2)           # α ∈ [10⁻², 10²]
        alpha = 10.0 ** log_alpha
        a = rng.dirichlet(alpha * np.ones(n))

        # short exponentiated‑gradient (top‑K) – cheap but useful
        a, _ = _expgrad_topk(
            a, n,
            rs_start,
            rs_deadline,
            best_a,
            best_val,
            top_k=3,
            target_val=TARGET
        )

        # brief hill‑climb
        a, _ = _hillclimb_simple(
            a, n,
            rs_start,
            rs_deadline,
            best_a,
            best_val,
            start_frac=0.25,
            end_frac=0.001,
            target_val=TARGET
        )

        # tiny uniform mixing
        a, val = _mix_with_uniform(a, n)

        # update global best
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Restart {restart_counter:3d}] new global best = {best_val:.6f}")

        if best_val <= TARGET:
            print("[Target reached] exiting early.")
            break

        restart_counter += 1

    # ------------------------------------------------------------------
    # Final verification via the supplied secure wrapper (optional)
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {reported:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return plain Python list (will be summed to ≈1)
    return best_a.tolist()
