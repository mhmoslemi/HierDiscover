def propose_candidate(seed=42, budget_s=1000):
    """
    Gradient‑descent optimiser for the min‑max auto‑convolution problem.
    ...

    Returns list of floats (Σa≈1) that minimises evaluate_sequence.
    """
    import time, numpy as np

    # constants
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000  # length of step function

    # size for FFT (next power of two)
    conv_len = 2 * n - 1
    revconv_len = 3 * n - 2
    size_fft = 1 << ((max(conv_len, revconv_len) - 1).bit_length())

    # ---- helper functions -------------------------------------------------
    def objective(a):
        """3‑line internal evaluation: 2·n·max(conv(a,a))/(sum(a)**2)."""
        A = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(A * A, size_fft)[:conv_len]
        maxc = conv.max()
        s = a.sum()
        if s < 1e-12:
            return np.inf
        return float(2.0 * n * maxc / (s * s)), conv

    # ---- initialise -------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        # fallback – symmetric exponential
        r = 0.997
        a0 = np.power(r, np.arange(n, dtype=np.float64))
        a0 = a0 + np.power(r, np.arange(n-1, -1, -1, dtype=np.float64))
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

    x = np.log(a0 + 1e-30)               # soft‑max parameters
    best_a = a0.copy()

    print(f"[Init] n={n} objective={best_val:.6f}")

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.05

            x += jitter
            no_improve = 0
            print(f"[Iter {it:07d}] jitter (stagnation)")

    # Final deterministic polishing – tiny uniform mixing (same as earlier)
    # ----------------------------------------------------------------------

        best_local_val = best_val
        uniform = np.full_like(a, 1.0 / n)

                best_local = cand

        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:  # pragma: no cover

4. obtain the gradient of this smooth objective w.r.t. the soft‑max parameters,
5. perform an Adam update on the parameters,
6. periodically add a small random jitter if no improvement is seen.

# Helper: evaluate the true objective (identical to evaluate_sequence when Σa=1)

def _objective(a: np.ndarray, n: int) -> float:
