def grad_max_convolution(a, k):
    n = len(a)
    grad = np.zeros_like(a)
    low = max(0, k-(n-1))
    high = min(n-1, k)

    grad[i_range] = 2 * a[j_range]

"""
Fast multi‑phase heuristic for the min‑max self‑convolution problem.

The algorithm proceeds in three stages:

1.  **Parametric sweep** – a collection of hand‑crafted families (exponential,
    inverse‑power, sine‑power, half‑sine) and a handful of random draws are
    evaluated on a fixed length (default 8000).  The best vector is kept.
2.  **Projected sub‑gradient descent** – starting from the best vector,
    we repeatedly minimise the exact objective
        f(a)=2·n·max(conv(a,a))
    using the sub‑gradient of the maximal convolution entry and projection
    onto the probability simplex.  The step size decays geometrically and the
    loop runs until the time budget is exhausted.
3.  **Uniform mixing** – as a final touch a tiny convex combination with the
    uniform distribution is tried; if it improves the objective the new
    vector is kept.

All vectors are kept non‑negative and normalised (sum = 1); the objective is
scale‑invariant, so the normalisation does not affect the score.  The routine
honours the 1000 s time budget and returns the best sequence it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of a with itself via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector a."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _grad_max_convolution(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)_k at the currently maximal index k_max."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k_max - i
    # derivative of a_i a_j appears twice except when i == j
    grad[i] = 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
# Small helper to evaluate and possibly improve a candidate by mixing with uniform
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Try a few tiny convex combinations with the uniform distribution."""
    best_a, best_val = a, _objective_value(a)
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list of floats that (approximately) minimise `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5
    total_budget = deadline - start_time

    # ------------------------------------------------------------------
    # Problem size – 8000 is a good compromise between expressiveness
    # and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 1) Parametric sweep (≈ 20 % of the total budget)
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 0.20 * total_budget
    best_a = np.full(n, 1.0 / n)          # start with uniform
    best_val = _objective_value(best_a)

    # ---- helper for evaluating a candidate and updating the best ----------
    def _try_candidate(a):
        nonlocal best_a, best_val
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[sweep] new best = {best_val:.6f}")

    # ----- 1a) Exponential U‑shaped families -----
    for r in np.geomspace(0.001, 0.999, 30):
        if time.time() >= sweep_deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        a = np.power(r, idx) + np.power(r, n - 1 - idx)
        _try_candidate(a)

    # ----- 1b) Inverse‑power U‑shaped families -----
    pos = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(0.1, 5.0, 25):
        if time.time() >= sweep_deadline:
            break
        a = pos ** (-p) + (n + 1 - pos) ** (-p)
        _try_candidate(a)

    # ----- 1c) Sine‑power families (central dip) -----
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    base = np.sin(theta)                     # values in (0,1)
    for p in np.linspace(0.1, 8.0, 25):
        if time.time() >= sweep_deadline:
            break
        a = base ** p
        _try_candidate(a)

    # ----- 1d) Half‑sine (already known to be decent) -----
    if time.time() < sweep_deadline:
        a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1)))
        _try_candidate(a)

    # ----- 1e) Random draws (Dirichlet‑like) -----
    for _ in range(120):
        if time.time() >= sweep_deadline:
            break
        # raise uniform random numbers to a random power to obtain varied sparsity
        power = rng.uniform(0.2, 4.0)
        a = rng.random(n) ** power
        _try_candidate(a)

    # ------------------------------------------------------------------
    # 2) Projected sub‑gradient descent (≈ 75 % of the remaining budget)
    # ------------------------------------------------------------------
    sg_deadline = start_time + 0.95 * total_budget  # leave a little for final mix
    eta0 = 0.5 / n          # initial step size (≈ half the average entry)
    eta = eta0
    decay = 0.9997          # gentle decay
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 300_000

    a = best_a.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    while time.time() < sg_deadline:
        # compute full convolution and current objective
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        k_max = int(conv.argmax())

        # Update best if we made progress
        if cur_val < best_local_val - 1e-14:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[sgd {it:7d}] new best = {best_local_val:.6f}")
        else:
            no_improve += 1

        # Sub‑gradient of the active convolution entry
        grad = _grad_max_convolution(a, k_max)

        # Gradient step + projection onto the simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # Decay step size
        eta *= decay

        # If we stagnate, inject a tiny amount of noise and reset step size
        if no_improve > IMPROVE_LIMIT:
            noise = rng.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[sgd {it:7d}] perturb after stagnation")

        it += 1

    # Record the best found during the sub‑gradient phase
    best_a, best_val = best_local_a, best_local_val

    # ------------------------------------------------------------------
    # 3) Final tiny uniform mixing (often yields a marginal improvement)
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a)

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
