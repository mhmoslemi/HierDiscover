"""
Fast heuristic for the min‑max self‑convolution problem.

The algorithm proceeds in three stages:
  1. *Parameter sweep* – explore simple one‑parameter families
     (monotone exponentials, power‑law decays, mixtures with the uniform
     distribution) using a fast FFT‑based convolution to obtain a good
     starting point.
  2. *Simulated‑annealing refinement* – hill‑climb from the best candidate.
     Moves transfer a random fraction of mass from a source index to a
     target index while updating the full convolution incrementally
     (O(n) per trial).  A temperature schedule lets occasional uphill
     moves be accepted, and a restart mechanism avoids stagnation.
  3. *Final check* – evaluate the best sequence with the supplied
     ``evaluate_sequence`` function and return it.

The whole routine respects the 1000 s time budget; intermediate progress
is printed to aid debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself, length 2·len(a)‑1, using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simple parametric families (used for a quick initial search)
# ----------------------------------------------------------------------
def _exp_decay(n: int, r: float) -> np.ndarray:
    """Monotone decreasing exponential a_i ∝ r**i, 0<r<1."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    """Power‑law decreasing a_i ∝ (i+1)^{-p}, p>0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    a /= a.sum()
    return a


def _mix_with_uniform(a: np.ndarray, weight: float) -> np.ndarray:
    """Convex combination (1‑weight)·a + weight·uniform, then renormalise."""
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    mixed = (1.0 - weight) * a + weight * uniform
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (incremental convolution updates)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         start: float,
                         deadline: float,
                         rng: np.random.Generator,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Hill‑climbing with temperature; returns best (a, value) found."""
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_a = a.copy()
    best_val = cur_val

    # initial temperature – 2 % of the starting value
    T0 = cur_val * 0.02
    no_improve = 0
    IMP_LIMIT = 250_000
    it = 0

    while time.time() < deadline:
        prog = (time.time() - start) / (deadline - start)
        temperature = T0 * (1.0 - prog)          # linear cooling

        # index where the convolution attains its current maximum
        k_max = int(np.argmax(b))

        # contributions of each i to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # source index – either weighted by its contribution to the max entry
        # or uniform (to keep exploration)
        if total > 0 and rng.random() < 0.9:
            probs_src = contrib / total
            i = rng.choice(n, p=probs_src)
        else:
            i = rng.integers(n)

        # target index – uniformly chosen different from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 1e-12:
            continue

        # move a random fraction (up to 30 %) of a[i] to a[j]
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        # ----- incremental update of the full convolution -----
        b_old = b.copy()
        # linear part: +2·δ·a shifted by j, –2·δ·a shifted by i
        b[j:j + n] += 2.0 * delta * a
        b[i:i + n] -= 2.0 * delta * a
        # quadratic part:  +δ² at 2j and 2i,  –2δ² at i+j
        two_j = 2 * j
        two_i = 2 * i
        ij = i + j
        b[two_j] += delta * delta
        b[two_i] += delta * delta
        b[ij]   -= 2.0 * delta * delta

        # apply the mass transfer
        a[i] -= delta
        a[j] += delta

        new_val = 2.0 * n * b.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            if temperature > 0.0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SA   {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # revert everything
            b = b_old
            a[i] += delta
            a[j] -= delta
            no_improve += 1

        # restart if we have been idle for a long time
        if no_improve > IMP_LIMIT:
            print(f"[SA   {it:7d}] restart after {no_improve} idle steps")
            noise = rng.normal(scale=0.012, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the 1000 s budget.

    Returns a list of floats (the sequence of heights).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # tiny safety margin for the final call to evaluate_sequence
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 9000 is a good compromise between expressivity
    # and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 9000

    # ------------------------------------------------------------------
    # 1. Quick parametric sweep to obtain a solid starting point.
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)                 # uniform
    best_val = _objective(best_a)
    print("[init] uniform value =", best_val)

    # exponential decay (monotone)
    r_vals = np.concatenate((
        np.linspace(0.90, 0.99, 15, endpoint=False),
        np.linspace(0.99, 0.9999, 25, endpoint=False),
        np.linspace(0.9999, 0.999999, 25)
    ))
    for r in r_vals:
        a = _exp_decay(n, r)
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[exp   ] r={r:.6f} val={val:.6f}")

        # also try a slight uniform mixing (helps stability)
        a_mix = _mix_with_uniform(a, weight=0.07)
        val = _objective(a_mix)
        if val < best_val:
            best_val, best_a = val, a_mix.copy()
            print(f"[expMix] r={r:.6f} val={val:.6f}")

    # power‑law decay
    p_vals = np.linspace(0.1, 5.0, 30)
    for p in p_vals:
        a = _power_law(n, p)
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[pow   ] p={p:.3f} val={val:.6f}")

        a_mix = _mix_with_uniform(a, weight=0.07)
        val = _objective(a_mix)
        if val < best_val:
            best_val, best_a = val, a_mix.copy()

    # 3. Final report (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
