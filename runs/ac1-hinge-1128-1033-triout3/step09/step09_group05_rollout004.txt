Implementation of projection onto simplex:

def project_simplex(v):
    """Project vector v onto the simplex {x >=0, sum x = 1}."""
    # algorithm: sort v descending, find rho = max{i | v_i - (cumsum(v_i) - 1)/ (i+1) > 0}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Works for n >0.

def _project_simplex(v):
    # ensure non-negative and sum to 1 after projection.
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort w descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if rho.size == 0:
        # all <= threshold, return uniform?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # due to floating errors, ensure sum is 1
    if w.sum() == 0:
        w = np.full(n, 1.0/n)
    else:
        w /= w.sum()
    return w

import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution (a * a) via FFT
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a normalised vector."""
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return uniform
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)

    # Renormalise (to correct tiny rounding errors)
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Parametric families for Phase‑0 (random exploration)
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power(n: int, p: float) -> np.ndarray:
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)                 # stabilise
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Phase‑0 – random exploration of rich parametric families
# ----------------------------------------------------------------------
def _phase0_random(n: int, start_time: float, deadline: float,
                   best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng(123456)
    while time.time() < deadline:
        choice = rng.integers(0, 5)
        if choice == 0:                               # pure exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_exp(n, r)
        elif choice == 1:                             # reversed exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_rev_exp(n, r)
        elif choice == 2:                             # mixture of two exponentials
            r1 = rng.uniform(0.90, 0.99999)
            r2 = rng.uniform(0.90, 0.99999)
            w = rng.random()
            cand = _gen_mix_exp(n, r1, r2, w)
        elif choice == 3:                             # pure power‑law
            p = rng.uniform(-5.0, 5.0)
            cand = _gen_power(n, p)
        else:                                         # power‑law × exponential
            p = rng.uniform(-5.0, 5.0)
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_power_exp(n, p, r)

        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Phase‑0] new best = {best_val:.6f}")

        # occasional Dirichlet draws (low concentration -> sparse)
        if rng.random() < 0.03:
            conc = rng.uniform(0.05, 5.0)
            cand = rng.dirichlet(np.full(n, conc))
            val = _objective(cand, n)
            if val < best_val:
                best_val, best_a = val, cand.copy()
                print(f"[Phase‑0 Dir] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑1 – Smoothed (log‑sum‑exp) gradient descent with Adam
# ----------------------------------------------------------------------
def _phase1_sgd(a0: np.ndarray, n: int,
                start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)          # first‑moment accumulator
    v = np.zeros_like(a)          # second‑moment accumulator
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta = 0.05                     # base step size for Adam
    beta_min, beta_max = 0.5, 20.0
    size_corr = 1 << ((3 * n - 2).bit_length())   # FFT size for correlation

    it = 0
    no_improve = 0
    IMPATIENT = 4000               # trigger jitter after this many non‑improves

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # smoothing parameter β – increase linearly with progress
        progress = (now - start) / (deadline - start)
        beta = beta_min + progress * (beta_max - beta_min)

        # smoothed weights (log‑sum‑exp)
        shift = max_b                              # avoid overflow
        exp_vals = np.exp(beta * (b - shift))
        w = exp_vals / exp_vals.sum()             # shape (2n‑1,)

        # gradient g = 2 * (a_rev * w)_{n‑1 .. n‑1+n-1}
        a_rev = a[::-1]
        fa = np.fft.rfft(a_rev, size_corr)
        fw = np.fft.rfft(w, size_corr)
        conv_rev = np.fft.irfft(fa * fw, size_corr)[:3 * n - 2]
        g = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = _project_simplex(a_new)

        # jitter if we have been stuck for a while
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.004, size=n)
            a = _project_simplex(a + noise)
            no_improve = 0
            print(f"[SGD {it:7d}] jitter after stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑2 – Deterministic greedy hill‑climbing
# ----------------------------------------------------------------------
def _phase2_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 3000
    max_frac_start, max_frac_end = 0.6, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Hill {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index of the maximal convolution entry
        k = int(np.argmax(b))

        # build the relevant reversed fragment (rev[i] = a_{k-i} if defined)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        rev = np.zeros(n, dtype=np.float64)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        # choose donor (largest contribution) and receiver (smallest)
        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = -np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            # donor exhausted – use jitter if we are stuck
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = _project_simplex(cur + noise)
                no_improve = 0
                print(f"[Hill {it:7d}] jitter after stagnation")
            it += 1
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            it += 1
            continue

        cur[i] -= delta
        cur[j] += delta
        # ensure feasibility
        cur = _project_simplex(cur)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑3 – final tiny uniform mixing
# ----------------------------------------------------------------------
def _phase3_mix_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
