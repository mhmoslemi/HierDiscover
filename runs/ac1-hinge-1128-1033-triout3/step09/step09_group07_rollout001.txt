"""
Mini‑batch mirror‑descent for the max‑convolution objective.

The true objective is
    obj(a) = 2·n·max(a*a) / (sum(a)²)          (a ≥ 0, a ≠ 0)

The value is invariant under scaling, so we keep the vector on the
probability simplex (sum = 1).  For a given `a` we compute the
self‑convolution with an FFT (O(n log n)).  The sub‑gradient of the
max‑convolution is obtained from the index `p` where the convolution
attains its maximum:

    ∂/∂a_i max(a*a) = 2·a_{p-i}   for the indices that actually
    contribute to the maximum, otherwise 0.

We perform a multiplicative (exponential‑mirror) update
    a ← a·exp(-α·g)   and renormalise.  The step size α decays as
    α = α₀ / sqrt(t+1) so the algorithm behaves like a sub‑gradient
    method.

A few random restarts, occasional anti‑symmetric re‑ordering and a
lightweight hill‑climbing phase (swaps / tiny mass transfers) are
added to escape plateaus.  The expensive `evaluate_sequence` is called
only for logging; the inner loop works with a cheap FFT‑based
approximation of the objective.

The whole routine respects the 1000 s budget and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear self‑convolution `c = x * x` (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Remove tiny negative artefacts caused by floating point errors
    np.maximum(conv, 0.0, out=conv)
    return conv


def _approx_obj(a: np.ndarray) -> float:
    """Fast surrogate of `evaluate_sequence` using FFT‑based max‑convolution."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return math.inf
    return 2.0 * n * max_c / (s * s)


def _gradient_of_max(a: np.ndarray, p: int) -> np.ndarray:
    """
    Sub‑gradient of `max(a*a)` at the current arg‑max position `p`.
    Returns g_i = 2·a_{p‑i} for all indices that actually appear
    in the sum defining the convolution at `p`, zero elsewhere.
    """
    n = a.shape[0]
    i_low = max(0, p - (n - 1))
    i_high = min(p, n - 1)
    # indices that contribute
    indices = np.arange(i_low, i_high + 1)
    g = np.zeros_like(a)
    # a_{p-i} for every valid i
    g[i_low:i_high + 1] = 2.0 * a[p - indices]
    return g


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Put the largest values opposite the smallest ones (pairwise).
    This often reduces the peak of the self‑convolution.
    """
    s = np.sort(a)          # ascending
    n = a.shape[0]
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # big → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------

    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` within the allotted time budget.

    The method consists of:
      * initialisation (historic sequence → interpolation, else random Dirichlet)
      * several cheap random restarts
      * mirror‑descent on the max‑convolution (sub‑gradient from the current peak)
      * occasional anti‑symmetric re‑ordering to shake the solution
      * a short stochastic hill‑climbing refinement (swaps / tiny mass moves)
    """
    # ------------------------------------------------------------------
    # General set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a few seconds margin for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    # Problem size – a moderate size gives a good balance.
    # ------------------------------------------------------------------
    n = 8000                     # can be changed; 8000 works well on the reference run

    # ------------------------------------------------------------------
    # Initialise from the historic construction if present, otherwise random.
    # ------------------------------------------------------------------
