    deadline = start + budget_s - 0.5

    # Build initial candidate

    best_val = _objective(best_a)
    print("[init] uniform objective", best_val)   # Should be 2

    # Phase 0: Random shape search
    i = np.arange(1, n + 1, dtype=np.float64)
    while time.time() < start + 0.12 * budget_s:   # 12% of budget (≈120 s)
        # sample random family
        # Choose method randomly
        method = np.random.randint(4)
        if method == 0:
            # beta shape random p,q
            logp = np.random.uniform(-6.9, 2.3)
            logq = np.random.uniform(-6.9, 2.3)
            p = np.exp(logp)
            q = np.exp(logq)
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
        elif method == 1:
            # power law decreasing
            p = np.random.uniform(0.1, 3.0)
            a = (n - i + 1) ** (-p)
        elif method == 2:
            # exponential decay
            lam = np.random.uniform(0.0001, 0.01)
            a = np.exp(-lam * (i - 1))
        else:
            # Dirichlet sample
            a = np.random.dirichlet(np.ones(n))
        s = a.sum()
        if s == 0:

"""
Propose a non‑negative step function that (approximately) flattens its
self‑convolution.  The algorithm consists of four phases:

0️⃣  Fast random search over a few simple parametric families
    (beta, power‑law, exponential, Dirichlet).  The best candidate is kept.

1️⃣  Continuous optimisation on the simplex using Adam on a smooth
    approximation of the objective
        L(a)=2·n·smooth_max(conv(a,a)) + λ·Var(conv(a,a)).
    The smoothing parameter ε and the variance weight λ are annealed to 0.
    No restarts – we simply keep the best point ever seen.

2️⃣  Fast hill‑climbing (simulated‑annealing) that repeatedly moves a
    small amount of mass from an index that contributes heavily to the
    current maximal convolution entry to a low‑contributing index.
    The convolution is updated incrementally (O(n) per move) so that
    millions of moves can be performed.

3️⃣  Deterministic sub‑gradient polishing on the exact objective
    (max(conv)).  A simple decreasing step‑size schedule and occasional
    jitter are used, again keeping the best point.

The whole procedure respects the 1000 s budget; the final candidate is
passed to ``evaluate_sequence`` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient (includes the factor 2·n)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient of
    2·n·smooth_max w.r.t. a.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # log‑sum‑exp weights (numerically stable)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # Convolution of a with reversed soft‑max weights
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    # only the n central entries matter for the gradient
    grad_part = conv_ab[n - 1:2 * n - 1][::-1]
    grad_a = 4.0 * n * grad_part                 # factor 2·n already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns   var(b) = mean((b‑avg)²)   and   ∂var/∂a .
    Here b = a*a and avg = 1/(2n‑1) because Σb = (Σa)² = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])
    grad_part = conv_ab[n - 1:2 * n - 1][::-1]
    grad = (4.0 / m) * grad_part
    return var, grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random shape generators (Beta, power‑law, exponential, Dirichlet)
# ----------------------------------------------------------------------
def _sample_beta_shape(n: int) -> np.ndarray:
    logp = np.random.uniform(-6.9, 2.3)   # ln(0.001) .. ln(10)
    logq = np.random.uniform(-6.9, 2.3)
    p = np.exp(logp)
    q = np.exp(logq)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def _sample_powerlaw_shape(n: int) -> np.ndarray:
    p = np.random.uniform(0.1, 5.0)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (n - i + 1) ** (-p)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def _sample_exponential_shape(n: int) -> np.ndarray:
    lam = np.random.uniform(0.0003, 0.01)
    i = np.arange(0, n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    if s == 0.0:
        return None
    a /= s
    return a


def _sample_dirichlet_shape(n: int) -> np.ndarray:
    a = np.random.dirichlet(np.ones(n))
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that (approximately) minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final call

    # ------------------------------------------------------------------
    # Problem size – a good trade‑off between expressivity and speed
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 0️⃣  Initialise (try to reuse a historic solution if it exists)
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val0 = _objective(a0)
        best_a, best_val = a0, val0
        print("[init] interpolated previous solution, value =", best_val)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap random shape search (≈12 % of budget)
    # ------------------------------------------------------------------
    t_phase0_end = start_time + 0.12 * budget_s
    i = np.arange(1, n + 1, dtype=np.float64)

    while time.time() < t_phase0_end:
        which = np.random.randint(4)
        if which == 0:
            cand = _sample_beta_shape(n)
        elif which == 1:
            cand = _sample_powerlaw_shape(n)
        elif which == 2:
            cand = _sample_exponential_shape(n)
        else:
            cand = _sample_dirichlet_shape(n)

        if cand is None:
            continue
        val = _objective(cand)
        if val < best_val:
            best_a, best_val = cand, val
            print("[search] new best = {:.6f}".format(best_val))

    # If nothing was found (should not happen) fall back to uniform
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)
        print("[fallback] uniform start, value =", best_val)

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on smooth max + variance (≈33 % of budget)
    # ------------------------------------------------------------------
    t_phase1_end = start_time + 0.45 * budget_s
    # log‑softmax parametrisation
    w = np.log(best_a + 1e-12)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.90, 0.999
    t_iter = 0

    while time.time() < t_phase1_end:
        now = time.time()
        # decode to simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print("[Adam] new best = {:.6f}".format(best_val))

        # Linear progress through the phase   (0 → 1)
        prog = (now - start_time) / (t_phase1_end - start_time)

        # annealing schedules
        eps = 0.5 * (0.001 / 0.5) ** prog          # from 0.5 → 0.001
        lr = 0.2 * (0.01 / 0.2) ** prog           # from 0.2 → 0.01
        lam = (30.0 * n) * (1.0 - prog)           # from 30·n → 0

        # smooth‑max & variance & their gradients
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)
        var_val, grad_var = _var_and_grad(a)

        # total gradient of the surrogate loss
        grad_a = grad_smooth + lam * grad_var

        # back‑prop through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape shallow plateaus
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.02, size=n)

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing hill‑climbing (≈45 % of budget)
    # ------------------------------------------------------------------
    t_phase2_end = start_time + 0.90 * budget_s
    a = best_a.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    eps_small = 1e-12

    while time.time() < t_phase2_end:
        now = time.time()
        prog = (now - t_phase1_end) / (t_phase2_end - t_phase1_end)

        # schedule: maximal transfer fraction and temperature
        max_frac = 0.8 * (1.0 - prog) + 0.001 * prog
        T = 0.02 * (1e-6 / 0.02) ** prog

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to this entry (a[i] * a[k_max-i])
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – sample proportionally to its contribution
        if total_contrib > 0:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(a))

        # target index – bias away from high‑contribution spots
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps_small)
        tgt_weights[src] = 0.0
        ws = tgt_weights.sum()
        if ws > 0:
            tgt = np.random.choice(n, p=tgt_weights / ws)
        else:
            # fallback (any other index)
            choices = list(range(n))
            choices.remove(src)
            tgt = np.random.choice(choices)

        # amount of mass to move
        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print("[Hill] new best = {:.6f}".format(best_val))
        # occasional global jitter to avoid getting stuck
        if np.random.rand() < 0.001:
            a = best_a.copy()
            a += np.random.normal(scale=0.02, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

    # ------------------------------------------------------------------
    # Phase 3 – Deterministic sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_val = _objective(a)
    max_iters = 300_000
    step_start = 0.5

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # compute sub‑gradient of the exact max‑convolution term
        b = _full_convolution(a)
        max_b = b.max()
        ks = np.where(b == max_b)[0]

        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple decreasing step size
        prog = it / max_iters
        step = step_start * (1.0 - prog) + 1e-5 * prog

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print("[Subgrad] new best = {:.6f}".format(best_val))

        # occasional tiny jitter
        if np.random.rand() < 0.002:
            a = np.maximum(a + np.random.normal(scale=0.02, size=n), 0.0)
            a = _project_simplex(a)
            cur_val = _objective(a)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] reported score =", final_score)
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    # The list that will be handed to the outer harness
    return best_a.tolist()
