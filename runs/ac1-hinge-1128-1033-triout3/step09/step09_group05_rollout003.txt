"""
sidon_optimizer.py

Proposes a step‑function whose non‑zero locations form a *Sidon set*
(a B₂‑set: every sum i+j is unique).  With such a set of size m the
discrete convolution a∗a contains at most one product term at each
position, so max(a∗a)= (1/m)² when the non‑zero entries are weighted
uniformly.  The evaluation function therefore equals

        2·n·max(a∗a) = 2·n / m² .

Choosing n≈8000 gives the theoretical bound m≈√(2n)≈126, which makes the
objective close to 1.0 – well below the target 1.503.  The algorithm
searches for a large Sidon set by repeatedly running a cheap random‑greedy
construction; the best set found is turned into a candidate sequence.
Only a single call to ``evaluate_sequence`` is made at the end.

The whole routine respects the 1000 s time budget and returns a list of
non‑negative floats (Σa = 1).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: greedy construction of a B₂‑set (Sidon set) given an order.
# ----------------------------------------------------------------------
def greedy_sidon(order: np.ndarray, n: int) -> list[int]:
    """
    Build a Sidon set using the supplied order of indices.
    The returned list contains the chosen positions (0‑based).
    """
    chosen = []                     # positions already in the set
    used_sums = set()               # sums i+j that are already taken

    for i in order:
        # Self‑sum must be unused
        if (i + i) in used_sums:
            continue

        # Check sums with all previously chosen positions
        conflict = False
        for s in chosen:
            if (i + s) in used_sums:
                conflict = True
                break
        if conflict:
            continue

        # No conflict – add i to the Sidon set
        for s in chosen:
            used_sums.add(i + s)    # the sum i+s (s+i) is now occupied
        used_sums.add(i + i)        # also occupy the self‑sum
        chosen.append(i)

    return chosen


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a large Sidon set and return a uniform distribution on it.
    The algorithm spends almost the whole budget on many random greedy tries
    and finally returns the best vector it found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # Problem size – 8000 works well and matches the previous experiments.
    n = 8000

    rng = np.random.default_rng(seed + 12345)

    best_set: list[int] = []
    best_m = 0
    attempts = 0

    # ------------------------------------------------------------------
    # Random‑greedy search for a maximal Sidon set
    # ------------------------------------------------------------------
    while time.time() < deadline - 0.2:      # keep a little time for final eval
        # a random permutation of the indices 0 … n‑1
        order = rng.permutation(n)
        cand_set = greedy_sidon(order, n)
        m = len(cand_set)

        if m > best_m:
            best_set = cand_set
            best_m = m
            attempts += 1
            # theoretical objective value for this candidate (no need to convolve)
            approx_obj = 2.0 * n / (m * m)
            print(f"[Attempt {attempts:4d}] size={m:3d}, approx objective={approx_obj:.6f}")

    # ------------------------------------------------------------------
    # Build the final sequence: uniform weight on the chosen indices
    # ------------------------------------------------------------------
    a = np.zeros(n, dtype=float)

    if best_m == 0:                     # fallback – should never happen
        a[:] = 1.0 / n
        final_obj = 2.0                 # uniform objective
        print("[Fallback] using uniform distribution.")
    else:
        weight = 1.0 / best_m
        a[best_set] = weight
        final_obj = 2.0 * n * (weight * weight)   # max_b = weight²
        print(f"[Result] n={n}, m={best_m}, final objective≈{final_obj:.6f}")

    # ------------------------------------------------------------------
    # Verify with the provided security‑wrapped evaluator
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:            # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return a.tolist()
