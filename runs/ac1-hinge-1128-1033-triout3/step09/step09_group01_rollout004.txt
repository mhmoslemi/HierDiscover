def _expgrad_opt(a0, n, deadline, start_time, best_a, best_val):
    a = a0.copy()
    best_val = best_val
    best_a = best_a.copy()
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 50000
    eta0 = 5.0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute conv and objective
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        # update best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[ExpGrad {iteration}] new best {best_val:.6f}")
        else:
            no_improve += 1
        # if stagnation, random perturb
        if no_improve > IMPROVE_LIMIT:
            a = np.random.dirichlet(np.ones(n))
            best_a = a.copy()
            best_val = 2.0 * n * _full_convolution(a).max()
            no_improve = 0
            print(f"[ExpGrad {iteration}] stagnation – random restart")
        # compute gradient for max k
        k = int(np.argmax(b))
        # compute g
        g = np.zeros_like(a)
        if k < n:
            g[:k+1] = a[k::-1]
        else:
            start = k - (n - 1)
            # slice reversed portion of a[n-1:start-1:-1]
            g[start:] = a[n-1:start-1:-1]
        # step size schedule: decaying
        progress = (now - start_time) / (deadline - start_time)
        eta = eta0 * (1.0 - progress) + 0.01 * progress
        # multiplicative update
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0 or np.isnan(s) or np.isinf(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        iteration += 1
    return best_a, best_val

"""
Exponential‑gradient (mirror descent) optimizer for the step‑function problem.

The algorithm works as follows:

* Warm‑start: reuse the previous best sequence (if available), otherwise
  pick the best candidate from a simple two‑parameter Beta family or some
  random Dirichlet draws.
* Main loop: repeatedly
    – compute the full self‑convolution (FFT based),
    – locate the index k where the convolution attains its maximum,
    – form the sub‑gradient of the max‑convolution term:
          g_i = a_{k-i}  (zero when the index is out of range),
    – apply an exponential‑gradient (multiplicative) update
          a_i ← a_i·exp(‑η·g_i)   and renormalise to the simplex,
    – keep the best sequence seen so far.
  The step size η decays slowly so that the search lasts the whole time
  budget.  A stagnation counter triggers a random restart when no
  improvement is observed for many iterations.
* Finally the best sequence is evaluated with ``evaluate_sequence`` and
  returned.

The procedure respects the 1000 s budget, uses only a single CPU and
relies on NumPy (FFT) for fast convolutions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # nearest power‑of‑two length for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Cheap Beta‑family sweep (used only for warm‑start)
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient (exponential‑gradient) optimiser
# ----------------------------------------------------------------------
def _expgrad_opt(a0: np.ndarray,
                 n: int,
                 deadline: float,
                 start_time: float,
                 best_a: np.ndarray,
                 best_val: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using an exponential‑gradient update.
    The update direction is the sub‑gradient of the maximal convolution term.
    """
    a = a0.copy()
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000          # trigger a restart after many useless steps
    eta0 = 5.0                        # initial step size (will decay)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- evaluate current state -----
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[ExpGrad {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            a = np.random.dirichlet(np.ones(n))
            best_a = a.copy()
            best_val = 2.0 * n * _full_convolution(a).max()
            no_improve = 0
            print(f"[ExpGrad {iteration:7d}] stagnation – random restart")
            continue

        # ----- sub‑gradient of the active convolution entry -----
        k = int(np.argmax(b))
        g = np.zeros_like(a)
        if k < n:
            # the overlap lies completely inside the vector
            g[:k + 1] = a[k::-1]
        else:
            # the overlap is shifted to the right edge
            start = k - (n - 1)      # first index that participates
            # reversed slice from the rightmost element down to start
            g[start:] = a[n - 1:start - 1:-1]

        # ----- step‑size schedule (slow decay) -----
        progress = (now - start_time) / (deadline - start_time)
        eta = eta0 * (1.0 - progress) + 0.01 * progress

        # ----- exponential‑gradient update (mirrored on simplex) -----
        a = a * np.exp(-eta * g)
        total = a.sum()
        if total <= 0.0 or not np.isfinite(total):
            # safety net – fall back to a uniform distribution
            a = np.full(n, 1.0 / n)
            total = 1.0
        a /= total

        iteration += 1

    return best_a, best_val
