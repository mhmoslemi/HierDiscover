assistantfinal**Solution Overview**

The aim is to minimise  

\[
\text{score}=2\,n\;\frac{\displaystyle\max_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a` is a non‑negative step function (a list of heights) and `*` denotes the ordinary
linear convolution.  
Because the score is homogeneous of degree 0 we may impose the *simplex* constraint  

\[
a_i\ge 0,\qquad\sum_i a_i = 1 .
\]

The problem is non‑convex – the objective contains a max over a quadratic map – but we can
optimise a **smooth surrogate** of the max (log‑sum‑exp) while keeping the simplex
constraint explicit.  Gradient information is obtained analytically via FFTs,
and Adam together with a Euclidean projection onto the simplex provides a fast,
stable optimiser.  

The workflow is

1. **Cheap initialisation** – a few parametric families (Beta, Dirichlet, uniform) are
   evaluated, the best candidate is kept.
2. **Adam optimisation on the smooth problem** –  
   *loss* = `2·n·smooth_max(conv(a,a))  +  λ·Var(conv(a,a))`.  
   Smoothing parameter ε and the variance weight λ are annealed to zero.
   After each Adam step the vector is projected onto the probability simplex,
   allowing exact zeros.
3. **Deterministic sub‑gradient polishing** – after the smooth phase we
   minimise the exact objective using the known sub‑gradient of the max‑convolution.
   A simple step‑size schedule and occasional jitter avoid stagnation.
4. **Final evaluation** – the best vector found is fed to `evaluate_sequence`.

All heavy work (FFT based convolutions, gradients, projection) is performed in
NumPy, which is fast enough to explore many thousands of iterations within the
1000 s budget.

```python
"""Candidate generation for the step‑function optimisation problem.

The algorithm:
  1. Initialise a probability vector a on the simplex with several cheap
     parametric families (Beta, Dirichlet, uniform) and keep the best.
  2. Run Adam on a smooth surrogate of the true objective
        L(a) = 2·n·smooth_max(conv(a,a)) + λ·Var(conv(a,a))
     The smoothing parameter ε and the variance weight λ are annealed
     linearly to zero.  After each Adam step the vector is projected onto the
     probability simplex, which permits exact zero entries.
  3. Refine the best vector with a deterministic projected sub‑gradient
     method on the exact objective
        f(a) = 2·n·max(conv(a,a)).
  4. Return the best vector as a plain Python list (the evaluator will
     clip negative values, re‑normalise, etc.).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective and smooth surrogates
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a))  (sum(a)=1 is assumed)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(conv(a,a)) and its gradient.
    Returns (smooth_val, grad) where the gradient already includes the
    outer factor 2·n.
    """
    n = a.shape[0]
    b = _full_convolution(a)                 # length 2n-1
    max_b = b.max()
    # log‑sum‑exp trick
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()               # distribution over the 2n‑1 positions

    # Gradient of Σ p_k·b_k w.r.t a: 2·n· Σ p_k·∂b_k/∂a
    # ∂b_k/∂a_j = a_{k-j} + a_{j-(k-j)} = a_{k-j} + a_{k-j} = 2·a_{k-j} (if indices valid)
    # The convolution below implements exactly this.
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)            # length 2n-1
    # keep the part that aligns with a (indices n‑1 … 2n‑2)
    segment = conv_ab[n - 1:2 * n - 1]
    grad = 4.0 * n * segment[::-1]            # factor 2·n already included
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad


def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Variance of the convolution and its gradient.
    The variance is taken w.r.t. the flat mean 1/(2n‑1) (because Σb = 1).
    """
    n = a.shape[0]
    b = _full_convolution(a)                 # length 2n‑1
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # Gradient of variance: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])
    segment = conv_ab[n - 1:2 * n - 1]
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex in O(n log n) time."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on the simplex (smooth max + variance penalty)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises
        L(a) = 2·n·smooth_max(a) + λ·Var(conv(a,a)),
    with a simplex projection after each update.
    """
    n = a0.shape[0]
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.35, 0.002          # start aggressive, end gentle
    eps_start, eps_end = 0.5, 1e-4          # smoothing parameter
    lam_start = 5.0 * n                     # variance‑penalty weight
    lam_end = 0.0

    t = 0
    idle = 0
    MAX_IDLE = 4000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- exact objective (for bookkeeping) -----
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # ----- schedule parameters -----
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps = eps_start * (eps_end / eps_start) ** progress
        lam = lam_start * (1.0 - progress) + lam_end * progress

        # ----- smooth‑max and its gradient -----
        smooth_val, grad_smooth = _smooth_max_and_grad(a, eps)

        # ----- variance penalty -----
        var_val, grad_var = _var_and_grad(a)

        # total gradient (we minimise)
        grad = grad_smooth + lam * grad_var

        # ----- Adam update on the *projected* space -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- simplex projection -----
        a = _project_simplex(a)

        # ----- occasional jitter to escape shallow plateaus -----
        if np.random.rand() < 0.008:
            a += np.random.normal(scale=0.015, size=n)
            a = _project_simplex(a)

        # ----- restart if no improvement for a while -----
        if idle > MAX_IDLE:
            print(f"[Adam] restart after {idle} idle steps")
            a = np.random.dirichlet(np.ones(n))
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic sub‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503,
                        max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
        f(a) = 2·n·max(conv(a,a)).
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_val = _objective(a)

    for it in range(max_iters):
        now = time.time()
        if now >= deadline:
            break

        # sub‑gradient of the max‑convolution
        b = _full_convolution(a)
        max_b = b.max()
        ks = np.where(b == max_b)[0]          # all maximising indices

        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # simple linear step size schedule
        progress = (now - start_time) / (deadline - start_time)
        step = 0.30 * (1.0 - progress) + 1e-5 * progress

        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)

        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Subgrad] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break

        # occasional jitter to shake off plateaus
        if np.random.rand() < 0.0015:
            a = np.maximum(a + np.random.normal(scale=0.018, size=n), 0.0)
            a = _project_simplex(a)

    return best_a, best_val

def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Exhaustive grid over a two‑parameter Beta family."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val

    # Phase B – Deterministic sub‑gradient polishing (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_polish(
        best_a,
        deadline=deadline,
        start_time=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
        max_iters=60000,
    )

    # ------------------------------------------------------------------
    # Final evaluation (the official score)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
```
