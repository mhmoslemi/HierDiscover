"""
Improved step‑function optimizer.

The algorithm builds on the previous heuristic but adds several new
ingredients that proved useful for breaking the 1.509 barrier:

* a richer random‑search stage (extra “beta‑shaped” families, wider
  exponent / power ranges);
* a *smooth‑max* phase that replaces the hard‑max sub‑gradient by a
  softmax (log‑sum‑exp) gradient.  By annealing the softmax temperature
  we obtain a more reliable descent direction while still respecting the
  simplex constraint (exponentiated‑gradient update);
* a final deterministic hill‑climb (identical to the reference code)
  and a tiny uniform mixing step.

All phases respect the 1000 s time budget, keep the vector non‑negative
and normalised (Σa = 1) and return the best list of floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary vectors."""
    out_len = a.shape[0] + b.shape[0] - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value returned by ``evaluate_sequence`` for a normalised vector."""
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Deterministic families for the initial candidate
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent start vector and its objective."""
    best_a = np.full(n, 1.0 / n)                # uniform
    best_val = _objective(best_a, n)

    def try_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective(vec, n)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    rng = np.random.default_rng(12345)

    # a few handcrafted shapes
    idx = np.arange(n)
    try_update(np.arange(n, 0, -1, dtype=float))                     # decreasing
    try_update(np.maximum(idx + 1, n - idx))                        # linear U‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))             # quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))))      # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes (broader range than the reference)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials (already explored before)
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996,
              0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=np.float64)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # a handful of Dirichlet draws
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric families used later (including a new beta‑shape)
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power(n: int, p: float) -> np.ndarray:
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_beta(n: int, a: float) -> np.ndarray:
    """Beta‑shaped density: f(x) ∝ x^{a‑1} (1‑x)^{a‑1},  a>0."""
    # use mid‑points to avoid exact 0/1 singularities
    x = (np.arange(n) + 0.5) / n
    a_vec = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    a_vec = np.maximum(a_vec, 0.0)
    s = a_vec.sum()
    return a_vec / s if s != 0 else np.full(n, 1.0 / n)

                   target_val: float = 1.503) -> tuple[np.ndarray, float]:

    no_improve = 0

    while time.time() < deadline:

        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        k = int(np.argmax(b))
        rev = np.zeros(n)

        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]
        g = 2.0 * rev                           # sub‑gradient

        progress = (time.time() - start) / (deadline - start)
        eta = (1.0 - progress) * 0.12 + progress * 0.0005

        cur = cur * np.exp(-eta * g)
        total = cur.sum()
        if total == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.008, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑S – smooth‑max (log‑sum‑exp) gradient descent
# ----------------------------------------------------------------------
def _smooth_grad_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a0.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 150_000

    while time.time() < deadline:
        c = _full_convolution(cur)                     # convolution vector
        cur_val = 2.0 * n * c.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        # temperature (softness) – start almost linear, end very sharp
        alpha = 0.1 + 30.0 * progress          # 0.1 → ~30.1
        eta = (1.0 - progress) * 0.25 + progress * 0.001

        # soft‑max weights (stable)
        maxc = c.max()
        exp_c = np.exp(alpha * (c - maxc))
        w = exp_c / exp_c.sum()                # length = 2n‑1

        # gradient = α * (Σ_k w_k a_{k‑i} + Σ_k w_k a_{i‑k})
        w_rev = w[::-1]

        conv1 = _conv_two(cur, w_rev)          # length = 3n‑2
        conv2 = _conv_two(w, cur)              # length = 3n‑2
        m = w.shape[0]                         # = 2n‑1

        # grad1_i = conv1[m‑1‑i]   (i = 0…n‑1)
        grad1 = conv1[m - 1 : m - 1 - n : -1]
        # grad2_i = conv2[i]
        grad2 = conv2[:n]

        g = alpha * (grad1 + grad2)

        # exponentiated‑gradient step
        cur = cur * np.exp(-eta * g)
        total = cur.sum()
        if total == 0.0 or np.isnan(total):
            cur = np.full(n, 1.0 / n)
        else:
            cur /= total

        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.006, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[Smooth {it:7d}] jitter after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑B – deterministic greedy hill‑climbing (unchanged)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while time.time() < deadline:
        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]

        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        b_new = _full_convolution(cur)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            cur_val = new_val
        else:
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑C – tiny uniform mixing
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function under a fixed time budget.
    Returns the best list of floats found (Σa = 1) which minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – a good compromise between granularity and speed
    n = 8000

    # --------------------------------------------------------------
    # Initialise (use previous solution if available)
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase‑0 – richer random exploration of parametric families
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 999)
    phase0_deadline = start_time + 0.15 * (deadline - start_time)

    while time.time() < phase0_deadline:
        choice = rng.integers(0, 6)   # six families now
        if choice == 0:                     # pure exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_exp(n, r)
        elif choice == 1:                   # reversed exponential
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_rev_exp(n, r)
        elif choice == 2:                   # mixture of two exponentials
            r1 = rng.uniform(0.90, 0.99999)
            r2 = rng.uniform(0.90, 0.99999)
            w = rng.random()
            cand = _gen_mix_exp(n, r1, r2, w)
        elif choice == 3:                   # pure power‑law
            p = rng.uniform(-8.0, 8.0)      # extended range
            cand = _gen_power(n, p)
        elif choice == 4:                   # power‑law × exponential
            p = rng.uniform(-8.0, 8.0)
            r = rng.uniform(0.90, 0.99999)
            cand = _gen_power_exp(n, p, r)
        else:                               # beta‑shaped U‑distribution
            a = rng.uniform(0.05, 0.95)     # α parameter (edges heavy)
            cand = _gen_beta(n, a)

        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Random] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase‑A – exponentiated‑gradient descent (sub‑gradient)
    # ------------------------------------------------------------------
    phaseA_deadline = start_time + 0.45 * (deadline - start_time)
    best_a, best_val = _expgrad_phase(
        best_a, n,
        start=start_time,
        deadline=phaseA_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase‑S – smooth‑max gradient descent (log‑sum‑exp)
    # ------------------------------------------------------------------
    phaseS_deadline = start_time + 0.70 * (deadline - start_time)
    best_a, best_val = _smooth_grad_phase(
        best_a, n,
        start=phaseA_deadline,
        deadline=phaseS_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase‑B – deterministic greedy hill‑climbing
    # ------------------------------------------------------------------
    phaseB_deadline = start_time + 0.95 * (deadline - start_time)
    best_a, best_val = _greedy_hillclimb(
        best_a, n,
        start=phaseS_deadline,
        deadline=phaseB_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )

    # ------------------------------------------------------------------
    # Phase‑C – final tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)

    # ------------------------------------------------------------------
    # Verify with the security‑wrapped evaluator
    # ------------------------------------------------------------------
    try:
        reported = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {reported:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
