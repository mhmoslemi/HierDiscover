        max_frac = 0.001 + 0.2 * (1 - progress) # start larger, shrink

        b_new = cur_b.copy()

            # accept
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
        # optional random restart if no improvement for many steps

        max_frac = 0.2 * (1 - progress) + 0.001 * progress
        delta = a_cur[i] * np.random.rand() * max_frac
        if delta <= 0: continue

        # update b incrementally
        b_new = b_cur.copy()
        b_new[j:j+n] += 2.0 * delta * a_cur
        b_new[i:i+n] -= 2.0 * delta * a_cur
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i+j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        dE = new_val - cur_val
        if dE < 0 or np.random.rand() < math.exp(-dE / max(T,1e-12)):
            # accept move

            a_cur[j] += delta
            b_cur = b_new

            if new_val < best_local_val - 1e-12:
                best_local_val = new_val
                best_local_a = a_cur.copy()
                no_improve = 0
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")

            cur_val = 2.0 * n * b_cur.max()
            no_improve = 0
            print(f"[SA {it:7d}] restart after stagnation")
        it += 1
    return best_local_a, best_local_val

"""
Optimise a non‑negative step function for the self‑convolution objective.

The algorithm proceeds in five stages (all respecting the 1000 s budget):
  1.  A broad parametric sweep (exponential, power‑law, beta, sine‑power,
      piecewise‑constant, logistic, random Dirichlet) to obtain a good
      starting point.
  2.  Adam optimiser on a log‑sum‑exp smooth surrogate of the max‑convolution.
  3.  Targeted hill‑climbing that moves mass from indices that contribute
      most to the current maximal convolution entry.
  4.  Projected sub‑gradient descent on the exact (non‑smooth) objective.
  5.  Simulated‑annealing style random walk that accepts occasional
      uphill moves, helping escape shallow local minima.

A final tiny mixing with the uniform distribution is performed.
The best sequence found is returned as a plain Python list (its sum is 1,
so the provided `evaluate_sequence` evaluates the correct objective).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based for speed)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective handling (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a probability vector (Σa = 1)."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x ≥ 0, Σx = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families – broad sweep for a good starting point
# ----------------------------------------------------------------------
def _initial_candidate(n: int) -> tuple[np.ndarray, float]:
    """Return the best vector found among a diverse collection of parametric families."""
    best_a = np.full(n, 1.0 / n)                # uniform starts the race
    best_val = _objective_value(best_a)

    def _maybe_update(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)          # 0 … n‑1
    i1 = i + 1.0                           # 1 … n

    # 1. Exponential tails (decreasing & symmetric)
    for r in np.linspace(0.9995, 0.99999, 12):
        _maybe_update(r ** i)                         # monotone decreasing
        _maybe_update(r ** i + r ** (n - 1 - i))      # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 5.0, 15):
        _maybe_update(i1 ** (-p))
        _maybe_update(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.2, 5.0, 8):
        for b_param in np.linspace(0.2, 5.0, 8):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _maybe_update(a)

    # 4. Sine‑power family (half‑wave sinusoid raised to a power)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 10):
        _maybe_update(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges” shapes
    for k_frac in np.linspace(0.01, 0.2, 10):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.5, 10):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _maybe_update(a)

    # 6. Logistic‑type tails (asymmetric, with mirror)
    for slope in np.linspace(0.005, 0.05, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _maybe_update(a)
        _maybe_update(np.flip(a))

    # 7. Random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        _maybe_update(a)

    # 8. Interpolation of a previous solution if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _maybe_update(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp approximation of the max‑convolution."""
    # logits → simplex via soft‑max
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.14, 0.004
    eps_start, eps_end = 1e-1, 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80_000

    it = 0
    while time.time() < deadline:
        # simplex point
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        # d/da 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, rev(p))
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]

        # back‑prop through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.009, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.25
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Greedy mass transfer from high‑contrib. indices, with occasional restarts."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # active convolution index
        k_max = int(np.argmax(b))

        # contribution of each index to the active entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # choose a target index biased away from the active side
        if np.random.rand() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = np.random.choice(n, p=probs_j)
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (linear‑time)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Exact sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.6 / n
    eta = eta0
    decay = 0.9995
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 180_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient at the active index
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        eta *= decay

        if no_improve > IMPROVE_LIMIT:
            # small random shake and restart step size
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase D – Simulated‑annealing style random walk
# ----------------------------------------------------------------------
def _simulated_annealing_phase(a0: np.ndarray, n: int,
                               start: float, deadline: float,
                               best_a: np.ndarray, best_val: float) -> tuple[np.ndarray, float]:
    """Random walk with occasional uphill moves to escape shallow minima."""
    a_cur = a0.copy()
    b_cur = _full_convolution(a_cur)
    cur_val = 2.0 * n * b_cur.max()

    best_local_a = best_a.copy()
    best_local_val = best_val

    # initial temperature is a fraction of the current objective gap
    T0 = max(0.02 * cur_val, 1e-4)

    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T0 * (0.005 ** progress)          # exponential cooling

        # active convolution index
        k_max = int(np.argmax(b_cur))

        # source index weighted by its contribution to the active entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a_cur)
        contrib[mask] = a_cur[mask] * a_cur[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_i = contrib / total_contrib
            i = np.random.choice(n, p=probs_i)
        else:
            i = np.random.randint(n)

        if a_cur[i] <= 1e-16:
            continue

        # target index biased toward low contribution to the active side
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a_cur[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0:
            probs_j = w / w.sum()
            j = np.random.choice(n, p=probs_j)
        else:
            j = (i + np.random.randint(1, n)) % n

        max_frac = 0.15 * (1 - progress) + 0.001 * progress
        delta = a_cur[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update
        b_new = b_cur.copy()
        b_new[j:j + n] += 2.0 * delta * a_cur
        b_new[i:i + n] -= 2.0 * delta * a_cur
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        dE = new_val - cur_val

        # Metropolis acceptance
        if dE < 0 or np.random.rand() < math.exp(-dE / max(T, 1e-12)):
            a_cur[i] -= delta
            a_cur[j] += delta
            b_cur = b_new
            cur_val = new_val
            if new_val < best_local_val - 1e-12:
                best_local_val = new_val
                best_local_a = a_cur.copy()
                no_improve = 0
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a_cur = best_local_a.copy()
            b_cur = _full_convolution(a_cur)
            cur_val = 2.0 * n * b_cur.max()
            no_improve = 0
            print(f"[SA {it:7d}] restart after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass – useful for a final tiny tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to avoid accidental overtime
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 12000                     # a bit larger than the previous 9000

    # ------------------------------------------------------------------
    # Initialise a – exhaustive parametric sweep.
    # ------------------------------------------------------------------
    print("[Init] starting parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on a smooth surrogate (≈35 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * (deadline - start_time)
    best_a, best_val = _adam_phase(
        best_a, n,
        start=start_time,
        deadline=phase_a_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase A] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – targeted hill‑climbing (≈25 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.60 * (deadline - start_time)
    best_a, best_val = _hillclimb_targeted(
        best_a, n,
        start=phase_a_deadline,
        deadline=phase_b_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase B] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase C – projected sub‑gradient descent (≈20 % of budget)
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.80 * (deadline - start_time)
    best_a, best_val = _subgradient_phase(
        best_a, n,
        start=phase_b_deadline,
        deadline=phase_c_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase C] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – simulated annealing / random walk (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _simulated_annealing_phase(
        best_a, n,
        start=phase_c_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
    )
    print(f"[Phase D] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
