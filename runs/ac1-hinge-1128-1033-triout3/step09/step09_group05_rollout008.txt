def _gradient_smooth(a, T):
    b = _full_convolution(a)  # size 2n-1
    shift = b.max()
    exp_b = np.exp((b - shift) / T)
    w = exp_b / exp_b.sum()
    # compute correlation using FFT
    L = 1 << ((len(a) + len(w) - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conj(fa) * fw, L)[:len(a) + len(w) - 1]
    offset = len(w) - 1
    grad = 2.0 / T * corr[offset:offset + len(a)]
    return grad

def project_simplex(v):
    """Project v onto the probability simplex."""
    # Implementation: Sort v descending, find threshold.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

        a0 = np.asarray(height_sequence_1, dtype=float)

"""
Step‑function optimizer (new version).

We start from a diverse set of parametric families, then run a
*temperature‑annealed smooth exponential‑gradient descent*.
The gradient is the exact derivative of the soft‑max approximation to
max(conv(a,a)).  After the smooth phase we finish with the deterministic
hill‑climb and a tiny uniform mixing.

All vectors are kept on the probability simplex (Σa = 1, a_i ≥ 0).
The code respects the 1000 s budget and prints intermediate best values.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast (full) convolution via FFT – length 2·n‑1.
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (assumes Σa = 1):  f(a) = 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    maxb = _full_convolution(a).max()
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families for the quick random sweep.
# ----------------------------------------------------------------------
def _gen_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_rev_exp(n: int, r: float) -> np.ndarray:
    a = np.power(r, np.arange(n - 1, -1, -1, dtype=np.float64))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_mix_exp(n: int, r1: float, r2: float, w: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w * np.power(r1, i) + (1.0 - w) * np.power(r2, n - 1 - i)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power(n: int, p: float) -> np.ndarray:
    a = np.power(np.arange(1, n + 1, dtype=np.float64), p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gen_power_exp(n: int, p: float, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    log_a = p * np.log(i + 1.0) - (np.log(r) * i)
    log_a -= np.max(log_a)
    a = np.exp(log_a)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Deterministic families for a cheap first sweep.
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent start vector and its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a, n)

    def try_update(vec):
        nonlocal best_a, best_val
        vec = np.asarray(vec, dtype=np.float64)
        if vec.ndim != 1 or vec.shape[0] != n:
            return
        if np.any(vec < 0):
            return
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        val = _objective(vec, n)
        if val < best_val:
            best_val, best_a = val, vec.copy()

    rng = np.random.default_rng(12345)

    # a few handcrafted shapes
    idx = np.arange(n)
    try_update(np.arange(n, 0, -1, dtype=float))                # decreasing
    try_update(np.maximum(idx + 1, n - idx))                   # linear U‑shape
    try_update((np.arange(n, 0, -1, dtype=float) ** 2))        # quadratic
    try_update(np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))) # half‑wave sine
    try_update(1.0 + np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))  # cosine ends

    # power‑law U‑shapes
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        i = np.arange(n, dtype=float) + 1.0
        a = i ** (-p) + (n - i + 1) ** (-p)
        try_update(a)

    # symmetric exponentials
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.997, 0.996,
              0.995, 0.99, 0.98, 0.95):
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
        try_update(a)

    # a few random Dirichlet draws
    for _ in range(12):
        try_update(rng.dirichlet(np.ones(n)))

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase S – temperature‑annealed smooth exponential‑gradient descent.
# ----------------------------------------------------------------------
def _smooth_expgrad_phase(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the soft‑max approximation of max(conv(a,a)).
    The update rule is an exponentiated‑gradient step:
        a ← a ⊙ exp(−η·∇f)  (⊙ = element‑wise product)
    with a temperature T that is annealed together with the step size η.
    """
    cur = a0.copy()
    no_improve = 0
    IMPATIENT = 100_000

    # FFT size that can accommodate both the convolution (2n‑1) and the
    # correlation (n+2n‑2 = 3n‑2).
    L = 1 << ((3 * n - 2).bit_length())

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- forward FFT of the current vector -----
        fa = np.fft.rfft(cur, L)

        # ----- compute full self‑convolution (length 2n‑1) -----
        conv_full = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
        max_b = conv_full.max()
        cur_val = 2.0 * n * max_b

        # ----- record improvement -----
        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[SmoothGrad {int(now-start):7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- annealing schedules -----
        progress = (now - start) / (deadline - start)
        # step size η
        eta = (1.0 - progress) * 0.5 + progress * 0.001
        # temperature factor: start = 5·max_b, end = 0.01·max_b
        T_factor_start = 5.0
        T_factor_end = 0.01
        T_factor = T_factor_start * (T_factor_end / T_factor_start) ** progress
        T = max_b * T_factor
        if T < 1e-12:
            T = 1e-12

        # ----- soft‑max weights -----
        shift = max_b
        exp_vals = np.exp((conv_full - shift) / T)
        w = exp_vals / exp_vals.sum()

        # ----- correlation a ⋆ w (full length n+2n‑2) -----
        fw = np.fft.rfft(w, L)
        corr_full = np.fft.irfft(np.conj(fa) * fw, L)[: n + len(w) - 1]

        # gradient of the smooth objective
        offset = len(w) - 1
        grad = 2.0 / T * corr_full[offset:offset + n]

        # ----- exponentiated‑gradient update -----
        exp_arg = -eta * grad
        # avoid overflow in the exponentials
        exp_arg = np.clip(exp_arg, a_min=-700, a_max=700)
        # stable scaling
        max_arg = np.max(exp_arg)
        scale = np.exp(exp_arg - max_arg)
        cur = cur * scale
        s = cur.sum()
        if s == 0.0:
            cur = np.full(n, 1.0 / n)
        else:
            cur /= s

        # ----- jitter when stuck -----
        if no_improve > IMPATIENT:
            noise = np.random.normal(scale=0.006, size=n)
            cur = np.maximum(cur + noise, 0.0)
            cur /= cur.sum()
            no_improve = 0
            print(f"[SmoothGrad {int(now-start):7d}] jitter after stagnation")

    return best_a, best_val

# Phase B – deterministic greedy hill‑climbing (unchanged).
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray, n: int,
                      start: float, deadline: float,
                      best_a: np.ndarray, best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    cur = a_start.copy()
    it = 0
    no_improve = 0
    IMPATIENT = 300_000
    max_frac_start, max_frac_end = 0.5, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break

        b = _full_convolution(cur)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-15:
            best_val, best_a = cur_val, cur.copy()
            no_improve = 0
            print(f"[HillClimb {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k = int(np.argmax(b))
        rev = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        rev[idx] = cur[k - idx]
        i = int(np.argmax(rev))
        rev_tmp = rev.copy()
        rev_tmp[i] = np.inf
        j = int(np.argmin(rev_tmp))

        if cur[i] <= 1e-15:
            if no_improve > IMPATIENT:
                noise = np.random.normal(scale=0.006, size=n)
                cur = np.maximum(cur + noise, 0.0)
                cur /= cur.sum()
                no_improve = 0
                print(f"[HillClimb {it:7d}] jitter after stagnation")
            continue

        delta = cur[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        new_val = 2.0 * n * _full_convolution(cur).max()
        if new_val < cur_val - 1e-15:
            cur_val = new_val
        else:
            cur[i] += delta
            cur[j] -= delta

        it += 1

    return best_a, best_val

# Phase C – final tiny uniform mixing (unchanged).

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    best_a, best_val = a, _objective(a, n)
    uniform = np.full(n, 1.0 / n)

        cand /= cand.sum()
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val
