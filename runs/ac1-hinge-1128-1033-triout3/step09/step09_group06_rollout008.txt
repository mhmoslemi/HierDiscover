def _simulated_annealing(a, cur_val, deadline):
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    # parameters
    temp = 0.01
    decay = 0.9999
    while time.time() < deadline - 2:
        # propose move
        if rng.random() < 0.3:
            # swap
            i, j = rng.integers(0, n, size=2)
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # transfer
            i, j = rng.integers(0, n, size=2)
            if cur_a[i] < 1e-12:
                continue
            delta = cur_a[i] * 0.1 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        # project onto simplex (nonneg already)
        new_a = np.maximum(new_a, 0.0)
        sum_new = new_a.sum()
        if sum_new == 0:
            continue
        new_a /= sum_new
        # compute objective fast
        val = _objective(new_a)
        # acceptance
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / max(temp, 1e-12)):
            cur_a = new_a
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = new_a.copy()
                print(f"[anneal] new best {best_val:.6f}")
        # temperature decay
        temp *= decay
    return best_a, best_val

"""
Propose a non‑negative step function that (approximately) minimises
`evaluate_sequence`.  The algorithm consists of three stages:

1. **L2‑flattening** – gradient descent on the squared deviation of the
   self‑convolution from its average value (which equals 1/(2n‑1) after
   normalising the sum to 1).  This pushes the whole convolution towards a
   flat profile.

2. **Smooth‑max descent** – a log‑sum‑exp surrogate of the L∞‑norm is
   optimised with multiplicative mirror‑descent.  The temperature (ε)
   decreases from 0.3 down to 0.001, focusing the update on the largest
   convolution peaks.  After each ε‑stage an anti‑symmetric re‑ordering
   (pairing large with small entries) is applied – this never worsens
   the true maximum.

3. **Simulated‑annealing refinement** – a cheap stochastic local search
   (random swaps, tiny mass transfers) with a slowly cooling temperature
   accepts uphill moves with probability `exp(‑Δ/Temp)`.  The objective is
   evaluated by a fast FFT‑based convolution (the true value is
   `2·n·max(conv)` when the sum of the coefficients equals 1).

The whole search respects a 1000 s time budget, performs a few
restarts with different initialisations and returns the best sequence
found.  All heavy work (FFT convolutions, gradients) happens without
calling the expensive `evaluate_sequence`; that function is used only
for the final verification.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based convolution, gradient, projection, re‑order)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted_peaks(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)[k] w.r.t. a.
    Accepts signed weights w (length 2n‑1).  Returns a vector of length n.
    """
    n = a.shape[0]
    # reverse the weight vector because convolution with reversed kernel
    w_rev = w[::-1]
    size = _next_pow2(n + w_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[: n + w_rev.shape[0] - 1]
    # relevant slice corresponds to indices n‑1 … 2n‑2 (inclusive)
    seg = conv[n - 1 : 2 * n - 1]
    # reverse again to obtain the gradient
    return seg[::-1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]               # large → left side
        b[-1 - i] = s[i]               # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _objective(a: np.ndarray) -> float:
    """True objective for a vector with Σa = 1: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Stage 1 – L2‑flattening (quadratic loss on the whole convolution)
# ----------------------------------------------------------------------
def _flatten_l2(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Gradient descent on Σ (conv - avg)^2.  Returns best vector/value."""
    n = a.shape[0]
    avg = 1.0 / (2 * n - 1)          # because Σ(conv)=1 when Σa=1
    best_a = a.copy()
    best_val = _objective(a)

    lr = 0.6                         # initial additive step size
    max_iters = 2500
    it = 0
    while it < max_iters and time.time() < deadline - 5:
        conv = _full_convolution(a)
        diff = conv - avg             # signed weight vector
        # gradient of Σ diff^2 ≈ 4·(diff * a)  (convolution with signed weights)
        grad = 4.0 * _gradient_weighted_peaks(a, diff)

        # additive mirror‑descent step, then simplex projection
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        # occasional re‑ordering (does not increase the max)
        if it % 200 == 0:
            a = _anti_symmetric_reorder(a)

        # evaluate every few iterations
        if it % 25 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[L2] it {it:4d} new best = {best_val:.6f}")

        # decay learning rate slowly
        lr *= 0.9998
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 2 – Smoothed‑max descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smoothed_descent(a_start: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Minimise a log‑sum‑exp surrogate of the max‑convolution."""
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    eps_schedule = [0.3, 0.2, 0.1, 0.07, 0.04,
                    0.02, 0.01, 0.005, 0.002, 0.001]
    total_iters = 3500
    it_per_eps = max(1, total_iters // len(eps_schedule))

    for eps in eps_schedule:
        eta = 0.7
        for it in range(it_per_eps):
            if time.time() >= deadline - 5:
                break

            conv = _full_convolution(a)
            mx = conv.max()
            # stable softmax over conv values
            shift = mx
            exp_vals = np.exp(eps * (conv - shift))
            p = exp_vals / exp_vals.sum()          # Σp = 1, p≥0

            # gradient of the smooth surrogate
            grad = 2.0 * _gradient_weighted_peaks(a, p)

            # multiplicative mirror‑descent (keeps entries ≥0)
            delta = -eta * grad
            delta = np.clip(delta, -15.0, 0.0)      # avoid overflow/underflow
            a = a * np.exp(delta)
            a = _project_simplex(a)

            # occasional checks and re‑ordering
            if it % 20 == 0:
                cur_val = _objective(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[smooth eps {eps:.3f} it {it:4d}] new best = {best_val:.6f}")
                if it % 200 == 0:
                    a = _anti_symmetric_reorder(a)
            eta *= 0.9997
        # one final re‑ordering after each ε‑stage (cannot increase the max)
        a = _anti_symmetric_reorder(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – Simulated‑annealing local refinement
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Light stochastic hill‑climbing with temperature decay.
    Allows occasional uphill moves to escape local minima.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    cur_a = a.copy()
    cur_val = cur_val

    rng = np.random.default_rng()
    temp = 0.008                      # initial temperature
    decay = 0.99985

    while time.time() < deadline - 2:
        # propose a move
        if rng.random() < 0.3:
            # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-14:
                continue
            delta = cur_a[i] * 0.08 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑normalise onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s <= 0:
            continue
        new_a /= s

        # compute fast objective (no call to evaluate_sequence)
        val = _objective(new_a)

        # Metropolis acceptance
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / max(temp, 1e-12)):
            cur_a = new_a
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = new_a.copy()
                print(f"[anneal] new best = {best_val:.6f}")

        temp *= decay
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` under a 1000 s time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 5.0     # safety margin

    # Problem size – a few thousand coefficients give a good trade‑off.
    n = 8000

    # ------------------------------------------------------------------
    # Try to initialise from a previously discovered construction, if any.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        a0 = a0 / a0_sum if a0_sum > 0 else np.full(n, 1.0 / n)
    except Exception:
        a0 = None

    best_a = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Multiple restarts with diverse seeds
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 30 and restart < 6:
        restart += 1

        # ---- initialise -------------------------------------------------
        if restart == 1 and a0 is not None:
            a = a0.copy()
            init_name = "previous"
        else:
            strat = rng.choice(["uniform", "exp", "dirichlet"])
            if strat == "uniform":
                a = np.full(n, 1.0 / n)
                init_name = "uniform"
            elif strat == "exp":
                a = rng.exponential(scale=1.0, size=n)
                a /= a.sum()
                init_name = "exp"
            else:  # dirichlet with random concentration
                conc = rng.uniform(0.04, 0.3)
                a = rng.dirichlet(np.full(n, conc))
                init_name = f"dirichlet_{conc:.3f}"

        # ---- Stage 1: L2 flattening ------------------------------------
        a, val = _flatten_l2(a, deadline)
        # ---- Stage 2: smooth‑max descent --------------------------------
        a, val = _smoothed_descent(a, deadline)
        # ---- Stage 3: local simulated‑annealing refinement -------------
        a, val = _simulated_annealing(a, val, deadline)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[global] new best {best_val:.6f} (restart {restart}, init={init_name})")

    # ------------------------------------------------------------------
    # Final polish on the current best vector (if time permits)
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 5:
        best_a, best_val = _simulated_annealing(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final verification with the supplied evaluate_sequence
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
