def propose_candidate(seed=0, budget_s=1000, **kwargs):
   np.random.seed(seed)
   rng = np.random.default_rng(seed)

   start = time.time()
   deadline = start + budget_s - 5.0

   n = 8000

   # initialise with random exponential draws
   a = rng.exponential(scale=1.0, size=n)
   a /= a.sum()

   # optionally start from height_sequence_1 as better initial guess
   try:
      init = np.asarray(height_sequence_1, dtype=float)
      if init.shape[0] != n:
         old_x = np.linspace(0,1,init.shape[0])
         new_x = np.linspace(0,1,n)
         init = np.interp(new_x, old_x, init)
      init = np.maximum(init,0.0)
      s = init.sum()
      if s > 0:
         init /= s
      # Evaluate its bound (cheap)
      conv_init = _full_convolution(init)
      eval_init = 2 * n * conv_init.max()
      # Keep if better
      if eval_init < 2 * n * _full_convolution(a).max():
         a = init.copy()
   except Exception:
      pass

   # evaluation shortcut
   def eval_a(vec):
       conv = _full_convolution(vec)
       return 2.0 * n * conv.max()

   best_a = a.copy()
   best_val = eval_a(best_a)
   print(f"[init] evaluation = {best_val:.6f}")

   # gradient descent with Adam and smoothing
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   alpha0 = 0.5
   beta_start = 1.0
   beta_end = 5000.0
   beta_ramp = 150000  # iterations over which beta increases linearly

   it = 0
   last_improvement = 0
   max_iter = 500000

   while time.time() < deadline and it < max_iter:
       conv = _full_convolution(a)
       max_conv = conv.max()
       # compute smoothed weights
       delta = conv - max_conv
       beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp)
       w = np.exp(beta * delta)   # w <= 1, w[max]=1
       w_rev = w[::-1]
       g = _gradient_weighted(a, w_rev)  # shape n

       # Adam update
       m = beta1 * m + (1 - beta1) * g
       v = beta2 * v + (1 - beta2) * (g * g)
       m_hat = m / (1 - beta1**(it + 1))
       v_hat = v / (1 - beta2**(it + 1))

       # step
       step = -alpha0 * (m_hat / (np.sqrt(v_hat) + eps))

       # limit step for numeric stability
       step = np.clip(step, -20.0, 20.0)
       a = a * np.exp(step)

       # renormalize
       a = np.maximum(a, 1e-15)
       a /= a.sum()

       # logging every 500 iterations
       if (it % 500) == 0:
           cur_val = 2.0 * n * (np.max(conv))  # uses old conv; compute again to be safe after update
           # recompute after update for accurate cur_val
           conv2 = _full_convolution(a)
           cur_val = 2.0 * n * np.max(conv2)
           if cur_val < best_val - 1e-12:
               best_val = cur_val
               best_a = a.copy()
               last_improvement = it
               print(f"[iter {it}] new best = {best_val:.6f}")

           # occasional anti-symmetric reorder
           if (it % 2000) == 0:
               a_ord = _anti_symmetric_reorder(a)
               conv_ord = _full_convolution(a_ord)
               val_ord = 2.0 * n * np.max(conv_ord)
               if val_ord < best_val - 1e-12:
                   best_val = val_ord
                   best_a = a_ord.copy()
                   a = a_ord.copy()
                   print(f"[reorder {it}] improved to {best_val:.6f}")

       # if stagnated for >10000 iters, random blend
       if (it - last_improvement) > 15000:
           # restart partially
           noise = rng.exponential(scale=1.0, size=n)
           noise /= noise.sum()
           a = 0.6 * a + 0.4 * noise
           a = np.maximum(a, 1e-15)
           a /= a.sum()
           # reset Adam moments
           m.fill(0.0)
           v.fill(0.0)
           last_improvement = it
           print(f"[perturb {it}] blended with random noise")
       it += 1

   # final hill climbing
   cur_a = best_a.copy()
   cur_val = best_val
   while time.time() < deadline:
       if rng.random() < 0.2:
           # swap two entries
           i, j = rng.integers(0, n, size=2)
           if i != j:
               cand = cur_a.copy()
               cand[i], cand[j] = cand[j], cand[i]
           else:
               continue
       else:

           i, j = rng.integers(0, n, size=2)
           if i == j or cur_a[i] <= 1e-12:
               continue
           delta = cur_a[i] * 0.02 * rng.random()
           cand = cur_a.copy()
           cand[i] -= delta
           cand[j] += delta

       cand = np.maximum(cand, 0.0)
       s = cand.sum()
       if s == 0:

       cand /= s

       # evaluate cheaply
       conv_c = _full_convolution(cand)
       cand_val = 2.0 * n * np.max(conv_c)
       if cand_val < cur_val - 1e-12:
           cur_a = cand
           cur_val = cand_val
           best_a = cand.copy()
           best_val = cand_val
           print(f"[hillclimb] new best = {cur_val:.6f}")

   final_score = evaluate_sequence(best_a.tolist())
   print(f"[final] evaluation = {final_score:.6f}")
   return best_a.tolist()

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  

The search consists of three stages:

1. **Multiple short Adam‑mirrored‑descent runs** – each run starts from a fresh
   random Dirichlet vector (exponential draws, normalised to sum = 1).  The
   objective that is minimised is a smooth approximation of the
   infinity‑norm of the self‑convolution:
        Φ(a) = (1/β)·log(∑_k exp(β·(c_k−max(c)))) ,
   where c = a∗a.  The smoothing parameter β is increased from 1 up to a
   large value (≈ 5000), so the algorithm first follows a soft landscape and
   then concentrates on the true maximum.  Updates are performed by an
   exponentiated‑gradient (mirror‑descent) step with Adam‑style adaptive
   moments; after each step the vector is projected back onto the probability
   simplex.

2. **Periodic re‑ordering** – every few thousand iterations the current vector
   is reordered antipodally (largest values placed opposite the smallest ones);
   this often reduces the peak of the convolution.

3. **Stochastic hill‑climbing with simulated‑annealing acceptance** – starting
   from the best vector found in stage 1 a cheap local‐search is run until the
   time budget is exhausted.  Random swaps or tiny mass‑transfers are tried;
   moves are accepted if they improve the true objective or with a small
   probability that decays over time (simulated‑annealing).  The best solution
   encountered is finally returned.

The whole routine respects the 1000 s budget, prints intermediate
evaluations and always returns a valid (non‑negative, normalised) sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (useful for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of a real vector `x`.
    Returns an array of length 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can produce tiny negatives – clamp them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   w.r.t. `a` (the factor 2 is included).
    `w_rev` must be `w` reversed.
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    # The relevant slice starts at index w_rev.size‑1 (=2n‑2) and has length n
    start = w_rev.shape[0] - 1
    seg = conv[start:start + n]
    return 2.0 * seg


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Re‑order `a` so that the largest values are placed opposite the smallest
    ones (pairwise).  This often reduces the convolution peak.
    """
    n = a.shape[0]
    s = np.sort(a)  # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]          # large → left side
        b[-1 - i] = s[i]          # small → right side
    if n % 2:
        b[half] = s[half]
    return b

# Main optimisation routine
# ----------------------------------------------------------------------

    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence` within the given time budget.
    """

    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a few seconds at the end for final evaluation / tidy‑up
    deadline = start_time + budget_s - 5.0

    # Problem size – a few thousand coefficients give a reasonable
    # trade‑off between expressiveness and FFT speed.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Helper to compute the cheap proxy of the true objective:
    #   2·n·max(conv)  (sum is forced to 1, so denominator = 1)
    # --------------------------------------------------------------
    def cheap_score(a_vec: np.ndarray) -> float:
        return 2.0 * n * np.max(_full_convolution(a_vec))

    # --------------------------------------------------------------
    # Initialise the global best from the historic construction if it
    # exists, otherwise fall back to a random Dirichlet vector.
    # --------------------------------------------------------------
    try:
        # `height_sequence_1` may be present in the global namespace.
        base = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)

        base = np.maximum(base, 0.0)
        s = base.sum()
        if s > 0:

        else:

            base /= base.sum()

        base /= base.sum()

    global_best_a = base.copy()
    global_best_val = cheap_score(global_best_a)
    print(f"[init] cheap score = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the Adam–mirrored‑descent phase
    # ------------------------------------------------------------------
    alpha0 = 0.5                # base step size for the exponentiated update
    beta1 = 0.9                 # Adam first‑moment decay
    beta2 = 0.999               # Adam second‑moment decay
    eps = 1e-8
    beta_start = 1.0
    beta_end = 5000.0
    beta_ramp_iters = 150_000   # linear increase of the smoothing factor

    # ------------------------------------------------------------------
    # Run a series of independent short optimisation runs until the
    # remaining time does not allow another full run.
    # ------------------------------------------------------------------

    while time.time() + min_run_time < deadline:
        # ---- random start (Dirichlet) --------------------------------
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()

        # Adam moments (reset for each run)
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        best_a_run = a.copy()
        best_val_run = cheap_score(a)

        it = 0
        last_improve = 0
        max_iters = 200_000

        while time.time() < deadline and it < max_iters:
            # ----- convolution and smooth weights -----
            conv = _full_convolution(a)
            max_conv = conv.max()
            delta = conv - max_conv                      # ≤ 0
            beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)

            a = np.maximum(a, 1e-15)

            if (it % 500) == 0:

                if cur_val < best_val_run - 1e-12:

                        best_a_run = a_ord.copy()

                m.fill(0.0)
                v.fill(0.0)
                last_improve = it
                print(f"[run perturb {it}] blended with random noise")
                # do not reset it – we keep counting iterations

        if best_val_run < global_best_val - 1e-12:

            global_best_a = best_a_run.copy()
            print(f"[global] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Fine‑grained hill‑climbing (with a tiny simulated‑annealing component)
    # ------------------------------------------------------------------
    cur_a = global_best_a.copy()
    cur_val = global_best_val

    hc_start = time.time()

        if rng.random() < 0.25:               # pure swap

            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                 # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.03 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # ---- re‑project onto the simplex ----
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = cheap_score(new_a)

        # ---- simulated‑annealing acceptance ----

    # ------------------------------------------------------------------
