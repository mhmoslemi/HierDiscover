"""
Propose a step‑function that minimises 2·n·max(conv(a,a)).

Algorithm
---------
1. **Initialisation**
   * Try to reuse the previously found sequence ``height_sequence_1`` if
     it exists, otherwise generate a set of hand‑crafted candidates:
     exponential, power‑law, cosine‑window families and a few Dirichlet
     draws.  The best of these (in terms of the exact objective) seeds the
     optimisation.

2. **Exponentiated‑gradient (EG) sub‑gradient descent**
   * Work on the probability simplex via log‑weights `log_w`.
   * At each iteration compute the exact self‑convolution (FFT) and the
     index `k_max` of its largest entry.
   * The sub‑gradient of the max‑convolution entry is
       `grad[i] = 2·a[k_max-i]` for indices that contribute.
   * Update the log‑weights multiplicatively:
       `log_w ← log_w – η·grad`  (η decays slowly).
   * Occasionally inject a small random jitter and, after a long period
     without improvement, re‑initialise around the current best solution.

3. **Uniform mixing (tiny)**
   * Try convex combinations (1‑α)·a + α·uniform for several α∈[0,0.02] and
     keep the best.

4. **Return**
   * The best probability vector (sum = 1) is returned as a plain Python
     list of floats.  The harness will invoke ``evaluate_sequence`` on it.

The whole search respects the 1000 s budget, runs on a single CPU and
uses only NumPy (no external files or network access).  It improves on
the previous heuristic by employing multiplicative updates, a finer
parameter sweep and a simpler yet effective refinement stage.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simplex mixing with uniform (tiny α values)
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Try (1‑α)·a + α·uniform for several α and keep the best."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val


# ----------------------------------------------------------------------
# Parameter‑family generator (used when no previous solution is present)
# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, rng: np.random.Generator):
    """Yield a sequence of candidate vectors built from simple parametric families."""
    # 1. Exponential (U‑shaped) family: a_i ∝ r**i + r**(n‑1‑i)
    for r in np.linspace(0.55, 0.9999, 30):
        i = np.arange(n, dtype=np.float64)
        a = np.power(r, i) + np.power(r, n - 1 - i)
        yield a

    # 2. Power‑law (edge‑peaked) family: a_i ∝ ((i+1)*(n-i))**(-p)
    for p in np.linspace(-4.0, 4.0, 30):
        i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
        base = i * (n + 1 - i) + 1e-12                     # avoid zero
        a = np.power(base, -p)
        yield a

    # 3. Cosine‑window family with shape exponent β
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))      # Hann window (non‑negative)
    for beta in np.linspace(0.5, 4.0, 20):
        a = np.power(w, beta)
        yield a

    # 4. Random Dirichlet draws – a few quick samples
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        yield a


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient descent (mirror descent on simplex)
# ----------------------------------------------------------------------
def _eg_subgradient_descent(
    a0: np.ndarray,
    n: int,
    start: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) descent using the sub‑gradient
    of the max‑convolution entry.  Returns the best vector seen.
    """
    eps = 1e-12
    log_w = np.log(a0 + eps)                 # initialise log‑weights
    eta = 0.5                                 # initial step size
    decay = 0.99995                           # geometric decay
    best_a = a0.copy()
    best_val = _objective_value(best_a)

    no_improve = 0
    IMPROVE_LIMIT = 30000
    it = 0

    while time.time() < deadline:
        # --- compute current probability vector ---
        max_log = np.max(log_w)
        w = np.exp(log_w - max_log)          # stabilised exponentiation
        a = w / w.sum()

        # --- evaluate objective ---
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[EG  {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # --- sub‑gradient of the maximal convolution entry ---
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n, dtype=int)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # --- multiplicative update on the simplex ---
        log_w -= eta * grad
        eta *= decay

        # --- occasional jitter to escape shallow plateaus ---
        if rng.random() < 0.003:
            log_w += rng.normal(scale=0.005, size=n)

        # --- stagnation handling ---
        if no_improve > IMPROVE_LIMIT:
            # re‑centre around the current best with a moderate perturbation
            log_w = np.log(best_a + eps)
            log_w += rng.normal(scale=0.02, size=n)
            eta = 0.5
            no_improve = 0
            print(f"[EG  {it:7d}] perturb after stagnation")

        it += 1

        # safety break if we are within a small margin of the deadline
        if time.time() > deadline - 0.2:
            break

    return best_a, best_val
