"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1️⃣  Initialise a candidate:
    * Interpolate the historic best ``height_sequence_1`` if available.
    * Augment with a few Dirichlet draws and a random Beta‑family sweep.
2️⃣  Refine by *projected sub‑gradient descent* on the exact objective
    f(a)=2·n·max(conv(a,a)).  The sub‑gradient is taken from all
    maximising indices and a variance‑penalty term is added (with a
    schedule that starts strong and vanishes).
3️⃣  Perform a fast *simulated‑annealing hill‑climber* that repeatedly
    moves a small amount of mass from an index that contributes heavily
    to the current maximal convolution entry to an index that contributes
    little.  Acceptance follows a Metropolis rule with a temperature that
    cools linearly.
4️⃣  Return the best sequence seen (projected onto the simplex).  The
    final score is obtained via the provided ``evaluate_sequence`` call.

The whole optimisation respects the 1000 s budget and works for any
reasonable length (chosen here as 4000 for a good speed/quality trade‑off).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Variance of the convolution and its gradient (exact, no scaling)
# ----------------------------------------------------------------------
def _var_and_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns   var(b) = mean((b‑avg)²)   and   ∂var/∂a  (no extra scaling).
    Here b = a*a and avg = 1/(2n‑1) because Σb = (Σa)² = 1.
    """
    n = a.shape[0]
    b = _full_convolution(a)
    m = 2 * n - 1
    avg = 1.0 / m
    v = b - avg
    var = np.mean(v * v)

    # gradient: (4/m) * cross‑correlation of a with v
    conv_ab = _conv_fft(a, v[::-1])
    segment = conv_ab[n - 1:2 * n - 1]
    grad = (4.0 / m) * segment[::-1]
    return var, grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent (max‑conv) with variance penalty
# ----------------------------------------------------------------------
def _subgrad_descent(a0: np.ndarray,
                     deadline: float,
                     start_time: float,
                     best_a: np.ndarray,
                     best_val: float,
                     lam_start: float = 30.0,
                     lam_end: float = 0.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on
          f(a) = 2·n·max(conv(a,a)) + λ·Var(conv(a,a))
    λ is linearly annealed from lam_start to lam_end.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)

    it = 0
    idle = 0
    MAX_IDLE = 20000          # restart after long stagnation
    while True:
        now = time.time()
        if now >= deadline:
            break

        # recompute convolution & max
        b = _full_convolution(a)
        max_val = b.max()
        cur_val = float(2.0 * n * max_val)  # sum(a)=1 => denominator =1

        # update best if improved
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Subgrad {it:6d}] new best = {best_val:.6f}")
        else:
            idle += 1

        # progression for schedules
        progress = (now - start_time) / (deadline - start_time)
        lam = lam_start * (1.0 - progress) + lam_end * progress
        step = 0.6 / np.sqrt(it + 1)          # classic 1/√t schedule

        # ---- gradient of the max term (sub‑gradient) ----
        k_maxes = np.where(b == max_val)[0]
        grad_max = np.zeros_like(a)
        for k in k_maxes:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad_max[mask] += 2.0 * a[idx[mask]]
        grad_max /= max(1, len(k_maxes))

        # ---- variance gradient (exact) ----
        var_val, grad_var = _var_and_grad(a)

        # total gradient
        grad = grad_max + lam * grad_var

        # gradient step and projection onto simplex
        a = a - step * grad
        a = _project_simplex(a)

        # occasional jitter to escape flat regions
        if it % 4000 == 0 and np.random.rand() < 0.15:
            a += np.random.normal(scale=0.04, size=n)
            a = _project_simplex(a)

        # restart if stuck for a long time
        if idle > MAX_IDLE:
            print(f"[Subgrad {it:6d}] restart after {idle} idle steps")
            a = np.random.dirichlet(np.ones(n))
            idle = 0
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (fast local search)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               deadline: float,
               start_time: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from an index that contributes heavily
    to the current maximal convolution entry towards an index that
    contributes little.  Uses a decreasing temperature schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = float(2.0 * n * b.max())
    it = 0
    no_improve = 0
    IMP_LIMIT = 12000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # schedule
        progress = (now - start_time) / (deadline - start_time)
        max_frac = 0.7 * (1.0 - progress) + 0.001 * progress
        temperature = 0.02 * (0.5 ** progress)  # exponential cooling

        # find current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to this entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – prob. ∝ contribution
        if total_contrib > 0:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(a))

        # target index – bias towards low contribution
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + 1e-12)
        tgt_weights[src] = 0.0
        if tgt_weights.sum() > 0:
            tgt = np.random.choice(n, p=tgt_weights / tgt_weights.sum())
        else:
            tgt = (src + n // 2) % n

        # amount to move
        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0:
            continue

        # update a and recompute the convolution (full recompute – cheap enough)
        a[src] -= delta
        a[tgt] += delta
        b = _full_convolution(a)
        new_val = float(2.0 * n * b.max())

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if temperature > 0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:6d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMP_LIMIT:
            # small random perturbation around current best
            a = best_a.copy()
            a += np.random.normal(scale=0.03, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = float(2.0 * n * b.max())
            no_improve = 0
            print(f"[Hill {it:6d}] stagnation – perturb")
        it += 1

    return best_a, best_val

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # keep a small safety margin

    # problem size – a trade‑off between quality and speed
    n = 4000

        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        val = _objective(a0)

    except Exception:

    # Final reliable evaluation (mandatory)
    # ------------------------------------------------------------------
