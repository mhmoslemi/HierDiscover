"""
Propose a non‑negative step function (list of floats) that minimises the
`evaluate_sequence` objective.

Algorithm
---------
1. **Parameterisation** – the vector `a` is represented as a soft‑max of an
   unrestricted vector `x`.  This guarantees `a_i ≥ 0` and `∑ a_i = 1`
   (the sum appears in the denominator of the objective and can be fixed to
   1 without loss of generality).

2. **Smooth proxy for the max** – instead of minimising the non‑smooth
   quantity `max(conv)`, we minimise a smooth approximation

        f(x) = (1/β)·log( Σ_k exp(β·conv_k) ),

   where `conv = a * a` (linear convolution).  For large β the proxy tends to
   the true maximum, yet it stays differentiable.

3. **Gradient** – using the convolution structure the gradient of `f`
   w.r.t. `a` can be written in closed form:
        ∂f/∂a_i = 2·Σ_j a_j·w_{i+j},
   where `w_k = exp(β·conv_k) / Σ_m exp(β·conv_m)` .
   The gradient w.r.t. `x` follows from the soft‑max Jacobian:
        ∂f/∂x_i = a_i·(∂f/∂a_i – Σ_j a_j·∂f/∂a_j).

   All convolutions are performed with an FFT (O(n log n)).

4. **Optimisation** – we run a limited‑memory BFGS (`L‑BFGS‑B`) on the
   unconstrained variables `x`.  Several random restarts are performed
   (different initialisations and a few β‑values).  The best candidate seen
   by the *exact* `evaluate_sequence` function is kept.

5. **Local refinement** – after the L‑BFGS phase a cheap stochastic hill‑climbing
   (tiny mass transfers and random swaps) is applied until the time budget
   expires.

The whole routine respects the 1000 s budget, prints intermediate progress,
and finally returns the best list of heights found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT‑based utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(v: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution of a real vector `v`.
    Returns an array of length 2·len(v)‑1.
    """
    n = v.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fv = np.fft.rfft(v, size)
    conv = np.fft.irfft(fv * fv, size)[:out_len]
    # eliminate tiny negative round‑off errors
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_weighted(a: np.ndarray, w_rev: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to `a`.
    The input `w_rev` must be the reversed weight vector (length 2·n‑1).
    Returns g_i = 2·Σ_j a_j·w_{i+j}.
    """
    n = a.shape[0]
    conv_len = n + w_rev.shape[0] - 1          # = 3·n‑2
    size = _next_pow2(conv_len)
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(fa * fw, size)[:conv_len]
    start = w_rev.shape[0] - 1                 # = 2·n‑2
    seg = conv[start:start + n]                # length n
    return 2.0 * seg


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The function runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a few seconds margin

    # --------------------------------------------------------------
    # Problem size -------------------------------------------------
    # --------------------------------------------------------------
    # If a historical construction exists we reuse its length,
    # otherwise we fix a convenient power of two for fast FFTs.
    try:
        init = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.shape[0]
    except Exception:
        n = 2048  # power of two, fast FFT, enough degrees of freedom

    # Guarantee a minimal size for a reasonable search space
    if n < 1024:
        n = 1024

    print(f"[info] problem size n = {n}")

    # --------------------------------------------------------------
    # Helpers for the smooth proxy objective
    # --------------------------------------------------------------
    def smooth_max_and_grad(x_vec: np.ndarray, beta: float):
        """
        Given the unconstrained vector `x_vec`, compute the smooth max of
        the self‑convolution of the corresponding probability vector,
        together with its gradient w.r.t. `x_vec`.

        Returns
        -------
        f : float          – smooth approximation of max(conv)
        g_x : np.ndarray   – gradient ∂f/∂x (same shape as `x_vec`)
        """
        # soft‑max to obtain a on the simplex
        max_x = np.max(x_vec)
        exp_x = np.exp(x_vec - max_x)          # shift for stability
        a = exp_x / exp_x.sum()

        # convolution a*a
        conv = _full_convolution(a)            # length 2n‑1

        # smooth max via log‑sum‑exp
        beta_conv = beta * conv
        max_bc = np.max(beta_conv)
        exp_beta = np.exp(beta_conv - max_bc)
        sum_exp = exp_beta.sum()
        # f = (log Σ exp(β·conv) ) / β
        f = (np.log(sum_exp) + max_bc) / beta

        # softmax weights w_k = exp(β·conv_k) / Σ exp(...)
        w = exp_beta / sum_exp                  # shape (2n‑1,)
        w_rev = w[::-1]

        # gradient w.r.t a : g_a_i = 2 Σ_j a_j w_{i+j}
        g_a = _gradient_weighted(a, w_rev)      # already includes the factor 2

        # gradient w.r.t x using soft‑max Jacobian
        avg = np.dot(g_a, a)                    # scalar  Σ a_j g_a_j
        g_x = a * (g_a - avg)                  # elementwise
        return f, g_x

    # --------------------------------------------------------------
    # Multi‑restart L‑BFGS optimisation
    # --------------------------------------------------------------
    best_a = None
    best_val = float("inf")
    beta_schedule = [5.0, 10.0, 20.0, 40.0]   # a few β‑values to explore

    restart = 0
    while time.time() < deadline and restart < 15:
        beta = beta_schedule[restart % len(beta_schedule)]

        # random start (log‑softmax of an exponential vector)
        x0 = np.log(rng.exponential(scale=1.0, size=n) + 1e-12)

        # L‑BFGS‑B on the smooth proxy
        def fun(x):
            f, g = smooth_max_and_grad(x, beta)
            return f, g

        options = {"maxiter": 2500, "disp": False}
        res = minimize(fun, x0, method="L-BFGS-B", jac=True,
                       options=options)

        # Recover `a` from the optimiser's solution
        max_x = np.max(res.x)
        exp_x = np.exp(res.x - max_x)
        a_opt = exp_x / exp_x.sum()

        # Exact evaluation of the original objective
        val = evaluate_sequence(a_opt.tolist())
        print(f"[restart {restart}] β={beta:.1f}  "
              f"L‑BFGS f≈{res.fun:.6f}  eval={val:.6f}")

        if val < best_val:
            best_val = val
            best_a = a_opt.copy()
            print(f"   *** new best = {best_val:.6f}")

        # early exit if we are already very close to the theoretical optimum
        if best_val < 1.1:
            break

        restart += 1

    # --------------------------------------------------------------
    # Hill‑climbing refinement (tiny stochastic moves)
    # --------------------------------------------------------------
    if best_a is None:
        # fall back to a uniform vector (should never happen)
        best_a = np.full(n, 1.0 / n)
        best_val = evaluate_sequence(best_a.tolist())

    cur_a = best_a.copy()
    cur_val = best_val

    while time.time() < deadline:
        # with probability 0.3 perform a small mass transfer,
        # otherwise do a random swap of a few entries.
        if rng.random() < 0.3:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] < 1e-15:
                continue
            delta = cur_a[i] * 0.04 * rng.random()   # move ≤ 4 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            idx = rng.choice(n, size=4, replace=False)
            new_a = cur_a.copy()
            new_a[idx] = np.random.permutation(new_a[idx])

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            best_a = new_a.copy()
            best_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    # --------------------------------------------------------------
    # Final reporting
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_a.tolist()
