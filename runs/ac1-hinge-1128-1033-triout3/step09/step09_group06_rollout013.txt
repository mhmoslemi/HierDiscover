def _smoothed_descent_inc_beta(a_start, deadline):

  n = len(a)

  best_val = 2*n*(_full_convolution(a).max()) # compute objective directly.

  beta_schedule = [0.5, 1, 2, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
  total_iters = 5000  # maybe more
  it_per_beta = max(1, total_iters // len(beta_schedule))

  for beta in beta_schedule:
     eta = 0.9  # step size
     for it in range(it_per_beta):
        if time.time() >= deadline - 5:
            break
        conv = _full_convolution(a)
        shift = conv.max()
        exp_vals = np.exp(beta*(conv - shift))
        p = exp_vals / exp_vals.sum()
        grad = 2.0 * _gradient_weighted_peaks(a, p)
        delta = -eta * grad
        delta = np.clip(delta, -20.0, 0.0)

        if it % 20 == 0:

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s budget)
--------------------------------
1. **Initial restarts** – a handful of different seeds (uniform, sinusoid,
   exponential random, or a previously discovered construction).  All vectors
   are normalised to sum = 1 (the objective is scale‑invariant).

2. **Temperature‑increasing soft‑max descent** – minimise a smooth surrogate
   of the max‑convolution

        f_β(a) = (1/β)·log∑_k exp(β·b_k) ,   b = a*a .

   For each temperature β we perform mirror‑descent steps
        a ← a·exp(−η·∇f_β) ,   a ← Π_simplex(a) .
   β is increased (0.5 → … → 1e4) so that the optimisation first sees a
   smooth landscape and then focuses on the true maximum.  The gradient
   ∇f_β is obtained with a single FFT via the helper ``_gradient_weighted_peaks``.
   After each β‑stage we apply the anti‑symmetric re‑ordering, which never
   increases the convolution peak.

3. **Targeted local search** – after the descent we have a vector ``a`` and its
   full convolution ``c = a*a``.  The current worst shift ``s = argmax c`` is
   identified.  The pair of indices (i, s‑i) contributing most to that peak is
   located, and a small amount of mass is transferred from the larger of the
   two to a random distant position.  The convolution can be updated in O(n)
   time using slice operations, allowing thousands of cheap proposals.  Only
   moves that strictly decrease the maximum convolution value are kept.

4. **Final polishing** – one more run of the local search on the best
   candidate.

All heavy work (FFT‑based gradient steps, O(n) updates in the local search)
avoids calling ``evaluate_sequence``; the true objective is queried only
occasionally to report progress.  The routine respects the 1000 s limit and
returns the best sequence found.

The function ``propose_candidate`` is the entry point required by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT‑based convolution, projection, re‑ordering, …)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


def _gradient_weighted_peaks(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k·b_k w.r.t. a, where b = a*a and p ≥ 0, Σp = 1.
    Result: g_i = Σ_k p_k·a_{k‑i}   (indices out of range give 0).
    Computed with a single FFT.
    """
    n = a.shape[0]
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    # the slice corresponding to indices n‑1 … 2n‑2
    seg = conv[n - 1 : 2 * n - 1]
    g = seg[::-1]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0 , Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values opposite the smallest ones (pairwise)."""
    n = a.shape[0]
    s = np.sort(a)                       # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[-1 - i]                 # large → left side
        b[-1 - i] = s[i]                 # small → right side
    if n % 2:
        b[half] = s[half]
    return b


# ----------------------------------------------------------------------
# 1️⃣  Temperature‑increasing soft‑max descent
# ----------------------------------------------------------------------
def _softmax_descent(a_start: np.ndarray, deadline: float):
    """
    Mirror‑descent on a log‑sum‑exp surrogate of max(b) with an
    *increasing* temperature schedule.
    Returns the best vector (sum = 1) found and its true objective.
    """
    a = a_start.copy()
    n = a.shape[0]

    # current best
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"[descent init] objective = {best_val:.6f}")

    # temperature schedule – start smooth, finish very sharp
    beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0,
                     100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0, 10000.0]

    # total inner iterations – we target a few thousand FFT steps
    total_iters = 8000
    it_per_beta = max(1, total_iters // len(beta_schedule))

    for beta in beta_schedule:
        eta = 0.8                     # base step size
        for it in range(it_per_beta):
            if time.time() >= deadline - 5.0:
                break

            conv = _full_convolution(a)          # length 2n‑1
            shift = conv.max()
            # stable softmax
            exp_vals = np.exp(beta * (conv - shift))
            p = exp_vals / exp_vals.sum()        # Σp = 1

            # gradient of the smooth surrogate
            grad = 2.0 * _gradient_weighted_peaks(a, p)

            # exponential mirror‑descent step (keep a ≥ 0)
            delta = -eta * grad
            delta = np.clip(delta, -20.0, 0.0)    # avoid overflow in exp
            a = a * np.exp(delta)
            a = _project_simplex(a)               # enforce Σa = 1

            # occasional true‑objective check (cheap via max of conv)
            if it % 30 == 0:
                cur_max = conv.max()
                cur_val = 2.0 * n * cur_max
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[beta {beta:.1e} it {it:04d}] new best = {best_val:.6f}")

                # anti‑symmetric re‑ordering every few hundred steps
                if it % 300 == 0:
                    a_ord = _anti_symmetric_reorder(a)
                    conv_ord = _full_convolution(a_ord)
                    val_ord = 2.0 * n * conv_ord.max()
                    if val_ord < best_val:
                        best_val = val_ord
                        best_a = a_ord.copy()
                        a = a_ord.copy()
                        print(f"[reorder β {beta:.1e} it {it}] improved to {best_val:.6f}")

            # slowly decay step size – helps stability for large beta
            eta *= 0.9995

        # final re‑ordering after each temperature stage (cannot increase max)
        a = _anti_symmetric_reorder(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# 2️⃣  Targeted local search that directly reduces the maximum convolution entry
# ----------------------------------------------------------------------
def _targeted_local_search(a: np.ndarray, n: int, deadline: float):
    """
    Fast O(n) hill‑climbing that always reduces the current maximum of the
    self‑convolution.  The convolution is updated incrementally, so we can
    afford many thousands of proposals without calling ``evaluate_sequence``.
    Returns the (still normalised) vector and its objective.
    """
    conv = _full_convolution(a)
    cur_max = conv.max()
    best_val = 2.0 * n * cur_max
    best_a = a.copy()
    rng = np.random.default_rng()

    while time.time() < deadline - 2.0:
        # 1) locate the worst shift
        s = int(np.argmax(conv))

        # 2) find the pair (i, s‑i) that contributes most to that shift
        i_low = max(0, s - (n - 1))
        i_high = min(n - 1, s)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        prod = a[idx] * a[s - idx]
        i_max = int(idx[np.argmax(prod)])

        # 3) pick a distant receiver index
        j = rng.integers(0, n)
        while j == i_max:
            j = rng.integers(0, n)

        # 4) propose a small transfer from i_max to j
        max_delta = a[i_max] * 0.25                 # move at most 25 % of the source
        if max_delta < 1e-14:
            # source is essentially empty – pick another i
            continue
        delta = max_delta * rng.random()
        if delta <= 1e-16:
            continue

        # 5) incremental update of the convolution (using the *old* a)
        new_conv = conv.copy()
        # linear part: 2·δ·(shifted_a_at_j – shifted_a_at_i)
        new_conv[j:j + n] += 2.0 * delta * a
        new_conv[i_max:i_max + n] -= 2.0 * delta * a
        # quadratic part
        ii = 2 * i_max
        jj = 2 * j
        ij = i_max + j
        new_conv[ii] += delta * delta
        new_conv[jj] += delta * delta
        new_conv[ij] -= 2.0 * delta * delta

        # ensure non‑negativity (numerical safety)
        if new_conv.min() < -1e-12:
            continue

        new_max = new_conv.max()
        if new_max < cur_max - 1e-12:               # strict improvement
            # accept the move
            a[i_max] -= delta
            a[j] += delta
            conv = new_conv
            cur_max = new_max
            cur_val = 2.0 * n * cur_max
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[local] improved to {best_val:.6f}")

        # otherwise reject – try another random move

    return best_a, best_val
