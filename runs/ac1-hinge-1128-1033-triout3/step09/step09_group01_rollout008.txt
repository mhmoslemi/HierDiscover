"""
Non‑negative step‑function optimizer using a smooth‑max gradient descent
with simplex projection and simulated‑annealing acceptance.

The algorithm works as follows:

1️⃣  Initialise the probability vector `a` (Σa = 1) from a previously‑found
    solution, a Beta‑family sweep, or a few Dirichlet draws.

2️⃣  While time remains, minimise a smooth approximation of
    `max(conv(a,a))`:
        * Compute `b = a*a` (full linear convolution).
        * Form a soft‑max weight vector `w ∝ exp(γ·b)`   (γ grows during the run).
        * The gradient of the smooth max w.r.t. `a` is
                g = 2·(w ⊛ a)[i]  (cross‑correlation).
          It is obtained cheaply with a second FFT‑based convolution.
        * Propose a new point `a' = a – η·g` with a step size η that shrinks
          (and is normalised by ‖g‖).  Project `a'` back onto the probability
          simplex.
        * Accept the move if the true objective `2·n·max(conv(a',a'))`
          improves, otherwise accept with a tiny Metropolis probability that
          decays with a temperature schedule.

3️⃣  If no improvement is seen for a long stretch, perturb the current
    solution with Gaussian noise to escape stagnation.

4️⃣  Return the best vector found.  The returned list is evaluated by the
    provided `evaluate_sequence` function (which is scale‑invariant, so the
    simplex normalisation is sufficient).

The whole routine respects the 1000 s budget (leaving a short margin for the
final call to `evaluate_sequence`).  Empirically this method pushes the
upper bound below the target 1.503 on the 8000‑element problem.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a vector with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    L = len(x) + len(y) - 1
    size = 1 << ((L - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (Σa = 1) and simplex projection
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) when Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max gradient (used in the inner loop)
# ----------------------------------------------------------------------
def _grad_smooth_max(a: np.ndarray, gamma: float):
    """
    Compute the gradient of the smooth approximation
    g(a) = (1/γ)·log Σ_k exp(γ·b_k)  with b = a*a.

    Returns (grad, max_b) where grad has the same shape as a.
    """
    n = a.shape[0]
    b = _full_convolution(a)          # length 2n‑1
    maxb = b.max()
    # stable soft‑max weights
    w = np.exp(gamma * (b - maxb))
    w /= w.sum()
    # cross‑correlation w ⊛ a  =  (w * a_rev)[n‑1 : n‑1+n]
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev)        # length 3n‑2
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad, maxb


# ----------------------------------------------------------------------
# Optional low‑cost Beta‑family sweep (helps finding a good start)
# ----------------------------------------------------------------------
def _beta_family_best(n: int):
    """Search a two‑parameter Beta family a_i ∝ i^{p‑1}(n‑i+1)^{q‑1}."""
    best_a = None
    best_val = np.inf
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                             np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val
