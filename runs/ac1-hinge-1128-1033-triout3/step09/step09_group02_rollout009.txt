def _refine_exact_phase(a, n, start, deadline, best_a, best_val):
    a = a.copy()
    best_local_a = best_a.copy()
    best_local_val = best_val
    no_improve = 0
    step_factor = 0.4  # start moving up to 40% of source mass
    # reduce step factor gradually
    decay = 0.9995
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_local_val - 1e-12:
            best_local_val = cur_val
            best_local_a = a.copy()
            no_improve = 0
            print(f"[Refine {int(time.time()-start):6d}] new best = {best_local_val:.6f}")
        else:
            no_improve += 1

        # termination
        if no_improve > 150000:
            break

        # active set tolerance relative
        eps = max_b * 1e-7
        active_idx = np.where(b >= max_b - eps)[0]
        if len(active_idx) == 0:
            # no active set? just break
            break

        # compute weight per position
        w = np.zeros(n)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # idx = np.arange(i_start, i_end+1)
            # contribution = a[k - idx]
            # We can vectorize:
            idx = slice(i_start, i_end + 1)
            w[idx] += a[k - np.arange(i_start, i_end+1)]
        # Choose source and target
        src = np.argmax(w)
        tgt = np.argmin(w)
        if src == tgt or a[src] <= 1e-16:
            # random fallback
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))
        delta = a[src] * step_factor
        if delta <= 1e-16:
            # reduce step factor
            step_factor *= 0.5
            continue
        # Apply move
        a[src] -= delta
        a[tgt] += delta

        # update step factor
        step_factor *= decay
        if step_factor < 1e-6:
            step_factor = 1e-6
    return best_local_a, best_local_val

"""
search_step_function.py

A time‑bounded optimiser for the “max‑convolution” evaluation problem.

Strategy
--------
1. **Parametric sweep** – generate a diverse pool of candidate shapes
   (exponential, power‑law, beta, sinusoidal, piecewise‑constant,
   logistic, random Dirichlet) and keep the best normalised vector.

2. **Adam on a smooth surrogate** – minimise a log‑sum‑exp approximation
   of the max‑convolution.  The surrogate temperature ε is annealed from
   1e‑1 to 1e‑12 while the learning rate decays, giving a fast coarse
   descent.

3. **Targeted hill‑climbing** – greedily transfer a small fraction of
   mass from indices that contribute most to the current peak to
   indices that contribute least.

4. **Projected sub‑gradient descent** – use the exact sub‑gradient of
   the max‑convolution (active index) and project onto the probability
   simplex.

5. **Exact refinement** – for the remaining time an *active‑set*
   refinement is performed.  All indices that belong to the
   (approximately) maximal convolution entries are collected; a weight
   w_i = Σ_{k∈active} a_{k−i} is computed and a small amount of mass
   is moved from the index with the largest weight to the index with the
   smallest weight.  The move is accepted only if it lowers the true
   objective; otherwise it is rejected.  This loop repeatedly shrinks
   the peak and often pushes the bound below the previous record.

6. **Uniform mixing** – a final convex combination with a tiny amount of
   uniform mass (α∈{0.001,…,0.02}) can smooth out residual spikes.

The whole pipeline is allocated inside the 1000 s budget; intermediate
best values are printed for monitoring.  The returned sequence is
non‑negative, sums to 1 and can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple objective (sum of a is 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generate a good starting point by scanning a rich collection of families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best vector among a large set of parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a):
        nonlocal best_a, best_val
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    i = np.arange(n, dtype=float)
    i1 = i + 1.0

    # 1. Exponential tails (decreasing & symmetric)
    for r in np.linspace(0.995, 0.99999, 14):
        _try(r ** i)                        # decreasing
        _try(r ** i + r ** (n - 1 - i))     # symmetric U‑shape

    # 2. Power‑law tails
    for p in np.linspace(0.2, 6.0, 18):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Beta‑distribution shapes
    for a_param in np.linspace(0.2, 6.0, 7):
        for b_param in np.linspace(0.2, 6.0, 7):
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 4. Sine‑power family
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 6.0, 12):
        _try(base_sin ** p)

    # 5. Piecewise‑constant “high‑at‑edges”
    for k_frac in np.linspace(0.005, 0.15, 12):
        k = int(round(k_frac * n))
        if k == 0 or 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 6. Logistic tails (asymmetric + mirrored)
    for slope in np.linspace(0.003, 0.04, 6):
        a = 1.0 / (1.0 + np.exp(slope * (i - n / 2.0)))
        _try(a)
        _try(np.flip(a))

    # 7. Random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(40):
        a = rng.dirichlet(np.ones(n))
        _try(a)

    # 8. Interpolation of a previously found solution, if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth log‑sum‑exp surrogate of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int, start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp (smooth max) surrogate."""
    w = np.log(a0 + 1e-12)          # logits → simplex via soft‑max
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.003
    eps_start, eps_end = 1e-1, 1e-13

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 80_000

    it = 0
    while time.time() < deadline:
        # point on simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        #   ∂/∂a  2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        # convolution a * p_rev (full length 3n‑2)
        conv_ab = _full_convolution(np.concatenate([a, np.zeros_like(a)]))  # dummy to reuse
        # Instead use FFT directly for speed:
        n_a = a.shape[0]
        n_p = p_rev.shape[0]
        out_len = n_a + n_p - 1
        size = 1 << (out_len - 1).bit_length()
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p_rev, size)
        conv = np.fft.irfft(fa * fp, size)[:out_len]
        # keep the central segment that corresponds to indices 0 … n‑1 of the true gradient
        grad_a = 4.0 * n * conv[n_a - 1: 2 * n_a - 1][::-1]

        # back‑prop through the log‑softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.008:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.2
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy mass transfer that reduces the exact peak."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the active (max) entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose source proportional to contribution
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # choose a target that is weakly coupled to the max entry
        if np.random.rand() < 0.45:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            if w.sum() > 0.0:
                probs_j = w / w.sum()
                j = int(np.random.choice(n, p=probs_j))
            else:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (exact but cheap)
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb current best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray, n: int,
                       start: float, deadline: float,
                       best_a: np.ndarray, best_val: float,
                       target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    eta0 = 0.6 / n
    eta = eta0
    decay = 0.9994
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[SGD {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        k_max = int(np.argmax(b))

        # sub‑gradient (active index)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # step + projection onto simplex
        v = a - eta * grad
        a = _project_simplex(v)

        # decay step size
        eta = max(eta * decay, 1e-12)

        if no_improve > IMPROVE_LIMIT:
            # shake and restart
            noise = np.random.normal(scale=0.011, size=n)
            a = np.maximum(best_local_a + noise, 0.0)
            a /= a.sum()
            eta = eta0
            no_improve = 0
            print(f"[SGD {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase D – Exact active‑set refinement (mass transfer based on weights)
# ----------------------------------------------------------------------
def _refine_exact_phase(a0: np.ndarray, n: int,
                        start: float, deadline: float,
                        best_a: np.ndarray, best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Exact refinement using active‑set weights."""
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val

    step_factor = 0.4            # start moving up to 40 % of source mass
    decay = 0.9997
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000
    MIN_STEP = 1e-12

    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_local_val - 1e-12:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Refine {it:7d}] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # final shake and exit
            break

        # active set – indices whose value is within a tiny tolerance of the max
        eps = max_b * 5e-7
        active = np.where(b >= max_b - eps)[0]
        if active.size == 0:
            break

        # compute weight w_i = Σ_{k∈active} a_{k‑i}
        w = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)
            w[idx] += a[k - idx]

        # source = index with largest weight, target = smallest weight
        src = int(np.argmax(w))
        tgt = int(np.argmin(w))
        if src == tgt or a[src] <= MIN_STEP:
            # fall back to random move
            src = int(np.argmax(a))
            tgt = int(np.argmin(a))

        delta = a[src] * step_factor
        if delta < MIN_STEP:
            step_factor *= 0.5
            continue

        # Apply move and evaluate new objective (exact)
        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        b_new = _full_convolution(a_candidate)
        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-15:
            a = a_candidate
            cur_val = new_val
            no_improve = 0
            if new_val < best_local_val - 1e-12:
                best_local_val, best_local_a = new_val, a.copy()
        else:
            # reject move, shrink step size
            step_factor *= 0.7

        # decay step size slowly
        step_factor = max(step_factor * decay, MIN_STEP)
        it += 1

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Small polishing: mixture with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a, best_val = a, _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand
    return best_a, best_val

    # 5️⃣ Phase D – Exact active‑set refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _refine_exact_phase(
        best_a, n,
        start=phase_c_deadline,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=TARGET,
    )
    print(f"[Phase D] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Final polishing – tiny uniform mixing
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣ Report final score using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
