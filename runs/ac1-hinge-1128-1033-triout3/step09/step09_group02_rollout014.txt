def simulated_annealing(a, n, start, deadline, best_a, best_val):
    a = a.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_val = best_val
    best_a = best_a.copy()
    # parameters
    max_frac_start = 0.6
    max_frac_end = 0.001
    total_iter = 0
    no_improve = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        # temperature schedule: initial T high, e.g., 0.01*cur_val, decays to near 0.
        T0 = 0.05 * cur_val # maybe
        T = T0 * (1 - progress)**2  # decays quadratically

        # choose source index i based on contributions to current maximum
        # compute k_max = argmax b
        k_max = int(np.argmax(b))
        # compute contributions = a_i * a_{k_max-i}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # choose target index j: preferably with low contribution
        # define rev_contrib of the symmetric pair as above but for j?
        # We'll pick j randomly weighted inversely to contribution (or uniform)
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros(n)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        # Inverse weighting:
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0  # exclude i
        w_sum = w.sum()
        if w_sum > 0:
            probs_j = w / w_sum
            j = np.random.choice(n, p=probs_j)
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0:
            continue

        # compute new b via incremental update
        b_new = b.copy()
        # decrement contributions of losing mass i
        b_new[i:i+n] -= 2.0 * delta * a
        # increment contributions of gaining mass j (using old a before update)
        b_new[j:j+n] += 2.0 * delta * a
        # adjust self-interaction for i and j
        b_new[2*i] += delta * delta
        b_new[2*j] += delta * delta
        # cross term)
        b_new[i+j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # acceptance
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            # accept move; update a and b, cur_val
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-15:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA {total_iter:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasionally reproject to ensure sum stable and no negative due to rounding
        if total_iter % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        if no_improve > 200000:
            # perturb best and restart
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {total_iter:7d}] perturb after stagnation")

        total_iter += 1

    return best_a, best_val

"""
Simulated‑annealing optimiser for the step‑function problem.
We start from a diverse parametric sweep (exponential, power‑law, beta,
inverse‑sqrt, arcsine, piecewise constant) to obtain a good seed.
A temperature‑controlled hill‑climbing (mass transfer) phase then refines
the candidate by moving small amounts of probability mass while tracking
the exact max‑convolution.  The final result is blended with a tiny
fraction of the uniform distribution to smooth out any remaining spikes.
"""
import time
import math
import numpy as np

# convolution helpers
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv

def _objective_value(a: np.ndarray) -> float:
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _initial_candidate(n: int):
    """Search a rich set of parametric families for a good seed."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    def _try(a_raw):
        nonlocal best_a, best_val
        a = np.asarray(a_raw, dtype=np.float64)
        if a.ndim != 1 or a.shape[0] != n:
            return
        if np.any(a < 0):
            return
        s = a.sum()
        if s <= 0:
            return
        a = a / s
        val = _objective_value(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    i = np.arange(n, dtype=float)                # 0 … n‑1
    i1 = i + 1.0                                 # 1 … n

    # 1. Exponential tails (decaying and symmetric)
    for r in np.linspace(0.995, 0.999999, 20):
        _try(r ** i)                                   # left‑hand only
        _try(r ** i + r ** (n - 1 - i))                # symmetric

    # 2. Power‑law tails
    for p in np.linspace(0.1, 3.0, 30):
        _try(i1 ** (-p))
        _try(i1 ** (-p) + (n - i) ** (-p))

    # 3. Inverse‑sqrt (a special case of power‑law)
    _try(1.0 / np.sqrt(i1) + 1.0 / np.sqrt(n - i))    # symmetric

    # 4. Beta densities – fine grid around the U‑shape (a,b<1)
    beta_vals = np.linspace(0.1, 1.5, 15)
    for a_param in beta_vals:
        for b_param in beta_vals:
            a = (i1 ** (a_param - 1.0)) * ((n - i) ** (b_param - 1.0))
            _try(a)

    # 5. Arcsine distribution (Beta(0.5,0.5))
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    _try(arcs)

    # 6. Sine‑power family (half‑wave)
    base_sin = np.sin(np.pi * (i1) / (n + 1.0))
    for p in np.linspace(0.5, 5.0, 12):
        _try(base_sin ** p)

    # 7. Piecewise constant high‑edges shapes
    for k_frac in np.linspace(0.005, 0.1, 12):
        k = max(1, int(round(k_frac * n)))
        if 2 * k >= n:
            continue
        for high_frac in np.linspace(0.05, 0.4, 12):
            total_high = 2.0 * k * high_frac
            if total_high >= 1.0:
                continue
            low = (1.0 - total_high) / (n - 2 * k)
            a = np.full(n, low)
            a[:k] = high_frac
            a[-k:] = high_frac
            _try(a)

    # 8. Random Dirichlet draws
    rng = np.random.default_rng(123456)
    for _ in range(80):
        _try(rng.dirichlet(np.ones(n)))

    # 9. If a previous solution exists, interpolate it
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        _try(a0)
    except Exception:
        pass

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Mass‑transfer SA that directly works with the exact max‑convolution."""
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = best_a.copy()
    best_local_val = best_val

    max_frac_start, max_frac_end = 0.6, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        # temperature schedule – linear decay to tiny value
        T0 = 0.02 * cur_val          # start about 2 % of current value
        T = max(T0 * (1.0 - progress), 1e-12)

        # locate the index that is currently maximal
        k_max = int(np.argmax(b))

        # source index – weighted by its contribution to the maximal bin
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            # nothing to move from this index
            continue

        # target index – bias towards positions that affect the maximal
        # bin as little as possible
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0.0:
            probs_j = w / w.sum()
            j = int(np.random.choice(n, p=probs_j))
        else:
            j = (i + np.random.randint(1, n)) % n

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (exact)
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_local_val - 1e-15:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        # periodic re‑projection to guard against drift
        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        # restart after long stagnation
        if no_improve > IMPROVE_LIMIT:
            # perturb the best solution and continue
            a = best_local_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val

def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Blend a tiny amount of uniform mass – often smooths the peak."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Main entry point used by the harness.
    Produces a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine first builds a good seed by
    sweeping several parametric families, then refines it with a
    simulated‑annealing mass‑transfer phase, and finally applies a small
    uniform mixing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – trade‑off between resolution and speed
    n = 9000

    # ------------------------------------------------------------------
    # Phase 0 – broad parametric sweep
    # ------------------------------------------------------------------
    print("[Init] parametric sweep")
    best_a, best_val = _initial_candidate(n)
    print(f"[Init] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – simulated‑annealing refinement (≈80 % of remaining time)
    # ------------------------------------------------------------------
    phase1_deadline = deadline
    best_a, best_val = _simulated_annealing(
        best_a, n,
        start=start_time,
        deadline=phase1_deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
    )
    print(f"[SA] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final tiny uniform blend
    # ------------------------------------------------------------------
    best_a, best_val = _mix_with_uniform(best_a, n)
    print(f"[Final] best after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Evaluate with the provided wrapper (just to display the score)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()

"""
Simulated‑annealing optimiser for the step‑function problem.

1.  A wide parametric sweep (exponential, power‑law, beta, inverse‑sqrt,
    arcsine, sine‑power, piecewise‑constant, random Dirichlet) produces a
    strong seed.
2.  A temperature‑controlled mass‑transfer phase works with the exact
    max‑convolution: a small amount of probability mass is moved from an
    index that contributes heavily to the current peak to an index that
    contributes little.  The Metropolis rule lets the search escape local
    minima.
3.  Finally the candidate is blended with a tiny fraction of the uniform
    distribution – an inexpensive polish that often lowers the peak a
    little further.

The routine respects the required interface `propose_candidate` and
returns a list of non‑negative floats (Σ ≈ 1) ready for `evaluate_sequence`.
"""

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – exact)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Result length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:out_len]
    return conv


def _objective_value(a: np.ndarray) -> float:
    """Exact objective for a vector with Σ a = 1."""
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)

# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {w ≥ 0, Σ w = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Phase 1 – simulated‑annealing mass‑transfer refinement
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, n: int,
                         start: float, deadline: float,
                         best_a: np.ndarray, best_val: float,
                         target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Direct optimisation of the exact max‑convolution.
    A small amount of probability mass is moved from a source index i
    (large contribution to the current peak) to a target index j
    (small contribution).  Acceptance follows the Metropolis rule.
    """
    a = a0.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_local_a = best_a.copy()
    best_local_val = best_val

    max_frac_start, max_frac_end = 0.6, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # progress‐dependent parameters
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T0 = 0.02 * cur_val                       # start temperature ~2 % of value
        T = max(T0 * (1.0 - progress), 1e-12)     # linear cooling

        # index of the current maximal convolution bin
        k_max = int(np.argmax(b))

        # ---------- choose source index i ----------
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue  # nothing to move from this index

        # ---------- choose target index j ----------
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0                     # never move mass back to i immediately
        if w.sum() > 0.0:
            probs_j = w / w.sum()
            j = int(np.random.choice(n, p=probs_j))
        else:
            j = (i + np.random.randint(1, n)) % n

        if j == i:
            continue

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the full convolution -----
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a     # removal from i
        b_new[j:j + n] += 2.0 * delta * a     # addition to j
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_local_val - 1e-15:
                best_local_val = cur_val
                best_local_a = a.copy()
                print(f"[SA {it:7d}] new best = {best_local_val:.6f}")
                if best_local_val <= target_val:
                    break
        else:
            no_improve += 1

        # occasional re‑normalisation to curb accumulation of round‑off error
        if it % 5000 == 0:
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

            a = best_local_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {it:7d}] perturbing after stagnation")

        it += 1

    return best_local_a, best_local_val

# Small final polishing – blend with a tiny share of the uniform distribution
# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, n: int) -> tuple[np.ndarray, float]:
    """Convex combination with uniform mass; useful for a final tiny tweak."""
    best_a = a.copy()
    best_val = _objective_value(a)
    uniform = np.full(n, 1.0 / n)
    for alpha in (0.001, 0.003, 0.005, 0.010, 0.020):
        cand = (1.0 - alpha) * a + alpha * uniform
        cand /= cand.sum()
        val = _objective_value(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    return best_a, best_val
