"""
Propose a step‑function sequence that minimises the supplied
`evaluate_sequence` metric.

Algorithm Overview
------------------
1. **Parameter sweep** – generate a modest number of analytically
   motivated families (power‑law, exponential, beta‑type and the
   classical (1‑x)^(-1/2) series).  All sequences are normalised to
   sum = 1 and a fast FFT‑based score `2·n·max(conv)` is computed.
   The best few candidates for each length are kept.

2. **Smoothed‑max mirror descent** – starting from each kept candidate,
   minimise a *smooth* approximation of the max‑convolution:
   `M_λ(a)= (1/λ)·log Σ_k exp(λ·(a∗a)_k)`.  Its gradient is a
   weighted correlation of the current sequence with the soft‑max
   weights and can be obtained by two FFTs.  An exponentiated‑gradient
   (mirror) step is performed and the result is projected onto the
   probability simplex.

3. **Refinement with the exact max‑subgradient** – after the smooth
   phase the smoothing parameter λ is large, so the objective is
   essentially the true max.  A few hundred exponentiated‑gradient
   updates using the active‑set subgradient (as in the previous
   solution) polish the solution.

All heavy work uses only FFTs; the expensive `evaluate_sequence`
routine is called only for the current global best.  The whole
procedure respects the 1000 s wall‑clock limit and prints progress
messages.

The returned list is a non‑negative step‑function (sum is automatically
scaled to 1, which leaves the evaluation invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex using the algorithm of Duchi et al."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:     # all entries zero → return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution / score (identical to evaluate_sequence for sum=1)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) using an FFT of appropriate length."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic “square‑root” start (coefficients of (1‑x)^(-1/2))
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^(-1/2), normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Families of analytic shapes (used in the parameter sweep)
# ----------------------------------------------------------------------
def _power_law(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-α) , normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha)
    a /= a.sum()
    return a


def _exponential_decay(n: int, beta: float) -> np.ndarray:
    """a_i ∝ exp(-β·i) , normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-beta * i)
    a /= a.sum()
    return a


def _beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """a_i ∝ (i+1)^p · (n-i)^q , normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p * (n - i) ** q
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient (λ–softmax over convolution entries)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float) -> (np.ndarray, float):
    """
    Compute gradient of M_λ(a)= (1/λ)·log Σ_k exp(λ·(a∗a)_k)
    and return both the gradient and the (unsmoothed) fast score.
    """
    n = a.shape[0]
    # ---- convolution a∗a -------------------------------------------------
    size_conv = 1 << ((2 * n - 2).bit_length())
    Fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(Fa * Fa, size_conv)[: 2 * n - 1]

    # ---- soft‑max weights ------------------------------------------------
    mx = conv.max()
    # shift by max for numerical stability
    exps = np.exp(lam * (conv - mx))
    w = exps / exps.sum()               # Σ w_k = 1

    # ---- gradient = 2·corr(a, w) ---------------------------------------
    Lw = conv.size                     # = 2*n-1
    offset = Lw - 1                    # position of lag 0 in linear correlation
    size_corr = 1 << ((n + Lw - 2).bit_length())   # ≥ n+Lw‑1
    FA = np.fft.fft(a, size_corr)
    FW = np.fft.fft(w, size_corr)
    corr = np.fft.ifft(FA * np.conj(FW)).real
    grad = 2.0 * corr[offset: offset + n]

    fast = float(2.0 * n * mx)
    return grad, fast


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution (used for final polishing)
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k at the active indices `active_idx`.
    For each active k we add 2·a_{k-i} to component i whenever the term
    participates in the convolution.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    if active_idx.size > 1:
        grad /= float(active_idx.size)
    return grad

            pool.append(_power_law(n, alpha))

        # exponential decay family
        for beta in np.linspace(0.0005, 0.02, 20):
            pool.append(_exponential_decay(n, beta))

        # simple beta‑type shapes (few discrete combos)
        for p in (0.0, 0.5, 1.0):
            for q in (0.0, 0.5, 1.0):
                pool.append(_beta_shape(n, p, q))

        # mixtures of the square‑root series and a power‑law
        sqrt_seq = _sqrt_series(n)
        for mix in (0.2, 0.5, 0.8):
            for alpha in (0.4, 0.5, 0.6):
                pl = _power_law(n, alpha)
                mixed = mix * sqrt_seq + (1 - mix) * pl
                mixed /= mixed.sum()
                pool.append(mixed)

        # --------------------------------------------------------------
        # Score all pool members quickly and keep the best few.
        # --------------------------------------------------------------
        scored = []
        for a in pool:
            fast = _fast_score(a)
            scored.append((fast, a))
        scored.sort(key=lambda x: x[0])
        top_k = 7
        top_candidates = [a for _, a in scored[:top_k]]
        print(f"[init] best raw score among pool = {scored[0][0]:.9f}")

        # --------------------------------------------------------------
        # Smoothed‑max mirror descent for each promising candidate.
        # --------------------------------------------------------------
        for start_idx, a0 in enumerate(top_candidates):
            if time.time() >= deadline:
                break

            a = a0.copy()
            a = _project_simplex(a)                # guarantee feasibility
            cur_fast = _fast_score(a)
            _maybe_update_best(a, cur_fast)

            # ---------- phase 1 : smoothed‑max descent ----------
            lam = 0.5                # initial smoothing λ
            eta0 = 0.6               # initial step size
            max_iters = 8000
            it = 0
            while it < max_iters and time.time() < deadline:
                # increase λ slowly (more focus on the true max)
                lam = 0.5 + 0.001 * it          # grows to ≈8.5 after 8000 iters
                eta = eta0 / math.sqrt(1.0 + it)

                grad, fast_val = _smooth_gradient(a, lam)

                # mirror (exponentiated) update on the simplex
                log_a = np.log(a + 1e-30)
                a_new = np.exp(log_a - eta * grad)
                a_new = _project_simplex(a_new)

                # cheap evaluation with the fast score
                new_fast = _fast_score(a_new)

                # accept if we made progress (greedy)
                if new_fast <= cur_fast:
                    a = a_new
                    cur_fast = new_fast
                    _maybe_update_best(a, cur_fast)

                # occasional tiny random jitter to avoid stagnation
                if it % 2000 == 0 and it > 0:
                    jitter = rng.randn(n) * 2e-4
                    a = _project_simplex(a + jitter)
                    cur_fast = _fast_score(a)

                it += 1

            # ---------- phase 2 : exact max sub‑gradient polishing ----------
            # Now λ is large enough that the smooth objective is essentially the true max.
            # Perform a few hundred exponentiated‑gradient steps using only the active constraints.
            max_iters2 = 2000
            it2 = 0
            while it2 < max_iters2 and time.time() < deadline:
                # compute current convolution and the active set
                Fa = np.fft.rfft(a, size_conv)
                conv = np.fft.irfft(Fa * Fa, size_conv)[: 2 * n - 1]
                mx = conv.max()
                active = np.flatnonzero(conv >= mx - 1e-14)

                grad = _max_conv_gradient(a, active)

                eta2 = 0.12 / math.sqrt(1.0 + it2)   # smaller step for fine tuning
                log_a = np.log(a + 1e-30)
                a_new = np.exp(log_a - eta2 * grad)
                a_new = _project_simplex(a_new)

                new_fast = _fast_score(a_new)
                if new_fast <= cur_fast:
                    a = a_new
                    cur_fast = new_fast
                    _maybe_update_best(a, cur_fast)

                it2 += 1

            print(f"[finished n={n} start={start_idx}] local best score {cur_fast:.9f}")

    # ------------------------------------------------------------------
    # If nothing survived (should not happen) fall back to a uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(lengths[0], 1.0 / lengths[0])

    print(f"\n[final] overall best evaluation = {best_score:.9f}")
    return best_seq.tolist()
