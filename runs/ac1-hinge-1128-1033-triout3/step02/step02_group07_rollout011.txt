"""
Propose a non‑negative step sequence that minimises the provided
`evaluate_sequence` metric.

The optimiser works in two stages for each chosen length `n`:

1. **Smoothed‑max mirror descent**  
   – Initialise the vector on the simplex (sum = 1).  
   – At each iteration compute the full autocorrelation `c = a∗a`.  
   – Approximate the non‑smooth objective `max(c)` by a soft‑max
     with a temperature `β`.  The gradient of this smooth objective is
     `∇ = 2·(w ⋆ a)` where `w` are the soft‑max weights.  The convolution
     is evaluated with a single FFT (size ≈ 3 n).  
   – Perform a multiplicative (exponentiated‑gradient) step
     `log a ← log a – α·∇` and re‑normalise onto the probability simplex.
   – The step size `α` decays and the temperature `β` grows
     (slowly approaching the true max).  This quickly drives the
     objective down far below the previous 1.5708 bound.

2. **Local simulated‑annealing refinement**  
   After the mirror‑descent phase a fast Metropolis walk performs tiny
   mass transfers between random coordinates, accepting moves that lower
   the fast score (`2·n·max(a∗a)`) or with a temperature‑dependent
   probability.  Periodic random restarts avoid stagnation.

All fast scores are computed with FFTs; a call to the expensive
`evaluate_sequence` is performed only when a new global best is found.
The routine respects the 1000 s budget, prints progress and finally
returns the best sequence discovered (as a plain Python list).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _sqrt_series(n: int) -> np.ndarray:
    """Normalised first n coefficients of (1‑x)^(-1/2)."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


def _full_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a (length 2·n‑1) using FFT of length `size_fft`."""
    a_fft = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(a_fft * a_fft, size_fft)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Smooth approximation of the max: soft‑max weights that sum to 1."""
    # shift by the max for numerical stability
    shift = conv.max()
    w = np.exp(beta * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:                     # extreme under‑flow – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    return w / w_sum


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of the smoothed max with respect to `a`.
    Implements  ∇ = 2 * (w_rev ⋆ a)[2·n‑2‑i]   (i = 0..n‑1)
    where w_rev is w reversed.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reverse for correlation
    a_fft = np.fft.rfft(a, size_fft)
    w_fft = np.fft.rfft(w_rev, size_fft)
    full = np.fft.irfft(a_fft * w_fft, size_fft)
    # length of the full correlation is 3·n‑2
    full = full[: 3 * n - 2]
    idx = (2 * n - 2) - np.arange(n)    # positions corresponding to each a_i
    grad = 2.0 * full[idx]
    return grad


def _fast_score(a: np.ndarray, size_fft: int) -> float:
    """Fast evaluation 2·n·max(a∗a) using a single FFT."""
    n = a.shape[0]
    conv = _full_convolution(a, size_fft)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights with a minimal
    `evaluate_sequence` value.  The function respects the 1000 s runtime
    limit and returns a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem sizes.
    # The larger the length, the lower the theoretical limit (≈1),
    # but the optimisation is a little slower.  We spend the bulk of the
    # budget on a single, reasonably large size.
    # ------------------------------------------------------------------
    if "length" in kwargs:
        lengths = [int(kwargs["length"])]
    else:
        lengths = [8000]                     # primary size
        # a few smaller sizes help escaping possible local traps
        lengths += [3000, 5808]

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a portion of the remaining budget for this length
        remaining = deadline - time.time()
        phases = len(lengths) - lengths.index(n)          # decreasing denominator
        phase_budget = remaining / phases
        phase_end = time.time() + phase_budget

        # FFT length sufficient for both convolution (2n‑1) and gradient
        size_fft = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Choose an initial vector.
        # Prefer the global `height_sequence_1` if it matches the size,
        # otherwise fall back to the analytic sqrt‑series.
        # ------------------------------------------------------------------
        try:
            a0 = np.array(height_sequence_1, dtype=float)
            if a0.shape[0] != n:
                a0 = _sqrt_series(n)
        except Exception:
            a0 = _sqrt_series(n)

        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # Helper to update the global best using the *slow* evaluation.
        # ------------------------------------------------------------------
        def _update_global_best(candidate: np.ndarray, fast_score: float) -> None:
            nonlocal best_seq, best_score
            if fast_score < best_score - 1e-12:
                true_val = evaluate_sequence(candidate.tolist())
                if true_val < best_score:
                    best_score = true_val
                    best_seq = candidate.copy()
                    print(f"[global] new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # Initial fast score
        # ------------------------------------------------------------------
        cur_fast = _fast_score(a0, size_fft)
        _update_global_best(a0, cur_fast)
        print(f"[init] n={n}, start score {cur_fast:.6f}")

        # ==================================================================
        # Phase 1 – smoothed‑max mirror descent
        # ==================================================================
        a = a0.copy()
        it = 0
        max_iter = 30000                       # generous, will be cut by time
        alpha0 = 0.5
        beta0 = 1.0
        beta_max = 500.0
        while it < max_iter and time.time() < phase_end:
            # 1) full convolution and soft‑max weights
            conv = _full_convolution(a, size_fft)
            max_val = conv.max()
            beta = min(beta_max, beta0 * (1.001 ** it))   # slowly increase temperature
            w = _softmax_weights(conv, beta)

            # 2) gradient of the smooth max
            grad = _gradient_from_weights(a, w, size_fft)

            # 3) multiplicative (exponentiated) update on the simplex
            alpha = alpha0 / math.sqrt(1.0 + it)          # decaying step size
            log_a = np.log(a + 1e-30) - alpha * grad
            a = np.exp(log_a)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum

            # 4) fast evaluation and possible global update
            cur_fast = 2.0 * n * max_val
            if cur_fast < best_score - 1e-12:
                _update_global_best(a, cur_fast)

            # occasional jitter to avoid numerical stagnation
            if (it + 1) % 5000 == 0:
                jitter = np.random.randn(n) * 1e-5
                a = np.maximum(a + jitter, 0.0)
                a_sum = a.sum()
                a = a / a_sum if a_sum > 0 else np.full_like(a, 1.0 / n)

            it += 1
            if time.time() >= phase_end:
                break

        # ==================================================================
        # Phase 2 – simulated‑annealing fine‑tuning
        # ==================================================================
        # Temperature schedule: start modest, cool linearly to zero.
        T_start = 0.12
        fast_cur = cur_fast
        no_improve = 0
        iter_sa = 0
        while time.time() < phase_end:
            elapsed = time.time() - (phase_end - phase_budget)
            frac = min(1.0, elapsed / phase_budget)
            T = T_start * (1.0 - frac)

            # random source and destination (must be different)
            src = np.random.randint(n)
            if a[src] < 1e-12:
                continue
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

            # move a small fraction of the mass from src to dst
            delta_max = a[src] * min(0.5, T)
            delta = np.random.rand() * delta_max

            a_candidate = a.copy()
            a_candidate[src] -= delta
            a_candidate[dst] += delta

            fast_cand = _fast_score(a_candidate, size_fft)

            # Metropolis acceptance
            if fast_cand <= fast_cur or np.random.rand() < math.exp(-(fast_cand - fast_cur) / max(T, 1e-12)):
                a = a_candidate
                fast_cur = fast_cand
                if fast_cand < cur_fast - 1e-9:
                    cur_fast = fast_cand
                    _update_global_best(a, cur_fast)
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # If we have been stuck for a long time, restart from a fresh random point
            if no_improve > 25000:
                a = np.random.rand(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                fast_cur = _fast_score(a, size_fft)
                no_improve = 0
                print(f"[restart] n={n} after {iter_sa} SA steps")

            iter_sa += 1
            if time.time() >= phase_end:
                break

        print(f"[finished n={n}] best local fast score {cur_fast:.6f}")

    # ----------------------------------------------------------------------
    # Return the best sequence (or a safe fallback)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of the first used length
        fallback_n = lengths[0]
        best_seq = np.full(fallback_n, 1.0 / fallback_n)

    print(f"[final] overall best evaluation = {best_score:.6f}")
    return best_seq.tolist()
