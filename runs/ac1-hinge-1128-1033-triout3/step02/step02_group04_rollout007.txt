"""
Simple simulated‑annealing / hill‑climbing search for a low‑objective
step‑function.  The objective to minimise is

    score = 2 * n * max(a * a) / (sum(a)**2)

where * denotes the discrete self‑convolution.  Because the objective
is invariant under scaling we keep the vector on the probability simplex
(sum = 1, a_i ≥ 0).  The algorithm repeatedly moves a tiny amount of
mass from one index to another, accepts the move if it improves the
maximum convolution value (or with a temperature‑dependent probability
to escape local minima) and keeps the best solution seen.  Occasional
jitter and a restart scheme help to explore the landscape.

A small local refinement with pairwise transfers is performed at the
end.  The returned list is already clipped to the range required by
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x | x_i ≥ 0 , Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: next power of two (used for zero‑padded FFT).
# ----------------------------------------------------------------------
def _next_power_of_two(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Helper: compute the maximum entry of the self‑convolution and the
# full convolution array (needed for the arg‑max index).
# ----------------------------------------------------------------------
def _max_convolution(p: np.ndarray, m_fft: int, n: int):
    """Return max(b) and the full convolution b = p * p (length 2n‑1)."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
    max_val = float(np.max(conv))
    return max_val, conv


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑objective step function.

    The routine works on the simplex (∑a_i = 1) and uses a
    simulated‑annealing style hill climbing with occasional restarts.
    The final sequence is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size and initial vector
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name‑defined]
    except Exception:
        n = 1153  # fallback length used in the benchmark

    # initialise from the previously found construction if available,
    # otherwise use a random probability vector.
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name‑defined]
        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total > 0.0:
            p = seq / total
        else:
            p = np.full(n, 1.0 / n)
    else:
        p = rng.random(n)
        p = p / p.sum()

    # --------------------------------------------------------------
    # 2. FFT preparation
    # --------------------------------------------------------------
    # we need a size large enough for a linear convolution of length 2n‑1
    m_fft = _next_power_of_two(2 * n - 1)

    # --------------------------------------------------------------
    # 3. Initial evaluation
    # --------------------------------------------------------------
    cur_max, cur_conv = _max_convolution(p, m_fft, n)
    cur_score = 2.0 * n * cur_max
    best_score = cur_score
    best_p = p.copy()

    print(f"Initial score (sum=1): {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing / hill climbing loop
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0  # leave a tiny safety margin
    iteration = 0
    last_improve_iter = 0

    # temperature schedule
    T_start = 1.0

    # after about 200 000 accepted moves we reach T_end.
    decay_factor = (T_end / T_start) ** (1.0 / 200_000)

    T = T_start

    # donor entry).  The fraction is reduced slowly together with the temperature.
    max_move_fraction = 0.2

    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # 4.1  Choose a move
        # ----------------------------------------------------------
        # Identify the shift that currently attains the maximum.
        k_max = int(np.argmax(cur_conv))

        # The set of indices that contribute to this shift.
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)

        donor_candidates = np.arange(i_low, i_high + 1)
        donor_weights = p[donor_candidates] * p[k_max - donor_candidates]
        w_sum = donor_weights.sum()
        if w_sum == 0.0:
            # fallback to uniform random donor
            i = rng.integers(i_low, i_high + 1)
        else:
            donor_weights /= w_sum
            i = rng.choice(donor_candidates, p=donor_weights)

        # Choose a recipient index that does *not* belong to the same shift
        # (this tends to break the large peak).  Simple rejection sampling.
        while True:
            j = rng.integers(0, n)
            if j < i_low or j > i_high:
                break

        # Amount to transfer (keeping non‑negativity)
        eps = max_move_fraction * T * p[i] * rng.random()
        if eps <= 0.0 or p[i] < eps:
            continue
        new_p = p.copy()
        new_p[i] -= eps
        new_p[j] += eps
        # renormalise – the total sum is already 1 but floating‑point drift
        # may appear, so we project back onto the simplex.
        new_p = _simplex_projection(new_p)

        # ----------------------------------------------------------
        # 4.2  Evaluate the candidate
        # ----------------------------------------------------------
        new_max, new_conv = _max_convolution(new_p, m_fft, n)
        new_score = 2.0 * n * new_max

        # ----------------------------------------------------------
        # 4.3  Acceptance test (simulated annealing)
        # ----------------------------------------------------------
        if new_score < cur_score:

        else:

            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            p = new_p
            cur_score = new_score
            cur_max = new_max
            cur_conv = new_conv

            # keep the best solution seen so far
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:7d} – new best: {best_score:.6f}")

        # ----------------------------------------------------------
        # 4.4  Temperature and step‑size decay
        # ----------------------------------------------------------
        T = max(T_end, T * decay_factor)
        max_move_fraction = max(0.01, max_move_fraction * decay_factor)

        # ----------------------------------------------------------
        # 4.5  Restart if we have stagnated for a long time
        # ----------------------------------------------------------
        if iteration - last_improve_iter > 30_000:
            # perturb the best solution and continue from there
            noise = rng.normal(scale=0.02, size=n)
            p = best_p + noise
            p = np.clip(p, 0.0, None)
            p = p / p.sum()
            p = _simplex_projection(p)
            cur_max, cur_conv = _max_convolution(p, m_fft, n)
            cur_score = 2.0 * n * cur_max
            last_improve_iter = iteration
            T = T_start
            max_move_fraction = 0.2
            print(f"Iter {iteration:7d} – restart (stagnation)")

    # --------------------------------------------------------------
    # 5. Final local refinement (pairwise transfers)
    # --------------------------------------------------------------
    p = best_p.copy()
    cur_max, cur_conv = _max_convolution(p, m_fft, n)
    cur_score = 2.0 * n * cur_max

    swap_attempts = 3000
    for _ in range(swap_attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j or p[i] < 1e-12:
            continue
        eps = 1e-4 * rng.random() * p[i]
        new_p = p.copy()
        new_p[i] -= eps
        new_p[j] += eps
        # projection is not needed – the move keeps the sum exactly 1
        new_max, _ = _max_convolution(new_p, m_fft, n)
        new_score = 2.0 * n * new_max
        if new_score < cur_score:
            p = new_p
            cur_score = new_score
            cur_max = new_max
            best_score = new_score
            best_p = p.copy()
            print(f"Swap improvement → {best_score:.6f}")

    print(f"Final best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Return the sequence in the required format
    # --------------------------------------------------------------
    return np.clip(best_p, 0.0, 1000.0).tolist()
