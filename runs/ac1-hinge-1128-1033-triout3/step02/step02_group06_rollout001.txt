    import time, numpy as np
    from scipy.optimize import linprog

    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0
    # Step 1: generate central binomial prefix
    b = _binomial_prefix(n)
    sum_b = b.sum()
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / sum_b
    f = b * scale

    # compute initial M
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    best_f = f.copy()
    best_M = M

    it = 0
    no_improve = 0
    max_no_improve = 200

    while time.time() < deadline and no_improve < max_no_improve:
        # Construct constraint matrix A
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f[i]
        # bounds vector
        b_ub = np.full(2*n-1, M, dtype=np.float64)

        # Solve LP max sum(g) => min -sum(g)
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print('LP failed, break')
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        # Scale g to have same target sum
        g = g0 * (target_sum / sum_g0)

        # convex combination
        new_f = (1 - t) * f + t * g
        conv_new = np.convolve(new_f, new_f)
        new_M = conv_new.max()

        if new_M < best_M - 1e-8:
            best_M = new_M
            best_f = new_f.copy()
            f = new_f
            M = new_M
            no_improve = 0
            print(f"[iter {it}] improved M={best_M:.6f}")
        else:
            no_improve += 1

        it += 1

    # final local search: random tweaks
    cur_f = best_f.copy()
    cur_M = best_M
    sigma = 0.05
    while time.time() < deadline:
        idx = np.random.randint(n)

"""
Propose a low‑score step function.

The evaluation function is

    score = 2 * n * max(f * f) / (sum(f) ** 2)

with `*` denoting the linear (zero‑padded) convolution.
If we normalise the sequence so that sum(f) = sqrt(2 n) the score
coincides with the maximum convolution value `M = max(f*f)`.  The
theoretical lower bound is `M >= 1` (or 2 n/(2 n‑1) for finite n), i.e.
the optimum score is about 1.

We start from the classic “central‑binomial’’ sequence
(b_k = C(2k,k)/4^k) which has a perfectly flat interior convolution
(`max = 1`) but a too small total mass (∑b_k ≈ 2√(n/π)).  By repeatedly
solving a linear programme of the form

    maximise   ∑ g_j
    subject to f * g ≤ M               (pointwise)
                g_j ≥ 0

and mixing the solution `g` back into `f` we can increase the total
mass while keeping the convolution ceiling `M` nearly unchanged.
When we renormalise after every step the objective value reduces.
The LP is solved with scipy.optimize.linprog (“highs” method) – it is
fast for a few hundred variables and thus we can perform many
iterations within the 1000 s budget.

The algorithm:
  1. pick a length `n` (default 800);
  2. build the central‑binomial prefix, scale it to
     `sum = sqrt(2 n)`;
  3. iterate
        – build the constraint matrix for `f * g ≤ M`;
        – solve the LP, obtain `g₀`;
        – scale `g₀` to the same normalisation;
        – replace `f ← (1‑t) f + t g` (t = 0.5);
        – recompute `M = max(f*f)`;
        – keep the best `f` seen.
  4. finish with a cheap stochastic hill‑climb that perturbs a few
     coordinates while re‑normalising.
  5. return the best sequence found.

If SciPy is unavailable the routine falls back to the pure
central‑binomial construction (the previous baseline).

The whole search respects the 1000 s time budget and returns a
non‑negative list of floats.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper: central‑binomial coefficients (C(2k,k)/4**k)
# ----------------------------------------------------------------------
def _binomial_prefix(max_n: int) -> np.ndarray:
    """Return the first `max_n` coefficients b_k = C(2k,k)/4**k."""
    b = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, max_n):
        # recurrence b_k = b_{k‑1} * (2k‑1)/(2k)
        cur = cur * (2.0 * k - 1.0) / (2.0 * k)
        b[k] = cur
    return b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     n: int = 800,
                     t: float = 0.5,
                     max_no_improve: int = 200,
                     **kwargs) -> List[float]:
    """
    Produce a sequence of non‑negative heights that (heuristically) minimises
    the evaluation function defined in `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for any stochastic steps.
    budget_s : int
        Total time budget in seconds (the caller ensures ≤1000).
    n : int
        Length of the candidate step‑function.  Larger values give a
        tighter theoretical bound but increase the LP size (O(n²)).
    t : float
        Mixing coefficient for the update f ← (1‑t)·f + t·g.
    max_no_improve : int
        Stop the LP loop after this many consecutive iterations without
        improvement.

    Returns
    -------
    List[float]
        The best candidate sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0           # safety margin

    # --------------------------------------------------------------
    # 1.  Initialise with a central‑binomial prefix, normalised so that
    #     sum(f) = sqrt(2*n).
    # --------------------------------------------------------------
    b = _binomial_prefix(n)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / b.sum()
    f = b * scale                                    # current sequence
    conv_f = np.convolve(f, f)
    M = float(conv_f.max())                          # current max‑convolution

    best_f = f.copy()
    best_M = M

    # --------------------------------------------------------------
    # Helper: build the matrix A such that (f * g)[k] = A[k]·g .
    # --------------------------------------------------------------
    def _build_A(seq: np.ndarray) -> np.ndarray:
        """Return matrix A of shape (2n‑1, n) for convolution with `seq`."""
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        # column i contains `seq[i]` on rows i … i+n‑1
        for i in range(n):
            A[i:i + n, i] = seq[i]
        return A

    # --------------------------------------------------------------
    # 2.  Iterative LP / mixing loop
    # --------------------------------------------------------------
    try:
        from scipy.optimize import linprog
        use_lp = True
    except Exception:  # scipy not available
        use_lp = False
        print("[fallback] scipy not available – using pure central‑binomial.")

    it = 0
    no_improve = 0
    while use_lp and time.time() < deadline and no_improve < max_no_improve:
        # build constraints for the current f
        A = _build_A(f)                               # (2n‑1) × n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise sum(g)  ⇔ minimise –sum(g)
        c = -np.ones(n, dtype=np.float64)

        # solve LP
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': deadline - time.time()})

        if not res.success:
            # LP failed (unlikely); break and fall back to a simple hill‑climb
            print(f"[lp] iteration {it}: LP failed ({res.status}) – exiting loop.")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        # If the LP did not manage to increase the mass we stop.
        if sum_g0 <= 0.0:
            print(f"[lp] iteration {it}: zero mass returned – exiting loop.")
            break

        # ----------------------------------------------------------
        # 2.1  Renormalise g to the same target sum as f
        # ----------------------------------------------------------
        g = g0 * (target_sum / sum_g0)

        # ----------------------------------------------------------
        # 2.2  Convex combination and recompute M
        # ----------------------------------------------------------
        new_f = (1.0 - t) * f + t * g
        conv_new = np.convolve(new_f, new_f)
        new_M = float(conv_new.max())

        # ----------------------------------------------------------
        # 2.3  Accept if we made progress
        # ----------------------------------------------------------
        if new_M + 1e-12 < best_M:          # strict improvement
            best_M = new_M
            best_f = new_f.copy()
            f = new_f
            M = new_M
            no_improve = 0
            print(f"[iter {it:03d}] improved M = {best_M:.9f}")
        else:
            no_improve += 1

        it += 1

    # --------------------------------------------------------------
    # 3.  Light stochastic refinement (Gaussian perturbations)
    # --------------------------------------------------------------
    cur_f = best_f.copy()
    cur_M = best_M
    sigma = 0.04                     # relative size of perturbations
    while time.time() < deadline:
        idx = np.random.randint(n)
        old_val = cur_f[idx]
        # Gaussian perturbation proportional to the current magnitude
        delta = np.random.randn() * sigma * max(old_val, 1e-12)
        cand = cur_f.copy()
        cand[idx] = max(0.0, old_val + delta)

        # Re‑normalise to keep the sum fixed (required for a fair score)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand = cand * (target_sum / cand_sum)

        conv_cand = np.convolve(cand, cand)
        cand_M = float(conv_cand.max())

        if cand_M < cur_M - 1e-12:
            cur_f = cand
            cur_M = cand_M
            if cand_M < best_M:
                best_M = cand_M
                best_f = cand.copy()
                print(f"[refine] new best M = {best_M:.9f}")

    # --------------------------------------------------------------
    # 4.  Final sanity – clip to the limits required by `evaluate_sequence`
    # --------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0).astype(np.float64)

    # Optional: report the real score (useful for debugging)
    try:
        real_score = evaluate_sequence(best_f.tolist())
        print(f"[final] length={n}, max_conv={best_M:.9f}, real_score={real_score:.9f}")
    except Exception:
        pass

    return best_f.tolist()
