"""
Propose a low‑objective step‑function.

The objective used by the evaluation framework is
    score = 2·n·max(conv_self) / (sum(a)^2)

Because the problem is homogeneous we fix the sum to 1 and minimise
    2·n·max(conv_self)

The algorithm consists of three phases

1. **Multiplicative (exponential) gradient descent** on the exact L∞‑norm.
   The worst convolution index `s*` is found, the sub‑gradient
   ∂max/∂a_i = 2·a_{s*‑i} (zero when the index is out of range) is computed
   and a mirror‑descent step  a_i ← a_i·exp(‑η·∂)  is performed.
   After each step we project back onto the probability simplex.

2. **Fine‑tuning with Adam** on a smooth log‑sum‑exp approximation of the
   maximum.  This is similar to the reference implementation but starts
   from the best point found in phase 1 and uses a reduced learning‑rate.

3. **Local pairwise swaps** – a cheap greedy search that tries to move a
   tiny amount of mass from one coordinate to another and keeps the move
   if the objective improves.

The whole search runs for the allotted time (default ≈ 1000 s) and returns
the best sequence found.  All numbers are non‑negative and ≤ 1000 as required
by `evaluate_sequence`.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex (Σx_i = 1, x_i ≥ 0)
# --------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of a vector onto the probability simplex. """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    # 1. Determine the problem size and initialise a probability vector
    # ----------------------------------------------------------
    try:
        # `height_sequence_1` may be defined in the outer scope
        init_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_seq.shape[0]
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        total = init_seq.sum()
        if total > 0.0:
            p = init_seq / total
        else:
            p = np.full(n, 1.0 / n)
    except Exception:
        n = 1153          # fallback length used in the benchmark
        p = rng.random(n)
        p = p / p.sum()

    # FFT size large enough for the full self‑convolution (length 2·n‑1)
    m_fft = 1 << ((3 * n - 2).bit_length())

    # ----------------------------------------------------------
    # 2. Phase‑1: exponential‑gradient descent on the exact max
    # ----------------------------------------------------------
    lr0 = 0.20                 # initial step size η
    decay = 0.9998             # geometric decay of η
    jitter_std = 5e-4          # size of occasional Gaussian jitter
    jitter_interval = 2000
    restart_interval = 20000  # restart if no improvement for this many iter
    best_score = np.inf
    best_p = p.copy()
    best_iter = 0
    iteration = 0
    lr = lr0
    deadline = time.time() + budget_s - 2.0  # small safety margin

    while time.time() < deadline:
        iteration += 1

        # ----- current objective (fast approximation) -----------------
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            best_iter = iteration
            print(f"Iter {iteration:6d} – new best (phase‑1): {best_score:.6f}")

        # ----- sub‑gradient of the max (exact) -----------------------
        s_max = int(conv.argmax())
        i_start = max(0, s_max - (n - 1))
        i_end = min(n - 1, s_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        counterpart = s_max - idx
        grad = np.zeros_like(p)
        grad[idx] = 2.0 * p[counterpart]          # ∂max/∂p_i

        # ----- exponential‑gradient update (mirror descent) -------
        #   p_i ← p_i · exp(‑η·grad_i)   then renormalise to Σp_i = 1
        p = p * np.exp(-lr * grad)
        # numeric safety – avoid all‑zero vectors
        s = p.sum()
        if s == 0.0 or not np.isfinite(s):
            p = np.full(n, 1.0 / n)
        else:
            p /= s

        # ----- occasional jitter to inject diversity ----------------
        if iteration % jitter_interval == 0:
            p = p + rng.normal(scale=jitter_std, size=n)
            p = np.clip(p, 0.0, None)
            p /= p.sum()

        # ----- decay learning‑rate ---------------------------------
        lr *= decay

        # ----- restart if stagnant ----------------------------------
        if iteration - best_iter > restart_interval:
            # restart near the current best solution with a modest perturbation
            p = best_p + rng.normal(scale=0.01, size=n)
            p = np.clip(p, 0.0, None)
            p /= p.sum()
            lr = lr0
            best_iter = iteration
            print(f"Iter {iteration:6d} – restart (phase‑1)")

    # ----------------------------------------------------------
    # 3. Phase‑2: fine‑tuning with Adam on a smooth log‑sum‑exp max
    # ----------------------------------------------------------
    p = best_p.copy()
    m1 = np.zeros_like(p)
    m2 = np.zeros_like(p)
    lr_adam = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 5.0            # smoothing parameter for log‑sum‑exp
    alpha_mul = 1.02
    alpha_max = 500.0
    max_adam_iters = 3000

    for it in range(1, max_adam_iters + 1):
        # stop if we run out of time
        if time.time() >= deadline:
            break

        # ----- smooth max and its gradient -------------------------
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

        z = alpha * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)          # stability trick
        w = exp_z / exp_z.sum()            # soft‑max weights (sum = 1)

        # correlation needed for the gradient (identical to the reference code)
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
        corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]

        # map correlation indices back to gradient of the original vector
        grad = 2.0 * corr_full[2 * n - 2 - np.arange(n)]

        # ----- Adam update -----------------------------------------
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

        m1_hat = m1 / (1.0 - beta1 ** it)
        m2_hat = m2 / (1.0 - beta2 ** it)

        step = lr_adam * m1_hat / (np.sqrt(m2_hat) + eps)
        p = p - step
        p = _simplex_projection(p)

        # ----- tighten the smooth approximation gradually ----------
        alpha = min(alpha * alpha_mul, alpha_max)

        # ----- evaluate current objective ----------------------------
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"Adam fine‑tune iter {it:5d} – new best: {best_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # ----------------------------------------------------------
    # 4. Phase‑3: cheap pairwise‑swap local refinement
    # ----------------------------------------------------------
    p = best_p.copy()
    swap_attempts = 5000
    for _ in range(swap_attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_move = 1e-4 * rng.random()
        if p[i] <= eps_move:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move

        p_fft = np.fft.rfft(new_p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
        new_score = 2.0 * n * conv.max()
        if new_score < best_score:
            p = new_p
            best_score = new_score
            best_p = p.copy()
            print(f"Swap improvement → {best_score:.6f}")

    # ----------------------------------------------------------
    # 5. Return the best sequence found (clipped to the allowed range)
    # ----------------------------------------------------------
    print(f"Final best score: {best_score:.6f}")
    # The evaluation routine will clip again, but we stay inside the limits.
    final_seq = np.clip(best_p, 0.0, 1000.0).tolist()
    return final_seq
