"""
Hybrid projected gradient descent (hard‑max subgradient) + pairwise swap
refinement for the step‑function evaluation problem.

The algorithm works on the simplex (∑a_i = 1, a_i ≥ 0).  The objective
2·n·max(conv(a,a)) is invariant to scaling, so we keep the sum fixed at
one.  At each iteration we:

1. Compute the self‑convolution a*a via FFT.
2. Identify all positions attaining the current maximum (the “active set”).
3. Build a uniform weight vector on the active set and obtain the
   sub‑gradient of the max‑operator by correlating a with this weight.
4. Perform an Adam update and project back onto the simplex.
5. Occasionally add a tiny random jitter to escape plateaus.

When the time budget is almost exhausted a cheap local search swaps a
small mass between two coordinates and keeps any improvement.

The method starts from the provided `height_sequence_1` if available,
otherwise from a random simplex point.  It typically improves the
previous best score 1.569… down towards the target 1.503 within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x | x_i ≥ 0 , Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Compute the evaluation score for a probability vector (sum = 1).
# ----------------------------------------------------------------------
def _compute_score(p: np.ndarray, m_fft: int, n: int) -> float:
    """Return 2·n·max(conv(p,p)) for a probability vector p."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑operator via the active set (hard‑max).
# ----------------------------------------------------------------------
def _hardmax_gradient(p: np.ndarray,
                      p_fft: np.ndarray,
                      conv: np.ndarray,
                      m_fft: int,
                      n: int) -> np.ndarray:
    """
    Compute a sub‑gradient of f(p)=max(conv(p,p)).
    The active set of indices attaining the maximum is used with uniform
    weights.
    """
    max_val = conv.max()
    # tolerance to collect indices that are numerically equal to the max
    tol = 1e-14
    active = np.where(conv >= max_val - tol)[0]

    # weight vector on the active indices (sum to 1)
    w = np.zeros(2 * n - 1, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size

    # correlation p * w_rev gives the gradient of the max (up to factor 2)
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)              # map to gradient components
    grad = 2.0 * corr_full[idx]
    return grad
