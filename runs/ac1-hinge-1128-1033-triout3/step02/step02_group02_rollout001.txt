    best = a.copy()
    best_score = compute_score(a)
    current = a.copy()
    cur_score = best_score
    temperature = 1.0
    eta = 0.1
    while time.time() < deadline:
        # compute gradient subgradient of max conv
        n = len(current)
        # compute conv via fft
        size = next_pow2(2*n-1)
        A = np.fft.rfft(current, n=size)
        conv = np.fft.irfft(A*A, n=size)[:2*n-1]
        max_val = conv.max()

        # compute subgradient as average over ks
        grad = np.zeros_like(current)
        for k in ks:
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range

        # each term counts each pair twice: for each (i,j) we added a_j for each k where i+j=k? hmm maybe need factor 2 from earlier. We'll multiply by 2.
        grad *= 2

        grad_proj = grad - grad.mean()

        cand = project_to_simplex(cand)

"""
Propose a non‑negative step function that minimizes `evaluate_sequence`.

Algorithm Overview
------------------
1. **Parametric sweep** –  Enumerate several simple symmetric families
   (U‑shaped power law, sum‑of‑tails, exponential, Gaussian, symmetric
   Beta) for a modest set of lengths and shape parameters.  For each
   candidate the maximal self‑convolution entry is obtained with a fast
   real‑FFT (O(n log n)).  The best candidate of this phase is kept as a
   warm start.

2. **Projected sub‑gradient descent with occasional random moves** –
   Starting from the warm‑start, iteratively improve the sequence on the
   probability simplex.
   * The sub‑gradient of the current maximal convolution entry is
     `grad_i = 2 * a_{k*‑i}` (zero if the index is out of range), where
     `k*` is an index attaining the maximum.
   * An additive step `a ← a – η·grad` is taken and the result is
     projected back onto the simplex using the O(n) algorithm of
     Duchi et al.
   * With probability `p_rand` a random mass‑transfer move (swap a
     fraction of mass between two positions) is performed instead,
     allowing the search to escape flat regions.
   * Temperature and step‑size are annealed geometrically.

3. **Final exact evaluation** – The best sequence found is passed to the
   provided `evaluate_sequence` once to obtain the official score, which
   is printed and returned.

The whole routine respects a total time budget (`budget_s`) and prints
progress information.  The returned value is a plain Python list of
floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Small utility returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def max_conv_fft(a: np.ndarray) -> float:
    """
    Compute the maximal entry of the self‑convolution a*a using a real FFT.
    The input may be any real vector; the result is the exact max up to
    floating‑point error.
    """
    n = a.shape[0]
    size = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def score_fft(a: np.ndarray) -> float:
    """Fast proxy for the official score (without the sum‑square denominator)."""
    n = a.shape[0]
    return 2.0 * n * max_conv_fft(a)


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the O(n) algorithm of
    Duchi, Shalev‑Shwartz, Singer & Chandra (2008).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i, n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def gaussian_distribution(n: int, sigma_factor: float) -> np.ndarray:
    """
    Symmetric Gaussian centered in the middle.
    sigma = sigma_factor * n  (σ relative to length).
    """
    sigma = max(sigma_factor * n, 1e-8)
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2) + 1e-12
    a /= a.sum()
    return a


def beta_symmetric_distribution(n: int, a_exp: float) -> np.ndarray:
    """
    Symmetric Beta( a_exp , a_exp ) shape (up‑down U‑shape if a_exp<1,
    centre‑peaked if a_exp>1).  a[i] ∝ (i+1)^{a_exp‑1} (n‑i)^{a_exp‑1}.
    """
    i = np.arange(1, n + 1)                 # 1 … n
    a = (i ** (a_exp - 1)) * ((n - i + 1) ** (a_exp - 1)) + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core refinement routine
# ----------------------------------------------------------------------


def refine_sequence(initial: np.ndarray,
                    deadline: float,
                    p_rand: float = 0.35,
                    eta0: float = 0.12,
                    temp0: float = 1.0,
                    cooling: float = 0.9996,
                    eta_cooling: float = 0.9999) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with occasional random mass‑transfer
    moves.  Returns the best sequence encountered and its fast score.
    """
    n = initial.shape[0]
    current = initial.copy()
    best = current.copy()
    cur_score = score_fft(current)
    best_score = cur_score

    temperature = temp0
    eta = eta0
    it = 0

    while time.time() < deadline:
        it += 1

        # Decide move type
        if np.random.rand() < p_rand:
            # ---- Random mass transfer move ----
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] <= 0.0:
                continue
            # move a random fraction of mass from i to j
            frac = current[i] * np.random.rand() * 0.5
            cand = current.copy()
            cand[i] -= frac
            cand[j] += frac
            cand = project_to_simplex(cand)
        else:
            # ---- Sub‑gradient step on the maximal convolution entry ----
            size = next_pow2(2 * n - 1)
            A = np.fft.rfft(current, n=size)
            conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
            max_val = conv.max()
            kstar = int(conv.argmax())

            # gradient of the max entry (sub‑gradient)
            # grad_i = 2 * current[kstar - i]  if inside range else 0
            idx = kstar - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(current)
            grad[mask] = 2.0 * current[idx[mask]]

            # Project gradient onto the tangent space of the simplex
            # (i.e. make it sum to zero)
            grad -= grad.mean()

            # Additive step + projection
            cand = current - eta * grad
            cand = project_to_simplex(cand)

        # Evaluate candidate (fast proxy)
        cand_score = score_fft(cand)

        # Simulated‑annealing style acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = math.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                print(f"Iter {it:7d}: new global best (fast) = {best_score:.6f}")

        # Anneal temperature and step size
        temperature = max(temperature * cooling, 1e-8)
        eta = max(eta * eta_cooling, 1e-5)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function that minimises

    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for the final exact evaluation
    deadline = start_time + budget_s - 5.0

    best_seq = None

    except NameError:

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 16000]

        # Exponential decay

                print(f"New best: n={n:5d}, shape=beta, a={bexp:4.2f}, fast score={sc:.6f}")

    # --------------------------------------------------------------

        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best fast score  : {best_score:.6f}")
    print(f"Best exact score: {exact_score:.6f}")
    print(f"Best length      : {len(best_seq)}")
