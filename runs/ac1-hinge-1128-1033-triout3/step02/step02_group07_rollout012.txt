    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _beta_shape(n, alpha):
    """Return symmetric beta(α,α) shape on {0,…,n‑1} (unnormalised)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # For α<1 the density has integrable singularities at the ends.
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # Numerical safety – replace NaNs (possible if α==0) by zero.
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    return a

def _fast_score(a, size_fft):
    """2·n·max(a∗a) using FFT of length size_fft."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def _max_conv(a, size_fft):
    """Maximum entry of a∗a (no factor 2·n)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return conv.max()

def propose_candidate(seed=0, budget_s=1000, length=4000):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm consists of three stages:
    1) a cheap grid search over symmetric beta‑shaped distributions;
    2) Adam‑style projected gradient descent on a smooth log‑sum‑exp
       approximation of the ``max``‑convolution objective (FFT‑based gradients);
    3) a short simulated‑annealing fine‑tuning phase.
    All heavy evaluations are performed with an FFT based surrogate; the
    expensive ``evaluate_sequence`` is called only when a new global best is
    discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    n = int(length)
    # FFT lengths needed throughout the run
    size_fft_small = 1 << ((2 * n - 1).bit_length())    # for a∗a
    size_fft_big   = 1 << ((3 * n - 2).bit_length())    # for w∗a_rev

    best_seq = None
    best_score = float("inf")          # true ``evaluate_sequence`` value

    # ------------------------------------------------------------------
    # 1)  Grid search over beta shapes (α parameter) and uniform mixing.
    # ------------------------------------------------------------------
    alphas = np.linspace(0.2, 1.2, 11)          # 0.2 … 1.2
    mix_factors = np.linspace(0.0, 1.0, 6)      # λ∈{0,0.2,…,1}
    uniform = np.ones(n, dtype=np.float64)

    for alpha in alphas:
        base = _beta_shape(n, alpha)
        base /= base.sum()
        for lam in mix_factors:
            cand = lam * base + (1.0 - lam) * uniform
            cand /= cand.sum()
            fast_val = _fast_score(cand, size_fft_small)
            if fast_val < best_score * 1.2:   # cheap pre‑filter
                # we have a candidate that may be interesting – evaluate
                true_val = evaluate_sequence(cand.tolist())
                if true_val < best_score:
                    best_score = true_val
                    best_seq = cand.copy()
                    print(f"[grid] new best {best_score:.6f} (α={alpha:.3f}, λ={lam:.2f})")
    # if grid did not produce anything (should not happen) start from uniform
    if best_seq is None:
        best_seq = uniform / n
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] uniform init score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2)  Adam projected‑gradient descent on a smoothed max objective.
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current iterate on the simplex
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05
    t = 0
    # temperature (softmax inverse temperature) schedule parameters
    tau0 = 1.0
    tau_max = 200.0
    tau_inc = (tau_max - tau0) / 20000.0    # reach tau_max after ~20k steps

    # keep a fast value for the current iterate
    cur_max = _max_conv(a, size_fft_small)
    cur_fast = 2.0 * n * cur_max

    while time.time() < deadline:
        t += 1
        # update softmax temperature (grows linearly, bounded)
        tau = min(tau_max, tau0 + tau_inc * t)

        # ------- compute convolution and softmax weights -------
        fa = np.fft.rfft(a, size_fft_small)
        p_full = np.fft.irfft(fa * fa, size_fft_small)[: 2 * n - 1]
        # softmax over (tau * p_full)
        p_max = p_full.max()
        exp_w = np.exp(tau * (p_full - p_max))
        w = exp_w / exp_w.sum()

        # ------- gradient w.r.t. a (2 * (w * a_rev)) -------
        a_rev = a[::-1]
        fw = np.fft.rfft(w, size_fft_big)
        fa_rev = np.fft.rfft(a_rev, size_fft_big)
        g_full = np.fft.irfft(fw * fa_rev, size_fft_big)[: 3 * n - 2]
        grad = 2.0 * g_full[n - 1 : n - 1 + n]   # slice length n

        # Adam update on the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate fast score of candidate
        new_max = _max_conv(a_new, size_fft_small)
        new_fast = 2.0 * n * new_max

        # accept if improvement (greedy).  Occasionally accept worsening
        # moves to escape shallow plateaus – we simply keep the better of
        # the two candidates.
        if new_fast < cur_fast:
            a = a_new
            cur_fast = new_fast
            cur_max = new_max
        else:
            # with small probability allow uphill move (temperature‑like)
            if np.random.rand() < 0.001:
                a = a_new
                cur_fast = new_fast
                cur_max = new_max

        # check for a new true global best
        if cur_fast < best_score * 1.01:      # cheap guard
            true_val = evaluate_sequence(a.tolist())
            if true_val < best_score:
                best_score = true_val
                best_seq = a.copy()
                print(f"[adam] new global best {best_score:.6f}")

        # periodic random jitter to avoid stagnation
        if t % 2000 == 0:
            a = _project_simplex(a + np.random.randn(n) * 1e-4)

        # abort if we are out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3)  Simulated‑annealing fine‑tuning (cheap moves).
    # ------------------------------------------------------------------
    # Use the best iterate from the Adam phase as starting point.
    a = best_seq.copy()
    cur_max = _max_conv(a, size_fft_small)
    cur_fast = 2.0 * n * cur_max
    no_improve = 0

    while time.time() < deadline:
        # temperature that slowly cools to zero
        remaining = deadline - time.time()
        T = max(1e-4, remaining / (budget_s * 2.0))

        src = np.random.randint(n)
        if a[src] < 1e-12:
            continue
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        delta = a[src] * np.random.rand() * min(0.5, T)
        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta

        cand_max = _max_conv(a_candidate, size_fft_small)
        cand_fast = 2.0 * n * cand_max

        if cand_fast < cur_fast or np.random.rand() < math.exp(-(cand_fast - cur_fast) / (T + 1e-12)):
            a = a_candidate
            cur_fast = cand_fast
            if cand_fast < best_score:
                true_val = evaluate_sequence(a.tolist())
                if true_val < best_score:
                    best_score = true_val
                    best_seq = a.copy()
                    print(f"[sa] new global best {best_score:.6f}")

        # simple restart heuristic
        no_improve += 1
        if no_improve > 50000:
            a = _project_simplex(np.random.rand(n))
            cur_fast = _fast_score(a, size_fft_small)
            no_improve = 0
            print("[sa] restart after many non‑improving steps")
    # ------------------------------------------------------------------
    # Return the best discovered sequence (as plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] best evaluation {best_score:.6f}")
    return best_seq.tolist()

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex using the algorithm of
    Duchi et al. (2008).  Works for any non‑negative array ``v`` and returns a
    new array with the same shape that satisfies sum(v)=1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
