"""
Propose a step‑function that minimizes `evaluate_sequence`.

The algorithm uses a smooth‐max approximation of the convolution peak and
Adam gradient descent on the log‑softmax parametrisation of the vector.
The loss is

    L(a) = 2·n·max_k (a∗a)_k   (the sum of a is kept equal to 1)

which is exactly the value returned by `evaluate_sequence` for a normalised
vector.  The max is approximated by the log‑sum‑exp (softmax) with a
temperature that grows linearly during the run, turning the smooth loss
into the true max as the optimisation proceeds.

The optimisation proceeds independently for a few problem sizes (default:
3000, 5808 and 8000).  For each size an Adam optimiser runs until the
allocated time slice expires, restarting from a random perturbation if no
improvement is seen for a while.  The best sequence seen across all runs
is returned (as a plain Python list).

All heavy work (FFT based convolutions, gradients) is performed with NumPy.
Only when a new global best is discovered is the expensive
`evaluate_sequence` routine called.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^(-1/2), normalised to sum 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sums to 1)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fast_score(a: np.ndarray, size_fft: int) -> float:
    """Return 2·n·max(a∗a) using an FFT of length `size_fft`."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      length: int | None = None,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total wall‑clock time (seconds) allowed for the search.
    length : int | None
        Optional fixed length for the step function.
    kwargs : dict
        Ignored (kept for compatibility with the previous interface).

    Returns
    -------
    list[float]
        The best sequence discovered (list of floats, sum ≈ 1).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the caller

    # ------------------------------------------------------------------
    # Problem sizes – if a fixed length is requested we use only that.
    # ------------------------------------------------------------------
    if length is not None:
        lengths = [int(length)]
    else:
        # modest variety – works well in practice and fits into the time budget
        lengths = [3000, 5808, 8000]

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper: evaluate with the external (slow) routine only when we have a
    # new global best.  This keeps the expensive call minimal.
    # ------------------------------------------------------------------
    def _maybe_update_best(candidate: np.ndarray, fast_val: float) -> None:
        nonlocal best_seq, best_score
        if fast_val < best_score - 1e-12:
            # the external function expects a plain list
            true_val = evaluate_sequence(candidate.tolist())
            if true_val < best_score:
                best_score = true_val
                best_seq = candidate.copy()
                print(f"[global] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Loop over the selected lengths.  Time is split equally (subject to
    # early termination if the global deadline is hit).
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        # remaining lengths including current one
        remaining_lengths = len(lengths) - lengths.index(n)
        alloc = remaining / remaining_lengths
        phase_end = time.time() + alloc

        print(f"[init] length {n}, allocated {alloc:.2f}s")

        # ------------------------------------------------------------------
        # Initialise the variables.
        #   - If a global variable `height_sequence_1` of the same length
        #     exists we reuse it (normalised).
        #   - Otherwise start from the analytic sqrt‑series with a tiny jitter.
        # ------------------------------------------------------------------
        try:
            a0 = np.asarray(height_sequence_1, dtype=float)
            if a0.shape[0] != n:
                raise ValueError
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            if a0_sum < 1e-12:
                raise ValueError
            a0 = a0 / a0_sum
        except Exception:
            a0 = _sqrt_series(n)
            a0 += np.random.randn(n) * 1e-4
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()

        # work in the log‑softmax domain: x such that a = softmax(x)
        x = np.log(a0 + 1e-30)               # ensures positivity

        # Adam state
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t = 0

        # Pre‑compute FFT lengths
        size_fft_conv = _next_pow2(2 * n - 1)          # for a∗a
        size_fft_corr = _next_pow2(3 * n - 2)          # for w∗a_rev

        # ------------------------------------------------------------------
        # Hyper‑parameters for the smooth‑max schedule and Adam learning rate.
        # ------------------------------------------------------------------
        lr = 0.5                         # base learning rate for Adam
        alpha_start = 0.5                # start temperature (more smoothing)
        alpha_end = 50.0                 # final temperature (close to true max)
        no_improve = 0
        max_no_improve = 3000            # restart after this many stagnant steps

        # ------------------------------------------------------------------
        # Main optimisation loop for this length.
        # ------------------------------------------------------------------
        while time.time() < phase_end:
            # --------------------------------------------------------------
            # Current probability vector and its autocorrelation
            # --------------------------------------------------------------
            # softmax with numeric stability
            a = np.exp(x - np.max(x))
            a_sum = a.sum()
            a = a / a_sum

            # FFT of a (zero‑padded) – reused for both convolution and gradient
            a_fft = np.fft.rfft(a, size_fft_corr)

            # full convolution a∗a (length 2n‑1)
            conv_full = np.fft.irfft(a_fft * a_fft, size_fft_corr)[: 2 * n - 1]
            conv_max = conv_full.max()

            # fast score (exact objective for sum=1)
            fast_cur = 2.0 * n * conv_max
            _maybe_update_best(a, fast_cur)

            # --------------------------------------------------------------
            # Smooth‑max weights w (softmax over convolution entries)
            # --------------------------------------------------------------
            # temperature grows linearly with the fraction of allocated time used
            elapsed = time.time() - (phase_end - alloc)
            frac = min(1.0, elapsed / alloc)
            alpha = alpha_start + (alpha_end - alpha_start) * frac

            # stable softmax on the shifted convolution
            shift = conv_full - conv_max
            w = np.exp(alpha * shift)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # --------------------------------------------------------------
            # Gradient of the smooth‑max objective w.r.t. a
            # --------------------------------------------------------------
            # grad_a = 2 * (w ⋆ a)  (cross‑correlation)
            w_fft = np.fft.rfft(w, size_fft_corr)
            a_rev_fft = np.fft.rfft(a[::-1], size_fft_corr)
            corr_full = np.fft.irfft(w_fft * a_rev_fft, size_fft_corr)[: 3 * n - 2]
            # slice the central n entries -> correlation values for each index
            grad_a = 2.0 * corr_full[n - 1: n - 1 + n]

            # scalar term needed for the log‑softmax chain rule
            c = np.dot(a, grad_a)

            # --------------------------------------------------------------
            # Gradient w.r.t. the log‑softmax parameters x
            # --------------------------------------------------------------
            grad_x = 2.0 * n * a * (grad_a - c)

            # --------------------------------------------------------------
            # Adam update
            # --------------------------------------------------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            x -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # --------------------------------------------------------------
            # Simple restart logic if we have been stagnant for a while
            # --------------------------------------------------------------
            if fast_cur < best_score - 1e-9:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= max_no_improve:
                    # re‑initialise from the analytical start with noise
                    a0 = _sqrt_series(n) + np.random.randn(n) * 1e-3
                    a0 = np.maximum(a0, 0.0)
                    a0 /= a0.sum()
                    x = np.log(a0 + 1e-30)
                    m.fill(0.0)
                    v.fill(0.0)
                    t = 0
                    no_improve = 0
                    print(f"[restart] length {n} after {t} steps")

        # end of inner while
        print(f"[finished] length {n}, best local fast score {fast_cur:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of the smallest length tried
        n_fallback = lengths[0]
        best_seq = np.full(n_fallback, 1.0 / n_fallback)
        print("[fallback] returning uniform vector")

    print(f"[final] overall best evaluation = {best_score:.6f}")
    return best_seq.tolist()
