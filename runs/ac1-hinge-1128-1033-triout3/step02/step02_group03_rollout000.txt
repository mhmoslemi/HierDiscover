"""
Algorithm: three‑phase optimization of a non‑negative step function.

*Phase‑0 – Adam‑based smoothed gradient descent*  
  A soft‑max approximation of the max‑operator (parameter α) is used.
  The gradient of   α·softmax(conv(p,p))   is obtained by a single
  correlation and the iterate is updated by Adam; after each step we
  project back onto the probability simplex.

*Phase‑1 – Exponentiated‑gradient sub‑gradient descent*  
  Once the smoothing becomes tight we switch to the exact sub‑gradient
  of the true max (the active convolution index).  The update is performed
  in the mirror‑descent (exponentiated‑gradient) fashion which keeps the
  vector on the simplex.  Adaptive learning‑rate decay and occasional
  jitter help to escape plateaus.

*Phase‑2 – Simulated‑annealing refinement*  
  A simple Metropolis‑Hastings move transfers a random amount of mass
  from one coordinate to another while preserving the ℓ₁‑norm.  Moves
  that increase the objective are accepted with probability exp(−Δ/T);
  the temperature T is slowly cooled.  This stochastic search often
  discovers improvements that deterministic first‑order methods miss.

The routine stops a few seconds before the allotted budget and returns
the best sequence (as a plain list) discovered so far.  All numbers are
non‑negative, sum ≥ 0.01, and lie in the interval [0,1000] – therefore
the returned list passes the security checks of ``evaluate_sequence``.
"""

import time
from typing import List

import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv(p: np.ndarray) -> float:
    """Return max(conv(p,p)).  n = len(p)."""
    return np.max(np.convolve(p, p))


def _smooth_gradient(p: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the soft‑max approximation of max(conv(p,p))."""
    n = p.shape[0]
    conv = np.convolve(p, p)                # length 2n‑1
    # soft‑max weights
    z = alpha * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()
    # gradient: 2 * (p * w) convolution, sliced to size n
    w_rev = w[::-1]
    corr = np.convolve(p, w_rev)           # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)         # mapping to gradient components
    grad = 2.0 * corr[idx]
    return grad


def _subgradient_max(p: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv(p,p)) at the active index."""
    n = p.shape[0]
    k_max = int(np.argmax(conv))
    grad = np.zeros_like(p)
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    if i_start <= i_end:
        idx_range = np.arange(i_start, i_end + 1)
        # For each i in [i_start, i_end] we have contribution 2 * p[k_max-i]
        grad[i_start:i_end + 1] = 2.0 * p[k_max - idx_range]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """Search for a sequence of heights that minimises the evaluation function."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size and initial point
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name‑defined]
    except Exception:
        n = 1153  # historic default

    # initialise a probability vector (sum == 1)
    if rng.random() < 0.5 and 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name‑defined]
        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total > 0.0:
            p = seq / total
        else:
            p = np.full(n, 1.0 / n)
    else:
        p = rng.random(n)
        p = p / p.sum()

    best_p = p.copy()
    best_score = 2.0 * n * np.max(np.convolve(p, p))

    # --------------------------------------------------------------
    # 2. Time management
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 3. Phase‑0: Adam + soft‑max smoothing
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    lr = 0.15
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(p)
    v = np.zeros_like(p)

    alpha = 1.0          # start smoothing
    alpha_mul = 1.02     # make soft‑max sharper each iteration

    it = 0
    while time.time() < deadline:
        it += 1

        # gradient of smooth objective
        grad = _smooth_gradient(p, alpha)

        # Adam update (on the simplex, so we project afterwards)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        p_next = p - step
        p_next = _simplex_projection(p_next)

        p = p_next

        # improve hyper‑parameters
        lr *= 0.9995
        alpha *= alpha_mul

        # evaluate
        cur_score = 2.0 * n * np.max(np.convolve(p, p))
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[Phase‑0] Iter {it:6d} – new best score: {best_score:.6f}")

        # occasional jitter to help escape flat regions
        if it % 500 == 0:
            noise = rng.normal(scale=0.0008, size=n)
            p = _simplex_projection(p + noise)

        # stop the soft‑max phase once α is large enough that the soft‑max
        # becomes a very tight approximation of the true max.
        if alpha > 5000.0:
            break

    # --------------------------------------------------------------
    # 4. Phase‑1: exponentiated‑gradient sub‑gradient descent
    # --------------------------------------------------------------
    eta = 0.6               # initial multiplicative step size
    eta_decay = 0.9997
    sub_it = 0
    no_improve = 0

    while time.time() < deadline:
        sub_it += 1

        conv = np.convolve(p, p)
        grad = _subgradient_max(p, conv)

        # mirror‑descent (exponentiated) update
        p = p * np.exp(-eta * grad)
        s = p.sum()
        if s == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p = p / s

        # jitter every few hundred steps
        if sub_it % 400 == 0:
            noise = rng.normal(scale=0.0007, size=n)
            p = _simplex_projection(p + noise)

        eta *= eta_decay

        cur_score = 2.0 * n * np.max(np.convolve(p, p))
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
            print(f"[Phase‑1] Iter {sub_it:6d} – new best score: {best_score:.6f}")
        else:
            no_improve += 1

        # random restart when stuck for a long time
        if no_improve >= 3000:
            p = best_p + rng.normal(scale=0.004, size=n)
            p = np.clip(p, 0.0, None)
            p = p / p.sum()
            no_improve = 0

        # early exit if we have already crossed the target
        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 5. Phase‑2: Simulated annealing refinement
    # --------------------------------------------------------------
    # Temperature schedule
    T = (best_score - 1.5) * 0.02 if best_score > 1.5 else 0.001
    cooling = 0.99995
    move_factor = 0.35   # up‑to this fraction of the source mass may be moved

    cur_score = 2.0 * n * np.max(np.convolve(p, p))

    sa_it = 0
    while time.time() < deadline:
        sa_it += 1

        # propose a move: transfer δ from i to j
        i = rng.integers(n)
        if p[i] == 0.0:
            continue
        # pick a different index j
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * p[i] * move_factor
        p_new = p.copy()
        p_new[i] -= delta
        p_new[j] += delta

        # compute new objective
        new_score = 2.0 * n * np.max(np.convolve(p_new, p_new))

        # metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            p = p_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                print(f"[SA] Iter {sa_it:6d} – new best score: {best_score:.6f}")

        # temperature cooling
        T *= cooling

        # occasional global perturbation to keep the chain alive
        if sa_it % 1200 == 0:
            noise = rng.normal(scale=0.001, size=n)
            p = _simplex_projection(p + noise)
            cur_score = 2.0 * n * np.max(np.convolve(p, p))

        # stop early if we already achieved the target
        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 6. Return the best sequence (plain Python list)
    # --------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Best score achieved: {best_score:.6f} (n={n})")
    return result
