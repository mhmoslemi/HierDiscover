"""
two‑phase optimizer for the step‑function bound

The algorithm works under the time budget supplied to ``propose_candidate``.
It first **flattens the auto‑convolution** by minimizing the L2 distance to
the uniform distribution (the average value of the convolution).  This is
performed with Adam on the simplex (Euclidean projection after each update).
A second phase refines the result by **minimising the true max** using a
soft‑max approximation that is gradually tightened – essentially the same
idea as the original code but with a smaller learning‑rate and Adam‑style
momentum.  Random jitter and occasional restarts help to escape plateaus.
The best sequence found (scaled to the classic normalisation
∑a_i = √(2 n)) is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0 , Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a sequence of heights that minimises the evaluation function."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name‑defined]
    except Exception:
        n = 1153  # historic default

    # ------------------------------------------------------------------
    # 2. Initial point (probability simplex)
    # ------------------------------------------------------------------
    if rng.random() < 0.5 and 'height_sequence_1' in globals():
        p = np.array(height_sequence_1, dtype=float)  # type: ignore[name‑defined]
        p = np.clip(p, 0.0, 1000.0)
        total = p.sum()
        if total > 0.0:
            p = p / total
        else:
            p = np.full(n, 1.0 / n)
    else:
        p = rng.random(n)
        p = p / p.sum()

    best_p = p.copy()
    # objective with sum(p)=1 is 2·n·max(conv)
    cur_conv = np.convolve(p, p)
    best_score = 2.0 * n * cur_conv.max() / (p.sum() ** 2)

    # ------------------------------------------------------------------
    # 3. Timing
    # ------------------------------------------------------------------
    start_time = time.time()
    total_budget = budget_s - 5.0            # safety margin
    deadline = start_time + total_budget
    phase1_deadline = start_time + 0.65 * total_budget  # ~65 % for the flattening phase

    # ------------------------------------------------------------------
    # Pre‑computed constants
    # ------------------------------------------------------------------
    idx = 2 * n - 2 - np.arange(n)          # indices used for gradient extraction
    L_len = 2 * n - 1
    avg = 1.0 / L_len                       # target uniform value of the convolution

    # ------------------------------------------------------------------
    # 4. Phase 1 – flatten the convolution (Adam on L2 loss)
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.05
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    t = 0
    stagnation = 0

    while time.time() < phase1_deadline:
        t += 1
        conv = np.convolve(p, p)                     # length 2n‑1
        diff = conv - avg                            # deviation from uniform
        # gradient of (1/L)·∑ diff²  w.r.t. p
        corr = np.convolve(p, diff[::-1])            # length 3n‑2
        grad_raw = 2.0 * corr[idx]                   # 2·∂(∑ w·conv)/∂p  with w = diff
        grad = (4.0 / L_len) * corr[idx]             # factor 4/L from the L2 loss derivation

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step
        p = _simplex_projection(p)

        # occasional jitter
        if t % 500 == 0:

        # evaluate current objective

        if cur_score + 1e-12 < best_score:

            best_p = p.copy()
            stagnation = 0

    # ------------------------------------------------------------------

        # schedule updates
        lr2 *= lr_decay
        alpha *= alpha_mul

        # evaluate
        cur_score = 2.0 * n * np.max(np.convolve(p, p)) / (p.sum() ** 2)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_p = p.copy()
            stagnation2 = 0
            total_it = t + it2
            print(f"[Phase 2] it {total_it:6d} – new best score: {best_score:.6f}")
        else:
            stagnation2 += 1

        if best_score <= 1.503:
            break

        # occasional restart if no progress for a while
        if stagnation2 >= 5000:
            p = best_p + rng.normal(scale=0.003, size=n)
            p = np.clip(p, 0.0, None)
            p = p / p.sum()
            stagnation2 = 0

    # ------------------------------------------------------------------
    # 6. Final random sampling (use any remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        p_rand = rng.random(n)
        p_rand /= p_rand.sum()
        cur_score = 2.0 * n * np.max(np.convolve(p_rand, p_rand)) / (p_rand.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_p = p_rand.copy()
            print(f"[Random] new best score: {best_score:.6f}")
        # thin out the loop
        if time.time() + 0.01 > deadline:
            break

    # 7. Rescale to the classic normalisation (∑a = √(2 n))
    # ------------------------------------------------------------------
    scale = np.sqrt(2.0 * n) / best_p.sum()
    result = np.clip(best_p * scale, 0.0, 1000.0).tolist()
    print(f"Best score achieved: {best_score:.6f}")
    return result
