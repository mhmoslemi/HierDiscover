def gradient_of_smooth_max(a, eta):
    n = len(a)
    # compute conv = a * a via FFT
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    # compute smoothed weights
    maxc = conv.max()
    # to avoid overflow, compute w = exp(eta * (conv - maxc))
    w = np.exp(eta * (conv - maxc))
    # compute conv of w with reversed a using FFT
    size2 = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size2)
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=size2)
    ga_full = np.fft.irfft(W * A_rev, n=size2)  # length size2
    # extract gradient slice
    start = n-1
    g = 2 * ga_full[start : start + n]
    return g

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that minimises the given evaluation metric.

Algorithm
---------
1. **Parametric sweep** –  Enumerate several simple families of
   symmetric “U‑shaped’’ and “beta‑shaped’’ distributions for a range of
   lengths and exponents.  For each candidate we evaluate the
   convolution peak with an O(n log n) FFT routine and keep the best.

2. **Exponentiated‑gradient refinement** –  Starting from the best
   candidate we minimise a smooth approximation of the objective
   ``max(conv)``:
       f(a) = (1/η)·log∑_k exp( η·(a*a)[k] )
   The gradient of f can be written as a convolution, so each iteration
   costs only a few FFTs.  An exponentiated‑gradient step (mirror descent
   on the probability simplex) moves mass away from indices that
   contribute most to the current peak.  A small probability of a random
   mass‑transfer move is added to escape shallow minima.  The learning
   rate is slowly decreased and the smoothing parameter η is slowly
   increased, effectively turning the smooth objective into the true
   max as optimisation proceeds.

3. **Exact evaluation** –  The best sequence found is finally passed to
   ``evaluate_sequence`` for the official score.

The whole search respects a 1000 s time budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] = (dist+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Two power‑law tails added together."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete beta‑like shape:
        a[i] ∝ (i+1)^{alpha‑1} · (n‑i)^{beta‑1}
    """
    i = np.arange(n)
    a = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1) + 1e-12
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones(n)
        a_sum = n
    a /= a_sum
    return a


def arcsine_distribution(n: int, eps: float = 1e-12) -> np.ndarray:
    """
    Discretised Beta(0.5,0.5) – the classic arcsine law.
    a[i] ∝ 1 / sqrt( (i+0.5)(n‑0.5‑i) ).
    """
    i = np.arange(n)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def mix_with_uniform(shape: np.ndarray, lam: float) -> np.ndarray:
    """
    Convex combination with the uniform distribution.
    lam=0 → pure uniform, lam=1 → pure shape.
    """
    n = shape.shape[0]
    uniform = np.ones(n) / n
    mixed = (1.0 - lam) * uniform + lam * shape
    mixed /= mixed.sum()
    return mixed
