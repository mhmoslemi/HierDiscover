"""
Propose a step‑function (non‑negative heights) that minimises

        M = 2·n·max(conv(a, a)) / (sum(a)**2)

Algorithm
---------

The search runs three phases inside the allotted time (≈ 1000 s).

1. **Adam on a smooth maximum** – we replace the max‑operator by a
   soft‑max with a temperature α.  The gradient of the smooth objective
   is obtained by a single weighted correlation, and the iterate is
   projected onto the probability simplex.  Adam‑style momentum and
   adaptive step‑sizes accelerate convergence; α is increased
   geometrically so that the approximation tightens.

2. **Exact sub‑gradient mirror descent** – once α is large enough we
   switch to the true sub‑gradient of `max(conv)`.  An exponentiated‑gradient
   (mirror‑descent) update respects the simplex constraints naturally.
   Small Gaussian jitter is added periodically to escape plateaus and
   occasional random restarts avoid stagnation.

3. **Simulated‑annealing refinement** – the best solution from phase 2 is
   further refined by a simple Metropolis‑Hastings walk: propose a
   zero‑sum Gaussian perturbation, project onto the simplex and accept
   it either if it improves the score or with probability exp(−Δ/T).
   The temperature and proposal scale are slowly cooled.

All convolutions are performed by FFT (zero‑padded to a fixed power‑of‑two
size) which makes each iteration cheap (≈ 10‑µs for n≈1150).  The best
sequence found is returned as a plain Python list; scaling of the
heights does not affect the evaluation because the objective is
scale‑invariant.
"""

import time
import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0 , Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small helper – next power of two greater or equal to x."""
    return 1 << (x - 1).bit_length()


def _convolve_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    The result length is len(x)+len(y)-1.
    """
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    z = np.fft.irfft(X * Y, size)
    return z[: len(x) + len(y) - 1]


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a sequence of non‑negative heights that minimise the objective."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)  # type: ignore[name‑defined]
        except Exception:
            n = 1153
    else:
        n = 1153

    # ------------------------------------------------------------------
    # 2. Initialise a probability vector (sum == 1)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        seq0 = np.array(height_sequence_1, dtype=float)  # type: ignore[name‑defined]
        seq0 = np.clip(seq0, 0.0, 1000.0)
        s = seq0.sum()
        if s > 0:
            p = seq0 / s
        else:
            p = np.full(n, 1.0 / n)
    else:
        p = rng.random(n)
        p /= p.sum()

    best_p = p.copy()
    # pre‑compute FFT size that works for every convolution we need
    _fft_size = _next_pow2(3 * n)          # enough for 3n‑2 length results

    def max_conv(a: np.ndarray) -> float:
        """Return max of the linear convolution a*a."""
        conv = _convolve_fft(a, a, _fft_size)
        return float(conv.max())

    best_score = 2.0 * n * max_conv(p)
    print(f"[Init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Time budget
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 4. Phase 1 – Adam on the smooth maximum
    # ------------------------------------------------------------------
    m = np.zeros_like(p)    # first moment
    v = np.zeros_like(p)    # second moment
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = beta1, beta2
    lr = 0.12               # initial learning rate
    alpha = 1.0             # smoothing factor for soft‑max
    alpha_mul = 1.006
    eps = 1e-8
    it = 0

    while time.time() < deadline:
        it += 1

        # --- smooth approximation of max(conv) ---
        conv = _convolve_fft(p, p, _fft_size)               # length 2n‑1
        z = alpha * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)                           # stability trick
        w = exp_z / exp_z.sum()                             # soft‑max weights (sum=1)

        # --- gradient of the smooth objective ---
        w_rev = w[::-1]
        corr = _convolve_fft(p, w_rev, _fft_size)          # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                    # map to gradient indices
        grad = 2.0 * corr[idx]

        # --- Adam update + simplex projection ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p_new = p - step
        p = _simplex_projection(p_new)

        # --- hyper‑parameter updates ---
        lr *= 0.9995
        alpha *= alpha_mul

        # --- evaluate ---
        cur_score = 2.0 * n * max_conv(p)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            print(f"[Phase 1] it={it:6d} – new best {best_score:.6f}")

        # early exit if already very good
        if best_score <= 1.45:
            break

    # ------------------------------------------------------------------
    # 5. Phase 2 – exact max sub‑gradient with exponentiated updates
    # ------------------------------------------------------------------
    p = best_p.copy()
    eta = 0.5
    eta_decay = 0.9997
    sub_it = 0
    no_improve = 0

    while time.time() < deadline:
        sub_it += 1

        conv = _convolve_fft(p, p, _fft_size)
        max_val = conv.max()
        # collect all indices that are (practically) maximal
        ties = np.where(conv >= max_val - 1e-12)[0]

        grad = np.zeros_like(p)
        for k_max in ties:
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                grad[i_start:i_end + 1] += 2.0 * p[k_max - idx]
        if len(ties) > 1:
            grad /= len(ties)

        # exponentiated (mirror‑descent) update
        p = p * np.exp(-eta * grad)
        s = p.sum()
        if s == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p = p / s

        # occasional jitter to shake the system
        if sub_it % 400 == 0:
            noise = rng.normal(scale=0.001, size=n)
            p = _simplex_projection(p + noise)

        eta *= eta_decay

        cur_score = 2.0 * n * max_conv(p)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
            print(f"[Phase 2] it={sub_it:6d} – new best {best_score:.6f}")
        else:
            no_improve += 1

        # random restart if we are stuck for a while
        if no_improve >= 5000:
            p = best_p + rng.normal(scale=0.005, size=n)
            p = np.clip(p, 0.0, None)
            p /= p.sum()
            no_improve = 0

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 6. Phase 3 – simulated‑annealing refinement
    # ------------------------------------------------------------------
    p = best_p.copy()
    cur_score = best_score
    T = 0.02                 # initial temperature
    sigma = 0.02             # proposal scale
    sa_it = 0

    while time.time() < deadline:
        sa_it += 1
        # propose a zero‑sum Gaussian perturbation and project back
        delta = rng.normal(scale=sigma, size=n)
        p_new = p + delta
        p_new = _simplex_projection(p_new)

        new_score = 2.0 * n * max_conv(p_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            p = p_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_p = p_new.copy()
                print(f"[SA] it={sa_it:6d} – new best {best_score:.6f}")

        # cooling schedules
        T *= 0.9995
        sigma *= 0.9995
        if T < 1e-8:
            T = 0.02
            sigma = 0.02

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 7. Final random sampling (tiny budget remaining)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        q = rng.random(n)
        q /= q.sum()
        sc = 2.0 * n * max_conv(q)
        if sc < best_score:
            best_score = sc
            best_p = q.copy()
            print(f"[Random] new best {best_score:.6f}")
        if time.time() + 0.02 > deadline:
            break

    # ------------------------------------------------------------------
    # 8. Return the best sequence (plain list, values already non‑negative)
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"[Result] best score achieved = {best_score:.6f}")
    return result
