def solve_lp_max_sum(A, M, n):
    c = -np.ones(n) # minimize negative sum
    bounds = [(0, None)]*n
    # inequality constraints: A @ x <= M
    # SciPy expects A_ub @ x <= b_ub
    res = linprog(c, A_ub=A, b_ub=np.full(A.shape[0], M), bounds=bounds, method='highs')
    if not res.success:
        return None, -np.inf
    x = res.x
    sum_x = np.sum(x)
    return x, sum_x

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    from scipy.optimize import linprog

    start = time.time()
    deadline = start + budget_s - 5

    best_seq = None
    best_score = float('inf')
    # Choose n
    for n in [150, 200, 250, 300]:
        if time.time() > deadline:
            break
        # compute constant-convolution coefficients up to n
        coeffs = np.empty(n, dtype=np.float64)
        cur = 1.0
        coeffs[0] = cur
        for k in range(1,n):
            cur = cur * (2*k-1)/(2*k)
            coeffs[k] = cur
        # scale to target sum sqrt(2n)
        target_sum = np.sqrt(2.0*n)
        sum_coeff = coeffs.sum()
        a = coeffs * (target_sum / sum_coeff)
        # compute initial M
        conv = np.convolve(a, a)
        M = conv.max()
        # iterative improvement
        iteration = 0
        while time.time() < deadline and iteration < 2000:
            iteration += 1
            # Build A
            A = np.zeros((2*n-1, n), dtype=np.float64)
            for i in range(n):
                A[i:i+n, i] = a[i]
            # solve LP
            c = -np.ones(n)
            b_ub = np.full(2*n-1, M)
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
            if not res.success:
                print(f"LP failed at n={n} iteration {iteration}")
                break
            b = res.x
            sum_b = b.sum()
            if sum_b <= target_sum * 0.999999:
                # cannot improve sum, maybe M optimal
                break
            # scale to target sum
            g = b * (target_sum / sum_b)
            # Mixed candidate
            t_vals = [0.5, 0.3, 0.2, 0.1, 0.05]
            improved = False
            for t in t_vals:
                a_candidate = (1 - t) * a + t * g
                M_candidate = np.convolve(a_candidate, a_candidate).max()
                if M_candidate < M - 1e-7:
                    a = a_candidate
                    M = M_candidate
                    improved = True
                    break
            if not improved:
                # no improvement with simple mixing, try replace if g better
                M_g = np.convolve(g, g).max()
                if M_g < M:
                    a = g
                    M = M_g
                else:
                    # break out; can't improve further
                    break
        # Compute final score
        seq = a.tolist()
        score = evaluate_sequence(seq)
        print(f"[n={n}] final score={score:.10f}, M={M:.10f}, length={n}")
        if score < best_score:
            best_score = score
            best_seq = seq

    # Return best_seq
    if best_seq is None:
        # fallback: simple constant shape length 20000 (previous) as last resort
        best_seq = coeffs[:20000].tolist()
    return best_seq

"""
Propose a candidate step‑function using an iterative convex‑mixing scheme.

The algorithm starts from the classic “constant‑convolution’’ coefficients
(a_k = C(2k,k)/4**k) scaled so that sum(a) = sqrt(2·n).  With this scaling the
objective 2 n·max(conv)/sum² equals the current maximal convolution value M.
For a fixed n the theoretical lower bound on M is 1 (because the average of
the convolution is sum²/(2n−1) = 1 after the scaling).  The constant‑convolution
construction yields M ≈ π/2 ≈ 1.57, far from the bound.

To improve, we repeatedly solve a linear program:

    maximise   Σ b_i
    subject to a * b ≤ M   (component‑wise convolution)
               b_i ≥ 0     (and a modest upper bound)

The LP gives a vector b whose sum is larger than that of a while still keeping
a·b ≤ M.  After rescaling b to the required total sum we mix it with a:
        a_new = (1‑t)·a + t·b   (t chosen from a small set)
The mixed vector has the same L₁ norm, but its self‑convolution peak
generally drops.  By iterating this step the peak M decreases towards the
optimal value.  The whole procedure runs inside the allotted time budget
and keeps the best sequence found (across several problem sizes).

The final sequence is returned as a plain list of non‑negative floats;
the evaluator will clamp values to the allowed range.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (placed at module level – no closures)
# ----------------------------------------------------------------------
def build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """Return the matrix A (shape (2n‑1, n)) such that A @ b = a * b (linear conv)."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column i contributes a[i] to rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A


def solve_max_sum_lp(A: np.ndarray, M: float, ub: float):
    """Linear program:
           maximise   Σ b_i
           subject to A @ b ≤ M, 0 ≤ b_i ≤ ub
       Returns (b, Σb) or (None, None) on failure.
    """
    n = A.shape[1]
    c = -np.ones(n)                     # linprog minimises, so use -Σb
    b_ub = np.full(A.shape[0], M)
    bounds = [(0.0, ub) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None, None
    b = res.x
    return b, b.sum()


def mixed_maximum(conv_aa: np.ndarray, conv_ag: np.ndarray,
                  conv_bb: np.ndarray, t: float) -> float:
    """Maximum value of the self‑convolution of (1‑t)·a + t·b
       using the three pre‑computed convolutions.
    """
    a_fac = 1.0 - t
    # (a_fac)²·conv_aa + 2·t·a_fac·conv_ag + t²·conv_bb
    mix = (a_fac ** 2) * conv_aa + 2.0 * t * a_fac * conv_ag + (t ** 2) * conv_bb
    return mix.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that yields a low value of the evaluation
    function.  The routine tries a few problem sizes (n≈150…300) and for each
    runs the iterative LP‑mixing process described above.  The best sequence
    found before the time budget expires is returned.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    best_seq = None
    best_score = float("inf")

    # Problem sizes to try – larger n gives more freedom but LP becomes slower
    for n in (150, 200, 250, 300):
        if time.time() > deadline:
            break

        # ----- initialise with the constant‑convolution coefficients -----
        a = np.empty(n, dtype=np.float64)
        cur = 1.0
        a[0] = cur
        for k in range(1, n):
            cur = cur * (2 * k - 1) / (2 * k)
            a[k] = cur

        target_sum = np.sqrt(2.0 * n)                # we enforce Σa = √(2n)
        a = a * (target_sum / a.sum())
        eps = 1e-12                                   # avoid exact zeros
        a = np.maximum(a, eps)

        # current maximal self‑convolution (also the current objective value)
        conv_a = np.convolve(a, a)
        M = conv_a.max()

        # upper bound for LP variables – a generous but finite cap
        ub = max(1.0, target_sum * 5.0)

        iteration = 0
        improved = True

        while time.time() < deadline and iteration < 2000 and improved:
            iteration += 1

            # ----- linear program: maximise Σb under a·b ≤ M -----
            A = build_convolution_matrix(a)
            b, sum_b = solve_max_sum_lp(A, M, ub)
            if b is None:
                # LP failure – abort this n
                print(f"[n={n}] LP failed at iteration {iteration}")
                break

            # If LP cannot increase the L1‑norm, we have reached a stationary point
            if sum_b <= target_sum * (1.0 + 1e-12):
                break

            # Rescale b to have the required total sum
            scale = target_sum / sum_b
            g = b * scale
            g = np.maximum(g, eps)            # keep positivity

            # Pre‑compute the three needed convolutions
            conv_ag = np.convolve(a, g)
            conv_g = np.convolve(g, g)

            # ----- find a mixing proportion that lowers the peak -----
            best_t = None
            best_M = M
            # a small set of candidate t values (more aggressive mixing first)
            for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02):
                M_t = mixed_maximum(conv_a, conv_ag, conv_g, t)
                if M_t < best_M - 1e-9:
                    best_M = M_t
                    best_t = t

            if best_t is not None:
                # perform the mixing
                a = (1.0 - best_t) * a + best_t * g
                M = best_M
                conv_a = np.convolve(a, a)   # recompute exact convolution for next round
                improved = True
                continue

            # If no mixing help, try a direct replacement when g itself is better
            M_g = conv_g.max()
            if M_g < M - 1e-9:
                a = g
                M = M_g
                conv_a = conv_g
                improved = True
                continue

            # No improvement achievable
            improved = False

        # ----- evaluate the final candidate for this n -----
        seq = a.tolist()
        score = evaluate_sequence(seq)
        print(f"[result n={n}] iterations={iteration}, final_score={score:.10f}, "
              f"final_M={M:.10f}, length={n}")

        if score < best_score:
            best_score = score
            best_seq = seq

    # Fallback: if everything failed, return a long constant‑convolution prefix
    if best_seq is None:
        MAX_LEN = 20000
        coeffs = np.empty(MAX_LEN, dtype=np.float64)
        cur = 1.0
        coeffs[0] = cur
        for k in range(1, MAX_LEN):
            cur = cur * (2 * k - 1) / (2 * k)
            coeffs[k] = cur
        best_seq = coeffs.tolist()

    # final clamping to the evaluator's range
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return best_seq
