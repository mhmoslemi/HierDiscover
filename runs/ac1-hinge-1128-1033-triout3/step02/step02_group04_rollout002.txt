"""
Sub‑gradient + simulated‑annealing optimizer for the “max self‑convolution”
objective used in the competition.  

The algorithm works on the probability simplex (sum = 1, non‑negative).  It
repeatedly:
  * evaluates the self‑convolution,
  * identifies the shift k giving the current maximum,
  * moves a modest amount of probability mass from the index that contributes
    the most to that shift towards an index that contributes little,
  * accepts the move if it improves the objective, otherwise with a
    temperature‑dependent probability (simulated annealing),
  * occasionally perturbs the best solution to escape stagnation.

A final greedy pair‑wise swap pass tries to squeeze a little more improvement.
The routine respects the 1000 s budget, works for any length (uses the
provided `height_sequence_1` if present) and returns a list of heights ready
for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection on the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i >= 0 , sum x_i = 1} (Duchi et al. ICML 2008)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core utility: compute self‑convolution and the objective value.
# ----------------------------------------------------------------------
def _conv_score(p: np.ndarray, m_fft: int) -> tuple[np.ndarray, float]:
    """
    Return (full_conv, objective) for a probability vector p.
    Full convolution length is 2*len(p)-1.
    The objective is 2*n*max(conv) because sum(p)=1.
    """
    n = p.shape[0]
    fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(fft * fft, n=m_fft)[: 2 * n - 1]
    max_b = conv.max()
    # sum(p) == 1, so denominator is 1
    score = float(2.0 * n * max_b)
    return conv, score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Heuristic optimiser that searches for a step‑function with a low
    ``evaluate_sequence`` value.  The method combines a sub‑gradient move
    (focused on the currently worst convolution shift) with a simulated‑annealing
    acceptance rule and occasional random restarts.  A final local swap phase
    tries to obtain a final polish.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size and initial distribution
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        # start from the best known construction (already on the simplex)
        init = np.array(height_sequence_1, dtype=float)
        init = np.clip(init, 0.0, 1000.0)
        total = init.sum()
        if total > 0:
            p = init / total
        else:
            # unlikely, fall back to uniform
            n = init.shape[0]
            p = np.full(n, 1.0 / n)
    else:
        # no previous construction – pick a modest size (fast to evaluate)
        n = 600
        # uniform start (simplex)
        p = np.full(n, 1.0 / n)

    n = p.shape[0]

    # FFT parameters: next power‑of‑two that can hold a convolution of length 2n‑1
    m_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Initial evaluation
    # ------------------------------------------------------------------
    conv, cur_score = _conv_score(p, m_fft)
    best_score = cur_score
    best_p = p.copy()
    print(f"[init] n={n}, score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Hyper‑parameters for the annealing loop
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # annealing schedule (linear decay)
    T_start = 0.08
    T_end = 1e-6

    # step size for moving mass (fraction of the source entry)
    eps_frac_start = 0.5   # early moves can be large
    eps_frac_end = 0.001   # later moves become tiny

    # restart / jitter schedule
    jitter_every = 150000   # iterations
    no_imp_limit = 120000   # if no improvement for this many iters, jitter

    last_improve_iter = 0
    iteration = 0

    # ------------------------------------------------------------------
    # 4. Main stochastic sub‑gradient loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # a) identify the worst shift k (where conv is maximal)
        # --------------------------------------------------------------
        k = int(np.argmax(conv))

        # --------------------------------------------------------------
        # b) compute contributions a_i * a_{k-i} for this shift
        # --------------------------------------------------------------
        idx = np.arange(n)
        jdx = k - idx
        valid = (jdx >= 0) & (jdx < n)

        contrib = np.zeros_like(p)
        contrib[valid] = p[idx[valid]] * p[jdx[valid]]

        # source of the largest contribution
        i_src = int(np.argmax(contrib))
        # destination where contribution is (almost) zero
        zero_idxs = np.where(contrib == 0.0)[0]
        if zero_idxs.size > 0:
            i_dst = int(rng.choice(zero_idxs))
        else:
            # fallback – pick the minimal contributor (different from source)
            i_dst = int(np.argmin(contrib))
            if i_dst == i_src:
                # pick any other index
                i_dst = (i_src + 1) % n

        # --------------------------------------------------------------
        # c) decide how much mass to move (epsilon)
        # --------------------------------------------------------------
        # linear schedules for temperature and step fraction
        prog = iteration / (budget_s * 10)  # very rough normalisation
        # clamp to [0,1]
        prog = min(max(prog, 0.0), 1.0)

        T = T_start * (1.0 - prog) + T_end * prog
        eps_frac = eps_frac_start * (1.0 - prog) + eps_frac_end * prog

        eps = min(p[i_src] * eps_frac, 0.02)  # cap absolute move for stability
        if eps <= 0.0:
            # nothing to move – recompute conv for the next iteration
            conv, cur_score = _conv_score(p, m_fft)
            continue

        # --------------------------------------------------------------
        # d) propose the new distribution
        # --------------------------------------------------------------
        p_new = p.copy()
        p_new[i_src] -= eps
        p_new[i_dst] += eps
        # sum is preserved, all entries stay non‑negative by construction

        # --------------------------------------------------------------
        # e) evaluate the proposal
        # --------------------------------------------------------------
        _, new_score = _conv_score(p_new, m_fft)

        # --------------------------------------------------------------
        # f) acceptance (simulated annealing)
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis rule
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            p = p_new
            cur_score = new_score
            conv, _ = _conv_score(p, m_fft)  # recompute conv for next loop
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                last_improve_iter = iteration
                print(f"[{iteration:7d}] new best: {best_score:.6f}")

        # --------------------------------------------------------------
        # g) occasional jitter / restart when stuck
        # --------------------------------------------------------------
        if (iteration - last_improve_iter) >= no_imp_limit:
            # perturb best solution and continue from there
            p = best_p + rng.normal(scale=0.015, size=n)
            p = _simplex_projection(p)       # keep on simplex
            conv, cur_score = _conv_score(p, m_fft)
            last_improve_iter = iteration
            print(f"[{iteration:7d}] restart (jitter) – best still {best_score:.6f}")

        if iteration % jitter_every == 0:
            # small random walk to avoid tiny cycles
            jitter = rng.normal(scale=0.003, size=n)
            p = _simplex_projection(p + jitter)
            conv, cur_score = _conv_score(p, m_fft)

        # ------------------------------------------------------------------
        # early exit if we have already passed the target
        # ------------------------------------------------------------------
        if best_score <= 1.48:      # a comfortable margin below the required 1.503
            print(f"[{iteration:7d}] target reached (score={best_score:.6f})")
            break

    # ------------------------------------------------------------------
    # 5. Final local refinement – cheap pairwise swaps
    # ------------------------------------------------------------------
    p = best_p.copy()
    conv, cur_score = _conv_score(p, m_fft)

    swap_trials = 3000
    for _ in range(swap_trials):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j or p[i] <= 1e-8:
            continue
        # tiny mass transfer
        eps = min(p[i] * 0.5, 1e-4 * rng.random())
        if eps <= 0:
            continue
        p2 = p.copy()
        p2[i] -= eps
        p2[j] += eps
        _, new_score = _conv_score(p2, m_fft)
        if new_score < cur_score:
            p = p2
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                print(f"[swap] improved to {best_score:.6f}")

    print(f"[final] best score found: {best_score:.6f}")
    # Clip to the limits enforced by evaluate_sequence and return as plain list
    return np.clip(best_p, 0.0, 1000.0).tolist()
