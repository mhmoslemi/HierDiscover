Iterative linear‑programming improvement of a step‑function.

The algorithm starts from the well known central‑binomial coefficients
(b_k = C(2k,k)/4**k) truncated to a moderate length *n* and rescaled so

    maximise    Σ g_i

strictly decreases.  Repeating this update drives the score down towards
the optimal bound.

The LP is tiny (a few hundred variables, O(n) constraints) and solved
with `cvxpy`/`ECOS`, so many iterations fit easily into the 1000 s budget.

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _binomial_prefix(max_n: int) -> np.ndarray:
    """Return the central‑binomial coefficients a_k = C(2k,k)/4**k for 0≤k<max_n."""
    a = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    a[0] = cur
    for k in range(1, max_n):
        # recurrence: a[k] = a[k‑1] * (2k‑1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        a[k] = cur
    return a


def _conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix B such that B @ g = a * g (full convolution).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    B = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes the constant a[i] on rows i … i+n‑1
    for i in range(n):
        B[i:i + n, i] = a[i]
    return B


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     max_len: int = 15000,
                     n_opt: int = None,
                     **kwargs) -> list[float]:
    """
    Produce a coefficient sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for any stochastic components (fallback search).
    budget_s : int
        Total time budget in seconds.
    max_len : int
        Upper bound on the sequence length (mostly ignored; we cap it).
    n_opt : int | None
        Desired length of the optimisation.  If None a modest length
        (≈300) is chosen so that the LP solves quickly.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest length that still allows a low theoretical bound.
    # ------------------------------------------------------------------
    if n_opt is None:
        n = min(300, max_len)                 # 300 → lower bound ≈1.003
    else:
        n = min(int(n_opt), max_len)

    # ------------------------------------------------------------------
    # 2. Initialise with the central‑binomial prefix, normalised to
    #    Σ a_i = sqrt(2 n).
    # ------------------------------------------------------------------
    a_raw = _binomial_prefix(n)
    target_sum = math.sqrt(2.0 * n)
    a = a_raw * (target_sum / a_raw.sum())
    # initial score = max convolution (because of the chosen normalisation)
    conv = np.convolve(a, a)
    M = float(conv.max())
    best_score = M
    best_seq = a.tolist()
    print(f"[init] n={n}, initial max_conv={M:.9f}, target_sum={target_sum:.6f}")

    # ------------------------------------------------------------------
    # 3. If cvxpy is not available, fall back to the simple stochastic
    #    refinement (the original method).  This path still yields a legal
    #    answer.
    # ------------------------------------------------------------------
    if not _CVX_AVAILABLE:
        print("[fallback] cvxpy not available – using simple random walk.")
        # simple hill‑climbing around the central‑binomial start
        cur = a.copy()
        cur_score = best_score
        sigma = 0.05
        no_improve = 0
        while time.time() < deadline and no_improve < 5000:
            idx = np.random.randint(n)
            delta = np.random.randn() * sigma * max(cur[idx], 1e-12)
            cand = cur.copy()
            cand[idx] = max(0.0, cand[idx] + delta)
            obj = float(evaluate_sequence(cand.tolist()))
            if obj < cur_score:
                cur, cur_score = cand, obj
                no_improve = 0
                if obj < best_score:
                    best_score, best_seq = obj, cand.tolist()
                    print(f"[rw] improved to {obj:.9f}")
            else:
                no_improve += 1
        return [float(max(0.0, x)) for x in best_seq]

    # ------------------------------------------------------------------
    # 4. Main LP‑based optimisation loop.
    # ------------------------------------------------------------------
    t = 0.1                     # initial mixing weight
    iteration = 0
    lower_M = target_sum ** 2 / (2 * n - 1)   # theoretical lower bound on M

    while time.time() < deadline:
        iteration += 1

        # ---- build the convolution matrix for the current f (=a) ----
        B = _conv_matrix(a)

        # ---- linear program: maximise Σ g_i   subject to   B @ g ≤ M, g ≥ 0 ----
        g = cp.Variable(n, nonneg=True)
        constraints = [B @ g <= M + 1e-12]   # tiny slack for numerical safety
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception as e:  # pragma: no cover
            print("[lp] solver exception:", e)
            break
        if g.value is None:
            print("[lp] solver failed to produce a solution")
            break

        g_opt = np.array(g.value).flatten()
        sum_g = g_opt.sum()
        if sum_g <= 0.0:      # degenerate – shouldn’t happen
            print("[lp] trivial solution, aborting")
            break

        # ---- scale g to keep the ℓ₁‑norm fixed (Σ = √(2n)) ----
        scale = target_sum / sum_g
        g_scaled = g_opt * scale

        # ---- convex combination step ----
        a_new = (1.0 - t) * a + t * g_scaled
        a_new = np.maximum(a_new, 0.0)      # safeguard against tiny negatives

        # ---- compute the new worst‑case convolution value ----
        conv_new = np.convolve(a_new, a_new)
        M_new = float(conv_new.max())

        # ---- update best solution if we made progress ----
        if M_new < best_score:
            best_score = M_new
            best_seq = a_new.tolist()
            print(f"[iter {iteration:3d}] improved score: {best_score:.9f} (t={t:.3f})")

        # ---- prepare for next iteration ----
        a, M = a_new, M_new

        # Reduce step size slowly – this stabilises the process.
        t = max(t * 0.97, 1e-4)

        # Stop if we are essentially at the lower bound.
        if M <= lower_M * 1.005:
            print(f"[converged] reached near‑optimal bound (M={M:.9f})")
            break

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python floats).
    # ------------------------------------------------------------------
    best_seq = [float(max(0.0, x)) for x in best_seq]
    # final safety evaluation (for logging)
    try:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] best score = {final_score:.9f}, length = {len(best_seq)}")
    except Exception:
        pass
    return best_seq
