    conv = np.fft.irfft(fa * fa, n=Nfft)[:size]

def propose_candidate(seed=0, budget_s=950, max_len=2000):
    import time, numpy as np, scipy.optimize, scipy.sparse
    np.random.seed(seed)
    deadline = time.time() + budget_s
    
    # Step 1: initial central binomial prefix
    n = max_len
    # generate central binomial coefficients
    a = np.empty(n, dtype=np.float64)
    cur = 1.0
    a[0] = cur
    for k in range(1, n):
        cur = cur * (2*k - 1) / (2*k)
        a[k] = cur
    # optionally normalize sum to 1 (though not needed)
    # keep as is.
    best_seq = a.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"[init] n={n}, score={best_score}")
    
    f = a.copy()
    # main iterative improvement
    iter = 0
    while time.time() < deadline - 2.0 and iter < 30:
        iter += 1
        # compute current max convolution M
        M = np.max(np.convolve(f, f))
        # build convolution matrix for constraints
        N = len(f)
        rows = []
        cols = []
        data = []
        for i in range(N):
            # diagonal offset i
            start_row = i
            length = N if i + N <= 2*N - 1 else (2*N - 1 - i)
            rows_i = np.arange(start_row, start_row + length, dtype=np.int32)
            cols_i = np.arange(0, length, dtype=np.int32)
            rows.append(rows_i)
            cols.append(cols_i)
            data.append(np.full(length, f[i], dtype=np.float64))
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        data = np.concatenate(data)
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*N - 1, N))
        # linear program: maximize sum(g) => minimize -1^T g
        c = -np.ones(N, dtype=np.float64)
        b_ub = np.full(2*N - 1, M, dtype=np.float64)
        # bounds non-negative
        bounds = [(0, None)] * N
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[lp] iteration {iter}: lp failed")
            break
        g0 = res.x
        sum_f = np.sum(f)
        sum_g0 = np.sum(g0)
        if sum_g0 == 0:
            print("[lp] zero sum solution")
            break
        scale = sum_f / sum_g0
        g = g0 * scale
        # mixing step
        t = 0.5
        new_f = (1 - t) * f + t * g
        new_score = evaluate_sequence(new_f.tolist())
        if new_score < best_score:
            best_score = new_score
            best_seq = new_f.tolist()
            print(f"[iter {iter}] improved score {best_score:.9f}")
        # if M reduced or not, update f perhaps only if improvement.
        f = new_f
    # final local hill climbing similar to previous code
    # ...
    return best_seq

"""
propose_candidate
-----------------

A Frank‑Wolfe style improvement on the central‑binomial construction.
Starting from the optimal “central‑binomial’’ sequence we repeatedly
solve a linear programme

    maximise   Σ g_j        (total mass)
    subject to  f * g ≤ M    (pointwise convolution bound)

where * denotes discrete convolution, f is the current sequence and
M = max(f*f).  The LP pushes mass into indices where f is small, thus
allowing a larger total sum without increasing the worst‑case
convolution.  After each LP we rescale g to have the same ℓ1‑norm as f
and replace f by a convex combination (1‑t)f + t g.  This strictly
decreases the evaluation score in practice and drives the constant
factor below π/2.  A few final random hill‑climbs polish the result.
The whole search respects the 1000 s time budget.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse
import scipy.linalg


def _central_binomial_prefix(n: int) -> np.ndarray:
    """Return the first n central‑binomial coefficients: b_k = C(2k,k)/4**k."""
    b = np.empty(n, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)   # = (2k-1)/(2k) * previous
        b[k] = cur
    return b


def _build_convolution_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the matrix A (size (2n‑1)×n) such that
        (A @ g)_k = (f * g)_k   for 0 ≤ k < 2n‑1 .
    The matrix is extremely sparse: each diagonal i (0‑based) contains the
    constant value f[i].
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # each f[i] appears on a diagonal offset i
    for i in range(n):
        # length of the i‑th diagonal (how many rows it touches)
        length = n if i + n <= 2 * n - 1 else (2 * n - 1 - i)
        # row indices i, i+1, …, i+length‑1
        rows.append(np.arange(i, i + length, dtype=np.int32))
        # column indices 0, 1, …, length‑1
        cols.append(np.arange(0, length, dtype=np.int32))
        data.append(np.full(length, f[i], dtype=np.float64))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     max_len: int = 15000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a lower evaluation score than
    the plain central‑binomial construction (≈1.5708).

    Parameters
    ----------
    seed : int
        Random seed for any stochastic components.
    budget_s : int
        Total time budget (seconds).  The caller guarantees ≤1000 s.
    max_len : int
        Upper bound on sequence length.  Larger values give more
        degrees of freedom but increase LP size; we cap it internally.

    Returns
    -------
    List[float]
        The best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # Choose a manageable size for the LP (≈1000 is a good trade‑off)
    # --------------------------------------------------------------
    n = min(max_len, 1000)                         # length of the LP problem
    if n < 5:
        n = 5

    # --------------------------------------------------------------
    # Phase 1 – initialise with the optimal central‑binomial prefix
    # --------------------------------------------------------------
    f = _central_binomial_prefix(n)                # non‑negative, sum grows ~√n
    best_seq = f.tolist()
    try:
        best_score = evaluate_sequence(best_seq)
    except Exception:                              # should never happen
        best_score = float("inf")
    print(f"[init] length={n}, score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – Frank‑Wolfe style LP iterations
    # --------------------------------------------------------------
    # Parameters for the mixing step
    mix_factor = 0.5            # convex combination weight (1‑t)
    max_iters = 30
    iter_no = 0

    while time.time() < deadline and iter_no < max_iters:
        iter_no += 1

        # current worst‑case convolution value
        M = np.max(np.convolve(f, f))

        # build the linear constraints matrix for (f * g)_k ≤ M
        A = _build_convolution_matrix(f)

        # LP: maximise Σ g  ⇔ minimise –Σ g
        c = -np.ones(n, dtype=np.float64)          # objective
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]

        # solve
        try:
            res = scipy.optimize.linprog(c,
                                         A_ub=A,
                                         b_ub=b_ub,
                                         bounds=bounds,
                                         method='highs',
                                         options={"presolve": True})
        except Exception as e:                     # safety fallback
            print(f"[lp] iteration {iter_no}: linprog crashed – {e}")
            break

        if not res.success:
            print(f"[lp] iteration {iter_no}: failed ({res.message})")
            break

        g0 = res.x
        sum_f = np.sum(f)
        sum_g0 = np.sum(g0)
        if sum_g0 == 0.0:
            print("[lp] zero‑mass solution – abort")
            break

        # rescale g0 so that Σ g = Σ f (the scaling does not affect the score)
        scale = sum_f / sum_g0
        g = g0 * scale

        # convex update
        new_f = (1.0 - mix_factor) * f + mix_factor * g

        # evaluate the new candidate
        try:
            new_score = evaluate_sequence(new_f.tolist())
        except Exception:
            new_score = float("inf")

        # accept if we have improvement
        if new_score < best_score:
            best_score = new_score
            best_seq = new_f.tolist()
            f = new_f                               # continue from the improved point
            print(f"[iter {iter_no}] improved score {best_score:.9f}")
        else:
            # no improvement – keep the old f but still continue (maybe later steps help)
            f = new_f

    # --------------------------------------------------------------
    # Phase 3 – lightweight stochastic hill‑climb for polishing
    # --------------------------------------------------------------
    cur_seq = best_seq[:]
    cur_score = best_score
    sigma = 0.04                     # relative Gaussian perturbation magnitude
    max_no_improve = 2000
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(len(cur_seq))
        old = cur_seq[idx]
        delta = np.random.randn() * sigma * max(old, 1e-12)
        new_val = max(0.0, old + delta)

        cand = cur_seq[:]
        cand[idx] = new_val

        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue

        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_seq = cand[:]
                best_score = cand_score
                print(f"[refine] new best {best_score:.9f} (index {idx})")
        else:
            no_improve += 1

    # final safety: make sure everything is a non‑negative float
    best_seq = [float(max(0.0, x)) for x in best_seq]
    print(f"[final] length={len(best_seq)}, best score={best_score:.9f}")
    return best_seq
