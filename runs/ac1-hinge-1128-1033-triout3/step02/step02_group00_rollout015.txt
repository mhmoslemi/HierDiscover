def build_toeplitz(a):
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n), dtype=float)
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # row k: columns i correspond to a[k-i]
        A[k, i_min:i_max+1] = a[k - np.arange(i_min, i_max+1)]
    return A

def _build_constraint_matrix(a):
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    # For each k row
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # columns i = i_min..i_max inclusive
        # A[k,i] = a[k-i]
        idx = np.arange(i_min, i_max+1)
        A[k, idx] = a[k - idx]
    return A

"""
Hybrid optimizer for the self‑convolution bound.

* Initialise the step heights on a “arcsine” profile (heavy at the ends,
  light in the middle) – a good heuristic for flattening the self‑convolution.
* Run a fast sub‑gradient descent with a simple Adam‑style momentum
  and periodic random mass‑transfer jumps.  This quickly reduces the
  maximum convolution value.
* Every few ten‑thousand sub‑gradient steps we solve a linear programme:
        maximise   Σ b_j
        subject to  a * b ≤ M        (componentwise)
                    b_j ≥ 0
  where M = max (a*a).  The optimum b has a larger ℓ₁‑norm while still
  respecting the current bound M.  By moving a towards the normalised
  direction g = b / Σb (using a line‑search) the bound M never increases.
  This “Frank‑Wolfe / exchange” step often pushes the solution past the
  plateau observed by plain sub‑gradient descent.
* The loop runs until the time budget expires; the best sequence seen
  (according to the provided `evaluate_sequence`) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy support for the linear‑programming refinement step
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog  # type: ignore
    from scipy.sparse import csr_matrix  # type: ignore
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full self‑convolution using a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full (length 2·n‑1) convolution a*a."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    np.maximum(conv, 0, out=conv)            # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Helper: build a sparse Toeplitz‑like matrix for the constraint a*b ≤ M
# ----------------------------------------------------------------------
def _build_constraint_matrix_sparse(a: np.ndarray) -> csr_matrix:
    """Return a (2n‑1) × n sparse matrix with rows containing shifted a."""
    n = a.shape[0]
    m = 2 * n - 1
    # count of non‑zero entries = n*(2n‑1) – O(n) (every row is a slice of a)
    data = []
    rows = []
    cols = []
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        col_idx = np.arange(i_min, i_max + 1, dtype=int)
        row_idx = np.full_like(col_idx, k)
        # a_{k‑col}
        vals = a[k - col_idx]
        rows.extend(row_idx.tolist())
        cols.extend(col_idx.tolist())
        data.extend(vals.tolist())
    return csr_matrix((data, (rows, cols)), shape=(m, n))


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Hybrid optimisation: sub‑gradient descent + occasional LP refinement.
    Returns the best non‑negative sequence (list of floats) found before
    the allotted time expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------
    # Initialise the vector a (try to reuse a previous construction)
    # ------------------------------------------------------------
    try:                                   # use the global if it exists
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = a.shape[0]
    except Exception:                     # fresh start – arcsine shape
        n = 2000
        i = np.arange(n, dtype=float)
        # arcsine‑like density: large at the ends, small in the centre
        a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
        a = np.maximum(a, 0.0)
        a /= a.sum()

    # Normalise to a probability distribution (Σa = 1)
    a = _project_onto_simplex(a)

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    conv = _full_convolution(a)
    cur_max = conv.max()
    cur_val = float(2.0 * n * cur_max)          # equals evaluate_sequence for Σa=1
    best_a = a.copy()
    best_val = cur_val
    iteration = 0
    # Adam‑style moment estimates
    m_exp = np.zeros_like(a)
    v_exp = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    print(f"[init] n={n}, initial evaluation={cur_val:.6f}")

    # ------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # --------------------------------------------------------
        # Current state (convolution, max, objective)
        # --------------------------------------------------------
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = float(2.0 * n * max_conv)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # --------------------------------------------------------
        # Sub‑gradient of the active peak (k*)
        # --------------------------------------------------------
        k_star = int(np.argmax(conv))
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        subgrad = np.zeros_like(a)
        subgrad[idx] = 2.0 * a[k_star - idx]       # g_i = 2·a_{k*‑i}

        # --------------------------------------------------------
        # Adam step‑size schedule (linear cooling)
        # --------------------------------------------------------
        progress = (now - start_time) / (deadline - start_time)
        eta_start = 0.20
        eta_end = 1e-4
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # Adam momentum update
        m_exp = beta1 * m_exp + (1.0 - beta1) * subgrad
        v_exp = beta2 * v_exp + (1.0 - beta2) * (subgrad ** 2)
        m_hat = m_exp / (1.0 - beta1 ** (iteration + 1))
        v_hat = v_exp / (1.0 - beta2 ** (iteration + 1))
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------
        # Gradient step + projection onto the simplex
        # --------------------------------------------------------
        a_candidate = a - step

        # occasional random mass‑transfer to escape plateaus
        if np.random.rand() < 0.01:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a_candidate[src] * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        a = _project_onto_simplex(a_candidate)

        iteration += 1

        # --------------------------------------------------------
        # Periodic LP refinement (Frank‑Wolfe style)
        # --------------------------------------------------------
        if _SCIPY_AVAILABLE and iteration % 20000 == 0:
            # Current bound M
            M = max_conv
            # Build the Toeplitz‑type constraint matrix a * b ≤ M
            A = _build_constraint_matrix_sparse(a)
            b_ub = np.full(A.shape[0], M, dtype=float)

            # Linear programme: maximise Σ b_j  (i.e. minimise –Σ b_j)
            res = linprog(
                -np.ones_like(a),
                A_ub=A,
                b_ub=b_ub,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True},
            )
            if res.success:
                b_opt = res.x
                sum_b = b_opt.sum()
                # if we can increase the ℓ₁‑norm, move a toward the normalised direction
                if sum_b > 1.0 + 1e-6:
                    g = b_opt / sum_b
                    # simple back‑tracking line search on the mixing parameter t
                    for t in [0.5, 0.25, 0.125, 0.0625, 0.03125]:
                        a_new = (1.0 - t) * a + t * g
                        conv_new = _full_convolution(a_new)
                        M_new = conv_new.max()
                        if M_new <= M * (1.0 + 1e-12):
                            a = a_new
                            # update current records after a successful step
                            cur_val = float(2.0 * n * M_new)
                            if cur_val < best_val:
                                best_val = cur_val
                                best_a = a.copy()
                                print(
                                    f"[LP{iteration:7d}] new best after LP = {best_val:.6f}"
                                )
                            break
            # (if the LP failed or gave no improvement we simply continue)

    # ------------------------------------------------------------
    # Final reporting using the provided evaluation routine
    # ------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
