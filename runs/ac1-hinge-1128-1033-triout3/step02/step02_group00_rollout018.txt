"""
Smoothed‑max gradient descent for the self‑convolution objective.

The objective  f(a) = max_k (a*a)[k]  (with Σ a_i = 1, a_i ≥ 0) can be
approximated by the smooth function

        L_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).

∂L_β/∂a_i = 2·Σ_k w_k·a_{k‑i},
with w_k = exp(β·(a*a)[k]) / Σ_j exp(β·(a*a)[j])  (a probability vector).

The algorithm:
  * start from a simple decreasing (triangular) shape,
  * iterate:
        – compute the full self‑convolution,
        – build the soft‑max weights w,
        – compute the gradient via a correlation,
        – take a step a ← a – η·grad,
        – project back onto the probability simplex,
        – periodically perform a tiny random mass‑transfer (to escape plateaus).
  * β is increased linearly from a small value (smooth) to a large value
    (close to the true max), while the step‑size η is decreased.
  * The best sequence seen during the run is returned.

The method is cheap (O(n log n) per iteration) and typically finds
solutions better than the previous sub‑gradient approach within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector v onto the simplex {x ≥ 0, Σ x = 1}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safeguard – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to counter rounding errors
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full convolution a*a using a real FFT (fast for many iterations)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full (length 2·n‑1) convolution a*a."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()            # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    np.maximum(conv, 0, out=conv)               # eliminate tiny negatives
    return conv
