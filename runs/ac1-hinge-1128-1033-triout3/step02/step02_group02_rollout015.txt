def gradient_softmax(a, eps):
    n = a.shape[0]
    size1 = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size1)
    conv = np.fft.irfft(A * A, n=size1)[:2*n-1]
    # compute weights
    maxc = conv.max()
    w = np.exp(eps * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # gradient: correlation a with w
    size2 = 1 << ((3*n - 2).bit_length())
    A2 = np.fft.rfft(a, n=size2)
    W = np.fft.rfft(w, n=size2)
    corr = np.fft.irfft(np.conj(A2) * W, n=size2)[: n + (2*n - 1) - 1]  # = 3n-2
    grad = 2.0 * corr[:n]
    return grad, conv.max()

def simplex_projection(v):
    # v: numpy array
    # returns projected onto simplex (sum 1, non-negative)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all negative
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    """Project onto the probability simplex (sum=1, non-negative)."""
    n = v.shape[0]
    # shift to ensure sum positive?
    # Solve via sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative, return uniform
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # Ensure sum is exactly 1 (numerical tolerance)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0/n)
    return w
```

"""
Propose a non‑negative step function that minimizes the evaluation metric.

Algorithm
---------
1.  **Parametric sweep** –  Enumerate a few simple families of symmetric
    “U‑shaped’’ distributions (plain power‑law, power‑law with an offset,
    sum‑of‑tails, exponential and a mixed power‑law/exponential).  For a
    range of lengths and exponents the maximum entry of the self‑convolution
    is obtained with an FFT‑based routine, which is much faster than the
    naïve ``np.convolve`` used by the evaluator.  The best candidate of this
    sweep is kept as a warm start.

2.  **Smoothed gradient descent** –  Starting from the warm start we minimise
    a smooth approximation of the objective:

          fε(a) = (1/ε)·log Σk exp(ε·(a∗a)[k]) ,

    whose gradient can be written in closed form and computed by FFTs.
    At each iteration we
       * compute the gradient of fε,
       * take a step opposite to the gradient,
       * orthogonally project the result onto the probability simplex
         (non‑negative, sum = 1),
       * accept the move if it improves the true objective or with a
         decreasing Metropolis probability.

    The smoothing parameter ε is increased gradually (annealing) so that the
    optimisation first follows a smooth landscape and later concentrates on
    the true max‑convolution objective.

3.  **Random mass‑transfer moves** –  To escape shallow basins, with a small
    probability we perform a random transfer of a fraction of mass between
    two positions (as in simulated annealing).

The routine respects a time budget (default 1000 s) and returns the best
list of floats found.  A final exact call to ``evaluate_sequence`` reports the
score that will be used for scoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x ≥ 1)."""
    return 1 << (x - 1).bit_length()


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to guard against tiny numerical drift
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Parametric families of symmetric step‑functions
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Power‑law U‑shape a[i] = (dist+1)^(-p)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = (d + 1) ** (-p) + 1e-15
    a /= a.sum()
    return a


def u_shaped_offset_distribution(n: int, p: float, offset: float) -> np.ndarray:
    """
    Power‑law U‑shape with an additive offset:
        a[i] = (dist + offset + 1)^(-p) .
    The offset widens the high‑plateau near the edges.
    """
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = (d + offset + 1) ** (-p) + 1e-15
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails a[i] = (i+1)^(-p)+(n-i)^(-p)."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-15
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges a[i] = exp(-α·dist)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-15
    a /= a.sum()
    return a


def mixed_power_exponential_distribution(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Mixed shape a[i] = (dist+1)^(-p) * exp(-α·dist).
    Allows independent control of the power‑law tail and the exponential decay.
    """
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = (d + 1) ** (-p) * np.exp(-alpha * d) + 1e-15
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the smooth approximation
# ----------------------------------------------------------------------


def grad_and_max(a: np.ndarray, eps: float, size_fft: int):
    """
    Compute the gradient of the smoothed max‑convolution objective
    (log‑sum‑exp) together with the true maximal convolution entry.
    Returns (grad, max_conv).
    """
    n = a.shape[0]

    # FFT of the current vector (zero‑padded to size_fft)
    A = np.fft.rfft(a, n=size_fft)

    # Linear convolution a*a  (length 2n–1)
    conv = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
    max_c = conv.max()

    # Soft‑max weights over the 2n‑1 convolution entries
    # subtract max_c for numerical stability
    w_unnorm = np.exp(eps * (conv - max_c))
    w_sum = w_unnorm.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w_unnorm / w_sum

    # Gradient = 2 * correlation(a, w)
    # correlation(a,w)[i] = Σ_j a[j] * w[i + j]
    # This can be obtained as the convolution of a with w reversed.
    W_rev = np.fft.rfft(w[::-1], n=size_fft)
    conv_rev = np.fft.irfft(A * W_rev, n=size_fft)[: 3 * n - 2]

    # Extract the needed n entries: grad[i] = 2 * conv_rev[(2n‑2)‑i]
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv_rev[idx]

    return grad, max_c


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = np.inf   # this stores the *approximate* score 2·n·maxconv

    # ------------------------------------------------------------------
    # 0.  Seed from a previously supplied sequence (if present)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            if external.sum() < 1e-12:
                external = np.ones_like(external) / external.size
            else:
                external = external / external.sum()
            sc = 2.0 * len(external) * max_conv_fft(external)
            if sc < best_score:
                best_seq = external.copy()
                best_score = sc
                print(f"Seeded from height_sequence_1: score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic parametric sweep
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000,
               12000, 16000, 20000, 25000, 30000]
    p_vals = np.linspace(0.2, 3.0, 28)            # step ≈0.1
    offsets = [0.0, 0.5, 1.0, 2.0, 5.0]
    alphas = np.linspace(0.0002, 0.02, 20)      # exponential decay
    mix_params = [(p, a) for p in np.linspace(0.2, 2.0, 12)
                        for a in np.linspace(0.0002, 0.01, 8)]

    print("\n=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # ---- plain power‑law U‑shape ----
        for p in p_vals:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=U‑pow, p={p:4.2f}, score={sc:.6f}")

        # ---- power‑law with offset (plateau) ----
        for p in p_vals:
            if time.time() > deadline:
                break
            for off in offsets:
                a = u_shaped_offset_distribution(n, p, off)
                sc = 2.0 * n * max_conv_fft(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"New best: n={n:5d}, shape=U‑off, p={p:4.2f}, off={off:3.1f}, score={sc:.6f}")

        # ---- sum‑of‑tails ----
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=tails, p={p:4.2f}, score={sc:.6f}")

        # ---- exponential decay ----
        for alp in alphas:
            if time.time() > deadline:
                break
            a = exponential_distribution(n, alp)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=exp, α={alp:6.4f}, score={sc:.6f}")

        # ---- mixed power‑law / exponential ----
        for p, alp in mix_params:
            if time.time() > deadline:
                break
            a = mixed_power_exponential_distribution(n, p, alp)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=mixed, p={p:4.2f}, α={alp:6.4f}, score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Fallback to a random normalised vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = 2.0 * n * max_conv_fft(a)
        print(f"Fallback random init: n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Smoothed gradient descent with occasional random transfers
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = len(current)
    current_max = max_conv_fft(current)
    best_seq = current.copy()
    best_score = 2.0 * n * current_max

    size_fft = next_pow2(3 * n - 2)        # large enough for convolutions and correlations
    eps = 1.0
    eps_max = 30.0
    eps_factor = 1.15

    eta = 0.5                # base step size for gradient move
    eta_decay = 0.9999

    temperature = 0.2
    temp_decay = 0.9995

    iteration = 0
    no_improve = 0

    print("\n=== Smoothed gradient descent ===")
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Random mass‑transfer move (exploration)
        # --------------------------------------------------------------
        if np.random.rand() < 0.25:
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] <= 0.0:
                continue
            delta = current[i] * np.random.rand() * 0.3
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = project_simplex(cand)
            cand_max = max_conv_fft(cand)
        else:
            # --------------------------------------------------------------
            # Gradient descent step on the smoothed objective
            # --------------------------------------------------------------
            grad, conv_max = grad_and_max(current, eps, size_fft)
            norm_grad = np.linalg.norm(grad)
            if norm_grad < 1e-12:
                # Gradient vanished – abort the descent part of this iteration
                continue
            step = eta / (norm_grad + 1e-12)
            cand = current - step * grad
            cand = project_simplex(cand)
            cand_max = max_conv_fft(cand)

        # --------------------------------------------------------------
        # Acceptance test (simulated‑annealing style)
        # --------------------------------------------------------------
        accept = False
        if cand_max < current_max:
            accept = True
        else:
            prob = np.exp((current_max - cand_max) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_max = cand_max
            cand_score = 2.0 * n * cand_max
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Iter {iteration:7d}: new global best = {best_score:.6f}")
                no_improve = 0
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Scheduler updates
        # --------------------------------------------------------------
        eta = max(eta * eta_decay, 1e-6)
        temperature = max(temperature * temp_decay, 1e-9)

        if iteration % 250 == 0 and eps < eps_max:
            eps = min(eps * eps_factor, eps_max)

        if no_improve >= 5000:
            print("No improvement for a long time – terminating search.")
            break

    # ------------------------------------------------------------------
    # 4.  Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = None

    print("\n=== Search finished ===")
    print(f"Best approx. score : {best_score:.6f}")
    print(f"Best length        : {len(best_seq)}")
    return best_seq.tolist()
