"""
Propose a step‑function (non‑negative heights) that minimises

        M = 2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm is an **exponentiated‑gradient mirror descent** on a
smooth‐max approximation of the max‑convolution.  At each iteration we
1. compute the self‑convolution `c = conv(p, p)` (the distribution `p`
   always satisfies Σp = 1);
2. replace the hard max by a soft‑max with temperature `α`,
   i.e. `w_k ∝ exp(α·c_k)`;
3. compute the gradient of the smooth objective
   `g = 2·corr(w, p)` (correlation gives the exact ∂/∂p of Σw_k·c_k);
4. perform a multiplicative (exponentiated) update
   `p_i ← p_i·exp(−η·(g_i−min(g)))` and renormalise – this automatically
   respects the simplex constraints;
5. slowly increase `α` (tightening the approximation) and decay the
   step‑size `η`.

The simplex is never left, all entries stay non‑negative and the sum
remains ≥ 0.01, so the security checks in `evaluate_sequence` are passed.
Occasional Gaussian jitter and a simple stagnation‑restart keep the search
from getting stuck.  When the time budget is exhausted the best distribution
found so far is returned as a plain Python list.
"""

import time
import numpy as np


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return soft‑max weights w_k ∝ exp(alpha·conv_k) (numerically stable)."""
    z = alpha * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of heights that minimises the evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (a few seconds are kept as safety margin).

    Returns
    -------
    list[float]
        The best non‑negative sequence found (already normalised to sum = 1).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size and initial distribution
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name‑defined]
        init = np.array(height_sequence_1, dtype=float)  # type: ignore[name‑defined]
        init = np.clip(init, 0.0, 1000.0)
        tot = init.sum()
        p = init / tot if tot > 0 else np.full(n, 1.0 / n)
    except Exception:                       # height_sequence_1 not defined
        n = 1153
        p = np.full(n, 1.0 / n)

    # best solution seen so far (always normalised)
    best_p = p.copy()
    best_score = 2.0 * n * np.max(np.convolve(p, p))

    # ------------------------------------------------------------------
    # 2. Hyper‑parameters
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    alpha = 0.5          # start smoothing for soft‑max
    alpha_mul = 1.02    # tighten approximation slowly
    eta = 0.12           # multiplicative‑gradient step size
    eta_decay = 0.9995   # gradual decay of η

    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # 3. Main loop – exponentiated‑gradient on the smooth max
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # self‑convolution
        conv = np.convolve(p, p)               # length 2n‑1
        cur_score = 2.0 * n * np.max(conv)

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
            print(f"[Iter {it:6d}] new best score: {best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 3.1 soft‑max weights for the smooth max
        # --------------------------------------------------------------
        w = _softmax_weights(conv, alpha)

        # --------------------------------------------------------------

        #     g_i = 2 * Σ_j p_j * w_{i+j}

        factor = np.exp(-eta * grad_shift)      # 0 < factor ≤ 1
        p = p * factor
        p = np.clip(p, 0.0, None)               # safety
        s = p.sum()
        if s == 0.0 or np.isnan(s):

            p = best_p.copy()
        else:
            p /= s

            p = np.clip(p + jitter, 0.0, None)
            p /= p.sum()

        eta *= eta_decay

            p /= p.sum()
            no_improve = 0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()

    return result
