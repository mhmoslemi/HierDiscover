Gradient‑based optimizer for the step‑function problem.

The score to minimise is   S(a) = 2 n·max(a*a) / (sum(a))² .
Since the expression is homogeneous, we normalise sum(a)=1 and only
need to minimise  max(a*a).  The optimiser works on the logits of a
soft‑max parametrisation (so the constraints a≥0, Σa=1 are automatic).

A smoothed version of the max‑operator is used:
    f(a) = (1/β)·log Σ_k exp(β·(a*a)_k) .
For small β this is a smooth average; β is increased
geometrically during the run so that the optimisation gradually
focuses on the true maximum.

Gradient:
    ∂f/∂a_i = 2·Σ_k w_k·a_{k‑i}   with   w_k = exp(β·(a*a)_k) / Σ_j exp(β·(a*a)_j)

The convolution and the required correlation are evaluated with
real FFTs (O(N log N)).  An Adam‑style update is applied on the
logits.  The routine starts from a fast parametric sweep over a few
simple families (U‑shaped, sum‑of‑tails, exponential) to obtain a
warm start, then runs the gradient optimiser until the time budget
expires.  Exact scores (using the provided ``evaluate_sequence``) are
computed only occasionally to keep the overhead low.

The function returns the best sequence (as a plain list) found within
the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Return max entry of the self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n-1-i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12           # avoid exact zeros
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def correlation_linear(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation of a (length n) with w (length m):
        (a ⋆ w)[i] = Σ_j a_j·w_{i+j},   i=0..n‑1
    Returns an array of length n.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(A * np.conj(W), n=size)
    offset = m - 1
    return corr_full[offset: offset + n]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best list of floats found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    best_seq = None
    best_score = np.inf
    best_approx = np.inf

    # --------------------------------------------------------------
    # 0.  Warm start from the previously supplied sequence (if any)
    # --------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external = external / s
            approx = 2.0 * len(external) * max_conv_fft(external)
            if approx < best_approx:
                best_approx = approx
                best_seq = external.copy()
                print(f"Seeded from height_sequence_1: approx score={approx:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 1.  Light parametric sweep (fast FFT based) for a warm start
    # --------------------------------------------------------------
    lengths = [3000, 5000, 8000, 12000, 16000]
    p_vals = np.linspace(0.2, 2.0, 19)               # 0.2,0.3,…,2.0
    alphas = np.linspace(0.0002, 0.006, 10)          # exponential decay

    print("=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for p in p_vals:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_approx:
                best_approx = sc
                best_seq = a.copy()
                print(f"new best (U‑pow) n={n:5d} p={p:4.2f} ≈{sc:.6f}")
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_approx:
                best_approx = sc
                best_seq = a.copy()
                print(f"new best (tails) n={n:5d} p={p:4.2f} ≈{sc:.6f}")
        for a_alpha in alphas:
            if time.time() > deadline:
                break
            a = exponential_distribution(n, a_alpha)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_approx:
                best_approx = sc
                best_seq = a.copy()
                print(f"new best (exp)   n={n:5d} α={a_alpha:6.4f} ≈{sc:.6f}")

    # --------------------------------------------------------------
    # 2.  Fallback to a random normalised vector if still empty
    # --------------------------------------------------------------
    if best_seq is None:
        n = 8000
        a = np.random.rand(n) + 1e-9
        a /= a.sum()
        best_seq = a.copy()
        best_approx = 2.0 * n * max_conv_fft(a)
        print(f"fallback random init: n={n}, approx score={best_approx:.6f}")

    # --------------------------------------------------------------
    # 3.  Gradient‑based refinement on the simplex (Adam + smoothed max)
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # Initialise logits (soft‑max parameters) – ensure positivity
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()
    z = np.log(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_opt = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    # Smoothed‑max schedule
    beta_sm = 5.0          # start temperature for soft‑max
    beta_inc = 1.02        # multiplicative increase per iteration
    beta_max = 300.0

    # Learning‑rate (pre‑scaled by Adam's bias‑correction later)
    lr = 0.5

    iteration = 0
    print("\n=== Gradient‑based refinement ===")
    while time.time() < deadline:
        iteration += 1

        # ----- soft‑max to obtain a valid probability vector -----
        z_shift = z - np.max(z)                      # for numerical stability
        a = np.exp(z_shift)
        a = a / a.sum()

        # ----- compute self‑convolution via FFT -----
        conv_len = 2 * n - 1
        size = 1 << ((3 * n - 1).bit_length())       # enough for correlation as well
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)[:conv_len]

        # ----- smoothed max (log‑sum‑exp) -----
        max_conv = conv.max()
        # subtract max to avoid overflow
        exp_shift = np.exp(beta_sm * (conv - max_conv))
        w_sum = exp_shift.sum()
        smooth_max = max_conv + np.log(w_sum) / beta_sm
        weight = exp_shift / w_sum                     # softmax weights (length conv_len)

        # ----- gradient w.r.t. a (using correlation) -----
        # da_i = 2 * Σ_k weight_k * a_{k-i}
        da = 2.0 * correlation_linear(a, weight)

        # ----- gradient w.r.t. logits (soft‑max Jacobian) -----
        grad_z = da - a * np.sum(da)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_opt)
        # optional clipping of the step to keep logits bounded
        step_norm = np.linalg.norm(step)
        if step_norm > 0.5:
            step = step * (0.5 / step_norm)
        z -= step
        # keep logits in a reasonable window
        z = np.clip(z, -50.0, 50.0)

        # ----- record best approximated score occasionally -----
        if iteration % 100 == 0:
            approx_score = 2.0 * n * conv.max()
            if approx_score < best_approx:
                best_approx = approx_score
                best_seq = a.copy()
                print(f"Iter {iteration:6d}: new approx best = {best_approx:.6f}")

        # ----- increase smoothing parameter slowly -----
        beta_sm = min(beta_max, beta_sm * beta_inc)

        # ----- exact evaluation sparsely (to keep overhead low) -----
        if iteration % 500 == 0 or (deadline - time.time() < 30):
            try:
                exact = evaluate_sequence(a.tolist())  # noqa: F821
                if exact < best_score:
                    best_score = exact
                    best_seq = a.copy()
                    print(f"Iter {iteration:6d}: exact score improved to {best_score:.6f}")
            except Exception as e:
                # any problem with the external evaluator – just skip
                print(f"Iter {iteration:6d}: evaluate_sequence failed ({e})")

    # ------------------------------------------------------------------
    # 4.  Final exact evaluation (once) and return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())  # noqa: F821
        print(f"\nFinal exact score: {final_score:.6f}")
    except Exception as e:
        final_score = best_score
        print(f"Final exact evaluation failed ({e}) – using best approximated score.")

    print("\n=== Search finished ===")
    print(f"Best approximated score : {best_approx:.6f}")
    print(f"Best exact score        : {final_score:.6f}")
    print(f"Best length             : {len(best_seq)}")
    return best_seq.tolist()
