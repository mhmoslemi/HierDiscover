def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex {x | x_i ≥ 0 , Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where the condition holds
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
