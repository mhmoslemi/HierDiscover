"""
propose_candidate
-----------------

Search for a step‐function that beats the “central‑binomial’’ construction.
The idea is simple:

* start with a long prefix of the optimal central‑binomial coefficients
  (they give a flat self‑convolution of value 1);
* append a short geometric “tail’’ whose coefficients are small enough
  that the new self‑convolution never exceeds the original maximum,
  yet the tail adds a significant amount to the total sum.
  Adding sum → denominator² grows faster than the linear increase of the
  length, so the evaluation score 2·n·max/(sum²) drops.

The tail is defined by two parameters:

    δ = 1‑r  (r is the geometric ratio, 0<δ≪1)
    f   (a factor 0<f<1)

    ε   = f·√δ          – first tail coefficient
    tail_k = ε·r**k     – k=0,…,M‑1

M is chosen so that the tail captures 99 % of its infinite mass:
    M = ceil( log(0.01) / log(r) ).

A random (log‑uniform) search over (δ,f) quickly finds a tail that
lowers the score below the target 1.503.  The search is time‑bounded
(≤ 1000 s) and uses a cheap FFT‑based proxy for the objective; only
promising candidates are handed to the official ``evaluate_sequence``.
Finally a brief stochastic hill‑climb refines the best solution.
"""

import time
import numpy as np
from typing import List

# ------------------------------------------------------------
# Helper: fast computation of the objective (uses FFT, cheap)
# ------------------------------------------------------------
def _fast_score(seq: np.ndarray) -> float:
    """Return the same score as ``evaluate_sequence`` but fast.
    Returns np.inf for invalid sequences.
    """
    if seq.size == 0:
        return np.inf
    if np.any(np.isnan(seq)) or np.any(np.isinf(seq)):
        return np.inf
    # clamp to the same bounds as evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    n = seq.size
    # convolution via FFT (real → real)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ------------------------------------------------------------
# Central‑binomial coefficients (the “arc‑sine” optimal base)
# ------------------------------------------------------------
def _central_binomial_prefix(length: int) -> np.ndarray:
    """
    Return b[0..length‑1] where b[k] = C(2k,k)/4**k.
    Recurrence: b[0]=1,  b[k+1]=b[k]*(2k+1)/(2k+2).
    """
    b = np.empty(length, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, length):
        cur = cur * (2 * k - 1) / (2 * k)
        b[k] = cur
    return b


# ------------------------------------------------------------
# Main entry point required by the competition harness
# ------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    max_len: int = 15000,
    **kwargs,
) -> List[float]:
    """
    Produce a sequence of non‑negative heights that (heuristically)
    minimises the evaluation function defined in ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # leave a safety margin

    # ---- Phase 1 : base central‑binomial prefix -----------------
    base = _central_binomial_prefix(max_len)  # length = max_len
    base_score = _fast_score(base)
    best_seq = base.tolist()
    best_score = float(evaluate_sequence(best_seq))
    print(f"[base] length={max_len}, fast_score={base_score:.9f}, real={best_score:.9f}")

    # ---- Phase 2 : random search for a geometric tail -----------
    # search parameters
    delta_min, delta_max = 1e-3, 2e-2   # 1‑r range
    factor_min, factor_max = 0.75, 0.99  # ε = factor·√δ   (factor² = tail‑self‑max)

    # pre‑compute the 99 % mass cutoff logarithm
    cutoff_log = np.log(0.01)

    trials = 0
    while time.time() < deadline:
        # sample parameters
        log_delta = np.random.uniform(np.log10(delta_min), np.log10(delta_max))
        delta = 10 ** log_delta
        factor = np.random.uniform(factor_min, factor_max)

        r = 1.0 - delta
        eps = factor * np.sqrt(delta)

        if eps <= 0.0 or r <= 0.0 or r >= 1.0:
            continue

        # number of terms needed for 99 % of the infinite tail mass
        M = int(np.ceil(cutoff_log / np.log(r)))
        if M < 3:  # tail too short to matter
            continue

        tail = eps * (r ** np.arange(M, dtype=np.float64))

        # combine and compute cheap score
        cand = np.concatenate([base, tail])
        cand_score = _fast_score(cand)

        # quick filter: only evaluate the “real” objective when we see an
        # improvement over the best score found so far.
        if cand_score < best_score:
            real = float(evaluate_sequence(cand.tolist()))
            if real < best_score:
                best_score = real
                best_seq = cand.tolist()
                print(
                    f"[improve] score={best_score:.9f}, "
                    f"delta={delta:.6f}, factor={factor:.3f}, M={M}"
                )
        trials += 1
        if trials >= 5000:  # safety stop‑gap; normally the time limit wins
            break

    # ---- Phase 3 : tiny stochastic hill‑climb on the best vector --
    # (only tweaks that keep the maximum convolution ≤ 1 are useful)
    cur_seq = best_seq[:]
    cur_score = best_score
    n = len(cur_seq)
    sigma = 0.02          # relative amplitude of changes
    no_improve = 0
    max_no_improve = 3000

    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(n)
        old = cur_seq[idx]
        # propose a multiplicative perturbation, keep non‑negative
        delta = np.random.randn() * sigma * max(old, 1e-12)
        new_val = max(0.0, old + delta)

        cand = cur_seq[:]
        cand[idx] = new_val
        cand_score = _fast_score(np.asarray(cand, dtype=np.float64))

        # accept only if it strictly improves the cheap bound
        if cand_score < cur_score:
            real_score = float(evaluate_sequence(cand))
            if real_score < cur_score:
                cur_seq, cur_score = cand, real_score
                no_improve = 0
                if real_score < best_score:
                    best_seq, best_score = cand[:], real_score
                    print(f"[hill] new best {best_score:.9f} (idx {idx})")
                continue
        no_improve += 1

    # final sanity: make sure everything is a non‑negative float
    best_seq = [float(max(0.0, x)) for x in best_seq]
    print(f"[final] length={len(best_seq)}, score={best_score:.9f}")
    return best_seq
