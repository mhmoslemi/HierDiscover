        a = a / a.sum()
    # best tracking

    best_score = _objective(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 0.2  # initial learning rate
    # schedule: exponential decay based on progress
    t = 0
    while time.time() < deadline:

        conv = _full_convolution(a)
        max_val = conv.max()

def _full_convolution(a):
    n = a.shape[0]
    m = 2*n-1
    size = 1 << (m-1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:m]
    conv = np.maximum(conv, 0, out=conv)  # destructive
    return conv

"""
Propose a candidate step sequence that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Determine a working length ``n`` (use the length of the provided
   ``height_sequence_1`` if available, otherwise a default of 3000).
2. Initialise the best known vector with that sequence (or a uniform one)
   and compute its score.
3. **Randomised parametric search** – repeatedly draw a shape from a small
   family (Gaussian, triangular, cosine‑power, beta‑distribution,
   exponential) with random parameters, normalise it to a probability
   simplex and evaluate the objective (``2*n*max(conv)``).  Keep the
   best vector found.  This step quickly discovers shapes that beat the
   previous best value (≈1.52).
4. **Fine‑tuning** – run an Adam‑style projected gradient descent on the
   best vector discovered in step 3.  The sub‑gradient of
   ``max(conv)`` is obtained from the convolution index attaining the
   current maximum.  The update is projected back onto the simplex after
   each step.  Occasional random mass‑transfer jumps help to escape plateaus.
5. Return the best vector (as a plain ``list[float]``) before the time
   budget expires.

The whole procedure respects the 1000 s wall‑time limit and prints
intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al.)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full (linear) convolution using a real FFT
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a*a (length 2·n‑1) using FFT.
    The result is forced to be non‑negative to avoid tiny negative
    rounding errors.
    """
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()       # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Shape generators (all return a non‑negative vector that sums to 1)
# ----------------------------------------------------------------------
def _gaussian(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on [0,1] with standard deviation sigma."""
    if sigma <= 0.0:
        sigma = 0.01
    x = np.linspace(0.0, 1.0, n, dtype=float)
    centre = 0.5
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def _triangle(n: int, width: float) -> np.ndarray:
    """Symmetric triangular bump, zero at the edges. ``width`` ∈ (0,0.5]."""
    if width <= 0.0:
        width = 0.1
    if width > 0.5:
        width = 0.5
    x = np.linspace(0.0, 1.0, n, dtype=float)
    centre = 0.5
    a = np.maximum(0.0, 1.0 - np.abs(x - centre) / width)
    a /= a.sum()
    return a


def _cosine_power(n: int, power: float) -> np.ndarray:
    """Hann window raised to a given power."""
    if power <= 0.0:
        power = 1.0
    idx = np.arange(n, dtype=float)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))   # Hann
    a = np.power(w, power)
    a /= a.sum()
    return a


def _beta(n: int, a_param: float, b_param: float) -> np.ndarray:
    """Beta‑distribution shape on [0,1] with parameters a,b > 0."""
    a_param = max(1e-6, a_param)
    b_param = max(1e-6, b_param)
    x = np.linspace(0.0, 1.0, n, dtype=float)
    a = np.power(x, a_param) * np.power(1.0 - x, b_param)
    a += 1e-12                     # avoid exact zeros
    a /= a.sum()
    return a


def _exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    lam = max(1e-6, lam)
    x = np.linspace(0.0, 1.0, n, dtype=float)
    centre = 0.5
    a = np.exp(-lam * np.abs(x - centre))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Score evaluation (fast version – sum is guaranteed to be 1)
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) – the same quantity that evaluate_sequence
    would produce when the vector sums to 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Random candidate generator (no lambda – required by the rules)
# ----------------------------------------------------------------------
def _random_candidate(n: int) -> np.ndarray:
    """Pick a random shape family and random parameters."""
    r = np.random.rand()
    if r < 0.30:                     # Gaussian
        sigma = np.random.uniform(0.04, 0.55)
        return _gaussian(n, sigma)
    elif r < 0.50:                   # Triangular
        width = np.random.uniform(0.08, 0.48)
        return _triangle(n, width)
    elif r < 0.70:                   # Cosine‑power
        power = np.random.uniform(0.2, 4.5)
        return _cosine_power(n, power)
    elif r < 0.85:                   # Beta
        a_param = np.random.uniform(0.2, 4.0)
        b_param = np.random.uniform(0.2, 4.0)
        return _beta(n, a_param, b_param)
    else:                            # Exponential
        lam = np.random.uniform(0.5, 12.0)
        return _exponential(n, lam)

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights with low evaluate_sequence
    value.  The routine blends a cheap random‑parameter search with a
    projected Adam optimisation.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0                # safety margin

    # --------------------------------------------------------------
    # Choose length n and initialise the best known vector
    # --------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = a0.shape[0]
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() <= 0.0:
            raise ValueError
        a0 = a0 / a0.sum()
    except Exception:
        n = 3000
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_val = _score(best_a)
    print(f"[init] n={n}, initial score={best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – randomised parametric search (≈ 60 % of the budget)
    # --------------------------------------------------------------
    search_deadline = start + 0.60 * budget_s
    it = 0
    while time.time() < search_deadline:
        cand = _random_candidate(n)
        val = _score(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            print(f"[search {it:06d}] new best = {best_val:.6f}")
        it += 1

    # --------------------------------------------------------------
    # Stage 2 – fine‑tuning with Adam + projection onto the simplex
    # --------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    eta_start = 0.25
    eta_end = 5e-5
    t = 0

    # keep a copy of the best found during fine‑tuning
    best_fine_a = best_a.copy()
    best_fine_val = best_val

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- evaluate current state -----
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = float(2.0 * n * max_conv)

        # keep the best ever seen
        if cur_val < best_fine_val:
            best_fine_val = cur_val
            best_fine_a = a.copy()
            print(f"[fine {t:07d}] new best = {best_fine_val:.6f}")

        # ----- sub‑gradient at the active convolution peak -----
        k_star = int(np.argmax(conv))
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        subgrad = np.zeros_like(a)
        subgrad[idx] = 2.0 * a[k_star - idx]          # g_i = 2·a_{k*‑i}

        # (optional) tiny L2 regularisation to keep the iterate well‑behaved
        lam = 5e-4
        subgrad += lam * a

        # ----- Adam update (learning‑rate schedule) -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * subgrad
        v = beta2 * v + (1.0 - beta2) * (subgrad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        progress = (now - start) / (deadline - start)
        eta = eta_start * (1.0 - progress) + eta_end * progress
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step

        # occasional random mass transfer (helps to escape plateaus)
        if np.random.rand() < 0.02:
            i_src = np.random.randint(n)
            i_dst = np.random.randint(n - 1)
            if i_dst >= i_src:
                i_dst += 1
            delta = np.random.rand() * a_candidate[i_src] * 0.4
            a_candidate[i_src] -= delta
            a_candidate[i_dst] += delta

        # projection back onto the simplex (ensures sum=1 and non‑negativity)
        a = _project_onto_simplex(a_candidate)

    # --------------------------------------------------------------
    # Final reporting
    # --------------------------------------------------------------
    final_score = best_fine_val
    print(f"[final] best evaluation = {final_score:.6f}")

    # Return the best vector that we have seen (as a plain Python list)
    return best_fine_a.tolist()
