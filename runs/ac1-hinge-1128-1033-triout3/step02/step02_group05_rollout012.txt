"""
propose_candidate
-----------------

Search for a non‑negative step‑function that beats the classic
“constant‑convolution’’ construction (value ≈ π/2 ≈ 1.5708).
The optimisation problem is

    minimise  2·n·max(conv(a,a)) / (∑a)²

which is scale‑invariant.  By fixing ∑a = 1 the objective becomes
`2·n·max(conv)`.  We therefore minimise the maximum of the
self‑convolution while keeping the vector on the probability simplex.

The algorithm runs several optimisation runs (different lengths) inside the
time budget:

* initialise `a` with the constant‑convolution coefficients
  (truncated and normalised to sum = 1);
* use a smooth approximation of the max (`softmax`) with a slowly
  increasing temperature parameter `γ`;
* compute the gradient of the smooth objective analytically
  (the gradient of each convolution entry is 2·a shifted);
* apply Adam updates, then project back onto the simplex (clip to ≥0
  and renormalise);
* keep the best sequence ever seen (by the exact objective).

If the optimiser does not improve on the fallback (the 20 000‑term
constant‑convolution prefix) we return that fallback instead.
All numbers are clamped to the range required by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to unit sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        # fall back to the uniform distribution
        v[:] = 1.0 / v.shape[0]
    return v


def _softmax_weights(x: np.ndarray, gamma: float) -> np.ndarray:
    """Return a soft‑max weight vector approximating the max of `x`."""
    # shift for numerical stability
    m = np.max(x)
    e = np.exp(gamma * (x - m))
    return e / e.sum()


def _constant_convolution_prefix(length: int) -> np.ndarray:
    """
    Return the first `length` coefficients of the infinite series
    a_k = binom(2k,k) / 4**k, normalised to sum = 1.
    """
    coeffs = np.empty(length, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, length):
        cur = cur * (2 * k - 1) / (2 * k)  # recurrence for the central binomial ratio
        coeffs[k] = cur
    coeffs /= coeffs.sum()
    return coeffs


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that yields a low value of
    `evaluate_sequence`.  The routine tries a few Adam‑optimised
    runs on different lengths; if none improve on the classic
    construction it falls back to a long constant‑convolution prefix.
    """
    # ---------- timing ----------
    # we keep a small safety margin (≈5 s) for the harness
    deadline = time.time() + budget_s - 5.0

    np.random.seed(seed)

    # ---------- fall‑back construction (the one already known) ----------
    MAX_FALLBACK_LEN = 20000
    fallback_coeffs = _constant_convolution_prefix(MAX_FALLBACK_LEN)
    fallback_score = evaluate_sequence(fallback_coeffs.tolist())
    best_score = float(fallback_score)
    best_seq = fallback_coeffs.copy()
    print(f"[fallback] length={MAX_FALLBACK_LEN}, score={best_score:.10f}")

    # ---------- optimisation parameters ----------
    # lengths we will try – larger lengths give a smaller theoretical lower bound
    # but each iteration gets more expensive (O(n²)).  We keep a modest set.
    lengths = [128, 256, 512, 1024]   # feel free to adjust if time permits

    # Adam hyper‑parameters (single set for all runs)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05                     # base learning rate (scaled by Adam)

    # ---------- main optimisation loop ----------
    for n in lengths:
        if time.time() > deadline:
            break

        # initialise a with the constant‑convolution shape, add a tiny random jitter
        a = _constant_convolution_prefix(n)
        a += 1e-3 * np.random.rand(n)
        a = _project_to_simplex(a)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        it = 0
        # we cap the number of iterations per length by a cheap heuristic
        max_iters = 8000
        while time.time() < deadline and it < max_iters:
            it += 1

            # ---- convolution and smooth max ----
            conv = np.convolve(a, a)                 # length 2n‑1
            # γ grows slowly so early steps are smooth, later they focus on the true max
            gamma = 10.0 + 0.02 * it
            w = _softmax_weights(conv, gamma)        # weights sum to 1
            soft_max = np.dot(w, conv)               # smooth approximation of max(conv)

            # ---- gradient of the smooth max ----
            # grad_M = 2 * cross‑correlation of a and w
            cross = np.convolve(a[::-1], w, mode="full")   # length 3n‑2
            grad_M = 2.0 * cross[n - 1 : 2 * n - 1]        # vector length n

            # ---- gradient of the objective 2·n·soft_max (∑a=1) ----
            grad_obj = 2.0 * n * grad_M - 4.0 * n * soft_max * np.ones_like(a)

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * grad_obj
            v = beta2 * v + (1.0 - beta2) * (grad_obj ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = base_lr / (np.sqrt(v_hat) + eps) * m_hat

            a = a - step
            a = _project_to_simplex(a)

            # ---- occasional exact evaluation (cheap) ----
            if it % 250 == 0:
                max_b = np.max(np.convolve(a, a))
                exact_score = 2.0 * n * max_b
                if exact_score < best_score:
                    best_score = float(exact_score)
                    best_seq = a.copy()
                    print(f"[improve] n={n} it={it} score={best_score:.10f}")

        # end of one length
        print(f"[finished] n={n}, best_score_sofar={best_score:.10f}")

    # ----------------------------------------------------------------------
    # final sanity checks & return
    # ----------------------------------------------------------------------
    # Clamp the final sequence to the limits required by `evaluate_sequence`.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # As a final safeguard we ask the evaluator for the exact score.
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] length={len(best_seq)}, evaluated_score={final_score:.10f}")

    return best_seq.tolist()
