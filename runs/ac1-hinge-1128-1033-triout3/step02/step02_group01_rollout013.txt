"""
Propose a low‑score step‑function.

The algorithm consists of three stages:

1.  Deterministic sweep over several parametric families (U‑shapes,
    beta‑shapes, exponential decay, pure harmonic).  The best vector
    found is kept as a seed.

2.  A refined stochastic search that combines three move types:
        – projected sub‑gradient step (as in the reference code);
        – mirror‑descent style multiplicative update;
        – directed random mass‑transfer from a high‑gradient index to a
          low‑gradient index.
    The active set for the sub‑gradient is enlarged to include all
    convolution entries that are within a tiny relative tolerance of the
    current maximum, which yields a smoother descent direction.

3.  Simple simulated‑annealing acceptance and occasional reheating to
    escape plateaus.

All convolutions are evaluated by a single FFT, which makes the inner
loop fast enough to explore millions of candidates within the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length 2*len(a)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is non‑negative, sum=1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a vector already normalized."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to the two edges: ((i+1)*(n-i))^(-p)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def beta_distribution(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}.  a[i] ∝ (i+1)^{α‑1} (n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * (j ** (beta_param - 1.0)) + 1e-14
    a /= a.sum()
    return a


def exp_decay_distribution(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the edges: exp(-γ·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-gamma * d) + 1e-14
    a /= a.sum()
    return a


def harmonic_left_distribution(n: int, p: float) -> np.ndarray:
    """One‑sided power law: a[i] ∝ (i+1)^{-p} (mass concentrated on the left)."""
    i = np.arange(n) + 1
    a = i.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) discovered before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over several families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # lengths to try – keep them modest so the sweep finishes quickly
    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 16000, 20000]
    # U‑shape exponent grid
    p_grid = np.arange(0.1, 2.01, 0.1)
    # Beta parameters (symmetric for now)
    beta_grid = np.arange(0.2, 2.01, 0.2)
    # exponential decay rates
    gamma_grid = np.arange(0.001, 0.021, 0.001)

    print("=== Deterministic sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # --- U‑shape families ------------------------------------------------
        for p in p_grid:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (u_prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # --- Beta family ----------------------------------------------------
        for alpha in beta_grid:
            # keep symmetry (alpha == beta) for speed; asymmetric cases could be added later
            a = beta_distribution(n, alpha, alpha)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (beta)    n={n:5d} α={alpha:4.2f}  score={sc:.6f}")

        # --- Exponential decay -----------------------------------------------
        for gamma in gamma_grid:
            a = exp_decay_distribution(n, gamma)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f" new best (exp)    n={n:5d} γ={gamma:5.3f}  score={sc:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. Include an external construction if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            total = external.sum()
            if total > 0.0:
                external /= total
                sc_ext = score_from_vector(external)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement (projected sub‑gradient / mirror / directed moves)
    # ------------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score
    n = len(current)

    # hyper‑parameters (tuned for speed / robustness)
    temperature = 1.0
    cooling_rate = 0.9998          # slightly slower cooling & step decay
    base_step = 0.25               # initial magnitude of updates
    step_size = base_step

    iteration = 0
    last_improve_iter = 0

    print("\n=== Starting stochastic refinement ===")
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 4a – compute convolution and an enlarged active set
        # --------------------------------------------------------------
        conv = full_convolution_fft(current)
        max_val = conv.max()
        # include all entries within a very tight relative band of the maximum
        rel_tol = 1e-5
        active_idx = np.where(conv >= max_val * (1.0 - rel_tol))[0]

        # --------------------------------------------------------------
        # 4b – sub‑gradient assembled from the active set
        # --------------------------------------------------------------
        grad = np.zeros_like(current)
        for k in active_idx:
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            i_vals = np.arange(start_i, end_i + 1)
            j_vals = k - i_vals
            grad[i_vals] += 2.0 * current[j_vals]

        # --------------------------------------------------------------
        # 4c – choose a move type
        # --------------------------------------------------------------
        r = np.random.rand()
        if r < 0.35:
            # ---- mirror‑descent (multiplicative) update ----
            cand = current * np.exp(-step_size * grad)
            cand = simplex_projection(cand)

        elif r < 0.65:
            # ---- directed random mass transfer (high‑grad → low‑grad) ----
            # pick source from top‑few gradient entries, sink from bottom‑few
            top_k = 6
            low_k = 6
            src_indices = np.argpartition(-grad, top_k)[:top_k]
            dst_indices = np.argpartition(grad, low_k)[:low_k]
            src = np.random.choice(src_indices)
            dst = np.random.choice(dst_indices)
            if src != dst and current[src] > 1e-12:
                # move a fraction of source mass towards the sink
                delta = current[src] * np.random.rand() * 0.5
                cand = current.copy()
                cand[src] -= delta
                cand[dst] += delta
                cand = simplex_projection(cand)
            else:
                continue  # nothing changed – skip this iteration

        else:
            # ---- projected sub‑gradient step (additive) ----
            cand = current - step_size * grad
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)

        # --------------------------------------------------------------
        # 4d – evaluate candidate (cheap internal score)
        # --------------------------------------------------------------
        cand_score = score_from_vector(cand)

        # --------------------------------------------------------------
        # 4e – simulated‑annealing acceptance
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            # global best update
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 4f – anneal temperature and step size
        # --------------------------------------------------------------
        temperature *= cooling_rate
        step_size *= cooling_rate

        # --------------------------------------------------------------
        # 4g – stagnation handling (re‑heat occasionally)
        # --------------------------------------------------------------
        if iteration - last_improve_iter > 50000:
            bump = np.random.rand(n) * 1e-4
            current = current + bump
            current = simplex_projection(current)
            current_score = score_from_vector(current)
            last_improve_iter = iteration
            temperature = 1.0
            step_size = base_step
            print(f" * Re‑heat at iter {iteration}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # external evaluator will clip and renormalise again, so we return the raw vector
    return best_seq.tolist()
