def grad_softmax(a, beta):
    n = len(a)
    # compute convolution a*a via FFT (size for conv)
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]  # length 2n-1

    # compute softmax weights w
    # shift to avoid overflow
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()

    # compute gradient: 2 * (w * a_rev) at offset
    a_rev = a[::-1]
    size2 = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=size2)
    A_rev = np.fft.rfft(a_rev, n=size2)
    conv2 = np.fft.irfft(W * A_rev, n=size2)[: len(w) + len(a_rev) - 1]  # length 3n-2

    # grad_i = 2 * conv2[i + n - 1]
    grad = 2.0 * conv2[n - 1: n - 1 + n]  # slice n elements
    return conv.max(), grad, w

def gradient_step(a, beta, eta):
    n = len(a)
    # compute conv
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]

    # compute softmax weights
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()

    a_rev = a[::-1]
    size2 = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size2)
    A_rev = np.fft.rfft(a_rev, n=size2)

"""
Propose a step‑function that minimises the given evaluation metric.

Algorithm
---------
1. **Reuse** an externally supplied sequence (``height_sequence_1``) if it exists.
2. **Fast parametric sweep** over several families (U‑shaped power law,
   sum‑of‑tails, exponential, geometric) and a variety of lengths.
   The best few candidates (by the fast FFT based score) are kept as warm
   starts.
3. **Smooth‑max gradient descent** on the simplex.
   - The convolution‐max is approximated by a soft‑max (temperature β).
   - The gradient of the smooth max is obtained by a single extra FFT
     (cross‑correlation of the soft‑max weights with the reversed vector).
   - An exponentiated‑gradient (mirror‑descent) update keeps the iterate in
     the probability simplex.
   - β is slowly increased, the step size η is slowly decreased.
   - Occasionally a tiny Gaussian jitter is added to escape shallow minima.
   The best solution over all restarts is retained.
4. **Final exact evaluation** with the official ``evaluate_sequence``.
   The function returns the best list of floats found before the time budget
   expires.

The whole search respects a user‑supplied time budget (default 1000 s) and
prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """
    Fast computation of max entry of the self‑convolution a*a using a real FFT.
    a may be any real vector (non‑negative for our problem).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def fast_score(a: np.ndarray) -> float:
    """
    Approximate evaluation used inside the optimizer:
        score ≈ 2·n·max(a*a)
    (identical to the true value except for the cheap FFT convolution.)
    """
    return 2.0 * a.shape[0] * max_conv_fft(a)


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i, n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^(-p)+(n‑i)^(-p)."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def geometric_distribution(n: int, r: float) -> np.ndarray:
    """Geometric decay from the edges: a[i] ∝ r**min(i,n‑1‑i). 0<r<1."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = (r ** d) + 1e-12
    a /= a.sum()
    return a


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max weights for a convolution vector.
    The shift by the max value prevents overflow.
    """
    max_val = conv.max()
    exp_vals = np.exp(beta * (conv - max_val))
    return exp_vals / exp_vals.sum()


def gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max (soft‑max) of the self‑convolution.
    For a fixed weight vector w (length 2·n‑1) the gradient is
        grad_i = 2 * Σ_k w_k * a_{k-i}
    which can be obtained by a single FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]                          # reversed a
    # length needed for convolution of w (len 2n‑1) with a_rev (len n)
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv2 = np.fft.irfft(W * A_rev, n=size)[: len(w) + n - 1]
    # relevant slice corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv2[n - 1 : n - 1 + n]
    return grad


def exponentiated_gradient_step(a: np.ndarray, beta: float, eta: float) -> np.ndarray:
    """
    Perform a single mirror‑descent step on the simplex:
        a' = a * exp(-η·∇)
    followed by renormalisation.  A tiny floor prevents numerical zeros.
    """
    # compute convolution and soft‑max weights
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

    w = softmax_weights(conv, beta)

    # gradient of the smooth max
    grad = gradient_smooth_max(a, w)

    # exponentiated update
    factor = np.exp(-eta * grad)
    a_new = a * factor
    # avoid exact zeros (they would never leave the simplex)
    eps = 1e-12
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new


def random_candidate(n: int) -> np.ndarray:
    """A random point on the simplex (Dirichlet(1))."""
    a = np.random.rand(n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Seed from the externally supplied sequence if present.
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)  # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
            else:
                external = external / s
            score_ext = fast_score(external)
            best_score = score_ext
            best_seq = external.copy()
            print(f"Seeded from height_sequence_1: score≈{score_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Deterministic sweep over lengths & parametric families.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 20000, 30000]
    p_vals = np.linspace(0.2, 2.0, 19)      # 0.2 … 2.0
    alphas = np.linspace(0.0003, 0.006, 12)
    r_vals = np.linspace(0.85, 0.995, 12)

    # keep the K best candidates (by fast_score) for later refinement
    K = 12
    candidate_pool = []                     # list of (score, array)

    print("=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # ---- U‑shaped power‑law ----
        for p in p_vals:
            a = u_shaped_distribution(n, p)
            sc = fast_score(a)
            if len(candidate_pool) < K:
                candidate_pool.append((sc, a))
                candidate_pool.sort(key=lambda x: x[0])
            elif sc < candidate_pool[-1][0]:
                candidate_pool[-1] = (sc, a)
                candidate_pool.sort(key=lambda x: x[0])

        # ---- Sum‑of‑tails ----
        for p in p_vals:
            a = sum_tails_distribution(n, p)
            sc = fast_score(a)
            if len(candidate_pool) < K:
                candidate_pool.append((sc, a))
                candidate_pool.sort(key=lambda x: x[0])
            elif sc < candidate_pool[-1][0]:
                candidate_pool[-1] = (sc, a)
                candidate_pool.sort(key=lambda x: x[0])

        # ---- Exponential decay ----
        for a_alpha in alphas:
            a = exponential_distribution(n, a_alpha)
            sc = fast_score(a)
            if len(candidate_pool) < K:
                candidate_pool.append((sc, a))
                candidate_pool.sort(key=lambda x: x[0])
            elif sc < candidate_pool[-1][0]:
                candidate_pool[-1] = (sc, a)
                candidate_pool.sort(key=lambda x: x[0])

        # ---- Geometric decay ----
        for r in r_vals:
            a = geometric_distribution(n, r)
            sc = fast_score(a)
            if len(candidate_pool) < K:
                candidate_pool.append((sc, a))
                candidate_pool.sort(key=lambda x: x[0])
            elif sc < candidate_pool[-1][0]:
                candidate_pool[-1] = (sc, a)
                candidate_pool.sort(key=lambda x: x[0])

        print(f"Length {n:5d} finished, current pool best score≈{candidate_pool[0][0]:.6f}")

    # Ensure we have at least one candidate
    if not candidate_pool:
        # fallback: a random normalised vector of moderate size
        n = 8000
        a = random_candidate(n)
        candidate_pool.append((fast_score(a), a))

    # ------------------------------------------------------------------
    # 2.  Refine each candidate with smooth‑max gradient descent.
    # ------------------------------------------------------------------
    # Sort pool by initial score (ascending)
    candidate_pool.sort(key=lambda x: x[0])
    print("\n=== Gradient‑descent refinement ===")

    for init_idx, (init_score, a0) in enumerate(candidate_pool):
        if time.time() > deadline:
            break
        n = a0.shape[0]
        a = a0.copy()
        cur_score = fast_score(a)
        # hyper‑parameters – they evolve during the run
        beta = 5.0             # start temperature for soft‑max
        beta_max = 300.0
        eta = 0.6              # initial step size for exponentiated update
        eta_min = 1e-4
        it = 0

        while time.time() < deadline:
            it += 1

            # ----------------------------------------------------------
            # evaluate current solution (fast)
            # ----------------------------------------------------------
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[cand {init_idx+1}/{len(candidate_pool)}] "
                      f"iter {it:5d} new global best≈{best_score:.6f} (n={n})")
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break

            # ----------------------------------------------------------
            # gradient‑descent step
            # ----------------------------------------------------------
            a_new = exponentiated_gradient_step(a, beta, eta)

            # occasional jitter to avoid stagnation
            if it % 250 == 0:
                jitter = np.random.normal(scale=1e-3, size=n)
                a_new = a_new + jitter
                a_new = np.maximum(a_new, 0.0)
                a_new /= a_new.sum()

            # evaluate the candidate
            new_score = fast_score(a_new)

            # ----------------------------------------------------------
            # acceptance (monotone with a little simulated‑annealing fallback)
            # ----------------------------------------------------------
            accept = False
            if new_score <= cur_score:
                accept = True
            else:
                # prob. to accept a worsening move (temperature≈η)
                prob = np.exp((cur_score - new_score) / max(eta, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                a = a_new
                # optionally tighten parameters after a successful move
                beta = min(beta * 1.005, beta_max)
                eta = max(eta * 0.999, eta_min)

            # decay parameters even when not accepted (helps convergence)
            beta = min(beta * 1.001, beta_max)
            eta = max(eta * 0.9999, eta_min)

            # safety break if iterations become too many without improvement
            if it % 2000 == 0 and it > 4000 and cur_score - best_score < 1e-6:
                break

        # early stop if we already met the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3.  Final exact evaluation (once) and report.
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but guard against it
        raise RuntimeError("Search failed to produce any candidate.")

    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence (fallback to fast score): {e}")
        exact_score = fast_score(best_seq)

    print("\n=== Search finished ===")
    print(f"Best fast score : {best_score:.6f}")
    print(f"Exact score     : {exact_score:.6f}")
    print(f"Best length     : {len(best_seq)}")
    return best_seq.tolist()
