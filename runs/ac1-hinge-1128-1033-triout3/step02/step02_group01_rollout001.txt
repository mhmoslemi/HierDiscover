"""
Propose a step‑function with a lower evaluation score.

Algorithm Overview
------------------
1. **Initialization**
   * Use an existing good construction (`height_sequence_1`) if available,
     otherwise start from a random point on the probability simplex.
   * Choose a problem size `n`.  We keep the length of the starting vector
     (or round up to the next power‑of‑two for FFT efficiency).

2. **Smooth‑max optimisation**
   * The true objective is `max(conv)` where `conv = a * a` (full convolution).
   * We replace the nondifferentiable max by the *log‑sum‑exp* smooth
     approximation:
          Lα(a) = (1/α) log Σ exp(α·conv_i)
     As α ↑ ∞, `Lα → max`.  For a fixed α,
     the gradient w.r.t. `a` is
          ∇Lα = 2·(a ⋆ w)      where  w_i = exp(α·conv_i)/Σexp(α·conv_i)
     (`⋆` denotes cross‑correlation).  Both the convolution and the
     cross‑correlation are evaluated by FFT (O(n log n)).

3. **Projected gradient descent with annealing**
   * Perform gradient steps on the simplex:
        a' = Π_S ( a − η·∇Lα )
     where `Π_S` is the Euclidean projection onto the simplex.
   * Step size `η` decays like `η₀ / sqrt(iter)`.
   * The smoothing parameter α starts small (α≈2) and is multiplied
     by 1.005 each iteration, up to a configurable `α_max` (≈30).
   * A simple simulated‑annealing acceptance rule allows occasional
     uphill moves and helps escape shallow local minima.

4. **Escapes and restarts**
   * Every few thousand iterations a tiny random perturbation is added
     and the vector is re‑projected.
   * If no improvement is seen for a long stretch we optionally re‑seed
     from a fresh random simplex point (keeping the best ever found).

5. **Termination**
   * The loop runs until the time budget is exhausted (≈ 1000 s).
   * The best sequence discovered (as a plain list of floats) is returned.

All heavy operations are performed with NumPy FFTs, and the code does not

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of (fixed) size L."""
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def cross_corr_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Compute the cross‑correlation (a ⋆ w)[k] = Σ_j a_j·w_{j+k}
    for k=0,…,len(a)-1 using FFT of size L.
    """
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w, n=L)
    # conj(A) * W gives the linear cross‑correlation
    corr = np.fft.irfft(A.conj() * W, n=L)
    return corr[: a.shape[0]]  # keep only the first n entries


def logsumexp(x: np.ndarray) -> float:
    """Stable log‑sum‑exp."""
    x_max = np.max(x)
    return float(x_max + np.log(np.sum(np.exp(x - x_max))))


def smooth_max_and_grad(a: np.ndarray, alpha: float, L: int):
    """
    Compute the smooth approximation of max(conv) and its gradient.

    Returns
    -------
    smooth_max : float          (≈ max(conv) when α large)
    grad       : np.ndarray    (shape = a.shape)
    conv       : np.ndarray    (full convolution, length 2n‑1)  (for diagnostics)
    """
    # full convolution a*a
    conv = full_convolution_fft(a, L)

    # smooth max via log‑sum‑exp
    scaled = alpha * conv
    lse = logsumexp(scaled)
    smooth_max = lse / alpha

    # softmax weights w_i  (sum_i w_i = 1)
    w = np.exp(scaled - lse)

    # gradient = 2 * (a ⋆ w)
    grad = 2.0 * cross_corr_fft(a, w, L)
    return smooth_max, grad, conv


def max_convolution(a: np.ndarray, L: int) -> float:
    """Return max_k (a*a)[k] (fast, FFT based)."""
    conv = full_convolution_fft(a, L)
    return float(conv.max())

    """
    Exact score returned by evaluate_sequence for a vector already normalised
    to sum == 1 (i.e. 2 * n * max(conv)).
    """
    n = a.shape[0]
    return 2.0 * n * max_convolution(a, L)
