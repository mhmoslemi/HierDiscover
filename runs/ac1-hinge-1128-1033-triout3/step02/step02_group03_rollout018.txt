def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 1153

    # baseline binomial shape
    a = np.empty(n, dtype=np.float64)

    a = a / a.sum()

    # keep best
    best = a.copy()
    best_score = float('inf')
    # evaluate initial score
    conv = np.convolve(best, best)
    best_score = 2 * n * conv.max() / (best.sum()**2)  # sum ==1

    start = time.time()
    deadline = start + budget_s - 5.0
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        conv = np.convolve(a, a)
        max_conv = conv.max()
        cur_score = 2 * n * max_conv  # sum=1

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[Iter {iteration}] new best {best_score:.6f}")

        # Build LP matrix
        A = np.zeros((2*n-1, n), dtype=np.float64)
        # each column i corresponds to shifted f[i] values
        for i in range(n):
            A[i:i+n, i] = a  # f

        # Solve LP: maximize sum(b) subject to A b <= max_conv, b>=0
        c = -np.ones(n, dtype=np.float64)  # minimize -sum(b)
        # bound vector
        b_ub = np.full(2*n-1, max_conv, dtype=np.float64)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True})
        except Exception as e:
            print(f"LP failed: {e}")
            break

        if not res.success:
            # fallback to simple multiplicative update
            grad = np.zeros_like(a)
            k = int(np.argmax(conv))
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            grad[i_start:i_end+1] = 2.0 * a[k - np.arange(i_start, i_end+1)]
            eta = 0.01

Propose a step‑function (non‑negative heights) that minimises

        M = 2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm follows the Frank‑Wolfe style scheme described in the
statement:

1. Initialise a distribution `a` from the exact series whose
   self‑convolution is constant on the first `n` positions:
         a_i ∝ C(2i,i) / 4**i   (central binomial coefficients).

   After normalisation Σ a_i = 1 the starting bound is close to π/2
   (≈ 1.57).

2. Iteratively improve the bound:
   * Compute `M = max(conv(a, a))`.
   * Solve a linear programme

         maximise   Σ b_i
         subject to conv(a, b) ≤ M      (2n‑1 linear inequalities)
                    b_i ≥ 0

     The LP finds a vector `b` that can carry the largest total mass
     while not increasing the maximal convolution value.
   * Scale `b` to a probability vector `g` (Σ g_i = 1) and move the
     current iterate towards `g` by a back‑tracking line‑search on the
     mixing weight `t`.  The line‑search guarantees a strict decrease
     of the objective.
   * Keep the best iterate seen so far.

3. When progress stalls, a cheap projected‑gradient step (soft‑max
   smoothing) is used as a fallback, together with occasional random
   jitter to escape plateaus.

The whole procedure respects the 1000 s time budget and returns the
best sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _build_conv_matrix(p: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that  conv(p, b)[k] = (A @ b)[k]
    for any vector b (full convolution with *zero padding*).

    A has shape (2*n-1, n).  Entry A[k, j] = p[k-j] if 0 <= k-j < n,
    otherwise 0.
    """
    n = p.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A.fill(0.0)
    # each column j is the vector p shifted by +j positions
    for j in range(n):
        A[j : j + n, j] = p
    return A


def _score(p: np.ndarray, n: int) -> float:
    """Return the evaluation value M for a distribution with sum(p)=1."""
    conv = np.convolve(p, p)
    return 2.0 * n * np.max(conv)

    a = np.empty(n, dtype=np.float64)
