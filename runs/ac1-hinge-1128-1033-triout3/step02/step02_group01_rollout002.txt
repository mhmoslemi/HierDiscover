"""
Propose a step‑function with a lower evaluation score.

Algorithm
---------
1. **Parametric sweep** –   Exhaustively test several simple families of
   symmetric, non‑negative vectors (power‑law “U‑shapes”, product‑distance,
   exponential decay, linear decay) for a range of lengths and exponents.
   The best candidate (lowest 2·n·max(conv) score) is kept as a seed.

2. **Smooth Adam optimisation** –   Starting from the seed, minimise the
   (smoothed) maximal self‑convolution using Adam on the *log‑softmax*
   parametrisation `a = softmax(x)`.  The smooth surrogate of the maximum
   is a soft‑max with an increasing temperature parameter `γ`.  The gradient
   of the surrogate with respect to the probabilities `a` is obtained
   efficiently by a cross‑correlation (via FFT).  After each Adam step the
   vector stays on the probability simplex automatically, so no projection is
   required.

3. **Stagnation handling** –   If no improvement is seen for many
   iterations a small random perturbation is added to the logits to escape
   flat regions.

All time checks respect the global budget (default 1000 s).  The routine
returns the best raw list of floats found; the external evaluator will
re‑normalise it anyway.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    idx = np.where(rho_arr > 0)[0]
    if idx.size == 0:
        theta = 0.0
    else:
        rho = idx[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return float(full_convolution_fft(a).max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector with sum==1."""
    n = len(a)
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Families of candidate vectors
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n-1-i)+1)^(-p)  (U‑shape based on distance to edges)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n-i))^(-p)  (U‑shape based on product of distances)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_min_distribution(n: int, p: float) -> np.ndarray:
    """Exponential decay from the centre: a[i] ∝ exp(-p*min(i,n-1-i))."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = np.exp(-p * d) + 1e-14
    a /= a.sum()
    return a


def linear_min_distribution(n: int, p: float) -> np.ndarray:
    """Linear decay to the centre: a[i] ∝ max(1 - p*min(i,n-1-i), eps)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = np.maximum(1.0 - p * d, 1e-14)
    a /= a.sum()
    return a


def mixture_with_uniform(base: np.ndarray, alpha: float) -> np.ndarray:
    """Convex combination (alpha)·base + (1‑alpha)·uniform, then normalise."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n)
    a = alpha * base + (1.0 - alpha) * uniform
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth optimisation (Adam on logits)
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    init_a: np.ndarray,
    deadline: float,
    *,
    lr: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    gamma_start: float = 1.0,
    gamma_max: float = 50.0,
    gamma_mul: float = 1.0005,
    stagnation_iters: int = 20000,
) -> np.ndarray:
    """
    Adam optimisation of the soft‑max surrogate of max(conv).
    Returns the best vector seen (sum == 1, non‑negative).
    """
    n = init_a.shape[0]

    # logits (unconstrained) – initialise from the current distribution
    x = np.log(np.maximum(init_a, 1e-14))

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = init_a.copy()
    best_score = score_from_vector(best_a)

    t = 0
    last_improve = 0
    gamma = gamma_start

    # pre‑compute FFT size for the correlation step
    size_corr = 1 << ((3 * n - 2).bit_length())

    while time.time() < deadline:
        t += 1

        # ----- current probabilities -----
        x_shift = x - np.max(x)           # for numerical stability
        a = np.exp(x_shift)
        a /= a.sum()

        # ----- exact score (for bookkeeping) -----
        conv = full_convolution_fft(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score + 1e-12 < best_score:          # strict improvement
            best_score = cur_score
            best_a = a.copy()
            last_improve = t
            print(f"Iter {t:7d}: new best score = {best_score:.6f}")

        # ----- soft‑max weights for the surrogate -----
        scaled = gamma * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()   # length 2n‑1, sum to 1

        # ----- gradient wrt a (smooth max) via correlation -----
        fa = np.fft.rfft(a, n=size_corr)
        fw = np.fft.rfft(w, n=size_corr)
        corr = np.fft.irfft(fa * np.conj(fw), n=size_corr)
        grad_a = 2.0 * corr[:n]                         # linear‑correlation result

        # ----- chain rule: gradient wrt logits x -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- increase temperature (sharpen the surrogate) -----
        gamma = min(gamma_max, gamma * gamma_mul)

        # ----- stagnation handling -----
        if t - last_improve > stagnation_iters:
            # inject a tiny random kick into the logits
            x += np.random.randn(n) * 0.01
            last_improve = t
            # give the optimiser a fresh temperature
            gamma = gamma_start

    return best_a

        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Smooth Adam refinement (log‑softmax + soft‑max surrogate)
    # --------------------------------------------------------------
    print("\n=== Smooth Adam refinement ===")
    refined = smooth_adam_optimize(best_seq, deadline)

    # final check – keep the best of the two
    refined_score = score_from_vector(refined)
    if refined_score < best_score:
        best_score = refined_score
        best_seq = refined.copy()
        print(f"Refinement improved score to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return result
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
