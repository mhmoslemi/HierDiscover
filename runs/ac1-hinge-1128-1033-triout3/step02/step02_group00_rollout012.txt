def full_convolution(a, b, size=None):
    n = a.shape[0]; m = b.shape[0]
    out_len = n + m - 1
    if size is None:

    fa = np.fft.rfft(a, size)

    conv = np.fft.irfft(fa*fb, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

def _full_convolution(x):
    """Full convolution x*x using FFT."""
    n = x.shape[0]
    size = 1 << ((2*n-1) - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:2*n-1]
    np.maximum(conv, 0, out=conv)
    return conv

def _conv_fft(x, y):
    """Full convolution x*y using FFT."""
    n = x.shape[0]; m = y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

    2·n·max(conv(a,a))

smooth log‑sum‑exp approximation

    Lε(a) = 2·n·ε·log(∑_k exp(conv_k/ε))

which converges to the original objective when ε → 0.  For a fixed ε the
gradient is explicit:

    ∂L/∂a_i = 4·n·∑_k p_k·a_{k-i},
    where p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε).

The inner sum is a sliding dot‑product that can be computed by a single FFT
convolution (a * reverse(p)).  We perform Adam updates on the
unconstrained variables ``w`` and gradually decrease ε (from 5·10⁻³ to
10⁻⁷).  The soft‑max guarantees feasibility at every step, and we keep the

formula).  Random perturbations and occasional restarts avoid stagnation.

The whole optimisation runs until the time budget (≈1000 s) is exhausted.

import time

# ----------------------------------------------------------------------
# FFT‑based full convolution (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full convolution x*x (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)            # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv
