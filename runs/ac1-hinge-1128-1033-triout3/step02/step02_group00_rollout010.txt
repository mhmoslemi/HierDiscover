"""
Exponentiated‑gradient optimisation of the self‑convolution L∞ norm.

The evaluation of a sequence a (with non‑negative entries) after normalising
to a probability distribution is

    R(a) = 2·n·max_k (a*a)[k]      (n = len(a))

which equals the ratio of the maximal value of the discrete
self‑convolution to its average (the theoretical lower bound is 1).
We minimise a smooth surrogate of the max – a log‑sum‑exp with temperature
β – and perform exponentiated‑gradient steps on the simplex.  The temperature
β and the step size η are increased / decreased piece‑wise linearly, and a
few random “mass‑swap’’ perturbations are injected to avoid stagnation.
The routine works for any length; it tries the length of a previously
found construction (if the global variable ``height_sequence_1`` exists)
and also a larger length, keeping the best sequence discovered before the
time budget expires.

# ----------------------------------------------------------------------
# Fast linear convolution via real FFT (zero‑padded to the next power of two)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the full (linear) convolution of two real vectors using FFT.
    Result length = len(x) + len(y) – 1.
    Small negative rounding errors are clipped to zero.
    """
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: n + m - 1]
    # Clip tiny negatives that stem from floating‑point noise
    np.maximum(conv, 0, out=conv)
    return conv

        lengths.append(2000)
        lengths.append(4000)

    # Preserve order while removing duplicates
    seen = set()
    lengths = [L for L in lengths if not (L in seen or seen.add(L))]

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- initialise a probability vector a on the simplex  -----
        if init is not None and len(init) == n:
            a = init.copy()
        else:
            if init is not None:
                # linear interpolation of the old vector to the new size
                x_old = np.linspace(0.0, 1.0, len(init))
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, init)
                a = np.maximum(a, 0.0)
            else:
                a = np.random.rand(n)
            a = np.maximum(a, 1e-12)      # keep strictly positive
        a = a / a.sum()                    # normalise to sum = 1

        # ----- schedule of temperature β and step size η  -----
        # (beta, eta_start, eta_end, iterations)
        stages = [
            (0.5, 0.5, 0.05, 4000),
            (2.0, 0.3, 0.02, 4000),
            (5.0, 0.2, 0.01, 4000),
            (15.0, 0.1, 0.005, 4000),
            (30.0, 0.05, 0.001, 4000),
        ]

        iteration = 0

        # ----- core optimisation (exponentiated‑gradient) -----
        for beta, eta_start, eta_end, max_iter in stages:
            for it in range(max_iter):
                if time.time() >= deadline:
                    break

                # self‑convolution and current objective value
                conv = _full_convolution_fft(a, a)            # length 2n‑1
                max_conv = conv.max()
                cur_val = 2.0 * n * max_conv
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[n={n:5d} it={iteration:6d}] new best = {best_val:.6f}")

                # smooth approximation of the max via a soft‑max weight vector
                # w_i ∝ exp(beta * (conv_i - max_conv))
                exp_arg = beta * (conv - max_conv)
                w = np.exp(exp_arg)
                w_sum = w.sum()
                if w_sum == 0.0 or not np.isfinite(w_sum):
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the smooth max (size n)
                a_rev = a[::-1]
                grad_full = _full_convolution_fft(w, a_rev)   # length 3n‑2
                grad = 2.0 * grad_full[n - 1 : n - 1 + n]      # slice of length n

                # linear cooling of the step size within the stage
                eta = eta_start * (1.0 - it / max_iter) + eta_end * (it / max_iter)

                # exponentiated‑gradient update on the simplex
                mult = np.exp(-eta * grad)
                a = a * mult + 1e-16          # avoid exact zeros
                a = a / a.sum()

                # occasional random “mass‑swap’’ to escape plateaus
                if np.random.rand() < 0.001:
                    i, j = np.random.choice(n, 2, replace=False)
                    delta = 0.05 * min(a[i], a[j])
                    a[i] -= delta
                    a[j] += delta
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()

                iteration += 1
            if time.time() >= deadline:
                break

        # ----- final local search (tiny random moves) -----
        # only performed if time permits
        while time.time() < deadline - 0.2:
            i, j = np.random.choice(n, 2, replace=False)
            delta = 0.02 * min(a[i], a[j])
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            conv_cand = _full_convolution_fft(cand, cand)
            val_cand = 2.0 * n * conv_cand.max()
            if val_cand < best_val:
                best_val = val_cand
                best_seq = cand.copy()
                a = cand
                print(f"[n={n:5d} local] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but ensures a valid return value
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # Return a plain Python list as required by the competition framework
    return best_seq.tolist()
