Now define projection onto simplex:

def proj_simplex(v):
    # Implements Euclidean projection onto simplex {x >= 0, sum x = 1}
    # Reference: Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function that (approximately) minimises
``evaluate_sequence``.  

The algorithm works in three stages:

1. **Initial shape** –  a discrete arcsine distribution (high mass at the
   edges, low mass in the centre) is known to give a nearly flat
   self‑convolution.  The vector is normalised so that its sum equals 1.

2. **Smooth‑gradient descent** –  The objective
   ``max(conv)`` is approximated by a soft‑max
   ``F_β(a)= (1/β)·log Σ_k exp(β·(a*a)_k)``.
   Its gradient can be expressed using a single FFT‑based convolution.
   An exponentiated‑gradient (mirror‑descent on the simplex) step
   ``a ← a·exp(-η·∇F_β(a))`` keeps the vector non‑negative; a final
   normalisation projects it onto the probability simplex.
   The temperature‐like parameter β is increased linearly from a small
   value (β≈5) to a large value (β≈300) so that the algorithm first
   makes large, smooth moves and then focuses on the true max.

3. **Multi‑scale refinement** –  the optimisation is performed first on a
   modest size (n≈8000) to obtain a good shape quickly, then the vector
   is interpolated to a larger size (n≈20000) where the theoretical
   lower bound 2·n/(2n−1)≈1.0 is almost reachable.  The same gradient
   loop is continued until the time budget (default 1000 s) expires or a
   target score (< 1.503) is reached.

The routine never calls ``evaluate_sequence`` inside the loop (the FFT
based max‑convolution is exact up to round‑off and is far faster).  A
single final call returns the precise score for reporting.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_weights(beta: float, conv: np.ndarray) -> np.ndarray:
    """Stable soft‑max on β·conv."""
    maxv = conv.max()
    exps = np.exp(beta * (conv - maxv))       # max exponent = 0
    return exps / exps.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law on {0,…,n‑1}: a[i] ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1} (Duchi et al. 2008)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising `evaluate_sequence`.

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 0.  Starting shape – arcsine distribution, sum normalised to 1.
    # ------------------------------------------------------------------
    # choose a modest size for the first phase; later we upscale.
    n_small = 8000
    a = _arcsine_distribution(n_small)

    # FFT size that can hold a*a (2n‑1) and a*w_rev (3n‑2) in one go.
    size_fft = _next_pow2(3 * n_small - 2)

    # Hyper‑parameters (tuned empirically)
    beta_start = 5.0
    beta_end   = 300.0
    beta_ramp_iters = 2500          # linear increase over these iterations
    eta0 = 0.30                     # initial step size for exponentiated update
    eta_decay = 2e-4                # gradual decay of the step size
    max_iters_small = 6000          # hard cap for the first phase
    best_seq = a.copy()
    best_score = np.inf

    it = 0
    print("\n=== Phase 1: optimisation on n =", n_small, "===\n")
    while time.time() < deadline and it < max_iters_small:
        # ----- convolution and current score ---------------------------------
        A = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(A * A, n=size_fft)[: 2 * n_small - 1]   # a*a
        cur_score = 2.0 * n_small * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {it:5d}: new best score ≈ {best_score:.6f}")

        # ----- schedule for β ------------------------------------------------
        if it < beta_ramp_iters:
            beta = beta_start + (beta_end - beta_start) * it / beta_ramp_iters
        else:
            beta = beta_end

        # ----- soft‑max weights ------------------------------------------------
        w = _softmax_weights(beta, conv)               # length 2n‑1

        # ----- gradient of the smoothed max (via one FFT) --------------------

        W_rev = np.fft.rfft(w[::-1], n=size_fft)        # FFT of reversed w

        # ----- convolution & current score ------------------------------------
        A = np.fft.rfft(a, n=size_fft)

        cur_score = 2.0 * n_large * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {it:5d}: new best score ≈ {best_score:.6f}")

        # ----- β schedule (same linear ramp, but with more iterations) -------
        if it < beta_ramp_iters:

        else:
            beta = beta_end

        # ----- soft‑max weights ------------------------------------------------
        w = _softmax_weights(beta, conv)

        W_rev = np.fft.rfft(w[::-1], n=size_fft)
