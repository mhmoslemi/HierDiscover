"""
Gradient‑descent search (softened max) for a low‑score step function.

The algorithm works with a *log‑softmax* parametrisation a_i = exp(x_i) / Σ exp(x)
so that the simplex constraints are satisfied automatically.
We minimise a smooth approximation of the objective

    f(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

which converges to the true maximum as β grows.  The gradient of f w.r.t. a is

    ∂f/∂a_i = a_i + Σ_j w_{i+j}·a_j                (1)

where w is the soft‑max of the convolution entries.
Using the chain rule we obtain the gradient on the x‑variables,
project it onto the simplex tangent space and apply the Adam optimiser.
β is gradually increased so that the optimisation first sees a smooth
landscape and later focuses on the true peak.  The best vector seen
throughout the run (evaluated with the exact “evaluate_sequence”
function) is returned.

The routine respects the supplied time budget and works both with the
previously discovered construction (available as `height_sequence_1`)
and with a fresh uniform start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    n = len(a)
    m = len(b)
    size = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: n + m - 1]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return conv_full_fft(a, a)


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (as a Python list) found before the time budget expires.
    """
    np.random.seed(seed)

    # small safety margin for the runner
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # 0. initialise a – try to use the previously discovered vector
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)
        base = np.maximum(base, 0.0)
        if base.size == 0 or base.sum() == 0.0:
            raise ValueError
        base /= base.sum()
        n = base.size
        a0 = base
        print(f"Starting from provided height_sequence_1 (n={n})")
    except Exception:
        # fallback: modestly sized uniform start
        n = 8000
        a0 = np.full(n, 1.0 / n, dtype=float)
        print(f"Starting from uniform distribution (n={n})")

    # Log‑softmax parametrisation (x can be any real vector)
    eps_log = 1e-14
    x = np.log(a0 + eps_log)

    # Adam optimiser state
    lr = 0.12                       # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-9
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # Temperature for the soft‑max over convolution entries
    beta = 1.0
    beta_max = 30.0
    beta_increase_every = 5000     # iterations
    beta_factor = 1.5

    best_score = np.inf
    best_a = None
    iteration = 0
    last_improve = 0

    print("\n=== Begin optimisation (soft‑max gradient descent) ===")
    while time.time() < deadline:
        iteration += 1

        # ---- a = softmax(x) ------------------------------------------------
        a = np.exp(x)
        a_sum = a.sum()
        a /= a_sum                     # now Σ a_i = 1, a_i ≥ 0

        # ---- exact score (for bookkeeping) ---------------------------------
        conv_aa = conv_self_fft(a)     # length 2n‑1
        max_c = conv_aa.max()
        cur_score = 2.0 * n * max_c    # exact value of evaluate_sequence for sum=1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---- smooth max: weights w_k ---------------------------------------
        # stability trick: shift by max before exponentiation
        shift = conv_aa.max()
        shifted = conv_aa - shift
        exp_beta = np.exp(beta * shifted)
        w = exp_beta / exp_beta.sum()               # shape (2n‑1,)

        # ---- compute f_i = Σ_j w_{i+j} a_j efficiently --------------------
        a_rev = a[::-1]                              # reversed a, length n
        conv_w_arev = conv_full_fft(w, a_rev)       # length 3n‑2
        # slice indices n‑1 … 2n‑2  (Python end‑exclusive)
        f = conv_w_arev[n - 1 : 2 * n - 1]           # shape (n,)

        # ---- gradient w.r.t a (see derivation (1)) -------------------------
        grad_a = a + f                               # shape (n,)

        # ---- map gradient to x‑space (tangent to simplex) -----------------
        avg = np.dot(a, grad_a)                      # scalar = Σ a_i ∂f/∂a_i
        grad_x = a * (grad_a - avg)                  # also sums to zero

        # ---- Adam update ---------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        # bias‑correction (powers can become tiny, safe with **)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- schedule for β ------------------------------------------------
        if iteration % beta_increase_every == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_factor)
            print(f"Iter {iteration:7d}: increased β to {beta:.2f}")

        # ---- occasional random kick to escape plateaus --------------------
        if iteration - last_improve > 25000:
            # add a tiny random perturbation to x
            noise = np.random.randn(n) * 0.02
            x += noise
            last_improve = iteration
            print(f"Iter {iteration:7d}: injected random noise to escape stagnation")

    # ----------------------------------------------------------------------
    #  Final reporting and return
    # ----------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score discovered: {best_score:.6f} (length {len(best_a)})")
    # the external evaluator will renormalise again, so we can return the raw vector
    return best_a.tolist()
