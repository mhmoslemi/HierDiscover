"""
Improved search for a step‑function that minimises
`evaluate_sequence`.

The algorithm has three stages:

1. **Analytic starter** – the normalised coefficients of
   (1‑x)^(-1/2) (the “square‑root” construction) give a sequence
   with evaluation ≈ π/2.  It is used as a deterministic warm‑start.

2. **Smooth‑max gradient descent** – we minimise the
   convex objective  f(a)=max_k (a*a)_k.
   The non‑smooth max is replaced by a smooth log‑sum‑exp
   approximation with a temperature τ that grows during the run.
   The exact gradient of the smooth surrogate is obtained by a
   second FFT‑based convolution, followed by an exponentiated‑gradient
   (mirror‑descent) update on the log‑variables and a projection onto
   the simplex (∑a_i=1, a_i≥0).  This drives the evaluation well
   below the naïve bound.

3. **Fine‑grained simulated‑annealing** – after the gradient phase
   a Metropolis walk that transfers a random amount of mass between
   two coordinates explores the remaining “wiggle‑room’’ of the
   discrete problem.  The walk is temperature‑controlled and restarts
   if progress stalls.

Both phases use the fast FFT based score   2·n·max(a∗a)   (identical to
`evaluate_sequence` for a normalised sequence) and only call the
expensive `evaluate_sequence` when a new global best is discovered.
The routine respects the 1000 s time limit, works on a single CPU and
requires only NumPy (SciPy is optional for a tiny LP‑based warm‑start,
which is used when available).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming warm‑start (requires SciPy)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog  # type: ignore
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex using the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “square‑root” start (coefficients of (1‑x)^(-1/2))
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n normalised coefficients of (1‑x)^(-1/2)."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()                     # normalise to sum = 1
    return w


# ----------------------------------------------------------------------
# Fast scalar score: 2·n·max(a∗a)  (identical to evaluate_sequence for sum=1)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(a∗a) using an FFT of length `fft_len`."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray,
                         conv: np.ndarray,
                         tau: float,
                         fft_len2: int) -> np.ndarray:
    """
    Compute ∇_{a}  ( 1/τ log Σ_k exp(τ·conv_k) )
    via a second convolution.
    """
    # weights proportional to exp(τ·(conv - max(conv)))  – numerically stable
    max_c = conv.max()
    w = np.exp(tau * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:               # can happen only for very large τ
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # gradient of each conv_k is 2 * a_{k-i}; the weighted sum is 2 * (w * a_rev)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len2)
    fa_rev = np.fft.rfft(a_rev, fft_len2)
    g_full = np.fft.irfft(fw * fa_rev, fft_len2)[: a.shape[0]]
    grad = 2.0 * g_full
    return grad


# ----------------------------------------------------------------------
# Optional LP warm‑start (maximise Σb_j  s.t. a*b ≤ M)
# ----------------------------------------------------------------------
def _lp_warm_start(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ b_j  subject to  b ≥ 0,  a*b ≤ M  (linear constraints).
    Returns the LP solution (already normalised to sum = 1).
    """
    if not _SCIPY_AVAILABLE:
        return a.copy()                     # fallback – no LP available

    n = a.shape[0]
    # Build the convolution matrix A where (A b)_k = Σ_i a_i b_{k-i}
    # This matrix is Toeplitz with each anti‑diagonal equal to a_i.
    rows = 2 * n - 1
    data = []
    row_ind = []
    col_ind = []

    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            data.append(a[i])
            row_ind.append(k)
            col_ind.append(j)

    # Build sparse matrix
    from scipy.sparse import coo_matrix
    A = coo_matrix((data, (row_ind, col_ind)), shape=(rows, n))

    # Upper‑bound vector
    ub = np.full(rows, M)

    # Linear program:
    # maximise 1ᵀ b   ⇔   minimise –1ᵀ b
    c = -np.ones(n)

    bounds = [(0, None) for _ in range(n)]

    # linprog works with dense matrices for the "highs" methods,
    # but we can pass the sparse A_ub directly.
    res = linprog(c,
                  A_ub=A,
                  b_ub=ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})

    if not res.success:
        return a.copy()
    b = res.x
    if b.sum() == 0.0:
        return a.copy()
    b /= b.sum()                     # normalise to the simplex
    return b

# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that minimises the
    supplied `evaluate_sequence` function.

    The routine runs a smooth‑max gradient descent followed by a
    short simulated‑annealing fine‑tuning.  Several problem sizes are
    tried; the best sequence seen over the whole budget is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a set of lengths – we allocate roughly equal time to each.
    # ------------------------------------------------------------------
    if "length" in kwargs:
        lengths = [int(kwargs["length"])]
    else:
        # a mix of powers‑of‑two and a few arbitrary sizes
        lengths = [1024, 2048, 4096, 8000, 12000, 16384]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper that evaluates with the external (slow) routine only when
    # a new global best is discovered.
    # ------------------------------------------------------------------
    def _maybe_update_best(candidate: np.ndarray, fast_val: float) -> None:
        nonlocal best_seq, best_score
        if fast_val < best_score - 1e-12:
            tru = evaluate_sequence(candidate.tolist())
            if tru < best_score:
                best_score = tru
                best_seq = candidate.copy()
                print(f"[global] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main loop over the selected lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (roughly equal split)
        remaining = deadline - time.time()
        alloc = remaining / (len(lengths) - lengths.index(n))
        phase_end = time.time() + alloc

        # --------------------------------------------------------------
        # Initial vector
        # --------------------------------------------------------------
        try:
            a0 = np.array(height_sequence_1, dtype=float)
            if a0.shape[0] != n:
                a0 = _sqrt_series(n)
        except Exception:            # pragma: no cover
            a0 = _sqrt_series(n)

        a = a0.copy()
        a = _project_simplex(a)                # guarantee sum=1, a_i≥0

        # FFT lengths (next power of two, reused later)
        size_fft1 = 1 << ((2 * n - 1).bit_length())
        size_fft2 = 1 << ((3 * n - 2).bit_length())

        # fast score of the start
        cur_fast = _fast_score(a, size_fft1)
        _maybe_update_best(a, cur_fast)
        print(f"[init] n={n}, start fast score {cur_fast:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – smooth‑max gradient descent
        # --------------------------------------------------------------
        max_iter = 25000
        eta0 = 0.9               # initial step size (in log‑space)
        tau_inc = 1.0 / max_iter   # increase τ each iteration
        tau = 0.5                # start temperature for the smooth max

        it = 0
        last_improve = 0
        while it < max_iter and time.time() < phase_end:
            # current convolution
            fa = np.fft.rfft(a, size_fft1)
            conv = np.fft.irfft(fa * fa, size_fft1)[: 2 * n - 1]

            # fast evaluation
            fast_val = 2.0 * n * conv.max()
            # keep the best seen in this phase
            if fast_val < cur_fast - 1e-12:
                cur_fast = fast_val
                _maybe_update_best(a, cur_fast)
                last_improve = it

            # smooth‑max temperature schedule
            tau = min(10.0, tau + tau_inc)    # slowly grow to ~10

            # gradient of smooth surrogate
            grad = _smooth_max_gradient(a, conv, tau, size_fft2)

            # exponentiated‑gradient step (mirror descent on the simplex)
            eta = eta0 / (1.0 + it * 5e-5)    # slight decay
            log_a = np.log(a + 1e-30)
            a_new = np.exp(log_a - eta * grad)
            a_new = _project_simplex(a_new)

            a = a_new

            # occasional random jitter to avoid stagnation
            if it % 1000 == 0 and it > 0:
                jitter = np.random.randn(n) * 1e-5 / math.sqrt(it + 1)
                a = _project_simplex(a + jitter)

            # restart if no improvement for a long stretch
            if it - last_improve > 8000:
                a = _sqrt_series(n)
                a = _project_simplex(a + np.random.randn(n) * 1e-3)
                last_improve = it
                print(f"[restart] n={n} after {it} gradient iters")

            it += 1

        # --------------------------------------------------------------
        # Phase 2 – simulated‑annealing fine‑tuning
        # --------------------------------------------------------------
        T_start = 0.12
        iter_sa = 0
        no_improve = 0
        fast_cur = cur_fast

        while time.time() < phase_end:
            # linear cooling schedule
            elapsed = time.time() - (phase_end - alloc)
            frac = min(1.0, elapsed / alloc)
            T = T_start * (1.0 - frac)

            # pick two distinct indices
            src = np.random.randint(n)
            if a[src] < 1e-14:
                continue
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

            # amount to move – scaled by temperature
            delta = np.random.rand() * a[src] * min(0.5, T)

            a_cand = a.copy()
            a_cand[src] -= delta
            a_cand[dst] += delta

            # fast evaluation
            fast_cand = _fast_score(a_cand, size_fft1)

            # Metropolis acceptance
            if fast_cand <= fast_cur or np.random.rand() < math.exp(-(fast_cand - fast_cur) / max(T, 1e-12)):

                fast_cur = fast_cand
                if fast_cand < cur_fast - 1e-12:

                else:

            else:
                no_improve += 1

                no_improve = 0

    # ------------------------------------------------------------------
    # Return the best sequence discovered (plain Python list)
    # ------------------------------------------------------------------

    print(f"[final] overall best evaluation = {best_score:.6f}")
    return best_seq.tolist()
