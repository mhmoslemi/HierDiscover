def compute_subgradient(a, conv):
    max_val = conv.max()
    eps = 1e-12 * max_val + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    n = len(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises the evaluation score

    2 * n * max_k (a * a)[k]

where * denotes the full linear convolution.  The algorithm works in two
phases:

1.  Deterministic sweep over several parametric “U‑shaped’’ families
    (inverse‑distance, inverse‑product‑distance, exponential decay) for
    many lengths and exponent/scale parameters.  The best vector from
    this sweep is taken as a seed.

2.  Stochastic refinement on the simplex using a projected sub‑gradient
    descent driven by the (sub‑)gradient of the maximal convolution
    entry, combined with simulated‑annealing acceptance, occasional
    random mass‑transfer moves and a simple re‑heat strategy to escape
    stagnation.

All heavy work is performed inside ``propose_candidate`` so that the
function respects the given time budget (default 1000 s).  The returned
list is a raw vector (sum≈1) – the external evaluator will re‑scale it
as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level as required)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex:
        { w >= 0 , sum(w) = 1 }.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for a vector that
    already satisfies sum(a) == 1 (the denominator becomes 1).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices where the convolution attains (or almost attains) its
    maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        # i ranges over indices where both i and k-i are valid
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each pair contributes 2*a[j]
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on distance to the nearest edge:
        a[i] = (min(i, n‑1‑i) + 1)^{-p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the product of distances to both edges:
        a[i] = ((i+1)*(n‑i))^{-p}
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """
    Symmetric exponential decay from the edges:
        a[i] = exp(-lam*i) + exp(-lam*(n‑1‑i))
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# Main search routine
# ----------------------------------------------------------------------
