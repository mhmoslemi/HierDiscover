        z = np.log(a)

        # Adam hyperparameters
        alpha = 0.1  # learning rate for logits
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        m = np.zeros_like(z)
        v = np.zeros_like(z)

        # smoothing gamma schedule
        gamma_start = 0.5
        gamma_end = 80.0

        max_iters = 200000  # generous but limited by time budget
        it = 0
        cur_fast = _fast_score(a, max(1 << ((2 * n - 1).bit_length()), size_fft))  # compute using shorter size maybe.

        # define inner maybe later.

        while time.time() < deadline and it < max_iters:

            A = np.fft.rfft(a, size_fft)

            # compute convolution a*a using same FFT size
            conv_full = np.fft.irfft(A * A, size_fft)[:2*n-1]

            w = np.exp(gamma * (b - b_max))   # max weight is 1

            # compute gradient wrt logits
            dot_a_grad = np.dot(a, grad_a)
            grad_z = a * (grad_a - dot_a_grad)

            # Adam updates
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * (grad_z ** 2)
            bias_correction1 = 1 - beta1 ** (it + 1)

            step = alpha * (np.sqrt(bias_correction2) / bias_correction1)

            # compute new a via softmax
            z_max = np.max(z)
            a = np.exp(z - z_max)
            a_sum = a.sum()
            a = a / a_sum

        # break loops

    intervals: a_i = 2·(√(i+0.5)‑√(i‑0.5)), with a_0 = 2·√0.5.

    right = np.sqrt(i + 0.5)
    a = 2.0 * (right - left)    # integral of t^{-1/2} over the midpoint cell

    return a

        fast_new = 2.0 * n * conv_full.max()

        # ----- record improvement ------------------------------------------------
        if fast_new < fast_val - 1e-12:
            fast_val = fast_new
            true_new = evaluate_sequence(a.tolist())
            if true_new < best_score:
                best_score = true_new
                best_seq = a.copy()
                print(f"[global] iter={it}, new best {best_score:.9f}")
            last_improve = it

        # ----- restart if stuck for a long time ---------------------------------
        if it - last_improve > 40_000:
            # re‑initialise logits from the analytic start (small random jitter)
            z = np.log(_midpoint_sqrt_series(n) + eps)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"[restart] iter={it}")

        # ----- occasional jitter to avoid plateaus --------------------------------
        if it % 5_000 == 0 and it > 0:
            z += np.random.randn(n) * 0.02 / np.sqrt(it + 1)

        it += 1

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    print(f"[final] best evaluation = {best_score:.9f}")
    return best_seq.tolist()
