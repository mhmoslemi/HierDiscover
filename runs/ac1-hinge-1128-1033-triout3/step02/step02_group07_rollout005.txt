"""
Propose a step‑function (non‑negative vector) that minimizes the
`evaluate_sequence` metric.

The routine follows three ideas:

1. **Arcsine‑type initialization** – a discretisation of the
   probability density  f(x)=1/(π√(x(1−x)))  (the “arcsine law”).  This
   distribution is known to give a very flat autocorrelation in the
   continuous setting.

2. **Exponentiated‑gradient descent on the simplex** – we minimise the
   smooth approximation of the maximal convolution value
   (soft‑max of the autocorrelation) using a mirror‑descent step
   (multiplicative update) and an exact projection onto the probability
   simplex.  The gradient is the usual sub‑gradient of the (soft‑max)
   of the convolution.

3. **Aggressive simulated‑annealing fine‑tuning** – after the gradient
   phase we perform a long Metropolis walk that transfers mass between
   random coordinates.  The temperature schedule is long‑ranged,
   restarts are triggered after stagnation, and every accepted improvement
   is checked with the exact `evaluate_sequence` function.

All heavy work is done with FFT‑based convolutions (O(n log n)).  The
expensive call to `evaluate_sequence` is performed only when a new
global best is found.  The whole optimisation is bounded by the supplied
time budget (≈1000 s) and runs on a single CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex using the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Arcsine‑type start (discretised 1/(π√(x(1−x)))).
# ----------------------------------------------------------------------
def _arcsine_series(n: int) -> np.ndarray:
    """Return a normalised arcsine‑law vector of length n."""
    i = np.arange(n, dtype=np.float64)
    # shift by 0.5 to avoid the singularities at the boundaries
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# FFT‑based evaluation: 2·n·max(a∗a) (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, size_fft: int) -> float:
    """Return 2·n·max(a∗a) using an FFT of length `size_fft`."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the (soft‑max) maximal convolution entry.
# For a given temperature β we use a soft‑max weighting to obtain a
# smoother direction than the raw max.
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ exp(β·conv_k) w.r.t. a, assuming Σ a = 1.
    It is 2 * (a  ⋆  p) where p_k = exp(β·conv_k) / Σ exp(β·conv)
    and '⋆' denotes cross‑correlation.
    """
    # soft‑max weights (avoid overflow)
    max_c = conv.max()
    exp_beta = np.exp(beta * (conv - max_c))
    p = exp_beta / exp_beta.sum()               # shape (2n‑1,)

    n = a.shape[0]
    # compute cross‑correlation a ⋆ p efficiently via FFT
    # we need a length at least n + (2n‑1) - 1 = 3n‑2 to avoid circular aliasing
    size = 1 << ((3 * n - 2).bit_length())
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    # cross‑correlation = ifft( A * conj(P) )
    corr = np.fft.irfft(A * np.conj(P), size)[:n]
    return 2.0 * corr


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      length: int = 8000,
                      **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The function respects the time budget, uses the
    arcsine start, performs exponentiated‑gradient descent followed by a
    long simulated‑annealing phase and returns the best sequence encountered.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = int(length)
    # initial vector – try the provided global if it matches the size,
    # otherwise fall back to the arcsine construction.

    except Exception:
        a0 = _arcsine_series(n)

    a = _project_simplex(a0)               # guarantee sum == 1, non‑negative

    # FFT size for the plain convolution (a∗a)
    size_fft = 1 << ((2 * n - 1).bit_length())
    cur_fast = _fast_score(a, size_fft)

    best_seq = a.copy()
    best_score = float('inf')              # will be filled after first check

    # ------------------------------------------------------------------
    # Helper that evaluates with the external (slow) routine only when
    # a new global best is discovered.
    # ------------------------------------------------------------------
    def _maybe_update_best(candidate: np.ndarray, fast_val: float) -> None:
        nonlocal best_seq, best_score
        if fast_val < best_score - 1e-12:
            true_val = evaluate_sequence(candidate.tolist())
            if true_val < best_score:
                best_score = true_val
                best_seq = candidate.copy()
                print(f"[global] new best {best_score:.6f}")

    _maybe_update_best(a, cur_fast)
    print(f"[init] n={n}, start fast score {cur_fast:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent on a smooth surrogate.
    # ------------------------------------------------------------------
    beta = 30.0                     # soft‑max sharpness (larger → closer to max)
    eta0 = 1.0                      # initial step size for exp‑gradient
    decay = 1e-5                    # slower decay to keep decent steps
    max_it = 40000                  # many iterations – limited by time

    it = 0
    while it < max_it and time.time() < deadline:
        # compute convolution a∗a

            a = _project_simplex(a + jitter)
            cur_fast = _fast_score(a, size_fft)

        it += 1
        if time.time() >= deadline:

    print(f"[phase‑1] finished after {it} iters, fast score {cur_fast:.6f}")

    # Phase 2 – Simulated‑annealing fine‑tuning.
    # ------------------------------------------------------------------
    # Temperature schedule: start modest, linearly cool to almost zero.
    T_start = 0.35
    total_sa_time = deadline - time.time()
    sa_start = time.time()
    no_improve = 0
    fast_cur = cur_fast

    while time.time() < deadline:
        elapsed = time.time() - sa_start
        frac = min(1.0, elapsed / total_sa_time)
        T = T_start * (1.0 - frac) + 0.001 * frac   # never reach exact zero

        # pick a source with non‑zero mass
        src = np.random.randint(n)
        if a[src] < 1e-14:
            continue
        # pick a destination distinct from src
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # amount to transfer – up to a fraction of the source mass
        delta_max = a[src] * min(0.9, T)           # allow larger moves early
        delta = np.random.rand() * delta_max

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta

        fast_cand = _fast_score(a_candidate, size_fft)

        # Metropolis criterion
        if fast_cand <= fast_cur or np.random.rand() < math.exp(
                -(fast_cand - fast_cur) / max(T, 1e-12)):
            a = a_candidate
            fast_cur = fast_cand

                cur_fast = fast_cand
                _maybe_update_best(a, cur_fast)
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # restart after long stagnation

            a += np.random.randn(n) * 5e-4   # small perturbation
            a = _project_simplex(a)
            fast_cur = _fast_score(a, size_fft)
            cur_fast = fast_cur
            no_improve = 0
            print(f"[restart] SA stagnated, re‑initialised at time {elapsed:.1f}s")

    print(f"[phase‑2] finished, best local fast score {cur_fast:.6f}")
    print(f"[final] overall best evaluated score = {best_score:.6f}")
