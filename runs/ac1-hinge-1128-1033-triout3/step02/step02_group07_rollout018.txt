"""
Propose candidate sequence that minimises `evaluate_sequence`.

The algorithm consists of two stages.

1. **Linear‑programming “mass‑inflation’’ step** –  
   Starting from the analytic √‑series (coefficients of (1‑x)^‑½) we
   compute the current worst convolution value M = max(a∗a).  With this M
   we solve a linear program

        maximize    Σ g_i
        subject to  (a∗g)_k ≤ M   for all k = 0…2n‑2
                     g_i ≥ 0.

   The LP is tiny (≤ 2 n – 1 constraints, n variables) and can be solved
   with `scipy.optimize.linprog` on a sparse matrix.  The solution
   provides a vector g whose total mass Σg is > 1, therefore after
   normalising g we obtain a new a′ with a strictly smaller
   `max(a′∗a′)`.  Repeating this step a few times yields a large drop
   of the objective.

2. **Fine‑tuning with exponentiated‑gradient descent** –  
   After the LP phase we perform a few hundred exponentiated‑gradient
   updates using the exact sub‑gradient of the current max‑convolution
   (active indices only).  This removes any remaining irregularities
   and often gives a few extra improvement.

Both stages are fast (the LP is dominant but only a handful of solves)
and respect the 1000 s wall‑time.  The best sequence observed is
returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic start: coefficients of (1‑x)^‑½, normalised to sum = 1
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the first n coefficients of (1‑x)^‑½, normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Fast convolution & score (exactly the same as evaluate_sequence for sum=1)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, size_fft: int) -> float:
    """Return 2·n·max(a∗a) (equivalent to evaluate_sequence for sum(a)=1)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return 2.0 * n * conv.max()


def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return the full linear convolution a∗a (length 2n‑1)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Build the (a∗g) ≤ M constraints as a sparse matrix
# ----------------------------------------------------------------------
def _build_constraint_matrix(a: np.ndarray) -> csr_matrix:
    """
    Return a CSR matrix A of shape (2n‑1, n) such that (A @ g)[k] = (a∗g)_k.
    """
    n = a.shape[0]
    # row indices i+j for all i,j
    i = np.arange(n)[:, None]           # shape (n,1)
    j = np.arange(n)                    # shape (n,)
    rows = (i + j).ravel()              # length n*n, values 0 … 2n‑2
    cols = np.tile(np.arange(n), n)     # each column repeated n times
    data = np.repeat(a, n)              # a_i repeated for each j
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


# ----------------------------------------------------------------------
# Linear‑programming “inflation’’ step
# ----------------------------------------------------------------------
def _lp_inflation_step(a: np.ndarray, M: float, max_iter: int,
                       time_deadline: float) -> np.ndarray:
    """
    Perform up to `max_iter` LP inflations.  Returns the improved vector
    (still normalised to sum = 1).  Stops earlier if the time limit is
    reached or if the LP yields no improvement.
    """
    n = a.shape[0]
    size_fft = 1 << ((2 * n - 1).bit_length())
    for _ in range(max_iter):
        # stop if we run out of time
        if time.time() >= time_deadline:
            break

        # ----- Build the LP -----
        A = _build_constraint_matrix(a)
        b = np.full(2 * n - 1, M, dtype=np.float64)      # (a∗g) ≤ M
        c = -np.ones(n, dtype=np.float64)                # maximise Σg ↔ minimise -Σg
        bounds = [(0, None)] * n

        # Solve the LP (highs works with sparse matrices)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – give up this length
            break

        g = np.maximum(res.x, 0.0)       # numerical clean‑up
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:         # no mass increase → stop
            break

        # Normalise g so that its sum becomes 1.
        a = g / sum_g
        # New worst convolution value (exact, because a is already normalised)
        conv = _full_conv(a, size_fft)
        M = conv.max()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (same idea as the reference code)
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k at the active indices `active_idx`.
    For every active k we add 2*a_{k-i} to the i‑th component whenever
    0 ≤ i < n and 0 ≤ k‑i < n.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    if len(active_idx) > 1:
        grad /= len(active_idx)
    return grad


def _expgrad_refine(a: np.ndarray, size_fft: int,
                    max_it: int, time_deadline: float) -> np.ndarray:
    """A few hundred exponentiated‑gradient steps (fast, improves locally)."""
    n = a.shape[0]
    cur_fast = _fast_score(a, size_fft)

    eta0 = 0.6
    decay = 5e-5
    for it in range(max_it):
        if time.time() >= time_deadline:
            break
        # current convolution and active indices
        conv = _full_conv(a, size_fft)
        mx = conv.max()
        active = np.flatnonzero(conv >= mx - 1e-14)

        grad = _max_conv_gradient(a, active)

        eta = eta0 / (1.0 + decay * it)
        log_a = np.log(a + 1e-30)
        a_new = np.exp(log_a - eta * grad)
        a_new = _project_simplex(a_new)

        fast_new = _fast_score(a_new, size_fft)
        if fast_new <= cur_fast:
            a = a_new
            cur_fast = fast_new

        # occasional tiny random jitter
        if it % 400 == 0 and it > 0:
            a = _project_simplex(a + np.random.randn(n) * 5e-5 / math.sqrt(it + 1))

    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that (approximately)
    minimises the supplied `evaluate_sequence` function.
    The routine:
        1. starts from the √‑series,
        2. carries out a few LP inflation steps,
        3. finishes with a short exponentiated‑gradient refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Desired length (user may override via keyword)
    n = int(kwargs.get("length", 1500))

    # ------------------------------------------------------------------
    # Initialise from the analytic construction
    # ------------------------------------------------------------------
    a = _sqrt_series(n)                      # already sums to 1
    size_fft = 1 << ((2 * n - 1).bit_length())

    # Evaluate the initial candidate (slow call only if it improves)
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, start score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Linear‑programming inflation (up to a few iterations)
    # ------------------------------------------------------------------
    conv = _full_conv(a, size_fft)
    M = conv.max()
    a = _lp_inflation_step(a, M, max_iter=8, time_deadline=deadline)
    # one final fast evaluation after the LP phase
    fast_score = _fast_score(a, size_fft)
    true_score = evaluate_sequence(a.tolist())
    if true_score < best_score:
        best_score = true_score
        best_seq = a.copy()
        print(f"[LP] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Exponentiated‑gradient fine‑tuning (few hundred iterations)
    # ------------------------------------------------------------------
    a = _expgrad_refine(a, size_fft, max_it=2000, time_deadline=deadline)
    fast_score = _fast_score(a, size_fft)
    true_score = evaluate_sequence(a.tolist())
    if true_score < best_score:
        best_score = true_score
        best_seq = a.copy()
        print(f"[refine] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # If time remains, try a couple of random restarts (lightweight)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # construct a random simplex point
        rnd = np.random.rand(n)
        rnd = _project_simplex(rnd)

        # LP inflation from the random start (very few steps)
        conv = _full_conv(rnd, size_fft)
        M = conv.max()
        rnd = _lp_inflation_step(rnd, M, max_iter=3, time_deadline=deadline)

        # short gradient polish
        rnd = _expgrad_refine(rnd, size_fft, max_it=500, time_deadline=deadline)

        cur_score = evaluate_sequence(rnd.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = rnd.copy()
            print(f"[restart] new best {best_score:.6f}")

        # break if we cannot afford another restart
        if time.time() >= deadline - 1.0:
            break

    print(f"[final] overall best evaluation = {best_score:.6f}")
    return best_seq.tolist()
