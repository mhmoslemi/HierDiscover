"""
Propose a low‑objective step function.

The objective is
    2 * n * max(conv(p, p))   (p is non‑negative, sum(p)=1)

Because the objective is scale‑invariant we may restrict p to the probability
simplex.  The algorithm works as follows:

* Initialise p from the previously found sequence (if present) and from a
  uniform distribution, keeping the best of them.
* Run many short optimisation “restarts”.  Each restart performs Adam updates
  on a smooth log‑sum‑exp approximation of max(conv).  The smoothing parameter
  α is increased aggressively (up to 1e5) so that the gradient quickly focuses
  on the true maximal entry.
* After every Adam step the iterate is projected onto the simplex
  (Duchi et al. 2008) and then sorted in decreasing order – this encourages a
  monotone shape that tends to reduce the maximal autocorrelation.
* Periodic jitter and a hard restart when no improvement has been seen for a
  while keep the search from stagnating.
* After the optimisation loop a cheap local “pairwise‑swap” refinement is
  carried out on the best solution found.
* The best probability vector (scaled to sum‑to‑1) is returned;  the evaluator
  rescales it automatically, so no further scaling is required.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
