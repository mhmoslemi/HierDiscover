"""
Propose a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. **Warm‑start** – if a global variable `height_sequence_1` exists, use it
   after normalising to a probability vector.

2. **Deterministic sweep** – generate several families of symmetric
   “U‑shaped’’ sequences (power‑law tails, sum‑of‑tails, exponential
   decay) for many lengths and parameters.  The best candidate is kept
   as the starting point for the next phase.

3. **Gradient‑descent refinement** – treat the normalised vector `a`
   (∑a=1) as a point on the simplex and minimise the smooth proxy of the
   maximal self‑convolution

        f(a) = (1/α)·log Σ_k exp(α·(a*a)[k])          (α≫1)

   The gradient of `f` w.r.t. `a` can be written as a convolution,
   which we evaluate by FFT in O(n log n).  We perform mirror‑descent
   updates on `log a` (ensuring positivity) and re‑normalise after each
   step.  A schedule for the smoothness parameter `α` and the learning
   rate is used, and the best vector seen (according to the true
   evaluation metric) is stored.

4. **Optional random restart** – if time permits a short second run is
   started from a uniform vector to escape possible local minima.

5. **Exact evaluation** – the best sequence is finally passed to the
   provided `evaluate_sequence` (once) and returned as a plain Python
   list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def max_conv_fft(a: np.ndarray) -> float:
    """
    Fast computation of max entry of the self‑convolution a*a via FFT.
    The vector a is assumed to be non‑negative; the function works for any
    real vector.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] = (min(i,n-1-i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Two power‑law tails: a[i] = (i+1)^(-p) + (n-i)^(-p)."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weights for the smooth maximum:
        w_k = exp(alpha·(conv_k - max(conv)))   (normalised)
    """
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    s = w.sum()
    if s == 0.0:
        # fall back to a hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        s = 1.0
    return w / s


def _gradient_of_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a f where f is the smooth maximum with weight vector w.
    The formula is:
        grad_i = 2 * Σ_k w_k * a_{k-i}
    which is a convolution of w with the reversed a.
    """
    n = a.shape[0]
    # reversed a
    a_rev = a[::-1]
    size = _next_pow2(3 * n - 2)
    fft_w = np.fft.rfft(w, n=size)
    fft_a_rev = np.fft.rfft(a_rev, n=size)
    conv_full = np.fft.irfft(fft_w * fft_a_rev, n=size)
    # slice (n‑1) … (n‑1+n‑1) gives the needed n entries
    grad = 2.0 * conv_full[(n - 1) : (n - 1 + n)]
    return grad


def gradient_descent_optimize(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    target_score: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on log‑probabilities with a smooth approximation of the
    max‑convolution.  Returns the best sequence seen and its approximate
    score (2·n·max(conv) with ∑a=1).
    """
    a = init_a.copy()
    n = a.shape[0]
    eps = 1e-30

    # initial best
    best_a = a.copy()
    best_score = 2.0 * n * max_conv_fft(a)

    # schedule parameters
    max_alpha = 2000.0
    min_alpha = 50.0
    init_lr = 0.5
    decay = 0.9997

    lr = init_lr
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # 1) full self‑convolution of current a
        size_conv = _next_pow2(2 * n - 1)
        A_fft = np.fft.rfft(a, n=size_conv)
        conv = np.fft.irfft(A_fft * A_fft, n=size_conv)[: 2 * n - 1]

        # 2) smooth weights (soft‑max over conv entries)
        alpha = min_alpha + (max_alpha - min_alpha) * (t / max_iters)
        w = _smooth_weights(conv, alpha)

        # 3) gradient of the smooth objective
        grad = _gradient_of_smooth_max(a, w)

        # subtract mean to keep updates within the simplex (constant shift
        # in log‑space does not affect the normalisation)
        grad -= grad.mean()

        # 4) mirror‑descent on log‑probabilities
        log_a = np.log(a + eps)
        log_a -= lr * grad
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.ones_like(a) / n
        else:
            a /= a_sum

        # decay learning rate
        lr = max(lr * decay, 1e-5)

        # 5) evaluate true objective (using fast approx)
        cur_score = 2.0 * n * max_conv_fft(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"GD iter {t:6d}: new best approx. score = {best_score:.6f}")

            if best_score < target_score:
                print("Target score reached during gradient descent.")
                break

    return best_a, best_score

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    deadline = time.time() + budget_s - 5.0  # keep a safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

            s = external.sum()
            if s < 1e-12:

                s = 1.0
            external = external / s
            score_ext = 2.0 * len(external) * max_conv_fft(external)

    except NameError:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lengths = [

    p_vals = np.linspace(0.2, 2.0, 37)          # finer grid for power‑law

        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
