"""
search for a low‑objective step‑function.

The algorithm is a two‑stage projected optimisation on the probability
simplex:

1.  Adam on a log‑sum‑exp (smooth‑max) surrogate of the self‑convolution
    maximum.  The smoothing parameter α is increased up to a large value
    (α≈4000) so that the surrogate becomes almost the true max.  After every
    iteration the iterate is projected onto the simplex with the
    Duchi‑et‑al. projection routine.

2.  After α has reached its maximum we switch to a true sub‑gradient step:
    the active indices where the convolution attains its maximum are
    identified, the gradient of the (non‑smooth) max is taken as the average
    of the corresponding gradients and a simple projected gradient descent
    with a decaying step size is performed.

Both phases keep the best point (according to the exact objective
`2·n·max(conv)/sum^2`) and occasionally apply a small random jitter or a
restart around the current best to escape plateaus.  Finally a cheap
pairwise‑swap hill‑climber is run for a few thousand moves.

The routine respects the 1000 s budget, works for any existing global
`height_sequence_1` (otherwise starts from a random simplex vector) and
returns the best sequence found as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper 2: Linear (non‑circular) self‑convolution using FFT.
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int, n: int):
    """Return (conv, p_fft) where conv = p * p (linear) and p_fft = rfft(p)."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
    return conv, p_fft

# ----------------------------------------------------------------------
# Helper 3: smoothed max (log‑sum‑exp) value and its gradient.
# ----------------------------------------------------------------------
def _smooth_max_and_grad(p: np.ndarray,
                         conv: np.ndarray,
                         p_fft: np.ndarray,
                         alpha: float,
                         m_fft: int,
                         n: int):
    """
    Compute the smoothed max  f(p)= (log Σ exp(alpha·b_k))/alpha
    and its gradient ∂f/∂p using FFT‑based correlation.
    Returns (smooth_value, gradient, w) where w are the soft‑max weights.
    """
    z = alpha * conv
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)          # stability trick
    sum_exp = np.sum(exp_z)
    w = exp_z / sum_exp                # soft‑max weights, sum to 1

    # correlation needed for the gradient
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]

    idx = 2 * n - 2 - np.arange(n)      # map correlation to gradient indices
    grad = 2.0 * corr_full[idx]         # ∂f/∂p (exact for the smoothed max)

    smooth_val = (max_z + np.log(sum_exp)) / alpha
    return smooth_val, grad, w

# ----------------------------------------------------------------------
# Helper 4: gradient of the *exact* max (sub‑gradient).
# ----------------------------------------------------------------------
def _exact_max_grad(p: np.ndarray,
                    conv: np.ndarray,
                    p_fft: np.ndarray,
                    m_fft: int,
                    n: int,
                    tol: float = 1e-12):
    """Return a sub‑gradient of max_k b_k at p (average over active indices)."""
    max_b = np.max(conv)
    active = np.where(conv >= max_b - tol)[0]
    if active.size == 0:               # should not happen, but guard anyway
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * corr_full[idx]
    return grad, max_b

# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with minimal `evaluate_sequence` score.
    Uses Adam on a smoothed max, then a sub‑gradient phase and a tiny
    hill‑climber.  The returned sequence respects the non‑negative
    and ≤1000 restrictions of `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size and initial vector
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # type: ignore[name‑defined]
        has_initial = True
    except Exception:
        n = 1153
        has_initial = False

    if has_initial:
        seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name‑defined]
        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total > 0.0:
            p = seq / total
        else:
            p = np.full(n, 1.0 / n)
    else:
        p = rng.random(n)
        p = p / p.sum()

    p = p.astype(float)

    # FFT size: next power‑of‑two that can hold convolution of length 3·n‑2
    m_fft = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # 2. Initial best score (exact objective)
    # ------------------------------------------------------------------
    conv, _ = _conv_self(p, m_fft, n)
    best_score = 2.0 * n * np.max(conv)
    best_p = p.copy()
    print(f"Initial exact objective: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimiser hyper‑parameters (smoothed‑max phase)
    # ------------------------------------------------------------------
    lr = 0.10
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    mom1 = np.zeros_like(p)
    mom2 = np.zeros_like(p)

    alpha = 1.0
    alpha_mul = 1.005          # a bit faster increase than the reference
    alpha_max = 4000.0         # much larger → tighter approximation of the true max

    jitter_interval = 2000
    restart_interval = 5000
    last_improve_iter = 0

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a few seconds margin
    iteration = 0

    # ------------------------------------------------------------------
    # 4. Phase‑1 : Adam on the smoothed maximum
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # compute convolution and smoothed max + gradient
        conv, p_fft = _conv_self(p, m_fft, n)
        _, grad, _ = _smooth_max_and_grad(p, conv, p_fft, alpha, m_fft, n)

        # Adam update
        mom1 = beta1 * mom1 + (1.0 - beta1) * grad
        mom2 = beta2 * mom2 + (1.0 - beta2) * (grad ** 2)

        m_hat = mom1 / (1.0 - beta1 ** iteration)
        v_hat = mom2 / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step
        p = _simplex_projection(p)

        # occasional jitter to shake the iterate
        if iteration % jitter_interval == 0:
            noise = rng.normal(scale=0.001, size=n)
            p = _simplex_projection(p + noise)

        # increase smoothing gradually, but never exceed alpha_max
        if alpha < alpha_max:
            alpha = min(alpha * alpha_mul, alpha_max)

        # exact objective for bookkeeping
        conv_exact, _ = _conv_self(p, m_fft, n)
        cur_score = 2.0 * n * np.max(conv_exact)

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            last_improve_iter = iteration
            print(f"Iter {iteration:7d} – new best (smoothed): {best_score:.6f}")

        # stop early if the target is already achieved
        if best_score <= 1.503:
            break

        # restart if we have been stagnant for a while
        if iteration - last_improve_iter > restart_interval:
            p = best_p + rng.normal(scale=0.02, size=n)
            p = np.clip(p, 0.0, None)
            p = p / p.sum()
            mom1.fill(0.0)
            mom2.fill(0.0)
            lr = max(lr * 0.5, 1e-5)
            last_improve_iter = iteration
            print(f"Iter {iteration:7d} – restart (perturb around best)")

    # ------------------------------------------------------------------
    # 5. Phase‑2 : exact‑max sub‑gradient descent (if time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        conv, p_fft = _conv_self(p, m_fft, n)
        grad, max_b = _exact_max_grad(p, conv, p_fft, m_fft, n, tol=1e-12)

        # simple step‑size schedule (decaying)
        step_size = 0.02 / (1.0 + iteration * 1e-4)
        p = p - step_size * grad
        p = _simplex_projection(p)

        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            last_improve_iter = iteration
            print(f"Iter {iteration:7d} – new best (exact‑grad): {best_score:.6f}")

        if best_score <= 1.503:
            break

        # safety break to avoid endless loop if budget almost exhausted
        if iteration % 500 == 0 and time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 6. Tiny local refinement using pairwise swaps
    # ------------------------------------------------------------------
    p = best_p.copy()
    conv, _ = _conv_self(p, m_fft, n)
    cur_score = 2.0 * n * np.max(conv)

    swap_attempts = 3000
    for _ in range(swap_attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_move = 1e-4 * rng.random()
        if p[i] <= eps_move:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move

        conv_new, _ = _conv_self(new_p, m_fft, n)
        new_score = 2.0 * n * np.max(conv_new)

        if new_score < cur_score:
            p = new_p
            cur_score = new_score
            best_p = p.copy()
            best_score = cur_score
            print(f"Swap improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Return the best sequence found (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    final_score = 2.0 * n * np.max(_conv_self(best_p, m_fft, n)[0])
    print(f"Final best score: {final_score:.6f}")

    return np.clip(best_p, 0.0, 1000.0).tolist()
