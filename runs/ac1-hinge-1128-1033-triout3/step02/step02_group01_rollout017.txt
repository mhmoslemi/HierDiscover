def subgradient_max(a, conv):
    # conv = full_convolution_fft(a)
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = len(a)
    grad = np.zeros_like(a)
    for k in max_idx:
        # indices i where both a[i], a[k-i] are defined
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

def smooth_gradient(a, conv, beta):
    n = len(a)
    max_c = conv.max()
    # stability shift
    shifted = conv - max_c
    w = np.exp(beta * shifted)
    w_sum = w.sum()

"""
Propose a low‑score step‑function.

Algorithm outline
-----------------
1.  Initialise a candidate:
    * If a global variable ``height_sequence_1`` exists, use it (normalised).
    * Otherwise, pick the best “U‑shaped’’ distribution from a tiny
      deterministic sweep (two families, a few exponents) for an initial
      size (≈2000).

2.  Multi‑stage refinement:
    * Stage 0 – optimise the initial size.
    * Stage 1 – upscale the best vector to a larger size (≈8000) by linear
      interpolation and continue optimisation.
    * Stage 2 – upscale again to a still larger size (≈20000) and finish
      optimisation.

   In each stage we run a projected gradient descent on a smooth
   approximation of the max‑convolution objective:

        fβ(a) = log‑sum‑exp( β·(a∗a) )          (β gradually grows)

   The gradient of fβ is computed via FFT‑based convolutions and the
   iterate is projected onto the probability simplex.  A simulated‑annealing
   acceptance rule with a slowly decaying temperature is used to escape
   shallow local minima.  Periodic random “bumps’’ re‑heat the search.

3.  The best vector found over all stages (lowest value of
    ``2·n·max(a∗a)``) is returned as a plain Python list.  The external
    evaluator will re‑normalise it, so we keep the sum exactly 1.
"""

# Helper utilities (all top‑level, no closures)

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {w≥0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """(min‑distance) U‑shape: a[i] ∝ (min(i,n-1-i)+1)^{-p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """(product‑distance) U‑shape: a[i] ∝ ((i+1)*(n-i))^{-p}."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    size = next_pow_two(2 * n)            # zero‑pad to avoid circular wrap‑around
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = next_pow_two(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: len(x) + len(y) - 1]


def subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a∗a)[k] using only the active (max) indices."""
    n = a.shape[0]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_vals = np.arange(lo, hi + 1, dtype=int)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def smooth_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth maximum
          fβ(a) = log Σ exp(β·(a∗a)[k]).
    Returns ∇fβ (same size as a).  ``conv`` must be a∗a.
    """
    n = a.shape[0]
    max_c = conv.max()
    # stability shift (max becomes 0)
    shifted = conv - max_c
    w = np.exp(beta * shifted)          # w[k] ≥ 0, w[max]=1
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    a_rev = a[::-1]
    # convolution w * a_rev  → length (2n‑1)+(n‑1) = 3n‑2
    conv_wa = convolve_fft(w, a_rev)
    grad = 2.0 * conv_wa[n - 1 : n - 1 + n]   # slice the relevant n entries
    return grad


def upscale_vector(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger size, followed by renormalisation."""
    old_n = len(vec)
    if new_n == old_n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up[:] = 1.0 / new_n
    else:
        up /= s
    return up


def find_best_ufam(n: int) -> np.ndarray:
    """Exhaustive search over the two U‑families for a given length."""
    best_vec = None
    best_score = np.inf
    exps = np.arange(0.1, 3.01, 0.2)
    for p in exps:
        a1 = u_min_distribution(n, p)
        conv1 = full_convolution_fft(a1)
        s1 = 2.0 * n * conv1.max()
        if s1 < best_score:
            best_score, best_vec = s1, a1.copy()
        a2 = u_prod_distribution(n, p)
        conv2 = full_convolution_fft(a2)
        s2 = 2.0 * n * conv2.max()
        if s2 < best_score:
            best_score, best_vec = s2, a2.copy()
    return best_vec


def optimise_stage(a0: np.ndarray,
                   deadline: float,
                   target_score: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Projected (smooth) gradient descent with simulated‑annealing acceptance.
    Returns the best vector found in the stage and its score.
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_conv = full_convolution_fft(cur)
    cur_score = 2.0 * n * cur_conv.max()

    best_vec = cur.copy()
    best_score = cur_score

    # hyper‑parameters (tuned empirically)
    temperature = 1.0
    cooling = 0.9999
    step = 0.12
    step_decay = 0.9997
    beta = 0.5
    beta_max = 8000.0
    beta_factor = 1.002

    it = 0
    last_improve = 0
    while time.time() < deadline:
        it += 1

        # ----- decide move -------------------------------------------------
        if np.random.rand() < 0.65:      # smooth‑gradient move
            grad = smooth_gradient(cur, cur_conv, beta)
            cand = cur - step * grad
        else:                             # simple random mass transfer
            i, j = np.random.choice(n, 2, replace=False)
            if cur[i] == 0.0:
                continue
            delta = cur[i] * np.random.rand() * 0.5
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta

        # ----- feasibility -------------------------------------------------
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)

        # ----- evaluate ----------------------------------------------------
        cand_conv = full_convolution_fft(cand)
        cand_score = 2.0 * n * cand_conv.max()

        # ----- acceptance --------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur, cur_conv, cur_score = cand, cand_conv, cand_score
            if cand_score < best_score:
                best_score, best_vec = cand_score, cand.copy()
                last_improve = it
                print(f"Iter {it:7d}: new best = {best_score:.6f}")

        # ----- schedule updates --------------------------------------------
        temperature *= cooling
        step *= step_decay
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # ----- stagnation handling -----------------------------------------
        if it - last_improve > 4000:
            # small random bump to escape plateaus
            bump = np.random.rand(n) * 1e-4
            cur = simplex_projection(cur + bump)
            cur_conv = full_convolution_fft(cur)
            cur_score = 2.0 * n * cur_conv.max()
            last_improve = it
            temperature = 1.0
            step = 0.12
            beta = 0.5
            print(f"Iter {it:7d}: re‑heat after stagnation")

        # early exit if we already beat the target
        if best_score <= target_score:
            break

    return best_vec, best_score


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    safety = 5.0
    deadline_total = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # Initialise candidate (use external construction if available)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            if external.sum() > 0.0:
                a_init = external / external.sum()
                print(f"Using external height_sequence_1 (len={len(a_init)})")
            else:
                a_init = None
        else:
            a_init = None
    except NameError:
        a_init = None

    if a_init is None:
        # deterministic U‑shape sweep for a modest size (≈2000)
        a_init = find_best_ufam(2000)
        print(f"Initial deterministic U‑shape (len={len(a_init)})")

    # ------------------------------------------------------------------
    # Multi‑stage optimisation (sizes roughly 2k → 8k → 20k)
    # ------------------------------------------------------------------
    stage_lengths = [len(a_init), 8000, 20000]
    # remove duplicates / non‑increasing entries
    stage_lengths = [l for i, l in enumerate(stage_lengths)
                     if i == 0 or l > stage_lengths[i - 1]]
    n_stages = len(stage_lengths)

    # allocate roughly equal time to each stage (later stages get a bit more)
    remaining_time = budget_s - safety
    stage_times = [remaining_time * (0.35 if i == n_stages - 1 else 0.3)
                   for i in range(n_stages)]

    best_global_seq = a_init.copy()
    best_global_score = 2.0 * len(best_global_seq) * full_convolution_fft(best_global_seq).max()

    for idx, (n_target, stage_time) in enumerate(zip(stage_lengths, stage_times)):
        stage_deadline = time.time() + stage_time
        if stage_deadline > deadline_total:
            stage_deadline = deadline_total

        if idx == 0:
            cur_vec = best_global_seq.copy()
        else:
            # upscale the best vector from the previous stage
            cur_vec = upscale_vector(best_global_seq, n_target)
            print(f"Upscaled to length {n_target}")

        # optimise this stage
        best_stage_vec, best_stage_score = optimise_stage(
            cur_vec, stage_deadline, target_score=1.503)

        # keep the absolute best across stages
        if best_stage_score < best_global_score:
            best_global_score = best_stage_score
            best_global_seq = best_stage_vec.copy()
            print(f"Stage {idx+1}/{n_stages}: new global best = {best_global_score:.6f}")

        if time.time() >= deadline_total:
            break

    print("\n=== Search finished ===")
    print(f"Best score found: {best_global_score:.6f}  (length {len(best_global_seq)})")
    # Return as a plain list – the external evaluator will clip/normalise again.
    return best_global_seq.tolist()
