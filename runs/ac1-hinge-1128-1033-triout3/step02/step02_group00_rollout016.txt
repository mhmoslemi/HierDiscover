"""
Mirror‑descent optimiser for the self‑convolution objective.

The evaluation function is
    f(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Because the ratio is homogeneous we may fix sum(a)=1.
The algorithm minimises the smoothed objective
    φ_β(a) = (1/β)·log( Σ_k exp(β·conv_k) )
which converges to the true maximum as β → ∞.
A mirror‑descent step on the probability simplex is performed
using the Kullback‑Leibler geometry (exponentiated‑gradient update).
The gradient of φ_β is a weighted sum of the sub‑gradients of each
convolution entry; the weights are just the soft‑max of the current
convolution values.  All heavy arithmetic (convolution, correlation)
is carried out by FFTs, so a few thousand iterations on a few‑thousand
element vector comfortably fit into the 1000 s budget.

Multiple restarts with different random seeds and a modest schedule for
β (starting smooth and ending sharp) are used; the best sequence found
so far is kept and finally returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _fft_size(min_len: int) -> int:
    """Return the next power‑of‑two not smaller than *min_len*."""
    return 1 << ((min_len - 1).bit_length())


def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Linear (full) convolution a*a using a real FFT of length *size*.
    The result has length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, size)
    conv_full = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    # tiny negative round‑off is removed
    np.maximum(conv_full, 0, out=conv_full)
    return conv_full


def _correlation_fft(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[i] = Σ_j a_j·w_{i+j}
    for i = 0 … len(a)-1, using FFT of length *size*.
    The returned array already contains the factor 2 required by the
    gradient of the soft‑max objective.
    """
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    # correlation is inverse FFT of A·conj(W)
    corr_full = np.fft.irfft(fa * np.conj(fw), size)
    start = w.shape[0] - 1          # index where lag 0 begins
    return 2.0 * corr_full[start : start + a.shape[0]]

    lengths = [3000, 6000, 10000]

    # Exponentiated‑gradient schedule

    beta_start = 3.0

    best_seq = None

    for n in lengths:
