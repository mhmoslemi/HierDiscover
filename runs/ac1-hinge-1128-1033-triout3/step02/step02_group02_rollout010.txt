```python
def simplex_projection(v):
    # project v onto simplex {x >= 0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

The goal is to find a non‑negative step function `a = (a₀,…,aₙ₋₁)` that minimises  

```
score(a) = 2·n·max (a * a) / ( Σ a )²                ( * )
```

The evaluator first normalises the vector (`a ← a / Σa`), therefore the
objective is *scale‑invariant* – only the **shape** of the vector matters.
If we write `p = Σ a = 1` the objective reduces to  

```
score(a) = 2·n·max (a * a)            ( ** )
```

Thus we have to make the maximal entry of the self‑convolution as small
as possible while keeping the entries non‑negative and summing to one.

The programme below consists of three stages

1. **Parametric sweep** – a fast O(n log n) scan over several simple,
   symmetric families (U‑shaped power laws, sum‑of‑tails, exponential
   decays) for a variety of lengths and parameters.  
   The best `k` candidates are kept.

2. **Smoothed gradient descent on the simplex** – starting from each of the
   retained candidates we minimise a *smooth* approximation of the max‑
   convolution

   ```
   F(a) = (1/β)·log Σₖ exp( β·(a*a)ₖ )
   ```

   Its gradient can be written in closed form

   ```
   ∂F/∂aᵢ = 2· Σₖ wₖ·aₖ₋ᵢ ,   wₖ ∝ exp(β·(a*a)ₖ) .
   ```

   The weights `wₖ` are computed from the current convolution, the gradient
   is obtained by a single FFT‑based convolution, and a step in the
   opposite direction is projected onto the probability simplex
   (Euclidean projection).  
   The smoothing parameter `β` is increased slowly, turning the smoothed
   problem into the original max‑problem.  A simple back‑tracking line‑search
   for the step size together with a tiny simulated‑annealing acceptance
   rule gives a robust local optimiser.

3. **Final exact evaluation** – the best vector found is fed to the
   provided `evaluate_sequence` function once and returned.

The whole search respects the 1000 s budget, prints intermediate progress,
and works for any length up to a few × 10⁴ (the FFTs stay cheap).

```python
"""
Propose a step‑function that minimises the given evaluation metric.

Algorithm
---------
1. Fast parametric sweep over several simple families (U‑shaped power‑law,
   sum‑of‑tails, exponential) for a handful of lengths.  The best few
   candidates (by the cheap approximation 2·n·max(a*a)) are kept.

2. Smoothed gradient descent on the simplex:
   * At iteration t we have a shape a (Σa = 1, a ≥ 0).
   * Compute the full self‑convolution c = a * a (FFT, length 2·n‑1).
   * Form soft‑max weights wₖ ∝ exp(β·(cₖ−max(c)))  (β starts small and
     grows slowly).
   * Gradient of the smoothed objective is
         gᵢ = 2· Σₖ wₖ·aₖ₋ᵢ .
     This is a single convolution a * w_rev and can be obtained with an FFT.
   * Update a ← a – η·g, then project onto the probability simplex.
   * Accept the update if the true (unsmoothed) score improves, otherwise
     shrink η.  Increase β gradually so the algorithm finally targets the
     true max‑convolution.
   * Repeat until time runs out or the target score is reached.

3. Return the best vector (as a plain list) after a final call to
   ``://evaluate_sequence`` for the exact score.

The search runs entirely on one CPU, uses only NumPy and standard Python
libraries, and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x ≥ 0 , Σ x = 1 }.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a normalised vector i.e. Σa = 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n-1-i)+1)⁻ᵖ."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12           # avoid exact zeros
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)⁻ᵖ + (n-i)⁻ᵖ."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·dist)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def random_distribution(n: int) -> np.ndarray:
    """Uniformly random normalised non‑negative vector."""
    a = np.random.rand(n)
    a /= a.sum()
    return a

# Smoothed gradient descent on the simplex
# ----------------------------------------------------------------------


def approx_score(a: np.ndarray) -> float:
    """Fast score 2·n·max(a*a) for a normalised vector."""
    n = a.shape[0]
    return 2.0 * n * max_conv_fft(a)


def gradient_descent_refine(
    a0: np.ndarray,
    deadline: float,
    target_score: float = 1.503,
    max_iters: int = 5000,
    beta0: float = 0.5,
    beta_factor: float = 1.02,
    eta0: float = 0.5,
    eta_growth: float = 1.05,
    eta_decay: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Refine a normalised vector a0 by minimising the (smoothed) max‑convolution.
    Returns the best vector seen and its (approx.) score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = approx_score(a)

    beta = beta0
    eta = eta0

    L = 2 * n - 1               # length of convolution / weight vector
    idx_slice = L - 1 - np.arange(n)   # indices to pick the gradient from conv

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- current convolution and true score -----
        conv = full_conv_fft(a)
        max_c = conv.max()
        cur_sc = 2.0 * n * max_c

        # keep global best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"Iter {it:6d}: new best approx. score = {best_sc:.6f}")

        # early exit if target reached
        if best_sc < target_score:
            print("Target reached inside refinement.")
            break

        # ----- soft‑max weights -----
        # wₖ ∝ exp(β·(cₖ - max_c))  (max exponent = 0 → numerical stability)
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (2 * a * w_rev) -----
        w_rev = w[::-1]                     # reversed weight vector
        # FFT based convolution a * w_rev (length 3·n‑2)
        size = 1 << ((n + L - 1).bit_length())
        A = np.fft.rfft(a, n=size)
        B = np.fft.rfft(w_rev, n=size)
        grad_full = np.fft.irfft(A * B, n=size)[: n + L - 1]
        grad = 2.0 * grad_full[idx_slice]

        # ----- tentative update -----
        a_candidate = a - eta * grad
        a_candidate = simplex_projection(a_candidate)

        # evaluate true score of the candidate
        conv_cand = full_conv_fft(a_candidate)
        cand_sc = 2.0 * n * conv_cand.max()

        # ----- acceptance -----
        if cand_sc < cur_sc:
            # accept improvement
            a = a_candidate
            eta *= eta_growth          # a bit larger step after success
        else:
            # reject – shrink step size
            eta *= eta_decay
            if eta < 1e-8:      # step became too tiny → stop
                break

        # ----- schedule -----
        beta = min(beta * beta_factor, 20.0)   # don't let β explode

    return best_a, best_sc

            deadline=deadline,
            target_score=1.503,
            max_iters=8000,
        )
        if sc_refined < global_best_score:
            global_best_score = sc_refined
            global_best_seq = a_refined.copy()
            print(f"  *** New global best approx. score = {global_best_score:.6f}")

        # early exit if we already beat the target
        if global_best_score < 1.503:
            print("Target score reached – stopping refinement.")
            break

    # ------------------------------------------------------------------
    # 4.  Final exact evaluation (once) and report.
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(global_best_seq.tolist())   # noqa: F821
        print("\nExact evaluation by evaluate_sequence: {:.6f}".format(exact_score))
    except Exception as e:
        print("\nCould not call evaluate_sequence:", e)
        exact_score = global_best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {global_best_score:.6f}")
    print(f"Best exact score       : {exact_score:.6f}")
    print(f"Best length            : {len(global_best_seq)}")
    return global_best_seq.tolist()
```
