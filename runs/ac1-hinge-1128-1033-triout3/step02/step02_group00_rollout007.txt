def full_convolution_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    np.maximum(conv, 0, out=conv)
    return conv

"""
Propose a low‑objective step‑function using a two‑phase optimizer.

Phase 1 – fast sub‑gradient descent with Adam‑style momentum.
            Works on the exact max‑self‑convolution (the non‑smooth part).

Phase 2 – refined smoothed‑max descent.
            The max is approximated by a log‑sum‑exp (soft‑max)
            which yields a smooth gradient; Adam is used again.
            The temperature (γ) of the soft‑max is increased
            linearly so the optimisation gradually focuses on the true
            maximum.

Both phases keep the vector on the probability simplex
( non‑negative, sum = 1 ) using the O(n log n) Duchi projection.
A small collection of analytically motivated initial shapes
(uniform, U‑shaped sin, power‑law, arcsine‑like) are tested and the
best one is taken as the start point.  Random transfers and occasional
re‑initialisations help escape plateaus.

The routine respects the 1000 s budget, logs progress and finally
returns the best sequence found (as a plain list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full linear convolution using a real FFT (fast for many iterations)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear (full) convolution of vectors x and y."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    np.maximum(conv, 0, out=conv)        # clip tiny negatives
    return conv


def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return _full_convolution_fft(a, a)


def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a)  – the quantity that evaluate_sequence returns for sum(a)=1."""
    n = a.shape[0]
    conv = _self_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialise a set of plausible starting shapes and pick the best one.
# ----------------------------------------------------------------------
def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised (sum=1) sequence of length n with low initial objective."""
    candidates = []

    # 1) uniform
    a = np.full(n, 1.0 / n)
    candidates.append(a)

    # 2) U‑shaped sine: high on the edges, zero in the centre
    x = np.arange(n, dtype=float)
    a = 1.0 - np.sin(np.pi * x / (n - 1))
    a = np.maximum(a, 0.0)
    if a.sum() > 0:
        a /= a.sum()
        candidates.append(a)

    # 3) power‑law families (edge‑heavy for negative exponents)
    for p in (-1.0, -0.75, -0.5, -0.25):
        idx = np.arange(n, dtype=float)
        prod = (idx + 1) * (n - idx)          # (i+1)*(n-i)
        a = prod ** p
        if np.any(np.isinf(a)):
            a = np.nan_to_num(a, posinf=0.0, neginf=0.0, nan=0.0)
        s = a.sum()
        if s > 0:
            a /= s
            candidates.append(a)

    # 4) arcsine‑like (beta(½,½) density discretised)
    idx = np.arange(n, dtype=float) + 0.5
    denom = np.sqrt(idx * (n - idx))
    a = 1.0 / np.maximum(denom, 1e-12)
    s = a.sum()
    if s > 0:
        a /= s
        candidates.append(a)

    # 5) a few random Dirichlet vectors (use rng for reproducibility)
    for _ in range(3):
        a = rng.dirichlet(np.ones(n))
        candidates.append(a)

    # Pick the candidate with the smallest objective
    best = candidates[0]
    best_val = _objective(best)
    for cand in candidates[1:]:
        val = _objective(cand)
        if val < best_val:
            best = cand
            best_val = val
    return best
