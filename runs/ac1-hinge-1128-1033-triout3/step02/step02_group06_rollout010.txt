The procedure:
    1. Scan p = 2 … 15 (any larger p only improves the bound).
    2. For each p compute the analytic score for the maximal allowed
       length (default 15000) using integer binomials.
    3. Choose the p with the smallest score.
    4. Build the corresponding a_i by a simple recurrence,
       scale to ≤1000 and return the list.
Only a single call to `evaluate_sequence` is made (to report the real
score); the heavy work is done analytically, so the 1000 s budget is
ample.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     max_len: int = 15000,
                     **kwargs) -> list[float]:
    """
    Generate a step‑function whose self‑convolution peak is extremely low.
    The construction uses the binomial family a_i = C(i+p‑1, p‑1) and
    picks the integer exponent p (2 ≤ p ≤ 15) that gives the smallest
    analytic upper bound on the evaluation function.
    The resulting list is scaled so that every entry lies in [0, 1000].

    Parameters
    ----------
    seed : int
        Unused – kept for API compatibility.
    budget_s : int
        Time budget (seconds).  The algorithm finishes well before this.
    max_len : int
        Desired length of the step function (capped at 15000).

    Returns
    -------
    List[float]
        The candidate sequence.
    """
    # ------------------------------------------------------------------
    # Safety margin and length handling
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0                     # leave buffer
    max_len = int(min(max_len, 15000))                         # hard cap

    # ------------------------------------------------------------------
    # Phase 1 – analytic search over p
    # ------------------------------------------------------------------
    best_score = float('inf')
    best_p = None

    # we only need a few p values – p≥2 gives a non‑trivial improvement,
    # p up to 15 already drives the constant below 2·10⁻⁴.
    for p in range(2, 16):
        n = max_len

        # sum S_n = C(n+p-1, p)  (small k = p)
        S = math.comb(n + p - 1, p)

        # max convolution value = C(n+2p-2, 2p-1)
        # use the smaller k = 2p-1 (since 2p-1 << n for our range)
        max_b = math.comb(n + 2 * p - 2, 2 * p - 1)

        # analytic objective (exact rational, then to float)
        score = 2.0 * n * max_b / (S * S)

        if score < best_score:
            best_score = score
            best_p = p

    # ------------------------------------------------------------------
    # Phase 2 – build the concrete sequence for the chosen p
    # ------------------------------------------------------------------
    p = best_p
    n = max_len

    # a_0 = 1, a_i = a_{i-1} * (i + p - 1) / i   (i≥1)
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (i + p - 1) / i

    # Scale the whole vector so that its maximum becomes exactly 1000.
    # Scaling does NOT change the evaluation score because the objective
    # is homogeneous of degree zero.
    scale = 1000.0 / seq[-1]
    seq *= scale

    # Convert to plain Python floats (evaluate_sequence expects a list)
    candidate = [float(x) for x in seq]

    # ------------------------------------------------------------------
    # Phase 3 – do a single sanity check with the provided evaluator
    # ------------------------------------------------------------------
    try:
        real_score = evaluate_sequence(candidate)
        print(f"[result] p={p}, length={n}, analytic={best_score:.9e}, "
              f"real={real_score:.9e}")
    except Exception as exc:
        real_score = best_score
        print(f"[result] p={p}, length={n}, analytic={best_score:.9e}, "
              f"evaluation error ({exc})")

    # Final safety: clamp any accidental negatives (should not happen)
    candidate = [float(max(0.0, x)) for x in candidate]

    return candidate
