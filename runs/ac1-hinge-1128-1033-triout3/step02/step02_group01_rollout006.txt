from scipy.optimize import linprog

def lp_move(f, M):
    n = len(f)
    conv = full_convolution_fft(f)
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - eps)[0]
    # Build A_ub: each row length n. For each active k:
    # A_ub[k, j] = f_{k-j} if 0 <= k-j < n else 0.
    # This is like f shifted.
    A = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        # for each j: b_j contributes to conv at position k via f_{k-j}
        # So A[idx, j] = f_{k-j} if within bounds
        start = max(0, k-(n-1))
        end = min(k, n-1)
        # j runs from start to end, and for each j, f_{k-j}
        # We'll fill vector:
        js = np.arange(start, end+1)
        A[idx, js] = f[k - js]
    # Constraints: A @ b <= M
    # Objective: maximize sum(b) --> minimize -sum(b)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(len(active), M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        b = res.x
        # scale to sum=1
        if b.sum() > 0:
            b = b / b.sum()
        return b
    else:
        # fallback: return original f
        return f
```

"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Deterministic seed generation** – search a few parametric “U‑shaped’’
   families (min‑distance and product‑distance) for several lengths and
   exponents.  The best candidate becomes the initial point.

2. **LP‑guided mixing** – given the current vector `f` and its maximal
   convolution value `M`, solve a linear program that maximises the
   ℓ₁‑norm of a non‑negative vector `g` under the constraints
   (f∗g)[k] ≤ M for all indices `k` where the convolution of `f` with
   itself attains its maximum.  Mixing `f ← (1‑t)f + t g` with a
   decreasing step size provably never increases the objective
   `2·n·max(f∗f)`.  Repeating this yields a monotone descent.

3. **Fine‑grained refinement** – after the LP phase, perform projected
   sub‑gradient descent on the simplex, driven by the sub‑gradient of
   the maximal convolution entry, together with occasional random mass‑
   transfer moves (simulated‑annealing acceptance).  This phase
   smoothes out remaining irregularities.

All convolutions are computed with a single FFT, giving an overall fast
implementation that can explore many millions of candidates well within
the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w >= 0, sum(w) = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to closest edge
    a = (d + 1) ** (-p) + 1e-14           # avoid exact zeros
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a∗a (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n).bit_length())   # next power of two ≥ 2n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a vector already normalised."""
    n = len(a)
    max_c = max_convolution(a)
    return float(2.0 * n * max_c)


def active_set(conv: np.ndarray, rel_tol: float = 1e-12) -> np.ndarray:
    """Indices where conv is (numerically) maximal."""
    mx = conv.max()
    eps = mx * rel_tol + 1e-15
    return np.where(conv >= mx - eps)[0]

def build_constraint_matrix(f: np.ndarray, active: np.ndarray) -> np.ndarray:
    """

    coefficients f_{k‑j} (or 0) so that (f * b)[k] = (A·b)[k].

    for row, k in enumerate(active):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        js = np.arange(j_min, j_max + 1, dtype=int)
        A[row, js] = f[k - js]
    return A


def lp_optimize(f: np.ndarray, M: float, active: np.ndarray) -> np.ndarray:
    """
    Solve   max Σ b_j   s.t.   (f * b)[k] ≤ M  (k in active),  b ≥ 0.
    Returns the feasible vector normalised to sum = 1.
    If the LP fails, the original `f` is returned.
    """
    n = f.shape[0]
    if active.size == 0:
        return f.copy()

    A = build_constraint_matrix(f, active)
    b_ub = np.full(active.size, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σb → minimise -Σb
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    except Exception:  # pragma: no cover – safety net
        return f.copy()

    if not res.success:
        return f.copy()

    b = res.x
    s = b.sum()
    if s > 0:
        b /= s
    else:
        b = f.copy()
    return b


# ----------------------------------------------------------------------
# Sub‑gradient refinement utilities
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the objective 2·n·max(a∗a).
    Returns the candidate vector (already normalised to sum = 1).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    mx = conv.max()
    eps = mx * 1e-12 + 1e-15
    idx = np.where(conv >= mx - eps)[0]

    grad = np.zeros_like(a)
    for k in idx:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]

    cand = a - step * grad
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s > 0:
        cand /= s
    else:
        cand = a.copy()
    return cand


def random_mass_transfer(a: np.ndarray, amount_factor: float = 0.5) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    if a[i] == 0.0:
        return a.copy()
    delta = a[i] * np.random.rand() * amount_factor
    b = a.copy()
    b[i] -= delta
    b[j] += delta
    # re‑normalise (sum stays 1, but numerical noise may appear)
    s = b.sum()
    if s > 0:
        b /= s
    return b

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the given time budget and returns the best sequence
    (as a plain Python list) discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Deterministic seed generation (U‑shapes)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    lengths = [2000, 4000, 6000, 8000, 12000, 16000, 20000]
    exponents = np.arange(0.1, 3.01, 0.1)

    print("=== Exhaustive sweep over parametric families ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for p in exponents:
            # --- min‑distance family ---
            a1 = u_min_distribution(n, p)
            sc1 = score_from_vector(a1)
            if sc1 < best_score:
                best_score, best_seq = sc1, a1.copy()
                print(f" new best (min)   n={n:5d} p={p:4.2f}  score={sc1:.6f}")

            # --- product‑distance family ---
            a2 = u_prod_distribution(n, p)
            sc2 = score_from_vector(a2)
            if sc2 < best_score:
                best_score, best_seq = sc2, a2.copy()
                print(f" new best (prod)  n={n:5d} p={p:4.2f}  score={sc2:.6f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Use external construction if available (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            if external.sum() > 0:
                external = external / external.sum()
                sc_ext = score_from_vector(external)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Fallback (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 6000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  LP‑guided monotone descent
    # ------------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score
    n = current.shape[0]

    lp_iter = 0
    max_lp_iters = 300   # enough to push the bound down, cheap overall
    start_t = 0.2
    min_t = 0.01

    print("\n=== LP‑guided mixing phase ===")
    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1

        # convolution of current with itself
        conv = full_convolution_fft(current)
        M = conv.max()
        act = active_set(conv)

        # LP solve
        g = lp_optimize(current, M, act)

        # If LP returns (almost) the same vector, stop early
        if np.linalg.norm(g - current) < 1e-12:
            print(" LP solution coincides with current – stopping LP phase")
            break

        # Try a geometric schedule of mixing rates
        t = start_t
        improved = False
        while t >= min_t:
            cand = (1.0 - t) * current + t * g
            cand_score = score_from_vector(cand)
            if cand_score <= current_score:
                current = cand
                current_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score, best_seq = cand_score, cand.copy()
                    print(f" LP iter {lp_iter:4d} new best = {best_score:.6f}")
                break
            t *= 0.5

        if not improved:
            # no monotone improvement – inject a tiny random bump
            bump = np.random.rand(n) * 1e-4
            current = simplex_projection(current + bump)
            current_score = score_from_vector(current)

        # optional cooling of step size
        start_t *= 0.98
        if start_t < min_t:
            start_t = min_t

        # occasional sanity check
        if lp_iter % 20 == 0:
            elapsed = time.time() + budget_s - deadline - budget_s
            print(f" LP iter {lp_iter:4d}: score={current_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Fine‑grained stochastic refinement (sub‑gradient + random moves)
    # ------------------------------------------------------------------
    print("\n=== Stochastic refinement phase ===")
    temperature = 1.0
    cooling_rate = 0.9995
    step_size = 0.2
    step_decay = 0.9999
    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1

        # ---------- pick move ----------
        if np.random.rand() < 0.4:
            cand = random_mass_transfer(current, amount_factor=0.5)
        else:
            cand = subgradient_step(current, step_size)

        cand_score = score_from_vector(cand)

        # ---------- accept / reject ----------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current, current_score = cand, cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---------- annealing ----------
        temperature *= cooling_rate
        step_size *= step_decay

        # ---------- stagnation handling ----------
        if iteration - last_improve_iter > 25000:
            # small random perturbation to escape flat region
            bump = np.random.rand(n) * 1e-4
            current = simplex_projection(current + bump)
            current_score = score_from_vector(current)
            last_improve_iter = iteration
            temperature = 1.0

    # ------------------------------------------------------------------
    # 6️⃣  Return best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will re‑clip/normalise, so we return raw values.
    return best_seq.tolist()
