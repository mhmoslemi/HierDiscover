"""
Propose a low‑score step‑function.

The search proceeds in three stages:

1.  A deterministic sweep over two simple “U‑shaped’’ families (based on the
    distance to the nearest edge and on the product of the distances to both
    edges).  A few hundred candidate lengths and exponents are examined and the
    best raw vector (normalised to sum = 1) is remembered.

2.  A smoothed‑max gradient descent on the simplex using the Adam optimiser.
    The objective is τ·log(∑exp((a∗a)/τ)), whose gradient can be obtained by a
    single correlation computed with FFTs.  τ is slowly annealed to zero, so the
    optimisation converges to the true max‑convolution objective.
    After each step the vector is projected onto the probability simplex.

3.  Optional up‑sampling of the final result to a longer length (if time
    permits) and a short re‑optimisation at the larger resolution.

The routine respects the supplied time budget (default ≈ 1000 s) and returns
the best sequence it discovered as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Small utility returning the next power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {w ≥ 0, sum(w) = 1}.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on distance to the nearest edge:
        a[i] = (min(i, n‑1‑i) + 1)^{‑p}.
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the product of distances to both edges:
        a[i] = ((i+1)*(n‑i))^{‑p}.
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a∗a using a real‑valued FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a (sum(a)=1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a vector already normalised
    to sum = 1.  The factor 2·n·max_c is the same as evaluate_sequence's
    output when the sum is 1.
    """
    n = a.shape[0]
    max_c = max_convolution(a)
    return 2.0 * n * max_c


def correlation_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlate w (length 2·n‑1) with a (length n) and return the first n
    lags, i.e.   (w ⋆ a)[i] = Σ_k w_{i+k} a_k  (0 ≤ i < n).
    Implemented via FFT with zero‑padding to the next power of two.
    """
    n = a.shape[0]
    size = next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a[::-1], n=size)
    corr_full = np.fft.irfft(fw * fa_rev, n=size)
    start = n - 1
    return corr_full[start:start + n]


# ----------------------------------------------------------------------
# Main proposal routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence it discovered.
    """
    np.random.seed(seed)
    # small safety margin for the host runner
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 1. Deterministic sweep over parametric families
    # --------------------------------------------------------------
    # lengths adapted to the time budget – larger lengths tend to give lower
    # constants but cost more per iteration.
    lengths = [4000, 8000, 16000, 32000]
    # exponent grid (covers weak to strong U‑shapes)
    p_vals = np.concatenate(
        [np.arange(0.1, 2.01, 0.1), np.arange(2.1, 5.01, 0.2)]
    )

    print("=== Deterministic U‑shape sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for p in p_vals:
            # family 1: distance‑to‑edge
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (min)  n={n:5d} p={p:5.2f}  score={sc:.6f}")

            # family 2: product‑distance
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (prod) n={n:5d} p={p:5.2f}  score={sc:.6f}")

        # a few random exponents for extra coverage
        for _ in range(5):
            p = np.random.uniform(0.05, 5.0)
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (rand) n={n:5d} p={p:5.2f}  score={sc:.6f}")

    # --------------------------------------------------------------
    # 2. External construction (if present)
    # --------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score = sc
                    best_seq = external.copy()
                    print(f" using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3. Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Smoothed‑max Adam optimisation on the simplex
    # --------------------------------------------------------------
    n = len(best_seq)
    current = best_seq.copy()
    best_seq_global = best_seq.copy()
    best_score_global = best_score

    # Adam state
    m = np.zeros_like(current)
    v = np.zeros_like(current)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    t_iter = 0

    # initial temperature for the smoothed max
    conv = full_convolution_fft(current)
    max_c = conv.max()
    tau = max_c * 0.5 if max_c > 0 else 1e-4
    tau_min = max_c * 1e-6 + 1e-12
    tau_decay = 0.9995

    # Adam learning rate schedule
    lr = 0.2
    lr_decay = 0.9999

    print("\n=== Starting smoothed‑max Adam refinement ===")
    while time.time() < deadline:
        t_iter += 1

        # current max‑convolution and exact score
        conv = full_convolution_fft(current)
        max_c = conv.max()
        score = 2.0 * n * max_c

        if score < best_score_global:
            best_score_global = score
            best_seq_global = current.copy()
            print(f"Iter {t_iter:7d}: new global best score = {best_score_global:.6f}")

        # ----- smoothed max (soft‑max) weights -----
        shift = max_c  # stability trick
        w = np.exp((conv - shift) / max(tau, 1e-12))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed max -----
        grad = 2.0 * correlation_fft(w, current)

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = current - step
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)

        # evaluate candidate (exact objective)
        conv_cand = full_convolution_fft(cand)
        score_cand = 2.0 * n * conv_cand.max()

        # simulated‑annealing style acceptance
        if score_cand < score:
            accept = True
        else:
            prob = np.exp((score - score_cand) / max(tau, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = cand
            if score_cand < best_score_global:
                best_score_global = score_cand
                best_seq_global = cand.copy()
                print(f"Iter {t_iter:7d}: new best after accept score = {best_score_global:.6f}")

        # schedule updates
        tau = max(tau * tau_decay, tau_min)
        lr *= lr_decay

        # occasional random perturbation to escape flat regions
        if t_iter % 5000 == 0:
            bump = np.random.rand(n) * 1e-4
            current = simplex_projection(current + bump)

        # safety exit
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 5. Optional up‑sampling to a longer length (if time permits)
    # --------------------------------------------------------------
    # A simple linear interpolation often yields a slightly better constant
    # when the length is increased.
    max_len = 32000
    up_factors = [2, 3]  # try to double or triple the size
    for fac in up_factors:
        if time.time() >= deadline:
            break
        n_big = n * fac
        if n_big > max_len:
            continue
        # linear interpolation onto a finer grid
        x_small = np.linspace(0, 1, n)
        x_big = np.linspace(0, 1, n_big)
        up_seq = np.interp(x_big, x_small, best_seq_global)
        up_seq = np.maximum(up_seq, 0.0)
        up_seq /= up_seq.sum()
        up_score = score_from_vector(up_seq)
        print(f"Upsample factor {fac:1d}: length {n_big}  score {up_score:.6f}")
        if up_score < best_score_global:
            best_score_global = up_score
            best_seq_global = up_seq.copy()
            # a short fine‑tune at the larger size
            current = up_seq.copy()
            m = np.zeros_like(current)
            v = np.zeros_like(current)
            tau = max_c * 0.5 if max_c > 0 else 1e-4
            lr = 0.2
            t_iter = 0
            while time.time() < deadline:
                t_iter += 1
                conv = full_convolution_fft(current)
                max_c = conv.max()
                score = 2.0 * n_big * max_c
                if score < best_score_global:
                    best_score_global = score
                    best_seq_global = current.copy()
                    print(f"  [refine] iter {t_iter:5d} best={best_score_global:.6f}")
                shift = max_c
                w = np.exp((conv - shift) / max(tau, 1e-12))
                w /= w.sum()
                grad = 2.0 * correlation_fft(w, current)
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad ** 2)
                m_hat = m / (1 - beta1 ** t_iter)
                v_hat = v / (1 - beta2 ** t_iter)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                cand = current - step
                cand = np.maximum(cand, 0.0)
                cand = simplex_projection(cand)
                conv_cand = full_convolution_fft(cand)
                score_cand = 2.0 * n_big * conv_cand.max()
                if score_cand < score or np.random.rand() < 0.001:
                    current = cand
                tau = max(tau * tau_decay, tau_min)
                lr *= lr_decay
                if t_iter % 2000 == 0:
                    bump = np.random.rand(n_big) * 1e-4
                    current = simplex_projection(current + bump)
                if time.time() >= deadline:
                    break

    # --------------------------------------------------------------
    # 6. Return the best sequence found
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score_global:.6f}   (length {len(best_seq_global)})")
    # The external evaluator will re‑normalise again, so we return the raw vector.
    return best_seq_global.tolist()
