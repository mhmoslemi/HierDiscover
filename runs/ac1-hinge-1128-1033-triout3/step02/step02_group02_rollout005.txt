def compute_conv_fft(a):
    n=len(a)
    size=1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    return conv

def smooth_max(conv, alpha):
    mx = conv.max()
    # compute log-sum-exp safely
    # subtract mx for numeric stability
    exps = np.exp(alpha * (conv - mx))
    sum_exps = exps.sum()
    return (mx + np.log(sum_exps) / alpha)

The evaluation function depends only on the *shape* of the step‑function (the vector is first normalised, and the objective is  
`2·n·max(a∗a)` where `∗` denotes the self‑convolution).  
Hence the problem is:

*given a non‑negative vector `a` with `sum(a)=1`, minimise `max(conv)`*  
where `conv = a∗a`.

The code below searches this space in two stages:

1. **Initialisation** – use the previously discovered best sequence
   (`height_sequence_1` if it exists) or a simple U‑shaped distribution.
   The vector is resampled to several lengths (4000 … 20000) so that we
   can try to improve both the shape and the length.

2. **Gradient‑descent on a smooth approximation of the max**  
   The non‑smooth `max(conv)` is replaced by a *soft‑max*  

   ```
   Sα(a) = (1/α)·log Σ_k exp(α·conv_k)      (α>0)
   ```

   For large `α` the soft‑max approaches the true maximum, but for small
   `α` it is smooth and its gradient can be written in closed form.
   Using the convolution theorem the gradient can be obtained with a few
   FFTs (O(n log n) per iteration).

   The update rule is a multiplicative mirror‑descent step

   ```
   a ← a · exp(–η·∇Sα(a)) ,   then renormalise to sum = 1
   ```

   `η` (step size) and `α` (smoothness) are adapted on‑the‑fly:
   * `α` is slowly increased so that the algorithm first moves in a
     smooth landscape and later focuses on the true maximum.
   * `η` is increased after a successful step and decreased after a
     failure (simple adaptive scheme).

   The loop stops when the time budget is exhausted or when the
   experimentally observed bound drops below the target `1.503`.

All heavy work is performed with real FFTs (`np.fft.rfft`) and the
implementation never stores huge dense matrices, therefore it comfortably
fits into the 1000 s time limit on a single CPU.

```python
"""
Propose a step‑function (non‑negative vector) that minimises the
evaluation metric used by `evaluate_sequence`.

Algorithm
---------

1.  Initialise from the previously supplied best construction
    (`height_sequence_1`) if it exists, otherwise from a simple
    symmetric U‑shaped distribution.  The initial vector is resampled
    to a set of candidate lengths (4000‑20000).

2.  For each length run a mirror‑descent optimiser on the *soft‑max*
    of the self‑convolution.
      * The soft‑max `Sα(a) = (1/α)·log Σ exp(α·conv)` is a smooth
        upper bound of `max(conv)`.  Its gradient is
        `∇Sα(a) = Σ_k w_k ∂conv_k/∂a` with weights `w_k` = soft‑max
        probabilities.
      * Using the convolution theorem the gradient can be obtained
        with three FFTs per iteration (O(n log n)).
      * The update is multiplicative: `a ← a·exp(–η·∇Sα(a))` and then
        renormalised to the probability simplex.
      * `α` is increased slowly (making the bound tighter) and `η`
        is adapted depending on whether the last step improved the
        true objective `2·n·max(conv)`.

3.  The best vector seen over all lengths is finally evaluated with the
    official `evaluate_sequence` and returned.

The routine respects the 1000 s budget, prints progress information,
and never uses any file or network I/O.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ------------------------------------------------------------

def max_conv_fft(a: np.ndarray) -> float:
    """
    Return the maximum entry of the self‑convolution a*a using a real FFT.
    a must be a 1‑D numpy array (non‑negative, typically sum(a)=1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the *full* self‑convolution a*a (length 2·n-1) using the
    supplied FFT size (must be >= 2·n-1).  Returned array is real.
    """
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * a.shape[0] - 1]
    return conv


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the normalised soft‑max weights w_k = exp(α·conv_k) / Σ exp().
    The exponent is stabilised by subtracting the maximum entry.
    """
    mx = conv.max()
    exps = np.exp(alpha * (conv - mx))
    s = exps.sum()
    # guard against division by zero (should never happen)
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def gradient_softmax(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Gradient of the soft‑max of the convolution w.r.t. a.
    a : vector of length n
    w : weights of length 2·n‑1 (sum to 1)
    size : FFT length (>= 3·n‑2)
    Returns a vector of length n.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of w
    # FFT of a (already normalised) – reuse A if already known?
    A = np.fft.rfft(a, n=size)

    # FFT of w and of its reversal
    W = np.fft.rfft(w, n=size)
    W_rev = np.fft.rfft(w[::-1], n=size)

    # Convolution a * w  (length n+L‑1 = 3·n‑2)
    conv_aw = np.fft.irfft(A * W, n=size)[: n + L - 1]
    conv_aw_rev = np.fft.irfft(A * W_rev, n=size)[: n + L - 1]

    # Gradient = conv_aw[:n] + conv_aw_rev[L-1 - np.arange(n)]
    idx = L - 1 - np.arange(n)                     # 2·n‑2, …, n‑1
    grad = conv_aw[:n] + conv_aw_rev[idx]
    return grad


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the O(n log n) algorithm
    from Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def resample_distribution(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resample a probability vector `old` (sum=1) to length `new_n`
    using linear interpolation and re‑normalise.
    """
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    # avoid exact zeros (helps the gradient step)
    new += 1e-14
    new /= new.sum()
    return new


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Simple symmetric power‑law “U‑shape” used for a quick fallback
    when no external seed is available.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)               # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


# ------------------------------------------------------------
# Main search routine
# ------------------------------------------------------------

def propose_candidate(budget_s: int = 1000, seed: int = 12345) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the given time budget and
    returns the best list of floats it found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # tiny safety margin

    # --------------------------------------------------------
    # 0.  Seed from the previously supplied construction, if any
    # --------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            external_sum = external.sum()
            if external_sum < 1e-12:
                external = np.ones_like(external) / external.size
                external_sum = 1.0
            external = external / external_sum
            cur_score = 2.0 * external.size * max_conv_fft(external)
            best_seq = external.copy()
            best_score = cur_score
            print(f"Seeded from height_sequence_1: length={external.size}, "
                  f"score≈{cur_score:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------
    # 1.  If no seed, fall back to a simple U‑shape (n=8000)
    # --------------------------------------------------------
    if best_seq is None:
        n0 = 8000
        best_seq = u_shaped_distribution(n0, p=0.8)
        best_score = 2.0 * n0 * max_conv_fft(best_seq)
        print(f"Fallback initialisation: n={n0}, score≈{best_score:.6f}")

    # --------------------------------------------------------
    # 2.  Lengths we want to try – larger lengths give a larger factor 2·n
    # --------------------------------------------------------
    lengths_to_try = [5808, 8000, 10000, 12000, 16000, 20000]
    # ensure they are increasing (larger n → larger 2·n factor)
    lengths_to_try = sorted(set(lengths_to_try))

    # --------------------------------------------------------
    # 3.  Main optimisation loop (mirror‑descent on the soft‑max)
    # --------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_score = best_score

    for n in lengths_to_try:
        if time.time() >= deadline:
            break

        # Build an initial vector for this length
        if best_seq.shape[0] != n:
            a = resample_distribution(best_seq, n)
        else:
            a = best_seq.copy()

        # Pre‑compute FFT size (must be ≥ 3·n‑2 for the gradient step)
        fft_size = 1 << ((3 * n - 2).bit_length())

        # Parameters for the adaptive optimisation
        step_eta = 0.4          # initial multiplicative step size
        eta_min = 1e-6
        eta_max = 5.0
        alpha = 2.0            # start with a smooth approximation
        alpha_max = 200.0
        alpha_mul = 1.001      # growth factor per iteration
        iteration = 0
        max_iters = 5000       # generous upper bound; actual stop is time‑based

        print(f"\n=== Optimising length n={n} ===")
        while time.time() < deadline and iteration < max_iters:
            iteration += 1

            # ---- current convolution and objective ----
            conv = full_conv_fft(a, fft_size)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # update global best
            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best_seq = a.copy()
                print(f"Iter {iteration:5d} – new global best: {cur_score:.6f}")

                # early exit if we have already beaten the target
                if global_best_score < 1.503:
                    print("Target reached – terminating early.")
                    break

            # ---- soft‑max weights ----
            w = softmax_weights(conv, alpha)

            # ---- gradient of the soft‑max ----
            grad = gradient_softmax(a, w, fft_size)

            # ---- multiplicative mirror‑descent step ----
            # clamp exponent to avoid overflow / underflow
            exp_arg = -step_eta * grad
            exp_arg = np.clip(exp_arg, -50.0, 50.0)
            a_candidate = a * np.exp(exp_arg)
            s = a_candidate.sum()
            if s > 0:
                a_candidate /= s
            else:
                # numerical disaster – fall back to the current point
                a_candidate = a.copy()

            # ---- evaluate candidate (exact max) ----
            conv_cand = full_conv_fft(a_candidate, fft_size)
            cand_max = conv_cand.max()
            cand_score = 2.0 * n * cand_max

            # ---- acceptance rule (simple greedy + tiny back‑off) ----
            if cand_score < cur_score:
                a = a_candidate
                # successful step → become slightly more aggressive
                step_eta = min(step_eta * 1.02, eta_max)
            else:
                # unsuccessful → shrink step size
                step_eta = max(step_eta * 0.90, eta_min)

            # ---- slowly tighten the smooth‑max approximation ----
            alpha = min(alpha * alpha_mul, alpha_max)

            # optional early stop per‑length if we see no improvement for
            # a while (not strictly required, but saves time)
            if iteration % 1000 == 0:
                print(f"Iter {iteration:5d}: "
                      f"score≈{cur_score:.6f}, α={alpha:.1f}, η={step_eta:.4f}")

        # after finishing this length, keep the best seen so far
        best_seq = a.copy()   # reuse the last vector as seed for next length

        # stop completely if target already reached
        if global_best_score < 1.503:
            break

    # --------------------------------------------------------
    # 4.  Final official evaluation & return
    # --------------------------------------------------------
    try:
        exact_score = evaluate_sequence(global_best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = global_best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {global_best_score:.6f}")
    print(f"Best length          : {global_best_seq.shape[0]}")
    return global_best_seq.tolist()
```
