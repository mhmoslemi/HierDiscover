"""
Propose a candidate step‑function using the iterative LP refinement
described in the statement.

The algorithm:
1. Start from the known “constant‑convolution’’ coefficients
   a_k = C(2k,k)/4**k (the extremal sequence for the bound M=1).
2. Scale it so that Σ a = sqrt(2·n) (the normalization used in the
   paper).  After scaling the evaluation score equals the current
   maximum of the self‑convolution (M).
3. Repeatedly solve a linear program:
        maximise   Σ g_j
        subject to  (a * g)_k ≤ M   for all k,
                    g_j ≥ 0 .
   The constraints are linear because (a * g)_k = Σ_i a_i·g_{k‑i}.
   The LP finds a vector g that puts as much total mass as possible
   into positions where a is small, while never violating the current
   max‑convolution bound M.
4. Rescale g to the same ℓ1‑norm (sqrt(2·n)), and mix it with the old
   vector a (convex combination).  This reduces the peak of the
   self‑convolution and strictly improves the objective in practice.
5. Iterate until the time budget expires or no improvement is seen.
   The best sequence encountered is returned.

The LP step is cheap (O(n²) constraints) for modest n (≈ 300‑500) and
fits easily into the 1000 s budget, allowing the objective to drop
below the π/2 ≈ 1.5708 bound (target ≈ 1.503).
"""

import time
import numpy as np

# Try to import a fast LP solver; fall back to cvxpy if necessary.
try:
    from scipy.optimize import linprog
    _USE_SCIPY = True
except Exception:  # pragma: no cover
    _USE_SCIPY = False
    import cvxpy as cp


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that (A @ g) = conv(f, g) (full mode).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes f to rows j … j+n‑1
    for j in range(n):
        end = j + n
        if end > rows:
            end = rows
        length = end - j
        A[j:end, j] = f[:length]
    return A


def _solve_lp_max_sum(f: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   subject to  conv(f,g) ≤ bound, g ≥ 0.
    Returns the optimal g (as a NumPy array) or None if the solver fails.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(A.shape[0], bound, dtype=np.float64)

    if _USE_SCIPY:
        # scipy uses a minimisation interface → minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception:  # pragma: no cover
            return None
        if not res.success:
            return None
        return res.x
    else:
        # cvxpy fallback
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= b]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        except Exception:  # pragma: no cover
            return None
        if g.value is None:
            return None
        return np.asarray(g.value, dtype=np.float64)


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function (list of heights) with a low evaluation score.
    The method starts from the optimal constant‑convolution prefix and
    iteratively refines it by linear programming, staying within the given
    time budget.
    """
    # ------------------------------------------------------------------
    # Parameters (tuned for the 1000 s budget)
    # ------------------------------------------------------------------
    # length of the sequence – a few hundred gives a good trade‑off
    n = 400
    # mixing factor for the convex update (0 < t ≤ 1)
    mix_t = 0.5
    # maximum number of LP iterations (will stop earlier if time runs out)
    max_iters = 300

    deadline = time.time() + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – build the constant‑convolution coefficients.
    # ------------------------------------------------------------------
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for k in range(1, n):
        coeffs[k] = coeffs[k - 1] * (2 * k - 1) / (2 * k)

    # Normalise so that Σ a = sqrt(2·n) (the same normalisation used in the
    # reference algorithm).  After normalisation the current objective equals
    # the current max‑convolution value.
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / np.sum(coeffs)
    f = coeffs * scale

    # Initial score (also the current bound M)
    best_score = float("inf")
    best_f = f.copy()
    try:
        # evaluate_sequence is assumed to be provided in the execution environment.
        from __main__ import evaluate_sequence  # type: ignore
        current_score = evaluate_sequence(f.tolist())
        best_score = current_score
        best_f = f.copy()
        print(f"[init] n={n}, initial_score={current_score:.10f}")
    except Exception:  # pragma: no cover
        # If the harness does not inject evaluate_sequence, fall back to the
        # analytic expression (valid for the constant‑convolution seed).
        cum = np.sum(coeffs)
        current_score = 2.0 * n / (cum * cum)
        best_score = current_score
        best_f = f.copy()
        print(f"[init] n={n}, analytic_score={current_score:.10f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # Current max‑convolution (the bound M)
        conv_ff = np.convolve(f, f)
        M = np.max(conv_ff)

        # Solve the LP to find a g with larger ℓ1‑norm but respecting f*g ≤ M
        g = _solve_lp_max_sum(f, M)
        if g is None:
            print("[lp] solver failed – terminating")
            break

        # Rescale g to the target ℓ1‑norm
        sum_g = np.sum(g)
        if sum_g <= 0:
            print("[lp] zero solution – terminating")
            break
        g = g * (target_sum / sum_g)

        # Convex combination with mixing factor t
        f_new = (1.0 - mix_t) * f + mix_t * g

        # Evaluate the new candidate
        try:
            new_score = evaluate_sequence(f_new.tolist())
        except Exception:  # pragma: no cover
            # fallback analytic value (only valid when max(conv)=M)
            new_score = 2.0 * n * M / (np.sum(f_new) ** 2)

        # Keep the best seen so far
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            print(f"[iter {it}] improved score = {new_score:.10f}")

        # Simple acceptance rule: keep the update if it does not worsen the
        # objective; otherwise revert to the previous best (helps stability).
        if new_score <= current_score:
            f = f_new
            current_score = new_score
        else:
            # No improvement – reduce the mixing factor to be more conservative
            mix_t = max(0.05, mix_t * 0.7)

    print(f"[final] best_score={best_score:.10f}, length={n}")

    # Clamp to the evaluator's limits and return as a plain Python list.
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]
    return best_seq
