"""
propose_candidate
-----------------

Search for a probability vector (non‑negative numbers that sum to 1) that
minimises

    2 * n * max(conv_self)

where ``conv_self`` is the self‑convolution of the vector.  The objective
is exactly the value returned by ``evaluate_sequence`` because the
returned sequence is normalised to have unit sum.

The algorithm works as follows

1. Build a few promising initial points:
   * an “arcsine” shape  p_i ∝ 1/√((i+½)(n‑½‑i))  (known to flatten the
     convolution),
   * a few random Dirichlet draws,
   * the optional historic construction ``height_sequence_1``.
2. For every start run a short *sub‑gradient* phase that only looks at
   the current maximal convolution entry.  The gradient of this entry is
   cheap to compute (just a reversed slice of the current vector).  A
   decaying step size together with a projection onto the probability
   simplex drives the mass away from the peak.
3. Refine the result with a few hundred Adam steps on a smooth
   log‑sum‑exp approximation of the max (the same technique used in the
   reference code, but with a smaller smoothing schedule).
4. Finish with a tiny local‑swap hill‑climber that moves a tiny amount
   of probability from one index to another if it improves the objective.
5. Return the best vector found, clipped to the limits enforced by
   ``evaluate_sequence`` (0 ≤ x ≤ 1000).

The whole procedure respects the 1000 s budget, prints progress and
always returns a valid sequence.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0 , Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, reused many times)
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv

def _score(p: np.ndarray, m_fft: int) -> float:
    """Return the evaluation score 2·n·max(conv_self) for a probability vector."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * p.size * np.max(conv))

# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        n = 1153                               # fallback length used in the benchmark

    # FFT size: next power‑of‑two that can contain a convolution of length 2·n‑1
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Helper for timing
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0    # safety margin

    best_score = float("inf")
    best_p = None

    # --------------------------------------------------------------
    # Build a small pool of promising initial vectors
    # --------------------------------------------------------------
    init_vectors = []

    # 1) Arcsine‑shaped distribution  p_i ∝ 1/√((i+½)(n‑½‑i))
    i = np.arange(n, dtype=float)
    eps = 1e-12
    inv = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    p_arcsine = inv / inv.sum()
    init_vectors.append(p_arcsine)

    # 2) A few random Dirichlet draws
    for _ in range(4):
        p_rand = rng.random(n)
        p_rand /= p_rand.sum()
        init_vectors.append(p_rand)

    # 3) Historic construction if it exists
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        seq = np.clip(seq, 0.0, 1000.0)
        s = seq.sum()
        if s > 0:
            init_vectors.append(seq / s)

    # --------------------------------------------------------------
    # Core optimisation loops
    # --------------------------------------------------------------
    for start_id, p0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break

        print(f"=== Start {start_id + 1}/{len(init_vectors)} =====================")
        p = p0.copy()

        # ----------------------------------------------------------
        # (a) Sub‑gradient descent on the *actual* max entry
        # ----------------------------------------------------------
        step = 0.2
        decay = 0.9995
        min_step = 1e-5
        for it in range(5_000):
            if time.time() >= deadline:
                break

            conv = _conv_self(p, m_fft)
            max_val = conv.max()
            max_idx = conv.argmax()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_p = p.copy()
                print(f"  Subgrad iter {it:5d} – new best {best_score:.6f}")
                if best_score <= 1.503:
                    break

            # gradient of the maximal entry (sub‑gradient)
            g = np.zeros_like(p)
            if max_idx < n:
                # overlap size = max_idx + 1
                g[: max_idx + 1] = 2.0 * p[max_idx :: -1]
            else:
                start = max_idx - n + 1
                # elements from start … n‑1 correspond to reversed tail
                g[start:] = 2.0 * p[n - 1 : start - 1 : -1]

            # gradient step + simplex projection
            p = p - step * g
            p = _simplex_projection(p)

            # step‑size decay
            step = max(step * decay, min_step)

        # ----------------------------------------------------------
        # (b) Adam optimisation on a log‑sum‑exp smooth max
        # ----------------------------------------------------------
        # Adam state
        m1 = np.zeros_like(p)
        m2 = np.zeros_like(p)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr = 0.05

        # smoothing schedule (log‑sum‑exp)
        alpha = 1.0
        alpha_mul = 1.001
        alpha_max = 50.0

        for it in range(20_000):
            if time.time() >= deadline:
                break

            # smoothed max and its (soft) gradient
            p_fft = np.fft.rfft(p, n=m_fft)
            conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)                # stability trick
            w = exp_z / np.sum(exp_z)                # soft‑max weights

            # correlation needed for the gradient
            w_rev = w[::-1]
            w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
            corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]

            idx = 2 * n - 2 - np.arange(n)           # mapping to gradient indices
            grad = 2.0 * corr_full[idx]

            # Adam update
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            m_hat = m1 / (1.0 - beta1 ** (it + 1))
            v_hat = m2 / (1.0 - beta2 ** (it + 1))

            step_vec = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            p = p - step_vec
            p = _simplex_projection(p)

            # tighten the smooth approximation
            alpha = min(alpha * alpha_mul, alpha_max)

            # evaluation
            cur_score = 2.0 * n * np.max(conv)
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_p = p.copy()
                print(f"  Adam iter {it:5d} – new best {best_score:.6f}")
                if best_score <= 1.503:
                    break

    # ------------------------------------------------------------------
    # Final tiny local refinement using pairwise swaps
    # ------------------------------------------------------------------
    if best_p is None:
        # Fallback – uniform distribution (never happens in practice)
        best_p = np.full(n, 1.0 / n)
        best_score = _score(best_p, m_fft)

    p = best_p.copy()
    cur_score = _score(p, m_fft)
    swap_attempts = 5_000

    for _ in range(swap_attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # tiny mass transfer
        eps_move = 1e-4 * rng.random()
        if p[i] <= eps_move:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move
        new_score = _score(new_p, m_fft)
        if new_score < cur_score:
            p = new_p
            cur_score = new_score
            best_p = p.copy()
            best_score = cur_score
            print(f"Swap improvement → {best_score:.6f}")
            if best_score <= 1.503:
                break

    print(f"Final best score: {best_score:.6f}")

    # Clip to the range enforced by evaluate_sequence and return as list
    return np.clip(best_p, 0.0, 1000.0).tolist()
