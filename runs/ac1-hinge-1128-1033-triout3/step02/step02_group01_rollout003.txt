assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\,\max_k (a*a)[k]}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

All sequences are normalised to sum = 1, therefore the score is simply  

\[
\text{score}=2\,n\;\max_k (a*a)[k].
\]

The search consists of three stages

1. **Deterministic seeding** – a short sweep over the two classic symmetric “U‑shapes’’  
   \((\min(i,n-1-i)+1)^{-p}\) and \(((i+1)(n-i))^{-p}\) for several lengths and exponents.  
   The best of those seeds (or the external `height_sequence_1` if it exists) becomes the
   starting point.

2. **Frank‑Wolfe / LP step** – for a *moderate* length (≈ 3000) we improve the seed by a
   Frank‑Wolfe‐style update.  
   For a current vector `f` we compute `M = max_k (f*f)[k]` and solve the linear programme  

   \[
   \max_{g\ge0}\;\sum_i g_i \quad\text{s.t.}\quad (f*g)_k \le M,\;k=0,\dots,2n-2 .
   \]

   The solution `g` is the direction that most increases the feasible total mass while
   keeping the convolution bound `M`.  We move a small step towards the normalised
   version of `g`.  Repeating this step reduces the maximal convolution entry.

   The LP is formulated with **CVXPY** (`cp.conv` creates an affine expression because
   `f` is a constant parameter).

3. **Fine‑grained stochastic refinement** – the improved sequence is up‑sampled to a
   larger length (5 k – 12 k) by linear interpolation, re‑normalised and then refined
   with the sub‑gradient / random‑mass‑transfer scheme from the previous state‑of‑the‑art
   implementation.  This stage is cheap (FFT‑based convolutions) and finishes the
   remaining time budget.

The whole routine respects the 1000 s limit, prints progress information and finally
returns the best sequence found as a plain Python list.

```python
# -*- coding: utf-8 -*-
"""
Propose a low‑score step‑function.

The algorithm works in three phases:

1. Deterministic exploration of two symmetric “U‑shaped’’ families to obtain a
   good seed (also uses the external `height_sequence_1` if present).

2. Frank‑Wolfe style improvement on a moderate length (≈ 3000) by solving,
   in each iteration, the linear programme
       maximise   Σ g_i
       subject to (f * g)_k ≤ M   for every k,
   where M = max (f * f).  The LP direction g is combined with the current f
   using a small step size; this quickly lowers the maximal convolution entry.

3. Upsample the improved seed to a larger length (5 k – 12 k) and apply a
   stochastic refinement consisting of random mass‑transfer moves and projected
   sub‑gradient steps (identical to the method that previously achieved 1.532).

All convolutions are evaluated with a single FFT, making the inner loop fast.
The code respects a 1000 s budget, prints intermediate scores and finally
returns the best sequence as a list of floats.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the probability simplex {w >= 0, sum(w) = 1}.
    O(n log n) implementation (Wang & Carreira‑Perpiñán).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on distance to the nearest edge:
        a[i] = (min(i, n-1-i) + 1)^{-p}
    Returns a probability vector (sum = 1).
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the product of distances to both edges:
        a[i] = ((i+1)*(n-i))^{-p}
    Returns a probability vector (sum = 1).
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Output length = 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a * a (full linear convolution)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact score returned by evaluate_sequence for a vector already
    normalised to sum == 1 (no clipping needed).
    """
    n = len(a)
    return 2.0 * n * max_convolution(a)


def solve_lp_direction(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise   sum(g)
        subject to (f * g)_k ≤ M  for all k,
                   g_i ≥ 0.
    Returns the optimal g (as a numpy array) or None if the solver fails.
    """
    n = f.shape[0]
    # CVXPY variable and constant parameter
    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, value=f)
    conv_fg = cp.conv(f_param, g)       # length 2n-1
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=2000, verbose=False)
    except Exception:  # pragma: no cover
        return None
    if g.value is None:
        return None
    return np.asarray(g.value, dtype=float)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from the original length to ``new_n``.
    The result is re‑normalised to sum == 1.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total > 0:
        new_seq /= total
    else:
        new_seq = np.full(new_n, 1.0 / new_n)
    return new_seq

        # ----- random mass‑transfer move -----
        if np.random.rand() < 0.5:
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] == 0.0:
                continue
            delta = current[i] * np.random.rand() * 0.5
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # ----- projected sub‑gradient move -----
            conv = full_convolution_fft(current)
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-15
            max_idx = np.where(conv >= max_val - eps)[0]

            grad = np.zeros_like(current)
            for k in max_idx:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                i_vals = np.arange(lo, hi + 1)
                j_vals = k - i_vals
                grad[i_vals] += 2.0 * current[j_vals]

            cand = current - step_sz * grad
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)

        cand_score = score_from_vector(cand)

        # simulated‑annealing acceptance
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < refined_score:
                refined_seq = cand.copy()
                refined_score = cand_score
                last_imp = it
                print(f"Iter {it:7d}: new best (refine) = {cand_score:.6f}")

        # annealing
        temperature *= cooling_rate
        step_sz *= step_decay

        # re‑heat if stuck
        if it - last_imp > 60000:
            bump = np.random.rand(n) * 1e-4
            current = simplex_projection(current + bump)
            current_score = score_from_vector(current)
            last_imp = it
            temperature = 1.0
            print(f"Iter {it:7d}: re‑heat (refine)")

        if time.time() > deadline:
            break

    print("\n=== Search finished ===")
    print(f"Best score achieved: {refined_score:.6f} (length {len(refined_seq)})")
    # Return raw list – the external evaluator will clip / normalise again.
    return refined_seq.tolist()
```
