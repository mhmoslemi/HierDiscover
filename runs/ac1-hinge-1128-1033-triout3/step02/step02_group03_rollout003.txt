  1. Solve a linear program that, for the current distribution p, finds the

  3. Refine the result with an exponentiated‑gradient sub‑gradient step.
Multiple random restarts are performed until the time budget expires.
The best sequence encountered is returned.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse


def _conv_max(p: np.ndarray) -> float:
    """Return max(p * p) convolution."""
    # using FFT for speed (optional)
    # conv = np.convolve(p, p)   # O(n^2) but fine for n~1k
    conv = np.convolve(p, p)
    return conv.max()


def _score(p: np.ndarray, n: int) -> float:
    """Objective value = 2·n·max(conv(p,p))."""
    return 2.0 * n * np.max(np.convolve(p, p))


def _constraint_matrix(p: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the (2n‑1)×n matrix A such that (A @ g)[k] = (p * g)[k].
    Each column i contains p[i] on rows i … i+n‑1, zero elsewhere.
    """
    n = p.shape[0]
    # rows: for each column i we need rows i, i+1, …, i+n-1
    # we can generate by broadcasting
    rows = (np.arange(n)[:, None] + np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(p, n)
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_max_sum(p: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise   1ᵀ g   subject to   (p * g) ≤ M   and   g ≥ 0.
    Returns the optimal g (not normalised).  If the LP fails, None is returned.
    """
    n = p.shape[0]
    A = _constraint_matrix(p)
    b = np.full(2 * n - 1, M + 1e-12)   # tiny slack for numerical safety
    c = -np.ones(n)                     # minimise -∑g  → maximise ∑g

        options={"presolve": True, "time_limit": 5.0},
    )
    if not res.success:
        return None
    return res.x

    Perform a short run of exponentiated‑gradient sub‑gradient steps starting
    from p.  The loop stops when the deadline is reached.
    """

    eta = 0.5
    eta_decay = 0.9995
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution and locate its maximum
        conv = np.convolve(p, p)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max entry (as in the reference code)
        grad = np.zeros_like(p)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad[idx] = 2.0 * p[k_max - idx]

        s = p.sum()

            noise = rng.normal(scale=0.001, size=n)
            p = np.maximum(p + noise, 0.0)
            p = p / p.sum()

        eta *= eta_decay
    return p

M(p) = 2·n·max(conv(p,p))

The search consists of many short **random restarts**.  
Each restart performs two complementary optimisation phases:

1. **LP‑guided mass transfer** –  
   For the current `p` we compute `M = max(conv(p,p))`.  
   We then solve the linear programme  

   ```
   maximise   1ᵀ g
   subject to (p * g)(k) ≤ M   for k = 0 … 2n‑2
              g ≥ 0
   ```

   (`*` is ordinary convolution).  The solution `g` tells where additional mass can be placed without increasing the current peak.  
   After normalising `g` to sum = 1 we move part of the probability mass from `p` towards `g` (mixing coefficients 0.25, 0.5, 0.75, 1.0 are tried and the best is kept).  
   Repeating this step strictly reduces the objective or leaves it unchanged.

2. **Exponentiated‑gradient fine‑tuning** –  
   When the LP phase stalls we switch to a classic mirror‑descent step on the simplex.  
   The (sub‑)gradient of the *true* max‑convolution is obtained from the active lag `k_max`; an exponentiated update `p ← p·exp(-η·grad)` is applied and the vector is renormalised.  
   Small Gaussian jitter is added every few hundred steps to escape flat plateaus.

The whole process is repeated with different random initialisations (or a jittered version of the current best) until the time budget (≈ 1000 s) is exhausted.  
The best vector found is returned as a plain Python list – it already satisfies the security checks of `evaluate_sequence`.

def _exponentiated_gradient(p: np.ndarray, deadline: float) -> np.ndarray:
    """
    A short run of exponentiated‑gradient sub‑gradient steps.
    Stops when ``time.time() >= deadline``.
    """
    n = p.shape[0]
    rng = np.random.default_rng()
    eta = 0.5
    eta_decay = 0.9995
    it = 0
    while time.time() < deadline:
        it += 1
        conv = np.convolve(p, p)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) at the active index
        grad = np.zeros_like(p)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad[idx] = 2.0 * p[k_max - idx]

        # multiplicative (mirror‑descent) update on the simplex
        p = p * np.exp(-eta * grad)
        s = p.sum()
        if s == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p = p / s

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            noise = rng.normal(scale=0.001, size=n)
            p = np.maximum(p + noise, 0.0)
            p = p / p.sum()

        eta *= eta_decay
    return p


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The algorithm performs many brief random restarts; each restart
    first applies a few LP‑driven mass‑transfer steps, then refines the result
    with exponentiated‑gradient sub‑gradient steps.  The best sequence seen
    before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0                     # small safety margin

    # ------------------------------------------------------------------
    # 1. Problem size and initial distribution
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                           # type: ignore[name‑defined]
        base = np.array(height_sequence_1, dtype=float)
        base = np.clip(base, 0.0, 1000.0)
        p0 = base / base.sum() if base.sum() > 0 else rng.random(n)
        p0 = p0 / p0.sum()
    except Exception:
        n = 1153
        p0 = rng.random(n)
        p0 = p0 / p0.sum()

    best_p = p0.copy()
    best_score = _score(best_p)
    print(f"[Init] n={n} – initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Main search – repeated short restarts
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 10.0:
        restart += 1

        # start from the current best with a small jitter, or a fresh random point
        if rng.random() < 0.7:
            p = best_p + rng.normal(scale=0.005, size=n)
            p = np.clip(p, 0.0, None)
            p = p / p.sum()
        else:
            p = rng.random(n)
            p = p / p.sum()

        cur_score = _score(p)
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[Restart {restart}] New best from init {best_score:.6f}")

        # --------------------------------------------------------------
        # Phase A – a few LP‑based updates
        # --------------------------------------------------------------
        for lp_step in range(12):                               # at most 12 LP updates
            if time.time() >= deadline - 5.0:
                break
            M = _conv_max(p)
            g = _lp_max_sum(p, M)
            if g is None:
                break
            sum_g = g.sum()
            if sum_g <= 0:
                break
            g_scaled = g / sum_g

            # try several mixing coefficients and keep the best
            best_local = p
            best_local_score = cur_score
            for t in (0.25, 0.5, 0.75, 1.0):
                cand = (1.0 - t) * p + t * g_scaled
                cand_score = _score(cand)
                if cand_score < best_local_score:
                    best_local_score = cand_score
                    best_local = cand

            if best_local_score + 1e-12 < cur_score:            # accept improvement
                p = best_local
                cur_score = best_local_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_p = p.copy()
                    print(f"[Restart {restart} – LP] step {lp_step + 1} score {best_score:.6f}")

            # early termination if we have already hit the target
            if best_score <= 1.503:
                break

        if best_score <= 1.503:
            break

        # --------------------------------------------------------------
        # Phase B – exponentiated‑gradient fine‑tuning
        # --------------------------------------------------------------
        # use roughly 60 % of the time that is left for this restart
        remaining = deadline - time.time()
        phase_deadline = time.time() + 0.6 * remaining
        p = _exponentiated_gradient(p, phase_deadline)
        cur_score = _score(p)
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[Restart {restart} – EG] score {best_score:.6f}")

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3. Final cheap random sampling (uses any leftover time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        p_rand = rng.random(n)
        p_rand = p_rand / p_rand.sum()
        sc = _score(p_rand)
        if sc < best_score:
            best_score = sc
            best_p = p_rand.copy()
            print(f"[Random] new best {best_score:.6f}")
        if time.time() + 0.01 > deadline:
            break

    # ------------------------------------------------------------------
    # 4. Return the best sequence found
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Best score achieved: {best_score:.6f}")
    return result
