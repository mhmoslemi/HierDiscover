"""
Propose a step‑function (non‑negative vector) that minimizes
the evaluation metric used in the challenge.

The algorithm consists of three phases

1. **Warm‑start generation** – a fast parametric sweep over several
   families of symmetric “U‑shaped’’ distributions:
      * power‑law on the distance to the nearest edge,
      * product‑power on the product of the distances to both edges,
      * a convex combination of the two.
   For each candidate the maximal entry of the self‑convolution is
   computed with a real FFT (O(n log n)).  The best candidate becomes
   the warm start.

2. **Local stochastic refinement** – starting from the warm start a
   simulated‑annealing loop repeatedly applies three kinds of moves:
      * random transfer of a fraction of mass between two positions,
      * a gradient‑guided transfer from the index that contributes most
        to the current maximal convolution entry to the one that contributes
        least,
      * a multiplicative (mirror‑descent) update that follows the
        sub‑gradient of the maximal convolution entry.
   Moves are accepted if they improve the score or with a probability
   that depends on a slowly decreasing temperature, allowing escape
   from shallow local minima.  The best vector seen so far is kept.

3. **Exact evaluation** – after the time budget the best vector is
   evaluated once with the official ``evaluate_sequence`` routine.

The whole routine respects the 1000 s budget, prints occasional progress
information and returns a plain Python list of floats ready for the
scorer.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """
    Fast (real FFT) computation of max entry of the linear self‑convolution
    of a non‑negative vector a.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def shape_power_law(n: int, p: float, offset: float) -> np.ndarray:
    """
    Symmetric power‑law: a[i] = (dist_to_edge + offset)^{-p}
    where dist_to_edge = min(i, n-1-i).  Normalised to sum = 1.
    """
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = (d + offset) ** (-p)
    a += 1e-30                     # avoid exact zeros
    a /= a.sum()
    return a


def shape_product_power(n: int, p: float) -> np.ndarray:
    """
    Symmetric product‑power: a[i] = ((i+0.5)*(n-i-0.5))^{-p}.
    This places more mass near the edges while staying smooth.
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    a += 1e-30
    a /= a.sum()
    return a


def shape_mixed(n: int, p1: float, p2: float, lam: float) -> np.ndarray:
    """
    Convex combination of the two families above:
        lam * power_law(p1, offset=1) + (1‑lam) * product_power(p2)
    """
    a1 = shape_power_law(n, p1, offset=1.0)
    a2 = shape_product_power(n, p2)
    a = lam * a1 + (1.0 - lam) * a2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def fast_score(vec: np.ndarray) -> float:
    """
    Approximate score used during the search:
         2 * n * max_conv_fft(vec)
    The vector is assumed to be normalised (sum == 1).
    """
    n = vec.shape[0]
    return 2.0 * n * max_conv_fft(vec)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------

    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns a plain Python list of floats.
    """

    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 0.  Use an externally supplied sequence if it exists.
    # --------------------------------------------------------------

        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external = external / s
            score_ext = fast_score(external)
            if score_ext < best_score:
                best_score = score_ext
                best_seq = external.copy()
                print(f"Seeded from height_sequence_1: score={score_ext:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 1.  Deterministic parametric sweep – generate a strong warm start.
    # --------------------------------------------------------------
    lengths = [3000, 5000, 8000, 12000, 20000]
    p_vals = np.linspace(0.2, 2.0, 19)          # 0.2 … 2.0
    offsets = [0.2, 0.5, 1.0, 2.0]            # offset for power‑law
    lam_vals = [0.0, 0.25, 0.5, 0.75, 1.0]    # mixing coefficient

                    best_seq = a.copy()
                    print(f"New best (pow): n={n:5d}, p={p:4.2f}, off={off}, score={sc:.6f}")

        # ----- product‑power family -----
        for p in p_vals:
            a = shape_product_power(n, p)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best (prod): n={n:5d}, p={p:4.2f}, score={sc:.6f}")

        # ----- mixed family (reduced grid) -----
        # use a coarser grid for the two exponents to keep the sweep fast
        for p1 in p_vals[::3]:
            for p2 in p_vals[::3]:
                for lam in lam_vals:
                    a = shape_mixed(n, p1, p2, lam)
                    sc = fast_score(a)
                    if sc < best_score:
                        best_score = sc
                        best_seq = a.copy()
                        print(f"New best (mix): n={n:5d}, p1={p1:4.2f}, p2={p2:4.2f}, lam={lam}, score={sc:.6f}")

        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2.  If the sweep did not find anything (unlikely) fall back to random.
    # --------------------------------------------------------------
    if best_seq is None:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = fast_score(a)
        print(f"Fallback random init: n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Stochastic refinement on the simplex.
    # --------------------------------------------------------------
    current = best_seq.copy()
    n = len(current)
    current_score = best_score

    temperature = 1.0       # initial temperature for simulated annealing
    cooling_rate = 0.9997   # geometric cooling (slow – many iterations)
    transfer_frac = 0.5     # maximum fraction of source mass to move
    eta = 0.20              # base step size for mirror‑descent update

    it = 0
    print("\n=== Stochastic refinement ===")
    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # Choose a move type
        # ----------------------------------------------------------
        move_type = np.random.rand()
        if move_type < 0.5:
            # ---- Random mass transfer between two distinct positions ----
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] <= 0.0:
                continue
            delta = current[i] * np.random.rand() * transfer_frac
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            cand /= cand.sum()

        elif move_type < 0.8:
            # ---- Gradient‑guided deterministic transfer ----
            # Identify the convolution index where the maximum occurs
            size = 1 << ((2 * n - 1).bit_length())
            A = np.fft.rfft(current, n=size)
            conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
            kmax = int(conv.argmax())

            # Sub‑gradient of the maximal entry
            grad = np.zeros_like(current)
            i_start = max(0, kmax - (n - 1))
            i_end = min(n - 1, kmax)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = kmax - idx_i
            grad[idx_i] = 2.0 * current[idx_j]

            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))
            if current[i_max] <= 0.0:
                continue
            delta = current[i_max] * np.random.rand() * transfer_frac
            cand = current.copy()
            cand[i_max] -= delta
            cand[i_min] += delta
            cand /= cand.sum()

        else:
            # ---- Multiplicative mirror‑descent step ----
            size = 1 << ((2 * n - 1).bit_length())
            A = np.fft.rfft(current, n=size)
            conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
            kmax = int(conv.argmax())

            grad = np.zeros_like(current)
            i_start = max(0, kmax - (n - 1))
            i_end = min(n - 1, kmax)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = kmax - idx_i
            grad[idx_i] = 2.0 * current[idx_j]

            eta_t = eta * temperature
            expo = -eta_t * grad
            expo = np.clip(expo, -50.0, 0.0)       # avoid overflow
            cand = current * np.exp(expo)
            s = cand.sum()
            if s > 0.0:
                cand /= s
            else:
                cand = current.copy()

        # ----------------------------------------------------------
        # Fast score evaluation
        # ----------------------------------------------------------
        cand_score = fast_score(cand)

        # ----------------------------------------------------------
        # Simulated‑annealing acceptance test
        # ----------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Iter {it:7d}: new global best = {best_score:.6f}")

        # ----------------------------------------------------------
        # Anneal temperature
        # ----------------------------------------------------------
        temperature = max(temperature * cooling_rate, 1e-7)

    # --------------------------------------------------------------
    # 4.  Final exact evaluation (once) and report
    # --------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Exact evaluation failed: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {best_score:.6f}")
    print(f"Best exact score       : {exact_score:.6f}")
    print(f"Best length            : {len(best_seq)}")
    return best_seq.tolist()
