"""
Propose a candidate step function with a low evaluation score.

The algorithm builds a few simple initial probability vectors (uniform,
linearly decreasing, random Dirichlet) and then refines the best one by
sub‑gradient descent on the ∞‑norm of the self‑convolution.
The gradient of the worst convolution entry is exact (no smoothing),
the iterate is projected onto the probability simplex after each step.
Occasional jitter and hard restarts help to escape plateaus, and a final
pairwise‑swap phase performs a cheap local refinement.

The whole search respects a time budget (default 1000 s) and returns the
best sequence found.  All numbers are clipped to the range accepted by
``evaluate_sequence`` before returning.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex {x | x_i ≥ 0 , Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------
def _make_conv_fft_len(n: int) -> int:
    """
    Return the next power‑of‑two size sufficient for a linear convolution
    of two vectors of length n (i.e. length 2*n‑1).
    """
    return 1 << ((2 * n - 1).bit_length())


def _conv_full(p: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of p with itself, length 2*n‑1."""
    p_fft = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(p_fft * p_fft, n=fft_len)[: 2 * p.size - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution entry.
# ----------------------------------------------------------------------
def _max_conv_grad(p: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution entry at index k_max.
    For a given shift k, the gradient component i is 2 * p_{k-i} if
    k-i lies inside the vector, otherwise 0.
    """
    n = p.size
    grad = np.zeros_like(p)

    # valid i satisfy 0 ≤ i < n and 0 ≤ k_max-i < n
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)          # inclusive
    if i_start > i_end:                # no overlap (should not happen)
        return grad

    i_range = np.arange(i_start, i_end + 1, dtype=int)
    j_range = k_max - i_range
    grad[i_range] = 2.0 * p[j_range]   # 2 * p_{k-i}
    return grad

# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------

    Search for a low‑objective step‑function using sub‑gradient descent.
    The routine respects a wall‑clock budget (default 1000 seconds) and
    returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size and an initial feasible vector (probability simplex)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name‑defined]

    except Exception:

    # FFT length for convolution (shared across the whole run)
    fft_len = _make_conv_fft_len(n)

    # ------------------------------------------------------------------
    # 2. Helper to evaluate a probability vector (sum = 1)
    # ------------------------------------------------------------------
    def _score(p_vec: np.ndarray) -> float:
        """Return the value 2·n·max(conv(p,p))."""
        conv = _conv_full(p_vec, fft_len)
        return float(2.0 * n * np.max(conv))

    # ------------------------------------------------------------------
    # 3. Build a small portfolio of start points
    # ------------------------------------------------------------------
    candidates = []

    # a) uniform
    candidates.append(p0.copy())

    # b) linearly decreasing (p_i ∝ n-i) – often a decent heuristic
    linear = np.arange(n, 0, -1, dtype=float)

        print(f"Starter {idx:02d} – score: {sc:.6f}")
        if sc < best_score:

    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0   # safety margin

        step = base_lr / np.sqrt(iteration)

            p_new = _simplex_projection(p_new)

            step *= 0.5                     # shrink step and retry

            cur_score = new_score
            if new_score < best_score:

                best_p = p.copy()
                no_improve = 0

            else:

        else:

            p = _simplex_projection(p + noise)
            no_improve += 1

        # hard restart if we have been stuck for a while
        if no_improve >= restart_interval:

            p = np.clip(p, 0.0, None)
            p = p / p.sum()
            no_improve = 0
            iteration = 0
            print(f"--- restart with perturbed best (score {best_score:.6f}) ---")

        # occasional larger jitter to explore new basins
        if iteration % 5000 == 0:

            p = _simplex_projection(p + jitter)

        if best_score <= 1.503:

            break

    # ------------------------------------------------------------------
    p = best_p.copy()
    cur_score = _score(p)
    swap_trials = 2000
    for _ in range(swap_trials):

        if new_score < cur_score:
            p = p_new
            cur_score = new_score
            best_p = p.copy()
            best_score = new_score
            print(f"Swap improvement → {best_score:.6f}")

    print(f"Final best score: {best_score:.6f}")

    # Return a plain list of floats, clipped to the limits enforced by evaluate_sequence
    return np.clip(best_p, 0.0, 1000.0).tolist()
