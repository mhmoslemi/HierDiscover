def solve_lp(f, M):
    n = len(f)
    rows = []
    b = [M] * (2*n - 1)
    for k in range(2*n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # i indexes f; j = k - i indexes g
        for i in range(i_start, i_end+1):
            j = k - i
            row[j] = f[i]    # coefficient for g_j
        rows.append(row)
    A_ub = np.array(rows) # shape (2n-1,n)
    c = -np.ones(n) # maximize sum, but linprog minimizes, so negative

    res = linprog(c, A_ub=A_ub, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback handling
        return None
    return res.x

```
from scipy.optimize import linprog
def solve_g(f, M):
    n = len(f)
    A = []
    b = np.full(2 * n - 1, M)
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f[i]
        A.append(row)
    A = np.asarray(A)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    return res.x
```

```
def solve_for_g(f, M):
    n = len(f)
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vectorize: indices i = i_start..i_end; j = k - i
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        row[j_vals] = f[i_vals]
        rows.append(row)
    A = np.vstack(rows)
    # Linprog expects A_ub * x <= b_ub
    b = np.full(2*n - 1, M)
    c = -np.ones(n)   # maximize sum -> minimize negative sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError('LP failed')
    return res.x
```

assistantfinal**Solution Overview**

The score of a step function  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}
\]

is the ratio *max‑convolution* / *average‑convolution*.  
The theoretical lower bound of this ratio is 1 (all convolution
coefficients would be equal).  The infinite central‑binomial series

\[
b_k=\frac{\binom{2k}{k}}{4^k}
\]

has the property that its *self‑convolution* is exactly 1 for every
coefficient, and a prefix of length \(n\) therefore attains the
asymptotic value \(\pi/2\approx1.5708\).  It is optimal **under the
additional restriction** that the convolution never exceeds 1, but a
slightly larger maximum together with a larger \(\sum a_i\) can give a
smaller ratio.

The algorithm below builds on the linear‑programming (LP) idea that
appeared in the reference solution:

```
given f, let M = max(f*f)
find g  (non‑negative)   max   Σ g_i
           s.t.           (f * g)_k ≤ M   for all k
```

The constraints are linear because `f` is fixed, therefore a plain LP
solves for the “most generous’’ `g`.  After solving the LP we move a
small step towards `g`.  For a sufficiently small step the new
self‑convolution never exceeds the old bound `M`; consequently the
objective value never grows.  Repeating the procedure repeatedly
flattens the convolution, reduces the maximal entry and improves the
score.

The whole search consists of three phases

1. **Initial candidate** – a long central‑binomial prefix (the
   best known analytic construction).
2. **LP‑based refinement** – repeated solving of a small LP (length
   ≈ 300) with a back‑tracking line search on the step size.
3. **Light‑weight stochastic hill‑climb** – random Gaussian tweaks to
   escape possible local minima.

All work is performed inside the allowed 1000 s budget.  If SciPy
is not available the code falls back to the pure random hill‑climb
used in the previous run.

```python
import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions (must be top level – no closures)
# ----------------------------------------------------------------------


def _binomial_prefix(length: int) -> np.ndarray:
    """
    Central‑binomial coefficients:
        b[0]=1,
        b[k+1] = b[k] * (2*k+1) / (2*(k+1)).
    """
    b = np.empty(length, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, length):
        cur = cur * (2 * k - 1) / (2 * k)          # (2k‑1)/(2k) * previous
        b[k] = cur
    return b


def _max_self_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using full convolution."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Linear programme:
        maximise    Σ g_i
        subject to  (f * g)_k ≤ M   for all k = 0 … 2n‑2
                    g_i ≥ 0
    Returns the optimal g (numpy array of length n).

    The constraint matrix is Toeplitz:   (f * g)_k = Σ_i f_i g_{k‑i}.
    """
    try:
        from scipy.optimize import linprog
    except Exception as exc:                 # SciPy not available?
        raise RuntimeError("SciPy optimisation required for LP step") from exc

    n = f.shape[0]
    rows = []
    for k in range(2 * n - 1):
        # row for constraint (f * g)_k ≤ M
        row = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over the indices of f that appear in convolution term k
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals                     # corresponding indices of g
        row[j_vals] = f[i_vals]
        rows.append(row)
    A = np.vstack(rows)                         # shape (2n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)           # maximise sum → minimise –sum
    bounds = [(0, None)] * n                    # g_i ≥ 0

    # ‘highs’ is the modern, robust solver
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     length_lp: int = 300,
                     max_iter: int = 200,
                     t_initial: float = 0.2,
                     **kwargs) -> List[float]:
    """
    Produce a sequence of non‑negative heights that (heuristically) minimises
    ``evaluate_sequence``.

    The algorithm works in three stages:
    1.  Start with a long central‑binomial prefix (the analytic optimum for
        the special case max‑convolution = 1).
    2.  Repeatedly solve a linear programme that finds a direction ``g``
        increasing Σg while keeping the mixed convolution (f·g) bounded.
        A tiny convex step towards ``g`` never raises the maximal
        self‑convolution, therefore the score monotonically improves.
    3.  A few seconds of simple stochastic hill‑climbing are performed to
        polish the final result.
    """
    import random
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 5.0      # a small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – analytic starting point (central binomial)
    # ------------------------------------------------------------------
    # we keep a relatively long prefix only for the *analytic* score.
    # The optimisation itself works on a much shorter vector ``length_lp``.
    long_len = max(15000, length_lp * 5)         # ensure the analytic score is fine
    binom_full = _binomial_prefix(long_len)
    # pick a length that gives the best analytic score (always the longest)
    best_len = long_len
    candidate_full = binom_full[:best_len].tolist()
    try:
        best_real = evaluate_sequence(candidate_full)
        print(f"[init] analytic length={best_len}, real score={best_real:.9f}")
    except Exception:
        best_real = float('inf')
        print("[init] evaluation failed – will continue with LP refinement")

    # ------------------------------------------------------------------
    # Phase 2 – LP based refinement on a moderate size vector
    # ------------------------------------------------------------------
    n = length_lp                               # length of the vector used in the LP
    f = _binomial_prefix(n).astype(np.float64)  # initial f (central‑binomial)

    # keep the best sequence seen so far (full‑length version will be built
    # at the end from the refined short vector)
    best_score = best_real
    best_seq_lp = f.copy()

    t = t_initial                               # current step size
    t_min = 1e-5
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        M = _max_self_convolution(f)
        # --------------------------------------------------------------
        #  solve LP: maximise Σ g_i   subject to  (f * g)_k ≤ M
        # --------------------------------------------------------------
        try:
            g = _solve_lp_max_sum(f, M)
        except Exception as e:
            print(f"[lp] iteration {iteration}: LP failed – {e}")
            break

        # --------------------------------------------------------------
        #  line‑search on the step size to keep the max‑convolution ≤ M
        # --------------------------------------------------------------
        t_try = t
        while t_try > t_min:
            f_new = (1.0 - t_try) * f + t_try * g
            M_new = _max_self_convolution(f_new)
            if M_new <= M * (1.0 + 1e-12):      # non‑increase (tolerance)
                break
            t_try *= 0.5

        if t_try <= t_min:
            # couldn't make a step without raising the bound – shrink t
            t = max(t_min, t * 0.5)
            continue

        # accept the step
        f = f_new
        t = min(t * 1.1, 0.5)                  # slowly allow larger steps later

        # evaluate the current candidate (full length = best_len)
        # we embed the short vector into the long one by padding zeros at the end.
        full_candidate = np.zeros(best_len, dtype=np.float64)
        full_candidate[:n] = f
        try:
            cur_score = evaluate_sequence(full_candidate.tolist())
        except Exception:
            cur_score = float('inf')

        if cur_score < best_score:
            best_score = cur_score
            best_seq_lp = f.copy()
            print(f"[improve] it={iteration}, score={best_score:.9f}, "
                  f"M={M_new:.6f}, step={t_try:.5f}")

        # optional early stop if progress stalls
        if iteration % 20 == 0 and time.time() > deadline - 30:
            # use remaining time for stochastic refinement
            break

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic hill‑climb (few seconds)
    # ------------------------------------------------------------------
    cur_seq = best_seq_lp.tolist()
    cur_score = best_score
    sigma = 0.02
    no_improve = 0
    max_no_improve = 3000

    while time.time() < deadline and no_improve < max_no_improve:
        idx = random.randint(0, len(cur_seq) - 1)
        old = cur_seq[idx]
        delta = random.gauss(0.0, sigma) * max(old, 1e-12)
        new_val = max(0.0, old + delta)

        cand = cur_seq[:]
        cand[idx] = new_val

        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue

        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_seq_lp = cand[:]
                print(f"[hill] new best {best_score:.9f}")
        else:
            no_improve += 1

    # Build the final (full‑length) sequence: the refined short vector placed at
    # the beginning, zeros elsewhere – this never harms the score.
    final_seq = np.zeros(best_len, dtype=np.float64)
    final_seq[:len(best_seq_lp)] = best_seq_lp
    # ensure the output meets the evaluator’s safety checks
    final_seq = [float(max(0.0, x)) for x in final_seq.tolist()]

    print(f"[final] length={len(final_seq)}, best score={best_score:.9f}")
    return final_seq
```
