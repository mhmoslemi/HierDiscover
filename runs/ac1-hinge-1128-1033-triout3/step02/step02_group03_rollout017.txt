"""
Propose a step‑function (non‑negative heights) that minimises

        M = 2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm combines a few simple ideas that were not used together
before:

1. **Low‑dimensional families** – exponential, two‑sided exponential,
   and simple two‑level step functions are scanned on a grid.  This
   cheap pre‑search often discovers a very good starting point.

2. **Adam‑accelerated smoothed max descent** – the true max in the
   objective is replaced by a soft‑max with a temperature α that is
   increased gradually.  The gradient of the smooth objective is obtained
   analytically (a single correlation) and the iterate is projected onto
   the probability simplex.  Adam’s adaptive step‑sizes give fast,
   stable convergence.

3. **Exact max sub‑gradient (mirror descent)** – after α becomes large
   the soft‑max is essentially the true max.  An exponentiated‑gradient
   (mirror‑descent) update using the true sub‑gradient is performed.
   Occasional Gaussian jitter and random restarts keep the search from
   stagnating.

All the work fits into the allocated time budget; the best sequence found
is returned as a plain Python list that already satisfies the checks in
``evaluate_sequence``.
"""

import time
import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0 , Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(p: np.ndarray) -> float:
    """
    Return the objective 2·n·max(conv(p,p)) for a probability vector p.
    (sum(p) = 1, therefore the denominator of the original formula is 1.)
    """
    n = p.shape[0]
    conv = np.convolve(p, p)        # length 2·n‑1
    return 2.0 * n * np.max(conv)


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """soft‑max over the convolution entries, temperature = 1/α."""
    z = alpha * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)        # stable exponentiation
    return exp_z / np.sum(exp_z)


def _softmax_gradient(p: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed objective 2·n·∑_k w_k·conv_k(p,p)
    with respect to p (n‑dimensional).
    """
    n = p.shape[0]
    w_rev = w[::-1]
    # correlation of p with reversed soft‑max weights
    corr = np.convolve(p, w_rev)                     # length 3·n‑2
    idx = 2 * n - 2 - np.arange(n)                  # map to gradient indices
    return 2.0 * corr[idx]


def _max_subgradient(p: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(p,p)) at the active index.
    """
    n = p.shape[0]
    conv = np.convolve(p, p)
    k_max = int(np.argmax(conv))

    grad = np.zeros_like(p)
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    # contribution of the active window
    if i_start <= i_end:
        idx_range = np.arange(i_start, i_end + 1)
        grad[i_start:i_end + 1] = 2.0 * p[k_max - idx_range]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the evaluation
    function.  The function runs for at most ``budget_s`` seconds and returns
    the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose problem size (use the historic length if it exists)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                 # type: ignore[name‑defined]
        have_prev = True
    except Exception:
        n = 1153
        have_prev = False

    # ------------------------------------------------------------------
    # 2. Helper to turn a raw vector into a probability vector
    # ------------------------------------------------------------------
    def _normalise(v):
        total = v.sum()
        if total == 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / total

    # ------------------------------------------------------------------
    # 3. Initialise a collection of promising starting points
    # ------------------------------------------------------------------
    starters = []

    # a) reuse the historic construction if it exists
    if have_prev:
        seq = np.array(height_sequence_1, dtype=float)
        seq = np.clip(seq, 0.0, 1000.0)
        p = _normalise(seq)
        starters.append(p)

    # b) random Dirichlet draws
    for _ in range(4):
        p = rng.random(n)
        starters.append(_normalise(p))

    # c) monotone exponential families
    alphas_exp = np.logspace(-4, -1, 30)   # α = exp(-rate·i)
    for a in alphas_exp:
        p = np.exp(-a * np.arange(n))
        starters.append(_normalise(p))

    # d) two‑sided exponential (high at the ends, low in the middle)
    for a in alphas_exp:
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        p = np.exp(-a * dist)
        starters.append(_normalise(p))

    # e) simple 2‑level step functions (high / low)
    for _ in range(6):
        m = rng.integers(1, n - 1)            # split point
        low = rng.random()                    # low level in [0,1)
        p = np.empty(n, dtype=float)
        p[:m] = 1.0
        p[m:] = low
        starters.append(_normalise(p))

    # ------------------------------------------------------------------
    # 4. Global best tracking
    # ------------------------------------------------------------------
    best_score = np.inf
    best_p = None

    # ------------------------------------------------------------------
    # 5. Main optimisation loop over all initialisers
    # ------------------------------------------------------------------
    for init_id, p0 in enumerate(starters, 1):
        if time.time() >= deadline:
            break

        p = p0.copy()
        n = p.shape[0]

        # ---- Adam state for the smoothed‑max phase ----
        m_adam = np.zeros_like(p)
        v_adam = np.zeros_like(p)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr_adam = 0.2
        alpha = 4.0                # start temperature for the soft‑max
        alpha_mul = 1.003          # how fast we tighten the approximation

        # Phase‑A: smooth max descent (Adam)
        it = 0
        while time.time() < deadline:
            it += 1
            conv = np.convolve(p, p)
            w = _softmax_weights(conv, alpha)
            grad = _softmax_gradient(p, w)

            # Adam update (on the Euclidean space) then project back to simplex
            m_adam = beta1 * m_adam + (1.0 - beta1) * grad
            v_adam = beta2 * v_adam + (1.0 - beta2) * (grad * grad)
            m_hat = m_adam / (1.0 - beta1 ** it)
            v_hat = v_adam / (1.0 - beta2 ** it)
            step = lr_adam * m_hat / (np.sqrt(v_hat) + eps)

            p_next = p - step
            p_next = _simplex_projection(p_next)

            # occasional jitter to break symmetry
            if it % 500 == 0:
                jitter = rng.normal(scale=1e-3, size=n)
                p_next = _simplex_projection(p_next + jitter)

            p = p_next
            alpha *= alpha_mul

            cur_score = _objective(p)
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                print(f"[Init {init_id:02d}] Phase‑A iter {it:06d} – new best: {best_score:.6f}")

            # stop early if the smooth approximation is already tight
            if alpha > 1e4:
                break

        # ---- Phase‑B: exact max sub‑gradient with exponentiated updates ----
        eta = 0.5
        eta_decay = 0.9995
        no_improve = 0
        sub_it = 0
        while time.time() < deadline:
            sub_it += 1
            grad = _max_subgradient(p)

            # exponentiated (mirror‑descent) step
            p = p * np.exp(-eta * grad)
            s = p.sum()
            if s == 0.0:
                p = np.full(n, 1.0 / n)
            else:
                p = p / s

            # jitter every few hundred steps
            if sub_it % 400 == 0:
                jitter = rng.normal(scale=1e-3, size=n)
                p = _simplex_projection(p + jitter)

            eta *= eta_decay

            cur_score = _objective(p)
            if cur_score + 1e-12 < best_score:
                best_score = cur_score
                best_p = p.copy()
                no_improve = 0
                print(f"[Init {init_id:02d}] Phase‑B iter {sub_it:06d} – new best: {best_score:.6f}")
            else:
                no_improve += 1

            # random restart if stuck
            if no_improve >= 1800:
                p = best_p + rng.normal(scale=0.004, size=n)
                p = np.clip(p, 0.0, None)
                p = _normalise(p)
                no_improve = 0

            # early exit if we already beat the target
            if best_score <= 1.503:
                break

        # ------------------------------------------------------------------
        # 6. Light random sampling around the current best (still time left)
        # ------------------------------------------------------------------
        while time.time() < deadline - 0.2:
            p_rand = rng.random(n)
            p_rand = _normalise(p_rand)
            cur = _objective(p_rand)
            if cur < best_score:
                best_score = cur
                best_p = p_rand.copy()
                print(f"[Random] new best: {best_score:.6f}")
            # cheap break
            if time.time() > deadline - 0.2:
                break

        # stop completely if we have reached the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 7. Convert the best probability vector into a concrete height list
    # ------------------------------------------------------------------
    if best_p is None:
        # fallback – uniform distribution (never happens in practice)
        best_p = np.full(n, 1.0 / n)

    # Scale to the allowed maximum 1000 (scale does not affect the ratio)
    heights = np.clip(best_p * 1000.0, 0.0, 1000.0).tolist()
    print(f"Final best score: {best_score:.6f} (length={len(heights)})")
    return heights
