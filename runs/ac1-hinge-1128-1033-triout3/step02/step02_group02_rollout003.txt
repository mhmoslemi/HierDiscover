import time
import numpy as np

def max_conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n -1]
    return conv.max(), conv

def get_subgradient(a, mask):
    # mask is binary length 2n-1
    n = a.shape[0]
    # reverse mask
    mask_rev = mask[::-1]
    size = 1 << ((n + len(mask_rev) - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    M = np.fft.rfft(mask_rev, n=size)
    conv = np.fft.irfft(A * M, n=size)
    # conv length = n + len(mask_rev) - 1 = n + (2n-1) -1 = 3n -2
    conv = conv[: (3*n - 2)]
    # grad_i = 2 * conv[2*n - 2 - i]
    grad = 2.0 * conv[2*n - 2 - np.arange(n)]
    return grad

def simplex_projection(v):
    # Project v (nonnegative constraint automatically?) onto simplex {x>=0, sum x = 1}
    # Code from literature.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def gaussian_distribution(n, sigma_factor):
    # sigma in units of n; actual sigma = sigma_factor * n
    sigma = sigma_factor * n
    i = np.arange(n)
    center = (n-1)/2.0
    a = np.exp(- ((i-center)**2) / (2 * sigma**2)) + 1e-12
    a /= a.sum()
    return a

"""
Propose a non‑negative step‑function that (approximately) minimises
`evaluate_sequence`.  

The algorithm works in two phases  

1. **Fast parametric sweep** – several simple families (U‑shaped power law,
   sum‑of‑tails, exponential decay and a Gaussian bell) are generated for a
   range of lengths.  Convolutions are evaluated with a real‑FFT which is
   much faster than `np.convolve`.  The best few candidates are retained.

2. **Projected sub‑gradient descent** – for the current candidate we
   repeatedly
      * compute the full self‑convolution `b = a * a`,
      * identify all lags where `b` attains the maximal value,
      * compute a sub‑gradient of the objective `max(b)` w.r.t. the
        coefficients `a` (the gradient is
        `2 * (a  *  mask_rev)` where `mask` marks the maximal lags),
      * take a step opposite to the gradient and project back onto the
        probability simplex (non‑negative entries summing to one).
   The step size follows a `1/√t` schedule and a simple back‑tracking line
   search guarantees monotone improvement.  Random jitter is added from
   time to time to escape possible plateaus.

The whole search respects the 1000 s time budget; the best sequence found
is finally evaluated with the official `evaluate_sequence` function.

The implementation uses only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> (float, np.ndarray):
    """Maximum entry of a*a and the full convolution."""
    conv = _full_convolution(a)
    return float(conv.max()), conv


def _subgradient_max_convolution(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(a*a) at points where `mask` (length 2n‑1) is 1.
    The gradient is 2 * (a * mask_rev) restricted to the original
    indices.  Computed via FFT in O(n log n).
    """
    n = a.shape[0]
    mask_rev = mask[::-1]
    size = 1 << ((n + mask_rev.shape[0] - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    M = np.fft.rfft(mask_rev, n=size)
    conv = np.fft.irfft(A * M, n=size)[: n + mask_rev.shape[0] - 1]
    # grad_i = 2 * conv[2*n-2 - i]  for i = 0 … n‑1
    grad = 2.0 * conv[2 * n - 2 - np.arange(n)]
    return grad


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all non‑positive -> uniform
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1          # last true index (0‑based)
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all are normalised to sum = 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] = (min(i,n‑1‑i)+1)^{-p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from both edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def gaussian_distribution(n: int, sigma_factor: float) -> np.ndarray:
    """Bell‑shaped Gaussian centred in the middle, σ = sigma_factor·n."""
    sigma = max(sigma_factor * n, 1e-3)
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2) + 1e-12
    a /= a.sum()
    return a

            # ----- sub‑gradient of max(b) -----
            grad = _subgradient_max_convolution(current, mask)

            # Optional momentum (here disabled, but left for easy tweaking)
            if beta_momentum > 0:
                momentum = beta_momentum * momentum + (1 - beta_momentum) * grad
                direction = momentum
            else:
                direction = grad

            # ----- back‑tracking line search -----
            eta = base_eta / np.sqrt(it)          # initial step size
            # guarantee a decrease – shrink step until improvement
            for _ in range(12):
                cand = current - eta * direction
                cand = _simplex_projection(cand)
                maxb_cand, _ = _max_convolution(cand)
                cand_score = 2.0 * n * maxb_cand
                if cand_score <= current_score:
                    break
                eta *= 0.5

            # ----- accept the move -----
            if cand_score <= current_score:
                current, current_score = cand, cand_score
                if cand_score < best_score:
                    best_score, best_seq = cand_score, cand.copy()
                    print(f"Iter {it:6d} | new global best = {best_score:.6f}")

                last_improve_it = it
            else:
                # No improvement – occasionally inject random jitter
                if it - last_improve_it > 400:
                    jitter = (np.random.rand(n) - 0.5) * 0.001
                    cand = current + jitter
                    cand = _simplex_projection(cand)
                    maxb_cand, _ = _max_convolution(cand)
                    cand_score = 2.0 * n * maxb_cand
                    if cand_score < current_score:
                        current, current_score = cand, cand_score
                        if cand_score < best_score:
                            best_score, best_seq = cand_score, cand.copy()
                            print(f"Iter {it:6d} | jitter improved global best = {best_score:.6f}")
                    last_improve_it = it

            # Early exit if we have reached the target bound
            if best_score < 1.503:
                print("Target threshold reached – terminating search.")
                time_left = deadline - time.time()
                if time_left > 0:
                    # Give the exact evaluator a final chance before timeout
                    pass
                break

    # ------------------------------------------------------------------
    # 3.  Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print("\nExact evaluation by evaluate_sequence: {:.6f}".format(exact_score))
    except Exception as exc:
        print("\nCould not call evaluate_sequence ({}). Using approximate score.".format(exc))
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {best_score:.6f}")
    print(f"Best exact score       : {exact_score:.6f}")
    print(f"Best length            : {len(best_seq)}")
    return best_seq.tolist()
