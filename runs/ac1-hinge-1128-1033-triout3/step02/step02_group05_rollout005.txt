"""
propose_candidate
-----------------

Search for a step‑function that beats the classic constant‑convolution
construction.  The idea is to start from the optimal “constant‑convolution”
sequence `a_k = C(2k, k) / 4**k` and then improve it by a simple
convex‑combination / linear‑programming loop.

At each iteration we solve a linear program

    maximise   Σ g_i
    subject to   (f * g)_k ≤ M   for all convolution indices k
                 g_i ≥ 0

where `f` is the current sequence, `M = max(f*f)` and `*` denotes the
standard linear convolution.  The LP finds a vector `g` with the same
convolution bound `M` but with a larger ℓ₁‑norm.  By rescaling `g` to the
original ℓ₁‑norm and mixing it back into `f` we strictly reduce the bound
`M`.  Repeating this step drives the quantity

    2·n·M / (Σ a_i)²

downwards.  The loop runs until the time budget is exhausted (or no further
improvement is observed).  The final sequence is returned as a plain list of
non‑negative floats, ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# The evaluation function is provided by the harness.
# from <environment> import evaluate_sequence


def _constant_convolution_coeffs(n: int) -> np.ndarray:
    """Return the first `n` coefficients a_k = C(2k,k)/4**k."""
    coeffs = np.empty(n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that A @ g = conv(f, g).
    Column `j` of A contains a copy of `f` shifted down by `j` rows.
    """
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    return A


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a list of heights that aims for a score < 1.503."""
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    np.random.seed(seed)

    # --------------------------------------------------------------
    # 1. initialise with the classic constant‑convolution sequence.
    # --------------------------------------------------------------
    # Choose a length that balances quality and LP size.
    # 1200 gives a nice trade‑off (≈ 2 400 × 1 200 matrix ≈ 20 MiB).
    n = 1200 if budget_s > 300 else 800

    f = _constant_convolution_coeffs(n)
    sum_f = float(f.sum())
    M = float(np.convolve(f, f).max())

    # best solution found so far
    best_seq = f.copy()
    best_obj = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, sum={sum_f:.4f}, M={M:.6f}, score={best_obj:.10f}")

    # --------------------------------------------------------------
    # 2. iterative improvement via a linear programme.
    # --------------------------------------------------------------
    max_iters = 200
    blend_factor = 0.20          # initial convex‑combination weight
    eps_improve = 1e-9           # required improvement in M to accept a step

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # Build the convolution matrix for the current f.
        A = _conv_matrix(f)                     # shape (2n‑1, n)
        m = A.shape[0]

        # Linear programme: maximise Σ g_i  (i.e. minimise -Σ g_i)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]
        b = M * np.ones(m, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options={"presolve": True})

        if not res.success:
            print(f"[iter {it}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g <= 0.0:
            print(f"[iter {it}] LP produced zero sum – stop.")
            break

        # Rescale g to have the same ℓ₁‑norm as f.
        scale = sum_f / sum_g
        g_scaled = g * scale

        # Try the pure scaled vector first.
        conv_g = np.convolve(g_scaled, g_scaled)
        M_g = float(conv_g.max())

        # If it improves M we can take it directly.
        if M_g < M - eps_improve:
            f_new = g_scaled
            M_new = M_g
        else:
            # Otherwise blend a little with the current f.
            f_new = (1.0 - blend_factor) * f + blend_factor * g_scaled
            conv_new = np.convolve(f_new, f_new)
            M_new = float(conv_new.max())
            if M_new >= M - eps_improve:
                # no improvement – shrink the blend factor and retry.
                blend_factor = max(0.01, blend_factor * 0.5)
                continue  # skip the update, try next iteration

        # Evaluate the new candidate.
        obj_new = evaluate_sequence(f_new.tolist())
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = f_new.copy()
            print(f"[iter {it}] new best score={best_obj:.10f}  M={M_new:.6f}")

        # Accept the update (even if the score did not improve, we keep the
        # reduced bound M – this drives the objective down in later steps).
        f = f_new
        M = M_new
        sum_f = float(f.sum())   # numerically should stay constant

    # --------------------------------------------------------------
    # 3. Return the best sequence found.
    # --------------------------------------------------------------
    print(f"[final] best score={best_obj:.10f}, length={len(best_seq)}")
    # Clamp to the limits enforced by `evaluate_sequence`.
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    return best_seq
