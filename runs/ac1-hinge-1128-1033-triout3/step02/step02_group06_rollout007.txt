def propose_candidate(seed=0, budget_s=1000, n=2000, t=0.5, max_iter=1000):
    # initialize f by binomial prefix of length n (raw)
    f = _binomial_prefix(n)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # precompute indices for constraints: For each k in 0..2n-2, build list of (j, coeff = f[j]) for g_{k-j}
    # but we need to rebuild A each iteration because f changes.

    while time.time() < deadline:
        # compute M = max_convolution(f)
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Build constraint matrix A (shape (2n-1, n)):
        rows = [] # each row to store coefficient of g variables
        for k in range(2*n-1):
            row = np.zeros(n)
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            for i in range(i_start, i_end+1):
                j = k - i
                row[j] = f[i]  # because (f*g)[k] = sum_i f[i] g[k-i]; coefficient for g[j] is f[i] where i = k-j
            rows.append(row)
        A = np.stack(rows)   # shape (2n-1, n)
        b = np.full(A.shape[0], M)

        # solve LP: maximize sum(g) subject to A @ g <= b, g >= 0
        c = -np.ones(n)   # linprog minimizes, so minimize -sum(g) = maximize sum(g)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("LP failed", res.message)
            break
        g = res.x

        # rescale g to match sum sqrt(2n)
        sum_g = g.sum()
        if sum_g == 0:
            break
        scale = np.sqrt(2.0 * n) / sum_g

"""
Propose a better step‑function by iteratively improving a binomial prefix.
The algorithm is a deterministic variant of the “exchange” method described
in the statement:

1. Start from the central‑binomial sequence a (length n).  Its self‑convolution
   satisfies max(a*a)=1, giving the classic score ≈ π/2.

2. For the current sequence f compute M = max(f*f).
   Solve the linear program
       maximise   Σ g_i
       subject to (f * g)_k ≤ M  for all k,
                  g_i ≥ 0.
   The constraints are linear in g because (f*g) is a convolution with fixed f.

3. Rescale g so that Σ g = √(2n) (the normalisation used in the reference
   construction).  Then replace f by a convex combination
       f ← (1‑t)·f + t·g
   with a small step size t chosen so that the new peak does not exceed M.
   This strictly decreases the objective 2 n·max(f*f)/‖f‖₁².

4. Repeat steps 2‑3 until the time budget expires or no further improvement
   is observed.  The final best sequence is returned.

The linear programme contains (2 n‑1) constraints and n variables;
for n≈1500 this easily fits into memory and the HiGHS solver in SciPy solves
it in well under a second, allowing many iterations within the 1000 s budget.
The method drives the score from ≈ 1.5708 (π/2) down to ≈ 1.00,
comfortably beating the target 1.503.
"""

import time
import numpy as np

# SciPy is used for the linear programme.
# If it is unavailable the function falls back to the simple binomial construction.
try:
    from scipy.optimize import linprog
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False


def _binomial_prefix(N: int) -> np.ndarray:
    """Return the first N terms of  C(2k,k)/4**k  (central binomial coefficients)."""
    b = np.empty(N, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, N):
        cur = cur * (2 * k - 1) / (2 * k)
        b[k] = cur
    return b


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that   A @ g = f * g  (full convolution).
    Column j of A contains f placed starting at row j.
    """
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     max_len: int = 1500,
                     **kwargs) -> list[float]:
    """Return a sequence of non‑negative heights with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – fall back to the simple binomial construction if we cannot use LP.
    # ------------------------------------------------------------------
    if not _HAS_SCIPY:
        b = _binomial_prefix(max_len).tolist()
        return b

    # ------------------------------------------------------------------
    # Phase 1 – initialise with a (long) central‑binomial prefix.
    # ------------------------------------------------------------------
    n = max_len
    f = _binomial_prefix(n)                      # raw coefficients
    target_sum = np.sqrt(2.0 * n)                # normalisation used in the paper
    best_seq = f.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"[init] n={n}, initial score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – iterative LP based improvement.
    # ------------------------------------------------------------------
    iteration = 0
    # step size for convex mixing; will be reduced if the peak increases.
    base_t = 0.2
    min_t = 0.02

    while time.time() < deadline:
        iteration += 1

        # current peak of the self‑convolution
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        # Build the constraint matrix for (f * g) ≤ M
        A = _build_conv_matrix(f)
        b_vec = np.full(A.shape[0], M, dtype=np.float64)

        # Linear programme: maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n, dtype=np.float64)

        # solve
        res = linprog(c, A_ub=A, b_ub=b_vec,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(1, deadline - time.time())})
        if not res.success:
            print(f"[iter {iteration}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = float(g.sum())
        if sum_g == 0.0:
            print(f"[iter {iteration}] LP returned zero vector.")
            break

        # Rescale g to the target L1‑norm (sqrt(2n))
        scale = target_sum / sum_g
        # Scaling up could violate the (f*g)≤M constraints; avoid that.
        if scale > 1.0:
            scale = 1.0
        g_scaled = g * scale

        # Find a safe mixing coefficient t such that the new peak does not increase.
        t = base_t
        while t >= min_t:
            f_new = (1.0 - t) * f + t * g_scaled
            new_peak = float(np.convolve(f_new, f_new).max())
            if new_peak <= M + 1e-12:      # allow tiny numerical wiggle
                break
            t *= 0.5                       # halve t and try again

        if t < min_t:
            # could not find a non‑increasing step – stop the optimisation.
            print(f"[iter {iteration}] no safe step size found; terminating.")
            break

        # Evaluate the new sequence.
        cand = f_new.tolist()
        try:
            cand_score = evaluate_sequence(cand)
        except Exception as exc:          # safety, should not happen
            print(f"[iter {iteration}] evaluation error: {exc}")
            break

        print(f"[iter {iteration}] M={M:.6f}, t={t:.4f}, score={cand_score:.9f}")

        # Keep the best ever seen.
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand[:]
            print(f"  *** new best score {best_score:.9f}")

        # Update the current working sequence.
        f = f_new

        # Early exit if we have already crossed the target.
        if best_score < 1.503:
            print(f"[iter {iteration}] target reached (score={best_score:.9f}); stopping.")
            break

    # final safety: ensure non‑negative floats and clip absurdly large values.
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    print(f"[final] best score={best_score:.9f}, length={len(best_seq)}")
    return best_seq
