def compute_gradient(a, w):
    # a: np.ndarray shape (n,)
    # w: np.ndarray shape (2n-1,)
    n = a.shape[0]
    # compute convolution of a_rev and w
    size = 1 << ((3*n-2).bit_length())
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv_full = np.fft.irfft(A_rev * W, n=size)  # length size
    # conv_full length size>=3n-2; we need first (3n-2) elements.

def max_conv_fft(a: np.ndarray) -> float:
    """Return the maximum entry of the self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution vector (length 2n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv.astype(np.float64)


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution vector with temperature 1/β."""
    # shift by max for numerical stability
    shift = conv.max()
    exp_part = np.exp(beta * (conv - shift))
    s = exp_part.sum()
    if s == 0.0:
        # fallback to uniform weights if under‑flow occurred
        return np.full_like(conv, 1.0 / conv.size)
    return exp_part / s


def gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of soft‑max surrogate w.r.t a."""
    # g_i = 2 * sum_j w_{i+j} * a_j   (j runs 0..n‑1, i+j must be <2n‑1)

    W = np.fft.rfft(w, n=size)

    # slice n‑1 … n‑1+n‑1 (inclusive) -> n elements
    g = 2.0 * conv_full[n - 1 : n - 1 + n]
    return g


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Implementation of the algorithm from
    # https://arxiv.org/pdf/1101.6081.pdf
    if v.min() < 0:
        # shift to avoid negative numbers before sorting
        v = np.maximum(v, 0)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """High at the edges, low in the centre: (i+1)*(n-i) raised to -p."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = (prod + 1) ** (-p) + 1e-12

        # full convolution (needed for the gradient)
        conv = full_conv_fft(cur)
        maxb = conv.max()
        cur_score = 2.0 * n * maxb

        # record global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"Iter {it:7d}: new global best = {best_score:.6f}")

        # soft‑max weights over the convolution entries
        w = softmax_weights(conv, beta)

        # gradient of the smooth surrogate
        grad = gradient_softmax(cur, w)

        # additive projected gradient step
        cand = cur - eta * grad
        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total > 0.0:
            cand /= total
        else:
            cand = cur.copy()

        # evaluate candidate (only the max entry is needed for the true objective)
        cand_maxb = max_conv_fft(cand)
        cand_score = 2.0 * n * cand_maxb

        # simulated‑annealing acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = cand
            stagnant = 0
        else:
            stagnant += 1

        # schedule updates
        beta = min(beta * beta_mul, beta_max)
        eta = max(eta * eta_decay, min_eta)
        temperature = max(temperature * temp_decay, 1e-6)

        # occasional jitter if we have been stuck for a long time
        if stagnant > 5000:
            jitter = np.random.randn(n) * 0.02
            cand = cur + jitter
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cur = cand
            stagnant = 0
            print(f"Iter {it:7d}: jitter injected")

        # early exit when the target is already surpassed
        if best_score < 1.503:
            print(f"Target reached at iteration {it}, exiting early.")
            break

    # --------------------------------------------------------------
    # 4.  Final exact evaluation (once) and report.
    # --------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
