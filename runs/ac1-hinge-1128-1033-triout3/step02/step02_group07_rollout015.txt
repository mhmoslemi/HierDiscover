"""
minimise_step_function.py

Search for a non‑negative step‑function (a list of heights) that minimises
the supplied ``evaluate_sequence`` metric.

Strategy
--------
1. **Parameter sweep** –  the family  a_i ∝ (i+1)^{-α}  (α∈(0,1)) contains the
   classic “square‑root’’ construction (α=0.5).  A very cheap grid search over
   α finds a point on this curve that is already a little better than the
   analytic start for the concrete length *n*.

2. **Smoothed‑max gradient descent** –  instead of the sub‑gradient of the
   exact maximum we optimise the smooth approximation  

        L(a) = (1/β)·log( Σ_k exp(β·(a*a)_k) ) ,

   whose gradient can be written as a convolution and is cheap to evaluate
   with FFTs.  β is increased gradually so that the optimisation first moves
   in a well‑behaved landscape and then concentrates on the true supremum.

3. **Simulated‑annealing polishing** –  a classic Metropolis walk with
   temperature‑controlled mass transfers adds the ability to escape shallow
   local minima and to fine‑tune the solution.

All steps use only the fast FFT‑based score

        fast_score(a) = 2·n·max(a*a)

which is exactly the value returned by ``evaluate_sequence`` after the
normalisation ``sum(a)=1``.  The expensive ``evaluate_sequence`` is called
only when a new global best is found.

The routine respects the 1000 s time budget, works for any length supplied
via the ``length`` keyword, and prints a few progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast score (identical to the expensive evaluate_sequence for sum==1)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, size_fft: int) -> float:
    """2·n·max(a*a) computed by FFT."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Build a power‑law family a_i ∝ (i+1)^{-α}
# ----------------------------------------------------------------------
def _power_law(n: int, alpha: float) -> np.ndarray:
    """Return the normalised (i+1)^(-alpha) vector."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** (-alpha)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Smoothed‑max gradient (soft‑max with temperature β)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float, size_fft: int) -> np.ndarray:
    """
    Gradient of L_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k).

    Returns ∇L_β (same shape as a).  Computation uses FFTs:
        w_k ∝ exp(β·conv_k)   (softmax weights)
        grad_i = 2· Σ_k w_k a_{k-i}
    """
    n = a.shape[0]

    # convolution a*a
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]

    # softmax weights (avoid overflow)
    max_c = conv.max()
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()        # shape (2n-1,)

    # compute gradient = 2 * (w * a)  (cross‑correlation)
    # cross‑correlation can be done with one FFT: reverse a and convolve with w
    a_rev = a[::-1]
    fw = np.fft.rfft(w, size_fft)
    fa_rev = np.fft.rfft(a_rev, size_fft)
    grad_full = np.fft.irfft(fw * fa_rev, size_fft)[: n + n - 1]

    # the relevant part of the cross‑correlation is the middle n entries
    grad = 2.0 * grad_full[n - 1 : 2 * n - 1]   # length n
    return grad


# ----------------------------------------------------------------------
# Simulated‑annealing proposal (mass transfer between two indices)
# ----------------------------------------------------------------------
def _anneal_step(a: np.ndarray, cur_fast: float, size_fft: int,
                 temperature: float) -> tuple[np.ndarray, float, bool]:
    """
    Propose a tiny mass transfer and decide acceptance.
    Returns (new_a, new_fast, accepted_flag).
    """
    n = a.shape[0]

    # pick a source with non‑zero mass
    src = np.random.randint(n)
    if a[src] < 1e-12:
        return a, cur_fast, False

    # destination must be different
    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1

    # amount transferred – proportional to current temperature
    max_move = a[src] * min(0.5, temperature)
    delta = np.random.rand() * max_move

    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta

    fast_new = _fast_score(a_new, size_fft)

    # Metropolis acceptance
    if fast_new <= cur_fast:
        return a_new, fast_new, True
    else:
        prob = math.exp(-(fast_new - cur_fast) / max(temperature, 1e-12))
        if np.random.rand() < prob:
            return a_new, fast_new, True
        else:
            return a, cur_fast, False

        a = _project_simplex(a)    # guarantee feasibility (sum=1)

        cur_fast = _fast_score(a, size_fft)
        _maybe_update_best(a, cur_fast)

        beta = 5.0                 # start with a gentle smoothing
        beta_inc = 1.05           # multiplicative increase each outer loop
        max_outer = 15            # number of β‑increments
        for outer in range(max_outer):
            if time.time() >= phase_end:
                break
            # inner loop – simple gradient steps on the smoothed loss
            step_size = 0.7
            for it in range(1200):
                if time.time() >= phase_end:
                    break
                grad = _smooth_gradient(a, beta, size_fft)
                # simple projected gradient step (no exponentiation)
                a_new = a - step_size * grad
                a_new = _project_simplex(a_new)

                fast_new = _fast_score(a_new, size_fft)
                if fast_new < cur_fast:
                    a = a_new
                    cur_fast = fast_new
                    _maybe_update_best(a, cur_fast)
                # slowly decay step size
                step_size *= 0.9995
            # increase temperature to focus on the true max
            beta *= beta_inc

        print(f"[{n}] after smoothed‑grad: score {cur_fast:.6f}")

        # --------------------------------------------------------------
        # 3. Simulated‑annealing fine‑tuning
        # --------------------------------------------------------------
        T_start = 0.12
        cur_fast_sa = cur_fast
        no_improve = 0
        iter_sa = 0

        while time.time() < phase_end:
            # linear cooling
            elapsed = time.time() - (phase_end - alloc)
            frac = min(1.0, elapsed / alloc)
            T = T_start * (1.0 - frac)

            a, cur_fast_sa, improved = _anneal_step(a, cur_fast_sa,
                                                    size_fft, T)

            if improved and cur_fast_sa < cur_fast - 1e-9:
                cur_fast = cur_fast_sa
                _maybe_update_best(a, cur_fast)
                no_improve = 0
            else:
                no_improve += 1

            # occasional restart if stuck
            if no_improve > 25000:
                a = _power_law(n, 0.5)                 # back to square‑root start
                a += np.random.randn(n) * 1e-4
                a = _project_simplex(a)
                cur_fast_sa = _fast_score(a, size_fft)
                cur_fast = cur_fast_sa
                no_improve = 0
                print(f"[{n}] restart after {iter_sa} SA steps")

            iter_sa += 1

        print(f"[finished n={n}] best local evaluation {cur_fast:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (guaranteed to be valid)
        best_seq = np.full(lengths[0], 1.0 / lengths[0])
        print("[fallback] returning uniform sequence")

    print(f"[final] overall best evaluation = {best_score:.6f}")
    return best_seq.tolist()
