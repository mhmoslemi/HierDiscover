"""
propose_candidate
-----------------

Iterative improvement of the classic “central‑binomial” step function.
The method starts from the optimal sequence
    b_k = C(2k,k)/4**k      (k=0,…,n‑1)
which gives a flat self‑convolution (max = 1) and a score ≈π/2.
We then solve a linear programme

    maximise    Σ b_j
    subject to  (f * b)[k] ≤ M_f    for all k
                b_j ≥ 0

where ``f`` is the current sequence and ``M_f`` its maximal
self‑convolution value.  The LP adds mass at the “slack’’ edges of the
convolution, increasing the L₁‑norm while keeping the L∞‑norm ≤ M_f.
After rescaling the LP solution to the original L₁‑norm we obtain a new
sequence with a strictly smaller maximal convolution; the process is
repeated until the time budget expires or no further improvement is
possible.  Finally a tiny stochastic hill‑climb refines the best result.

The algorithm runs in ≤ 1000 s, uses only NumPy and SciPy, and returns a
list of non‑negative floats.
"""

import time
import numpy as np

try:
    # SciPy is optional – if unavailable we fall back to the simple binomial method.
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:                    # pragma: no cover
    _SCIPY_AVAILABLE = False


def _binomial_prefix(n: int) -> np.ndarray:
    """Return the first ``n`` central‑binomial coefficients
    b_k = C(2k,k) / 4**k  (k = 0,…,n‑1)."""
    b = np.empty(n, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)   # recurrence: b_{k}=b_{k-1}*(2k-1)/(2k)
        b[k] = cur
    return b


def _max_convolution(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of ``seq``."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ b_j   subject to   (f * b)[k] ≤ M   (all k) , b≥0.
    Returns the optimal `b` (numpy array) or None on failure.
    """
    n = f.size
    rows = 2 * n - 1

    # Build the (rows × n) matrix A where A @ b = f * b (full convolution)
    A = np.zeros((rows, n), dtype=np.float64)
    for i, val in enumerate(f):
        A[i:i + n, i] = val

    # Linear programme: minimise -1·b  (i.e. maximise Σ b)
    c = -np.ones(n, dtype=np.float64)
    ub = np.full(rows, M, dtype=np.float64)          # RHS of A @ b ≤ M
    bounds = [(0.0, 1000.0) for _ in range(n)]       # clip to the evaluator’s ceiling

    # Use HiGHS – fast for medium‑size dense LPs
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    Generate a step‑function sequence with a low evaluation score.
