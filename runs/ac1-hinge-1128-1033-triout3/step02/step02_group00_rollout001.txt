# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (Duchi et al.)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# FFT‑based full convolution (any two 1‑D arrays)
# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Return the full convolution u * v using a real FFT."""
    m = len(u) + len(v) - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    fu = np.fft.rfft(u, size)
    fv = np.fft.rfft(v, size)
    conv = np.fft.irfft(fu * fv, size)[:m]
    np.maximum(conv, 0, out=conv)             # clip tiny negatives
    return conv

# ----------------------------------------------------------------------

    """Full (length 2·n‑1) autocorrelation a*a."""
    return _conv_fft(a, a)

        if w_sum == 0:
            w_norm = np.full_like(w, 1.0 / w.shape[0])
        else:
            w_norm = w / w_sum

        # --- gradient = 2·Σ_k w_k·a_{k‑i} (computed via convolution) ----------
        conv_rev = _conv_fft(w_norm, a_rev)                 # length = 3n‑2
        slice_start = n - 1
        slice_end = slice_start + n
        subgrad = 2.0 * conv_rev[slice_start:slice_end]     # shape (n,)

        # --- step size (linear cooling) ------------------------------------
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # --- gradient step + projection --------------------------------------
        a_candidate = a - eta * subgrad

        # occasional random mass‑transfer (helps escape plateaus)
        if np.random.rand() < 0.01:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = a_candidate[src] * np.random.rand() * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        a = _project_onto_simplex(a_candidate)
        a_rev = a[::-1]                                     # keep reverse up‑to‑date
        iteration += 1

        # occasional full recomputation of convolution to avoid drift
        if iteration % 5000 == 0:
            conv = _full_convolution(a)

    # ------------------------------------------------------------------
    # Polishing phase: pure sub‑gradient descent on the true max (no smoothing)
    # ------------------------------------------------------------------
    print("[polish] starting sub‑gradient refinement")
    a = best_a.copy()
    a_rev = a[::-1]

    # parameters for the polishing stage
    pol_eta_start = 0.03
    pol_eta_end = 1e-5
    pol_iter = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[polish {pol_iter:7d}] new best = {best_val:.6f}")

        # active index where the convolution attains its maximum
        k_star = int(np.argmax(conv))

        # sub‑gradient of the active peak
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        subgrad = np.zeros_like(a)
        subgrad[idx] = 2.0 * a[k_star - idx]

        # step‑size schedule (linear)
        progress = (now - start_time) / (deadline - start_time)
        eta = pol_eta_start * (1.0 - progress) + pol_eta_end * progress

        a_candidate = a - eta * subgrad

        # random jump (same probability as before)
        if np.random.rand() < 0.01:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = a_candidate[src] * np.random.rand() * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        a = _project_onto_simplex(a_candidate)
        a_rev = a[::-1]

        pol_iter += 1
        if pol_iter % 3000 == 0:
            conv = _full_convolution(a)

    # ------------------------------------------------------------------
    # Final reporting (using the provided evaluation routine)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
