from scipy.optimize import minimize

def optimize_smooth(alpha, q0, maxiter=200):
    def fun(q):
        # compute p via softmax
        maxq = np.max(q)  # for numerical stability
        exp_q = np.exp(q - maxq)
        p = exp_q / exp_q.sum()
        conv = np.convolve(p, p)
        # compute smoothmax
        z = alpha * conv
        maxz = np.max(z)
        expz = np.exp(z - maxz)
        lse = maxz + np.log(np.sum(expz))
        f = lse / alpha  # smoothmax
        # compute weights w = softmax(α conv)
        w = expz / np.sum(expz)
        # gradient wrt p
        w_rev = w[::-1]
        corr = np.convolve(p, w_rev)
        idx = 2 * n - 2 - np.arange(n)
        grad_p = 2.0 * corr[idx]   # gradient of smoothmax wrt p
        # gradient wrt q (softmax parameterization)
        grad_q = p * (grad_p - np.dot(p, grad_p))
        return f, grad_q
    res = minimize(fun, q0, jac=True, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'disp': False})
    return res

    corr = np.convolve(p, w_rev)
    idx = 2 * n - 2 - np.arange(n)
    grad_p = 2.0 * corr[idx]
    # gradient wrt q via chain rule
    dot = np.dot(p, grad_p)
    grad_q = p * (grad_p - dot)
    return f, grad_q

"""
Optimise the step‑function heights with a two‑phase strategy:

1️⃣  **Smooth optimisation (L‑BFGS‑B)** – the non‑smooth max in the
    objective  M = 2·n·max(conv(a,a)) is replaced by a
    log‑sum‑exp (soft‑max) with a temperature α.
    The decision variables are the unconstrained logits q,
    i.e. a = softmax(q).  Using the analytical gradient
    we run a few L‑BFGS‑B optimisations with increasing α.
    This quickly drives the distribution away from the uniform
    “triangular” shape of the baseline.

2️⃣  **Exact max refinement (exponentiated mirror descent)** – starting
    from the best soft‑max solution we switch to the true objective.
    The sub‑gradient at the active convolution index is used in an
    exponentiated‑gradient (mirror‑descent) update, which automatically
    respects non‑negativity and the simplex constraint.
    Small Gaussian jitter is added occasionally to escape shallow
    plateaus.

Both phases keep the best sequence seen so far and respect the
1000 s time budget.  The returned list is ready for ``evaluate_sequence``."""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(p: np.ndarray, n: int) -> float:
    """Exact evaluation of M = 2·n·max(conv(p,p)) / (sum(p)^2)."""
    conv = np.convolve(p, p)
    return float(2.0 * n * np.max(conv) / (np.sum(p) ** 2))


def _smooth_obj_and_grad(q: np.ndarray, n: int, alpha: float):
    """
    Smooth approximation of the max‑convolution objective.
    Variables are logits q, the actual distribution is a = softmax(q).
    Returns (f, grad_q) where f = (1/α)·log Σ exp(α·conv) and grad_q
    is the gradient w.r.t. q.
    """
    # soft‑max to obtain a valid probability vector
    max_q = np.max(q)
    exp_q = np.exp(q - max_q)
    p = exp_q / exp_q.sum()

    # convolution of p with itself
    conv = np.convolve(p, p)                     # length 2·n‑1

    # soft‑max weights over the convolution entries
    z = alpha * conv
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)
    sum_exp_z = np.sum(exp_z)
    f = (max_z + np.log(sum_exp_z)) / alpha      # smooth max(conv)

    w = exp_z / sum_exp_z                         # soft‑max probabilities (length 2·n‑1)

    # gradient of smooth max w.r.t. p:
    #   ∂f/∂p_i = 2· Σ_k w_k·p_{k-i}  →  2·(p * rev(w))[i]  (restricted to 0≤i<n)
    w_rev = w[::-1]
    corr = np.convolve(p, w_rev)                 # length 3·n‑2
    idx = 2 * n - 2 - np.arange(n)               # indices that give the n‑dimensional gradient
    grad_p = 2.0 * corr[idx]

    # chain rule to obtain gradient w.r.t. logits q (soft‑max Jacobian)
    dot = np.dot(p, grad_p)
    grad_q = p * (grad_p - dot)

    return f, grad_q


def _optimize_smooth(q0: np.ndarray, n: int, alpha: float, maxiter: int, deadline: float):
    """
    Run a limited L‑BFGS‑B optimisation of the smooth objective.
    Stops early if the global deadline is reached.
    """
    # The optimiser does not accept a deadline, so we simply bound the
    # number of iterations; the surrounding loop checks the remaining time.
    res = minimize(
        _smooth_obj_and_grad,
        q0,
        args=(n, alpha),
        jac=True,
        method="L-BFGS-B",
        options={"maxiter": maxiter, "disp": False},
    )
    return res.x, float(res.fun)

# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    M = 2·n·max(conv(a,a)) / (sum(a)**2).
    The routine respects the 1000 s time budget and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size (use the historic sequence length if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name‑defined]
    except Exception:
        n = 1153

    # ------------------------------------------------------------------
    # Baselines (uniform and optionally the historic best)
    # ------------------------------------------------------------------
    best_p = np.full(n, 1.0 / n, dtype=float)
    best_score = _compute_score(best_p, n)
    print(f"[Init] Uniform baseline score: {best_score:.6f}")

    if "height_sequence_1" in globals():
        seq = np.clip(np.array(height_sequence_1, dtype=float), 0.0, 1000.0)
        total = seq.sum()
        if total > 0.0:
            p_prev = seq / total
            score_prev = _compute_score(p_prev, n)
            if score_prev < best_score:
                best_score = score_prev
                best_p = p_prev.copy()
                print(f"[Init] Previous sequence score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth L‑BFGS‑B optimisation with increasing α
    # ------------------------------------------------------------------
    # initialise logits (q) from the current best distribution
    eps = 1e-12
    q = np.log(best_p + eps)                  # soft‑max parametrisation

    # schedule of temperatures (larger α → tighter approximation of the max)
    alphas = [0.5, 2.0, 6.0, 12.0, 24.0]

    for alpha in alphas:
        # remaining time check
        if time.time() > deadline:
            break
        # a modest iteration budget – we rely on the LBFGS curvature model
        maxiter = 200
        q, _ = _optimize_smooth(q, n, alpha, maxiter, deadline)

        # evaluate the *exact* objective for the current distribution
        p = np.exp(q - np.max(q))
        p /= p.sum()
        cur_score = _compute_score(p, n)
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[Smooth] α={alpha:.1f} – new best score: {best_score:.6f}")

        # early exit if we have already passed the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 2 – exact max sub‑gradient with exponentiated updates
    # ------------------------------------------------------------------
    p = best_p.copy()
    eta = 0.5                  # step size for exponentiated update
    eta_decay = 0.9995
    no_improve = 0
    sub_it = 0

    while time.time() < deadline:
        sub_it += 1

        conv = np.convolve(p, p)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) at the active index k_max
        grad = np.zeros_like(p)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        # entries i that actually participate in the active sum
        if i_start <= i_end:
            idx_range = np.arange(i_start, i_end + 1)
            grad[idx_range] = 2.0 * p[k_max - idx_range]

        # exponentiated (mirror) update
        p = p * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = p.sum()
        if s == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p = p / s

        # occasional jitter to escape plateaus
        if sub_it % 500 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            p = _simplex_projection(p + jitter)

        # decay learning rate
        eta *= eta_decay

        # evaluate exact objective
        cur_score = _compute_score(p, n)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
            print(f"[Mirror] iter {sub_it:6d} – new best score: {best_score:.6f}")
            if best_score <= 1.503:
                break
        else:
            no_improve += 1

        # if stagnating for a while, re‑initialise around the current best
        if no_improve >= 2500:
            p = best_p + rng.normal(scale=0.005, size=n)
            p = np.clip(p, 0.0, None)
            p = p / p.sum()
            no_improve = 0

    # ------------------------------------------------------------------
    # Final random sampling (uses any remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # tiny Dirichlet draws – cheap way to possibly improve a little
        p_rand = rng.random(n)
        p_rand = p_rand / p_rand.sum()
        score_rand = _compute_score(p_rand, n)
        if score_rand < best_score:
            best_score = score_rand
            best_p = p_rand.copy()
            print(f"[Random] new best score: {best_score:.6f}")
        # limit loop frequency
        if time.time() + 0.01 > deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Best score achieved: {best_score:.6f}")
    return result
