"""
Propose a better step‑function sequence.

Algorithm
---------
* Work with a probability vector a (a_i ≥ 0, Σ a_i = 1).  For such a vector
  the evaluation routine reduces to   score = 2·n·max(a∗a) .
* The objective   max(a∗a)   is non‑smooth.  We replace the hard max by a
  smooth “soft‑max”:
        f(a) = (1/τ)·log Σ_k exp(τ·(a∗a)_k) .
  Its gradient is  ∇f = 2·correlation( w , a )  where
        w_k = exp(τ·( (a∗a)_k – max(a∗a) )) / Σ exp(·) .
* An Adam optimiser (first‑order moments, adaptive step size) is applied to
  minimise f.  After each step the vector is projected onto the probability
  simplex (Duchi‑et‑al. O(n log n) algorithm).
* The temperature τ and the Adam learning‑rate are increased/decreased
  linearly with the elapsed time so that early iterations make large,
  smooth moves while later ones focus on the true maximum.
* Occasionally a tiny random mass‑transfer is performed to escape plateaus.
* The best sequence (lowest true score) seen during the run is returned.

The method is completely self‑contained, uses only NumPy and the supplied
`evaluate_sequence` function, and respects the 1000‑second time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fall back – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to guard against rounding errors
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Full self‑convolution using a real FFT (fast for many iterations)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a∗a (length 2·n‑1) using an FFT of length `size`."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    # negative rounding errors are harmless – clamp to zero
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth max (soft‑max) of the self‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      tau: float,
                      size: int) -> np.ndarray:
    """
    Compute ∇f where
        f(a) = (1/τ)·log Σ_k exp(τ·conv_k)  (soft‑max of the convolution).

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    conv : np.ndarray, shape (2n‑1,)
        Full self‑convolution a∗a.
    tau : float
        Temperature parameter (larger → closer to true max).
    size : int
        FFT length (power‑of‑two ≥ 3·n‑2).

    Returns
    -------
    grad : np.ndarray, shape (n,)
        Gradient of f w.r.t. a.
    """
    # soft‑max weights (numerically stable)
    max_c = conv.max()
    w = np.exp(tau * (conv - max_c))
    w /= w.sum()                     # w_k ≥ 0, Σ w_k = 1

    # correlation w ⋆ a = Σ_i w_{i+k} a_i  (k = 0 … n‑1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    corr_full = np.fft.irfft(fw * np.conj(fa), size)
    n = a.shape[0]
    # the “valid” part of the correlation has length n and starts at index n‑1
    corr_valid = corr_full[n - 1 : n - 1 + n]

    grad = 2.0 * corr_valid
    return grad
