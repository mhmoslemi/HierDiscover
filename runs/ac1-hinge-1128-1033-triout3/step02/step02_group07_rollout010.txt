    # a length n, w length 2n-1
    w_rev = w[::-1]

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. *Smooth max* –  minimise a differentiable approximation of the
   max‑convolution term:  

        S(a) = (1/β)·log ∑ₖ exp(β·(a∗a)ₖ)

   The gradient of S is a weighted sum of the convolution gradients.
   We compute it efficiently with FFTs.

2. *Adam on the simplex* –  a is kept on the probability simplex
   (∑aᵢ=1, aᵢ≥0) using the Duchi‐et‑al. projection.  Adam
   moments are used for an adaptive step size.

3. *β schedule* –  start with a small smoothing parameter β and increase
   it linearly, turning the smooth objective into the true max.

4. *Fine‑tuning* –  after the gradient phase a short simulated‑annealing
   (mass transfer) walk refines the solution.

5. *Multiple lengths* –  the time budget is split over several problem
   sizes; the best sequence seen (according to the costly
   ``evaluate_sequence``) is returned.

The routine works entirely with NumPy and FFTs (≈O(n log n) per
iteration) and respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic start: coefficients of (1‑x)^(-1/2)  (normalised to sum 1)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n terms of (1‑x)^(-1/2), normalised to unit L¹ norm."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Helper: next power of two (for FFT length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Fast full convolution (a∗a) using a pre‑allocated FFT length
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return a∗a (full, length 2·len(a)‑1) using FFT of given length."""
    Fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(Fa * Fa, size_fft)
    n = a.shape[0]
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Fast score = 2·n·max(a∗a) (identical to evaluate_sequence for sum=1)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, size_fft: int) -> float:
    n = a.shape[0]
    b = _conv_full(a, size_fft)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Softmax weights for a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Return softmax(b·β) – a probability vector of length len(b)."""
    b_max = b.max()
    # subtraction avoids overflow
    exp_vals = np.exp(beta * (b - b_max))
    w = exp_vals / exp_vals.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the smoothed objective S(a) = (1/β)·log∑exp(β·bₖ)
# using the identity ∂bₖ/∂aᵢ = 2·a_{k-i}
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Compute ∇ₐ Σₖ wₖ·(a∗a)ₖ.
    The formula reduces to 2·(a * w_rev) restricted to n entries.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reverse for the needed shift
    Fa = np.fft.rfft(a, size_fft)
    Fwr = np.fft.rfft(w_rev, size_fft)
    conv = np.fft.irfft(Fa * Fwr, size_fft)
    # conv length = n + (2n‑1) - 1 = 3n‑2
    # slice indices n‑1 … 2n‑2 (inclusive) and reverse
    grad_part = conv[n - 1 : 2 * n - 1][::-1]
    return 2.0 * grad_part


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Return a non‑negative sequence that (approximately) minimises
    ``evaluate_sequence``.  The routine combines a smoothed‑max Adam
    optimiser with a brief Simulated‑Annealing fine‑tuner.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose problem sizes (split the time roughly equally)
    # ------------------------------------------------------------------
    if "length" in kwargs:
        lengths = [int(kwargs["length"])]
    else:
        lengths = [3000, 5808, 8000]   # a small diverse set

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper: update the true best only when a better *fast* score is seen.
    # ------------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray, fast_val: float) -> None:
        nonlocal best_seq, best_score
        if fast_val < best_score - 1e-12:
            true_val = evaluate_sequence(candidate.tolist())
            if true_val < best_score:
                best_score = true_val
                best_seq = candidate.copy()
                print(f"[global] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Loop over the selected lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        alloc = remaining / (len(lengths) - lengths.index(n))
        phase_end = time.time() + alloc

        # --------------------------------------------------------------
        # Starting vector: prefer the global `height_sequence_1` if it matches n
        # --------------------------------------------------------------
        try:
            a0 = np.array(height_sequence_1, dtype=float)
            if a0.shape[0] != n:
                a0 = _sqrt_series(n)
        except Exception:
            a0 = _sqrt_series(n)

        a = _project_simplex(a0)                    # enforce ∑aᵢ = 1, aᵢ≥0

        # FFT length that covers both the full convolution (2n‑1) and the
        # convolution with the weight vector (3n‑2)
        size_fft = _next_pow2(3 * n - 2)

        cur_fast = _fast_score(a, size_fft)
        _maybe_update(a, cur_fast)
        print(f"[init] n={n}, start score {cur_fast:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – Smoothed‑max Adam optimisation
        # --------------------------------------------------------------
        beta = 0.5            # initial smoothing strength
        beta_inc = 0.001      # linear increase per iteration
        lr = 0.05             # base learning rate
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it = 0

        while time.time() < phase_end:
            # --- full convolution and current fast score ----------------
            b_full = _conv_full(a, size_fft)                 # length 2n‑1
            max_b = b_full.max()
            fast_cur = 2.0 * n * max_b

            # --- softmax weights (smooth approximation of the max) -----
            w = _softmax_weights(b_full, beta)

            # --- gradient of the smoothed objective --------------------
            grad = _smooth_gradient(a, w, size_fft)

            # --- Adam update -------------------------------------------
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = _project_simplex(a_new)

            # --- evaluate candidate ------------------------------------
            fast_new = _fast_score(a_new, size_fft)

            # Accept if we improve (or with a tiny chance to escape plateaus)
            if fast_new < fast_cur or np.random.rand() < 0.001:
                a = a_new
                cur_fast = fast_new
                _maybe_update(a, cur_fast)

            # increase smoothing parameter slowly
            beta = beta + beta_inc

            # occasional random jitter to keep the dynamics alive
            if it % 2000 == 0:
                jitter = np.random.randn(n) * 1e-4
                a = _project_simplex(a + jitter)

            if time.time() >= phase_end:
                break

        # --------------------------------------------------------------
        # Phase 2 – Simulated‑Annealing fine‑tuning (tiny mass swaps)
        # --------------------------------------------------------------
        T_start = 0.2
        fast_local = cur_fast
        a_local = a.copy()
        it_sa = 0
        no_improve = 0

        while time.time() < phase_end:
            elapsed = time.time() - (phase_end - alloc)
            frac = min(1.0, elapsed / alloc)
            T = T_start * (1.0 - frac)          # linear cooling

            # pick a source with positive mass
            src = np.random.randint(n)
            if a_local[src] < 1e-12:
                continue
            # pick a different destination
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

            # amount of mass to move (scaled with temperature)
            delta_max = a_local[src] * min(0.5, T)
            delta = np.random.rand() * delta_max

            a_cand = a_local.copy()
            a_cand[src] -= delta
            a_cand[dst] += delta

            fast_cand = _fast_score(a_cand, size_fft)

            # Metropolis acceptance
            if fast_cand <= fast_local or np.random.rand() < math.exp(
                    -(fast_cand - fast_local) / max(T, 1e-12)):
                a_local = a_cand
                fast_local = fast_cand
                if fast_cand < cur_fast - 1e-9:
                    cur_fast = fast_cand
                    _maybe_update(a_local, fast_cand)
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            it_sa += 1

            # long stagnation → restart from a noisy sqrt‑series
            if no_improve > 20000:
                a_local = _sqrt_series(n)
                a_local += np.random.randn(n) * 1e-3
                a_local = _project_simplex(a_local)
                fast_local = _fast_score(a_local, size_fft)
                no_improve = 0
                print(f"[restart] n={n} after {it_sa} SA steps")

            if time.time() >= phase_end:

        print(f"[finished n={n}] best local fast score {cur_fast:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of appropriate length
        best_seq = np.full(lengths[0], 1.0 / lengths[0])
    print(f"[final] overall best evaluation = {best_score:.6f}")
    return best_seq.tolist()
