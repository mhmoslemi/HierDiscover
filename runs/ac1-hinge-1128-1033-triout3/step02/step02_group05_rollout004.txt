"""
propose_candidate
-----------------

Generate a step‑function whose self‑convolution is close to flat.
The construction discretises the continuous *arcsine* density
    f(x) = 1 / (π·√(x·(1‑x))) ,  x∈(0,1) .
The arcsine law has the remarkable property that the sum of two
independent copies is uniformly distributed on (0,2), i.e.

    (f * f)(y) = ½   for 0 ≤ y ≤ 2 .

When we replace the continuous density by a discrete probability
vector a (mass = f(x_i)·Δx with Δx = 1/N) the convolution a*a
approximates a constant.  Because
    ∑ a_i = 1  (scale‐invariance of the score),
the objective reduces to 2·N·max(a*a).  The lower theoretical bound
is 2·N/(2·N‑1)≈1, so any reasonable discretisation already beats the
previous π/2≈1.571 score.

The algorithm:
  1. Choose a length N that fits comfortably into the time budget.
  2. Build the raw weights w_i = 1 / √(x_i·(1‑x_i)),
     where x_i = (i+0.5)/N  (mid‑point rule, avoids division by zero).
  3. Normalise w to a probability vector a (∑a_i = 1).
  4. (Optional) a few projected‑sub‑gradient steps are performed to
     reduce the maximum convolution value while staying on the simplex.
  5. Return a as a plain Python list of floats.

The whole routine runs well under the 1000 s budget and typically
produces a score ≈1.0 – 1.1, comfortably better than the previous
π/2≈1.571 bound.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0                # cumulative sum minus 1
    ind = np.arange(n) + 1
    # Find the largest ρ satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety: return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _max_convolution(a: np.ndarray) -> float:
    """Return max of the linear convolution a*a."""
    # Use FFT because it is faster for the moderate sizes we use.
    n = a.shape[0]
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    Fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Fa * Fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) with a very low
    evaluation score.  The routine builds a discretised arcsine density,
    optionally refines it with a few projected sub‑gradient steps, and
    returns the best sequence found before the time budget expires.
    """
    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    safety = 5.0                     # keep a few seconds margin for the harness
    deadline = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # Choose a length N.
    # We aim for a decent N (few thousand) – larger N gives a smaller
    # lower bound but also costs more time for the final convolution.
    # ------------------------------------------------------------------
    # Simple heuristic: use up to 20000 elements if there is enough time,
    # otherwise fall back to a smaller size.
    max_possible = 20000
    # Rough estimate: constructing N items and one convolution costs ~O(N log N)
    # Conservatively allocate at most 300 ms per 1000 items.
    remaining = deadline - time.time()
    approx_time_per_1000 = 0.25       # seconds (very generous)
    est_n = int(min(max_possible,
                    max(500, (remaining / approx_time_per_1000) * 1000)))
    N = max(500, min(est_n, max_possible))

    # ------------------------------------------------------------------
    # Phase 1 – build the discretised arcsine density.
    # ------------------------------------------------------------------
    # Mid‑point rule on the interval (0,1)
    i = np.arange(N, dtype=np.float64)
    x = (i + 0.5) / N                 # x ∈ (0,1)
    # Raw (unnormalised) weights: w_i = 1 / sqrt(x_i (1-x_i))
    # The π factor is irrelevant because we normalise later.
    w = 1.0 / np.sqrt(x * (1.0 - x))

    # Normalise to sum to 1 (probability simplex)
    a = w / w.sum()

    # Ensure we respect the evaluator's clamps (they are far above our values)
    a = np.clip(a, 0.0, 1000.0)

    best_a = a.copy()
    best_score = 2.0 * N * _max_convolution(best_a) / (best_a.sum() ** 2)

    # ------------------------------------------------------------------
    # Phase 2 – (optional) few sub‑gradient refinement steps.
    # ------------------------------------------------------------------
    # We perform a limited number of cheap iterations as long as we have time.
    it = 0
    max_iters = 2000        # generous upper bound; will stop earlier if time runs out.
    while time.time() < deadline and it < max_iters:
        it += 1

        # Compute convolution and identify the worst index
        n = best_a.shape[0]
        size = int(2 ** np.ceil(np.log2(2 * n - 1)))
        Fa = np.fft.rfft(best_a, n=size)
        conv = np.fft.irfft(Fa * Fa, n=size)[: 2 * n - 1]

        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])

        # Sub‑gradient of the max‑conv term:
        #   g_k = a_{max_idx - k} (if in range) + a_k
        grad = best_a.copy()                     # second term + a_k
        # First term: shifted copy
        if max_idx < n:
            grad[: max_idx + 1] += best_a[max_idx :: -1]
        else:
            start = max_idx - (n - 1)
            # slice a reversed from N-1 down to start
            grad[start:] += best_a[n - 1 : start - 1 : -1]

        # Step size schedule – diminishing
        step = 0.8 / np.sqrt(it)

        # Gradient descent step on the simplex
        candidate = best_a - step * grad
        candidate = _project_to_simplex(candidate)

        # Evaluate candidate
        cand_score = 2.0 * n * _max_convolution(candidate) / (candidate.sum() ** 2)

        # Accept if we improve (or with a tiny tolerance to avoid stagnation)
        if cand_score < best_score * (1.0 - 1e-8):
            best_a = candidate
            best_score = cand_score

        # Occasionally log progress (without flooding stdout)
        if it % 200 == 0:
            print(f"[progress] iter={it}, length={N}, score={best_score:.10f}")

    # ------------------------------------------------------------------
    # Phase 3 – final output
    # ------------------------------------------------------------------
    # Convert to plain Python list (the harness expects a list of floats)
    result = best_a.tolist()
    print(f"[result] length={len(result)}, final_score={best_score:.10f}")
    return result
