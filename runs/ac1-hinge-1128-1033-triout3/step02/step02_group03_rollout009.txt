def softmax_gradient(p, alpha):
    conv = np.convolve(p, p)
    z = alpha * conv
    max_z = np.max(z)   # biggest exponent argument
    exp_z = np.exp(z - max_z)
    w = exp_z / exp_z.sum()
    w_rev = w[::-1]
    corr = np.convolve(p, w_rev)
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * corr[idx]
    smooth_max = (np.log(exp_z.sum()) + max_z) / alpha  # smooth approximation of max conv.
    return grad, smooth_max, conv

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimizes

        M = 2·n·max(conv(p, p)) / (sum(p)**2)   (sum(p) = 1 ⇒ M = 2·n·max(conv))

The algorithm works in three stages inside the allocated time:

1. **Adam on a soft‑max approximation of the max‑operator**  
   A smooth surrogate   fα(p)= (1/α)·log Σ exp(α·conv)  is minimised.
   Its gradient is cheap (a single convolution).  Adam with a simplex
   projection gives fast, stable progress.  The smoothing factor α is
   increased geometrically so the surrogate tightens to the true max.

2. **Exact max sub‑gradient with exponentiated‑gradient updates**  
   Once the surrogate is tight we switch to the true sub‑gradient of
   max(conv).  Multiplicative (mirror‑descent) updates automatically
   respect the non‑negativity and the simplex constraint.

3. **Random sampling / tiny restarts**  
   Any remaining budget is spent on cheap random Dirichlet draws and on
   occasional jitter‑restarts when progress stalls.

All iterates are projected onto the probability simplex, guaranteeing that
the final list satisfies the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Helper: gradient of the soft‑max (log‑sum‑exp) surrogate
# ------------------------------------------------------------
def _softmax_gradient(p: np.ndarray, alpha: float):
    """
    Returns gradient of   fα(p) = (1/α)·log Σ exp(α·conv(p,p))
    and the current linear convolution conv(p,p).
    """
    n = p.shape[0]
    conv = np.convolve(p, p)               # length 2n‑1
    z = alpha * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)              # stability trick
    w = exp_z / exp_z.sum()                # soft‑max weights (sum to 1)

    # gradient = 2 * Σ_k w_k * (shifted copy of p)
    # implemented via a convolution with the reversed weight vector
    w_rev = w[::-1]
    corr = np.convolve(p, w_rev)           # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)         # indices that give the gradient
    grad = 2.0 * corr[idx]                 # shape (n,)

    return grad, conv
