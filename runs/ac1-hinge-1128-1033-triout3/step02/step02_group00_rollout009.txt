The algorithm works in three stages:

1. **Initialisation** – start from a known good construction
   (``height_sequence_1`` if it exists) or from a smooth “edge‑heavy”
   shape.  The vector is re‑parameterised as logits ``w`` and turned into
   a probability distribution ``a = softmax(w)`` (so ∑a = 1 and a ≥ 0

2. **Smoothed gradient descent** – we minimise a smooth approximation of
   the objective 2·n·max(conv(a,a)).  For a temperature parameter β we
   replace ``max`` by the smooth max

       M_β(conv) = (1/β)·log Σ_k exp(β·conv_k).

   analytically (using the Jacobian of the softmax).  The gradient step
   is performed with the Adam optimiser and the temperature β is
   increased linearly during the run, so the optimisation starts
   smooth and gradually focuses on the true max.

3. **Tracking the best discrete point** – after every iteration the
   exact objective (via ``evaluate_sequence``) is evaluated; the best
   sequence seen so far is stored and finally returned.

All operations are O(n log n) thanks to FFT‑based convolutions.

# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    shift = z - np.max(z)
    e = np.exp(shift)
    return e / e.sum()


def _next_pow2(x: int) -> int:
    """Small utility returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """
    Full convolution (length len(x)+len(y)-1) using real FFT.
    ``size`` must be a power of two ≥ len(x)+len(y)-1.
    """
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: len(x) + len(y) - 1]

        # --------------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * dw
        v = beta2 * v + (1.0 - beta2) * (dw * dw)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # learning‑rate schedule (starts larger, ends smaller)
        lr = 0.25 * (1.0 - progress) + 0.02 * progress

        w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------
        # 6️⃣  Occasional random perturbation to escape plateaus
        # --------------------------------------------------------
        if np.random.rand() < 0.005:
            # add a small Gaussian kick to a random subset of logits
            idx = np.random.choice(n, size=max(1, n // 200), replace=False)
            w[idx] += np.random.randn(len(idx)) * 0.05

        # --------------------------------------------------------
        # 7️⃣  Hard restart if we have been stuck for a long time
        # --------------------------------------------------------
        if no_improve_cnt >= RESTART_LIMIT:
            # re‑initialise logits around the current best but add noise
            w = np.log(best_a + eps) + 0.1 * np.random.randn(n)
            m.fill(0.0)
            v.fill(0.0)
            no_improve_cnt = 0
            print(f"[{t_iter:7d}] restart (no improvement)")

    # ------------------------------------------------------------
    # Final reporting using the provided evaluation routine
    # ------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return the best sequence found (as a plain Python list)
    return best_a.tolist()
