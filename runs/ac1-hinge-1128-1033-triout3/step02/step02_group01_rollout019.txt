"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a vector of heights) that
minimises the provided ``evaluate_sequence`` metric.

The algorithm combines three ideas:

1. **Seed generation**
   * Use the previously discovered construction (``height_sequence_1``)
     if it exists.
   * Generate a “flat‑convolution” seed by taking a square‑root in the
     Fourier domain of the ideal uniform self‑convolution.  After clipping
     negatives and normalising we obtain a vector whose convolution is
     already close to flat.
   * Sweep a few simple parametric families (U‑shapes) for several
     lengths and powers – a cheap deterministic sweep.

2. **Exponential‑gradient refinement**
   * The objective is
         f(a) = 2·n·max_k (a*a)[k] / (∑a)^2 .
     Because the ratio is scale‑invariant we keep the vector normalised
     (∑a = 1) and minimise ``max_k conv[k]``.
   * For a given index `k` the gradient of ``conv[k]`` w.r.t. ``a_i`` is
     ``2·a_{k‑i}`` (when the pair exists).  The sub‑gradient of the
     maximum is the sum of these gradients over all *active* indices
     ``k`` attaining the current maximum.
   * We perform exponentiated (mirror) gradient descent on the simplex:
         log_a ← log_a – η·g
         a ← softmax(log_a)                     (keeps a ≥ 0, sum = 1)
     The step size η decays as 1/√t.
   * Accepted updates follow a simulated‑annealing schedule so the
     search can escape shallow local minima.

3. **Random mass‑transfer perturbations**
   * Occasionally a simple random move (transfer a fraction of mass
     between two coordinates) is tried; it helps breaking symmetry and
     provides extra exploration.

The whole procedure respects the allotted wall‑clock budget
(`budget_s` seconds) and always returns the best sequence found so far.
"""

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w ≥ 0 , sum(w) = 1 }.
    Fast O(n log n) implementation (Wang & Carreira‑Perpiñán).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(logv: np.ndarray) -> np.ndarray:
    """
    Numerically stable softmax, returns a probability vector.
    """
    max_log = np.max(logv)
    e = np.exp(logv - max_log)
    return e / e.sum()


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14           # avoid exact zeros
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on the product of distances to both edges."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length 2·len(a).
    Returns a vector of length 2·len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())       # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a assumed non‑negative)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a vector already
    normalised to sum == 1.
    """
    n = len(a)
    return 2.0 * n * max_convolution(a)


def flat_convolution_root(n: int) -> np.ndarray:
    """
    Construct a seed whose *linear* self‑convolution would be uniform
    (all entries = 1/(2n-1)).  The exact square‑root in the Fourier domain
    inevitably yields some negative values; they are clipped to zero and the
    result is renormalised to sum = 1.
    """
    # target uniform convolution (length 2n-1)
    length_conv = 2 * n - 1
    target = np.full(length_conv, 1.0 / length_conv, dtype=np.float64)

    # pad to a power‑of‑two length for an efficient FFT
    L = 1 << ((length_conv).bit_length())
    padded = np.zeros(L, dtype=np.float64)
    padded[:length_conv] = target

    # FFT, square‑root, inverse FFT
    ft = np.fft.rfft(padded)
    # principal square‑root (complex)
    sqrt_ft = np.sqrt(ft.astype(np.complex128))

    a_full = np.fft.irfft(sqrt_ft, n=L)
    a = a_full[:n]

    # enforce non‑negativity and normalise
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.random.rand(n)
    a /= a.sum()
    return a


def subgradient_max_conv(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12):
    """
    Compute a sub‑gradient of the objective max_k conv[k] at vector a.
    ``conv`` must be the full self‑convolution of a.
    """
    max_val = conv.max()
    # treat numerical ties conservatively
    mask = conv >= max_val - max_val * eps - 1e-15
    idx = np.nonzero(mask)[0]

    n = a.shape[0]
    grad = np.zeros_like(a)

    for k in idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cur.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Anneal temperature
        # --------------------------------------------------------------
        temperature *= cooling_rate

        # --------------------------------------------------------------
        # Stagnation handling – occasional re‑heating
        # --------------------------------------------------------------
        if iteration - last_improve > 30000:
            # inject a tiny random jitter
            jitter = np.random.rand(n) * 1e-4
            cur = np.clip(cur + jitter, 0.0, None)
            cur /= cur.sum()
            log_cur = np.log(cur + 1e-20)
            cur_score = score_from_vector(cur)
            last_improve = iteration
            temperature = 1.0
            print(f"Iter {iteration:7d}: re‑heat (stagnation)")

    # ------------------------------------------------------------------
    # 4.  Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # Return a plain Python list – the external evaluator will clip/normalise.
    return best_seq.tolist()
