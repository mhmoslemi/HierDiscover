"""
propose_candidate – Find a low‑value step‑function for the evaluation metric.

Algorithm
---------
1. Choose a length `n` (2048 works well – large enough to give a tight bound,
   still cheap to convolve with NumPy’s C‑implementation).

2. Initialise the probability vector `p` with a discretised arcsine law:
      p_i ∝ 1 / sqrt(x_i (1‑x_i)),   x_i = (i+0.5)/n .
   The vector is normalised to sum to one (the simplex).

3. Optimise the *smooth* surrogate of the max‑convolution:
      f(p) = (1/α)·log Σ_k exp(α·(p∗p)_k).
   The gradient w.r.t. the soft‑max parameters `u` (where p = softmax(u))
   is derived analytically (including the missing `p_i` term that was omitted
   in the reference implementation).
   An Adam optimiser (first‑order momentum) is used and the smoothing
   parameter `α` is slowly increased (α ← α·α_mul).

4. When the time budget allows, switch to the exact sub‑gradient of the
   true max‑convolution and perform multiplicative (exponentiated‑gradient)
   updates, which keep the iterates on the simplex automatically.
   A tiny amount of Gaussian jitter and occasional random restarts help
   escape plateaus.

5. The best vector encountered during the whole run is kept, clipped to the
   allowed range [0,1000] and returned as a plain Python list.

The whole routine respects the 1000 s budget (a 5 s safety margin) and
produces a sequence with a much lower evaluation score than the previous
run (target ≈ 1.503).  All operations are NumPy‑based and avoid any
file‑ or network‑I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using NumPy (fast C implementation)."""
    return np.convolve(a, b)


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a step‑function (non‑negative heights) that minimises the
    evaluation function.  Returns a plain Python list of floats."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – a length a little larger than the historic 1153.
    # ------------------------------------------------------------------
    n = 2048                      # a power of two is convenient for FFT later

    # ------------------------------------------------------------------
    # 2. Initialise with a discretised arcsine law (Beta(½,½) density).
    # ------------------------------------------------------------------
    idx = np.arange(n) + 0.5
    x = idx / n                                   # points in (0,1)
    # avoid exact 0/1 to keep finite values
    eps = 1e-12
    p_raw = 1.0 / np.sqrt(x * (1.0 - x) + eps)    # 1/√(x(1−x))
    p = p_raw / p_raw.sum()                       # normalise to the simplex

    # we keep the best solution found so far
    best_p = p.copy()
    conv = _linear_convolution(p, p)
    best_score = 2.0 * n * np.max(conv)

    print(f"[Init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # 4. Phase‑1: smoothed objective minimisation (Adam on log‑softmax)
    # ------------------------------------------------------------------
    # Parameterisation: p = softmax(u)
    u = np.log(p + 1e-15)               # small offset to avoid log(0)

    # Adam hyper‑parameters
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # soft‑max temperature (α in the analysis)
    alpha = 5.0
    alpha_mul = 1.02   # slowly tighten the approximation

    it = 0
    eps_adam = 1e-8

    while time.time() < deadline:
        it += 1

        # ---------- current probabilities ----------
        p = np.exp(u)
        p /= p.sum()                       # soft‑max, guarantees simplex

        # ---------- convolution p * p ----------
        conv = _linear_convolution(p, p)    # length 2n‑1

        # ---------- smooth max (soft‑max) ----------
        z = alpha * conv
        w = _softmax(z)                    # weights sum to 1, length 2n‑1

        # ---------- gradient of the smooth surrogate ----------
        # s_i = Σ_k w_k p_{k-i}  (i–dependent weighted sum)
        w_rev = w[::-1]
        corr = _linear_convolution(p, w_rev)          # length 3n‑2
        # extracting s_i = corr[(2n‑2)‑i] for i=0..n‑1
        s = corr[(2 * n - 2) - np.arange(n)]

        grad_p = p + s                     # true gradient wrt p (including the missing +p term)

        # ---------- map gradient to the u‑space (soft‑max Jacobian) ----------
        inner = np.dot(p, grad_p)          # scalar ⟨p,∇_p⟩
        grad_u = p * (grad_p - inner)      # ∇_u f = Jᵀ ∇_p f

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- anneal smoothing ----------
        alpha *= alpha_mul

        # occasional jitter to avoid stagnation
        if it % 500 == 0:
            u += rng.normal(scale=0.02, size=n)

        # ---------- keep the best pure objective ----------
        cur_score = 2.0 * n * np.max(conv)   # *exact* objective, not the surrogate
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[Phase‑1] Iter {it:6d} – new best score: {best_score:.6f}")

        # early stop if we have reached the target region
        if best_score <= 1.55:
            break

    # ------------------------------------------------------------------
    # 5. Phase‑2: exact max sub‑gradient with exponentiated updates
    # ------------------------------------------------------------------
    p = best_p.copy()
    eta = 0.7                     # step size for multiplicative updates
    eta_decay = 0.9995
    sub_it = 0
    no_improve = 0
    while time.time() < deadline:
        sub_it += 1

        conv = _linear_convolution(p, p)
        k_max = int(np.argmax(conv))

        # ----- exact sub‑gradient of max(conv) at the active index -----
        grad = np.zeros_like(p)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        if i_start <= i_end:
            idx_range = np.arange(i_start, i_end + 1)
            # grad_i = p_i + p_{k_max-i}
            grad[idx_range] = p[idx_range] + p[k_max - idx_range]

        # ----- exponentiated (mirror) update -----
        p *= np.exp(-eta * grad)
        s = p.sum()
        if s == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p /= s

        # occasional jitter (helps escape shallow plateaus)
        if sub_it % 400 == 0:
            p = _simplex_projection(p + rng.normal(scale=0.001, size=n))

        eta *= eta_decay

        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
            total = it + sub_it
            print(f"[Phase‑2] Iter {total:6d} – new best score: {best_score:.6f}")
        else:
            no_improve += 1

        # restart if stuck for a long time
        if no_improve >= 3000:
            p = best_p + rng.normal(scale=0.005, size=n)
            p = np.clip(p, 0.0, None)
            p /= p.sum()
            no_improve = 0

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 6. Final quick random sampling (optional, uses any remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # draw a tiny Dirichlet‑like point
        p_rand = rng.random(n)
        p_rand /= p_rand.sum()
        score_rand = 2.0 * n * np.max(_linear_convolution(p_rand, p_rand))
        if score_rand < best_score:
            best_score = score_rand
            best_p = p_rand.copy()
            print(f"[Random] New best score: {best_score:.6f}")
        # break if only a trivial amount of time remains
        if time.time() + 0.02 > deadline:
            break

    # ------------------------------------------------------------------
    # 7. Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Best score achieved: {best_score:.6f}")
    return result
