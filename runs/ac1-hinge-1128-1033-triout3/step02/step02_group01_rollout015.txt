def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0
    best_seq = None
    best_score = np.inf

    # candidate lengths to try
    lengths = [2000, 4000, 8000, 12000, 20000, 40000] maybe truncated by time.

    for n in lengths:
        if time.time() > deadline:
            break
        # init sequence: start from best known or u_min shape p ~0.5
        if 'height_sequence_1' in globals():
            init = np.array(height_sequence_1, dtype=float)
            # if not same size, resample
            if init.size != n:
                # interpolate to size n
                x_old = np.linspace(0,1,init.size)
                x_new = np.linspace(0,1,n)
                init = np.interp(x_new, x_old, init)
        else:
            # start with u_min with p ~0.5 as baseline
            init = u_min_distribution(n, p=0.6) # guess
        # ensure sum=1 
        init = np.maximum(init,0)
        init_sum = init.sum()
        if init_sum==0: init = np.ones(n)/n
        else: init /= init_sum

        # run softmax gradient descent for a while
        seq, score = run_softmax_optim(init, n, deadline)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"new best after softmax length {n} score {score:.6f}")
    # final: return best_seq.tolist()
    return best_seq.tolist()

def optimize_sequence(a_init, deadline):
    a = a_init.copy()
    a = np.maximum(a, 0)
    a /= a.sum()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * full_convolution_fft(a).max()
    iteration = 0
    beta = 0.5
    beta_max = 40.0
    eta0 = 5.0
    # optionally Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    while time.time() < deadline:
        iteration += 1
        # compute conv once
        conv = full_convolution_fft(a)
        max_val = conv.max()
        # update beta (annealing)
        if iteration % 200 == 0 and beta < beta_max:
            beta *= 1.05
        # compute softmax weights
        # To avoid overflow, subtract max*beta as shift
        # Already have max_val, shift = max_val
        # compute exp(beta*(conv - max_val))
        exp_term = np.exp(beta * (conv - max_val))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp
        # compute gradient
        grad = gradient_smooth_max(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = eta0 / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat  # note: step is vector
        # projection
        a_new = simplex_projection(a_new)
        # evaluate
        conv_new = full_convolution_fft(a_new)
        score_new = 2.0 * n * conv_new.max()
        if score_new < best_score:
            best_score = score_new

"""
Propose a low‑score step‑function.

Algorithm outline
-----------------
1. Choose a moderate length (default 20 000).  Larger n gives a slightly
   better lower bound but costs more FFT work.
2. Build a handful of diverse starting vectors:
   * two parametric U‑shapes (distance‑to‑edge and product‑distance)
   * a random Dirichlet vector
   * the external construction (if the global variable `height_sequence_1`
     exists) interpolated to the chosen length.
3. For each start run a smooth‑max optimiser on the simplex.
   * The objective fβ(x) = (1/β)·log∑exp(β·(x*x)) is a smooth approximation
     of maxₖ (x*x)[k].  Its gradient is
        ∇fβ = 2·(x ⋆ w)   with   wₖ ∝ exp(β·(x*x)ₖ),
     where "⋆" denotes the linear correlation.
   * The gradient is evaluated with two FFT‑based convolutions:
        – self‑convolution (x*x) to obtain w,
        – correlation of x with w (implemented as a convolution with w reversed).
   * An Adam‑style update is applied and the iterate is projected back onto
     the probability simplex {x≥0, Σx=1}.
   * β starts small (≈0.5) and is slowly increased (≈1 % every few hundred
     iterations) so that early steps act on a smooth landscape and later
     steps focus on the true supremum norm.
   * If the iterate stalls for many iterations a tiny random jitter is added
     (re‑heating) and the Adam moments are reset.
4. The best vector seen over all runs (lowest exact score
   `2·n·max(x*x)`) is returned.  The external evaluator will clip and
   normalise again, so we return the raw vector.

The whole routine respects the given time budget; each optimisation run
gets at most 30 s (or the remaining budget) and the main loop stops before
the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {w ≥ 0, sum(w) = 1}.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using real FFT.
    Returns a vector of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on distance to the nearest edge:
        a[i] = (min(i, n-1-i) + 1)^{-p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14           # avoid exact zeros
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on product of distances to both edges:
        a[i] = ((i+1)*(n-i))^{-p}
    Normalised to sum = 1.
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max fβ(x) = (1/β)·log∑exp(β·(x*x))[k]
    with respect to a, given the soft‑max weights w (same length as a*a).
    Implements   ∇fβ = 2·(a ⋆ w)
    where ⋆ is the linear correlation.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reversed weight vector
    size = next_pow2(n + w_rev.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(fa * fw, n=size)
    conv = conv[: n + w_rev.shape[0] - 1]   # keep the linear part
    offset = w_rev.shape[0] - 1            # = 2*n-2
    # grad_i = 2 * conv[offset - i]
    idx = offset - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad


def smooth_max_step(a: np.ndarray, beta: float, eta: float,
                    m: np.ndarray, v: np.ndarray, it: int):
    """
    Perform one Adam‑style step on the smooth‑max surrogate.
    Returns the new iterate, its exact score, and updated Adam moments.
    """
    # 1) self‑convolution of a
    conv = full_convolution_fft(a)
    max_val = conv.max()

    # 2) soft‑max weights (stable formulation)
    exp_term = np.exp(beta * (conv - max_val))
    w = exp_term / exp_term.sum()

    # 3) gradient of the smooth‑max
    grad = gradient_smooth_max(a, w)

    # 4) Adam update
    m = 0.9 * m + 0.1 * grad
    v = 0.999 * v + 0.001 * (grad * grad)
    m_hat = m / (1 - 0.9 ** it)
    v_hat = v / (1 - 0.999 ** it)

    # 5) step and projection onto the simplex
    step = eta * m_hat / (np.sqrt(v_hat) + 1e-8)
    a_next = a - step
    a_next = simplex_projection(a_next)

    # 6) exact (non‑smooth) score for monitoring
    exact_score = 2.0 * a_next.shape[0] * full_convolution_fft(a_next).max()
    return a_next, exact_score, m, v


def optimise_on_simplex(a0: np.ndarray, deadline: float):
    """
    Gradient descent on the smooth‑max surrogate with Adam momentum.
    Returns the best vector seen before the deadline and its exact score.
    """
    a = a0.copy()
    n = a.shape[0]

    # best‑ever record
    best_a = a.copy()
    best_score = 2.0 * n * full_convolution_fft(a).max()

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    beta = 0.5
    beta_max = 30.0
    eta0 = 2.0                     # base learning rate
    stagnation = 0                 # iterations without improvement

    while time.time() < deadline:
        it += 1

        # slowly increase the smoothness parameter β
        if it % 200 == 0 and beta < beta_max:
            beta *= 1.03

        # step‑size decay
        eta = eta0 / (1.0 + it * 1e-4)

        a, score, m, v = smooth_max_step(a, beta, eta, m, v, it)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            stagnation = 0
            print(f"Iter {it:7d}: new best score {best_score:.6f} (β={beta:.2f})")
        else:
            stagnation += 1

        # occasional random re‑heating to escape plateaus
        if stagnation > 12000:
            jitter = np.random.rand(n) * 5e-5
            a = simplex_projection(a + jitter)
            m.fill(0.0)
            v.fill(0.0)
            stagnation = 0

    return best_a, best_score


def propose_candidate(seed: int = 2023, budget_s: int = 1000):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the given time budget and
    returns the best sequence it discovered as a plain Python list.
    """
    np.random.seed(seed)
    # safety margin for the host runner
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose a problem size
    # ------------------------------------------------------------------
    # 20 000 is a good trade‑off between flexibility and speed.
    # If a lot of time is left we automatically switch to 40 000.
    remaining_time = deadline - time.time()
    if remaining_time > 600:
        n = 40000
    else:
        n = 20000

    # ------------------------------------------------------------------
    # 2. Build a set of diverse initial vectors
    # ------------------------------------------------------------------
    initials = []

    # a) U‑shape (min‑distance) with a few exponents
    for p in (0.5, 0.8, 1.2):
        initials.append(u_min_distribution(n, p))

    # b) U‑shape (product‑distance)
    for p in (0.5, 0.8, 1.2):
        initials.append(u_prod_distribution(n, p))

    # c) Random Dirichlet (flat but noisy)
    initials.append(np.random.dirichlet(np.ones(n)))

    # d) External construction, interpolated to length n if available
    try:
        ext = np.array(height_sequence_1, dtype=float)
        if ext.size > 0:
            x_old = np.linspace(0.0, 1.0, ext.size)
            x_new = np.linspace(0.0, 1.0, n)
            ext_interp = np.interp(x_new, x_old, ext)
            initials.append(ext_interp)
    except NameError:
        pass

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 3. Optimise each start (time‑sliced)
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(initials):
        if time.time() >= deadline:
            break

        # normalise onto the simplex
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s <= 0.0:
            a0 = np.ones_like(a0) / a0.size
        else:
            a0 = a0 / s

        # allocate a modest slice of the remaining budget
        slice_deadline = min(deadline, time.time() + 30.0)

        print(f"Start optimisation {idx + 1}/{len(initials)} (n={n})")
        cand, cand_score = optimise_on_simplex(a0, slice_deadline)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"  >>> Global best updated: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = 2.0 * n * full_convolution_fft(best_seq).max()
        print("Fallback to uniform distribution.")

    print(f"Search finished – best score {best_score:.6f}")
    # The external evaluator will re‑normalise, so we simply return the raw vector.
    return best_seq.tolist()
