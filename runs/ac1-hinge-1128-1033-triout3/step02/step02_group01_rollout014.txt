    golden ratio phi = (sqrt(5)-1)/2 ~0.618

"""
Propose a low‑score step‑function.

The algorithm consists of three stages:

1. **Parametric sweep** – For several lengths we minimise the score
   over the two classic “U‑shaped’’ families (min‑distance and
   product‑distance) using a cheap golden‑section line search with a
   fine p‐resolution (≈10⁻³).  The best vector from this stage is used as
   a seed for the next stage.

2. **Momentum sub‑gradient descent with simulated annealing** –  The
   (non‑smooth) objective 2·n·max(a∗a) is minimised on the simplex.
   A projected (additive) sub‑gradient step with exponential
   moving‑average momentum is combined with a multiplicative (mirror)
   step and occasional random mass‑transfer moves.  The temperature is
   slowly decreased; if the search stagnates we re‑heat by injecting a
   tiny random bump.

3. **Return the best sequence** –  The evaluator will re‑normalise and
   clip the sequence, therefore we simply return the raw (already
   normalised) vector.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w >= 0, sum(w) = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape based on distance to the nearest edge."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14           # tiny epsilon avoids exact zeros
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape based on the product of distances to both edges."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a∗a using a single FFT (real‑valued)."""
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()    # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a (a is assumed non‑negative and sum(a)=1)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a vector already
    normalised to sum == 1.
    """
    n = len(a)
    max_c = max_convolution(a)
    return 2.0 * n * max_c


def golden_section_search(func, lo: float, hi: float,
                          tol: float = 1e-3, max_iter: int = 30):
    """
    Minimise a unimodal scalar function on [lo, hi] using golden‑section.
    Returns the best argument found.
    """
    phi = (np.sqrt(5) - 1) / 2      # ≈0.618
    a, b = lo, hi
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if (b - a) < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
    return (a + b) / 2.0


# ----------------------------------------------------------------------
# Main proposal routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the given time budget and
    returns the best sequence (list of floats) it discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # safety margin

    # --------------------------------------------------------------
    # 1.  Refined parametric sweep over U‑shapes
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # lengths to try – include a fairly large length to approach the
    # continuous limit, but keep them manageable for the sweep.
    lengths = [2000, 5000, 8000, 12000, 20000, 30000, 50000]

    # search interval for the exponent p (both families behave similarly)
    p_lo, p_hi = 0.02, 0.48          # avoid the divergent regime p>=0.5

    print("=== Refined parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # ---- min‑distance family ----
        def score_min(p):
            a = u_min_distribution(n, p)
            return score_from_vector(a)

        p_opt_min = golden_section_search(score_min, p_lo, p_hi, tol=1e-4)
        a_min = u_min_distribution(n, p_opt_min)
        sc_min = score_from_vector(a_min)
        if sc_min < best_score:
            best_score, best_seq = sc_min, a_min.copy()
            print(f" new best (min)  n={n:5d} p={p_opt_min:6.4f}  score={sc_min:.6f}")

        # ---- product‑distance family ----
        def score_prod(p):
            a = u_prod_distribution(n, p)
            return score_from_vector(a)

        p_opt_prod = golden_section_search(score_prod, p_lo, p_hi, tol=1e-4)
        a_prod = u_prod_distribution(n, p_opt_prod)
        sc_prod = score_from_vector(a_prod)
        if sc_prod < best_score:
            best_score, best_seq = sc_prod, a_prod.copy()
            print(f" new best (prod) n={n:5d} p={p_opt_prod:6.4f}  score={sc_prod:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2.  Include external construction if available
    # --------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            if external.sum() > 0:
                external = external / external.sum()
                sc_ext = score_from_vector(external)
                if sc_ext < best_score:
                    best_score, best_seq = sc_ext, external.copy()
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3.  Fallback (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Stochastic refinement (momentum sub‑gradient + random moves)
    # --------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score
    n = len(current)

    temperature = 1.0
    cooling_rate = 0.9995
    step_size = 0.15
    step_decay = cooling_rate
    momentum = np.zeros_like(current)
    beta = 0.9                     # exponential moving‑average momentum
    iteration = 0
    last_improve_iter = 0

    print("\n=== Starting stochastic refinement ===")
    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # 4a – pick a move type
        # ----------------------------------------------------------
        rnd = np.random.rand()
        if rnd < 0.35:
            # ----- random mass transfer (large‑scale) -----
            # pick source from the top 10 % and target from the bottom 10 %
            top_k = max(1, n // 10)
            bot_k = max(1, n // 10)
            src_idx = np.argpartition(-current, top_k)[:top_k]
            tgt_idx = np.argpartition(current, bot_k)[:bot_k]
            i = np.random.choice(src_idx)
            j = np.random.choice(tgt_idx)
            if current[i] == 0.0:
                continue
            delta = current[i] * np.random.rand() * 0.7   # move up to 70 % of source mass
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            # sum stays 1 → no projection needed
        elif rnd < 0.7:
            # ----- additive projected sub‑gradient step with momentum -----
            conv = full_convolution_fft(current)
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-15
            active = np.where(conv >= max_val - eps)[0]

            grad = np.zeros_like(current)
            for k in active:
                start_i = max(0, k - (n - 1))
                end_i = min(n - 1, k)
                i_vals = np.arange(start_i, end_i + 1)
                j_vals = k - i_vals
                grad[i_vals] += 2.0 * current[j_vals]

            # exponential moving‑average momentum
            momentum = beta * momentum + (1.0 - beta) * grad

            cand = current - step_size * momentum
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)
        else:
            # ----- multiplicative (mirror‑descent) sub‑gradient step -----
            conv = full_convolution_fft(current)
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-15
            active = np.where(conv >= max_val - eps)[0]

            grad = np.zeros_like(current)
            for k in active:
                start_i = max(0, k - (n - 1))
                end_i = min(n - 1, k)
                i_vals = np.arange(start_i, end_i + 1)
                j_vals = k - i_vals
                grad[i_vals] += 2.0 * current[j_vals]

            # mirror‑descent update: a_i ← a_i * exp(-η·g_i)
            exponent = -step_size * grad
            # to avoid underflow/overflow clip exponent
            exponent = np.clip(exponent, -50, 50)
            cand = current * np.exp(exponent)
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                cand = np.full_like(cand, 1.0 / n)
            else:
                cand = cand / cand_sum

        # ----------------------------------------------------------
        # 4b – evaluate candidate (cheap internal score)
        # ----------------------------------------------------------
        cand_score = score_from_vector(cand)

        # ----------------------------------------------------------
        # 4c – Simulated‑annealing acceptance
        # ----------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # 4d – anneal parameters
        # ----------------------------------------------------------
        temperature *= cooling_rate
        step_size *= step_decay

        # ----------------------------------------------------------
        # 4e – stagnation handling (re‑heat occasionally)
        # ----------------------------------------------------------
        if iteration - last_improve_iter > 15000:
            # inject a tiny random bump to escape a flat region
            bump = np.random.rand(n) * 1e-4
            current = current + bump
            current = simplex_projection(current)
            current_score = score_from_vector(current)
            last_improve_iter = iteration
            temperature = 1.0   # fresh temperature

    # --------------------------------------------------------------
    # 5.  Return the best sequence found
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will clip and normalise again,
    # so we just return the raw normalised vector.
    return best_seq.tolist()
