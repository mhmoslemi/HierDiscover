def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) convolution a*a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:m]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _arcsine_initial(n: int) -> np.ndarray:
    """Return a normalised arcsine‑shaped vector of length n.

    a_i ∝ 1 / sqrt((i+1)*(n-i))  (the +1 / -1 shift avoids division by zero).
    The vector is normalised to have unit sum.
    """
    i = np.arange(n, dtype=np.float64)
    # add a tiny epsilon to avoid infinities at the ends
    eps = 1e-12
    a = 1.0 / np.sqrt((i + eps + 1) * (n - i + eps))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # --------------------------------------------------
        # occasional random mass‑transfer jump (simulated annealing)
        # --------------------------------------------------
        if np.random.rand() < jump_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move at most 30 % of the source mass
            delta = np.random.rand() * 0.3 * a[src]
            a[src] -= delta
            a[dst] += delta
            # re‑normalise (numerical drift may happen)
            a = a / a.sum()

        # --------------------------------------------------
        # periodic LP refinement
        # --------------------------------------------------
        if iteration % lp_interval == 0:
            # current bound M
            M = max_conv
            # Build the Toeplitz constraint matrix A where
            #   (A @ g)[k] = (a * g)[k]
            rows = []
            cols = []
            data = []
            for k in range(2 * n - 1):
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                # all i such that a_i participates in shift k
                # column index for g is j = k - i
                for i_idx in range(i_start, i_end + 1):
                    rows.append(k)
                    cols.append(k - i_idx)
                    data.append(a[i_idx])
            A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

            # linear programme: maximise sum(g)  <=> minimise -sum(g)
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0, None)] * n
            # give the solver a small time budget to stay within the overall limit
            lp_time_left = max(0.0, deadline - time.time() - 0.2)
            options = {"time_limit": lp_time_left}
            try:
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                    bounds=bounds,
                    method="highs",
                    options=options,
                )
            except Exception as exc:  # pragma: no cover
                print(f"[LP] unexpected exception: {exc}")
                res = None

            if res is not None and res.success:
                g_opt = res.x
                if g_opt.sum() > 0:
                    g_opt = g_opt / g_opt.sum()
                # blend the LP solution into the current iterate
                a_new = (1.0 - lp_mix) * a + lp_mix * g_opt
                a_new = a_new / a_new.sum()
                # quick evaluation of the candidate
                conv_new = _full_convolution(a_new)
                val_new = float(2.0 * n * conv_new.max())
                if val_new < best_val:
                    best_val = val_new
                    best_a = a_new.copy()
                    print(f"[LP {iteration:7d}] improved best to {best_val:.6f}")
                a = a_new
            else:
                # LP failed – continue with the gradient steps
                if res is not None:
                    print(f"[LP {iteration:7d}] failed: {res.message}")

    # --------------------------------------------------
    # Final reporting
    # --------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
