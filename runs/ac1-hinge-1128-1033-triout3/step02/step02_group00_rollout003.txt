"""
Propose a candidate step sequence that minimizes ``evaluate_sequence``.

The algorithm is a projected accelerated gradient descent on a smooth
approximation of the max‑self‑convolution objective:

* The current self‑convolution `c = a * a` is computed with a real FFT.
* A soft‑max (log‑sum‑exp) weighting `w` is built from `c`.  As the
  temperature (`beta`) grows the weights concentrate on the indices where
  the convolution is maximal, so the gradient gradually approaches the
  true sub‑gradient of the ``max`` operator.
* The gradient of the smooth objective is
      g = 2·(w_rev * a)[n‑1 : n‑1+n]                    (full convolution)
  where `w_rev` is `w` reversed.
* A Nesterov‑style momentum term is used, the iterate is projected onto
  the probability simplex (non‑negative and sum = 1) after each step.
* The step size `eta`, momentum coefficient `mu` and soft‑max temperature
  `beta` follow linear schedules that end with a very small step size,
  zero momentum and a large `beta` (close to the true max).
* A tiny random mass‑transfer is performed occasionally to help escape
  plateaus.
* After the main optimisation phase a short local‑search phase performs
  random pairwise transfers and keeps improvements.

The routine respects the 1000 s time budget, prints occasional progress
messages and finally returns the best sequence found (as a plain list of
floats).  The returned sequence is guaranteed to be non‑negative and
its sum is one, which makes the evaluation equal to `2·n·max(conv)`.
"""

import numpy as np

# ------------------------------------------------------------
# Simplex projection (Duchi et al.)
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ------------------------------------------------------------
# Full convolution via real FFT (any two vectors)
# ------------------------------------------------------------
def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (len(x)+len(y)-1) convolution of x and y using FFT."""
    m = x.shape[0] + y.shape[0] - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:m]
    # Clip tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ------------------------------------------------------------
# Self‑convolution (a * a) shortcut
# ------------------------------------------------------------
def _self_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    np.maximum(conv, 0, out=conv)
    return conv

# Main entry point required by the competition framework
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) gives a low score."""
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – leave a couple of seconds for final I/O
    overall_deadline = start_time + budget_s - 2.0
    # keep a few seconds for a tiny local‑search refinement
    main_deadline = overall_deadline - 5.0

    # --------------------------------------------------------
    # Initialise the sequence a
    # --------------------------------------------------------
    try:
        # Use the best sequence found so far (if the variable exists)
        a = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = a.shape[0]
    except Exception:
        # Fallback: a modestly large edge‑heavy shape
        n = 2000
        x = np.arange(n, dtype=float)
        # A simple “U‑shape”: high at the ends, low in the middle
        a = 1.0 - np.abs(2.0 * x / (n - 1) - 1.0)   # triangular (peak in the centre)
        a = np.maximum(a, 0.0)

    # Normalise to a probability simplex (sum = 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_a = a.copy()
    conv = _self_convolution(a)
    best_val = float(2.0 * n * conv.max())
    print(f"[init] n={n}, initial evaluation={best_val:.6f}")

    # Momentum buffer for Nesterov‑type acceleration
    v = np.zeros_like(a)

    # Linear schedules (progress ∈ [0,1])
    eta_start, eta_end = 0.5, 1e-4          # step‑size
    mu_start, mu_end = 0.9, 0.0            # momentum decay
    beta_start, beta_end = 0.5, 200.0      # soft‑max “temperature”

    iteration = 0
    # --------------------------------------------------------
    # Main optimisation loop (projected accelerated gradient)
    # --------------------------------------------------------
    while True:
        now = time.time()
        if now >= main_deadline:
            break

        # ----- evaluate current objective -----
        conv = _self_convolution(a)
        cur_max = conv.max()
        cur_val = float(2.0 * n * cur_max)          # sum(a)=1 ⇒ denominator =1
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ----- schedules -----
        progress = (now - start_time) / (overall_deadline - start_time)
        eta = eta_start * (1.0 - progress) + eta_end * progress
        mu = mu_start * (1.0 - progress) + mu_end * progress
        beta = beta_start + (beta_end - beta_start) * progress   # increase beta

        # ----- smooth max weights (soft‑max) -----
        # shift by max for numerical stability
        shifted = conv - cur_max
        weights = np.exp(beta * shifted)
        weights_sum = weights.sum()
        if weights_sum == 0:
            # degenerate case – fall back to hard max
            weights = np.zeros_like(conv)
            weights[np.argmax(conv)] = 1.0
        else:
            weights /= weights_sum

        # ----- gradient of the smoothed objective -----
        w_rev = weights[::-1]                     # reverse for convolution
        # convolution w_rev * a
        grad_full = _full_convolution(w_rev, a)   # length = (2n‑1)+n‑1 = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # slice of length n

        # ----- momentum / gradient step -----
        v = mu * v - eta * grad
        a_candidate = a + v

        # ----- occasional random mass‑transfer (helps escape plateaus) -----
        if np.random.rand() < 0.02:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            # move up to 50 % of the source mass
            delta = np.random.rand() * a_candidate[i] * 0.5
            a_candidate[i] -= delta
            a_candidate[j] += delta

        # ----- project back onto the simplex -----
        a = _project_onto_simplex(a_candidate)

        iteration += 1

        # occasional full recomputation of the convolution (drift guard)
        if iteration % 5000 == 0:
            conv = _self_convolution(a)

    # --------------------------------------------------------
    # Local refinement: tiny random pairwise transfers
    # --------------------------------------------------------
    while time.time() < overall_deadline:
        # choose a random source and destination
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # try moving a small fraction of the source mass
        epsilon = 0.01 * np.random.rand()   # up to 1 % of the source
        delta = epsilon * best_a[i]
        if delta <= 0:
            continue
        trial = best_a.copy()
        trial[i] -= delta
        trial[j] += delta
        # project is not required – the move preserves the sum and stays non‑negative
        # evaluate quickly using the FFT‑based convolution
        conv_trial = _self_convolution(trial)
        val_trial = float(2.0 * n * conv_trial.max())
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial
            print(f"[refine] improved to {best_val:.6f}")

    # --------------------------------------------------------
    # Final reporting using the provided evaluation routine
    # --------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
