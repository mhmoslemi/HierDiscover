"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Fast parametric sweep** – generate several families of
   symmetric distributions (exponential, power‑law “U‑shape” and
   sum‑of‑tails) for a handful of lengths.  The convolution maximum is
   evaluated with a real‑FFT routine (`max_conv_fft`), which is far
   faster than the naive `np.convolve`.  The best candidate from this
   sweep is kept as a warm‑start.

2. **Mirror‑descent on the simplex** – starting from the warm‑start we
   minimise the L₂‑distance between the self‑convolution and the
   uniform value `1/(2n‑1)`.  For a vector `a` (∑a = 1) the loss is

        L(a) = Σ_k ( (a*a)[k] – 1/(2n‑1) )² .

   Its gradient on the simplex is

        ∇L = 4 * ( ( (a*a) – avg )  *  a_rev )_slice

   where `*` denotes convolution and `a_rev` is `a` reversed.
   A multiplicative (exponentiated‑gradient) update
   `a ← a * exp(‑η ∇L)` keeps the vector non‑negative; afterwards we
   renormalise to the simplex.  The step size η follows a
   1/√t schedule and the exponent is clipped to avoid overflow.
   The loop runs until the time budget is exhausted (≈ 1000 s) and we
   keep the best vector seen (according to the true objective
   `2·n·max(a*a)`).

3. **Final exact evaluation** – the best vector is evaluated once with
   the provided ``evaluate_sequence`` to obtain the score that will be
   reported.

The code respects the 1000 s budget, prints progress, and returns a
plain Python list of non‑negative heights.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def max_conv_fft(a: np.ndarray) -> float:
    """
    Maximum entry of the (linear) self‑convolution a*a using real FFT.
    The vector `a` may contain any real numbers; the function works for
    both the fast sweep (where a is already normalised) and the inner
    optimisation loop.
    """
    n = a.shape[0]
    size = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12           # avoid exact zeros
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a
