"""
Propose a low‐evaluation step sequence.

The evaluation function (already provided) is
    2·n·max(self‑convolution) / (sum a)²
where all aᵢ are forced non‑negative and are clipped to [0,1000].
If we normalise the vector to unit sum, the denominator becomes 1 and the
objective is simply 2·n·max(conv(a,a)).  The problem is convex:
minimise a scalar t subject to (a*a)[k] ≤ t for every k, aᵢ ≥ 0,
∑aᵢ = 1.  We therefore perform a projected (Adam) gradient descent on the
convex but non‑smooth objective max(conv).

Key ideas beyond the baseline implementation:

* **Active‑set sub‑gradient** – instead of using the gradient of a single
  maximal index we average the sub‑gradients of all indices whose convolution
  value is within a small relative tolerance of the maximum.  This yields a
  smoother descent direction.

* **Adam‑style adaptive step sizes** – first‑ and second‑moment estimates
  accelerate convergence while respecting the simplex constraint.

* **Multi‑scale optimisation** – we start with a modest length (n≈500) where
  iterations are cheap, obtain a good shape, then linearly upscale the vector
  to larger lengths (≈2000 and ≈6000).  The shape refined at a coarse scale is
  an excellent warm‑start for the finer scales.

* **Simplex projection** – after each update we project back onto the
  probability simplex (Duchi et al., 2008) to enforce non‑negativity and unit
  sum.

* **Occasional random mass‑transfer** – a tiny simulated‑annealing jump helps
  escape plateaus.

The routine spends the allotted time budget (≈1000 s) on these stages,
keeps the best sequence found, and finally reports its value via the
provided ``evaluate_sequence`` function.

The algorithm runs on a single CPU and uses only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector v onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full linear convolution via FFT (real, O(N log N))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    # Clip tiny negatives from numerical noise
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper: upscale a vector to a new length by linear interpolation
# ----------------------------------------------------------------------
def _upscale(vec: np.ndarray, new_len: int) -> np.ndarray:
    """Resize ``vec`` to ``new_len`` entries by linear interpolation."""
    old_len = vec.shape[0]
    if old_len == new_len:
        return vec.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, vec)

# ----------------------------------------------------------------------

    raw objective 2·n·max(conv).
    """
    # initialise
    a = a_init.astype(float, copy=False)
    a = np.maximum(a, 0.0)
    a = a / a.sum()                     # guarantee unit sum

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # best‑so‑far within this stage
    conv = _full_convolution(a)
    best_obj = float(2.0 * n * conv.max())
    best_a = a.copy()

    # constants for step‑size schedule
    lr_start = 0.20
    lr_end = 1e-5

    while time.time() < stage_deadline:
        # -------------------------------------------------------------
        # evaluate current objective
        # -------------------------------------------------------------
        conv = _full_convolution(a)
        cur_max = conv.max()
        cur_obj = float(2.0 * n * cur_max)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[stage n={n}]  new best = {best_obj:.6f}")

        # -------------------------------------------------------------
        # active‑set sub‑gradient (average over near‑max indices)
        # -------------------------------------------------------------
        # Relative tolerance shrinks linearly with time – wide at start,
        # tight near the end.
        elapsed = time.time() - global_start
        total = global_deadline - global_start
        progress = elapsed / total
        rel_eps = 0.05 * (1.0 - progress) + 0.005 * progress   # 5% → 0.5%

        thresh = cur_max * (1.0 - rel_eps)
        active_idx = np.where(conv >= thresh)[0]

        # safeguard: keep at most 200 active positions
        if active_idx.size > 200:
            active_idx = np.argsort(-conv)[:200]

        subgrad = np.zeros_like(a)
        for k in active_idx:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1, dtype=int)
            subgrad[idx] += 2.0 * a[k - idx]   # g_i = 2·a_{k-i}
        # average over the active set
        subgrad /= max(1, active_idx.size)

        # -------------------------------------------------------------
        # occasional random mass‑transfer (simulated annealing jump)
        # -------------------------------------------------------------
        if np.random.rand() < 0.01:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = a[src] * np.random.rand() * 0.5
            a[src] -= delta
            a[dst] += delta

        # -------------------------------------------------------------
        # Adam update + projection onto simplex
        # -------------------------------------------------------------
        t += 1
        # linear decay of the base learning rate
        lr = lr_start * (1.0 - progress) + lr_end * progress
        m = beta1 * m + (1.0 - beta1) * subgrad
        v = beta2 * v + (1.0 - beta2) * (subgrad * subgrad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = _project_onto_simplex(a_candidate)

    return best_a, best_obj

# Main entry point required by the competition framework

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑scale Adam projected descent on max‑self‑convolution.
    Returns the best non‑negative step sequence (list of floats) found
    before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin
    global_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Helper to obtain an initial shape for a given length
    # ------------------------------------------------------------------
    def _initial_shape(n):
        # Prefer the previously discovered construction if it exists
        try:
            a0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            return _upscale(a0, n)
        except Exception:
            # Fallback: a simple edge‑heavy sinusoidal shape
            x = np.arange(n, dtype=float)
            return np.maximum(1.0 - np.sin(np.pi * x / (n - 1)), 0.0)

    # ------------------------------------------------------------------
    # Multi‑scale schedule: (length, fraction of total time)
    # ------------------------------------------------------------------
    schedule = [
        (500, 0.22),
        (2000, 0.33),
        (6000, 0.40),
    ]

    best_overall_a = None
    best_overall_val = np.inf
    elapsed_cumulative = 0.0

    # ------------------------------------------------------------------
    # Run each stage sequentially, up‑scaling the best vector obtained so far
    # ------------------------------------------------------------------
    for idx, (n, frac) in enumerate(schedule):
        # absolute deadline for this stage
        stage_deadline = start_time + (budget_s - 2.0) * (elapsed_cumulative + frac)
        remaining = stage_deadline - time.time()
        if remaining <= 0.0:
            break

        if idx == 0:
            a_start = _initial_shape(n)
        else:
            # upscale the best vector from the previous stage
            a_start = _upscale(best_overall_a, n)

        print(f"[stage {idx+1}] length={n}, time allocated≈{remaining:.1f}s")
        best_a_stage, best_val_stage = _optimise_stage(
            a_start, n, stage_deadline, start_time, global_deadline
        )
        # update global best
        if best_val_stage < best_overall_val:
            best_overall_val = best_val_stage
            best_overall_a = best_a_stage.copy()
            print(f"[stage {idx+1}]  **global best updated** = {best_overall_val:.6f}")

        elapsed_cumulative += frac

    # ------------------------------------------------------------------
    # If any time is still left, fine‑tune the current global best
    # ------------------------------------------------------------------
    if best_overall_a is not None and time.time() < global_deadline:
        remaining = global_deadline - time.time()
        print(f"[fine‑tune] remaining time ≈{remaining:.1f}s")
        best_overall_a, best_overall_val = _optimise_stage(
            best_overall_a,
            best_overall_a.shape[0],
            global_deadline,
            start_time,
            global_deadline,
        )

    # ------------------------------------------------------------------
    # Safety fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_overall_a is None:
        n = 1000
        best_overall_a = np.full(n, 1.0 / n)
        best_overall_val = float(2.0 * n * (_full_convolution(best_overall_a).max()))

    # ------------------------------------------------------------------
    # Final reporting via the provided evaluation routine
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_overall_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_overall_a.tolist()
