"""
Propose a low‑score step‑function.

The algorithm consists of three stages:

1.  A deterministic sweep over several parametric families
    (U‑shapes based on distance, product of distances, and an
    exponential “edge‑decay’’ shape).  Random Dirichlet seeds are also
    generated.  All candidates are normalised to sum‑1 and the best
    ~20 vectors are kept.

2.  Each kept seed is refined with a projected Adam optimiser.
    The sub‑gradient of the maximal convolution entry is used;
    after each Adam step the vector is projected onto the probability
    simplex.  Small Gaussian jitter and a modest learning‑rate decay
    are added to help escape plateaus.

3.  Any remaining time is spent refining the current global best.

All operations are performed with FFT‑based convolutions
( O(n log n) ) and the routine respects the given time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w >= 0, sum(w) = 1 }.
    Implementation of the O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) convolution of a with itself using FFT.
    Returns a vector of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n                     # zero‑pad length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """
    Return max_k (a * a)[k] for a non‑negative vector a with sum == 1.
    Uses the FFT based convolution.
    """
    conv = full_convolution_fft(a)
    return float(conv.max())


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on distance to the nearest edge:
        a[i] = (min(i, n-1-i) + 1)^{-p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14           # avoid exact zeros
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the product of distances to both edges:
        a[i] = ((i+1)*(n-i))^{-p}
    Normalised to sum = 1.
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_exp_distribution(n: int, a_param: float) -> np.ndarray:
    """
    Symmetric exponential decay from the two ends:
        a[i] = exp(-a*i) + exp(-a*(n-1-i))
    Normalised to sum = 1.
    """
    i = np.arange(n)
    a = np.exp(-a_param * i) + np.exp(-a_param * (n - 1 - i))
    a += 1e-14
    a /= a.sum()
    return a


def compute_subgradient(a: np.ndarray):
    """
    Compute a sub‑gradient of the maximal convolution entry.
    Returns (grad, max_val) where grad has shape like a.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    # accumulate contributions from every active convolution index
    for k in active:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        seg_len = i1 - i0 + 1
        j0 = k - i0
        # segment of a reversed, length seg_len
        seg = a[j0 : j0 - seg_len : -1]
        grad[i0:i1 + 1] += 2.0 * seg
    return grad, max_val


def refine_sequence(initial: np.ndarray,
                    deadline: float,
                    best_score: float,
                    best_seq: np.ndarray):
    """
    Refine a candidate vector using projected Adam optimisation.
    Returns the (possibly improved) best_score and best_seq.
    """
    a = initial.copy()
    n = a.shape[0]

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # initial learning rate
    t = 0
    no_improve = 0
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1

        grad, max_val = compute_subgradient(a)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"Iter {iter_count:7d}: new global best = {best_score:.6f}")
        else:
            no_improve += 1

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        a = simplex_projection(a)

        # occasional tiny Gaussian jitter to escape flat regions
        if iter_count % 5000 == 0:
            a = a + np.random.randn(n) * 1e-5
            a = simplex_projection(a)

        # slow learning‑rate decay
        if iter_count % 2000 == 0:
            lr *= 0.95

        # give up if stuck for a long while
        if no_improve > 50000:
            break

    return best_score, best_seq

                best_seq = a_prod.copy()
                print(f" new best (prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")
            seed_vectors.append(a_prod.copy())

        for a_par in exp_params:
            if time.time() > deadline:
                break
            a_exp = u_exp_distribution(n, a_par)
            sc = 2.0 * n * max_convolution(a_exp)
            if sc < best_score:
                best_score = sc
                best_seq = a_exp.copy()
                print(f" new best (exp)   n={n:5d} a={a_par:4.2f}  score={sc:.6f}")
            seed_vectors.append(a_exp.copy())

        # a few random Dirichlet seeds for this length
        for _ in range(3):
            if time.time() > deadline:
                break
            rand_vec = np.random.rand(n)
            rand_vec /= rand_vec.sum()
            sc = 2.0 * n * max_convolution(rand_vec)
            if sc < best_score:
                best_score = sc
                best_seq = rand_vec.copy()
                print(f" new best (rand)  n={n:5d}          score={sc:.6f}")
            seed_vectors.append(rand_vec.copy())

    # --------------------------------------------------------------
    # 2.  Keep the most promising seeds (≈20) for refinement
    # --------------------------------------------------------------
    seed_scores = [(2.0 * len(v) * max_convolution(v), v) for v in seed_vectors]
    seed_scores.sort(key=lambda tup: tup[0])
    top_seeds = [vec for _, vec in seed_scores[:20]]

    # --------------------------------------------------------------
    # 3.  Include an external construction if it exists
    # --------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            if external.sum() > 0:
                external /= external.sum()
                sc_ext = 2.0 * len(external) * max_convolution(external)
                if sc_ext < best_score:
                    best_score = sc_ext
                    best_seq = external.copy()
                    print(f" using external height_sequence_1  score={sc_ext:.6f}")
                top_seeds.append(external.copy())
    except NameError:
        pass

    # --------------------------------------------------------------
    # 4.  Fallback in case nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = 2.0 * n * max_convolution(best_seq)
        print(f" fallback random n={n}  score={best_score:.6f}")

    print(f"\n=== Starting stochastic refinement (Adam) ===")
    print(f"Current global best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Allocate refinement time among the kept seeds
    # --------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        # no time left, just return what we have
        print("\n=== Search finished early (budget exhausted) ===")
        return best_seq.tolist()

    # allocate roughly equal time per seed, but guarantee a tiny budget
    min_per_seed = 5.0
    per_seed = max(min_per_seed,
                   (remaining_time - min_per_seed) / max(1, len(top_seeds)))

    for idx, seed in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seed_deadline = time.time() + per_seed
        best_score, best_seq = refine_sequence(seed, seed_deadline,
                                                best_score, best_seq)
        print(f" Seed {idx:02d} refined – global best now {best_score:.6f}")

    # --------------------------------------------------------------
    # 6.  Use any leftover budget to polish the current best
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_score, best_seq = refine_sequence(best_seq, final_deadline,
                                                best_score, best_seq)
        print(f" Final polishing – best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7.  Return the best sequence found
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will renormalise again, so returning the raw vector is fine.
    return best_seq.tolist()
