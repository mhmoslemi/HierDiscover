def solve_lp(f, M, eps=1e-12):
    n = len(f)
    K = 2*n - 1
    # Build matrix A
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]

    # bounds
    bounds = [(0, None)] * n
    # objective
    c = -np.ones(n)
    # b_ub
    b = np.full(K, M)
    # Solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Propose a step‑function with a smaller score than the basic central‑binomial
construction.

The score returned by `evaluate_sequence` is, after fixing the L¹ norm of the
vector to 1, 2·n·max_k (p∗p)[k] where p_i ≥ 0, Σp_i = 1.
Thus we minimise the largest entry of the self‑convolution of a probability
distribution on {0,…,n‑1}.

The algorithm works as follows:

1.  Initialise `p` with the normalised central‑binomial coefficients
    (the discrete arcsine law).  For this distribution the convolution is
    constant, giving a score ≈ π/2.

2.  Repeatedly solve the linear programme

        maximise   Σ g_i
        subject to g_i ≥ 0
                   (p ∗ g)_k ≤ M   for all k = 0,…,2n‑2

    where `M` is the current maximal convolution value of `p`.
    The programme finds a distribution `g` whose mass is larger than that of
    `p` while respecting the same convolution bound.

3.  Replace p by a convex combination of the old and the new distribution,
    renormalise to Σp_i = 1 and recompute `M`.  If the bound strictly
    decreases we keep the change; otherwise the iteration stops.

   The convex combination weight is fixed to 0.5 (fast convergence) and the
   loop is stopped after a few hundred iterations or when the time budget is
   exhausted.

4.  Finally a tiny stochastic hill‑climb is performed for a few seconds to
    polish the result.

The whole search respects the 1000 s budget, uses only NumPy and SciPy,
and returns a list of non‑negative floats.
"""

import time
from typing import List

import numpy as np
from scipy.optimize import linprog


def _central_binomial_prefix(n: int) -> np.ndarray:
    """Return the first n coefficients b_k = C(2k,k)/4**k."""
    b = np.empty(n, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, n):
        # recurrence b_{k} = b_{k-1} * (2k-1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        b[k] = cur
    return b


def _convolution_max(seq: np.ndarray) -> float:
    """Maximum of the discrete self‑convolution of `seq`."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _build_A_matrix(p: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ g)[k] = (p ∗ g)[k].
    Row k contains the coefficient p_i for all i with 0 ≤ i < n and
    0 ≤ k‑i < n (i.e. i ≤ k ≤ i+n‑1).
    """
    n = p.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = p[i]
    return A


def _solve_lp(p: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme that maximises Σ g_i under the constraints
    (p ∗ g)_k ≤ M   (k = 0,…,2n‑2)   and g_i ≥ 0.
    Returns the optimal g as a 1‑D NumPy array, or ``None`` on failure.
    """
    n = p.shape[0]
    A = _build_A_matrix(p)
    # Upper‑bound vector (all rows have the same bound M)
    b = np.full(A.shape[0], M, dtype=np.float64)

    # Objective: maximise Σ g_i  →  minimise −Σ g_i
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # Use the high‑performance HiGHS solver (built‑in to SciPy ≥1.6)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success and res.x is not None:
        return res.x
    return None


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    max_len: int = 15000,
    **kwargs,
) -> List[float]:
    """
    Return a step‑function (list of non‑negative heights) with a low
    `evaluate_sequence` score.  The method prefers a moderate length
    (default ~800) so that a linear programme can be solved many times
    within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # a small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise with the normalised central‑binomial coefficients.
    # ------------------------------------------------------------------
    # Use a length that comfortably fits inside the time budget for LPs.
    n = min(800, max_len)  # 800 is a good trade‑off between quality and speed
    p = _central_binomial_prefix(n).astype(np.float64)
    p /= p.sum()           # normalise to a probability distribution (L1 = 1)

    # current bound and best score
    M = _convolution_max(p)
    best_score = 2.0 * n * M
    best_p = p.copy()
    print(f"[init] n={n}, score={best_score:.9f}, M={M:.9e}")

    # ------------------------------------------------------------------
    # 2. Iteratively improve via the LP described in the statement.
    # ------------------------------------------------------------------
    max_iter = 300
    t_mix = 0.5                     # convex combination weight
    epsilon = 1e-12

    for it in range(max_iter):
        if time.time() > deadline:
            print("[time] budget exhausted")
            break

        # Solve LP to obtain a distribution with a larger total mass
        g = _solve_lp(p, M)
        if g is None:
            print("[lp] solve failed – terminating")
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-9:
            # No improvement possible under the current bound
            print("[lp] no mass increase (sum_g≈{:.9f})".format(sum_g))
            break

        # Mix old and new distributions, then renormalise to L1 = 1
        p_candidate = (1.0 - t_mix) * p + t_mix * g
        p_candidate = np.maximum(p_candidate, 0.0)
        p_candidate /= p_candidate.sum()

        M_candidate = _convolution_max(p_candidate)
        score_candidate = 2.0 * n * M_candidate

        # Accept the candidate if the bound actually decreased
        if M_candidate < M - epsilon:
            p = p_candidate
            M = M_candidate
            if score_candidate < best_score - 1e-12:
                best_score = score_candidate
                best_p = p.copy()
                print(
                    f"[iter {it:03d}] improved → score={best_score:.9f}, "
                    f"M={M:.9e}"
                )
        else:
            # No improvement in the bound – stop the loop
            print(f"[iter {it:03d}] no bound improvement (M={M_candidate:.9e})")
            break

    # ------------------------------------------------------------------
    # 3. Optional short random polishing phase (few thousand attempts).
    # ------------------------------------------------------------------
    cur_p = best_p.copy()
    cur_score = best_score
    while time.time() < deadline:
        # pick a random index and perturb it multiplicatively
        idx = np.random.randint(n)
        old = cur_p[idx]
        # Gaussian perturbation relative to the current magnitude
        delta = np.random.randn() * 0.02 * max(old, 1e-12)
        new_val = max(0.0, old + delta)
        if new_val == old:
            continue
        cand = cur_p.copy()
        cand[idx] = new_val
        cand /= cand.sum()  # keep L1 = 1
        M_cand = _convolution_max(cand)
        score_cand = 2.0 * n * M_cand
        if score_cand < cur_score:
            cur_p, cur_score = cand, score_cand
            if score_cand < best_score:
                best_score, best_p = score_cand, cand.copy()
                print(
                    f"[refine] new best {best_score:.9f} "
                    f"(idx={idx}, delta={delta:.2e})"
                )
        # stop after a few thousand successful attempts
        if np.random.rand() < 1e-4:
            break

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (scale it arbitrarily; scaling does not
    #    affect the score, but we keep the normalised values for safety).
    # ------------------------------------------------------------------
    result = [float(max(0.0, x)) for x in best_p.tolist()]
    # sanity check – we keep it short, but the caller may still invoke the
    # evaluator for the final score.
    try:
        final_score = evaluate_sequence(result)
        print(f"[final] length={len(result)}, final_score={final_score:.9f}")
    except Exception:
        pass
    return result
