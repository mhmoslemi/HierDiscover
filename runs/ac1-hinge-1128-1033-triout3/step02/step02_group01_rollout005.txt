def softmax_gradient(a, beta):
    conv = full_convolution_fft(a)  # length 2n-1
    max_val = conv.max()
    # compute exponentials
    # Use np.exp with shift to avoid overflow: exp(beta * (conv - max_val))
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        # fallback: assign equal weight to max entries
        w = np.zeros_like(conv)
        w[conv == max_val] = 1.0
        w_sum = w.sum()
    w = w / w_sum

    # compute gradient = 2 * cross-correlation of a with w (i.e., sum_j a_j w_{i+j})
    # Use convolution a * w_rev and slice
    w_rev = w[::-1]
    # compute convolution using FFT
    size = len(a) + len(w_rev) - 1
    L = 1
    while L < size:
        L <<= 1
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(fa * fw, n=L)
    # conv_rev length L, but valid part first size entries.
    conv_rev = conv_rev[:size]
    n = len(a)
    grad_slice = conv_rev[n-1:2*n-1][::-1]
    grad = 2.0 * grad_slice

def compute_softmax_gradient(a, beta):
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_val = conv.max()
    # shift conv for stability
    shifted = conv - max_val
    # compute exp(beta * shifted)
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        # fallback: uniform over max entries
        w = np.zeros_like(conv)
        w[conv == max_val] = 1.0
        w_sum = w.sum()
    w /= w_sum

    # Compute gradient via convolution a with reversed w
    w_rev = w[::-1]
    size = n + len(w_rev) - 1
    L = 1 << (size-1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(fa * fw, n=L)
    conv_rev = conv_rev[:size]
    grad_slice = conv_rev[n-1:2*n-1][::-1]
    grad = 2.0 * grad_slice
    return grad

"""
Propose a step‑function that minimizes `evaluate_sequence`.

The search consists of three stages:

1️⃣  Deterministic sweep over a few parametric families (U‑shapes,
    product‑U‑shapes and double‑exponential shapes) for several lengths
    and exponents.  The best vector (normalised to sum = 1) is kept as a
    seed.

2️⃣  Stochastic refinement using a *soft‑max* approximation to the
    max‑convolution.  The gradient of the smoothed objective is computed
    via FFT, a projected‑simplex update is performed, and a simulated‑
    annealing acceptance rule allows occasional uphill moves.

3️⃣  Random mass‑transfer moves are interleaved to help escape flat
    plateaus.  The algorithm stops when the time budget (≈ 1000 s) is
    exhausted and returns the best vector discovered.
"""

# Helper utilities

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the probability simplex {w >= 0, sum(w) = 1}.
    O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on distance to the nearest edge:
        a[i] = (min(i, n‑1‑i) + 1)^{‑p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ based on the product of distances to both edges:
        a[i] = ((i+1)*(n‑i))^{‑p}
    Normalised to sum = 1.
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exponential_distribution(n: int, lam: float) -> np.ndarray:
    """
    Double‑exponential (symmetric) shape:
        a[i] = exp(-lam*i) + exp(-lam*(n‑1‑i))
    Normalised to sum = 1.
    """
    i = np.arange(n)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (a is assumed non‑negative)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a vector already normalised to sum = 1."""
    n = a.shape[0]
    max_c = max_convolution(a)
    return 2.0 * n * max_c


def softmax_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the gradient of the soft‑max approximation
    `logsumexp(beta*(conv - max(conv)))` with respect to a.
    The gradient equals 2 * (a ⋆ w) where w is the soft‑max weight vector
    on the convolution entries.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)                     # length 2n‑1
    max_val = conv.max()
    # weights w_i ∝ exp(beta * (conv_i - max_val))
    # subtracting max_val avoids overflow
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:                                   # safety fallback
        w = np.zeros_like(conv)
        w[conv == max_val] = 1.0
        w_sum = w.sum()
    w /= w_sum                                          # now a probability vector

    # Cross‑correlation a ⋆ w  (i.e. sum_j a_j * w_{i+j}) can be obtained
    # via a convolution with the reversed weight vector.
    w_rev = w[::-1]

    size = n + len(w_rev) - 1
    L = 1 << (size - 1).bit_length()                   # next power of two
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(fa * fw, n=L)
    conv_rev = conv_rev[:size]

    # slice that corresponds to indices i = 0…n‑1
    grad_slice = conv_rev[n - 1 : 2 * n - 1][::-1]
    grad = 2.0 * grad_slice
    return grad

    # include the external construction if present
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        if external.sum() > 0.0:
            external /= external.sum()
            sc_ext = score_from_vector(external)
            if sc_ext < best_score:
                best_score = sc_ext
                best_seq = external.copy()
                print(f"det: using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # fallback (should never happen)
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f"det: fallback random seed n={n}  score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main proposal routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    The routine respects the supplied time budget (seconds) and returns the
    best sequence (list of floats) it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0               # small safety margin

    # --------------------------------------------------------------
    # 1. Deterministic initialization
    # --------------------------------------------------------------
    best_seq, best_score = initial_candidate()
    print(f"Initial best score: {best_score:.6f} (length {len(best_seq)})")

    # Normalise (just in case) – the internal score assumes sum = 1.
    best_seq = best_seq / best_seq.sum()
    current = best_seq.copy()
    current_score = best_score
    n = len(current)

    # --------------------------------------------------------------
    # 2. Stochastic refinement (soft‑max gradient + random moves)
    # --------------------------------------------------------------
    temperature = 0.2
    cooling = 0.9997
    step_size = 0.15
    step_decay = cooling

    beta = 1.0                  # initial soft‑max sharpness
    beta_inc = 0.0015
    beta_max = 80.0

    iteration = 0
    last_improve_iter = 0

    print("\n=== Starting stochastic refinement ===")
    while time.time() < deadline:
        iteration += 1

        # ----- 2a – choose move type -----
        if np.random.rand() < 0.3:                     # random mass transfer
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] <= 1e-12:
                continue
            delta = current[i] * np.random.rand() * 0.5
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            # sum stays 1 → no extra projection needed
        else:                                          # projected gradient step
            grad = softmax_gradient(current, beta)
            cand = current - step_size * grad
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)

        # ----- 2b – evaluate candidate (cheap internal score) -----
        cand_score = score_from_vector(cand)

        # ----- 2c – simulated‑annealing acceptance -----
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve_iter = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # ----- 2d – anneal parameters -----
        temperature *= cooling
        step_size *= step_decay
        beta = min(beta_max, beta + beta_inc)

        # ----- 2e – stagnation handling -----
        if iteration - last_improve_iter > 50000:
            # inject a tiny random bump to kick the search out of a plateau
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current_score = score_from_vector(current)
            last_improve_iter = iteration
            temperature = 0.2          # re‑heat

    # --------------------------------------------------------------
    # 3. Return the best sequence discovered
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will perform its own clipping/normalisation,
    # so we simply return the raw vector.
    return best_seq.tolist()
