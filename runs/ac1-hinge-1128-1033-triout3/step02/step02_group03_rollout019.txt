def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time= time.time()
    deadline = start_time + budget_s - 5

    # Determine n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 1153

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        s = evaluate_sequence(p.tolist())
        if s < best_score:
            best_score = s
            best_seq = p.copy()
            print(f"New best: {best_score:.6f}")
    
    # uniform
    p_uniform = np.full(n, 1.0/n)
    update_best(p_uniform)

    # Exponential grid
    a_vals = np.linspace(0.5, 5.0, 46)  # 0.5 to 5 step ~0.1
    for a in a_vals:
        if time.time() > deadline:
            break
        lam = a / n
        i = np.arange(n)
        p = np.exp(-lam * i)
        p = p / p.sum()
        update_best(p)

    # Mixture exponential random search
    max_mix_iter = 2000
    for _ in range(max_mix_iter):
        if time.time() > deadline:
            break
        a1 = rng.uniform(0.3, 5.0)
        a2 = rng.uniform(0.3, 5.0)
        w = rng.uniform(0.1, 0.9)
        lam1 = a1 / n
        lam2 = a2 / n
        i = np.arange(n)
        p = w * np.exp(-lam1 * i) + (1 - w) * np.exp(-lam2 * i)
        p = p / p.sum()
        update_best(p)

    # Logistic random search (monotonic decreasing)
    for _ in range(500):
        if time.time()>deadline:
            break
        beta = rng.uniform(0.001, 0.05)  # steepness
        shift = rng.uniform(0.7*n, 0.99*n)  # location of midpoint
        i = np.arange(n)
        p = 1.0/(1.0 + np.exp(beta * (i - shift)))
        p = p / p.sum()
        update_best(p)

    # Local refinement by random transfers
    if best_seq is None:
        # fallback to uniform
        best_seq = np.full(n, 1.0/n)
    p = best_seq.copy()
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two indices
        i, j = rng.choice(n, size=2, replace=False)
        # transfer fraction
        delta = rng.uniform(0, 0.02) * p[i]
        if delta <= 0:
            continue
        p_new = p.copy()
        p_new[i] -= delta
        p_new[j] += delta
        # optional small projection onto simplex not needed since sum unchanged
        # re-normalise to correct rounding errors
        if p_new[i] < 0:

import time
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Find a low‑M step‑function using a small parametric search + local tweaks."""
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # 1) Problem size
    # ---------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name‑defined]
    except Exception:
        n = 1153  # historic default

    best_seq = None
    best_score = np.inf

    # ---------------------------------------------------------
    # Helper to evaluate and possibly update the current best.
    # ---------------------------------------------------------
    def _maybe_update(p: np.ndarray):
        nonlocal best_seq, best_score
        # tiny safeguard – ensure sum is not accidentally zero
        if p.sum() < 1e-12:
            return
        score = evaluate_sequence(p.tolist())
        if score < best_score:
            best_score = score
            best_seq = p.copy()
            print(f"[Update] New best score: {best_score:.6f}")

    # ---------------------------------------------------------
    # Time handling
    # ---------------------------------------------------------
    start = time.time()
    limit = start + budget_s - 5.0   # keep a few seconds margin

    # ---------------------------------------------------------
    # 2) Very cheap baselines
    # ---------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    _maybe_update(uniform)

    # ---------------------------------------------------------
    # 3) Single‑exponential families (p_i ∝ exp(-a*i/n))
    # ---------------------------------------------------------
    a_grid = np.linspace(0.5, 5.0, 46)   # about 0.1 step
    i = np.arange(n, dtype=float)

    for a in a_grid:
        if time.time() > limit:
            break
        lam = a / n
        p = np.exp(-lam * i)
        p /= p.sum()
        _maybe_update(p)

    # ---------------------------------------------------------
    # 4) Mixture of two exponentials – random exploration
    # ---------------------------------------------------------
    max_mix = 500  # keep it modest to stay within the time budget
    for _ in range(max_mix):
        if time.time() > limit:
            break
        a1 = rng.uniform(0.3, 5.0)
        a2 = rng.uniform(0.3, 5.0)
        w = rng.uniform(0.1, 0.9)
        lam1 = a1 / n
        lam2 = a2 / n
        p = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
        p /= p.sum()
        _maybe_update(p)

    # ---------------------------------------------------------
    # 5) Logistic‑type decreasing profiles – random exploration
    # ---------------------------------------------------------
    for _ in range(300):
        if time.time() > limit:
            break
        beta = rng.uniform(0.001, 0.05)               # steepness
        shift = rng.uniform(0.6 * n, 0.99 * n)        # location of the midpoint
        p = 1.0 / (1.0 + np.exp(beta * (i - shift)))
        p /= p.sum()
        _maybe_update(p)

    # ---------------------------------------------------------
    # 6) Small stochastic local optimisation starting from the best found so far
    # ---------------------------------------------------------
    if best_seq is None:
        # fall back to uniform – should never happen
        best_seq = uniform.copy()
        best_score = evaluate_sequence(best_seq.tolist())

    p = best_seq.copy()
    cur_score = best_score
    it = 0
    while time.time() < limit:
        it += 1
        # pick two distinct indices and move a tiny amount of mass
        i1, i2 = rng.choice(n, size=2, replace=False)
        if p[i1] <= 0.0:
            continue
        delta = rng.uniform(0.0, 0.02) * p[i1]   # move at most 2 % of the donor
        if delta <= 0.0:
            continue
        p_new = p.copy()
        p_new[i1] -= delta
        p_new[i2] += delta
        # numerical safety – renormalise (sum is preserved up to round‑off)
        p_new = _simplex_projection(p_new)      # keeps exact sum == 1
        new_score = evaluate_sequence(p_new.tolist())
        if new_score < cur_score:
            p = p_new
            cur_score = new_score
            best_seq = p.copy()
            best_score = cur_score
            print(f"[Local] iter {it:6d} – improved score: {best_score:.6f}")

    # ---------------------------------------------------------
    # 7) Return plain Python list that satisfies the security checks.
    # ---------------------------------------------------------
    result = np.clip(best_seq, 0.0, 1000.0).tolist()
    print(f"Best final score: {best_score:.6f}")
    return result

"""
Search for a non‑negative step function that minimises

        M = 2·n·max(conv(a, a)) / (sum(a)**2)

The routine builds a handful of simple parametric families
(uniform, single exponential, mixture of two exponentials and a
logistic‑type decreasing profile), evaluates them, keeps the best
candidate and finally performs a cheap stochastic local optimisation
by moving tiny amounts of mass between random positions while always
projecting back onto the probability simplex.  The whole search respects
the security checks of ``evaluate_sequence`` and finishes safely before
the allotted 1000 s budget.
"""

import time
import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0 , Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
