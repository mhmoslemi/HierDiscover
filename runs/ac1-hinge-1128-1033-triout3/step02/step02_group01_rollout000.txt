"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Initialisation** – Use the best known construction (`height_sequence_1`)
   if it exists, otherwise start from a random simplex vector.  Enforce
   symmetry (the problem is invariant under reversal) and normalise to sum=1.

2. **Iterative refinement** – Repeatedly
   * compute the full self‑convolution by FFT,
   * locate the (near) maximal convolution entry/entries,
   * build the sub‑gradient of the max‑convolution,
   * form a descent direction with zero‑sum (to stay on the simplex),
   * perform a back‑tracking line‑search on the step‑size `λ`.  If a
     candidate improves the objective it is accepted; otherwise it may be
     accepted with a simulated‑annealing probability.

3. **Random perturbations** – When the line‑search fails, a small random
   mass‑transfer move is attempted (moving a fraction of mass from a
   source index to a target index).  This helps escape shallow local minima.

4. **Stagnation handling** – If no improvement occurs for many
   iterations we inject a tiny random bump and reset the temperature.

All operations are vectorised and use a single FFT per iteration,
allowing millions of iterations inside the 1000 s budget.  The routine
keeps track of the best vector seen and returns it as a plain Python
list.

The score computed internally is `2 * n * max(conv)`, which matches the
value returned by the provided `evaluate_sequence` when the vector is
already normalised to sum = 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w>=0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a (assumes sum(a)=1)."""
    return float(full_convolution_fft(a).max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a normalised vector."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


def asym_product_distribution(n: int, pl: float, pr: float) -> np.ndarray:
    """Two‑parameter “U‑shape’’: a[i] ∝ (i+1)^(-pl) * (n‑i)^(-pr)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-pl) * (n - i) ** (-pr) + 1e-14
    a /= a.sum()
    return a

    except Exception:

        init = (init + init[::-1]) / 2.0               # enforce symmetry
        init /= init.sum()

    # Enforce exact symmetry (helps the objective) and normalise
    current = (init + init[::-1]) / 2.0

    current_score = score_from_vector(current)
    best_seq = current.copy()
    best_score = current_score

    print(f"Starting score: {best_score:.6f} (n={n})")

    # --------------------------------------------------------------
    # 1. Stochastic refinement loop
    # --------------------------------------------------------------
    # algorithmic parameters
    step_size = 0.2          # initial step magnitude for line‑search
    min_lambda = 1e-8        # lower bound for line‑search step

    while time.time() < deadline:

    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # The external evaluator will clip and normalise again, so we simply
    # return the raw vector.
