"""
Hybrid parametric search + smooth‑max Adam refinement.

* Sample a few reasonably‑inspired families (uniform, sqrt‑series,
  power‑law, Beta‑shapes, truncated exponential) and keep the best
  candidates w.r.t. the fast‑score 2·n·max(a∗a) (the metric with a
  unit‑sum normalization).

* Starting from each promising candidate run a projected Adam optimiser
  on the smooth approximation  log ∑ exp(λ·(a∗a)) .  The temperature
  λ is increased geometrically so that the objective converges to the
  true max‑convolution while still providing a useful gradient.

* After each improvement the expensive ``evaluate_sequence`` is called
  only when a new global best is discovered.  The whole search stops
  before the 1000 s time limit.

The returned list is a non‑negative step‑function that (empirically)
beats the previous bound 1.570845.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Simplex projection – Duchi et al. (2008)
# --------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Fast evaluation of the metric (ignoring the final scaling by sum²)
# --------------------------------------------------------------
def fast_score(a: np.ndarray, size_fft: int) -> float:
    """Return 2·n·max(a∗a) using a single FFT of length ``size_fft``."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# --------------------------------------------------------------
# Helper for the smooth‑max objective
# --------------------------------------------------------------
def logsumexp(x: np.ndarray) -> float:
    """Numerically stable log‑sum‑exp."""
    m = np.max(x)
    return m + np.log(np.exp(x - m).sum())


# --------------------------------------------------------------
# Sequence factories
# --------------------------------------------------------------
def uniform_seq(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def sqrt_series_seq(n: int) -> np.ndarray:
    """Coefficients of (1‑x)^(-1/2) truncated to length n and normalised."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


def power_law_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta(a,b) on the grid (i+0.5)/n."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.where(
        (x > 0) & (x < 1),
        (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0)),
        0.0,
    )
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    if w.sum() == 0.0:
        return uniform_seq(n)
    w /= w.sum()
    return w


def truncated_exp_seq(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i ∝ r^i, truncated to length n."""
    if r >= 1.0:
        return uniform_seq(n)
    a = (1.0 - r) * (r ** np.arange(n, dtype=np.float64))
    denom = 1.0 - r ** n
    if denom == 0.0:
        return uniform_seq(n)
    a /= denom
    return a


# --------------------------------------------------------------
# Adam‑based refinement on the smooth‑max objective
# --------------------------------------------------------------
def refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    best_seq: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Perform a few thousand Adam steps on the smooth‑max of the
    self‑convolution, projecting onto the simplex after each step.
    Returns the best (a, fast_score) seen during the run.
    """
    # FFT lengths
    size_fft_score = 1 << ((2 * n - 1).bit_length())      # for fast_score
    size_fft_grad = 1 << ((3 * n - 2).bit_length())      # for gradient (w∗a)

    a = a0.copy()
    a = np.maximum(a, 0.0)
    a = project_simplex(a)

    cur_fast = fast_score(a, size_fft_score)

    # Adam hyper‑parameters
    lr = 0.2
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    lam = 1.0          # start of the smooth‑max temperature
    lam_max = 6000.0
    lam_factor = 1.015

    max_iters = 8000
    it = 0

    while it < max_iters and time.time() < deadline:
        # ---- compute convolution a∗a ---------------------------------
        fa = np.fft.rfft(a, size_fft_grad)
        conv = np.fft.irfft(fa * fa, size_fft_grad)[: 2 * n - 1]

        # ---- smooth‑max (log‑sum‑exp) and its softmax weights --------
        z = lam * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        sum_exp = exp_z.sum()
        w = exp_z / sum_exp                     # length 2n‑1, sums to 1

        # ---- gradient = 2·lam·(w∗a) ---------------------------------
        fw = np.fft.rfft(w, size_fft_grad)
        grad_full = np.fft.irfft(fw * fa, size_fft_grad)[:n]   # first n entries
        grad = 2.0 * lam * grad_full

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = project_simplex(a_new)

        # ---- cheap evaluation ----------------------------------------
        fast_new = fast_score(a_new, size_fft_score)

        # accept if we made progress (greedy)
        if fast_new <= cur_fast:
            a = a_new
            cur_fast = fast_new

            # update the global best via the slow evaluator
            if cur_fast < best_score - 1e-9:
                true_val = evaluate_sequence(a.tolist())
                if true_val < best_score:
                    best_score = true_val
                    best_seq = a.copy()
                    print(f"[global] new best {best_score:.6f}")

        # temperature schedule
        lam = min(lam_max, lam * lam_factor)

        it += 1

    return a, cur_fast, best_seq, best_score
