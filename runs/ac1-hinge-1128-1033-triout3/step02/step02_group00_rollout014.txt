def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    import time
    start = time.time()
    deadline = start + budget_s - 2

    # Choose length n
    n = 3000
    # initial decreasing linear shape
    a = np.arange(n, 0, -1, dtype=np.float64)   # n, n-1,...,1
    a = a / a.sum()  # sum 1

    # Alternatively, we might use logistic or power shape; but linear is fine.

    # Initialize log probabilities
    loga = np.log(a + 1e-30)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(loga)
    v = np.zeros_like(loga)

    # smoothing schedule
    beta_s_start = 0.5
    beta_s_end = 30.0

    # learning rate schedule
    lr_start = 0.2
    lr_end = 0.001

    best_score = float('inf')
    best_a = a.copy()
    iteration = 0

    # Determine FFT size
    size_conv = 1 << ((2*n - 2).bit_length())  # for conv (full)
    size_corr = 1 << ((3*n - 2).bit_length())  # for correlation (full)
    size = max(size_conv, size_corr)  # unify size

    while time.time() < deadline:
        iteration += 1

        # compute FFT of a
        A = np.fft.rfft(a, size)

        # compute full self-convolution
        conv_full = np.fft.irfft(A * A, size)[:2 * n - 1]

        # smoothing parameter beta
        progress = (time.time() - start) / (deadline - start)
        beta = beta_s_start * (1 - progress) + beta_s_end * progress

        # compute softmax weights
        max_c = conv_full.max()
        exps = np.exp(beta * (conv_full - max_c))
        w = exps / exps.sum()

        # correlation to get gradient of smooth max
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, size)
        corr = np.fft.irfft(A * W_rev, size)[:3 * n - 2]
        # gradient of smooth max w.r.t a: 2 * corr[2*n-2 - i]
        grad_s = 2.0 * corr[2 * n - 2 - np.arange(n)]

        # full gradient of f = 2*n * smooth_max
        grad_a = 2.0 * n * grad_s

        # compute gradient w.r.t loga using chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # learning rate schedule
        lr = lr_start * (1 - progress) + lr_end * progress

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        loga -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # compute new a via softmax
        a = np.exp(loga)
        a_sum = a.sum()
        if a_sum == 0:
            # reinitialize
            a = np.full(n, 1.0 / n)
            loga = np.log(a)
            continue
        a /= a_sum

        # evaluate occasionally
        if iteration % 100 == 0:
            try:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    print(f"[{iteration}] new best: {best_score:.6f}")
            except Exception as e:
                pass

        # occasional random perturbation to escape stagnation
        if np.random.rand() < 0.01:
            # move small mass between random indices
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.rand() * min(a[i], 0.1)  # take up to 10% from i
            a[i] -= delta
            a[j] += delta
            # reproject onto simplex via normalizing (but sum remains 1)
            # keep loga approx:
            loga = np.log(a + 1e-30)

    # final evaluation
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluation error:", exc)

    return best_a.tolist()

    rev = x[::-1]

    level = rev.copy()
    weight = np.ones_like(rev)

        if level[i] < level[i+1] - 1e-12:
            # need to pool
            total_weight = weight[i] + weight[i+1]
            new_level = (weight[i]*level[i] + weight[i+1]*level[i+1]) / total_weight

            if i > 0:

    # Expand pooled levels
    y_rev = np.repeat(level, weight.astype(int))

"""
Propose a candidate step sequence for the “max‑self‑convolution” objective.

The algorithm works on the probability simplex (sum = 1) and uses a
smooth approximation of the max‑operator:

    smooth_max_β(conv) = (1/β)·log( Σ_k exp(β·conv_k) ),

with β increasing from a small value to a large one (β≈30).  The gradient
of the smooth objective is a weighted combination of the individual
convolution gradients and can be computed efficiently via one FFT of the
current sequence and one FFT of the weight vector.  Optimization is
performed on the log‑probabilities (mirror descent) using the Adam
optimizer.  Periodic evaluation keeps the best vector seen; finally a
short “local‑search” phase applies tiny random mass‑transfers to squeeze
out the last improvements.

All operations respect the required constraints (non‑negative entries,
sum ≥ 0.01, elements ≤ 1000) – the final vector is normalised to sum = 1,
so the clamping in ``evaluate_sequence`` never activates.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: compute the full self‑convolution using a real FFT
# ------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a (full convolution, length 2·n‑1) using the given FFT size."""
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[: 2 * a.shape[0] - 1]
    # Clip tiny negative artefacts
    np.maximum(conv, 0, out=conv)
    return conv


# ------------------------------------------------------------
# Helper: correlation a * w (needed for the gradient of the smooth max)
# ------------------------------------------------------------
def _correlation_fft(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k·a_{k‑i} for all i using FFT.
    Returns a vector of length n with g_i values.
    """
    n = a.shape[0]
    # reverse w for correlation via convolution
    w_rev = w[::-1]
    A = np.fft.rfft(a, size)
    W_rev = np.fft.rfft(w_rev, size)
    full_corr = np.fft.irfft(A * W_rev, size)[: 3 * n - 2]  # length 3n‑2
    # g_i = full_corr[2n‑2‑i]
    g = full_corr[2 * n - 2 - np.arange(n)]
    return g


# ------------------------------------------------------------
# Main entry point required by the competition framework
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Adam‑based optimisation on the smooth maximum of the self‑convolution.
    Returns the best sequence (list of floats) found before the time budget
    expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final I/O

    # --------------------------------------------------------
    # Choose the length of the sequence and initialise a
    # --------------------------------------------------------
    try:
        # Re‑use the length of a previously discovered construction
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 3000                      # a moderately large size

    # Initialise with a simple decreasing linear shape (highly non‑uniform)
    a = np.arange(n, 0, -1, dtype=np.float64)
    a = a / a.sum()                   # normalise to sum = 1

    # Work in the log‑space (softmax parametrisation)
    loga = np.log(a + 1e-30)

    # --------------------------------------------------------
    # Adam state
    # --------------------------------------------------------
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(loga)
    v = np.zeros_like(loga)

    # --------------------------------------------------------
    # Schedules (time‑based linear interpolation)
    # --------------------------------------------------------
    beta_start = 0.5     # start of the smoothness parameter β
    beta_end = 30.0      # final β (approaches true max)
    lr_start = 0.20      # initial learning‑rate for Adam
    lr_end = 0.001       # final learning‑rate

    # --------------------------------------------------------
    # FFT size (large enough for both convolution and correlation)
    # --------------------------------------------------------
    size = 1 << ((3 * n - 2).bit_length())   # power of two ≥ 3n‑2

    # --------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------
    best_a = a.copy()
    # exact objective (since sum=1) is 2·n·max(conv)
    conv_cur = _full_convolution_fft(a, size)
    best_val = float(2.0 * n * conv_cur.max())
    iteration = 0

    print(f"[init] n={n}, initial objective={best_val:.6f}")

    # ========================================================
    # Main optimisation loop (gradient descent on smooth max)
    # ========================================================
    while time.time() < deadline:
        iteration += 1

        # ---- Full self‑convolution (needed for smooth max) ----
        conv = _full_convolution_fft(a, size)

        # ---- Update β (smoothness) ----
        progress = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start * (1.0 - progress) + beta_end * progress

        # ---- Softmax weights w_k = exp(β·conv_k) / Σ exp(β·conv) ----
        c_max = conv.max()
        # subtract c_max for numerical stability
        w_unnorm = np.exp(beta * (conv - c_max))
        w = w_unnorm / w_unnorm.sum()

        # ---- Gradient of smooth max (via correlation) ----
        corr = _correlation_fft(a, w, size)               # Σ_k w_k·a_{k-i}
        grad_smooth = 2.0 * corr                          # ∂smooth_max/∂a

        # ---- Full gradient of the objective f = 2·n·smooth_max ----
        grad_a = 2.0 * n * grad_smooth

        # ---- Gradient w.r.t. log‑probabilities (mirror‑descent) ----
        avg = np.dot(a, grad_a)                           # ⟨a,∇f⟩
        grad_log = a * (grad_a - avg)                     # ∂f/∂log a

        # ---- Learning‑rate schedule ----
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        bias_correction1 = 1.0 - beta1 ** iteration
        bias_correction2 = 1.0 - beta2 ** iteration
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        loga -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- Recover a from logits (softmax) ----
        shift = loga.max()
        a = np.exp(loga - shift)          # stable softmax
        a_sum = a.sum()
        if a_sum == 0.0:                  # extremely unlikely, re‑initialise
            a = np.full(n, 1.0 / n)
            loga = np.log(a)
            continue
        a /= a_sum                         # enforce Σ a_i = 1

        # ---- Occasionally evaluate the true (non‑smoothed) objective ----
        if iteration % 200 == 0:
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{iteration:6d}] new best = {best_val:.6f}")

        # ---- Small random perturbations to escape plateaus ----
        if np.random.rand() < 0.01:
            i, j = np.random.randint(n, size=2)
            if i != j:
                # move up to 5 % of a[i] from i to j
                delta = np.random.rand() * a[i] * 0.05
                a[i] -= delta
                a[j] += delta
                # recompute logits after the jump
                shift = np.log(a + 1e-30).max()
                loga = np.log(a + 1e-30) - shift

    # ========================================================
    # Local refinement: tiny random mass‑transfers accepted only if they improve
    # ========================================================
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # transfer a tiny amount (up to 1e‑4 of the source mass)
        delta = np.random.rand() * min(a[i], 1e-4)
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # compute objective for the candidate
        conv_new = _full_convolution_fft(a_new, size)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            print(f"[local] improved to {best_val:.6f}")

    # --------------------------------------------------------
    # Final reporting via the provided evaluation routine
    # --------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
