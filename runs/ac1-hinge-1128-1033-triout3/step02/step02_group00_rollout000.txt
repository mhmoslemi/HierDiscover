            best_a = a.copy()

            last_improve_iter = iteration

"""
Proposed algorithm
===================

We treat the optimisation problem

    minimise   F(a) = 2 * n * max(conv(a, a))           (a_i ≥ 0, Σ a_i = 1)

as a *smooth* optimisation problem.  The non‑smooth maximum is replaced by a
log‑sum‑exp (soft‑max) with an annealing temperature ``θ`` that is increased
linearly during the run.  For a given ``θ`` we minimise the smooth surrogate

    G_θ(a) = (1/θ)·log( Σ_k exp(θ·conv_k) ),

whose gradient can be written in closed form as a (cross‑)convolution that can
be evaluated with FFTs in O(n log n).  The iterates are updated with an Adam‑style
adaptive step and projected onto the probability simplex (Duchi et al., 2008)
to keep the constraints satisfied.

The algorithm starts from the best known construction (``height_sequence_1``)
if it exists, otherwise from a smooth cosine shape, and runs until the time
budget is exhausted (≈1000 s).  The current best candidate is kept and finally
returned as a plain Python list; its quality is also reported by calling the
provided ``evaluate_sequence`` routine.

The method improves on simple sub‑gradient descent by:
* smoothing the objective (helps to obtain useful gradients far from the peak),
* using Adam’s adaptive learning rates,
* annealing the smoothing parameter,
* occasional random perturbations and a simple restart mechanism
  when progress stalls.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x ≥ 0, Σ x = 1 }.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safeguard – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index of last positive entry
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to guard against rounding errors
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ------------------------------------------------------------
# Helper: full convolution a * a using a real FFT
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Returns the full (length 2·n‑1) convolution a*a using an FFT.
    """
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    np.maximum(conv, 0, out=conv)             # eliminate tiny negatives
    return conv
