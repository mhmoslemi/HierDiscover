def propose_candidate(seed=42, budget_s=1000): 
    rng = np.random.default_rng(seed)
    try: n = len(height_sequence_1)
    except: n = 1153
    # initial p based on previous best.
    p = np.array(height_sequence_1, dtype=float) if 'height_sequence_1' in globals() else rng.random(n)
    p = np.clip(p, 0, None)
    if p.sum() == 0: p = np.full(n, 1/n)
    p = p / p.sum()
    # Optionally add jitter to diversify.
    p = p + rng.normal(scale=1e-4, size=n)
    p = np.clip(p, 0, None)
    p = p / p.sum()
    conv = np.convolve(p, p)
    cur_score = 2 * n * conv.max()
    best_p = p.copy()
    best_score = cur_score

    start = time.time()
    deadline = start + budget_s - 5

    # set SA parameters
    T_start = 1e-2 # maybe relative to (best_score) scale
    T_end = 1e-6
    # We'll use exponential cooling: T = T_start * (T_end/T_start)**((t-elapsed)/(budget_t))
    # define max_iter, e.g., 5e6? We'll loop until time out.

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start
        # compute temperature
        progress = elapsed / (budget_s - 5)
        T = T_start * (T_end / T_start) ** progress  # exponential decay
        # propose move
        # sample source i such that p_i>0
        i = rng.choice(n, p=p)  # more likely to pick high mass? Maybe choose uniformly.
        # We want a source with decent mass so we can move some.

        i = rng.integers(n)

        conv_new[idx2i] += delta * delta
        conv_new[idx2j] += delta * delta
        conv_new[idxij] -= 2.0 * delta * delta

        new_score = 2.0 * n * new_max

    T_end = 1e-5

        elapsed = time.time() - start

        i = rng.choice(n, p=p)
        if p[i] <= 0.0:

        j = rng.integers(n - 1)

"""
Simulated‑annealing optimisation on the probability simplex.

The evaluation function is
    M = 2·n·max(conv(a,a)) / (sum a)^2 .
Because the expression is homogeneous, we may restrict the search to
probability vectors p (p_i ≥ 0, Σ p_i = 1).  For a given p the objective is
    f(p) = 2·n·max(p∗p) .
The algorithm starts from the best known construction (``height_sequence_1`` if
available, otherwise a random Dirichlet draw) and repeatedly moves a small
amount of mass from one index to another.  The self‑convolution is updated
incrementally:
    (p+Δp)∗(p+Δp) = p∗p + 2·p∗Δp + Δp∗Δp .
Thus each proposal can be evaluated in O(n) time by modifying two slices of the
convolution array and correcting three quadratic terms.  A temperature that
decays exponentially from 2·10⁻² to 10⁻⁵ controls the probability of accepting
up‑hill moves.  Periodic renormalisation removes accumulated floating‑point
drift and a long stagnation period triggers a small random restart around the
current best solution.  The best probability vector found before the time budget
expires is returned (clipped to [0,1000] and converted to a plain list) – it
satisfies all security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a step‑function (list of non‑negative heights) that minimises the
    evaluation metric.  The function runs for roughly ``budget_s`` seconds and
    prints intermediate progress.  It uses a simulated‑annealing search on the
    probability simplex with an exact incremental update of the self‑convolution.
    """

    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Problem size ---------------------------------------------------------
    # ----------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 1153                     # historic default

    # ----------------------------------------------------------------------
    # Initialise probability vector (sum == 1) ------------------------------
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals() and rng.random() < 0.5:
        p = np.array(height_sequence_1, dtype=float)
        p = np.clip(p, 0.0, None)
        tot = p.sum()
        if tot == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p = p / tot
    else:
        p = rng.random(n)
        p = p / p.sum()

    # tiny jitter to avoid exact zeros
    p += rng.normal(scale=1e-6, size=n)
    p = np.clip(p, 0.0, None)
    p = p / p.sum()

    # ----------------------------------------------------------------------
    # Initial convolution and score ----------------------------------------
    # ----------------------------------------------------------------------
    conv = np.convolve(p, p)                # length 2n‑1
    cur_score = 2.0 * n * conv.max()
    best_p = p.copy()
    best_score = cur_score

    # ----------------------------------------------------------------------
    # Time budget -----------------------------------------------------------
    # ----------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # ----------------------------------------------------------------------
    # Annealing schedule ----------------------------------------------------
    # ----------------------------------------------------------------------
    T_start = 0.02
    T_end = 1e-5
    max_delta_initial = 0.05   # maximum mass moved in a single step (absolute)

    stagnation = 0
    iteration = 0

    # ----------------------------------------------------------------------
    # Main simulated‑annealing loop ----------------------------------------
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 5.0)
        # exponential cooling
        T = T_start * (T_end / T_start) ** progress

        # ------------------------------------------------------------------
        # Propose a move: shift δ from i (source) to j (target) -------------
        # ------------------------------------------------------------------
        i = rng.choice(n, p=p)               # source biased by current mass
        if p[i] <= 0.0:
            continue
        j = rng.integers(n - 1)               # uniform target, j != i
        if j >= i:
            j += 1

        # step size adapts to temperature (larger early, smaller later)
        max_delta = max_delta_initial * (T / T_start + 0.1)
        max_delta = min(max_delta, p[i])
        if max_delta <= 1e-14:
            continue
        delta = rng.uniform(0.0, max_delta)

        # ------------------------------------------------------------------
        # Incremental update of the convolution -----------------------------
        # ------------------------------------------------------------------
        # copy the current convolution (cheap, length ≈ 2 n)
        conv_new = conv.copy()

        # linear part: 2·p∗Δp
        conv_new[i:i + n] -= 2.0 * delta * p
        conv_new[j:j + n] += 2.0 * delta * p

        # quadratic part: Δp∗Δp contributes at three positions
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        conv_new[idx2i] += delta * delta
        conv_new[idx2j] += delta * delta
        conv_new[idxij] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # ------------------------------------------------------------------
        # Metropolis acceptance ---------------------------------------------
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill move: accept with probability exp(-(ΔM)/T)
            prob = np.exp((cur_score - new_score) / (T + 1e-16))
            if rng.random() < prob:
                accept = True

        if accept:
            # apply the move
            p[i] -= delta
            p[j] += delta
            # numerical safeguard
            if p[i] < 0.0:
                p[i] = 0.0
            conv = conv_new
            cur_score = new_score

            # record global best
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_p = p.copy()
                stagnation = 0
                print(f"[SA] Iter {iteration:9d} – new best: {best_score:.6f}")
            else:
                stagnation += 1
        else:
            stagnation += 1

        # ------------------------------------------------------------------
        # Periodic renormalisation (removes drift) -------------------------
        # ------------------------------------------------------------------
        if iteration % 20000 == 0:
            total = p.sum()
            if abs(total - 1.0) > 1e-12:
                p = p / total
                conv = np.convolve(p, p)
                cur_score = 2.0 * n * conv.max()
                print(f"[Renorm] Iter {iteration:9d} – cur_score {cur_score:.6f}")

        # ------------------------------------------------------------------
        # Restart if stuck for a long time ---------------------------------
        # ------------------------------------------------------------------
        if stagnation >= 300_000:
            p = best_p + rng.normal(scale=0.001, size=n)
            p = np.clip(p, 0.0, None)
            p = p / p.sum()
            conv = np.convolve(p, p)
            cur_score = 2.0 * n * conv.max()
            stagnation = 0
            print(f"[Restart] Iter {iteration:9d} – renewed, cur_score {cur_score:.6f}")

        # ------------------------------------------------------------------
        # Early exit if target already reached -------------------------------
        # ------------------------------------------------------------------
        if best_score <= 1.503:
            print(f"[Target] reached {best_score:.6f} at iter {iteration}")
            break

    # ----------------------------------------------------------------------
    # Final cheap random sampling (uses any remaining budget) ---------------
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        p_rand = rng.random(n)
        p_rand = p_rand / p_rand.sum()
        score = 2.0 * n * np.max(np.convolve(p_rand, p_rand))
        if score < best_score:
            best_score = score
            best_p = p_rand.copy()
            print(f"[Random] New best {best_score:.6f}")
        # leave the loop shortly before the deadline
        if time.time() + 0.05 > deadline:
            break

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list) ---------------
    # ----------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Best score achieved: {best_score:.6f}")
    return result
