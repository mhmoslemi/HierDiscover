    n = len(p)
    rows = 2*n - 1
    # Build A
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):

# -*- coding: utf-8 -*-
"""
Hybrid optimizer for the “min‑max self‑convolution” problem.

The search works in two stages:

1.  A fast Adam‑based descent on a smooth log‑sum‑exp approximation of the
    maximum of the self‑convolution.  This quickly finds a decent region.
2.  An exact linear‑programming refinement.  For the current iterate ``p`` we
    compute ``M = max(p * p)`` and solve the linear program

        maximise    1ᵀ g
        subject to  conv(p, g) ≤ M ,   g ≥ 0

    The solution ``g`` has a larger ℓ₁‑norm while respecting the same convolution
    bound; after normalising ``g`` we move the iterate towards ``g`` (full
    replacement or a convex combination).  The LP step provably never increases
    the objective and often yields a noticeable drop.

Both stages respect the non‑negativity simplex constraint via a Euclidean
projection onto the probability simplex.  The procedure runs until the
allocated wall‑clock budget expires and returns the best sequence seen.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex {x | x_i ≥ 0 , Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------
def _compute_score(p: np.ndarray, m_fft: int) -> float:
    """Return 2·n·max(p∗p) for a probability vector p (sum(p)=1)."""
    n = p.shape[0]
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2 * n - 1]
    return float(2.0 * n * conv.max())

def _smooth_gradient(p: np.ndarray, alpha: float, m_fft: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp approximation
        f(p) = (1/α)·log∑ₖ exp(α·(p∗p)ₖ)
    Returns a vector of size n.
    """
    n = p.shape[0]
    p_fft = np.fft.rfft(p, n=m_fft)

    # self‑convolution
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2 * n - 1]

    # soft‑max weights
    z = alpha * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)               # stability
    w = exp_z / exp_z.sum()                 # ∑w = 1, gives soft‑max

    # gradient = 2 * (p ⋆ w_rev)  (⋆ = correlation)
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[:3 * n - 2]

    idx = 2 * n - 2 - np.arange(n)          # map correlation → gradient
    grad = 2.0 * corr_full[idx]
    return grad

def _lp_step(p: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀ g   s.t. conv(p, g) ≤ M , g ≥ 0 .
    Returns the normalised solution g / sum(g) (sum(g) > 0) or None
    if the linear program fails.
    """
    n = p.shape[0]
    rows = 2 * n - 1

    # Build the (rows × n) convolution matrix A where (A g)_k = ∑_i p_i g_{k‑i}
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = p  # each column is a shifted copy of p

    # Linear program data
    c = -np.ones(n)                     # maximize sum(g) → minimise -sum(g)
    b_ub = np.full(rows, M)             # upper bound for each convolution entry
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    return g / s                         # normalise to the simplex

def _try_mix(p: np.ndarray, g: np.ndarray,
             compute_score, best_score: float) -> tuple[np.ndarray, float]:
    """

# -*- coding: utf-8 -*-
"""
Hybrid Adam + linear‑programming optimizer for the min‑max self‑convolution
task.

The search proceeds in two stages:

1.  A fast Adam optimiser works on a smooth log‑sum‑exp approximation of
    the maximal entry of the self‑convolution.  This quickly finds a decent
    region of the feasible simplex.

2.  A deterministic LP refinement: for the current iterate *p* we compute
    M = max(p∗p) and solve

        maximise    1ᵀ g
        subject to   conv(p, g) ≤ M ,   g ≥ 0 .

    The LP yields a vector *g* whose ℓ₁‑norm can be larger than that of *p*
    while respecting the same convolution bound.  After normalising *g* to the
    simplex we either replace *p* by *g* or move part‑way towards *g*.
    The LP step never increases the objective and often produces a noticeable
    improvement.

Both phases respect the simplex constraint via a Euclidean projection onto
the probability simplex.  The algorithm runs until the allotted wall‑clock
budget expires and returns the best sequence observed.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex {x | x_i ≥ 0 , Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Core utilities
# ----------------------------------------------------------------------
def _compute_score(p: np.ndarray, m_fft: int) -> float:
    """Return 2·n·max(p∗p) for a probability vector p (∑p = 1)."""
    n = p.shape[0]
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2 * n - 1]
    return float(2.0 * n * conv.max())


def _smooth_gradient(p: np.ndarray, alpha: float, m_fft: int) -> np.ndarray:
    """
    Gradient of the smooth approximation
        f(p) = (1/α)·log Σₖ exp(α·(p∗p)ₖ)
    Returns a vector of size n.
    """
    n = p.shape[0]
    p_fft = np.fft.rfft(p, n=m_fft)

    # self‑convolution
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2 * n - 1]

    # soft‑max weights (log‑sum‑exp trick)
    z = alpha * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()               # Σ w = 1

    # gradient = 2·(p ⋆ w_rev)  (⋆ = correlation)
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[:3 * n - 2]

    idx = 2 * n - 2 - np.arange(n)        # map correlation → gradient
    grad = 2.0 * corr_full[idx]
    return grad


def _lp_step(p: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀ g   s.t. conv(p, g) ≤ M , g ≥ 0 .
    Returns the normalised solution g / sum(g) (or None if the LP fails).
    """
    n = p.shape[0]
    rows = 2 * n - 1

    # Build convolution matrix A: (A·g)_k = Σ_i p_i g_{k-i}
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = p

    c = -np.ones(n)                     # maximise sum(g) → minimise -sum(g)
    b_ub = np.full(rows, M)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return None
    return g / s                         # back to the simplex
