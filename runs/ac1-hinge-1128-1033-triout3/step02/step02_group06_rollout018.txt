"""
propose_candidate
-----------------

Search for a step‑function that beats the classic central‑binomial construction.
The idea is simple: the central binomial coefficients give a self‑convolution
that is exactly 1 for all interior shifts, which yields the asymptotic score
π/2 ≈ 1.5708.  By moving a tiny amount of mass from the large early
coefficients to later positions we can lower the peak of the convolution
while keeping the total L¹‑mass unchanged.  The objective
    2·n·max(conv) / (sum a)²
is invariant under a global scaling, so we keep the L¹‑mass fixed and
only try to reduce the maximal convolution value.

The algorithm:
1. Build a long central‑binomial prefix (default 8 000 entries).
2. Repeatedly pick an “early’’ index (where the coefficient is large) and a
   “late’’ index, move a small random amount of mass from the early to the
   late entry, and keep the change only if it improves the evaluation
   score.
3. The step size (σ) is slowly annealed; occasional re‑starts of the
   acceptance counter keep the search active.
4. The best sequence found before the time budget expires is returned.

The whole procedure is simple, uses only NumPy and the provided
`evaluate_sequence`, and fits comfortably inside the 1000 s limit.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper: central‑binomial coefficients (b_k = C(2k,k)/4**k)
# ----------------------------------------------------------------------
def _central_binomial_prefix(length: int) -> np.ndarray:
    """
    Return an array `a` with a[k] = C(2k,k) / 4**k for 0 <= k < length.
    The recurrence a[0]=1, a[k+1] = a[k] * (2k+1)/(2(k+1)) is used.
    """
    a = np.empty(length, dtype=np.float64)
    cur = 1.0
    a[0] = cur
    for k in range(1, length):
        # a_{k} = a_{k-1} * (2k-1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        a[k] = cur
    return a


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex.
# Guarantees non‑negative entries that sum to 1.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project vector `v` onto the simplex {x : x_i >= 0, sum x_i = 1}.
    Returns a new array with the projected values.
    """
    n = v.shape[0]
    # shift by the mean to speed up the sorting‑based algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     max_len: int = 8000,
                     **kwargs) -> List[float]:
    """
    Produce a non‑negative height sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget (seconds).  The function stops a few seconds
        before the limit to guarantee a return value.
    max_len : int
        Length of the initial central‑binomial prefix.  Larger values give
        a better starting point but increase the cost of each evaluation.

    Returns
    -------
    List[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – initialise with the classic central‑binomial sequence
    # ------------------------------------------------------------------
    a = _central_binomial_prefix(max_len)      # a[0] = 1, a decreasing
    n = max_len
    seq = a.copy()                             # work on a mutable copy

    # Normalise to unit L¹‑mass (scale is irrelevant for the objective)
    total = seq.sum()
    seq = seq / total

    # Initial score (for logging)
    best_score = evaluate_sequence(seq.tolist())
    best_seq = seq.copy()
    print(f"[init] length={n}, initial score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – local mass‑transfer search
    # ------------------------------------------------------------------
    # Parameters controlling the search
    sigma = 0.04                     # relative size of a transferred piece
    early_frac = 0.20                # first 20% of indices are regarded as “early”
    early_end = max(1, int(n * early_frac))
    iter_cnt = 0
    no_improve = 0
    max_no_improve = 5000           # after many failures we shrink sigma

    while time.time() < deadline:
        # pick an early (large) index and a later (small) index
        i = np.random.randint(early_end)          # 0 <= i < early_end
        j = np.random.randint(early_end, n)      # early_end <= j < n

        # propose moving a small random amount of mass from i to j
        max_delta = seq[i] * sigma
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * max_delta

        seq[i] -= delta
        seq[j] += delta

        # evaluate the perturbed sequence
        cand_score = evaluate_sequence(seq.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_seq = seq.copy()
            no_improve = 0
            # slowly tighten the step size to allow fine adjustments
            sigma = max(0.001, sigma * 0.995)
            print(f"[improve] iter={iter_cnt}, score={best_score:.9f}, "
                  f"sigma={sigma:.5f}, moved {delta:.6f} from {i}->{j}")
        else:
            # revert the move
            seq[i] += delta
            seq[j] -= delta
            no_improve += 1

        iter_cnt += 1

        # If we have been unsuccessful for a while, shrink sigma to explore
        # finer neighbourhoods.
        if no_improve >= max_no_improve:
            sigma = max(0.001, sigma * 0.5)
            no_improve = 0
            print(f"[anneal] iter={iter_cnt}, reducing sigma to {sigma:.5f}")

    # ------------------------------------------------------------------
    # Phase 3 – final clean‑up and return
    # ------------------------------------------------------------------
    # Small negative numerical noise is clipped; convert to plain Python floats.
    final_seq = [float(max(0.0, x)) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[final] length={len(final_seq)}, best score={final_score:.9f}")
    return final_seq
