"""
Propose a step‐function sequence that minimises ``evaluate_sequence``.

The algorithm has two phases:

1. **Smooth‐max optimisation** – minimise a smooth approximation of the
   max‑convolution using Adam and a soft‑max (log‑sum‑exp) over the full
   convolution.  The temperature (γ) is increased from a small value to a
   large one, turning the smooth objective into the true max as optimisation
   proceeds.  After each gradient step the vector is projected onto the
   probability simplex (non‑negative, sum = 1).

2. **Sub‑gradient refinement** – after the smooth phase we switch to the
   classic sub‑gradient method on the exact max‑convolution (as used in the
   previous state‑of‑the‑art).  This clean‑up often yields a slight further
   improvement.

Both phases run until the time budget is exhausted.  The best sequence seen
throughout the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1           # zero‑based
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:                  # numerical safety
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full (linear) convolution a*a using a real FFT (fast)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full (length 2·n‑1) convolution a*a."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    np.maximum(conv, 0, out=conv)               # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp) w.r.t. a.
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of (1/γ)·log Σ exp(γ·conv) w.r.t. a.
    Uses FFT‑based correlation to compute g_i = 2· Σ_k w_k·a_{k‑i},
    where w_k is the soft‑max weight of conv[k].
    """
    n = a.shape[0]
    # stable soft‑max on conv
    c_max = conv.max()
    exp_weights = np.exp(gamma * (conv - c_max))
    w = exp_weights / exp_weights.sum()            # shape (2n‑1,)

    # correlation w ⋆ a  (equivalently conv(w_rev, a))
    w_rev = w[::-1]                                 # reversed weights
    size = 1 << ((len(w_rev) + n - 1).bit_length())
    fw = np.fft.rfft(w_rev, size)
    fa = np.fft.rfft(a, size)
    corr_full = np.fft.irfft(fw * fa, size)[:len(w_rev) + n - 1]

    # extract b_i = Σ_k w_k·a_{k‑i} for i=0..n‑1
    # the needed slice starts at position 2n‑2‑i
    idx = 2 * n - 2 - np.arange(n)
    b = corr_full[idx]

    # gradient of the smooth max (we minimise, so move opposite)
    return 2.0 * b


# ----------------------------------------------------------------------
# Phase 1 – smooth‑max optimisation with Adam and simplex projection
# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Run the smooth‑max Adam optimiser until ``deadline``."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = float('inf')
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    start = time.time()

    # schedules
    eta_start, eta_end = 0.30, 1e-5          # base learning‑rate schedule
    gamma_start, gamma_end = 0.5, 80.0     # temperature schedule

    while True:
        now = time.time()
        if now >= deadline:
            break
        # ------------------------------------------------------------------
        # evaluate current point
        # ------------------------------------------------------------------
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[smooth] it={it:6d}  val={best_val:.6f}")

        # ------------------------------------------------------------------
        # compute adaptive parameters
        # ------------------------------------------------------------------
        progress = (now - start) / (deadline - start)   # in [0,1]
        eta = eta_start * (1.0 - progress) + eta_end * progress
        gamma = gamma_start + (gamma_end - gamma_start) * (progress ** 2)

        # ------------------------------------------------------------------
        # gradient of the smooth max
        # ------------------------------------------------------------------
        grad = _smooth_max_gradient(a, conv, gamma)

        # ------------------------------------------------------------------
        # Adam update (un‑biased estimates)
        # ------------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------------
        # take step and project back onto the simplex
        # ------------------------------------------------------------------
        a = _project_onto_simplex(a - step)

        # occasional tiny random jitter to escape plateaus
        if np.random.rand() < 0.005:
            # move a random fraction of mass from one entry to another
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = min(a[src], np.random.rand() * 0.05 * a[src])
            a[src] -= delta
            a[dst] += delta
            a = _project_onto_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – classic sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_refine(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Run a plain sub‑gradient descent (as in the previous best code)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    start = time.time()

    # linear step‑size schedule
    eta_start, eta_end = 0.12, 1e-5

    while True:
        now = time.time()
        if now >= deadline:
            break

        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgrad] it={it:6d}  val={best_val:.6f}")

        # ------------------------------------------------------------------
        # sub‑gradient of the active convolution peak
        # ------------------------------------------------------------------
        k_star = int(np.argmax(conv))
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        subgrad = np.zeros_like(a)
        subgrad[idx] = 2.0 * a[k_star - idx]

        # ------------------------------------------------------------------
        # step‑size (linear cooling)
        # ------------------------------------------------------------------
        progress = (now - start) / (deadline - start)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # ------------------------------------------------------------------
        # take a step and project back onto the simplex
        # ------------------------------------------------------------------
        a_candidate = a - eta * subgrad

        # small random jump occasionally
        if np.random.rand() < 0.01:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = min(a_candidate[src], np.random.rand() * 0.2 * a_candidate[src])
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        a = _project_onto_simplex(a_candidate)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step sequence.

    The routine:
      1. Builds an initial distribution (interpolated from the supplied
         ``height_sequence_1`` if it exists, otherwise a random Dirichlet).
      2. Runs the smooth‑max Adam optimiser for roughly half the time budget.
      3. Refines the result with a classic sub‑gradient descent for the
         remaining time.
      4. Returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # -------------------------------------------------------------
    # Choose a working length n
    # -------------------------------------------------------------
    # Longer vectors give more flexibility; 3000 is a good compromise.
    n = 3000

    # -------------------------------------------------------------
    # Initialise a: try to reuse the previously found construction,
    # otherwise fall back to a random Dirichlet vector.
    # -------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # Upsample / interpolate to the new length n
        old_n = old.shape[0]
        if old_n != n:
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() <= 0.0:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()
    except Exception:
        a0 = np.random.dirichlet(np.ones(n))

    # -------------------------------------------------------------
    # Phase 1 – smooth‑max optimisation (≈ 50 % of the time budget)
    # -------------------------------------------------------------
    phase1_deadline = start_time + (budget_s * 0.5)
    phase1_deadline = min(phase1_deadline, deadline)
    best_a, best_val = _smooth_optimize(a0, phase1_deadline)

    # -------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement (remaining time)
    # -------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_refine(best_a, deadline)

    # -------------------------------------------------------------
    # Final reporting (optional but useful)
    # -------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Ensure we return a Python list as required by the framework
    return best_a.tolist()
