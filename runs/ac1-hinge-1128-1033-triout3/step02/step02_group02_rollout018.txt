import time
import numpy as np

def max_conv_fft(a):
    """Return max entry of self-convolution of a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2 * n - 1]
    return float(conv.max())

def full_conv_fft(a):
    """Return full self-convolution array (length 2n-1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2 * n - 1]
    return conv

def correlation_fft(w, a):
    """Linear cross‑correlation (w ⋆ a) for non‑negative lag 0..len(a)-1.
    Returns array of length len(a) with (w⋆a)[i] = Σ_k w_k a_{k-i}.
    """
    n = a.shape[0]
    m = w.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a, n=size)
    corr_full = np.fft.irfft(W * np.conj(A), n=size)[: n + m - 1]
    # first n entries correspond to lag 0..n-1
    return corr_full[:n]

def decreasing_power_law(n, p):
    """Monotone decreasing power‑law a[i] ∝ (i+1)^{-p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + 1e-15
    a /= a.sum()
    return a

def decreasing_exponential(n, alpha):
    """Monotone decreasing exponential a[i] ∝ exp(-α i)."""
    i = np.arange(n)
    a = np.exp(-alpha * i) + 1e-15
    a /= a.sum()
    return a

def u_shaped_distribution(n, p):
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-15
    a /= a.sum()
    return a

def sum_tails_distribution(n, p):
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-15
    a /= a.sum()
    return a

def exponential_distribution(n, alpha):
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-15
    a /= a.sum()
    return a

def approx_score(a):
    """Score assuming sum(a)=1: 2 * n * max_conv_fft(a)."""
    n = len(a)
    return 2.0 * n * max_conv_fft(a)

    Returns the best vector found and its approximated score.

    n = len(a)
    best_a = a.copy()
    conv = full_conv_fft(a)

    A = np.fft.rfft(a, n=size)

    """Linear cross‑correlation (w⋆a) for non‑negative lags.

    n = len(a)
    return 2.0 * n * max_conv_fft(a)


def add_candidate(candidates, a, sc, K):
    """Maintain a sorted list of at most K best (score, array) pairs."""
    if len(candidates) < K:
        candidates.append((sc, a.copy()))
        candidates.sort(key=lambda x: x[0])
    else:
        if sc < candidates[-1][0]:
            candidates[-1] = (sc, a.copy())
            candidates.sort(key=lambda x: x[0])


def refine_on_simplex(start_a, deadline, target_approx):
    """
    Exponentiated‑gradient (mirror‑descent) refinement on the simplex.
    Uses a soft‑max over the convolution entries to obtain a smooth
    surrogate of the maximum.  Returns the best vector found and its
    approximated score (2·n·max_conv).
    """
    a = start_a.copy()
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * full_conv_fft(a).max()

    beta = 0.5          # temperature for the soft‑max
    max_beta = 80.0
    eta = 0.2           # base step size for the mirror update
    iteration = 0
    no_improve = 0

    while time.time() < deadline - 2.0:
        iteration += 1
        conv = full_conv_fft(a)
        max_c = conv.max()

        # soft‑max weights (shifted by max to keep numbers ≤ 1)
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient of the smooth maximum (mirror‑descent on the simplex)
        corr = correlation_fft(w, a)               # length n
        diag = w[2 * np.arange(n)] * a             # diagonal correction
        grad = 2.0 * corr - diag

        # back‑tracking line search on η
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            eta_try = eta * factor
            expo = -eta_try * grad
            expo = np.clip(expo, -50.0, 0.0)       # keep exp stable
            cand = a * np.exp(expo)
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            cand_score = 2.0 * n * full_conv_fft(cand).max()
            if cand_score < best_score:
                a = cand
                best_a = cand.copy()
                best_score = cand_score
                improved = True
                print(f"Iter {iteration:6d}, β={beta:.2f}, η={eta_try:.4f}, new best≈{best_score:.6f}")
                break

        if not improved:
            # occasional jitter to escape flat regions
            if np.random.rand() < 0.03:
                jitter = np.random.randn(n) * 1e-4
                cand = a + jitter
                cand = np.maximum(cand, 0.0)
                cand /= cand.sum()
                cand_score = 2.0 * n * full_conv_fft(cand).max()
                if cand_score < best_score:
                    a = cand
                    best_a = cand.copy()
                    best_score = cand_score
                    print(f"Iter {iteration:6d}, jitter accept, new best≈{best_score:.6f}")

        # schedule updates
        beta = min(beta * 1.02, max_beta)
        eta *= 0.9995

        # stagnation handling
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 800 and iteration % 100 == 0:
                a = best_a + np.random.rand(n) * 1e-3
                a = np.maximum(a, 0.0)
                a /= a.sum()

        if best_score < target_approx:
            print(f"Target approximated score reached: {best_score:.6f} at iteration {iteration}")
            break

    return best_a, best_score

# Main entry point required by the competition
# ----------------------------------------------------------------------
def propose_candidate(seed=123, budget_s=1000):
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The method consists of a wide parametric sweep (symmetric and
    asymmetric families) followed by a soft‑max‑based exponentiated‑gradient
    refinement.  It respects the given time budget and finally returns the
    best list of heights found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Seed from a provided global variable if available.
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
            else:
                external = external / s
            sc = approx_score(external)
            best_seq = external.copy()
            best_score = sc
            print(f"Seeded from height_sequence_1: approx score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Deterministic parametric sweep (symmetric + asymmetric families)
    # ------------------------------------------------------------------
    lengths = [5000, 8000, 12000, 16000, 20000, 30000]
    p_vals = np.linspace(0.2, 3.0, 28)       # 0.2 … 3.0
    alphas = np.linspace(0.0002, 0.008, 15) # 0.0002 … 0.008
    top_K = 6
    top_candidates = []  # list of (score, array)

    print("\n=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for p in p_vals:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            sc = approx_score(a)
            add_candidate(top_candidates, a, sc, top_K)
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            sc = approx_score(a)
            add_candidate(top_candidates, a, sc, top_K)
        for a_alpha in alphas:
            if time.time() > deadline:
                break
            a = exponential_distribution(n, a_alpha)
            sc = approx_score(a)
            add_candidate(top_candidates, a, sc, top_K)
        # asymmetric families
        for p in p_vals:
            if time.time() > deadline:
                break
            a = decreasing_power_law(n, p)
            sc = approx_score(a)
            add_candidate(top_candidates, a, sc, top_K)
        for a_alpha in alphas:
            if time.time() > deadline:
                break
            a = decreasing_exponential(n, a_alpha)
            sc = approx_score(a)
            add_candidate(top_candidates, a, sc, top_K)

    # ------------------------------------------------------------------
    # 2.  Initialise best solution from sweep (or fall back to random)
    # ------------------------------------------------------------------
    for sc, a in top_candidates:
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"Initial best from sweep: n={len(a)}, approx score={sc:.6f}")

    if best_seq is None:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = approx_score(a)
        print(f"Fallback random init: n={n}, approx score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Refine the few top candidates with exponentiated gradient.
    # ------------------------------------------------------------------
    target_approx = 0.7515          # ½ · 1.503 (we want 2·n·max < 1.503)
    refined_best_seq = best_seq
    refined_best_score = best_score

    # sort candidates by quality (best first)
    candidates = sorted(top_candidates, key=lambda x: x[0])

    for idx, (init_sc, init_a) in enumerate(candidates):
        if time.time() > deadline:
            break
        print(f"\n--- Refinement #{idx+1} (len={len(init_a)}) ---")
        cand_seq, cand_score = refine_on_simplex(init_a, deadline, target_approx)
        if cand_score < refined_best_score:
            refined_best_score = cand_score
            refined_best_seq = cand_seq.copy()
            print(f"Refinement #{idx+1} new global best approx={cand_score:.6f}")
        if refined_best_score * 2.0 < 1.503:
            print("Exact target reached in approximation, exiting early.")
            break

    # ------------------------------------------------------------------
    # 4.  Final exact evaluation (once) and reporting.
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(refined_best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = refined_best_score * 2.0   # fallback approximation

    print("\n=== Search finished ===")
    print(f"Best exact/evaluated score: {exact_score:.6f}")
    print(f"Best sequence length      : {len(refined_best_seq)}")
    return refined_best_seq.tolist()
