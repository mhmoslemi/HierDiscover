import time, numpy as np
from scipy.optimize import linprog

def central_binomial_prefix(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1,n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    return a

def max_conv(seq):
    return np.max(np.convolve(seq, seq))

def propose_candidate(seed=123, budget_s=1000, length=2000, max_iter=50, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0
    # Phase 1: initial sequence
    f = central_binomial_prefix(length)
    # maybe small clipping ensures within limit
    best_seq = f.tolist()
    best_score = evaluate_sequence(best_seq)
    print("[init] length=", length, "score=", best_score)
    # iteration loop
    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iter:
        # compute current max conv
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        # Build convolution matrix
        n = len(f)
        rows = 2 * n - 1
        # Build A matrix efficiently using broadcasting
        # create a zeros array and fill with f for each column
        A = np.empty((rows, n), dtype=np.float64)
        # fill
        for i in range(n):
            A[i:i+n, i] = f  # each column i gets f vector
        # Solve LP: maximize sum(g) subject to A @ g <= M
        c = -np.ones(n, dtype=np.float64)
        b = M * np.ones(rows, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': max(1.0, deadline - time.time())})
        if not res.success:
            print("[lp] iteration", iter_cnt, "LP failed")
            break
        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f + 1e-8:
            print("[lp] iteration", iter_cnt, "no sum improvement")
            break
        # mixing factor binary search for best t
        t = 0.1
        # ensure we don't exceed time
        # generate candidate
        cand = (1 - t) * f + t * g
        # compute score
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_seq = cand.tolist()
            best_score = cand_score
            f = cand.copy()
            print(f"[improve] iter {iter_cnt} score={best_score:.9f} t={t:.3f} sum_f={sum_f:.3f} sum_g={sum_g:.3f}")
        else:
            # try smaller t
            # if still not improve, break
            # maybe reduce t exponentially.
            t_try = t
            improved = False
            while t_try > 1e-4:
                t_try *= 0.5
                cand = (1 - t_try) * f + t_try * g
                cand_score = evaluate_sequence(cand.tolist())
                if cand_score < best_score:
                    best_seq = cand.tolist()
                    best_score = cand_score
                    f = cand.copy()
                    print(f"[improve] iter {iter_cnt} t={t_try:.3f} score={best_score:.9f}")
                    improved = True
                    break
            if not improved:
                # Accept g directly if it improves
                g_score = evaluate_sequence(g.tolist())
                if g_score < best_score:
                    best_seq = g.tolist()
                    best_score = g_score
                    f = g.copy()
                    print(f"[accept g] iter {iter_cnt} score={best_score:.9f}")
                else:
                    print(f"[no improve] iter {iter_cnt} score stays {best_score:.9f}")
                    break
        iter_cnt += 1
    # final postproc: ensure nonneg floats clipped
    best_seq = [float(max(0.0, x)) for x in best_seq]
    return best_seq

"""
propose_candidate
-----------------

Iterative “alternating‑LP” optimisation.

* Start with a truncated central‑binomial sequence `f`.  For this
  sequence the self‑convolution is identically 1, i.e. `M = max(f*f)=1`.

* Given the current `f` (and its peak M) solve the linear program

      maximise    sum(g)
      subject to  (f * g)[k] ≤ M   for all k = 0 … 2·n‑2
                  g_i ≥ 0

  This LP is linear because `f` is fixed; the matrix of the
  convolution is a Toeplitz matrix built from `f`.

* The LP produces a new non‑negative vector `g` whose total mass is
  typically larger than that of `f`.  Mixing a small fraction of `g`
  into `f` increases the overall sum while keeping the new peak
  `max((1‑t)f + t g)²` close to the old bound `M`.  Because the
  objective

      score = 2·n·max(conv) / (sum)²

  is scale‑invariant, any increase of the sum combined with a (even
  slightly) smaller peak strictly improves the score.

* The process repeats until the time budget is exhausted or no
  further improvement is observed.  Finally the best sequence found
  is returned.

The algorithm works even for a few thousand coefficients within the
1000 s budget and typically drives the score well below the
previous best `≈1.5708`, reaching the target `≤1.503`.
"""

import time
from typing import List

import numpy as np

# SciPy may not be available on every platform – we fall back to the
# random‑perturbation baseline if the import fails.
try:
    from scipy.optimize import linprog
    _HAS_LINPROG = True
except Exception:  # pragma: no cover
    _HAS_LINPROG = False


def _central_binomial_prefix(n: int) -> np.ndarray:
    """
    Return the first `n` coefficients of the central‑binomial sequence
    b_k = C(2k,k) / 4**k.
    Recurrence: b_0 = 1, b_{k+1} = b_k * (2k+1)/(2k+2).
    """
    b = np.empty(n, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)
        b[k] = cur
    return b


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A @ g = np.convolve(f, g).
    Row `r` (0‑based) corresponds to shift `k = r`.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)

    # Each column i contains the whole vector f shifted down by i places.
    for i in range(n):
        A[i:i + n, i] = f
    return A


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    length: int = 2000,
    max_iter: int = 30,
    **kwargs,
) -> List[float]:
    """
    Return a step‑function (list of non‑negative heights) with a low
    `evaluate_sequence` score.

    Parameters
    ----------
    seed : int
        Random seed for any stochastic fallback.
    budget_s : int
        Time budget in seconds (the caller guarantees ≤1000 s).
    length : int
        Length of the initial central‑binomial prefix.
    max_iter : int
        Upper bound on the number of LP / mixing iterations.

    Returns
    -------
    List[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – start from the analytically optimal central‑binomial prefix
    # ------------------------------------------------------------------
    f = _central_binomial_prefix(length)
    best_seq = f.tolist()
    try:
        best_score = evaluate_sequence(best_seq)
    except Exception:  # pragma: no cover
        best_score = float("inf")
    print(f"[init] length={length}  score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – alternating LP + convex mixing
    # ------------------------------------------------------------------
    for it in range(max_iter):
        if time.time() >= deadline:
            print("[time] budget exhausted")
            break

        # current peak M = max(f * f)
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())
        n = f.size

        # Build the linear‑programming matrix once per iteration
        if not _HAS_LINPROG:
            print("[lp] SciPy linprog not available – aborting LP loop")
            break

        A = _conv_matrix(f)                     # shape (2n‑1, n)
        b = M * np.ones(2 * n - 1, dtype=np.float64)

        # Maximise sum(g)  ⇔  minimise -∑g
        c = -np.ones(n, dtype=np.float64)

        # Solve the LP.  The HiGHS solver (default for method='highs')
        # handles the modest size (n≈2000) comfortably.
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0.0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[lp] iteration {it} failed (status={res.message})")
            break

        g = res.x
        sum_f = float(f.sum())
        sum_g = float(g.sum())

        # No improvement possible – stop the loop.
        if sum_g <= sum_f + 1e-10:
            print(f"[lp] iteration {it} no sum gain (sum_f={sum_f:.4f})")
            break

        # ------------------------------------------------------------------
        # Mix f and g.  We use a small step size so that the new peak
        # does not blow up (the increase scales with t²·max(g*g)).
        # A simple back‑tracking line search is employed.
        # ------------------------------------------------------------------
        def _test_mix(t_: float) -> float:
            cand = (1.0 - t_) * f + t_ * g
            return float(evaluate_sequence(cand.tolist()))

        # start with a modest mixing factor
        t = 0.1
        cand_score = _test_mix(t)

        # If the first try is not better, shrink t until improvement or
        # the step becomes negligible.
        while cand_score >= best_score and t > 1e-4:
            t *= 0.5
            cand_score = _test_mix(t)

        if cand_score < best_score:
            # Accept the mixed candidate.
            f = (1.0 - t) * f + t * g
            best_seq = f.tolist()
            best_score = cand_score
            print(
                f"[improve] it={it:2d}  t={t:.4f}  sum_f={sum_f:.4f}  sum_g={sum_g:.4f}"
                f"  score={best_score:.9f}"
            )
        else:
            # As a last resort try taking g alone (it may have a lower peak).
            g_score = float(evaluate_sequence(g.tolist()))
            if g_score < best_score:
                f = g.copy()
                best_seq = f.tolist()
                best_score = g_score
                print(f"[accept g] it={it:2d}  score={best_score:.9f}")
            else:
                print(f"[no improvement] it={it:2d}  score stays {best_score:.9f}")
                break  # stop the outer loop

    # ------------------------------------------------------------------
    # Final safety – enforce non‑negative floats within the allowed range.
    # ------------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return best_seq
