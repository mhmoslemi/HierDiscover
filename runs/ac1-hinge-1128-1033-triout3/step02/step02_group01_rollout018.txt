def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np, scipy.optimize as opt, scipy.sparse as sp

    deadline = time.time() + budget_s - 5

    np.random.seed(seed)

    # Choose length
    n = 2500
    # If external height_sequence_1 exists and length >= n, downsample it.
    try:
        ext = np.array(height_sequence_1, dtype=float)
        if ext.size >= n:
            # downsample by averaging consecutive blocks
            factor = ext.size // n
            f0 = ext[:factor*n].reshape(n, factor).mean(axis=1)
        else:
            # upsample or pad
            f0 = np.pad(ext, (0, n - ext.size), 'constant')
    except NameError:
        # fallback to uniform
        f0 = np.ones(n)

    # Normalize to sum = sqrt(2*n)
    target_sum = np.sqrt(2.0 * n)
    f = f0.copy()
    f = np.maximum(f, 0.0)
    if f.sum() == 0:
        f = np.ones(n)
    f = f / f.sum() * target_sum

    best_f = f.copy()
    # compute score via evaluate_sequence for vector
    best_score = evaluate_sequence(best_f.tolist())

    iteration = 0
    t = 0.5  # mixing parameter

    while time.time() < deadline:
        iteration += 1
        # compute convolution
        conv = np.convolve(f, f)
        M = conv.max()
        # Build sparse constraint matrix A: shape (2n-1, n)
        rows = []
        cols = []
        data = []
        for k in range(2*n-1):
            # i ranges
            i_start = max(0, k - (n - 1))
            i_end = min(n-1, k)
            # for each i, j = k-i
            # coefficient on variable g_j = f_i
            # we fill into row k, column j
            # i_start..i_end inclusive
            # For vectorized fill, we can loop over i
            for i in range(i_start, i_end+1):
                j = k - i
                rows.append(k)
                cols.append(j)
                data.append(f[i])
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # Set up LP
        c = -np.ones(n)  # maximize sum(g) => minimize -sum(g)
        # bounds: g_j >= 0
        bounds = [(0, None)] * n
        b_ub = np.full(2*n-1, M)
        # Solve LP
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fallback to gradient
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # scale g to have target_sum
        g = g * (target_sum / sum_g)

        # mixing
        f_new = (1 - t) * f + t * g
        # Prepare for next iteration
        f = f_new

        # Evaluate score
        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"Iter {iteration}: new best score {best_score:.6f}")

        # Reduce mixing parameter slowly to ensure convergence
        t = max(0.05, t * 0.99)

        # check time
        if time.time() > deadline:
            break

    print("Search finished, best score", best_score)
    return best_f.tolist()

"""
Propose a step‑function with a very low value of `evaluate_sequence`.

Algorithm
---------
1.  Initialise a non‑negative vector `f` of length *n* (default 2500).
    If a previous construction `height_sequence_1` exists it is down‑sampled;
    otherwise a uniform vector is used.  `f` is scaled so that
    sum(f) = sqrt(2·n), which makes the evaluation
    `evaluate_sequence(f)` equal to the maximal self‑convolution entry.

2.  Repeatedly improve `f` by solving a linear programme:
        maximise 1ᵀ·g
        s.t.    g ≥ 0
                f * g ≤ M            (element‑wise)
    where `M = max(f*f)`.  The convolution `f*g` is linear in `g`,
    therefore the constraints are linear.  The LP is solved with
    SciPy's HiGHS interior‑point solver.  The optimal `g` is
    rescaled to the same sum as `f` and mixed with `f`:
        f ← (1‑t)·f + t·g   (t slowly decays).

3.  After the LP phase a short projected sub‑gradient descent
    (with simulated‑annealing acceptance) cleans up remaining
    irregularities.

4.  The best vector found during the whole run is returned as a plain
    Python list.  All entries are non‑negative and comfortably below
    the clipping bound 1000, so the external `evaluate_sequence`
    sees exactly the intended values.

The whole routine respects the given time budget (default 1000 s)
and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w >= 0, sum(w) = 1 }.
    O(n log n) implementation from
    Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the simplex { w >= 0, sum(w) = target_sum }.
    """
    if target_sum <= 0:
        raise ValueError("target_sum must be positive")
    # Scale to unit‑sum, use the existing projector, then scale back.
    w = simplex_projection(v / target_sum)
    return w * target_sum


def convolution_max(a: np.ndarray) -> float:
    """Maximum entry of the full (linear) self‑convolution of a."""
    conv = np.convolve(a, a)
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """
    Exact score that `evaluate_sequence` would return.
    The vector must be non‑negative; no clipping is required here.
    """
    n = a.shape[0]
    max_c = convolution_max(a)
    s = a.sum()
    if s < 1e-8:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def build_convolution_matrix(f: np.ndarray) -> "scipy.sparse.csr_matrix":
    """
    Build the (2·n‑1) × n matrix A such that A @ g == f * g
    (full linear convolution).  The matrix is extremely sparse
    (each row contains at most n non‑zero entries) and is created
    with a diagonal (Toeplitz) representation.
    """
    import scipy.sparse as sp

    n = f.shape[0]
    # Offsets 0 … n‑1 correspond to the sub‑diagonals of the Toeplitz matrix.
    offsets = np.arange(n, dtype=np.int32)

    # data[i, j] = f[i]  (i = offset, j runs over the n columns)
    data = np.repeat(f[:, np.newaxis], n, axis=1)  # shape (n, n)

    # sp.diags expects a sequence of arrays, one per offset.
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A

                    best_f = cand.copy()
                    last_improve = it
                    print(f"Sub‑grad iter {it:5d}: new best = {best_score:.6f}")

            temperature *= cooling_rate
            step_size *= cooling_rate

            # occasional jitter to escape plateaus
            if it - last_improve > 5000:
                bump = np.random.rand(ncur) * 1e-4
                current = current + bump
                if current.sum() > 0.0:
                    current = current / current.sum() * target_sum
                else:
                    current = np.full_like(current, target_sum / ncur)
                last_improve = it
                temperature = 1.0

    # ------------------------------------------------------------------
    # 4. Return the best vector discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score = {best_score:.6f}   (length {best_f.shape[0]})")
    # The external evaluator will clip/normalise again, so we return the raw values.
    return best_f.tolist()
```
