def run_one_restart(p0, max_iters, deadline):
    p = p0.copy()
    best_score_local = compute_obj(p)
    best_p_local = p.copy()
    # initialize Adam moments zero
    mom1 = np.zeros_like(p)
    mom2 = np.zeros_like(p)
    alpha = 0.5
    for iteration in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute conv
        p_fft = np.fft.rfft(p, n=m_fft)
        conv_self = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2*n-1]
        # smoothing
        z = alpha * conv_self
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        w = exp_z / np.sum(exp_z)
        # compute gradient
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
        corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[:3*n-2]
        idx = 2*n-2 - np.arange(n)
        grad = 2.0 * corr_full[idx]
        # Adam update
        mom1 = beta1 * mom1 + (1 - beta1) * grad
        mom2 = beta2 * mom2 + (1 - beta2) * (grad**2)
        m_hat = mom1 / (1 - beta1**iteration)
        v_hat = mom2 / (1 - beta2**iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step
        # projection onto simplex
        p = _simplex_projection(p)
        # optionally jitter
        if iteration % 400 == 0:
            noise = rng.normal(scale=0.001, size=n)
            p = _simplex_projection(p + noise)
        # increase alpha
        alpha = min(alpha * alpha_mul, alpha_max)
        # evaluate objective
        p_fft = np.fft.rfft(p, n=m_fft)
        conv_self = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2*n-1]
        cur_score = 2.0 * n * np.max(conv_self)
        if cur_score < best_score_local:
            best_score_local = cur_score
            best_p_local = p.copy()
        # early exit maybe
        if cur_score <= target: break

    return best_p_local, best_score_local

"""
Propose a low‑objective step‑function.

The objective is                2·n·max(conv(a,a)) / (sum(a)²).
Because the expression is scale‑invariant we keep the vector on the
probability simplex (sum = 1, a_i ≥ 0).  The algorithm is a multi‑restart
projected Adam optimiser that minimises a smooth log‑sum‑exp approximation
of the maximum convolution entry.  After each Adam run a cheap local
pairwise‑swap refinement is performed.  Random restarts use Dirichlet
samples with a wide range of concentration parameters to explore both
dense (almost uniform) and sparse shapes.  The best vector found before the
time budget expires is returned.

The code is fully self‑contained – the only external function used by the
tester is `evaluate_sequence`, which is called only for a final sanity
check (the internal objective matches it exactly).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x | x_i ≥ 0 , Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core objective (scale‑invariant, works on the simplex)
# ----------------------------------------------------------------------
def _objective(p: np.ndarray, m_fft: int, n: int) -> float:
    """Return 2·n·max(conv(p,p)).  p is assumed to sum to 1."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))

              m_fft: int,
              n: int,
              deadline: float,
              target: float = 1.503) -> tuple[np.ndarray, float]:

    Perform a short Adam optimisation starting from p0.
    Returns (best_p, best_score) observed during the run.
    """

    lr = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # smoothing for the log‑sum‑exp (soft‑max) approximation of the max
    alpha = 0.8
    alpha_mul = 1.005
    alpha_max = 250.0

    # jitter to avoid stagnation
    jitter_interval = 400

    p = p0.copy()
    mom1 = np.zeros_like(p)
    mom2 = np.zeros_like(p)

    best_p = p.copy()
    best_score = _objective(p, m_fft, n)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        p_fft = np.fft.rfft(p, n=m_fft)

        # ----- self‑convolution (full length 2n‑1) -----

        z = alpha * conv_self
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)                # stable soft‑max numerator
        w = exp_z / np.sum(exp_z)                # soft‑max weights (sum = 1)

        # ----- gradient of the smoothed max (see paper) -----
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
        corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
        idx = 2 * n - 2 - np.arange(n)            # mapping to gradient entries
        grad = 2.0 * corr_full[idx]

        # ----- Adam update -----
        mom1 = beta1 * mom1 + (1.0 - beta1) * grad
        mom2 = beta2 * mom2 + (1.0 - beta2) * (grad ** 2)

        m_hat = mom1 / (1.0 - beta1 ** iteration)
        v_hat = mom2 / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step
        p = _simplex_projection(p)

        # occasional jitter
        if iteration % jitter_interval == 0:
            noise = np.random.default_rng().normal(scale=0.001, size=n)
            p = _simplex_projection(p + noise)

        # tighten the soft‑max
        alpha = min(alpha * alpha_mul, alpha_max)

        # ----- evaluate current exact objective -----
        cur_score = _objective(p, m_fft, n)

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            # print progress (optional – harmless)
            print(f"Iter {iteration:6d} – new best (Adam run): {best_score:.6f}")

        # early exit if we have already reached the target
        if best_score <= target:
            break

    return best_p, best_score


# ----------------------------------------------------------------------
# Simple pairwise‑swap local refinement
# ----------------------------------------------------------------------
def _pairwise_refine(p: np.ndarray,
                     m_fft: int,
                     n: int,
                     deadline: float,
                     cur_score: float,
                     max_attempts: int = 3000) -> tuple[np.ndarray, float]:
    """Randomly move ε mass from i→j; keep move if it improves the score."""
    rng = np.random.default_rng()
    best_p = p.copy()
    best_val = cur_score

    attempts = 0
    while attempts < max_attempts and time.time() < deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best_p[i] <= 1e-12:
            continue
        # propose a modest mass transfer
        eps = best_p[i] * rng.uniform(0.0005, 0.02)
        new_p = best_p.copy()
        new_p[i] -= eps
        new_p[j] += eps

        new_val = _objective(new_p, m_fft, n)
        if new_val < best_val:
            best_p = new_p
            best_val = new_val
            # optional progress message
            print(f"Swap improvement → {best_val:.6f}")

    return best_p, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The routine combines many Adam restarts (different Dirichlet
    initialisations) with a cheap pairwise‑swap refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            init_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name‑defined]
            init_seq = np.clip(init_seq, 0.0, 1000.0)
            if init_seq.sum() > 0.0:
                init_seq = init_seq / init_seq.sum()
            else:
                init_seq = None
        except Exception:
            init_seq = None
    else:
        init_seq = None

    # fallback size (same as the benchmark)
    n = 1153 if init_seq is None else init_seq.shape[0]

    # FFT size large enough for the largest convolution needed (3n‑2)
    m_fft = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Initialise the best known solution (uniform or provided)
    # ------------------------------------------------------------------
    if init_seq is not None:
        best_p = init_seq.copy()
    else:
        best_p = rng.random(n)
        best_p = best_p / best_p.sum()

    best_score = _objective(best_p, m_fft, n)
    print(f"Initial global best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin
    target_score = 1.503                     # target mentioned in the statement

    # ------------------------------------------------------------------
    # Main multi‑restart loop
    # ------------------------------------------------------------------
    restart_count = 0
    while time.time() < deadline:
        restart_count += 1

        # ---- random Dirichlet initialisation with random concentration ----
        conc = rng.uniform(0.05, 5.0)        # small -> sparse, large -> dense
        dirichlet_sample = rng.gamma(conc, 1.0, size=n)
        p0 = dirichlet_sample
        p0 = p0 / p0.sum()

        # Occasionally start from a jittered version of the current global best
        if restart_count % 7 == 0:
            jitter = rng.normal(scale=0.02, size=n)
            p0 = _simplex_projection(best_p + jitter)

        # Run a short Adam optimiser (time‑bounded)
        sub_deadline = min(deadline, time.time() + 4.0)   # at most a few seconds per restart
        p_opt, score_opt = _adam_run(p0, m_fft, n, sub_deadline, target=target_score)

        # Local pairwise refinement (quick, improves a few more points)
        if time.time() < deadline:
            p_opt, score_opt = _pairwise_refine(p_opt, m_fft, n,
                                                deadline, score_opt,
                                                max_attempts=1500)

        # Update global best if improvement found
        if score_opt < best_score:
            best_score = score_opt
            best_p = p_opt.copy()
            print(f"*** Global best improved to {best_score:.6f} after {restart_count} restarts ***")

        # Early termination if we have already reached the target
        if best_score <= target_score:
            print("Target score reached – terminating early.")
            break

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (optional but cheap)
    # ------------------------------------------------------------------
    try:
        # evaluate_sequence may perform extra clipping – our vector already respects limits
        from __main__ import evaluate_sequence  # will be present in the test harness
        check = evaluate_sequence(best_p.tolist())
        print(f"Evaluator sanity check: {check:.6f}")
        # If the evaluator returned inf something went wrong – fall back to uniform
        if not np.isfinite(check):
            raise ValueError("Invalid sequence according to evaluate_sequence.")
    except Exception:
        # safety fallback to a uniform distribution (never fails)
        best_p = np.full(n, 1.0 / n)
        best_score = _objective(best_p, m_fft, n)
        print("Fallback to uniform distribution due to evaluation error.")

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits imposed by evaluate_sequence)
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Final best score reported: {best_score:.6f}")
    print(f"Vector length: {len(result)}")
    return result
