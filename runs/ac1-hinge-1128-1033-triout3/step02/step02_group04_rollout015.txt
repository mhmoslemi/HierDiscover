def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 2000 # default size
    # exponent grid
    alphas = np.linspace(0.5, 6.0, 56)  # inclusive steps of 0.1
    best_score = float('inf')
    best_seq = None
    start = time.time()
    for alpha in alphas:
        # generate decreasing shape
        i = np.arange(n, dtype=float)
        t = i / (n - 1)
        seq = (1 - t) ** alpha
        total = seq.sum()
        if total == 0:
            continue
        seq /= total
        # evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"alpha {alpha:.2f} -> score {score:.6f}")
        # early exit if time near budget
        if time.time() - start > budget_s - 1:  # keep margin
            break
    # if none found, fallback to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
    return best_seq.tolist()
