def exponentiated_update(p, grad, eta):
    # p_i = p_i * exp(-eta * grad_i)
    # to avoid under/overflow, we can subtract max(eta * grad) before exponentiation.
    update = -eta * grad
    max_update = np.max(update)
    exp_update = np.exp(update - max_update)  # shift for numerical stability
    p = p * exp_update
    total = p.sum()
    if total > 0:
        p /= total
    else:
        p = np.full_like(p, 1.0/len(p))
    return p

    # initialize best

    deadline = time.time() + budget_s - 1.0

    # define evaluation function
    def evaluate(p):

        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2*n-1]

"""
Propose a candidate step‑function using exponentiated‑gradient descent on a
smooth log‑sum‑exp approximation of the maximum of the self‑convolution.
The algorithm runs several random restarts, gradually tightens the smooth
approximation, injects occasional jitter, and finishes with a simple
pairwise hill‑climbing and a short Adam refinement.  The returned sequence
is normalised to unit ℓ₁‑norm (the objective is scale‑invariant) and clipped
to the range required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex {x | x_i ≥ 0 , Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Compute the self‑convolution via FFT (full length 2n‑1)
# ----------------------------------------------------------------------
def _self_convolution(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Return p * p (full convolution, length 2·len(p)‑1)."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * len(p) - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max (log‑sum‑exp) of the self‑convolution.
# ----------------------------------------------------------------------
def _grad_logmax(p: np.ndarray, alpha: float, m_fft: int) -> np.ndarray:
    """
    Compute ∇_p  (1/α)·log Σ_k exp(α·(p*p)[k]).
    The gradient has the closed form 2·corr(p, w_rev) where w are the
    soft‑max weights derived from the smoothed max.
    """
    n = len(p)
    # FFT of p
    p_fft = np.fft.rfft(p, n=m_fft)

    # self‑convolution (full length 2n‑1)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

    # log‑sum‑exp weights
    z = alpha * conv
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)                # stability trick
    w = exp_z / np.sum(exp_z)                # soft‑max, Σ w = 1

    # correlation of p with reversed w (needed for gradient)
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)

    # full correlation (length 3n‑2)
    corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]

    # mapping to gradient components (see original source)
    idx = 2 * n - 2 - np.arange(n)           # maps 0…n‑1 ↦ proper indices
    grad = 2.0 * corr_full[idx]
    return grad

# ----------------------------------------------------------------------

    """Return 2·n·max(p*p)  (the value returned by ``evaluate_sequence`` for Σp=1)."""
    n = len(p)
    conv = _self_convolution(p, m_fft)
    return 2.0 * n * np.max(conv)
