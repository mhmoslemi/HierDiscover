"""
A simulated‑annealing / biased‑move optimiser for the
self‑convolution max‑norm problem.

The objective for a probability vector `p` (∑p = 1, p ≥ 0) is

    score(p) = 2·n·max_k (p * p)[k]               (1)

because `evaluate_sequence` is homogeneous of degree 0.
The optimiser works on the simplex, constantly updating the
full convolution by an O(n) incremental formula when a tiny
amount of mass is transferred from one coordinate to another.
Moves are biased toward indices that contribute most to the
currently maximal convolution coefficient, and a classic
temperature schedule (simulated annealing) allows uphill steps
early on.  Periodic full recomputation of the convolution and
small random jitters keep numerical errors in check.  After the
annealing phase a short greedy swap‑refinement is performed.

The returned list is a probability vector (sum=1);  `evaluate_sequence`
clips it to the allowed range, so the scaling does not affect the
objective.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex {x | x_i ≥ 0 , Σ x_i = 1}.
    Implementation of Duchi et al., ICML 2008.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Incremental convolution update utilities.
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full discrete convolution a * a (returns length 2·n‑1)."""
    return np.convolve(a, a)


def _increment_conv(conv: np.ndarray, a: np.ndarray,
                    i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving `delta` mass from index i
    to index j (i != j).  The update follows the exact algebraic formula:

        new_conv = conv
                   - 2·delta·shift_i(a)
                   + 2·delta·shift_j(a)
                   + delta²·e_{2i} + delta²·e_{2j}
                   - 2·delta²·e_{i+j}
    where shift_k(a) is `a` placed starting at position k.
    """
    n = a.shape[0]
    new = conv.copy()

    # linear part: +2·delta·(shift_j - shift_i)
    new[i:i + n] -= 2.0 * delta * a
    new[j:j + n] += 2.0 * delta * a

    # quadratic part
    new[2 * i] += delta * delta
    new[2 * j] += delta * delta
    new[i + j] -= 2.0 * delta * delta

    return new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative list) that minimises the
    evaluation function.  The routine works on the probability simplex,
    using a temperature‑controlled mass‑transfer Markov chain and a
    final greedy refinement.  It returns the best vector found before
    the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial vector (use the provided construction if
    # available, otherwise start from a random probability vector).
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name‑defined]
        have_initial = True
    except Exception:
        n = 1153
        have_initial = False

    if have_initial:
        p = np.array(height_sequence_1, dtype=float)
        p = np.clip(p, 0.0, 1000.0)
        s = p.sum()
        if s > 0.0:
            p = p / s
        else:
            p = np.full(n, 1.0 / n)
    else:
        p = rng.random(n)
        p = p / p.sum()

    # initial convolution and score
    conv = _full_convolution(p)
    cur_score = 2.0 * n * np.max(conv)
    best_score = cur_score
    best_p = p.copy()
    print(f"[init] n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    step_factor = 0.12          # maximum fraction of source mass moved
    T0 = 0.010                  # initial temperature
    Tmin = 1e-7
    decay = 0.99994             # exponential decay per iteration
    T = T0

    # restart logic
    last_improve_iter = 0
    restart_interval = 300000   # if no improvement for this many iters

    # recompute conv from scratch every this many iterations (numerical drift)
    recompute_every = 4000

    deadline = time.time() + budget_s - 2.0   # safety margin
    iteration = 0

    # ------------------------------------------------------------------
    # Main annealing loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) bias the source index towards coordinates that participate
        #    heavily in the current maximal convolution coefficient.
        # --------------------------------------------------------------
        k_max = int(np.argmax(conv))                     # index of max entry
        partners = k_max - np.arange(n)                  # partner index for each i
        mask = (partners >= 0) & (partners < n)

        if mask.any():
            idx = np.where(mask)[0]
            weights = p[idx] * p[partners[idx]]
            wsum = weights.sum()
            if wsum > 0.0:
                probs = weights / wsum
                i = rng.choice(idx, p=probs)
            else:
                i = rng.choice(idx)
        else:
            i = rng.integers(0, n)

        # ensure the source has some mass
        if p[i] < 1e-12:
            continue

        # pick a distinct target index
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # --------------------------------------------------------------
        # 2) propose a transfer of a small amount of mass
        # --------------------------------------------------------------
        delta = step_factor * p[i] * rng.random()
        if delta <= 0.0:
            continue

        # incremental update of convolution
        new_conv = _increment_conv(conv, p, i, j, delta)
        new_score = 2.0 * n * np.max(new_conv)

        # --------------------------------------------------------------
        # 3) accept / reject (Metropolis criterion)
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability to accept an uphill move
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # apply the move
            p[i] -= delta
            p[j] += delta
            conv = new_conv
            cur_score = new_score

            # record the best ever seen
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                last_improve_iter = iteration
                print(f"[{iteration:7d}] new best {best_score:.6f}")

        # --------------------------------------------------------------
        # 4) temperature decay
        # --------------------------------------------------------------
        T = max(Tmin, T * decay)

        # --------------------------------------------------------------
        # 5) occasional full recomputation (numerical stability)
        # --------------------------------------------------------------
        if iteration % recompute_every == 0:
            conv = _full_convolution(p)
            cur_score = 2.0 * n * np.max(conv)

        # --------------------------------------------------------------
        # 6) tiny jitter to shake the system (helps escape flat plateaus)
        # --------------------------------------------------------------
        if iteration % 2000 == 0:
            jitter = rng.normal(scale=5e-4, size=n)
            p = p + jitter
            p = np.clip(p, 0.0, None)
            p = _simplex_projection(p)
            conv = _full_convolution(p)
            cur_score = 2.0 * n * np.max(conv)

        # --------------------------------------------------------------
        # 7) restart if we have stagnated for a long time
        # --------------------------------------------------------------
        if iteration - last_improve_iter > restart_interval:
            # perturb the best solution fairly strongly and continue
            p = best_p + rng.normal(scale=0.025, size=n)
            p = np.clip(p, 0.0, None)
            p = _simplex_projection(p)
            conv = _full_convolution(p)
            cur_score = 2.0 * n * np.max(conv)
            T = T0
            last_improve_iter = iteration
            print(f"[{iteration:7d}] restart (stagnation) – score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy refinement – try tiny pairwise transfers.
    # ------------------------------------------------------------------
    p = best_p.copy()
    conv = _full_convolution(p)
    cur_score = 2.0 * n * np.max(conv)

    swap_attempts = 3000
    for _ in range(swap_attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j or p[i] <= 1e-12:
            continue
        eps = 1e-5 * rng.random()
        # propose moving eps from i to j
        new_conv = _increment_conv(conv, p, i, j, eps)
        new_score = 2.0 * n * np.max(new_conv)
        if new_score < cur_score:
            p[i] -= eps
            p[j] += eps
            conv = new_conv
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
                print(f"[swap] improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector found (already a probability vector;
    # evaluate_sequence will clip it to the required range).
    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return np.clip(best_p, 0.0, 1000.0).tolist()
