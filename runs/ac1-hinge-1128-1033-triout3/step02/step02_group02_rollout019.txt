"""
Propose a step‑function that minimises the evaluation metric.

The algorithm consists of three stages:

1. **Broad parametric sweep** –  Several simple families (U‑shaped power
   law, double‑tail power law, exponential decay) are evaluated for a
   range of lengths and exponents.  The best candidate found by the fast
   FFT based convolution estimator is kept as a warm start.

2. **Deterministic projected gradient descent on the simplex** –  Starting
   from the warm start we perform a mirror‑descent/gradient‑ascent loop.
   At each iteration the indices where the self‑convolution attains its
   current maximum are identified (a small active set).  The sub‑gradient
   of the maximum entry is computed analytically and an exponentiated
   update (`a_i ← a_i·exp(-η·g_i)`) is applied.  Afterwards the vector is
   projected back onto the probability simplex (non‑negative and
   sum = 1).  The step size η is reduced geometrically.

3. **Stochastic refinement** –  To escape possible local minima a
   simulated‑annealing loop is run.  Two move types are used: a random
   mass transfer between two positions and a full mirror‑descent step
   (as above).  Moves are accepted if they improve the exact score
   (computed with ``evaluate_sequence``) or with a temperature‑dependent
   probability.

The whole search respects the given time budget (default 1000 s).  The
best sequence found is returned as a plain list of Python ``float``s.
"""

import time
import numpy as np

# ---------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ---------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{-p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Two power‑law tails: a[i] ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).  Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("input must be a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_of_max_conv(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of the L∞‑norm of a*a.
    Only the entries that belong to the (approximately) active set
    (convolution values within ``eps`` of the maximum) are considered.
    """
    n = a.shape[0]
    # linear convolution via FFT (fast, exact for the used size)
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    max_val = conv.max()
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        # derivative of a_i*a_{k-i} + a_{k-i}*a_i = 2*a_{k-i}
        grad[idx_i] += 2.0 * a[idx_j]
    return grad


# ---------------------------------------------------------------
# Main search routine
# ---------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin

    best_seq = None
    best_score = np.inf

    # -----------------------------------------------------------
    # 0.   Use the external construction if it exists.
    # -----------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external = external / s
            sc = 2.0 * len(external) * max_conv_fft(external)
            if sc < best_score:
                best_score = sc
                best_seq = external.copy()
                print(f"Seeded from height_sequence_1: score={sc:.6f}")
    except NameError:
        pass

    # -----------------------------------------------------------
    # 1.   Deterministic parametric sweep (fast, FFT based)
    # -----------------------------------------------------------
    lengths = [4000, 6000, 8000, 10000, 12000, 16000, 20000]
    p_vals = np.linspace(0.2, 2.0, 19)          # 0.2, 0.3, …, 2.0
    alphas = np.linspace(0.0002, 0.008, 13)    # exponential decay rates

    print("\n=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # power‑law U‑shape
        for p in p_vals:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=U‑pow, p={p:4.2f}, score={sc:.6f}")

        # double‑tail power law
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=tails, p={p:4.2f}, score={sc:.6f}")

        # exponential decay
        for a_alpha in alphas:
            if time.time() > deadline:
                break
            a = exponential_distribution(n, a_alpha)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=exp, α={a_alpha:6.4f}, score={sc:.6f}")

    # -----------------------------------------------------------
    # 2.   If nothing was found (unlikely), fall back to a random start.
    # -----------------------------------------------------------
    if best_seq is None:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = 2.0 * n * max_conv_fft(a)
        print(f"Fallback: random init, n={n}, score={best_score:.6f}")

    # -----------------------------------------------------------
    # 3.   Deterministic mirror‑descent on the simplex.
    # -----------------------------------------------------------
    current = best_seq.copy()
    n = len(current)

    eta0 = 0.25                 # initial mirror‑descent step size
    eta_decay = 0.985           # geometric decay per iteration
    proj_every = 10             # projection frequency
    it = 0
    print("\n=== Deterministic mirror‑descent ===")
    while time.time() < deadline:
        it += 1
        grad = subgradient_of_max_conv(current, eps=1e-12)
        eta = eta0 * (eta_decay ** it)
        # multiplicative (exponential) update – mirror descent for the simplex
        cand = current * np.exp(-eta * grad)
        cand = project_onto_simplex(cand)

        # fast score estimate (FFT based)
        cand_score = 2.0 * n * max_conv_fft(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            current = cand
            print(f"Iter {it:5d}: deterministic improvement → score={cand_score:.6f}")

        # stop when step size becomes tiny
        if eta * grad.max() < 1e-12:
            break

    # -----------------------------------------------------------
    # 4.   Stochastic refinement with simulated annealing.
    # -----------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score
    temperature = 0.6
    cooling = 0.9997
    transfer_frac = 0.45
    it = 0
    print("\n=== Stochastic refinement (SA) ===")
    while time.time() < deadline:
        it += 1

        # Choose move type (70 % mass‑transfer, 30 % mirror step)
        if np.random.rand() < 0.7:
            # ---- Random mass transfer ---------------------------------
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] <= 0.0:
                continue
            delta = current[i] * np.random.rand() * transfer_frac
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = cand / cand.sum()
        else:
            # ---- Full mirror‑descent step (sub‑gradient of max) -------
            grad = subgradient_of_max_conv(current, eps=1e-12)
            step = temperature * 0.2
            cand = current * np.exp(-step * grad)
            cand = project_onto_simplex(cand)

        except Exception:
            # fallback to the fast FFT estimate if something goes wrong
            cand_score = 2.0 * n * max_conv_fft(cand)

        # Acceptance test
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Iter {it:7d}: new global best = {best_score:.6f}")

        # Cool down

        # Early exit if we beat the (unrealistic) target – helps the
        # benchmark harness when it is used.
        if best_score < 1.503:
            print("Target reached – exiting early.")
            break

    # -----------------------------------------------------------
    # 5.   Final exact evaluation and report
    # -----------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
