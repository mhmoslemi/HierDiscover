"""
propose_candidate – Adam‑based smooth‑max optimisation

The evaluator is homogeneous: 2·n·max(a∗a) for a non‑negative vector with sum=1.
Hence we simply minimise the maximal entry of the self‑convolution.

The algorithm works in three phases:

1️⃣  Deterministic sweep over a few one‑parameter families (U‑shapes
    based on distance to the edge, on product of distances, and a
    symmetric exponential).  The best candidate of this cheap search
    is taken as a seed.

2️⃣  Smooth‑max gradient descent with Adam.  The objective
    `fα(a)=log‑sum‑exp(α·(a∗a))` approximates `max(a∗a)`.  Its
    gradient is a convolution that can be evaluated with two FFTs
    per iteration.  After each Adam step we project back onto the
    probability simplex.

3️⃣  Simple stagnation handling – if no improvement is seen for a
    while we inject a tiny random bump and re‑heat the “temperature”
    (the soft‑max concentration α is increased gradually).

All work is done inside the given time budget (≈1000 s).  The best
vector found (normalised to sum = 1, non‑negative) is returned as a
list of floats.
"""

import time

# ----------------------------------------------------------------------

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w≥0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of power‑of‑2 size."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n]


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for self‑convolution a∗a."""
    return convolve_fft(a, a)


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i,n‑1‑i)+1)^{-p}  (U‑shape based on distance to the edge)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n-i))^{-p}  (U‑shape based on product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam·i) + exp(-lam·(n‑1‑i))  (symmetric exponential decay)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def internal_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a∗a) for a normalised (sum=1) vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return 2.0 * n * conv.max()


def smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Returns (grad, max_conv) where grad is the gradient of
    fα(a)=log‑sum‑exp(α·(a∗a)) and max_conv is max(a∗a).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)                # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (stable)
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(conv) / conv.size
    else:
        w /= w_sum

    # grad_i = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]
    grad_full = convolve_fft(w, a_rev)            # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]     # slice of length n
    return grad, max_c

        ext = np.array(height_sequence_1, dtype=float)
        ext = np.maximum(ext, 0.0)
        if ext.sum() > 0:
            ext /= ext.sum()
            sc_ext = internal_score(ext)
            if sc_ext < best_score:
                best_score, best_seq = sc_ext, ext.copy()
                print(f" using external height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3.  Fallback – random simplex point
    # --------------------------------------------------------------
    if best_seq is None:
        n = 8000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = internal_score(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Adam‑based smooth‑max optimisation
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.03                     # base learning rate
    iteration = 0
    last_improve = 0
    global_best_a = a.copy()
    global_best_score = best_score

    # schedule for the soft‑max concentration α
    alpha_min, alpha_max = 1.0, 30.0
    alpha_ramp_iters = 40000      # after this α stays at alpha_max

    print("\n=== Adam optimisation (smooth‑max) ===")
    while time.time() < deadline:
        iteration += 1

        # current concentration parameter
        if iteration < alpha_ramp_iters:
            alpha = alpha_min + (alpha_max - alpha_min) * iteration / alpha_ramp_iters
        else:
            alpha = alpha_max

        grad, max_c = smooth_max_gradient(a, alpha)

        # Adam update (projected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = simplex_projection(a_candidate)

        cand_score = internal_score(a_candidate)

        # Accept the new point (always) – Adam already performs a descent‑like step.
        a = a_candidate

        # Track the best ever seen
        if cand_score < global_best_score:
            global_best_score = cand_score
            global_best_a = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:7d}: new global best = {global_best_score:.6f}")

        # Stagnation handling – add a tiny random bump every few ten‑thousands iterations
        if iteration - last_improve > 50000:
            bump = np.random.rand(n) * 1e-4
            a = simplex_projection(a + bump)
            # reset Adam moments to avoid accumulated gradients blowing up
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat (random bump)")

        # occasional sanity check to avoid drifting due to numerical issues
        if iteration % 20000 == 0:
            # re‑normalise explicitly (project already does, but this is cheap)
            a = simplex_projection(a)

    # --------------------------------------------------------------
    # 5.  Return the best sequence discovered
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score = {global_best_score:.6f}   (length {len(global_best_a)})")
    # external evaluator will clip and normalise again, so we return the raw vector
    return global_best_a.tolist()
