# Helper: Euclidean projection onto the simplex {x≥0, Σx = 1}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        grad = 2.0 * corr_full[idx]
        # enforce symmetry on the gradient
        grad = (grad + grad[::-1]) / 2.0

        # ----- Adam update -----
        mom1 = beta1 * mom1 + (1.0 - beta1) * grad
        mom2 = beta2 * mom2 + (1.0 - beta2) * (grad ** 2)

        m_hat = mom1 / (1.0 - beta1 ** iteration)
        v_hat = mom2 / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step

        # ----- projection onto the scaled simplex & symmetry -----
        p = S * _simplex_projection(p / S)
        p = (p + p[::-1]) / 2.0
        p = S * _simplex_projection(p / S)

        # ----- occasional jitter to escape flat regions -----
        if iteration % jitter_interval == 0:
            noise = rng.normal(scale=0.002 * S, size=n)
            p = p + noise
            p = S * _simplex_projection(p / S)
            p = (p + p[::-1]) / 2.0
            p = S * _simplex_projection(p / S)

        # ----- tighten the smooth‑max approximation -----
        alpha = min(alpha * alpha_mul, alpha_max)

        # ----- evaluate current objective (max of the true convolution) -----
        p_fft = np.fft.rfft(p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
        cur_score = np.max(conv)

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            best_improve_iter = iteration
            print(f"Iter {iteration:6d} – new best: {best_score:.6f}")

            if best_score <= 1.503:
                # target reached – we can stop early
                break

        # ----- restart if stagnation persists -----
        if iteration - best_improve_iter > restart_interval:
            # perturb the current best and restart Adam
            p = best_p + rng.normal(scale=0.05 * S, size=n)
            p = np.clip(p, 0.0, None)
            p = S * _simplex_projection(p / S)
            p = (p + p[::-1]) / 2.0
            p = S * _simplex_projection(p / S)
            mom1.fill(0.0)
            mom2.fill(0.0)
            lr = max(lr * 0.7, 1e-4)
            best_improve_iter = iteration
            print(f"Iter {iteration:6d} – restart (lr now {lr:.5f})")

    # --------------------------------------------------------------
    # 6. Final greedy refinement – pairwise mass transfers
    # --------------------------------------------------------------
    p = best_p.copy()
    cur_score = best_score
    max_swaps = 5000

    for _ in range(max_swaps):
        if time.time() >= deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Move a tiny amount from i to j (and keep symmetry)
        eps_move = 1e-4 * S * rng.random()
        if p[i] < eps_move:
            continue

        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move

        # Mirror the moves to preserve exact symmetry
        mi = n - 1 - i
        mj = n - 1 - j
        if mi != i:
            if new_p[mi] < eps_move:
                continue
            new_p[mi] -= eps_move
            new_p[mj] += eps_move

        new_p = S * _simplex_projection(new_p / S)

        # Compute objective for the candidate
        p_fft = np.fft.rfft(new_p, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
        new_score = np.max(conv)

        if new_score < cur_score - 1e-7:
            p = new_p
            cur_score = new_score
            best_p = p.copy()
            best_score = cur_score
            print(f"Swap improvement → {best_score:.6f}")

    print(f"Final best objective (max‑conv): {best_score:.6f}")

    # Clip to the limits required by evaluate_sequence and return
    return np.clip(best_p, 0.0, 1000.0).tolist()
