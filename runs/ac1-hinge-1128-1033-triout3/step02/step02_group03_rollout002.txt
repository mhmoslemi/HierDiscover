"""
Propose a non‑negative step‑function (probability vector) that minimises

        M = 2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages, all fitting in the 1000 s budget.

1. **Adam on a smooth surrogate** – we minimise a smooth approximation of the
   max‑operator:
        fα(p) = (1/α)·log Σ exp(α·(p∗p))  +  λ·Var(p∗p) .
   The gradient of the first term is a soft‑max weighted convolution,
   the second term is the variance of the autocorrelation.
   Updates are performed with the Adam optimiser and the iterate is projected
   onto the probability simplex.

2. **Exact max sub‑gradient with exponentiated updates** – once α is large
   (the surrogate is close to the true max) we switch to a multiplicative
   mirror‑descent step using the true sub‑gradient of max(p∗p).  This respects
   non‑negativity automatically and the vector is renormalised after each step.

3. **Simulated‑annealing polishing** – with the remaining time we perform cheap
   local moves (transfer a tiny amount of mass between two coordinates) and
   accept them according to a decreasing temperature schedule.  Improvements are
   kept and the best vector seen throughout the run is returned.

Fast convolutions are performed with the FFT (zero‑padded to the next power of 2)
so each iteration costs O(N log N) and many thousands of iterations fit easily
into the time limit.

The returned list already satisfies the validation performed by
``evaluate_sequence`` (non‑negative, ≤ 1000).  The sum of the vector is 1,
hence the objective evaluated by the grader equals the one minimised here.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, b: np.ndarray, N: int) -> np.ndarray:
    """
    Linear convolution of real vectors a and b using FFT of size N (power of two).
    Returns the full convolution of length len(a)+len(b)-1.
    """
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = A * B
    c = np.fft.irfft(C, N)
    # keep only the needed part (caller will slice as required)
    return c


def _objective(p: np.ndarray, N: int) -> float:
    """Exact objective 2·n·max(p∗p) for a probability vector p."""
    n = p.shape[0]
    conv = _conv_fft(p, p, N)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)          # numerical safety
    return 2.0 * n * float(conv.max())


def _gradient_from_weights(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of Σ w_k·(p∗p)_k with respect to p.
    Implements the formula grad = 2·(p ⋆ w_rev) restricted to the central n entries.
    """
    n = p.shape[0]
    w_rev = w[::-1]
    corr = _conv_fft(p, w_rev, N)[: 3 * n - 2]       # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)                  # indices of the central part
    grad = 2.0 * corr[idx]
    return grad

    # 7. Final random sampling (cheap Dirichlet draws) while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline:
        p_rand = rng.random(n)
        p_rand = p_rand / p_rand.sum()
        score_rand = _objective(p_rand, NFFT)
        if score_rand < best_score:
            best_score = score_rand
            best_p = p_rand.copy()
            print(f"[Random] new best {best_score:.6f}")
        if time.time() + 0.01 > deadline:
            break

    # ------------------------------------------------------------------
    # 8. Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Best score achieved: {best_score:.6f}")
    return result
