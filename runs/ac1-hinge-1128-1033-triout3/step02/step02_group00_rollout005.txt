Implementation details for _full_convolution_fft(x, y) function:

def _full_convolution_fft(x, y):
    n = x.shape[0]
    m = y.shape[0]
    total_len = n + m - 1
    size = 1 << (total_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total_len]
    np.maximum(conv, 0, out=conv)
    return conv

Smoothed sub‑gradient (soft‑max) descent with momentum.

The objective to minimise is
    F(a) = 2·n·max_k (a*a)[k]      with   a_i ≥ 0 , Σ a_i = 1 .
For a fixed length n we keep a on the probability simplex.
At each iteration we compute the full auto‑convolution a*a, locate its
maximum M and build a soft‑max weight vector w over the whole
convolution:
        w_k ∝ exp(( (a*a)[k] – M) / τ) .
When τ → 0 the weights concentrate on the active index and we obtain the
usual sub‑gradient; for larger τ the gradient is a smooth average of many
active constraints, which helps to escape plateaux.

The gradient of  Σ_k w_k·(a*a)[k] is
        g_i = 2· Σ_k w_k·a_{k‑i}
which can be evaluated by a single FFT‑based convolution.
After a step a ← a – η·g we project back onto the simplex.
A momentum term (β≈0.9) and occasional random mass‑transfer improve
stability and exploration.

The routine runs for the whole time budget (default 1000 s) and returns
the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x >= 0, sum(x) = 1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient from soft‑max weights
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via a single convolution: reverse w, convolve with a,
    then pick the appropriate slice.
    """
    n = a.shape[0]
    # reverse the weight vector
    u = w[::-1]
    # full convolution u * a   -> length 3n‑2
    conv = _full_convolution_fft(u, a)
    # indices 2n‑2, 2n‑3, … , n‑1  (total n entries)
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g
