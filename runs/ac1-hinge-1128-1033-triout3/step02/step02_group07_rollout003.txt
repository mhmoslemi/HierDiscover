"""
A refined optimizer for the step‑function evaluation metric.

The algorithm consists of three stages:

1. **Warm‑start** – initialise from the analytically motivated
   square‑root series (coefficients of (1‑x)^(-1/2)) possibly with a tiny
   random perturbation.

2. **Momentum‑gradient descent** – the sub‑gradient of the maximal
   convolution entry is computed (active indices only).  A Nesterov‑type
   momentum term is accumulated and an additive update is performed.
   After each update the vector is projected onto the probability simplex
   (all entries non‑negative, sum = 1).  This phase runs for most of the
   allotted time and quickly drives the fast FFT‑based bound down.

3. **Deterministic mass‑transfer + annealed walk** – using the same
   gradient information we repeatedly move a small amount of mass from the
   index with the largest gradient to the index with the smallest.  The
   move is accepted if the fast score improves, otherwise with a
   temperature‑dependent Metropolis probability.  Periodic restarts
   prevent stagnation.  Whenever a new best fast score is encountered we
   call the expensive `evaluate_sequence` to record the true best.

The code respects the 1000 s budget, stays on a single CPU and
contains only top‑level helper functions (no closures or lambdas).  The
final sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex using the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “square‑root” start (coefficients of (1‑x)^(-1/2))
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the normalised first n coefficients of (1‑x)^(-1/2)."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max_k (a∗a)_k at the active indices `active`.
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k at the active indices `active`.
    For every active k we add 2*a_{k-i} to the i‑th component whenever
    0 ≤ i < n and 0 ≤ k-i < n.  The result is averaged over the active set.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    if active.size > 1:
        grad /= float(active.size)
    return grad


# ----------------------------------------------------------------------
# Fast FFT based evaluation: 2·n·max(a∗a) (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, size_fft: int) -> float:
    """Return 2·n·max(a∗a) using an FFT of length `size_fft`."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Global‑level helper to maintain the best (true) sequence found so far.
# ----------------------------------------------------------------------
def _maybe_update_best(best: list, candidate: np.ndarray, fast_val: float) -> None:
    """
    Update the holder `best` (list of [seq, score]) if `candidate` improves.
    The expensive `evaluate_sequence` is called only on a new global best.
    """
    if fast_val < best[1] - 1e-9:
        true_val = evaluate_sequence(candidate.tolist())
        if true_val < best[1]:
            best[1] = true_val
            best[0] = candidate.copy()
            print(f"[global] new best {best[1]:.6f}")


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that minimises the supplied
    `evaluate_sequence` function.

    The whole 1000 s budget is split 70 % for a momentum gradient descent
    and 30 % for a deterministic + annealed fine‑tuning phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    # ------------------------------------------------------------------
    # Determine problem size – use a power of two near the default length.
    # ------------------------------------------------------------------
    n = int(kwargs.get("length", 8000))
    # If the requested length is not a power of two, keep it (FFT will pad anyway)
    size_fft = 1 << ((2 * n - 1).bit_length())   # next power of two for convolution

    # ------------------------------------------------------------------
    # Initial vector – try to reuse a global sequence if it matches the size,
    # otherwise fall back to the analytically motivated square‑root series.
    # ------------------------------------------------------------------
    try:
        a0 = np.array(height_sequence_1, dtype=float)
        if a0.shape[0] != n:
            raise ValueError
    except Exception:
        a0 = _sqrt_series(n)

    # Slight random jitter helps to escape symmetric plateaus
    a = a0 + np.random.randn(n) * 1e-4
    a = np.maximum(a, 0.0)
    a = _project_simplex(a)                       # guarantee feasibility (sum = 1)

    # ------------------------------------------------------------------
    # Fast score of the starting point and initial global‑best update.
    # ------------------------------------------------------------------
    cur_fast = _fast_score(a, size_fft)

    best = [a.copy(), float("inf")]               # best = [seq, true_score]
    _maybe_update_best(best, a, cur_fast)
    print(f"[init] n={n}, start fast score {cur_fast:.6f}")

    # ==================================================================
    # Phase 1 – momentum based gradient descent (≈ 70 % of the budget)
    # ==================================================================
    grad_mom = np.zeros_like(a)
    beta = 0.9                                      # momentum coefficient
    it = 0
    phase1_end = start_time + (deadline - start_time) * 0.70

    while time.time() < phase1_end:
        it += 1
        # --- convolution and active set ---
        fa = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
        mx = conv.max()
        active = np.flatnonzero(conv >= mx - 1e-12)

        # --- sub‑gradient of the current max ---
        grad = _max_conv_gradient(a, active)

        # --- momentum update ---
        grad_mom = beta * grad_mom + (1.0 - beta) * grad

        # --- step size (decays like 1/sqrt(it)) ---
        step = 0.1 / (math.sqrt(it) + 1.0)

        # --- additive update, projection onto simplex ---
        a_new = a - step * grad_mom
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # --- fast evaluation and acceptance ---
        fast_new = _fast_score(a_new, size_fft)
        if fast_new <= cur_fast:
            a = a_new
            cur_fast = fast_new
            _maybe_update_best(best, a, cur_fast)
        else:
            # occasional stochastic acceptance to escape shallow plateaus
            if np.random.rand() < 1e-3:
                a = a_new
                cur_fast = fast_new

        # --- tiny jitter every few thousand iterations ---
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 5e-5
            a = _project_simplex(a + jitter)

    # ==================================================================
    # Phase 2 – deterministic mass transfer + simulated annealing (≈30 %)
    # ==================================================================
    # Temperature schedule: start modest and cool linearly to zero.
    T_start = 0.08
    no_improve = 0
    while time.time() < deadline:
        # recompute convolution and gradient for the current a
        fa = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
        mx = conv.max()
        active = np.flatnonzero(conv >= mx - 1e-12)
        grad = _max_conv_gradient(a, active)

        # indices with largest / smallest gradient magnitude
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        # propose moving a fraction of mass from i_max to i_min
        delta = min(a[i_max] * 0.5, 0.005)      # never move more than half of source
        if delta <= 1e-12:                     # fallback to a random tiny move
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = min(a[src] * 0.5, 1e-4)
            a_candidate = a.copy()
            a_candidate[src] -= delta
            a_candidate[dst] += delta
        else:
            a_candidate = a.copy()
            a_candidate[i_max] -= delta
            a_candidate[i_min] += delta

        fast_cand = _fast_score(a_candidate, size_fft)

        # linear cooling
        elapsed = time.time() - phase1_end
        total_sa = deadline - phase1_end
        frac = min(1.0, elapsed / total_sa)
        T = T_start * (1.0 - frac)

        # Metropolis acceptance
        if fast_cand <= cur_fast or np.random.rand() < math.exp(-(fast_cand - cur_fast) / max(T, 1e-12)):
            a = a_candidate
            cur_fast = fast_cand
            if fast_cand < best[1] - 1e-9:
                _maybe_update_best(best, a, cur_fast)
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # restart if stuck for a while
        if no_improve > 8000:
            a = _sqrt_series(n) + np.random.randn(n) * 1e-4
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            cur_fast = _fast_score(a, size_fft)
            no_improve = 0
            print("[restart] after stagnation")
            _maybe_update_best(best, a, cur_fast)

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain list of floats).
    # ------------------------------------------------------------------
    if best[0] is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n)
    else:
        best_seq = best[0]

    print(f"[final] best true evaluation = {best[1]:.6f}")
    return best_seq.tolist()
