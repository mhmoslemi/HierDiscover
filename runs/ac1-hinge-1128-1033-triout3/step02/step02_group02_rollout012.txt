Propose a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. **Warm‑start** – use the previously discovered sequence (`height_sequence_1`)

   families (sum‑of‑tails, exponential) for several lengths up to 20 000.
2. **Smooth‑max optimisation** – the true objective is
   `max(conv)`.  We replace the hard max by a soft‑max
   `log‑sum‑exp` with a temperature `β`.  The gradient of this smooth
   surrogate is a weighted sum of the gradients of the individual
   convolution entries, which can be computed efficiently by a single
   FFT‑based correlation.
3. **Mirror‑descent on the simplex** – we perform an exponentiated‑gradient
   (mirror‑descent) step
   `a_i ← a_i·exp(−η·g_i)`, where `g` is the smooth gradient, and renormalise
   to keep the vector on the probability simplex.  `β` is slowly increased
   (the surrogate becomes sharper) while the step size `η` is slowly
   decayed.
4. **Stochastic perturbations** – every few hundred iterations a tiny random
   mass‑transfer move is applied to avoid getting stuck.
5. **Tracking** – the best sequence (according to the fast FFT‑based score)
   is kept throughout the run.  At the end a single call to the official
   `evaluate_sequence` yields the exact score.

def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] = (min(i,n‑1‑i)+1)^{-p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] = (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a

    # 1.  Light parametric sweep (fast, works on the FFT bound only).
    # --------------------------------------------------------------
    lengths = [8000, 12000, 16000, 20000]
    p_vals = np.linspace(0.3, 1.5, 13)          # 0.3 … 1.5
    alphas = np.linspace(0.0005, 0.005, 10)    # exponential decay

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- parametric sweep length {n} ---")
        # sum‑of‑tails (usually best)
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (tails)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # exponential family (as a backup)
        for a_alpha in alphas:
            if time.time() > deadline:
                break
            a = exponential_distribution(n, a_alpha)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (exp)    n={n:5d} α={a_alpha:6.4f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2.  Fallback – random normalised vector (should never happen).
    # --------------------------------------------------------------
    if best_seq is None:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = 2.0 * n * max_conv_fft(a)
        print(f"Fallback random init: n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Smooth‑max mirror‑descent optimisation.
    # --------------------------------------------------------------
    a = best_seq.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    n = a.shape[0]

    # FFT size that can accommodate both 2n‑1 (convolution) and
    # n+(2n‑1)-1 = 3n‑2 (correlation).  Using a power of two simplifies things.
    size_fft = 1 << ((3 * n - 2).bit_length())

    # optimisation hyper‑parameters
    beta = 3.0               # start temperature for the soft‑max
    beta_mul = 1.001         # geometric increase
    eta = 4.0                # initial step size for the exponentiated update
    eta_mul = 0.9995         # geometric decay
    min_eta = 1e-4

    # Adam‑style moments for a more stable direction (optional)
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    print("\n=== Smooth‑max refinement ===")
    while time.time() < deadline:
        it += 1

        # ----- self‑convolution of the current vector -----
        A = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"Iter {it:6d}: new global best = {best_score:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            print("Target reached – exiting refinement.")
            break

        # ----- soft‑max weights over convolution entries -----
        # shift by the maximum for numerical stability
        shifted = conv - cur_max
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (linear correlation) -----
        # correlation a ⋆ w = ifft( A * conj(W) )
        W = np.fft.rfft(w, n=size_fft)
        corr_full = np.fft.irfft(A * np.conj(W), n=size_fft)
        # linear correlation length = n + (2n‑1) – 1 = 3n‑2,
        # the useful slice starts at len(w)‑1 = 2n‑2
        offset = 2 * n - 2
        grad = 2.0 * corr_full[offset: offset + n]            # shape (n,)

        # ----- Adam‑style momentum on the gradient -----
        m1 = beta1 * m1 + (1 - beta1) * grad
        m2 = beta2 * m2 + (1 - beta2) * (grad * grad)
        m1_hat = m1 / (1 - beta1 ** it)
        m2_hat = m2 / (1 - beta2 ** it)

        # ----- exponentiated mirror‑descent step -----
        # we use a normalised direction m1̂ / (sqrt(m2̂)+eps)
        direction = m1_hat / (np.sqrt(m2_hat) + eps)
        factor = np.exp(-eta * direction)
        a = a * factor
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            # numerical safeguard – revert to best known
            a = best_seq.copy()

        # ----- occasional random perturbation to escape plateaus -----
        if it % 500 == 0:
            # move a tiny random amount of mass between a random pair
            i, j = np.random.choice(n, 2, replace=False)
            if a[i] > 0:
                delta = a[i] * np.random.rand() * 0.02
                a[i] -= delta
                a[j] += delta
                a /= a.sum()

        # ----- schedule updates -----
        beta = min(200.0, beta * beta_mul)        # warm‑up temperature
        eta = max(min_eta, eta * eta_mul)        # shrink step size

    # ------------------------------------------------------------------
    # 4.  Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print(f"Could not call evaluate_sequence: {exc}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best approximate score : {best_score:.6f}")
    print(f"Best exact score       : {exact_score:.6f}")
    print(f"Best length            : {len(best_seq)}")
    return best_seq.tolist()
