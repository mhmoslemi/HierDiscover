    max_val = conv.max()

def refine_sequence(a, deadline):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps = 1e-8
    iter = 0
    best = a.copy()
    best_score = score_from_vector(a)
    mu = 0.1
    mu_decay = 0.998 # gradually reduce

    while time.time() < deadline:
        iter += 1
        # gradually reduce mu
        mu = max(mu*mu_decay, 1e-6)

        grad = smooth_gradient(a, mu)

        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**iter)
        v_hat = v / (1 - beta2**iter)
        step = 0.1 * m_hat / (np.sqrt(v_hat) + eps) # step size maybe 0.1

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = simplex_projection(a_new)

        # evaluate
        sc = score_from_vector(a_new)
        if sc < best_score:
            best_score = sc
            best = a_new.copy()

        # maybe accept based on SA if not better
        # For simplicity, always accept if improves else keep.
        if sc < score_from_vector(a):

"""
Propose a step‑function (non‑negative vector) with a low evaluation score.

Algorithm
---------

1.  **Exploratory sweep** of a few low‑dimensional parametric families
    (U‑shapes based on distance‑to‑edge, product of distances,
    double‑exponential) on several moderately large lengths.
    Simple mixtures of two families are also examined.
    The best sequence found by this deterministic search is kept as a
    seed.

2.  **Smooth‑max gradient refinement**.
    The objective `2·n·max(conv)` is approximated by a smooth log‑sum‑exp
    of the convolution entries:
        f_μ(a) = 2·n·μ·log∑_k exp(conv_k/μ) .
    Its gradient can be written analytically as a convolution, which we
    evaluate with a single FFT.  An Adam‑style adaptive step is applied,
    followed by projection onto the probability simplex.
    The temperature for a simulated‑annealing acceptance rule and the
    smoothing parameter μ are annealed during the run.

3.  The best vector seen (according to the *exact* `evaluate_sequence`
    score) is returned as a plain Python list.

The whole procedure respects the provided time budget (≈1000 s) and
should improve the previous best bound of ~1.532 to below the target
1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w >= 0, sum(w) = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: (min(i,n‑1‑i)+1)^{-p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    return a / a.sum()


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges:
    ((i+1)*(n‑i))^{-p}.
    """
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    return a / a.sum()


def exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Double‑exponential: exp(-lam·i) + exp(-lam·(n‑1‑i))."""
    i = np.arange(n)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i)) + 1e-14
    return a / a.sum()


# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = 2 * n  # power‑of‑two not required – rfft works with any length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a * a (linear convolution)."""
    return float(full_convolution_fft(a).max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a normalized vector."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp) utilities
# ----------------------------------------------------------------------


def smooth_gradient(a: np.ndarray, mu: float) -> np.ndarray:
    """
    Gradient of 2·n·μ·log∑_k exp((a*a)[k]/μ) w.r.t. a.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)                # length 2n‑1
    max_val = conv.max()
    # stable soft‑max weights
    w = np.exp((conv - max_val) / mu)
    w_sum = w.sum()
    w = w / w_sum

    # convolution rev_a * w  (only need the central n entries)
    rev_a = a[::-1]
    size = rev_a.shape[0] + w.shape[0] - 1
    # next power of two for efficient FFT
    size_fft = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(rev_a, n=size_fft)
    fw = np.fft.rfft(w, n=size_fft)
    conv_rev = np.fft.irfft(fa * fw, n=size_fft)[:size]

    grad = conv_rev[n - 1 : n - 1 + n]               # slice of length n
    grad = 4.0 * n * grad                           # overall factor for the smoothed score
    return grad


# ----------------------------------------------------------------------
# Local refinement with Adam + simulated annealing
# ----------------------------------------------------------------------


def refine_sequence(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Starting from a0 (already normalised), perform a few thousand
    gradient steps using the smooth‑max surrogate and Adam adaptation.
    The loop stops when the time budget is exhausted.
    """
    a = a0.copy()
    a = a / a.sum()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    # annealing schedules
    temperature = 1.0
    temp_decay = 0.9999
    mu = 0.1
    mu_decay = 0.9995
    mu_min = 1e-6
    base_lr = 0.05  # base learning‑rate for Adam

    best = a.copy()
    best_score = score_from_vector(best)

    while time.time() < deadline:
        it += 1

        # ---- compute gradient of the smoothed objective ----------
        grad = smooth_gradient(a, mu)

        # ---- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate = simplex_projection(a_candidate)

        # ---- evaluate (exact) score -------------------------------
        cand_score = score_from_vector(a_candidate)

        # ---- simulated‑annealing acceptance ------------------------
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            prob = np.exp((best_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_candidate
            if cand_score < best_score:
                best_score = cand_score
                best = a_candidate.copy()
                print(f"Iter {it:7d}: new best refined score = {best_score:.6f}")

        # ---- schedules --------------------------------------------
        temperature *= temp_decay
        mu = max(mu * mu_decay, mu_min)

        # occasional random bump to avoid stagnation
        if it % 5000 == 0:
            bump = np.random.rand(n) * 1e-4
            a = simplex_projection(a + bump)
            temperature = 1.0  # re‑heat

    return best
