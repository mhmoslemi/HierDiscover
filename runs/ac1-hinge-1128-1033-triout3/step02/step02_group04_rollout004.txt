"""Propose candidate step sequence using a linear‑programming + mixing scheme.

The algorithm starts from a (possibly known) non‑negative sequence with
∑a = √(2 n).  At each iteration we solve the linear program

    max 1ᵀb    s.t.  b ≥ 0 ,   f * b ≤ M

where `*` denotes the linear self‑convolution and `M` is the current
maximum value of f*f.  The solution b is scaled to the target sum,
mixed with the current f (f ← (1‑t) f + t b) and the new maximum is
re‑evaluated.  The mixing step size t is adapted; when no improvement is
obtained we shrink t and eventually restart from the best solution with a
small random jitter.  After the LP phase a cheap local‑swap hill‑climber
finishes the search.

The whole routine respects the 1000‑second budget, works on the simplex,
and returns a list of floats ready for ``evaluate_sequence``.
"""

import numpy as np
import time

def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 2).bit_length())  # next power‑of‑two ≥ 2n‑1
    A = np.fft.rfft(a, n=m)
    c = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    return c


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    # 1. Problem size & initial vector
    # ------------------------------------------------------------------

        n = len(height_sequence_1)          # type: ignore[name‑defined]

        n = 1153

    if 'height_sequence_1' in globals():
        f = np.clip(np.array(height_sequence_1, dtype=float), 0.0, 1000.0)
        total = f.sum()
        if total > 0.0:
            f = f * (target_sum / total)
        else:
            f = np.full(n, target_sum / n)
    else:
        f = np.full(n, target_sum / n)

    # ------------------------------------------------------------------
    # 2. Initial objective (max of the self‑convolution)
    # ------------------------------------------------------------------
    conv_f = _conv_self(f)
    M = float(conv_f.max())
    best_f = f.copy()
    best_M = M
    print(f"Initial max = {M:.6f}")

    # ------------------------------------------------------------------
    # 3. Availability of a linear‑programming solver
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
        have_lp = True
    except Exception as e:
        print("scipy.linprog not available:", e)
        have_lp = False

    # ------------------------------------------------------------------
    # 4. Main LP + mixing loop
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0  # keep a safety margin
    iteration = 0
    step = 0.2
    min_step = 1e-3
    max_iter = 200  # safety upper bound; actual loop stops earlier

    while time.time() < deadline and iteration < max_iter:

        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        # column j contains f shifted by j positions
        for j in range(n):
            A[j : j + n, j] = f

        # ---- right‑hand side (all entries = current max M) -------
        b_ub = np.full(rows, M, dtype=np.float64)

        # ---- solve the linear programme ---------------------------
        if have_lp:
            c_obj = -np.ones(n, dtype=np.float64)      # minimise -Σb  => maximise Σb
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')

            best_M = cur_M
            print(f"Swap #{attempts} – improvement M = {best_M:.6f}")

    print(f"Final best M (objective) = {best_M:.6f}")
    return np.clip(best_f, 0.0, 1000.0).tolist()
