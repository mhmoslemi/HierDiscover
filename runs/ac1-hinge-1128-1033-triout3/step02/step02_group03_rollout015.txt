"""
propose_candidate – a simple stochastic (simulated‑annealing) optimizer.

The objective to minimise is

        M(a) = 2·n·max(conv(a,a)) / (sum(a)^2),

with a_i ≥ 0 (the security checks of ``evaluate_sequence`` clip
negative numbers to 0 and values larger than 1000 to 1000).

The theoretical minimum of the objective is 2.0 (uniform
distribution on the whole support), but the historic runs of the
competition achieved a slightly lower *numeric* value because of
floating‑point effects.  This implementation therefore

1. Starts from the best known construction (``height_sequence_1``) if
   it is present, otherwise from the uniform distribution.

2. Repeatedly proposes a tiny “mass‑transfer’’ move: pick two
   indices i≠j, move a small amount δ from i to j, re‑normalise the
   whole vector to keep Σa_i unchanged (the objective is scale‑invariant).

3. Accepts the move if it improves the score, otherwise with a
   probability exp(‑Δ/T) where the temperature T is slowly cooled.
   This standard simulated‑annealing scheme is able to escape shallow
   plateaus that defeat simple gradient‑descent tricks.

4. Keeps the best vector seen so far and returns it when the time
   budget expires.

The routine respects the 1000‑second budget, uses only NumPy (no
external solvers) and never performs any disallowed I/O.
"""

import time
import numpy as np

def _normalise(v: np.ndarray) -> np.ndarray:
    """Return a copy of v scaled to have L1‑norm equal to 1.
    Guarantees non‑negativity (tiny negatives are clipped)."""
    v = np.clip(v, 0.0, None)
    total = v.sum()
    if total == 0.0:
        # avoid division by zero – fall back to a uniform vector
        return np.full_like(v, 1.0 / v.size)
    return v / total


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a step‑function (non‑negative heights) that minimises
    ``evaluate_sequence``.  The function returns a plain Python list that
    satisfies all security checks of ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size (length of the step function)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name‑defined]
    except Exception:
        n = 1153                              # historic default length

    # ------------------------------------------------------------------
    # 2. Initial solution – use the best known construction if present,
    #    otherwise fall back to the uniform distribution (theoretical optimum).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        # use the historic candidate as a warm start
        cur = np.array(height_sequence_1, dtype=float)
        cur = np.clip(cur, 0.0, 1000.0)
        cur = _normalise(cur)
        print("[Init] Starting from existing height_sequence_1")
    else:
        cur = np.full(n, 1.0 / n, dtype=float)
        print("[Init] Starting from uniform distribution")

    # best solution found so far
    best = cur.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"[Init] Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Simulated annealing parameters
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    temperature = 1.0                         # initial temperature
    cooling_rate = 0.9995                     # exponential cooling per iteration
    min_temperature = 1e-6                    # stop cooling at this value
    iteration = 0
    max_iter_without_improve = 5000
    no_improve_cnt = 0

    # Helper to evaluate a candidate quickly (no extra copying)
    def _score(vec):
        return evaluate_sequence(vec.tolist())

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- propose a neighbour by moving a tiny amount of mass -----
        i, j = rng.choice(n, size=2, replace=False)
        # amount to move – a small fraction of the source component
        delta = rng.normal(scale=0.01) * cur[i]
        # keep the move tiny and keep the vector non‑negative
        delta = np.clip(delta, -cur[i] + 1e-12, cur[i] - 1e-12)

        cand[i] -= delta
        cand[j] += delta
        cand = _normalise(cand)

        # ----- evaluate candidate -----
        cand_score = _score(cand)

        # ----- acceptance test (simulated annealing) -----
        accept = False
        if cand_score < best_score:
            accept = True
            no_improve_cnt = 0
        else:
            # Metropolis criterion
            if temperature > min_temperature:
                prob = np.exp(-(cand_score - best_score) / temperature)
                if rng.random() < prob:
                    accept = True
                else:
                    no_improve_cnt += 1
            else:
                no_improve_cnt += 1

        if accept:
            cur = cand
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[Iter {iteration:6d}] New best score: {best_score:.6f}")

        # ----- temperature schedule -----
        temperature *= cooling_rate

        # ----- occasional random restart to avoid stagnation -----
        if no_improve_cnt >= max_iter_without_improve:
            cur = _normalise(rng.random(n))
            no_improve_cnt = 0
            print(f"[Iter {iteration:6d}] Random restart triggered")

        # safety exit if we have already reached the (unrealistic) target
        if best_score <= 1.503:
            print("[Target] Desired bound reached – exiting early")
            break

    # ------------------------------------------------------------------
    # 5. Final tidy‑up and return
    # ------------------------------------------------------------------
    result = np.clip(best, 0.0, 1000.0).tolist()
    print(f"Best score achieved: {best_score:.6f} after {iteration} iterations")
    return result
