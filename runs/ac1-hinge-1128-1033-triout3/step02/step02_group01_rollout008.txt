            i, j = np.random.choice(n, 2, replace=False)

       * a projected sub‑gradient step w.r.t. the maximal convolution entry,
       * a random swap of two coordinates.
   Moves are accepted by a simple simulated‑annealing rule
   (always if they improve, otherwise with probability exp(Δ/temperature)).
   The temperature and a step‑size decay slowly.  If no improvement is
   seen for many iterations we re‑heat the search with a tiny random
   bump.

3. **Return** –  The best vector found (as a plain list) is returned.
   The external evaluator will clip and renormalise it again, so we
   work with vectors that already sum to one.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w ≥ 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearest edge: a[i] ∝ (min(i,n-1-i)+1)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: a[i] ∝ ((i+1)*(n‑i))^‑p."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Exponential decay from the centre: a[i] ∝ exp(-lam * min(i,n-1-i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-lam * d) + 1e-14
    a /= a.sum()
    return a


def two_power_distribution(n: int, p: float, q: float) -> np.ndarray:
    """Two‑parameter power law: a[i] ∝ (i+1)^‑p * (n-i)^‑q."""
    i = np.arange(n)
    a = (i + 1) ** (-p) * (n - i) ** (-q) + 1e-14
    a /= a.sum()
    return a


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself using an FFT of power‑of‑two length.
    Returns a vector of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of a (a is assumed non‑negative)."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a normalised vector (sum == 1).
    It is 2 * n * max(a*a) where n = len(a).
    """
    n = a.shape[0]
    max_c = max_convolution(a)
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# Main proposal routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects a time budget (default 1000 s) and returns the best
    sequence (list of floats) discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin

    # ------------------------------------------------------------------
    # 1. deterministic exploration of simple parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Lengths to try – we keep them modest so that the deterministic part
    # finishes quickly compared to the total budget.
    lengths = [2000, 4000, 8000, 12000, 16000]
    # exponents for the power‑law families
    exponents = np.arange(0.1, 4.01, 0.1)
    # lambdas for the exponential family
    lambdas = np.arange(0.1, 4.01, 0.1)

    print("=== Deterministic parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # ----- single‑parameter families -----
        for p in exponents:
            # min‑distance U‑shape
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            # product‑distance U‑shape
            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

        # ----- exponential decay -----
        for lam in lambdas:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f" new best (exp)   n={n:5d} λ={lam:4.2f}  score={sc:.6f}")

        # ----- two‑parameter power law -----
        for p in np.arange(0.0, 3.01, 0.5):
            for q in np.arange(0.0, 3.01, 0.5):
                a = two_power_distribution(n, p, q)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f" new best (pq)    n={n:5d} p={p:3.1f} q={q:3.1f}  score={sc:.6f}")

        # occasional break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. include the previously supplied construction if it exists
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score = sc
                    best_seq = external.copy()
                    print(f" using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f" fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. stochastic refinement (sub‑gradient + random moves + SA)
    # ------------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score
    n = current.shape[0]

    temperature = 1.0
    temp_decay = 0.99995
    step0 = 0.2

    iteration = 0
    last_improve = 0

    print("\n=== Stochastic refinement start ===")
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # select a move type
        # --------------------------------------------------------------
        r = np.random.rand()
        if r < 0.4:
            # ----- random mass transfer -----
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] <= 0.0:
                continue
            delta = current[i] * np.random.rand() * 0.5  # move at most 50 % of source mass
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
        elif r < 0.8:
            # ----- projected sub‑gradient step -----
            conv = full_convolution_fft(current)
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-15
            act = np.where(conv >= max_val - eps)[0]

            grad = np.zeros_like(current)
            # accumulate sub‑gradient from all active convolution entries
            for k in act:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                i_vals = np.arange(lo, hi + 1)
                j_vals = k - i_vals
                grad[i_vals] += 2.0 * current[j_vals]

            # diminishing step size (1/√iteration)
            step = step0 / np.sqrt(iteration)
            cand = current - step * grad
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)
        else:
            # ----- random swap of two coordinates -----
            i, j = np.random.choice(n, 2, replace=False)
            cand = current.copy()
            cand[i], cand[j] = cand[j], cand[i]      # sum unchanged, non‑negativity preserved

        # --------------------------------------------------------------
        # evaluate candidate (cheap internal score)
        # --------------------------------------------------------------
        cand_score = score_from_vector(cand)

        # --------------------------------------------------------------
        # acceptance rule (simulated annealing)
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # schedule updates
        # --------------------------------------------------------------
        temperature *= temp_decay

        # --------------------------------------------------------------
        # stagnation handling – occasional re‑heat
        # --------------------------------------------------------------
        if iteration - last_improve > 50000:
            # add a tiny random bump to escape a flat region
            bump = np.random.rand(n) * 1e-4
            current = current + bump
            current = simplex_projection(current)
            current_score = score_from_vector(current)
            last_improve = iteration
            temperature = 1.0
            print(f"Iter {iteration:7d}: re‑heat triggered")

        # occasional progress report
        if iteration % 10000 == 0:
            print(f"Iter {iteration:7d}: current={current_score:.6f}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. final report and return
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_seq)})")
    # the external evaluator will clip/normalise again, so we return the raw vector
    return best_seq.tolist()
