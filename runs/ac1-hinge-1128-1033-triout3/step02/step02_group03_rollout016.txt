import time
import numpy as np

def arcsine_distribution(n):
    """Return a probability vector of length n approximating the arcsine density on [0,1].
    Uses midpoint rule with step 1/n and then normalises."""
    i = np.arange(n)
    x = (i + 0.5) / n
    # arcsine pdf: 1/(π sqrt{x (1-x)})
    dens = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
    a = dens / n
    # Numerical normalisation (should already sum to ~1)
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a

def max_conv(p):
    """Return max value of self‑convolution of p (with sum(p)=1)."""
    return np.max(np.convolve(p, p))

        M = 2·n·max(conv(a,a)) / (sum(a)**2)

    The construction starts from a discretised arcsine distribution,
    which is the continuous solution of the problem and gives a
    near‑uniform self‑convolution.  Afterwards a short simulated‑annealing
    hill‑climbing phase performs tiny mass transfers between coordinates
    to shave off the remaining excess.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # type: ignore[name‑defined]
    except Exception:
        n = 1153

    # ------------------------------------------------------------------
    # 2. Initialise with the arcsine distribution (mid‑point rule)
    # ------------------------------------------------------------------
    p = arcsine_distribution(n)

    # ------------------------------------------------------------------
    # 3. Keep the best solution seen so far
    # ------------------------------------------------------------------
    best_p = p.copy()
    best_score = 2.0 * n * max_conv(p)
    print(f"Initial score (arcsine): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing refinement (mass transfer moves)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # pre‑compute the convolution of the current vector
    conv = np.convolve(p, p)
    cur_score = best_score

    # annealing parameters
    T = 0.02            # initial temperature (controls acceptance of worsening moves)
    T_decay = 0.99995
    max_delta_abs = 0.02   # maximal absolute mass moved in one step

    it = 0
    while time.time() < deadline:
        it += 1

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # skip if there is nothing to take from i
        if p[i] <= 0.0:
            continue

        delta = rng.random() * min(p[i], max_delta_abs)

        conv_new = conv.copy()

        conv_new[j:j + n] += 2.0 * delta * p

        conv_new[2 * i] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

    while time.time() < deadline:
        it += 1

        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if p[i] <= 0.0:
            continue

        # propose a tiny transfer δ from i to j
        delta = rng.random() * min(p[i], max_delta)

        # ---- incremental update of the convolution (O(n)) ----
        conv_new = conv.copy()
        # linear part: 2·δ·(shifted p at j) – 2·δ·(shifted p at i)
        conv_new[j:j + n] += 2.0 * delta * p
        conv_new[i:i + n] -= 2.0 * delta * p
        # quadratic part
        conv_new[2 * j] += delta * delta
        conv_new[2 * i] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta
        # -----------------------------------------------------

        new_score = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / T):
            # accept move
            p[i] -= delta
            p[j] += delta
            conv = conv_new
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                print(f"[Iter {it:6d}] New best score: {best_score:.6f}")

        # annealing temperature decay
        T *= T_decay

        # occasional jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            p = np.clip(p + jitter, 0.0, None)
            s = p.sum()
            p = p / s if s > 0 else np.full(n, 1.0 / n)
            conv = np.convolve(p, p)
            cur_score = 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 5. return the best sequence found
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Best score after refinement: {best_score:.6f}")
    return result
