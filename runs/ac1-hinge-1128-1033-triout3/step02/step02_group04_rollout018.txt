        S = []                     # list of selected positions
        nd_counts = {}             # unordered‑pair count per sum (0,1,2)
        diag_sums = set()          # sums of the form 2*i for i in S

        # ----- greedy insertion -----
        for i in order:
            # diagonal capacity test
            s_diag = i * 2
            if nd_counts.get(s_diag, 0) > 1:          # would exceed 4 ordered entries
                continue

            # test all sums with already chosen indices
            ok = True
            for j in S:
                s = i + j
                cnt = nd_counts.get(s, 0)
                # more than two unordered pairs would give >4 ordered entries
                if cnt >= 2:
                    ok = False
                    break
                # if this sum already has a diagonal, we may only add ONE unordered pair
                if s in diag_sums and cnt == 1:
                    ok = False
                    break
            if not ok:
                continue

            # ----- accept i -----
            S.append(i)
            diag_sums.add(s_diag)
            # update unordered‑pair counters
            for j in S[:-1]:
                s = i + j
                nd_counts[s] = nd_counts.get(s, 0) + 1

        # after the first pass try to add any still‑feasible indices
        available = set(range(n)) - set(S)
        added = True
        while added and time.time() < deadline:
            added = False
            # scan the remaining indices – the first feasible one is added
            for i in list(available):
                s_diag = i * 2
                if nd_counts.get(s_diag, 0) > 1:
                    continue
                ok = True
                for j in S:
                    s = i + j
                    cnt = nd_counts.get(s, 0)
                    if cnt >= 2:
                        ok = False
                        break
                    if s in diag_sums and cnt == 1:
                        ok = False
                        break
                if not ok:
                    continue
                # accept i
                S.append(i)
                diag_sums.add(s_diag)
                for j in S[:-1]:
                    s = i + j
                    nd_counts[s] = nd_counts.get(s, 0) + 1
                available.remove(i)
                added = True
                break   # restart scanning after a successful insertion

        m = len(S)
        if m == 0:
            continue

        # ----- evaluate the constructed set -----
        # binary vector (ordered pair counts are exact integers)
        bin_vec = np.zeros(n, dtype=int)
        bin_vec[S] = 1
        conv = np.convolve(bin_vec, bin_vec)          # length 2*n-1, integer counts
        max_c = int(conv.max())                       # maximum ordered pair count
        score = 2.0 * n * max_c / (m * m)

        if score < best_score:
            best_score = score
            best_m = m
            best_maxc = max_c
            # uniform weights on the chosen positions
            best_seq = np.zeros(n, dtype=float)
            best_seq[S] = 1.0 / m
            print(
                f"Trial {trial:5d} – |S|={m:3d}, max_c={max_c:2d}, "
                f"score={score:.6f}"
            )
            if best_score <= TARGET:
                break

    # ------------------------------------------------------------------
    # 3. Fallback: uniform distribution (never used when a better solution is found)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = 2.0 * n * (1.0 / n)        # = 2.0
        best_m = n
        best_maxc = n

    print(
        f"Best sequence: size={best_m}, max_c={best_maxc}, "
        f"final score={best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # 4. Return the sequence (already respects non‑negativity and the caps)
    # ------------------------------------------------------------------
    return np.clip(best_seq, 0.0, 1000.0).tolist()
