"""
propose_candidate
-----------------

Generate a monotone step‑function whose entries follow a power law
a_i = 1000 * ((i+1)/n) ** p  (i = 0 … n‑1).

For a shape a_i ∝ i^p with a_0 = 0 the self‑convolution satisfies

    max_k (a*a)[k] = 1000²·B(p+1,p+1)·n                (B = beta‑function)

while the sum of the coefficients is  Σ a_i = 1000·n/(p+1).
Hence the evaluation score equals the constant

    score(p) = 2·(p+1)²·B(p+1,p+1),

which decreases rapidly with the exponent p.
Choosing a modest length (n = 15000) and a fairly large exponent
(e.g. p = 10) yields a score around 6·10⁻⁵ – far below the target
1.5, with essentially no risk of overflow because the scaling factor
keeps every entry ≤ 1000.

The function tries a few exponents (3,5,7,9,11,13) within the time
budget, evaluates the true score using ``evaluate_sequence`` and
returns the best sequence found.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     max_len: int = 15000,
                     **kwargs) -> list[float]:
    """
    Produce a sequence of non‑negative heights that (heuristically) minimises
    the evaluation function defined in `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed (not used here, kept for API compatibility).
    budget_s : int
        Time budget in seconds.
    max_len : int
        Desired length of the step function (upper bound).

    Returns
    -------
    List[float]
        The candidate sequence.
    """
    # ------------------------------------------------------------------
    # Basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    n = int(max_len)                 # length of the sequence we will try
    best_seq = None
    best_score = float('inf')

    # Exponents to try – higher exponents give lower theoretical scores.
    # We keep the list short because each evaluation requires an O(n²)
    # convolution (np.convolve) which is the dominant cost.
    exponents = [3, 5, 7, 9, 11, 13]

    # ------------------------------------------------------------------
    # Try each exponent while we have time left
    # ------------------------------------------------------------------
    for p in exponents:
        if time.time() > deadline:
            break

        # Build the power‑law sequence.
        # (i+1)/n ∈ (0,1]; raising to a high power pushes early values
        # towards zero while keeping the last entry exactly 1000.
        idx = np.arange(1, n + 1, dtype=np.float64)   # 1 … n
        ratio = idx / n
        seq_np = 1000.0 * np.power(ratio, p)          # shape ∝ ((i+1)/n)^p
        candidate = seq_np.tolist()

        # Evaluate the candidate (the external grader will also evaluate it,
        # this is only for our own logging / sanity‑check).
        try:
            score = evaluate_sequence(candidate)
        except Exception as exc:      # should never happen, but be safe
            print(f"[skip] p={p} raised an exception: {exc}")
            continue

        print(f"[candidate] p={p:2d}, length={n}, score={score:.9f}")

        if score < best_score:
            best_score = score
            best_seq = candidate[:]

    # ------------------------------------------------------------------
    # Fallback: if for any reason none of the candidates succeeded,
    # fall back to a simple uniform sequence (score = 2).
    # ------------------------------------------------------------------
    if best_seq is None:
        print("[fallback] using uniform sequence as a safety net")
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] uniform length={n}, score={best_score:.9f}")

    # Ensure the returned list contains plain Python floats and no negatives.
    best_seq = [float(max(0.0, x)) for x in best_seq]

    print(f"[result] best exponent yields score {best_score:.9f} "
          f"with length={len(best_seq)}")
    return best_seq
