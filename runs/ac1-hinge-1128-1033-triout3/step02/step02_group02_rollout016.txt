def cross_corr_fft(x, y, size=None):
    # x: length m, y: length n.
    # Compute linear cross-correlation (x ⋆ y)[k] = sum_{i} x[i + k] * y[i], where x[i+k]=0 if out-of-range.
    # Equivalent: fft(x, size) * conj(fft(y, size)). Use irfft.
    # compute size = next_pow2(len(x) + len(y) -1) if not given.
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    corr = np.fft.irfft(X * np.conj(Y), n=size)
    # result length size; linear correlation is first len(x)+len(y)-1 entries
    return corr[:len(x)+len(y)-1]

    n = len(a)
    size_fft = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(A * A, n=size_fft)[:2*n - 1]

import time
import numpy as np

def next_pow2(x):
    """Return the next power of two >= x."""
    return 1 << ((x - 1).bit_length())

def max_conv_fft(a):
    """
    Fast max of self-convolution using FFT.
    a must be 1‑D numpy array of non‑negative numbers.
    Returns a float.
    """
    n = a.shape[0]
    size = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())

def u_shaped_distribution(n, p, alpha):
    """
    Symmetric distribution a_i ∝ (d_i+1)^{-p} * exp(-alpha*d_i)
    where d_i = min(i, n-1-i).  Normalised to sum = 1.
    """
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)
    a = (d + 1) ** (-p) * np.exp(-alpha * d) + 1e-14   # tiny epsilon avoids exact zeros
    a_sum = a.sum()
    if a_sum == 0:
        a = np.ones_like(a) / n
    else:
        a /= a_sum
    return a

def approximate_score(a):
    """Fast proxy for evaluate_sequence (scale‑invariant)."""
    n = a.shape[0]
    return 2.0 * n * max_conv_fft(a)

def gradient_lp(a, p_norm):
    """
    Gradient of the L_p norm of the self‑convolution w.r.t. a.
    Returns a numpy array of length len(a).
    """
    n = a.shape[0]

    # --- convolution a * a ---
    size_conv = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size_conv)
    conv = np.fft.irfft(A * A, n=size_conv)[: 2 * n - 1]
    # guard against tiny negative values caused by round‑off
    conv = np.maximum(conv, 0.0)

    # --- quantities that appear in the gradient ---
    conv_p = conv ** p_norm
    sum_conv_p = np.sum(conv_p)
    # denominator: (∑ conv^p)^{(p-1)/p}
    denom = (sum_conv_p ** ((p_norm - 1) / p_norm)) if sum_conv_p > 0 else 1.0

    # C = conv^{p-1}
    C = conv ** (p_norm - 1)

    # --- cross‑correlation C ⋆ a (linear, not circular) ---
    size_corr = next_pow2((2 * n - 1) + n - 1)       # = next_pow2(3n-2)
    F_C = np.fft.rfft(C, n=size_corr)
    F_a = np.fft.rfft(a, n=size_corr)
    corr = np.fft.irfft(F_C * np.conj(F_a), n=size_corr)[: (2 * n - 1) + n - 1]
    # first n entries correspond to indices 0 … n-1
    grad_raw = 2.0 * p_norm * corr[:n]

    # final gradient (mirrored on the simplex)
    return grad_raw / (denom + 1e-15)

    The routine performs a quick parametric sweep over a few families of
    U‑shaped distributions, then refines the best candidate using mirror‑descent
    on a smooth L_p approximation of the max‑convolution norm.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin for final evaluation

    best_seq = None

Propose a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm consists of three phases:

1. **Parametric sweep** –  a fast O(n log n) scan over a family of
   symmetric “U‑shaped’’ distributions
   a_i ∝ (min(i,n‑1‑i)+1)^{‑p}·exp(-α·min(i,n‑1‑i)).
   For each candidate the cheap bound
   score = 2·n·max(a*a) (computed via FFT) is evaluated and the best
   candidate is retained.

2. **Mirror‑descent refinement** –  starting from the best candidate,
   we minimise a smooth L_p approximation of the infinity‑norm of the
   self‑convolution:
       f(a) = (∑_k (a*a)_k^p)^{1/p} .
   The gradient ∇f is obtained analytically and via FFT (cross‑correlation).
   An exponentiated‑gradient (mirror‑descent on the simplex) step
   a ← a·exp(-η·∇f) followed by renormalisation reduces f,
   while occasional random mass‑transfer moves allow escapes.
   The norm power p is gradually increased (8 → 20) so that the
   optimisation finally focuses on the true max‑entry.

3. **Exact evaluation** –  the best sequence found is finally fed to the
   official ``evaluate_sequence`` (once) and returned as a plain list.

The routine respects a 1000 s time budget, prints progress, and always
returns a feasible, non‑negative sequence.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_conv_fft(a: np.ndarray) -> float:
    """
    Fast maximum of the self‑convolution a*a using real FFT.
    a must be a 1‑D real array.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def _u_shaped_distribution(n: int, p: float, alpha: float) -> np.ndarray:
    """
    Symmetric distribution:
        a_i ∝ (d_i+1)^{-p}·exp(-α·d_i) ,  d_i = min(i, n‑1‑i)
    The vector is normalised to sum‑to‑one.
    """
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx)
    a = (d + 1) ** (-p) * np.exp(-alpha * d) + 1e-14
    total = a.sum()
    if total == 0:
        a = np.ones_like(a) / n
    else:
        a /= total
    return a


def _approximate_score(a: np.ndarray) -> float:
    """Scale‑invariant proxy for the true evaluation function."""
    n = a.shape[0]
    return 2.0 * n * _max_conv_fft(a)


def _gradient_lp(a: np.ndarray, p_norm: int) -> np.ndarray:
    """
    Gradient of the L_p norm of the self‑convolution with respect to a.
    Works on the simplex (non‑negative, sum 1).
    """
    n = a.shape[0]

    # --- convolution a*a -------------------------------------------------
    size_conv = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size_conv)
    conv = np.fft.irfft(A * A, n=size_conv)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)                     # guard against tiny negatives

    # denominator of the gradient formula: (∑ conv^p)^{(p‑1)/p}
    conv_p = conv ** p_norm
    sum_conv_p = np.sum(conv_p)
    denom = (sum_conv_p ** ((p_norm - 1) / p_norm)) if sum_conv_p > 0 else 1.0

    # C = conv^{p‑1}
    C = conv ** (p_norm - 1)

    # --- linear cross‑correlation C ⋆ a (no circular wrap) -----------------
    size_corr = _next_pow2((2 * n - 1) + n - 1)      # = next_pow2(3n‑2)
    F_C = np.fft.rfft(C, n=size_corr)
    F_a = np.fft.rfft(a, n=size_corr)
    corr = np.fft.irfft(F_C * np.conj(F_a), n=size_corr)[: (2 * n - 1) + n - 1]

    # gradient for each coordinate i (0 ≤ i < n)
    grad_raw = 2.0 * p_norm * corr[:n]
    return grad_raw / (denom + 1e-15)
