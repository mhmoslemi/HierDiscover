import time, numpy as np, math, random

def max_conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A*A, n=size)[:2*n - 1]
    return float(conv.max())

def prepare_shape_powU(n, p):
    d = np.arange(n)
    d = np.minimum(d, n-1-d)
    a = (d+1)**(-p) + 1e-12
    a /= a.sum()
    return a

def prepare_shape_product(n, p):
    i = np.arange(n)
    # use i+0.5 to avoid zero product at edges
    prod = (i+0.5)*(n-1-i+0.5)
    a = prod**(-p/2) + 1e-12
    a /= a.sum()
    return a

def prepare_shape_exponential(n, alpha):
    d = np.minimum(np.arange(n), n-1-np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a

def approximate_score(a):
    n = a.shape[0]
    maxb = max_conv_fft(a)
    return 2.0 * n * maxb  # sum a = 1

"""
Propose a step‑function minimising the given evaluation metric.

The algorithm works in three stages:

1) **Seeding** – reuse any externally supplied sequence.
2) **Broad parametric sweep** – explore several analytical families
   (U‑shaped power‑law, product‑power, exponential, sine‑based) over a
   range of lengths and parameters.  Convolution maxima are obtained
   with an O(n log n) real‑FFT, keeping the best candidate (approximate
   score = 2·n·max(conv) because the vector is always normalised to sum=1).
3) **Local stochastic refinement** – starting from the best candidate,
   a simulated‑annealing loop alternates between a gradient‑descent
   (mirror‑descent on the simplex) step and a random mass‑transfer move.
   Each move is accepted if it improves the fast approximate score or
   with a probability decreasing with temperature.
   The loop stops when the time budget expires or the target score
   (≈1.503) is reached.

Finally the best sequence is evaluated with the official ``evaluate_sequence``
function (once) and returned as a plain Python list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based)
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """
    Return the maximal entry of the self‑convolution a*a using a real FFT.
    The input vector is assumed to be non‑negative and normalised to sum=1,
    but the routine works for any real vector.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def approx_score(a: np.ndarray) -> float:
    """
    Approximate the evaluation metric for a normalised (sum=1) vector.
    """
    n = a.shape[0]
    return 2.0 * n * max_conv_fft(a)


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def shape_powU(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law U‑shape:
        a[i] = (min(i, n-1-i) + 1)^{-p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1.0) ** (-p) + 1e-12          # avoid exact zeros
    a /= a.sum()
    return a


def shape_product_power(n: int, p: float) -> np.ndarray:
    """
    Symmetric product‑power shape:
        a[i] = ((i+½)*(n‑1‑i+½))^{-p/2}
    Normalised to sum = 1.  This corresponds to a Beta‑type distribution
    and includes the arcsine law as a special case (p≈1).
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - 1 - i + 0.5)
    a = prod ** (-p / 2.0) + 1e-12
    a /= a.sum()
    return a


def shape_exponential(n: int, alpha: float) -> np.ndarray:
    """
    Exponential decay from the edges:
        a[i] = exp(-alpha * min(i, n-1-i))
    Normalised to sum = 1.
    """
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def shape_sine(n: int, power: float) -> np.ndarray:
    """
    Positive sine‑based shape:
        a[i] = sin( π * (i+½) / n )^{power}
    Normalised to sum = 1.  For power<1 the distribution becomes more
    edge‑heavy, for power>1 it concentrates in the centre.
    """
    theta = np.pi * (np.arange(n) + 0.5) / n
    a = np.sin(theta) ** power + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main proposal routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = np.inf
    TARGET = 1.503

    # 0.  Seed from a possibly existing global variable.
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)   # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 1e-12:
                external = external / s
                score_ext = approx_score(external)
                best_seq = external.copy()
                best_score = score_ext
                print(f"Seeded from height_sequence_1 – approx. score={score_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic parametric sweep over several families.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 16000,
               20000, 24000, 28000, 32000]   # keep below ~35k for safety
    p_vals = np.linspace(0.2, 2.5, 23)           # 0.2,…,2.5
    prod_p_vals = np.linspace(0.4, 4.5, 22)      # 0.4,…,4.5
    alphas = np.linspace(0.0001, 0.009, 18)     # exponential decay
    sine_pows = np.linspace(0.2, 4.0, 20)       # sine exponent

    print("\n=== Parametric sweep ===")

        # ------------------------------------------------------------------

        else:
