"""
Propose a step‑function that minimizes the metric defined by
`evaluate_sequence`.  The algorithm combines three ideas:

1. **Fast parametric sweep** – generate a small family of symmetric
   Beta‑shaped or power‑law sequences (including the classic
   (1‑x)^{-1/2} series) and keep the best one according to a cheap
   FFT‑based evaluation of 2·n·max(conv).

2. **Smooth max optimisation** – starting from the best candidate,
   minimise the smoothed objective
       S(a)= (1/β)·log∑_k e^{β·(a∗a)_k}
   with projected mirror descent on the probability simplex.
   The softmax‑gradient is obtained via a correlation of the current
   vector with the soft‑max weights; an Adam optimiser (mirrored on the
   log‑probabilities) provides fast convergence.

3. **Fine‑grained simulated‑annealing** – after the gradient phase a
   short Metropolis walk transfers small amounts of mass between two
   random coordinates, accepting downhill moves or uphill moves with a
   temperature decreasing to zero.  This cleans up any remaining tiny
   irregularities.

All heavy work is done with O(n log n) FFTs; the expensive
`evaluate_sequence` function is only called when a new global best is
found.  The routine respects the 1000 s budget and works for a single
CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the metric for a unit‑sum vector.
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, size_fft: int) -> float:
    """Return 2·n·max(a∗a) for a vector a that already sums to 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic start: coefficients of (1‑x)^(-1/2) (the classic “square‑root” series)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^(-1/2), normalised to sum 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------
def _beta_series(n: int, a_param: float) -> np.ndarray:
    """Symmetric Beta(a_param,a_param) density on {0,…,n‑1} (discretised)."""
    x = (np.arange(n) + 0.5) / n          # points in (0,1)
    w = (x ** (a_param - 1.0)) * ((1.0 - x) ** (a_param - 1.0))
    w /= w.sum()
    return w


def _power_series(n: int, p: float) -> np.ndarray:
    """Monotonically decreasing power‑law a_i ∝ (i+1)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** (-p)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Return a non‑negative sequence that (approximately) minimises
    `evaluate_sequence`.  The function respects the time budget and
    prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – use a user supplied length if present, otherwise 8000
    # ------------------------------------------------------------------
    n = int(kwargs.get("length", 8000))
    if n <= 0:
        raise ValueError("length must be a positive integer")

    # FFT lengths needed for convolution and correlation
    size_fft = 1 << ((2 * n - 1).bit_length())          # >= 2n‑1
    size_fft_corr = 1 << ((3 * n - 2).bit_length())    # >= n+(2n‑1)-1

    # ------------------------------------------------------------------
    # Keep the best true result found so far
    # ------------------------------------------------------------------
    best_true_seq = None
    best_true_score = float("inf")

    # ------------------------------------------------------------------
    # ---- 1. Fast parametric sweep ------------------------------------
    # ------------------------------------------------------------------
    best_local_a = None
    best_local_fast = float("inf")

    # ---- a) Square‑root series ----------------------------------------
    a_sqrt = _sqrt_series(n)
    fast_sqrt = _fast_score(a_sqrt, size_fft)
    best_local_a, best_local_fast = a_sqrt, fast_sqrt

    # ---- b) Power‑law family (p from 0.1 to 1.0) --------------------
    for p in np.linspace(0.1, 1.0, 12):
        a_pow = _power_series(n, p)
        fast_val = _fast_score(a_pow, size_fft)
        if fast_val < best_local_fast:
            best_local_a, best_local_fast = a_pow, fast_val

    # ---- c) Symmetric Beta family (α from 0.2 to 1.2) -----------------
    for a_param in np.linspace(0.2, 1.2, 12):
        a_beta = _beta_series(n, a_param)
        fast_val = _fast_score(a_beta, size_fft)
        if fast_val < best_local_fast:
            best_local_a, best_local_fast = a_beta, fast_val

    # Evaluate the best of the sweep with the true function
    if best_local_fast < best_true_score - 1e-12:
        true_val = evaluate_sequence(best_local_a.tolist())
        if true_val < best_true_score:
            best_true_score = true_val
            best_true_seq = best_local_a.copy()
            print(f"[global] new best after sweep {best_true_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 2. Smooth‑max gradient descent (Adam on log‑probabilities) --
    # ------------------------------------------------------------------
    # Initialise the mirror variables (log‑probabilities) from the best sweep
    eps = 1e-30
    p_vec = np.log(best_local_a + eps)    # shape (n,)
    m_vec = np.zeros_like(p_vec)
    v_vec = np.zeros_like(p_vec)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12
    lr = 0.4

    t_iter = 0
    max_iters = 250_000                 # safety bound; actual stop is time‑driven
    beta_start = 5.0                    # initial β for the smooth max
    beta_max = 120.0                    # final β (approaches true max)

    while time.time() < deadline and t_iter < max_iters:
        # ---- compute current probability vector (simplex) ----------------
        max_p = np.max(p_vec)
        exp_p = np.exp(p_vec - max_p)    # stabilise
        a = exp_p / exp_p.sum()          # sums to 1, all positive

        # ---- convolution a∗a and smooth max weights --------------------
        fa = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
        max_conv = conv.max()
        fast_score = 2.0 * n * max_conv

        # update global best if we improve (ignore tiny numerical noise)
        if fast_score < best_true_score - 1e-12:
            true_val = evaluate_sequence(a.tolist())
            if true_val < best_true_score:
                best_true_score = true_val
                best_true_seq = a.copy()
                print(f"[global] new best {best_true_score:.6f}")

        # ---- smooth‑max (log‑sum‑exp) -----------------------------------
        # β is increased slowly during the run
        beta = beta_start + (beta_max - beta_start) * (t_iter / max_iters)
        # stability shift:
        shift = max_conv
        exp_w = np.exp(beta * (conv - shift))
        w = exp_w / exp_w.sum()          # shape (2n‑1,)

        # ---- gradient w.r.t. a (via correlation) ----------------------
        # linear correlation r[l] = Σ_i a_i * w_{i+l}, l = 0…n‑1
        # computed as IFFT(FA * conj(FW))
        FA = np.fft.rfft(a, size_fft_corr)
        FW = np.fft.rfft(w, size_fft_corr)
        corr_full = np.fft.irfft(FA * np.conj(FW), size_fft_corr)
        offset = len(w) - 1               # = 2n‑2
        grad_a = 2.0 * corr_full[offset: offset + n]   # shape (n,)

        # ---- mirror‑gradient (w.r.t. p) --------------------------------
        inner = np.dot(a, grad_a)          # scalar ⟨a,g⟩
        grad_p = a * (grad_a - inner)      # ∂S/∂p_i

        # ---- Adam update ------------------------------------------------
        t_iter += 1
        m_vec = beta1 * m_vec + (1.0 - beta1) * grad_p
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m_vec / (1.0 - beta1 ** t_iter)
        v_hat = v_vec / (1.0 - beta2 ** t_iter)
        p_vec -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- occasional jitter to escape tiny plateaus -----------------
        if t_iter % 5000 == 0:
            jitter = np.random.randn(n) * 1e-3 / math.sqrt(t_iter + 1)
            p_vec += jitter

    # ------------------------------------------------------------------
    # ---- 3. Simulated‑annealing fine‑tuning -------------------------
    # ------------------------------------------------------------------
    # Use the best vector obtained from the Adam phase as the starting point
    if best_true_seq is None:
        a_cur = best_local_a.copy()
    else:
        a_cur = best_true_seq.copy()
    fast_cur = _fast_score(a_cur, size_fft)

    # SA parameters
    T_start = 0.12
    elapsed_sa = 0.0
    iter_sa = 0
    no_improve = 0

    while time.time() < deadline:
        # linear cooling schedule based on elapsed SA time
        elapsed_sa = time.time() - (deadline - (budget_s - 0.5))
        frac = min(1.0, elapsed_sa / (budget_s - 0.5))
        T = T_start * (1.0 - frac)

        # pick two distinct indices (source has non‑zero mass)
        src = np.random.randint(n)
        if a_cur[src] < 1e-14:
            continue
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # transfer a random fraction of src's mass
        max_delta = a_cur[src] * min(0.5, T)
        delta = np.random.rand() * max_delta

        a_cand = a_cur.copy()
        a_cand[src] -= delta
        a_cand[dst] += delta

        fast_cand = _fast_score(a_cand, size_fft)

        # Metropolis acceptance
        if fast_cand <= fast_cur or np.random.rand() < math.exp(-(fast_cand - fast_cur) / max(T, 1e-12)):
            a_cur = a_cand
            fast_cur = fast_cand
            if fast_cand < best_true_score - 1e-12:
                true_val = evaluate_sequence(a_cur.tolist())
                if true_val < best_true_score:
                    best_true_score = true_val
                    best_true_seq = a_cur.copy()
                    print(f"[global] new best (SA) {best_true_score:.6f}")
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional restart if stuck for a long time
        if no_improve > 25000:
            a_cur = best_local_a.copy()
            a_cur += np.random.randn(n) * 5e-4
            a_cur = np.maximum(a_cur, 0.0)
            a_cur /= a_cur.sum()
            fast_cur = _fast_score(a_cur, size_fft)
            no_improve = 0
            print(f"[restart] SA restart after {iter_sa} iterations")

        iter_sa += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    if best_true_seq is None:
        # fallback – uniform distribution (should never happen)
        best_true_seq = np.full(n, 1.0 / n, dtype=np.float64)

    final_val = evaluate_sequence(best_true_seq.tolist())
    print(f"[final] best evaluation = {final_val:.6f}")
    return best_true_seq.tolist()
