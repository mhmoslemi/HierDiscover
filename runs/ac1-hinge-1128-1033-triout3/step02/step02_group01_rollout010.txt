        # We'll run at most 10000 iterations or until time.
        for it in range(20000):
            if time.time() > deadline:

            conv = full_convolution_fft(a)
            maxval = conv.max()
            # compute subgradient for all indices within tolerance 1e-12 relative
            eps_tol = maxval * 1e-12 + 1e-15
            max_idxs = np.where(conv >= maxval - eps_tol)[0]
            # compute gradient
            grad = np.zeros_like(a)
            for k in max_idxs:

            # Adam update

            a = a - step
            a = np.maximum(a, 0)
            a = simplex_projection(a)
            # compute score
            score = 2 * n * maxval
            if score < best_score:

                best_seq = a.copy()

            # optional simulated annealing acceptance: if score higher, could occasionally accept to escape.

    # Use simple hill climbing random transfers
    for it in range(50000):

            current = cand
            print(f"Random move iter {it}: new best {cand_score:.6f}")
    # final return
    print("Search finished. Best score:", best_score)
    return best_seq.tolist()

"""
Propose a low‑score step‑function.

Algorithm
---------
1.  Exhaustive sweep of a few parametric families (U‑shape based on
    distance to the nearest edge, product‑distance, exponential decay)
    for several lengths and exponents.  The best candidate (lowest
    2·n·max(a*a)) is kept.

2.  Refine the best candidates with a projected Adam optimiser
    (sub‑gradient of the maximal convolution entry, simplex projection
    after each step).  A small number of random restarts are performed
    and the global best is updated continuously.

3.  A final short hill‑climbing phase using random “mass‑transfer”
    moves helps to escape remaining local minima.

All convolutions are evaluated with a single FFT which makes the inner
loop fast enough to explore millions of candidates within the 1000 s
budget.  The returned vector is normalised to sum = 1 (the external
evaluator re‑normalises it anyway) and contains only non‑negative
numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {w >= 0, sum(w) = 1}.
    O(n log n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT (real‑valued)."""
    n = a.shape[0]
    size = 2 * n                       # zero‑pad to avoid circular wrap‑around
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (linear) self‑convolution of a."""
    conv = full_convolution_fft(a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a sum‑1 vector."""
    n = len(a)
    return 2.0 * n * max_convolution(a)


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] = (min(i, n‑1‑i) + 1)^{-p}   (U‑shape based on nearest edge)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] = ((i+1)*(n-i))^{-p}   (U‑shape based on product of distances)."""
    i = np.arange(n)
    prod = (i + 1) * (n - i)
    a = prod.astype(np.float64) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_exp_distribution(n: int, c: float) -> np.ndarray:
    """a[i] = exp(-c * min(i, n‑1‑i))   (exponential decay towards the centre)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = np.exp(-c * d) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (projected Adam)
# ----------------------------------------------------------------------


def adam_step(a: np.ndarray,
              grad: np.ndarray,
              m: np.ndarray,
              v: np.ndarray,
              t: int,
              lr: float,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """One projected Adam update for a simplex‑constrained vector."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = np.maximum(a_new, 0.0)
    a_new = simplex_projection(a_new)
    return a_new, m, v


def subgradient_maxconv(a: np.ndarray,
                        eps_rel: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the current a.
    We collect all indices whose convolution value is within
    eps_rel of the maximum and sum the corresponding contributions.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_val = conv.max()
    eps = max_val * eps_rel + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1)
        j_vals = k - i_vals
        # contribution 2 * a[j] to each i
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Main proposal routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the given time budget and
    returns the best sequence (list of floats) it discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # small safety margin

    # --------------------------------------------------------------
    # 1.  Deterministic sweep over parametric families
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    lengths = [2000, 4000, 8000, 12000, 16000, 20000, 24000, 28000, 32000]
    # exponents for the power‑law families (focus on the sub‑linear regime)
    p_vals = np.arange(0.05, 1.01, 0.05)   # 0.05, 0.10, …, 1.00
    # exponential decay constants
    c_vals = np.arange(0.001, 0.021, 0.002)   # 0.001, 0.003, …, 0.019

    print("=== Deterministic parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        for p in p_vals:
            # (i) min‑distance

            if sc1 < best_score:
                best_score = sc1
                best_seq = a1.copy()

                best_seq = a2.copy()
                print(f" new best (prod)  n={n:5d} p={p:4.2f}  score={sc2:.6f}")

        # --- exponential families ---

            sc3 = score_from_vector(a3)

                best_score = sc3
                best_seq = a3.copy()
                print(f" new best (exp)   n={n:5d} c={c:5.3f}  score={sc3:.6f}")

        # occasional time‑check
        if time.time() > deadline:

        best_seq = np.random.rand(n)

    # --------------------------------------------------------------
    # 4.  Refine with projected Adam (multiple restarts)
    # --------------------------------------------------------------
    n = best_seq.shape[0]
    print("\n=== Adam refinement (first seed) ===")
    # we keep a list of seeds – the best sequence plus a few random variants
    seeds = [best_seq.copy()]
    for _ in range(3):
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        seeds.append(rnd)

    # optimiser hyper‑parameters (they work reasonably for all n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # learning rate – we scale it loosely as 0.3 / sqrt(n)
    base_lr = 0.3 / np.sqrt(n)

    # run a short Adam optimisation for each seed

        if time.time() > deadline:

    # --------------------------------------------------------------
    # 5.  Final hill‑climbing with random mass‑transfer moves
    # --------------------------------------------------------------
    print("\n=== Random mass‑transfer hill climbing ===")
    current = best_seq.copy()
    n = len(current)
