A projected‑Adam optimiser for the aperiodic self‑convolution bound.

The objective to minimise (for a vector a with Σa = 1) is
    F(a) = 2·n·max_k (a*a)[k] .
The gradient of max_k (a*a)[k] at a point where the maximum is attained
k* is the sub‑gradient
    g_i = 2·a_{k*−i}   if 0 ≤ i ≤ n−1 and 0 ≤ k*−i ≤ n−1,
    g_i = 0           otherwise.

We keep the iterate on the probability simplex (a_i ≥ 0, Σa_i = 1) by an
Euclidean projection (the O(n log n) algorithm of Duchi et al.).
To speed‑up convergence we use Adam’s momentum and variance estimates.
A tiny random mass‑transfer is performed occasionally to escape plateaus.

The optimiser works on a symmetric vector (a[i] = a[n‑1‑i]); this halves
the effective dimension and matches the symmetry of the objective.
When the current run stagnates we optionally up‑sample the vector (linear
interpolation) and continue optimisation at twice the length – this lets
the algorithm discover finer‑grained shapes without restarting from
scratch.

The routine respects the 1000 s time limit, periodically prints progress
and finally returns the best sequence found (as a plain Python list).

def _project_onto_simplex(v):
    """
    Project a vector v onto the simplex {x ≥ 0, Σx = 1}.
    Implements the O(n log n) algorithm of Duchi, Shalev‑Shwartz,
    Singer and Chandra (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the largest rho that satisfies the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safeguard – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1            # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to avoid tiny rounding errors
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full linear convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution(a):
    """
    Compute the full (length 2·n‑1) convolution a*a using a real FFT.
    The result is clipped to non‑negative values to avoid spurious
    negative rounding errors.
    """
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0, out=conv)            # clip tiny negatives
    return conv

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Projected‑Adam optimiser for ``evaluate_sequence``.
    Returns a list of non‑negative heights (the best sequence found
    before the time budget expires).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so that the framework does not kill us
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initial length and initial vector (symmetric, on the simplex)
    # ------------------------------------------------------------------
    n = 512                         # start modestly; we will double later
    a = np.random.dirichlet(np.ones(n))          # Σa = 1
    a = (a + a[::-1]) / 2.0                       # enforce symmetry
    a = a / a.sum()                               # re‑normalise

    # If a previous construction exists, reuse it (truncating or interpolating)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)  # noqa: F821
        if prev.ndim == 1 and prev.size >= 2:
            # project onto simplex and symmetrise
            prev = _project_onto_simplex(prev.tolist())
            prev = (prev + prev[::-1]) / 2.0
            prev = prev / prev.sum()
            if prev.shape[0] >= n:
                a = prev[:n]
                a = a / a.sum()
                a = (a + a[::-1]) / 2.0
                print("[init] started from height_sequence_1 (truncated)")
    except Exception:
        # no usable global variable – we keep the random start
        pass

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_val = float('inf')
    best_seq = a.copy()
    iteration = 0
    last_improve = 0          # iteration index of the last improvement
    restart_gap = 200_000     # after this many iterations without gain we restart
    upsample_interval = 50_000
    upsampled = False

    # ------------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # --------------------------------------------------------------
        # Evaluate current objective
        # --------------------------------------------------------------
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv             # Σa = 1 ⇒ denominator = 1

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{iteration:9d}] n={n:<5d} new best = {best_val:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient of the active convolution peak
        # --------------------------------------------------------------
        k_star = int(np.argmax(conv))
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        subgrad = np.zeros_like(a)
        subgrad[idx] = 2.0 * a[k_star - idx]          # g_i = 2·a_{k*‑i}

        # --------------------------------------------------------------
        # Adam update (learning‑rate schedule: linear decay)
        # --------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * subgrad
        v = beta2 * v + (1.0 - beta2) * (subgrad * subgrad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # linear decay from 0.30 → 1e‑4 over the whole time budget
        progress = (now - start_time) / (deadline - start_time)
        lr = 0.30 * (1.0 - progress) + 1e-4 * progress
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step

        # occasional random perturbation (helps to escape plateaus)
        if np.random.rand() < 0.01:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta = np.random.rand() * a_candidate[src] * 0.5
            a_candidate[src] -= delta
            a_candidate[dst] += delta

        # --------------------------------------------------------------
        # Project back onto the simplex and re‑enforce symmetry
        # --------------------------------------------------------------
        a = _project_onto_simplex(a_candidate.tolist())
        a = (a + a[::-1]) / 2.0                     # symmetry
        a_sum = a.sum()
        if a_sum != 0.0:
            a = a / a_sum

        # --------------------------------------------------------------
        # Restart if we have been stuck for a long time
        # --------------------------------------------------------------
        if iteration - last_improve > restart_gap and (deadline - time.time()) > 30.0:
            # simple random restart (keep the best found so far)
            a = np.random.dirichlet(np.ones(n))
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[{iteration}] restart at n={n}")

        # --------------------------------------------------------------
        # Upsample (double the length) if enough time remains
        # --------------------------------------------------------------
        if (iteration % upsample_interval == 0) and (n < 4096):
            remaining = deadline - time.time()
            # we need at least ~60 s to make the up‑sampled phase worthwhile
            if remaining > 60.0:
                new_n = n * 2
                x_old = np.linspace(0.0, 1.0, n)
                x_new = np.linspace(0.0, 1.0, new_n)
                a = np.interp(x_new, x_old, a)
                a = a / a.sum()
                # re‑initialise Adam buffers for the larger space
                m = np.zeros_like(a)
                v = np.zeros_like(a)
                n = new_n
                upsampled = True
                print(f"[{iteration}] up‑sampled to n={n}")

    # ------------------------------------------------------------------
    # Final reporting (uses the official evaluation routine for sanity)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] official evaluation of best sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_seq.tolist()
