  `max(conv(p,p))`.  The sub‑gradient is known analytically (it only

import time
import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0 , Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise (may be off by 1e‑12)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n)
    return w

    while time.time() < deadline:
        it += 1

        conv = np.convolve(p, p)                     # length 2n‑1
        diff = conv - mu
        # gradient of ½‖conv‑μ‖² = 4·p * (diff reversed)
        corr = np.convolve(p, diff[::-1])            # length 3n‑2
        grad = 4.0 * corr[2 * n - 2 - np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        p_new = p - step
        p_new = _simplex_projection(p_new)

        # occasional jitter to prevent dead‑ends
        if it % 200 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            p_new = _simplex_projection(p_new + jitter)

        p = p_new

        cur_score = 2.0 * n * np.max(np.convolve(p, p))
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[Phase A] iter {it:6d} – new best {best_score:.6f}")

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 4. Phase B – Exact max sub‑gradient with exponentiated updates
    # ------------------------------------------------------------------
    eta = 0.5
    eta_decay = 0.9995
    sub_it = 0
    no_improve = 0

    while time.time() < deadline:
        sub_it += 1

        conv = np.convolve(p, p)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) at the active index
        grad = np.zeros_like(p)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad[idx] = 2.0 * p[k_max - idx]

        # exponentiated (mirror) update
        p = p * np.exp(-eta * grad)
        s = p.sum()
        if s == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p = p / s

        # jitter now and then
        if sub_it % 500 == 0:
            p = _simplex_projection(p + rng.normal(scale=0.001, size=n))

        eta *= eta_decay

        cur_score = 2.0 * n * np.max(np.convolve(p, p))
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
            total = it + sub_it
            print(f"[Phase B] iter {total:6d} – new best {best_score:.6f}")
        else:
            no_improve += 1

        # restart if we are stuck for a while
        if no_improve >= 2000:
            p = best_p + rng.normal(scale=0.005, size=n)
            p = np.clip(p, 0.0, None)
            p = p / p.sum()
            no_improve = 0

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 5. Phase C – Light random sampling with remaining time
    # ------------------------------------------------------------------
    while time.time() < deadline:
        p_rand = rng.random(n)
        p_rand /= p_rand.sum()
        cur_score = 2.0 * n * np.max(np.convolve(p_rand, p_rand))
        if cur_score < best_score:
            best_score = cur_score
            best_p = p_rand.copy()
            print(f"[Random] new best {best_score:.6f}")

        # quick exit when we are almost out of time
        if time.time() + 0.01 > deadline:
            break

    # ------------------------------------------------------------------
    # 6. Return result (clipped to the interval accepted by evaluate_sequence)
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Best final score: {best_score:.6f}")
    return result
