def _corr_weighted(a, w):
    n = a.shape[0]
    m = w.shape[0]  # 2n-1
    # reverse w
    w_rev = w[::-1]
    size = 1 << ((n + m - 1 - 1).bit_length())
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(A*W, size)[:n+m-1]  # length n+m-1
    T = m - 1
    g = 2.0 * conv[T - np.arange(n)]
    return g

"""
Propose a step‐function that minimizes ``evaluate_sequence``.

The algorithm combines three ideas that go beyond the plain
sub‑gradient method used previously:

1. **Smooth max approximation** – we replace the non‑smooth
   ``max(conv)`` by a softmax with a temperature ``ε`` that is
   gradually annealed.  The gradient of this smoothed objective is a
   weighted sum of the gradients of all convolution entries, not only
   the current maximiser.

2. **Adam‑style adaptive steps** – the (smoothed) gradient is fed to an
   Adam optimiser, which automatically rescales the step in each
   coordinate.  After the Adam step the vector is projected back onto
   the probability simplex (positivity and ∑a = 1), which leaves the
   evaluation function invariant.

3. **Exploration moves** – with a small probability we perform a
   random mass transfer between two positions, and every few thousand
   iterations we apply a larger random permutation of the current
   vector.  This helps to escape plateaus and prevents the optimiser
   from getting stuck in a poor local minimum.

The optimisation runs until the time budget (≈1000 s) is exhausted
and returns the best sequence seen.  The code is fully self‑contained
and only uses NumPy.  All helper functions are defined at module level
so that they may be inspected by the competition framework.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x>=0, sum(x)=1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full convolution (real FFT) of two arbitrary 1‑D arrays
# ----------------------------------------------------------------------
def _full_convolution_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1 - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[: n + m - 1]
    np.maximum(conv, 0, out=conv)  # clip tiny negatives from rounding
    return conv


# ----------------------------------------------------------------------
# Helper: self‑convolution (a*a) – a thin wrapper around the generic one
# ----------------------------------------------------------------------
def _full_convolution_self(a: np.ndarray) -> np.ndarray:
    return _full_convolution_two(a, a)


# ----------------------------------------------------------------------
# Helper: gradient of the soft‑max‑smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _smoothed_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Compute ∇_a   ε·log(∑_k exp(conv_k/ε))   where conv = a*a.
    The gradient is 2 * correlation(a, softmax(conv, eps)).
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _full_convolution_self(a)                 # length 2n‑1
    c_max = conv.max()

    # 2) soft‑max weights, numerically stable
    #    w_k ∝ exp((conv_k - c_max) / eps)
    scaled = (conv - c_max) / eps
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum == 0:          # eps became too small – fall back to hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # 3) correlation a ⋆ w  (g_i = 2 * Σ_j a_j w_{j+i})
    #    correlation can be obtained via a convolution with w reversed.
    w_rev = w[::-1]                              # length 2n‑1
    conv_rev = _full_convolution_two(a, w_rev)  # length 3n‑2
    T = len(w) - 1                               # = 2n‑2
    # extract g_i = 2 * conv_rev[T - i]  for i=0…n‑1
    g = 2.0 * conv_rev[T - np.arange(n)]
    return g
