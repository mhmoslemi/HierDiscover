def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0 , Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(p: np.ndarray, fft_size: int, out_len: int) -> np.ndarray:
    """
    Convolution of p with itself using FFT.
    `out_len` should be 2*len(p)-1.
    """
    fp = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(fp * fp, fft_size)[:out_len]
    # Numerical noise may produce tiny negative values – clip to zero.
    return np.maximum(conv_full, 0.0)


def _softmax_grad(p: np.ndarray, conv: np.ndarray, alpha: float,
                  fft_size: int, n: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation  Σ_k w_k (p*p)_k
    where w_k = softmax(alpha·conv_k).
    Returns a vector of length n.
    """
    # soft‑max weights (stable)
    z = alpha * conv
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / np.sum(exp_z)

    # compute gradient via convolution with reversed w
    w_rev = w[::-1]
    fp = np.fft.rfft(p, fft_size)
    fw = np.fft.rfft(w_rev, fft_size)
    # length of (p * w_rev) is 3*n-2
    corr_full = np.fft.irfft(fp * fw, fft_size)[: (3 * n - 2)]
    idx = 2 * n - 2 - np.arange(n)          # positions that give ∂/∂p_i
    grad = 2.0 * corr_full[idx]
    return grad


def _max_conv(p: np.ndarray, fft_size: int, n: int) -> float:
    """Return max(p * p) using FFT (no scaling, sum(p)=1)."""
    conv = _conv_fft(p, fft_size, 2 * n - 1)
    return float(np.max(conv))

        grad = _softmax_grad(p, conv, alpha, fft_size, n)

        # ----- Adam update on the simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps)
        p = _simplex_projection(p)

        # ----- bookkeeping -----
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[Adam] iter {it:6d} – new best: {best_score:.6f}")

        # ----- schedule updates -----
        alpha *= alpha_mul
        lr *= 0.9995               # gentle decay

        # early exit if already very good
        if best_score <= 1.55:
            break

    # ==============================================================
    # 2️⃣  Exact max sub‑gradient with exponentiated updates
    # ==============================================================

    eta = 0.45                   # base step size for exponentiated update
    eta_decay = 0.9997
    sub_it = 0
    max_sub_iters = 300_000

    # start from the best point obtained so far
    p = best_p.copy()
    cur_score = 2.0 * n * _max_conv(p, fft_size, n)

    while time.time() < deadline and sub_it < max_sub_iters:
        sub_it += 1

        # current convolution and the active index
        conv = _conv_fft(p, fft_size, 2 * n - 1)
        k_max = int(np.argmax(conv))

        # ----- sub‑gradient of the true max (active set) -----
        grad = np.zeros_like(p)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        if i_start <= i_end:
            idx_range = np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] = 2.0 * p[k_max - idx_range]

        # ----- exponentiated (mirror‑descent) update -----
        p = p * np.exp(-eta * grad)
        s = p.sum()
        if s == 0.0:
            p = np.full(n, 1.0 / n)
        else:
            p = p / s

        # occasional jitter to avoid stagnation
        if sub_it % 400 == 0:
            noise = rng.normal(scale=0.001, size=n)
            p = _simplex_projection(p + noise)

        # evaluate
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            print(f"[ExpGrad] iter {sub_it:6d} – new best: {best_score:.6f}")

        eta *= eta_decay

        if best_score <= 1.51:
            break

    # ==============================================================
    # 3️⃣  Simulated annealing fine‑tuning
    # ==============================================================

    # start SA from the best point discovered
    p = best_p.copy()
    cur_score = 2.0 * n * _max_conv(p, fft_size, n)

    # annealing parameters
    total_sa_time = deadline - time.time()
    T_start = 0.008                    # initial temperature (tuned empirically)
    step_max = 0.025                   # maximal mass moved in a single proposal

    sa_iters = 0
    while time.time() < deadline:
        sa_iters += 1
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # linear cooling
        T = T_start * (remaining / total_sa_time)

        # ----- propose a mass transfer -----
        # pick source proportionally to its weight (prefer moving from larger masses)
        i = rng.choice(n, p=p)
        if p[i] < 1e-12:
            continue
        # pick a distinct destination uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = rng.random() * min(p[i], step_max)
        if delta <= 0:
            continue

        p_new = p.copy()
        p_new[i] -= delta
        p_new[j] += delta

        # ----- evaluate the move -----
        new_score = 2.0 * n * _max_conv(p_new, fft_size, n)

        # ----- Metropolis acceptance -----
        if new_score < cur_score:
            p = p_new
            cur_score = new_score
        else:
            # accept with probability exp(-(Δ)/T)
            if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                p = p_new
                cur_score = new_score

        # ----- record improvement -----
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[SA] iter {sa_iters:6d} – new best: {best_score:.6f}")

        # occasional random jitter around the best to diversify search
        if sa_iters % 5000 == 0:
            p = best_p + rng.normal(scale=0.003, size=n)
            p = np.clip(p, 0.0, None)
            p = p / p.sum()
            cur_score = 2.0 * n * _max_conv(p, fft_size, n)

    # ------------------------------------------------------------------
    # Final random sampling (use any remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        p_rand = rng.random(n)
        p_rand = p_rand / p_rand.sum()
        score_rand = 2.0 * n * _max_conv(p_rand, fft_size, n)
        if score_rand < best_score:
            best_score = score_rand
            best_p = p_rand.copy()
            print(f"[Random] new best: {best_score:.6f}")
        # quick exit if we are less than a millisecond away from deadline
        if time.time() + 0.01 > deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence (plain Python list, respecting evaluate_sequence)
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Final best score: {best_score:.6f}")
    return result
