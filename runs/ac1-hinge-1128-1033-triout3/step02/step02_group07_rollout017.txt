"""
Optimisation of a non‑negative step function to minimise
evaluate_sequence.  The method combines:

1. A smooth‑max (log‑sum‑exp) formulation of the max‑convolution
   objective and L‑BFGS‑B optimisation on the soft‑max (log‑its)
   representation of the probability vector.  The smoothing parameter β
   is increased geometrically, turning the smooth objective into the
   true max‑convolution.

2. A long simulated‑annealing (Metropolis) stage that performs tiny
   mass‑transfer moves on the simplex while keeping the best sequence
   seen.  Only when a new global best is found is the expensive
   evaluate_sequence() called.

Both phases respect the 1000 s time budget, work on a single CPU and
output a list of non‑negative heights (the sequence) that achieves a
lower evaluation than the previous π/2 ≈ 1.5708 bound.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from functools import partial

# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “square‑root” start (coefficients of (1‑x)^(-1/2))
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient (used by L‑BFGS‑B)
# ----------------------------------------------------------------------
def _smooth_obj_grad(x: np.ndarray,
                     n: int,
                     size_fft: int,
                     beta: float):
    """
    x      – logits (real numbers) of length n.
    n      – length of the sequence.
    size_fft – FFT length (power of two ≥ 3·n‑2).
    beta   – smoothing parameter (β > 0).

    Returns
    -------
    value : float      – 2·n·smooth_max(conv)  (the fast surrogate of the true score)
    grad  : np.ndarray – gradient w.r.t. the logits x
    """
    # ----- soft‑max transformation (a lies on the simplex) -----
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)          # shift for stability
    a = exp_x / exp_x.sum()            # probability vector, sum(a) = 1

    # ----- convolution a * a (linear, not circular) -----
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]   # length 2n‑1

    # ----- smooth max via log‑sum‑exp -----
    conv_max = conv.max()
    # compute exp(beta*(conv - conv_max)) for numerical stability
    exp_beta = np.exp(beta * (conv - conv_max))
    sum_exp_beta = exp_beta.sum()
    smooth_max = conv_max + np.log(sum_exp_beta) / beta
    value = 2.0 * n * smooth_max

    # ----- gradient w.r.t. a (soft‑max probabilities) -----
    # softmax weights over the convolution entries
    s = exp_beta / sum_exp_beta          # ∂smooth_max/∂conv_k

    # grad_a = 2 * (a_rev  *  s)_{n‑1 … n‑1+n‑1}
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, size_fft)
    fs = np.fft.rfft(s, size_fft)
    conv_rev_full = np.fft.irfft(fa_rev * fs, size_fft)   # linear convolution
    start = n - 1
    grad_a = 2.0 * conv_rev_full[start:start + n]

    # ----- chain rule: gradient w.r.t. logits x -----
    dot = np.dot(a, grad_a)                       # scalar = a·grad_a
    grad_x = a * (grad_a - dot)                   # ∂value/∂x  (soft‑max Jacobian)

    return value, grad_x


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises the supplied `evaluate_sequence` metric.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000).
    length : int, optional
        Desired length of the sequence (default 8000).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("length", 8000))

    # ------------------------------------------------------------------
    # Initial vector (try the global variable if it exists)
    # ------------------------------------------------------------------
    try:
        a0 = np.array(height_sequence_1, dtype=float)
        if a0.shape[0] != n:
            a0 = _sqrt_series(n)
    except Exception:
        a0 = _sqrt_series(n)

    # Normalise to the probability simplex (sum = 1)
    a0 = a0 / a0.sum()
    # Log‑its for the soft‑max parametrisation used by L‑BFGS
    x0 = np.log(a0 + 1e-30)

    # ------------------------------------------------------------------
    # FFT size (must be ≥ 3·n‑2 to accommodate a_rev * s convolution)
    # ------------------------------------------------------------------
    size_fft = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate with the cheap FFT based score and,
    # if it improves the current best, call the expensive evaluate_sequence.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def _maybe_update_best(candidate_a: np.ndarray) -> None:
        nonlocal best_seq, best_score
        # fast score (identical to true score because sum(candidate)=1)
        fa = np.fft.rfft(candidate_a, size_fft)
        conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
        fast = 2.0 * n * conv.max()
        if fast + 1e-12 < best_score:            # genuine improvement
            true_val = evaluate_sequence(candidate_a.tolist())
            if true_val < best_score:
                best_score = true_val
                best_seq = candidate_a.copy()
                print(f"[global] new best {best_score:.6f}")

    # initial evaluation
    _maybe_update_best(a0)

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS‑B on the smooth‑max surrogate
    # ------------------------------------------------------------------
    beta = 0.5                     # initial smoothing
    beta_max = 30.0                # final (almost exact) max
    while time.time() < deadline and beta <= beta_max:
        # each stage runs a modest number of L‑BFGS iterations
        max_iter_stage = 200

        # minimise the smooth objective (value, grad) returned by _smooth_obj_grad
        result = minimize(fun=partial(_smooth_obj_grad,
                                      n=n,
                                      size_fft=size_fft,
                                      beta=beta),
                          x0=x0,
                          method='L-BFGS-B',
                          jac=True,
                          options={'maxiter': max_iter_stage,
                                   'ftol': 1e-12,
                                   'gtol': 1e-12,
                                   'disp': False})

        x0 = result.x                     # continue from the last point
        # reconstruct the probability vector from the logits
        max_x = np.max(x0)
        exp_x = np.exp(x0 - max_x)
        a_opt = exp_x / exp_x.sum()

        _maybe_update_best(a_opt)

        # increase the smoothing parameter (geometric schedule)
        beta = min(beta * 2.0, beta_max)

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    # start from the best vector obtained so far (or a0 if none)
    if best_seq is not None:
        a = best_seq.copy()
    else:
        a = a0.copy()

    # current fast score (used for Metropolis decisions)
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    cur_fast = 2.0 * n * conv.max()

    # basic SA schedule
    T_start = 0.25
    stagnation = 0
    iterations = 0

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / budget_s)
        T = T_start * (1.0 - frac)

        # pick two distinct indices
        src = np.random.randint(n)
        if a[src] < 1e-15:
            continue
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # amount of mass to move (proportional to temperature)
        delta = np.random.rand() * a[src] * min(0.4, T)

        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[dst] += delta

        # fast evaluation of the candidate
        fa_c = np.fft.rfft(a_cand, size_fft)
        conv_c = np.fft.irfft(fa_c * fa_c, size_fft)[: 2 * n - 1]
        fast_c = 2.0 * n * conv_c.max()

        # Metropolis acceptance
        if fast_c <= cur_fast or np.random.rand() < math.exp(-(fast_c - cur_fast) / max(T, 1e-12)):
            a = a_cand
            cur_fast = fast_c
            if fast_c + 1e-12 < best_score:
                _maybe_update_best(a)
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # occasional restart if stuck for a long time
        if stagnation > 250000:
            a = a + np.random.randn(n) * 1e-3
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)          # re‑normalise onto the simplex
            fa = np.fft.rfft(a, size_fft)
            conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
            cur_fast = 2.0 * n * conv.max()
            stagnation = 0
            print(f"[restart] SA after {iterations} steps")

        iterations += 1

    # ------------------------------------------------------------------
    # Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = a0
    print(f"[final] best evaluation = {best_score:.6f}")
    return best_seq.tolist()
