def simplex_projection(v):
    # ensures non-negative and sum=1
    # Implementation from literature.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # Based on algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all zeros?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step function that minimises the supplied evaluation
metric.

Algorithm
---------
1. **Broad parametric sweep** – generate many symmetric “U‑shaped’’ and
   exponential families for a variety of lengths (from a few thousand up to
   thirty‑thousand).  For each candidate we compute an *approximate* score
   `2·n·max(a*a)` using an FFT‑based convolution (much faster than the
   official evaluator).  The best candidate is kept as a warm start.

2. **Gradient‑descent refinement on the simplex** – starting from the warm
   start we run many iterations of projected gradient descent with an
   Adam‑style adaptive step.  At each iteration we:
      * compute the self‑convolution via FFT,
      * locate the *k* largest convolution entries (default k=5) and
        form the (sub‑)gradient of the maximum,
      * update the log‑variables with Adam, exponentiate and project back
        onto the probability simplex (ensuring non‑negativity and unit sum).

   The step‑size decays geometrically; the procedure stops when the time
   budget is exhausted.  The best sequence found during the search is
   finally evaluated with the official `evaluate_sequence` function and
   returned as a plain Python list.

The method is deterministic given a seed, but it also performs a few
random restarts if time permits, helping to avoid local minima.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Return the maximum entry of the self‑convolution of `a` using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law “U’’ shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from both edges: a[i] ∝ exp(-α·dist_to_edge)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x >= 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_of_topk_conv(a: np.ndarray, topk: int = 5) -> np.ndarray:
    """
    Compute a sub‑gradient of the maximal entry of the self‑convolution.
    Instead of the single maximum we average the gradients of the
    `topk` largest convolution entries (helps flatten the whole curve).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

    # indices of the `topk` largest entries
    if topk >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(conv, -topk)[-topk:]

    grad = np.zeros_like(a)
    for k in idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        # derivative of (a_i * a_j) w.r.t a_i is a_j, plus symmetric term
        grad[i] += 2.0 * a[j]

    grad /= len(idx)  # average
    return grad

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # leave a small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Seed from a previously supplied construction if available.
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)  # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            ssum = external.sum()
            if ssum < 1e-12:
                external = np.ones_like(external) / external.size
                ssum = 1.0
            external = external / ssum
            approx_score = 2.0 * external.size * max_conv_fft(external)
            if approx_score < best_score:
                best_score = approx_score
                best_seq = external.copy()
                print(f"Seeded from height_sequence_1: approx_score={approx_score:.6f}")
    except NameError:

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over lengths & parametric families.
    # ------------------------------------------------------------------
    # length candidates (balanced: not too small, not too huge)
    lengths = [

        12000, 15000, 18000, 20000, 25000, 30000,

    # more densely sampled exponents / decay rates
    p_vals = np.linspace(0.2, 2.5, 45)          # 0.2 … 2.5

    print("=== Parametric sweep ===")
    for n in lengths:

            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:

    # ------------------------------------------------------------------

    n = current.shape[0]

    beta2 = 0.999

    lr_base = 0.8                # initial learning rate (on log‑scale)

    v = np.zeros_like(current)

    while time.time() < deadline:
        t_iter += 1

        # --- compute gradient of the (approximated) max convolution ---
        grad = gradient_of_topk_conv(current, topk=topk)

        # --- Adam update on the *log* variables (to keep positivity) ---
        #   log_a = log(current + eps)  (we keep eps to avoid log(0))
        log_a = np.log(current + 1e-12)

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
