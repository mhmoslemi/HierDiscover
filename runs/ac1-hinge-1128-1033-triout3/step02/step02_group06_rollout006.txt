"""
propose_candidate
-----------------

Iterative improvement of a non‑negative step‑function using the LP‑based
procedure described in the challenge statement.

1.  Start with a “central‑binomial” sequence (the analytically optimal
    construction) of a modest length *n* (default 200).  It is scaled so
    that the sum of the coefficients equals √(2 n); with this normalization
    the evaluation function reduces to the maximal self‑convolution
    ‑‑ the quantity we want to shrink.

2.  At each iteration solve the linear program

        maximise   Σ g_j
        subject to   (f * g)_k ≤ M   for all convolution indices k
                     g_j ≥ 0

    where *M* = max(f * f) and *f* is the current sequence.  The constraints
    are linear in the unknown vector *g* (they are just weighted sums of the
    components of *g*).  After solving we scale *g* back to the same
    normalisation (sum = √(2 n)) and replace

        f ← (1‑t) f + t g               (t = 0.2)

    which provably reduces the maximal self‑convolution.  The loop stops
    when no further improvement is observed or the time budget runs out.

3.  Finally a tiny stochastic hill‑climb (Gaussian perturbations of a
    single coefficient) is performed for the remaining time; this can only
    improve the score.

The whole routine respects the 1000 s budget, works with `numpy` and
`scipy.optimize.linprog`, and returns a list of non‑negative floats ready
to be fed to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# SciPy is allowed in the environment.  If it is not present we fall back
# to the simple random perturbation version (the previous approach).
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _central_binomial_prefix(n: int) -> np.ndarray:
    """Return the first n central‑binomial coefficients:
       b_k = C(2k, k) / 4**k  (k = 0,…,n‑1).
    """
    b = np.empty(n, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, n):
        # recurrence b_k = b_{k‑1} * (2k‑1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        b[k] = cur
    return b


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2n‑1, n)) such that
        (A @ g)[k] = (f * g)[k]   for all k.
    The k‑th row contains the coefficients f_i that multiply g_{k-i}
    (zero when the index is out of range).
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)

    for k in range(m):
        # i runs over indices of f that are paired with a valid g entry.
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # vectorised fill of the row
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        A[k, j_range] = f[i_range]
    return A


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     max_len: int = 200,
                     **kwargs) -> list[float]:
    """
    Generate a step‑function sequence that (heuristically) minimises the
    evaluation function `evaluate_sequence`.  The algorithm uses the
    LP‑based improvement described above and a short stochastic refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 0. Choose a working length (n).  Larger n gives a better asymptotic
    #    bound but makes the linear program heavier.  200 is a good
    #    compromise for the 1000 s budget.
    # ------------------------------------------------------------------
    n = int(max_len)
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Initialise f with the central‑binomial prefix and normalise.
    # ------------------------------------------------------------------
    try:
        init = _central_binomial_prefix(n)
    except Exception as e:                         # pragma: no cover
        raise RuntimeError("Failed to compute central‑binomial coefficients") from e

    sum_init = init.sum()
    f = init * (target_sum / sum_init)             # now Σ f = √(2n)

    # ------------------------------------------------------------------
    # 2. Main LP‑based improvement loop.
    # ------------------------------------------------------------------
    best_seq = f.copy()
    best_score = float('inf')
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:                              # pragma: no cover
        pass

    # Variables that control the loop
    t = 0.2                                         # mixing factor
    max_iter = 40
    tol = 1e-12

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # Compute current maximal self‑convolution (M)
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # Build the convolution matrix for the constraints (f * g)_k ≤ M
        if not _SCIPY_AVAILABLE:
            # If SciPy is missing we cannot run the LP – fall back to random search.
            break

        A = _build_convolution_matrix(f)
        m = A.shape[0]

        # Linear programme: maximise Σ g_j  (i.e. minimise –Σ g_j)
        c = -np.ones(n, dtype=np.float64)           # objective vector for linprog (minimise)
        b_ub = np.full(m, M, dtype=np.float64)      # right‑hand side for all constraints

        # Solve the LP
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

        if not res.success:
            # No feasible improvement – break the loop.
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 < tol:
            break

        # Rescale g0 so that its sum matches the prescribed normalisation.
        g = g0 * (target_sum / sum_g0)

        # Convex combination of the old and the new sequence.
        f_new = (1.0 - t) * f + t * g

        # Evaluate the new candidate.
        try:
            score_new = evaluate_sequence(f_new.tolist())
        except Exception:                         # pragma: no cover
            # If the evaluator crashes we skip this iteration.
            continue

        # Update the best known solution.
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_seq = f_new.copy()
            print(f"[iter {it+1}] improvement: score={best_score:.9f}")

        # Decide whether to accept the update.
        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()
        # Accept only if the maximal self‑convolution decreased (or the score improved).
        if M_new < M - 1e-12 or score_new < best_score:
            f = f_new
        else:
            # No progress – reduce the mixing factor to try a finer step.
            t *= 0.5
            if t < 1e-3:
                break

    # ------------------------------------------------------------------
    # 3. Very light stochastic hill‑climb for any remaining time.
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    sigma = 0.03                        # relative Gaussian step size
    no_improve = 0
    max_no_improve = 2000

    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(n)
        old_val = cur_seq[idx]
        # Gaussian perturbation proportional to the current magnitude.
        delta = np.random.randn() * sigma * max(old_val, 1e-12)
        new_val = max(0.0, old_val + delta)

        cand = cur_seq.copy()
        cand[idx] = new_val

        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        if cand_score < cur_score - 1e-12:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[hill] new best {best_score:.9f}")
        else:
            no_improve += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as plain Python floats).
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, x)) for x in best_seq]
    # Final sanity check (optional)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] length={len(final_seq)} score={final_score:.9f}")
    except Exception:                             # pragma: no cover
        pass
    return final_seq
