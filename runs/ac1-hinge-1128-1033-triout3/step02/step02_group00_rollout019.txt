    n = a.shape[0]
    size = 1 << ((L + n - 2).bit_length())
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fw * np.conj(fa), size)[:L + n - 1]
    # corr[k] = sum_i w[i] * a[k - i] with zero padding
    # we need grad[j] = sum_i w[i + j] * a[i] = corr[j + (something)]?

    # With zero padding, a[k - i] means a at index... Let's derive proper index.

    # Using definition of linear correlation:
    # (w ⋆ a)[lag] = Σ_i w[i + lag] a[i]
    # This can be computed by: pad w and a appropriately and use FFT.
    # Implementation: compute convolution of w (reversed?) Wait typical cross-correlation can be computed as:

    # With zero padding, for lag >=0, the sum is as desired.

    # So corr[l] corresponds to sum_i w[i + l] a[i], where out-of-range indices are treated as zero.
    # So gradient array for l=0..n-1 is exactly corr[l].

"""
Gradient‑based optimisation of the step–function heights.

The objective to minimise is
    E(a) = 2·n·max(conv(a,a)) / (sum(a)^2) .
Because the denominator is homogeneous of degree 2 we can normalise the
vector to a probability simplex (sum = 1) and minimise
    F(a) = 2·n·max(conv(a,a)) .

We use a smooth approximation of the max:
    S_β(a) = (1/β)·log Σ_k exp(β·conv_k(a)) .
Grad S_β is a convex combination of the gradients of the individual
convolution terms, i.e. a weighted correlation with the softmax weights.
An Adam optimiser is employed on this smooth objective, with a linear
schedule for the smoothing parameter β (starting small, ending ≈ 15)
and a decaying step‑size.  After each gradient step the iterate is
projected back onto the simplex using the O(n log n) algorithm of
Duchi et al.  Occasionally a tiny random mass transfer and periodic
jitter are applied to escape plateaus.

The routine runs until the time budget expires and returns the best
sequence found (as a plain list of floats) which can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: full linear convolution via real FFT (fast for many iterations)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return the linear convolution a*a of length 2·n‑1 using FFT of length *size*."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negative round‑off
    return conv


# ----------------------------------------------------------------------
# Helper: correlation needed for the gradient of the smooth max
# ----------------------------------------------------------------------
def _correlation(p: np.ndarray, a: np.ndarray, size: int) -> np.ndarray:
    """
    Linear correlation (w ⋆ a)[ℓ] = Σ_i p[i+ℓ]·a[i] for ℓ≥0.
    The result is returned for ℓ = 0 … n‑1 (first n lags).
    """
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fp * np.conj(fa), size)[: len(p) + len(a) - 1]
    return corr[: a.shape[0]]   # first n entries are the needed lags


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step sequence with Adam on a smoothed version of
    the max self‑convolution objective.  The best sequence encountered before
    the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------
    # Choose length n and initialise the vector a
    # ------------------------------------------------------------

        a = np.asarray(height_sequence_1, dtype=float)   # use previous best if available
        n = a.shape[0]
    except Exception:
        n = 5000                                          # a moderately large length
        a = np.full(n, 1.0 / n)                           # uniform start

    # Normalise to the probability simplex
    a = a / a.sum()
    a = _project_onto_simplex(a)                          # defensive projection

    # ------------------------------------------------------------
    # Pre‑compute FFT size (power of two) sufficient for both
    # convolution (length 2n‑1) and correlation (length 3n‑2)
    # ------------------------------------------------------------
    fft_size = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------
    # Adam state initialisation
    # ------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # ------------------------------------------------------------
    # Scheduling parameters
    # ------------------------------------------------------------
    lr_start = 0.5                # initial learning rate
    lr_end = 1e-4                 # final learning rate
    beta_smooth_start = 0.5       # initial smooth‑max temperature
    beta_smooth_end = 15.0        # final temperature (close to true max)

    # ------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------
    conv = _full_convolution(a, fft_size)
    cur_max = conv.max()
    cur_score = float(2.0 * n * cur_max)
    best_a = a.copy()
    best_score = cur_score
    iteration = 0
    last_improve = start_time

    print(f"[init] n={n}, initial evaluation={cur_score:.6f}")

    # ------------------------------------------------------------
    # Optimisation loop
    # ------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------
        # Schedule updates (linear in elapsed time)
        # --------------------------------------------------------
        progress = (time.time() - start_time) / (deadline - start_time)
        lr = lr_start * (1.0 - progress) + lr_end * progress
        beta_smooth = beta_smooth_start + (beta_smooth_end - beta_smooth_start) * progress

        # --------------------------------------------------------
        # Convolution and true objective (used for bookkeeping)
        # --------------------------------------------------------
        conv = _full_convolution(a, fft_size)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{iteration:7d}] new best = {best_score:.6f}")

        # --------------------------------------------------------
        # Smooth approximation of the max via soft‑max weights
        # --------------------------------------------------------
        shifted = conv - cur_max                           # for numerical stability
        exp_vals = np.exp(beta_smooth * shifted)
        p = exp_vals / exp_vals.sum()                     # soft‑max over all bins

        # --------------------------------------------------------
        # Gradient of the smoothed objective: 2·corr(p, a)
        # --------------------------------------------------------
        grad = 2.0 * _correlation(p, a, fft_size)

        # --------------------------------------------------------
        # Adam update
        # --------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        a_tmp = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------
        # Occasional random mass‑transfer to aid exploration
        # --------------------------------------------------------
        if np.random.rand() < 0.01:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = np.random.rand() * a_tmp[i] * 0.4
            a_tmp[i] -= delta
            a_tmp[j] += delta

        # --------------------------------------------------------
        # Projection back onto the simplex
        # --------------------------------------------------------
        a = _project_onto_simplex(a_tmp)

        # --------------------------------------------------------
        # Periodic jitter when progress stalls
        # --------------------------------------------------------
        if iteration % 20000 == 0:
            # blend a tiny random Dirichlet component (≈1 % total mass)
            noise = np.random.rand(n)
            noise = noise / noise.sum() * 0.01
            a = a * 0.99 + noise
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)

    # ------------------------------------------------------------
    # Final reporting using the supplied evaluation routine
    # ------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
