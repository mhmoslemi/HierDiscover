"""
propose_candidate
-----------------
Generate a step‑function whose coefficients are the classic “constant‑convolution”
sequence a_k = C(2k,k) / 4**k.  For any prefix of this sequence the self‑convolution
is exactly 1 on the first ``len(prefix)`` positions, so the evaluation function
reduces to

    score = 2·n / ( Σ a_k )² .

The ratio is asymptotically π/2 ≈ 1.570796, and it decreases monotonically with
the length of the prefix.  The routine below builds the coefficients as fast as
possible and keeps extending the prefix until the wall‑clock budget is about to
expire.  The longest prefix obtained therefore gives the smallest (best) score
the algorithm can achieve within the allotted time.
"""

import time
import numpy as np


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (almost) attains the
    theoretical lower bound π/2 for the evaluation function.

    Parameters
    ----------
    seed : int, optional
        Unused – kept for API compatibility.
    budget_s : int, optional
        Total wall‑clock budget (seconds).  The function stops early if the
        remaining time is less than a small safety margin.
    **kwargs : dict
        Ignored – present only for compatibility with the harness.

    Returns
    -------
    list[float]
        The chosen step‑function.
    """
    # Leave a small safety margin so that the harness has time to finish
    # its own bookkeeping after we return.
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Phase 1 – generate the constant‑convolution coefficients.
    # ------------------------------------------------------------------
    # The recurrence a_k = a_{k-1} * (2k-1)/(2k) generates the exact values
    # without ever forming huge intermediate integers.
    coeffs = []                # list of a_k
    cur = 1.0                  # a_0 = 1
    coeffs.append(cur)

    # We'll keep producing terms until we run out of time or hit a
    # hard cut‑off that keeps the final convolution tractable.
    # Empirically, n ≈ 5·10⁴ still lets NumPy's convolution finish in a
    # reasonable amount of time on the evaluation side.
    HARD_MAX = 60_000

    k = 1                      # next index to generate
    while True:
        # Stop if we are too close to the deadline.
        if time.time() > deadline:
            break
        if k >= HARD_MAX:
            break

        # Recurrence: a_k = a_{k-1} * (2k-1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs.append(cur)
        k += 1

    n = len(coeffs)

    # ------------------------------------------------------------------
    # Phase 2 – compute the analytic score for the final prefix.
    # ------------------------------------------------------------------
    # The convolution bound is exactly 1, so we can evaluate the objective
    # analytically without calling ``evaluate_sequence``.
    total = float(np.sum(coeffs))
    analytic_score = 2.0 * n / (total * total)

    # ------------------------------------------------------------------
    # Phase 3 – final sanity‑check and return.
    # ------------------------------------------------------------------
    print(f"[result] length={n}, analytic_score={analytic_score:.10f}")

    # Apply the same clamping rules that ``evaluate_sequence`` does.
    # (All our coefficients are already in [0,1], but we keep the step
    # for completeness and future‑proofing.)
    final_seq = [float(max(0.0, min(1000.0, x))) for x in coeffs]

    return final_seq
