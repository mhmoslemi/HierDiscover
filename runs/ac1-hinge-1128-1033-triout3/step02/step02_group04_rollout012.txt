"""
Search for a low‑objective step‑function using exponentiated‑gradient
optimisation on a smooth approximation of the max of the self‑convolution.
The method works on the probability simplex (∑a_i = 1, a_i ≥ 0) and
includes periodic jitter, restarts and a cheap pair‑wise local refinement.
It is deliberately different from the previously‑shown Adam‑based optimiser.
"""

import time
import numpy as np


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Returns a list of non‑negative heights that (hopefully) minimises
    evaluate_sequence().  The search runs for at most ``budget_s`` seconds
    and is fully self‑contained – no file I/O, no external solvers.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Choose a problem size and an initial vector.
    # ------------------------------------------------------------------
    try:
        # If a previous construction is present we reuse it as a warm start.
        init_seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        total = init_seq.sum()
        if total > 0.0:
            p0 = init_seq / total                     # normalise to a probability vector
        else:
            raise ValueError
        n = len(p0)
    except Exception:
        # No historic data – fall back to a moderate size (power‑of‑two is convenient).
        n = 2048
        p0 = np.full(n, 1.0 / n)

    # FFT length – next power of two that can hold a linear convolution of length 2·n‑1.
    m_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2.  Helper to evaluate the objective quickly (avoids the full
    #     security checks of evaluate_sequence).
    # ------------------------------------------------------------------
    def fast_score(p_vec: np.ndarray) -> float:
        """2·n·max(p*p) where * denotes linear convolution."""
        p_fft = np.fft.rfft(p_vec, n=m_fft)
        conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # 3.  Keep the best solution found so far.
    # ------------------------------------------------------------------
    best_p = p0.copy()
    best_score = fast_score(best_p)
    print(f"Initial score (seed): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Hyper‑parameters.
    # ------------------------------------------------------------------
    base_lr = 0.6          # initial learning rate for the exponentiated‑gradient step
    lr_decay = 0.99995     # geometric decay per iteration
    alpha = 1.0            # smoothing for the log‑sum‑exp approximation of max
    alpha_mul = 1.02
    alpha_max = 3000.0

    jitter_interval = 800
    jitter_scale = 0.001

    max_no_improve = 25000          # restart if no improvement for this many iters
    max_iters_per_trial = 50000     # cap per restart

    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 5.  Main optimisation – many short restarts with exponentiated‑gradient.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 5.1  Initialise a fresh candidate.
        # --------------------------------------------------------------
        if best_score < 2.0 and rng.random() < 0.3:
            # Perturbed version of the current best (occasionally)
            p = best_p + rng.normal(scale=0.01, size=n)
            p = np.clip(p, 0.0, None)
            p /= p.sum()
        else:
            # Random Dirichlet (concentration < 1 encourages sparsity)
            conc = rng.random() * 1.0 + 0.3      # ∈ [0.3, 1.3]
            p = rng.dirichlet(np.full(n, conc))

        # Work in the log‑domain – the classical exponentiated‑gradient update.
        log_p = np.log(p + 1e-32)

        cur_lr = base_lr
        cur_alpha = alpha
        it_no_improve = 0

        # --------------------------------------------------------------
        # 5.2  Inner optimisation loop.
        # --------------------------------------------------------------
        for it in range(1, max_iters_per_trial + 1):
            if time.time() >= deadline:
                break

            # ----- recover the probability vector from the log‑variables.
            p = np.exp(log_p - np.max(log_p))
            p /= p.sum()

            # ----- forward FFT of the current candidate.
            p_fft = np.fft.rfft(p, n=m_fft)

            # ----- linear self‑convolution.
            conv_self = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

            # ----- smooth approximation of the max (log‑sum‑exp).
            z = cur_alpha * conv_self
            max_z = z.max()
            exp_z = np.exp(z - max_z)                # stability trick
            w = exp_z / exp_z.sum()                  # soft‑max weights, sum = 1

            # ----- gradient of the smoothed objective w.r.t. p.
            w_rev = w[::-1]
            w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
            corr_full = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 2 * n - 1]
            idx = 2 * n - 2 - np.arange(n)            # mapping back to gradient indices
            grad = 2.0 * corr_full[idx]

            # ----- exponentiated‑gradient step on the log‑variables.
            log_p -= cur_lr * grad

            # ----- occasional jitter to escape plateaus.
            if it % jitter_interval == 0:
                jitter = rng.normal(scale=jitter_scale, size=n)
                p = np.exp(log_p - np.max(log_p))
                p = np.clip(p + jitter, 0.0, None)
                p /= p.sum()
                log_p = np.log(p + 1e-32)

            # ----- evaluate the current objective (uses the *old* conv_self).
            cur_score = float(2.0 * n * conv_self.max())

            # ----- keep the best solution ever seen.
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                it_no_improve = 0
                print(f"Iter {it:6d} – new global best: {best_score:.6f}")
                if best_score <= 1.503:   # target reached → stop early
                    break
            else:
                it_no_improve += 1

            # ----- schedule updates.
            cur_lr *= lr_decay
            cur_alpha = min(cur_alpha * alpha_mul, alpha_max)

            # ----- restart if we have stagnated for a while.
            if it_no_improve >= max_no_improve:
                break

        if best_score <= 1.503:
            break   # early exit – we already beat the target

    # ------------------------------------------------------------------
    # 6.  Tiny local refinement (pairwise mass transfers).
    # ------------------------------------------------------------------
    p = best_p.copy()
    p_fft = np.fft.rfft(p, n=m_fft)
    conv_self = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]
    cur_score = float(2.0 * n * conv_self.max())

    swap_attempts = 5000
    for _ in range(swap_attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 5 % of the source mass
        max_move = min(p[i], 0.05 * p[i])
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        new_p = p.copy()
        new_p[i] -= delta
        new_p[j] += delta

        # evaluate new candidate
        new_fft = np.fft.rfft(new_p, n=m_fft)
        new_conv = np.fft.irfft(new_fft * new_fft, n=m_fft)[: 2 * n - 1]
        new_score = float(2.0 * n * new_conv.max())

        if new_score + 1e-12 < cur_score:
            p = new_p
            cur_score = new_score
            best_p = p.copy()
            best_score = cur_score
            print(f"Swap improvement → {best_score:.6f}")

    print(f"Final best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Return the solution (values clipped to the limits enforced by
    #     evaluate_sequence).
    # ------------------------------------------------------------------
    return np.clip(best_p, 0.0, 1000.0).tolist()
