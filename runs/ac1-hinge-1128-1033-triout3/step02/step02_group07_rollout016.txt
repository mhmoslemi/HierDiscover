"""
Propose a step‑function sequence that minimises the given
`evaluate_sequence` score.

The optimiser works in three stages:

1. **Parametric sweep** – search a tiny family of power‑law
   sequences a_i ∝ (i+δ)^‑p (p≈0.5).  The sweep is cheap (only FFTs)
   and quickly finds a better starting point than the plain
   (1‑x)^‑1/2 series.

2. **Smoothed Adam / exponentiated‑gradient descent** – minimise a
   smooth approximation of the max‑convolution
   f(a)=max_k (a∗a)_k using the soft‑max (log‑sum‑exp) surrogate.
   Gradients are obtained by FFTs and the update is performed in the
   log‑space (multiplicative) with Adam‑style momentum, followed by
   projection onto the probability simplex (∑a_i=1, a_i≥0).

3. **Fine‑grained simulated‑annealing** – once the smooth optimisation
   stalls a tiny Metropolis walk transfers a random amount of mass
   between two coordinates.  Improvements are accepted greedily,
   otherwise with a temperature‑dependent probability.

The whole routine respects the 1000 s budget, calls the expensive
`evaluate_sequence` only when a new global best is discovered, and
returns the best sequence found (as a plain Python list).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the objective (the same quantity that
# evaluate_sequence returns when the vector is normalised to sum=1)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, size_fft: int) -> float:
    """Return 2·n·max(a∗a) using an FFT of length `size_fft`."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Generate a normalised power‑law vector a_i ∝ (i+δ)^‑p
# ----------------------------------------------------------------------
def _power_law_seq(n: int, p: float, delta: float) -> np.ndarray:
    """Return the L1‑normalised vector (i+delta)^{-p}, i=0..n‑1."""
    idx = np.arange(n, dtype=np.float64)
    w = (idx + delta) ** (-p)
    # avoid infinite or NaN when delta=0 and i=0 -> set first entry manually
    if delta == 0.0:
        w[0] = 1.0   # (0)^-p is infinite; we cap it to 1 (the exact scale will be fixed by normalisation)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Soft‑max (log‑sum‑exp) gradient of the max‑convolution
#   g = ∂/∂a ( (1/α) log Σ_k exp(α·b_k) )
# with b = a∗a.
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray,
                     size_fft_a: int,
                     size_fft_g: int,
                     alpha: float) -> (np.ndarray, float):
    """
    Compute the gradient of the soft‑max approximation of max(b) and the
    current (unsmoothed) max value.  Returns (grad, max_b).
    """
    n = a.shape[0]

    # ----- b = a ∗ a  (full length 2n-1) -----
    fa = np.fft.rfft(a, size_fft_a)
    b_full = np.fft.irfft(fa * fa, size_fft_a)[: 2 * n - 1]

    # ----- soft‑max weights w_k  -----
    # stability: subtract the max before exponentiation
    max_alpha_b = np.max(alpha * b_full)
    w = np.exp(alpha * b_full - max_alpha_b)
    w_sum = w.sum()
    w /= w_sum                     # Σ_k w_k = 1

    # ----- gradient  g = 2 * (a ∗ w_rev)  -----
    # w_rev is w reversed; the convolution gives exactly the required
    # linear combination of the shifted a entries.
    w_rev = w[::-1]
    fa2 = np.fft.rfft(a, size_fft_g)
    fw2 = np.fft.rfft(w_rev, size_fft_g)
    conv_full = np.fft.irfft(fa2 * fw2, size_fft_g)
    # we need the slice (2n‑2)‑j for j=0..n‑1
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv_full[idx]

    max_b = b_full.max()
    return grad, max_b


# ----------------------------------------------------------------------
# Adam‑style optimisation in log‑space (multiplicative updates)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   time_deadline: float,
                   size_fft_a: int,
                   size_fft_g: int) -> (np.ndarray, float):
    """
    Run a smoothed Adam optimiser starting from a0.
    Returns the best vector seen (projected onto the simplex) and its
    fast_score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _fast_score(a, size_fft_a)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.12           # initial learning‑rate (in log‑space)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 2.0
    alpha_target = 30.0
    alpha_inc = (alpha_target / alpha) ** (1.0 / 5000)   # exponential growth

    it = 0
    while time.time() < time_deadline:
        # gradient of the current smooth objective
        grad, max_b = _smooth_gradient(a, size_fft_a, size_fft_g, alpha)

        # Adam moments (on the gradient itself)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # multiplicative (exponentiated‑gradient) update in log‑space
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a = np.log(a + eps) - step
        a = np.exp(log_a)

        # projection back onto the simplex (keeps sum=1, a≥0)
        a = _project_simplex(a)

        # record best if improved (fast_score uses exact max, not the smoothed one)
        cur_val = _fast_score(a, size_fft_a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[adam] iter {it:06d}  fast_score {cur_val:.6f}")

        # schedule updates
        alpha = min(alpha_target, alpha * alpha_inc)
        # slowly decay learning rate to stabilise
        lr = lr * 0.9999

        it += 1
        if it % 500 == 0 and time.time() >= time_deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine tuner (tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray,
                         time_deadline: float,
                         size_fft: int) -> (np.ndarray, float):
    """
    Perform a Metropolis walk that transfers mass between two random
    coordinates.  The temperature linearly cools to zero.
    Returns the best vector found and its fast_score.
    """
    a = a_start.copy()
    n = a.shape[0]
    cur_score = _fast_score(a, size_fft)
    best_a = a.copy()
    best_score = cur_score

    # SA schedule
    t_start = 0.12
    iter_cnt = 0
    no_improve = 0

    while time.time() < time_deadline:
        # linear cooling
        frac = (time.time() - (time_deadline - (time_deadline - time.time()))) / (time_deadline - time.time())
        T = t_start * max(0.0, 1.0 - frac)

        # propose a tiny transfer
        src = np.random.randint(n)
        if a[src] < 1e-15:
            continue
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        delta = np.random.rand() * a[src] * min(0.4, T)   # move at most 40 % of source when hot

        a_cand = a.copy()
        a_cand[src] -= delta
        a_cand[dst] += delta

        cand_score = _fast_score(a_cand, size_fft)

        # Metropolis acceptance
        if cand_score <= cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            a = a_cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                no_improve = 0
                print(f"[sa]  iter {iter_cnt:07d}  new best {best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random restart to escape stagnation
        if no_improve > 25000:
            a = _project_simplex(np.random.rand(n))
            cur_score = _fast_score(a, size_fft)
            no_improve = 0
            print(f"[sa] restart after {iter_cnt} iterations")

        iter_cnt += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that minimises the
    supplied `evaluate_sequence` function.

    The routine combines a fast parametric sweep, a smoothed Adam
    optimiser and a short simulated‑annealing phase.  It works on a
    single length (default 8000) and respects the total time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("length", 8000))

    # Pre‑compute FFT lengths
    size_fft_a = 1 << ((2 * n - 1).bit_length())          # for a∗a
    size_fft_g = 1 << ((3 * n - 2).bit_length())          # for gradient (a∗w_rev)

    # ------------------------------------------------------------------
    # Helper to keep the globally best sequence (uses the real evaluate)
    # ------------------------------------------------------------------
    best_global_seq = None
    best_global_score = float("inf")

    def _maybe_update_global(candidate: np.ndarray, fast_val: float) -> None:
        nonlocal best_global_seq, best_global_score
        if fast_val < best_global_score - 1e-9:
            true_val = evaluate_sequence(candidate.tolist())
            if true_val < best_global_score:
                best_global_score = true_val
                best_global_seq = candidate.copy()
                print(f"[global] NEW BEST {best_global_score:.6f}")

    # ------------------------------------------------------------------
    # 1) PARAMETRIC SWEEP
    # ------------------------------------------------------------------
    best_local = None
    best_local_score = float("inf")

    # try a few families: varying power p and delta
    p_vals = np.linspace(0.35, 0.65, 7)          # around 0.5
    delta_vals = np.linspace(0.0, 5.0, 11)      # shift away from the singularity

    for p in p_vals:
        for d in delta_vals:
            if time.time() >= deadline:
                break
            a = _power_law_seq(n, p, d)
            # fast evaluation (exact max, not the soft‑max)
            score = _fast_score(a, size_fft_a)
            if score < best_local_score:
                best_local_score = score
                best_local = a.copy()
                _maybe_update_global(a, score)

    # ------------------------------------------------------------------
    # If a global sequence is already available as a warm start use it
    # ------------------------------------------------------------------
    try:
        gl_seq = np.array(height_sequence_1, dtype=float)
        if gl_seq.shape[0] == n:
            gl_seq = _project_simplex(gl_seq)
            # evaluate
            gscore = _fast_score(gl_seq, size_fft_a)
            if gscore < best_local_score:
                best_local_score = gscore
                best_local = gl_seq.copy()
                _maybe_update_global(gl_seq, gscore)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) SMOOTH ADAM OPTIMISATION (log‑space, multiplicative updates)
    # ------------------------------------------------------------------
    if best_local is None:
        best_local = _project_simplex(np.full(n, 1.0 / n))

    # Allocate roughly half of the remaining budget to the Adam phase
    remaining = deadline - time.time()
    adam_deadline = time.time() + remaining * 0.55
    best_adam, best_adam_score = _adam_optimize(best_local,
                                                adam_deadline,
                                                size_fft_a,
                                                size_fft_g)
    _maybe_update_global(best_adam, best_adam_score)

    # ------------------------------------------------------------------
    # 3) SIMULATED ANNEALING REFINEMENT
    # ------------------------------------------------------------------
    # Use the remaining time for a short SA walk
    if time.time() < deadline:
        sa_deadline = deadline
        best_sa, best_sa_score = _simulated_annealing(best_adam, sa_deadline, size_fft_a)
        _maybe_update_global(best_sa, best_sa_score)

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------
    if best_global_seq is None:
        # fallback – uniform distribution (always feasible)
        best_global_seq = np.full(n, 1.0 / n)
    print(f"[final] overall best evaluation = {best_global_score:.6f}")
    return best_global_seq.tolist()
