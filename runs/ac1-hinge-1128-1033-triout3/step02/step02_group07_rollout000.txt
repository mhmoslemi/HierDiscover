"""
Propose a step‑function (non‑negative sequence) that minimizes the
`evaluate_sequence` metric.

Algorithm overview
------------------

The metric is invariant under scaling, therefore we keep the vector on the
probability simplex (non‑negative, sum = 1).  The objective is
    f(a) = max_k (a∗a)_k .
We minimise a smooth approximation of the maximum using a
log‑sum‑exp (soft‑max) with a temperature parameter α.  The optimisation
variables are the logits `s` such that a = softmax(s).

For a given α the smoothed objective is
    Φ_α(a) = (1/α)·log Σ_k exp(α·c_k)    where c = a∗a .
Its gradient w.r.t. a is
    ∇_a Φ_α = 2·(a ⋆ p)          (⋆ = cross‑correlation)
where p_k = exp(α·c_k) / Σ_j exp(α·c_j) is a probability vector
derived from the soft‑max.

Using the mirror map of the negative entropy we obtain the gradient on the
logits:
    ∇_s Φ_α = a ⊙ (∇_a Φ_α – ⟨a,∇_a Φ_α⟩)          (⊙ element‑wise)

We perform a projected gradient descent on s with a linearly decreasing
step size, while α is increased linearly from a smooth regime (α≈2) to a
sharp regime (α≈300) – a continuation method that helps escape bad local
minima.  All convolutions are evaluated with an FFT of length
≈3·n (linear convolution of a (length n) with the weight vector (length 2n‑1)).

During the search we keep the best sequence seen so far (according to the
fast evaluation `2·n·max(a∗a)`) and call the expensive `evaluate_sequence`
only when a new global best is found.  The routine works for any chosen
length; we try a few lengths, allocating the total time budget equally.

The implementation respects the 1000 s budget, uses only NumPy, and
contains no I/O other than occasional `print` statements for progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^(-1/2), normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


def _fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(a∗a) using an FFT of length fft_len (identical to evaluate_sequence for sum=1)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max()

# Core optimisation for a fixed length
# ----------------------------------------------------------------------
def _optimise_length(
    n: int,
    time_deadline: float,
    init_vec: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector of length n under the given time budget.
    Returns (best_vector, best_fast_score).
    """
    # FFT length for linear convolution a∗a (size 2n‑1)
    fft_len_conv = 1 << ((2 * n - 1).bit_length())
    # FFT length for gradient convolution a⋆p (size 3n‑2)
    fft_len_grad = 1 << ((3 * n - 2).bit_length())

    # Initialise a on the simplex
    if init_vec is not None and init_vec.shape[0] == n:
        a = np.maximum(init_vec, 0.0)
        a = _project_to_simplex(a)
    else:
        a = _sqrt_series(n)

    # Logits for the entropic mirror map
    s = np.log(a + 1e-30)

    # Parameters for continuation / step‑size schedules
    lr_start = 0.8               # learning rate at start
    lr_end = 0.02                # learning rate at end
    alpha_start = 2.0            # smoothness at start
    alpha_end = 300.0            # sharpness at end
    max_iter = 200_000           # safety cap
    eval_interval = 200          # how often we evaluate the fast score
    stall_limit = 5_000          # restart after this many iterations without improvement

    best_local_a = a.copy()
    best_local_score = _fast_score(a, fft_len_conv)

    last_improve_iter = 0
    start_time = time.time()
    for it in range(max_iter):
        # time check
        if time.time() >= time_deadline:
            break

        # fraction of allocated time elapsed (0→1)
        frac = (time.time() - start_time) / (time_deadline - start_time)

        # schedules
        lr = lr_start * (1.0 - frac) + lr_end * frac
        alpha = alpha_start + frac * (alpha_end - alpha_start)

        # ------- compute convolution a * a -------
        Fa = np.fft.rfft(a, fft_len_conv)
        conv_full = np.fft.irfft(Fa * Fa, fft_len_conv)[: 2 * n - 1]
        conv_max = conv_full.max()

        # ----- fast evaluation & best‑track -----
        if it % eval_interval == 0:
            fast_score = 2.0 * n * conv_max
            if fast_score < best_local_score - 1e-12:
                best_local_score = fast_score
                best_local_a = a.copy()
                last_improve_iter = it

        # ------- soft‑max over convolution entries -------
        # w_k = exp(alpha * (c_k - max(c)))  avoids overflow
        shifted = conv_full - conv_max
        w = np.exp(alpha * shifted)
        p = w / w.sum()                     # p is a probability vector, length 2n‑1

        # ------- gradient w.r.t a (cross‑correlation a ⋆ p) -------
        # we need convolution of a with reversed p
        p_rev = p[::-1]
        Fa = np.fft.rfft(a, fft_len_grad)
        Fp = np.fft.rfft(p_rev, fft_len_grad)
        corr = np.fft.irfft(Fa * Fp, fft_len_grad)  # linear convolution result
        # gradient_i = 2 * corr[2n‑2‑i]   for i=0…n‑1
        idx = 2 * n - 2 - np.arange(n)
        grad_a = 2.0 * corr[idx]

        # ------- mirror‑descent update on logits -------
        g_bar = np.dot(a, grad_a)
        ds = a * (grad_a - g_bar)          # ∇_s Φ_α
        s = s - lr * ds
        # recompute a = softmax(s)
        a = _softmax(s)

        # ------- stall detection – random restart -------
        if it - last_improve_iter > stall_limit:
            # perturb the current best a a little and continue
            perturb = np.random.randn(n) * 0.02
            s = np.log(best_local_a + 1e-30) + perturb
            a = _softmax(s)
            last_improve_iter = it
            # (optional) print a tiny message
            print(f"[restart] n={n} iter={it}")

    return best_local_a, best_local_score

        Total time budget in seconds (default 1000 s).
    length : int, optional
        Desired length of the sequence.  If omitted, a small set of lengths
        is tried.

    Returns
    -------
    list[float]
        The best sequence discovered (sum is 1, all entries ≥ 0).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the caller

    # ------------------------------------------------------------------
    # Determine which problem sizes to explore
    # ------------------------------------------------------------------
    if "length" in kwargs:
        lengths = [int(kwargs["length"])]
    else:
        # A mix of moderate sizes; longer sequences give more room for
        # improvement but cost more time.
        lengths = [4000, 6000, 8000]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate the *slow* true metric only when a new global best
    # is found.
    # ------------------------------------------------------------------
    def _maybe_update_global(candidate: np.ndarray, fast_val: float) -> None:
        nonlocal best_seq, best_score
        if fast_val < best_score - 1e-9:
            true_val = evaluate_sequence(candidate.tolist())
            if true_val < best_score:
                best_score = true_val
                best_seq = candidate.copy()
                print(f"[global] new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main loop over chosen lengths
    # ------------------------------------------------------------------
    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break

        # Allocate roughly equal remaining time to each remaining length
        remaining = deadline - time.time()
        alloc = remaining / (len(lengths) - idx)
        phase_deadline = time.time() + alloc

        # Starting vector: try to reuse the global best found so far,
        # otherwise fall back to the analytic sqrt series.
        init_vec = None
        if best_seq is not None and best_seq.shape[0] == n:
            init_vec = best_seq
        else:
            try:
                init_vec = np.array(height_sequence_1, dtype=float)
                if init_vec.shape[0] != n:
                    init_vec = None
            except Exception:
                init_vec = None

        # Run the optimiser for this length
        print(f"[start] length={n}, time budget≈{alloc:.1f}s")
        best_a, best_fast = _optimise_length(n, phase_deadline, init_vec)

        # Record the best local solution
        _maybe_update_global(best_a, best_fast)

        # Show progress for this length
        print(f"[finished] length={n}, best fast score {best_fast:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        # Uniform vector of the first chosen length as a safe default
        n0 = lengths[0]
        best_seq = np.full(n0, 1.0 / n0)

    print(f"[final] overall best evaluate_sequence value = {best_score:.6f}")
    return best_seq.tolist()
