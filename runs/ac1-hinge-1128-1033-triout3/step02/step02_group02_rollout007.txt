"""
Propose a non‑negative step function minimizing the provided evaluation metric.

Algorithm
---------
1. **Warm‑start** – If a global variable ``height_sequence_1`` exists,
   evaluate it (fast FFT based) and keep the best one.

2. **Deterministic parametric sweep** –  Several symmetric families
   (U‑shaped power‑law, sum‑of‑tails, exponential, product‑power‑law)
   are generated for many lengths (up to 30 000) and exponent values.
   The candidate with the smallest *approximate* score
   ``2·n·max(conv)`` (conv = a∗a, fast via FFT) is kept as a warm‑start.

3. **Local stochastic refinement** –  Starting from the best candidate,
   a simple simulated‑annealing / hill‑climbing loop is performed.
   Two move types are used:
   * *Mass transfer* – move a random fraction of the mass from one index
     to another.  The effect on the convolution peak is evaluated with
     an O(n) incremental formula based on a padded copy of the vector,
     thus avoiding repeated FFTs.
   * *Gradient step* – the index of the current maximal convolution entry
     is identified; the sub‑gradient of that entry is taken (g_i = 2·a_{k‑i})
     and a small mirror‑descent step away from the gradient is attempted.
   Moves are accepted if they improve the score or with a probability that
   decays with a temperature schedule (simulated annealing).

   After each accepted move the full convolution is recomputed (FFT) to
   keep the state accurate.

4. **Exact evaluation** – The best vector found is finally evaluated with
   the official ``evaluate_sequence`` routine and returned.

The implementation respects a time budget (default 1000 s).  All helper
functions are top‑level and no I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real part) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Parametric families (all return a normalized vector with sum == 1)
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """Sum of two power‑law tails: a[i] ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the edges: a[i] ∝ exp(-α·min(i,n‑1‑i))."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a


def product_powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """
    Product‑power‑law shape:
        a[i] ∝ ((i+1)*(n-i))^{‑p}
    (very small values in the middle, larger at the ends).
    """
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-12
    a /= a.sum()
    return a

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes the evaluation metric.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Seed from an externally supplied sequence if available.
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)  # noqa: F821
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external /= s
            approx = 2.0 * external.size * max_conv_fft(external)
            if approx < best_score:
                best_score = approx
                best_seq = external.copy()
                print(f"Seeded from height_sequence_1: approx_score={approx:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic parametric sweep
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000, 12000, 16000, 20000, 24000, 28000, 32000]
    p_vals = np.linspace(0.2, 2.5, 23)           # 0.2 … 2.5
    alphas = np.linspace(0.0002, 0.008, 12)      # exponential decay rates

    print("=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # ---- U‑shaped power‑law ----
        for p in p_vals:
            if time.time() > deadline:

            a = u_shaped_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best: n={n:5d}, shape=U‑pow, p={p:4.2f}, approx_score={score:.6f}")

        # ---- sum‑of‑tails ----
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best: n={n:5d}, shape=tails, p={p:4.2f}, approx_score={score:.6f}")

        for a_alpha in alphas:
            if time.time() > deadline:

            a = exponential_distribution(n, a_alpha)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best: n={n:5d}, shape=exp, α={a_alpha:6.4f}, approx_score={score:.6f}")

        # ---- product power‑law ----
        for p in p_vals:
            if time.time() > deadline:
                break
            a = product_powerlaw_distribution(n, p)
            score = 2.0 * n * max_conv_fft(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"New best: n={n:5d}, shape=prod‑pow, p={p:4.2f}, approx_score={score:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:

        a /= a.sum()
        best_seq = a

        print(f"Fallback random init: n={n}, approx_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local stochastic refinement on the simplex.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]

    # Full convolution of the current vector (exact, via FFT)
    conv = full_conv_fft(current)
    conv_max = conv.max()
    current_score = 2.0 * n * conv_max

    # Parameters for the annealing loop
    temperature = 1.0
    cooling_rate = 0.997
    max_frac = 0.35           # maximum fraction of source mass that can be moved
    grad_step_factor = 0.04   # step size multiplier for gradient moves
    batch_moves = 30          # candidate moves examined in each iteration
    it = 0

    print("\n=== Stochastic refinement ===")
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 3a. Decide which move type to try
        # --------------------------------------------------------------
        if np.random.rand() < 0.15:
            # ----- Gradient (mirror‑descent) step -----
            kmax = int(conv.argmax())
            i_start = max(0, kmax - (n - 1))
            i_end = min(n - 1, kmax)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            idx_j = kmax - idx_i
            grad = np.zeros_like(current)
            grad[idx_i] = 2.0 * current[idx_j]

            step = grad_step_factor * temperature
            cand = current - step * grad
            cand = np.clip(cand, 0.0, None)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum

            cand_max = max_conv_fft(cand)
            cand_score = 2.0 * n * cand_max
        else:
            # ----- Random mass‑transfer moves (evaluate a batch) -----
            # Pre‑compute padded array for fast incremental convolution updates
            pad = np.concatenate([np.zeros(n - 1), current, np.zeros(n - 1)])

            best_local_score = current_score
            best_move = None

            for _ in range(batch_moves):
                # source index drawn proportionally to its mass
                if np.all(current == 0):
                    break
                src = np.random.choice(n, p=current / current.sum())
                if current[src] <= 0.0:
                    continue

                # destination (different from source)
                dst = np.random.randint(0, n - 1)
                if dst >= src:
                    dst += 1

                # transfer a random fraction of the source mass
                delta = current[src] * np.random.rand() * max_frac
                if delta <= 0.0:
                    continue

                # incremental update of the convolution peak
                offset_src = n - 1 - src
                offset_dst = n - 1 - dst
                shift_src = pad[offset_src: offset_src + 2 * n - 1]
                shift_dst = pad[offset_dst: offset_dst + 2 * n - 1]

                # new maximal entry after the move (exact up to O(delta²) terms)
                new_max = np.max(conv + 2.0 * delta * (shift_dst - shift_src))
                cand_score = 2.0 * n * new_max

                if cand_score < best_local_score:
                    best_local_score = cand_score
                    best_move = (src, dst, delta, new_max)

            if best_move is None:
                # No improving move found – fall back to a tiny SA acceptance check
                cand_score = current_score + np.inf
                cand_max = conv_max
                cand = current  # dummy, will be rejected
                # no move performed this iteration
                accept = False
            else:
                src, dst, delta, new_max = best_move
                # apply the chosen move
                current[src] -= delta
                current[dst] += delta
                # recompute the exact convolution for the updated vector
                conv = full_conv_fft(current)
                conv_max = conv.max()
                current_score = 2.0 * n * conv_max
                # keep track of global best
                if current_score < best_score:
                    best_score = current_score
                    best_seq = current.copy()
                    print(f"Iter {it:6d}: new global best = {best_score:.6f}")
                # early exit if we already cross the target
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break
                continue  # skip the generic SA acceptance block for this iteration

        # --------------------------------------------------------------
        # 3b. SA acceptance decision (used for gradient moves or when
        #      no improving random move was found)
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # Accept the candidate (gradient move or the previously found move)
            current = cand
            conv = full_conv_fft(current)
            conv_max = conv.max()
            current_score = 2.0 * n * conv_max
            if current_score < best_score:
                best_score = current_score
                best_seq = current.copy()
                print(f"Iter {it:6d}: new global best = {best_score:.6f}")
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break

        # --------------------------------------------------------------
        # 3c. Anneal temperature
        # --------------------------------------------------------------
        temperature = max(temperature * cooling_rate, 1e-6)

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())  # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
