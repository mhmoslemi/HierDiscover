"""
propose_candidate
-----------------

Iterative linear‑programming refinement of a non‑negative step function.

The goal is to minimise
    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

For a fixed length *n* we first fix the *ℓ₁*‑norm
    sum(a) = sqrt(2·n)                                   (1)

so that the score equals the maximal self‑convolution value.
We start from the classic central‑binomial coefficients, which have
max(conv)=π/2≈1.5708.  By repeatedly solving a linear programme we
increase the ℓ₁‑norm that is possible while keeping the maximal
convolution ≤ the current value.  After each LP we blend the old
sequence with the improved one; this strictly reduces the maximal
convolution and therefore the score.

The LP in iteration *k* is

    maximise   Σ g_i
    subject to g_i ≥ 0
               (f⁽ᵏ⁾ * g)_j ≤ M⁽ᵏ⁾   for every convolution index j,

where f⁽ᵏ⁾ is the current sequence and M⁽ᵏ⁾ = max(conv(f⁽ᵏ⁾,f⁽ᵏ⁾)).
Because the convolution is linear in the second argument,
the constraints are linear and the problem is an ordinary LP.
We solve it with SciPy's HiGHS solver (fast and memory‑efficient).

The loop runs until the time budget is exhausted (≈1000 s) or no
improvement is observed.  Finally a tiny stochastic hill‑climb
polishes the best sequence found.

The returned list consists of non‑negative floats and is ready for
`evaluate_sequence`.
"""

import time
from typing import List

import numpy as np
from scipy.optimize import linprog


def _binomial_prefix(max_n: int) -> np.ndarray:
    """Return the central‑binomial coefficients b_k = C(2k,k)/4**k
    for k = 0 … max_n‑1 (recurrence b_{k+1}=b_k·(2k+1)/(2k+2))."""
    b = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, max_n):
        cur = cur * (2 * k - 1) / (2 * k)      # (2k‑1)/(2k) * previous
        b[k] = cur
    return b


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the (2·n‑1) × n matrix A such that A @ g = conv(f,g)."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contains f shifted by j positions
    for j in range(n):
        A[j : j + n, j] = f
    return A


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     n: int = 2000,
                     max_iters: int = 20,
                     blend_t: float = 0.2,
                     **kwargs) -> List[float]:
    """
    Produce a step‑function (list of non‑negative heights) with a low
    value of `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for the optional stochastic polishing.
    budget_s : int
        Overall time budget (seconds).  The function ensures it returns
        before the budget expires.
    n : int
        Length of the searched sequence.  2000 is a good trade‑off
        between quality and speed.
    max_iters : int
        Upper bound on the number of LP refinement iterations.
    blend_t : float
        Convex‑combination factor used when updating the sequence:
        f_{new}= (1‑t)·f + t·g .
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0     # keep a small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise with the central‑binomial coefficients, normalised
    #    so that sum = sqrt(2·n)  (then score == max(conv)).
    # ------------------------------------------------------------------
    b = _binomial_prefix(n)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / b.sum()
    f = b * scale                               # current sequence, length n
    conv_ff = np.convolve(f, f)
    M = conv_ff.max()                           # current maximal self‑conv.
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, sum={f.sum():.6f}, max_conv={M:.6f}, score={best_score:.9f}")

    # ------------------------------------------------------------------
    # 2. Iterative LP refinement
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # Build linear constraints: A @ g ≤ M ,  g ≥ 0
        A = _conv_matrix(f)                     # shape (2n‑1, n)
        ub = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)       # maximise sum(g)  → minimise -Σg

        # Solve the LP (HiGHS interior‑point, fast for dense problems)
        res = linprog(c, A_ub=A, b_ub=ub,
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print(f"[lp] iteration {it} – solver failure, stopping.")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0.0:
            print(f"[lp] iteration {it} – zero‑solution, stopping.")
            break

        # Rescale g0 so that it has the same ℓ₁‑norm as f
        g = g0 * (target_sum / sum_g0)

        # ------------------------------------------------------------------
        # 2a. Try a direct replacement (g may already have a smaller max)
        # ------------------------------------------------------------------
        conv_gg = np.convolve(g, g)
        M_g = conv_gg.max()
        if M_g < M - 1e-8:
            f = g
            M = M_g
            cur_score = evaluate_sequence(f.tolist())
            print(f"[iter {it}] direct replace → max={M:.6f}, score={cur_score:.9f}")
            if cur_score < best_score:
                best_f = f.copy()
                best_score = cur_score
                print(f"  ** new best (direct) score {best_score:.9f}")
            continue  # go to next iteration

        # ------------------------------------------------------------------
        # 2b. Otherwise blend the old and the improved sequence
        # ------------------------------------------------------------------
        f_new = (1.0 - blend_t) * f + blend_t * g
        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()
        new_score = evaluate_sequence(f_new.tolist())
        print(f"[iter {it}] blend → max={M_new:.6f}, score={new_score:.9f}")

        # Accept the blend if it strictly improves the objective
        if new_score < best_score - 1e-9:
            best_f = f_new.copy()
            best_score = new_score
            print(f"  ** new best (blend) score {best_score:.9f}")

        if M_new < M - 1e-7:
            f = f_new
            M = M_new
        else:
            # No progress – stop early
            print(f"[iter {it}] no improvement, terminating refinement.")
            break

    # ------------------------------------------------------------------
    # 3. Light stochastic polishing (optional, runs only if time permits)
    # ------------------------------------------------------------------
    cur_seq = best_f.tolist()
    cur_score = best_score
    sigma = 0.02
    no_improve = 0
    max_no_improve = 2000

    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(n)
        old = cur_seq[idx]
        delta = np.random.randn() * sigma * max(old, 1e-12)
        new_val = max(0.0, old + delta)
        cand = cur_seq[:]
        cand[idx] = new_val
        cand_score = evaluate_sequence(cand)
        if cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            no_improve = 0
            if cand_score < best_score:
                best_f = np.array(cand)
                best_score = cand_score
                print(f"[local] improved score {best_score:.9f}")
        else:
            no_improve += 1

    print(f"[final] best score = {best_score:.9f}, length = {len(best_f)}")
    # Ensure the output is a plain list of non‑negative floats
    return [float(max(0.0, x)) for x in best_f.tolist()]
