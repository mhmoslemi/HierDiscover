def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, cvxpy as cp, time
    deadline = time.time() + budget_s - 5.0

    # parameters
    n = 3000 # maybe adjust based on time
    S_target = np.sqrt(2.0 * n)

    # generate constant-convolution coefficients
    a = np.empty(n, dtype=np.float64)
    cur = 1.0
    a[0] = cur
    for k in range(1, n):
        cur = cur * (2*k-1) / (2*k)
        a[k] = cur
    # scale to target sum
    sum_a = a.sum()
    a = a * (S_target / sum_a)
    f = a.copy()
    
    # compute initial max conv
    def max_conv(x):
        c = np.convolve(x, x)
        return np.max(c)

    M = max_conv(f)
    best_f = f.copy()
    best_M = M

    t = 0.5  # mixing weight
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # solve LP to find g
        g = cp.Variable(n, nonneg=True)
        # constraints: for each k in 0..2n-2, conv(f,g)[k] <= M
        constraints = []
        for k in range(2*n-1):
            # compute linear expression: sum_i f_i * g_{k-i}
            i_start = max(0, k - (n-1))
            i_end = min(k, n-1)
            # For indices i from i_start..i_end, j = k-i
            expr = cp.sum([f[i] * g[k - i] for i in range(i_start, i_end+1)])
            constraints.append(expr <= M)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        # solve with ECOS, maybe set warm start? 
        prob.solve(solver=cp.ECOS, verbose=False, warm_start=True, max_iters=5000)
        if g.value is None:
            break  # solver failed
        g_val = np.array(g.value, dtype=np.float64)
        sum_g = g_val.sum()
        # scaling to target sum
        scale = S_target / sum_g
        g_scaled = g_val * scale
        # mix
        f = (1 - t) * f + t * g_scaled
        # compute new M
        M = np.max(np.convolve(f, f))
        # check improvement

        else:

def propose_candidate(seed=0,budget_s=1000,**kwargs):
    import numpy as np, time
    from scipy.optimize import linprog
    # 1. set up budget
    deadline = time.time() + budget_s - 5.0
    # 2. choose length
    n = 1500   # can adjust based on remaining time
    S_target = np.sqrt(2.0 * n)
    # 3. generate base coefficients (constant-convolution)
    a = np.empty(n, dtype=np.float64)
    cur = 1.0
    a[0] = cur
    for k in range(1, n):
        cur = cur * (2*k - 1) / (2*k)
        a[k] = cur
    # scale to target sum
    a = a * (S_target / a.sum())
    f = a.copy()
    # evaluate initial max
    def max_conv(x):
        return np.max(np.convolve(x, x))
    M = max_conv(f)
    best_f = f.copy()
    best_M = M
    t = 0.5
    iteration = 0
    # loop
    while time.time() < deadline:
        iteration += 1
        # Build constraint matrix
        rows = 2 * n - 1
        T = np.zeros((rows, n), dtype=np.float64)
        for i, fi in enumerate(f):
            T[i:i+n, i] = fi
        # linear program: max sum(g) s.t. T @ g <= M
        c = -np.ones(n)  # minimize -sum => maximize sum
        b = np.full(rows, M, dtype=np.float64)
        # bounds for variables: (0, None)
        bounds = [(0, None)] * n
        # Solve
        res = linprog(c, A_ub=T, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print(f"[debug] LP failed at iteration {iteration}, status: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        # scale g to keep total sum = S_target
        scale = S_target / sum_g
        g_scaled = g * scale
        # mixing
        f = (1.0 - t) * f + t * g_scaled
        # renormalize precisely (to avoid drift)
        f_sum = f.sum()
        f = f * (S_target / f_sum)
        # compute new max conv
        M_new = max_conv(f)
        print(f"[iter] {iteration}: M={M_new:.6f}, improvement={best_M - M_new:.6f}")
        if M_new < best_M - 1e-8:
            best_M = M_new
            best_f = f.copy()
        else:
            # no improvement, reduce t or break
            t *= 0.5
            if t < 1e-4:
                break
        M = M_new
    # final result
    result_seq = best_f.tolist()
    # clamp as required
    result_seq = [float(max(0.0, min(1000.0, x))) for x in result_seq]
    print(f"[result] length={n}, final_M={best_M:.6f}")
    return result_seq

"""
propose_candidate
-----------------
Generate a non‑negative step‑function that beats the classic constant‑convolution
construction.  The idea is to start from the known “Catalan” coefficients,
normalise the vector so that Σ a_i = sqrt(2·n) (for this normalisation the
evaluation score equals the maximal self‑convolution), then repeatedly solve a
linear programme

    max     Σ g_i
    s.t.    (f * g)_k ≤ M   for every k = 0 … 2·n‑2
            g_i ≥ 0

where f is the current sequence and M = max(f*f).  The LP returns a g that can
carry a larger ℓ₁‑norm than f while still respecting the same convolution
bound.  After scaling g back to the target ℓ₁‑norm we mix it with f
(f←(1‑t)·f + t·g).  This convex combination strictly reduces the maximal
convolution value M, and the process is iterated until the time budget expires.
The final sequence is returned (clamped to the evaluator’s [0, 1000] range).
"""

import time
import numpy as np
from scipy.optimize import linprog

def _build_conv_matrix(vec: np.ndarray) -> np.ndarray:
    """Return the dense matrix T such that T @ x = np.convolve(vec, x)."""
    n = vec.size
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=np.float64)
    for i, v in enumerate(vec):
        T[i:i + n, i] = v
    return T


def _max_convolution(vec: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution of vec."""
    return float(np.max(np.convolve(vec, vec)))

    The routine uses an LP‑based refinement of the classic Catalan
    coefficients and runs well within the 1000 s time limit.

    # ----------------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    # 5. Return the best sequence discovered (clamped to the evaluator limits).
    # ----------------------------------------------------------------------
    result = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]
    print(f"[result] length={n}, final_M={best_M:.6f}")
    return result
