def project_to_simplex(v):
    """Project v (non-negative maybe) onto the probability simplex."""
    # 1) sort v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho = max {j | u_j > cssv_j/j}
    rho = np.where(u > cssv / np.arange(1, len(v)+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Fast sub‑gradient / hill‑climbing optimizer for the
max‑convolution objective used in ``evaluate_sequence``.

The algorithm works in three stages:

1. **Initialisation** – start from the previously found best
   sequence (``height_sequence_1``) and linearly interpolate it to a
   larger length (default 8000).  The vector is projected onto the
   probability simplex (non‑negative, ℓ₁‑norm = 1).

2. **Projected sub‑gradient descent** – the true objective (with
   fixed ℓ₁‑norm) is 2·n·maxₖ(a∗a)[k].  Its sub‑gradient at the lag
   attaining the maximum is easy to compute:
   g_i = 2·a_{k‑i} (zero if the index is out of range).
   A step in the opposite direction is taken and the result is
   projected back onto the simplex.

3. **Guided mass‑transfer hill climbing** – for the current worst
   lag we identify the pair (i, j) that contributes most to the
   peak.  A small amount of mass is moved from that pair to a pair
   of low‑mass positions that do **not** recreate the offending lag.
   All candidate destination pairs are evaluated cheaply using an
   FFT based convolution; the best improving move is applied.
   The step‑size is adapted automatically.  Occasionally a tiny
   random perturbation is added to escape plateaus.

The whole routine respects a 1000 s wall‑clock budget; intermediate
scores are printed.  The final list of heights is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolution (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation a*a using FFT.
    Returns a vector of length 2·len(a)‑1.
    Small negative rounding errors are clipped to 0.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a real vector onto the simplex {x ≥ 0, Σx = 1}
    using the O(n) algorithm of Condat (2016).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, fallback
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided optimizer (sub‑gradient + exhaustive mass transfer)
# ----------------------------------------------------------------------
def _guided_optimize(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine ``a_start`` until ``deadline``.
    Returns the best sequence (as a NumPy array) and its exact score.
    """
    a = a_start.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = start_score

    # initial step fraction (relative amount of mass we move each iteration)
    step_frac = 0.07

    # keep an approximate max‑convolution value to avoid unnecessary exact evals
    approx_max = _full_convolution(a).max()

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) compute the current worst lag and the most contributing pair
        # --------------------------------------------------------------
        conv = _full_convolution(a)
        max_b = float(conv.max())

        # early‑exit if nothing improved for a long time
        if it % 2000 == 0 and np.abs(max_b - approx_max) < 1e-12:
            step_frac = max(step_frac * 0.5, 1e-5)

        approx_max = max_b

        # collect all lags attaining the maximum (numerical tolerance)
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i} for this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        if contrib.size == 0:
            continue

        # pick the pair with the largest contribution (break ties randomly)
        max_contrib = contrib.max()
        max_idxs = np.flatnonzero(contrib == max_contrib)
        idx = int(rng.choice(max_idxs))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # --------------------------------------------------------------
        # 2) decide how much mass to move
        # --------------------------------------------------------------
        if i_max != j_max:
            delta = min(a[i_max], a[j_max]) * step_frac
        else:
            delta = a[i_max] * step_frac

        if delta < 1e-14:
            # shrinking step size when mass becomes too tiny
            step_frac = max(step_frac * 0.5, 1e-6)
            continue

        # --------------------------------------------------------------
        # 3) generate a pool of candidate destination pairs (p, q)
        # --------------------------------------------------------------
        low_mass = np.where(a < a.mean())[0]
        if low_mass.size < 2:
            low_mass = np.arange(n)

        cand_pairs = []
        attempts = 0
        while len(cand_pairs) < 30 and attempts < 200:
            p, q = rng.choice(low_mass, size=2, replace=False)
            if p + q != k:                # avoid recreating the offending lag
                cand_pairs.append((int(p), int(q)))
            attempts += 1

        # --------------------------------------------------------------
        # 4) evaluate every candidate (cheap FFT based max convolution)
        # --------------------------------------------------------------
        best_candidate_a = None
        best_candidate_max = max_b

        for p, q in cand_pairs:
            new_a = a.copy()
            if i_max != j_max:
                new_a[i_max] -= delta
                new_a[j_max] -= delta
                new_a[p] += delta
                new_a[q] += delta
            else:
                # i_max == j_max: only one source index
                new_a[i_max] -= delta
                new_a[p] += delta

            if (new_a < -1e-12).any():
                continue

            # (re‑project is not needed – the move preserves the ℓ₁‑norm
            # and non‑negativity up to rounding)
            conv_new = _full_convolution(new_a)
            new_max = float(conv_new.max())
            if new_max < best_candidate_max - 1e-15:
                best_candidate_max = new_max
                best_candidate_a = new_a

        # --------------------------------------------------------------
        # 5) accept the best improving move, adapt the step size
        # --------------------------------------------------------------
        if best_candidate_a is not None:
            a = best_candidate_a
            approx_max = best_candidate_max
            step_frac = max(step_frac * 0.99, 1e-5)

            # occasional exact evaluation – cheap enough because it is rare
            if rng.random() < 0.015 or time.time() > deadline - 5.0:
                exact_score = evaluate_sequence(a.tolist())
                if exact_score < best_score:
                    best_score = exact_score
                    best_a = a.copy()
                    print(f"[improve] iteration {it}, exact score {best_score:.6f}, "
                          f"approx max {approx_max:.9f}, step {step_frac:.6f}")
        else:
            # no improving move found – shrink the step size a little
            step_frac = max(step_frac * 0.995, 1e-6)

        # --------------------------------------------------------------
        # 6) occasional random jitter to escape possible plateaus
        # --------------------------------------------------------------
        if rng.random() < 0.0015:
            noise = rng.normal(scale=step_frac, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            step_frac = min(step_frac * 1.4, 0.12)

    # Final exact evaluation (in case the last state is better)
    try:
        final_score = evaluate_sequence(a.tolist())
        if final_score < best_score:
            best_score = final_score
            best_a = a.copy()
            print(f"[final‑eval] improved to {best_score:.6f}")
    except Exception:
        pass

    return best_a, best_score

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence``.  The routine respects the supplied time
    budget (default 1000 s) and prints progress information.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a larger search space and a
    # potentially smaller theoretical optimum.  8000 keeps the exact
    # evaluation affordable (≈64 M operations).
    # ------------------------------------------------------------------
    TARGET_N = 8000

    # ------------------------------------------------------------------
    # Initialise from the previously discovered best (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != TARGET_N:
            # linear interpolation to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, TARGET_N)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        # fall back to a random Dirichlet vector (sum = 1)
        a0 = rng.dirichlet(np.ones(TARGET_N))

    # enforce feasibility (non‑negative, ℓ₁‑norm = 1)
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0.0:
        a0 = np.full(TARGET_N, 1.0 / TARGET_N)
    else:
        a0 = a0 / a0.sum()

    # Initial exact evaluation (may be slightly slower)
    try:
        best_score = evaluate_sequence(a0.tolist())
    except Exception:
        best_score = float("inf")
    best_seq = a0.copy()
    print(f"[init] n={TARGET_N}, initial exact score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Run the guided optimizer until the time budget runs out
    # ------------------------------------------------------------------
    best_seq, best_score = _guided_optimize(best_seq, best_score, deadline, rng)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] best exact score reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_seq.tolist()
