def propose_candidate(seed=0, budget_s=1000):
    deadline = time.time() + budget_s - 1
    rng = np.random.default_rng(seed)

    # Determine target length
    n_target = 16000  # or maybe 20000

    # Initialize a
    if 'height_sequence_1' in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = base_seq.shape[0]
        # Up-sample
        new_x = np.linspace(0, 1, n_target)
        old_x = np.linspace(0, 1, n0)
        a = np.interp(new_x, old_x, base_seq)
    else:
        a = None

    if a is None:
        # build initial shape: mixture of sine and semicircle or beta distribution
        x = np.arange(n_target, dtype=np.float64)
        t = 2 * x/(n_target-1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t*t, 0.0, None))
        sine = np.sin(np.pi * (x + 0.5) / n_target)
        a = (semicircle + sine) / 2.0

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = rng.random(n_target)
        a_sum = a.sum()
    a = a / a_sum

    # Keep best
    n = n_target
    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv

    # Hyperparameters
    lr = 0.05
    lr_min = 1e-6
    lr_decay = 0.99995
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    eps = max_conv * 0.5
    eps_min = max_conv * 1e-6
    eps_decay = 0.9999

    it = 0
    stagn = 0

    # iteration loop
    while time.time() < deadline - 5:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        current_val = 2.0 * n * max_conv
        if current_val < best_val - 1e-13:
            best_val = current_val
            best_seq = a.copy()
            stagn = 0
            print(f"[{it}] new best: {best_val:.12f}")
        else:
            stagn += 1

        # softmax weights
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0/conv.size)

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n-1:n-1+n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad*grad)

        beta1_t = beta1 ** it
        beta2_t = beta2 ** it
        m_hat = m / (1 - beta1_t)
        v_hat = v / (1 - beta2_t)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a + step
        a_new = _project_simplex(a_new)
        a = a_new

        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        if stagn > 2000:

            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            lr = max(lr_min, lr * 1.5)
            print(f"[perturb] iter {it}")

"""
propose_candidate
-----------------

An adaptive optimiser for the step‑function problem.

* Starts from the best known construction (`height_sequence_1` if it
  exists) and upsamples it to a larger size (≈3× more points).  A
  symmetric distribution is enforced (the optimum is believed to be
  mirror‑symmetric).

* The core optimisation is a projected Adam scheme that minimises a
  smooth approximation of the supremum of the self‑convolution.
  The smoothing is performed by a temperature‑controlled soft‑max
  applied to the convolution entries; the temperature decays
  geometrically to focus the updates on the truly maximal entries.

* Random perturbations are injected when progress stalls, and a final
  hard‑max polishing phase refines the solution with a very small
  constant step size.

* All iterates are projected onto the probability simplex, guaranteeing
  non‑negativity and unit L¹‑norm (the objective is scale‑invariant).

The routine respects a 1000 s wall‑time budget, prints occasional
progress messages and finally returns the best sequence found,
clamped to the limits required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1}.
    Implementation of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Full linear convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

# ----------------------------------------------------------------------
# Generate a random “nice’’ initial shape.
# ----------------------------------------------------------------------
def _random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative vector of length n that sums to 1."""
    x = np.arange(n, dtype=np.float64)
    # basic basis functions
    t = 2.0 * x / (n - 1) - 1.0                       # in [-1,1]
    semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
    sine = np.sin(np.pi * (x + 0.5) / n)
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
    triangle = np.minimum(x, n - 1 - x) + 1.0
    # a random Beta‑shaped term
    alpha = rng.uniform(0.3, 5.0)
    beta = rng.uniform(0.3, 5.0)
    tnorm = x / (n - 1)
    beta_shape = np.power(tnorm, alpha - 1.0) * np.power(1.0 - tnorm, beta - 1.0)
    beta_shape = np.nan_to_num(beta_shape)               # avoid 0⁰
    bases = [semicircle, sine, gauss, triangle, beta_shape]
    weights = rng.random(len(bases))
    weights /= weights.sum()
    a = sum(w * b for w, b in zip(weights, bases))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = rng.random(n)
        s = a.sum()
    a /= s
    return a

# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The function runs for at most `budget_s`
    seconds and returns the best sequence found.
    """
    deadline = time.time() + budget_s - 5.0   # a tiny safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose target length (larger sequences give more flexibility).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = base.shape[0]
        n_target = max(12000, int(3 * n0))       # ≈ three‑fold up‑sampling
        # up‑sample the existing construction by linear interpolation
        old_x = np.linspace(0.0, 1.0, n0)
        new_x = np.linspace(0.0, 1.0, n_target)
        a = np.interp(new_x, old_x, base)
    else:
        n_target = 15000
        a = None

    if a is None:
        a = _random_initial(n_target, rng)

    # enforce symmetry (mirror the vector) and normalise
    a = (a + a[::-1]) * 0.5
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = rng.random(n_target)
        a_sum = a.sum()
    a = a / a_sum

    n = n_target
    # ------------------------------------------------------------------
    # Keep track of the best solution seen so far.
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv          # sum(a) == 1 ⇒ denominator = 1
    best_seq = a.copy()
    print(f"[init] n={n}, initial_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Adam + temperature‑controlled soft‑max (Phase 1).
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    lr = 0.15
    lr_min = 1e-6
    lr_decay = 0.99994
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature

    eps_min = max(max_conv * 1e-6, 1e-12)
    eps_decay = 0.99990

    stagn = 0   # iterations without improvement
    it = 0

    # --- Main optimisation loop -------------------------------------------------
    while time.time() < deadline * 0.68:   # reserve ~32 % for polishing
        it += 1

        # current convolution and score
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        # keep best
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it:6d}] new_best={best_val:.12f}  lr={lr:.2e}  eps={eps:.2e}")
        else:
            stagn += 1

        # soft‑max weighting of the convolution entries
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0.0 and np.isfinite(w_sum):

        beta1_t = beta1 ** it
        beta2_t = beta2 ** it
        m_hat = m / (1.0 - beta1_t)

            stagn = 0

        if it % 4000 == 0:

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_seq = a.copy()
            stagn = 0

        else:
            stagn += 1

        w_sum = mask.sum()

            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask / w_sum

        step = - 2e-5 * grad

        a = _project_simplex(a)
        a = (a + a[::-1]) * 0.5

        if stagn > 5000:
            a = a + rng.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            a = (a + a[::-1]) * 0.5
            stagn = 0
            print(f"[jitter] iteration {it}")

    # ------------------------------------------------------------------
    # Final sub‑gradient polishing (few thousand pure max‑entry steps).
    # ------------------------------------------------------------------
    a = best_seq.copy()

        mx = conv.max()
        active = (conv >= mx - 1e-15).astype(np.float64)

            w = active / w_sum

    # ------------------------------------------------------------------

    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
