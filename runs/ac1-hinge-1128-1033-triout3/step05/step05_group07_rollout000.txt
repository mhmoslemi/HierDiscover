"""
Propose a sequence of non‑negative heights that (approximately) minimizes
``evaluate_sequence``.  The algorithm consists of two phases:

1. **Alternating linear‑programming (LP) step** – given a current
   distribution ``f`` (sum = 1) we solve a small LP:
       minimise M
       s.t.   (f * g)_k ≤ M   for all k = 0 … 2n‑2
              g_i ≥ 0
              Σ g_i = 1
   (``*`` is the ordinary linear convolution).  The optimal ``g`` is the
   most “flat’’ partner for ``f``.  We replace ``f`` by a convex combination
   ``(1‑t)·f + t·g`` with a decreasing mixing factor *t*.  This quickly
   reduces the peak of the self‑convolution.

2. **Fine‑grained hill‑climbing** – starting from the best sequence obtained
   from phase 1 we perform many tiny “mass‑transfer’’ moves
   ``a_i ← a_i – δ , a_j ← a_j + δ``.
   The move is chosen greedily: we locate the index ``k*`` where the
   convolution is maximal and move mass away from the pair ``(i, k*‑i)`` that
   contributes most to that peak.  The convolution is updated in O(n) using
   the exact formula from the reference implementation.

Both phases keep the vector normalised to unit L¹‑norm (scaling does not
change the evaluation value).  The best sequence found before the time
budget expires is returned.

The implementation falls back to the pure hill‑climbing method if SciPy
is not available.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast full convolution using real FFT (used for occasional re‑sync)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return linear convolution a*a (length 2*len(a)-1)."""
    n = a.shape[0]
    total = 2 * n - 1
    size = 1 << (total - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:total]
    # tiny negative values may appear because of round‑off
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Phase 1 – Linear‑programming improvement
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, mix: float):
    """
    Solve the LP that finds the best partner g for the current distribution f.
    Returns the convex combination (1‑mix)·f + mix·g (renormalised to sum=1).
    """
    try:
        from scipy.optimize import linprog
        from scipy import sparse
    except Exception:      # pragma: no cover
        # SciPy missing – just return f unchanged
        return f.copy()

    n = f.shape[0]
    rows = 2 * n - 1                     # number of convolution positions
    # Build the Toeplitz‑like convolution matrix for f.
    # Column i contains the constant value f[i] on rows i … i+n‑1.
    data = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A_f = sparse.diags(data, offsets, shape=(rows, n), format="csr")

    # Column for the auxiliary variable M (coefficient –1)
    M_col = sparse.csr_matrix((-np.ones(rows), (np.arange(rows), np.zeros(rows))),
                              shape=(rows, 1))

    # Full A_ub matrix: [A_f | M_col]
    A_ub = sparse.hstack([A_f, M_col], format="csr")
    b_ub = np.zeros(rows)

    # Equality: sum(g) = 1  (the last variable M is excluded)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Objective: minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Bounds: all variables >= 0
    bounds = [(0, None)] * (n + 1)

    # Solve the LP (HiGHS is the default high‑performance backend)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method="highs")
    if not res.success:
        # LP failed – fall back to current f
        return f.copy()

    g = res.x[:n]
    # Numerical noise may introduce tiny negatives
    g = np.maximum(g, 0.0)
    g_sum = g.sum()
    if g_sum == 0.0:
        g = np.full_like(g, 1.0 / n)
    else:
        g /= g_sum

    # Convex combination and re‑normalisation
    new_f = (1.0 - mix) * f + mix * g
    new_f /= new_f.sum()
    return new_f


# ----------------------------------------------------------------------
# Phase 2 – Greedy hill‑climbing with mass transfer
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float, best_a: np.ndarray,
                best_obj: float) -> tuple[np.ndarray, float]:
    """
    Perform many small mass‑transfer moves to flatten the self‑convolution.
    Starts from ``a`` and updates it in‑place.  Returns the best sequence
    and its objective seen during this phase.
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a)          # length 2n‑1
    cur_max = conv.max()
    cur_obj = float(2.0 * n * cur_max)

    # Hyper‑parameters (tuned for a 1000 s budget)
    max_frac = 0.30          # early moves may transfer up to 30 % of source mass
    min_frac = 0.02          # later moves shrink to 2 %
    jitter_prob = 0.015
    recompute_interval = 4000
    iteration = 0
    last_improve_iter = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        progress = (now - (deadline - (deadline - 0))) / (deadline - (deadline - 0))
        # Linear annealing of the move size
        move_frac = max_frac * (1.0 - progress) + min_frac * progress

        # Locate the current global peak of the convolution
        k_star = int(conv.argmax())
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner = k_star - idx
        prod = a[idx] * a[partner]
        i_src = int(idx[prod.argmax()])

        # Choose a different destination index uniformly at random
        j_dst = np.random.randint(n - 1)
        if j_dst >= i_src:
            j_dst += 1

        # Propose a mass transfer
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue

        a_old = a.copy()               # O(n) copy – cheap enough
        delta2 = delta * delta

        # Incremental update of the convolution using the exact formula
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[j_dst:j_dst + n] += 2.0 * delta * a_old
        conv[2 * i_src] += delta2
        conv[2 * j_dst] += delta2
        conv[i_src + j_dst] -= 2.0 * delta2

        # Apply the mass transfer
        a[i_src] -= delta
        a[j_dst] += delta

        new_max = conv.max()
        new_obj = float(2.0 * n * new_max)

        # Accept only downhill moves (uphill moves are handled by jitter)
        if new_obj < cur_obj:
            cur_obj = new_obj
            cur_max = new_max
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[hill‑climb {iteration:7d}] new best = {best_obj:.6f}")
                last_improve_iter = iteration
        else:
            # Revert the move
            a[:] = a_old
            conv[i_src:i_src + n] += 2.0 * delta * a_old
            conv[j_dst:j_dst + n] -= 2.0 * delta * a_old
            conv[2 * i_src] -= delta2
            conv[2 * j_dst] -= delta2
            conv[i_src + j_dst] += 2.0 * delta2

        # Random jitter (large transfer) to escape plateaus
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta_j = 0.5 * a[src] * np.random.rand()
            if delta_j > 0.0:
                a_tmp = a.copy()
                delta2j = delta_j * delta_j
                conv[src:src + n] -= 2.0 * delta_j * a_tmp
                conv[dst:dst + n] += 2.0 * delta_j * a_tmp
                conv[2 * src] += delta2j
                conv[2 * dst] += delta2j
                conv[src + dst] -= 2.0 * delta2j

                a[src] -= delta_j
                a[dst] += delta_j

                jitter_obj = float(2.0 * n * conv.max())
                if jitter_obj < cur_obj:
                    cur_obj = jitter_obj
                    cur_max = conv.max()
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[hill‑climb {iteration:7d}] jitter improved best = {best_obj:.6f}")
                        last_improve_iter = iteration
                else:
                    # revert jitter
                    a = a_tmp
                    conv[src:src + n] += 2.0 * delta_j * a_tmp
                    conv[dst:dst + n] -= 2.0 * delta_j * a_tmp
                    conv[2 * src] -= delta2j
                    conv[2 * dst] -= delta2j
                    conv[src + dst] += 2.0 * delta2j

        # Periodic full recompute to wipe numerical drift
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a)

        # Stagnation detection – re‑normalise occasionally
        if iteration - last_improve_iter > 250_000:
            a /= a.sum()
            conv = _full_convolution_fft(a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)
            last_improve_iter = iteration

    return best_a, best_obj

    best_obj = float(2.0 * n * _full_convolution_fft(best_a).max())

    # Phase 2 – intensive hill‑climbing until time runs out
    # ------------------------------------------------------------------
    best_a, best_obj = _hill_climb(a, deadline, best_a, best_obj)

    # ------------------------------------------------------------------
    # Final reporting (using the official evaluation function)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
