def _solve_lp_max_sum(f, M):
    n = f.shape[0]
    # Build Toeplitz matrix
    c_col = np.concatenate([f, np.zeros(n-1)])
    c_row = np.concatenate([f[:1], np.zeros(n-1)])  # first row size n
    T = toeplitz(c_col, c_row) # shape (2n-1, n)
    # Linear program: minimize -sum(b) subject to T @ b <= M, b >= 0
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=T, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        b = res.x
        return b
    else:
        raise RuntimeError("LP failed")

"""
Propose a new coefficient sequence for the self‑convolution upper‑bound problem.

Algorithm
---------
1. **Low‑dimensional LP refinement** (optional, uses SciPy):
   * Work on a reduced size `n_lp` (≈1500) where a linear program
     `max Σ b_j  s.t. f*b ≤ M, b ≥ 0` can be solved quickly.
   * Start from a simple distribution (uniform or a down‑sampled warm‑start).
   * Repeatedly:
        - compute the current bound `M = max(f*f)`,
        - solve the LP (HiGHS) to obtain a vector `b`,
        - normalize `b` to sum 1 (`g = b/Σb`) and mix
          `f ← (1‑t)·f + t·g` (projected onto the simplex).
   * Keep the best `f` (lowest `2·n_lp·M`).

2. **Upsampling**:
   * Interpolate the best low‑dimensional `f` to the final length
     `N` (the length of a warm‑start if present, otherwise 5000).

3. **Adam polishing**:
   * Perform projected Adam updates on the full‑size distribution.
   * The gradient is computed from a soft‑max weighting of the convolution
     entries (`w ∝ exp((conv‑max)/ε)`), which yields a sub‑gradient of
     `max(conv)`.
   * Learning‑rate and temperature decay slowly; occasional jitter
     avoids stagnation.

4. **Return** the best sequence found (clamped to `[0,1000]`).  The function
   respects the 1000 s budget and prints intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Low‑dimensional linear program: maximize Σ b_j  s.t. f*b ≤ M, b ≥ 0.
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float):
    """Solve the LP for a given f and bound M, return the optimal b."""
    try:
        from scipy.optimize import linprog
        from scipy.linalg import toeplitz
    except Exception as exc:
        raise RuntimeError("SciPy is required for the LP step") from exc

    n = f.shape[0]
    # Build Toeplitz matrix T where (T @ b) = f * b (full convolution)
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    T = toeplitz(col, row)                      # (2n‑1) × n matrix
    c = -np.ones(n, dtype=f.dtype)              # maximize Σ b  → minimize -Σ b
    bounds = [(0.0, None)] * n
    ub = np.full(2 * n - 1, M, dtype=f.dtype)
    res = linprog(c, A_ub=T, b_ub=ub, bounds=bounds, method="highs")
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

# ----------------------------------------------------------------------

    """

    else:
        N = 5000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    try:
