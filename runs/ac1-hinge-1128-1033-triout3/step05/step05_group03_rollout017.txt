    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        temperature = temp_start * (temp_end / temp_start) ** progress

"""
Propose a non‑negative step‑function that minimizes `evaluate_sequence`.

Algorithm
---------
1. Choose a relatively large problem size (default n = 12000).
2. Run several short Adam‑optimisation phases from diverse
   initialisations (uniform, Dirichlet, Gaussian bump).  The Adam
   optimiser minimises a smooth log‑sum‑exp approximation of the
   maximal self‑convolution value while staying on the simplex.
3. Take the best candidate and launch a *simulated‑annealing hill‑climber*.
   Each move transfers a small amount of mass from a “source” index
   (chosen proportionally to its contribution to the current maximal
   convolution entry) to a handful of random “target” indices.
   The best improving target is accepted; otherwise a worsening move
   may be accepted with a temperature‑dependent probability.
   Stagnation triggers a Gaussian perturbation of the current best.
4. When time permits, repeat the hill‑climber from fresh random starts,
   always keeping the globally best sequence.
5. Return the best sequence found (normalised to sum‑one).  The final
   evaluation call is performed with the provided `evaluate_sequence`
   wrapper.

The method combines global exploration (multiple restarts, annealing)
with fast FFT‑based convolution updates, and fits comfortably within
the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is assumed = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) (sum(a) is 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smoothed version of the max convolution.
    Returns the best (a, value) seen before *deadline*.
    """
    # soft‑max parametrisation (guarantees Σa = 1, a ≥ 0)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.0015
    eps_start, eps_end = 5e-2, 1e-13

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 25000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---------- decode soft‑max ----------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()                       # Σa = 1, a ≥ 0

        # ---------- exact objective for bookkeeping ----------
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve_cnt += 1

        # ---------- schedules ----------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---------- smooth max via log‑sum‑exp ----------
        b_full = _full_convolution(a)           # length 2·n‑1
        max_b = b_full.max()
        # stable log‑sum‑exp
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()             # softmax weights ≈ one‑hot

        # ---------- gradient of smooth objective w.r.t. a ----------
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)           # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]      # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]        # shape (n,)

        # ---------- chain rule through soft‑max ----------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.008, size=n)

        # restart on stagnation
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] restart after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.12
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – simulated‑annealing hill climbing with multi‑target moves
# ----------------------------------------------------------------------
def _hillclimb_annealed(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """
    Hill climbing that transfers a small amount of mass from a source
    index (chosen proportionally to its contribution to the current
    maximal convolution entry) to a few random targets, accepting the
    best improving target or, with a temperature‑dependent probability,
    a slight worsening move.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.02
    temp_start, temp_end = 1e-2, 1e-7
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    iteration = 0
    NUM_TARGETS = 7

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedules for transfer fraction and temperature
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        temperature = temp_start * (temp_end / temp_start) ** progress

        # ---- locate current maximal convolution entry ----
        k_max = int(np.argmax(b))

        # ---- contribution of each index to that entry ----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- pick source index (biased to large contributions) ----
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # ---- pick a few target indices (different from i) ----
        possible_js = np.arange(n)
        possible_js = possible_js[possible_js != i]
        if possible_js.size == 0:
            continue
        targets = np.random.choice(possible_js,
                                   size=min(NUM_TARGETS, possible_js.size),
                                   replace=False)

        # ---- amount of mass to move ----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- evaluate all candidates, keep the best improvement ----
        best_candidate_val = cur_val
        best_j = None
        best_b = None

        for j in targets:
            b_new = b.copy()
            b_new[j:j + n] += 2.0 * delta * a
            b_new[i:i + n] -= 2.0 * delta * a
            b_new[2 * j] += delta * delta
            b_new[2 * i] += delta * delta
            b_new[i + j] -= 2.0 * delta * delta

            new_val = 2.0 * n * b_new.max()

            if new_val < best_candidate_val:
                best_candidate_val = new_val
                best_j = j
                best_b = b_new

        # ---- decide whether to accept a move ----
        accept = False
        if best_j is not None and best_candidate_val < cur_val - 1e-14:
            # strict improvement
            accept = True
        else:
            # possibly accept a worsening move (simulated annealing)
            if np.random.rand() < np.exp((cur_val - best_candidate_val) / max(temperature, 1e-12)):
                # fallback to the first target in the list
                best_j = targets[0]
                best_candidate_val = 2.0 * n * (
                    b.copy()[best_j:best_j + n] + 2.0 * delta * a
                ).max()  # quick rough bound (will be recomputed below)
                # recompute accurate b for the chosen target
                b_new = b.copy()
                b_new[best_j:best_j + n] += 2.0 * delta * a
                b_new[i:i + n] -= 2.0 * delta * a
                b_new[2 * best_j] += delta * delta
                b_new[2 * i] += delta * delta
                b_new[i + best_j] -= 2.0 * delta * delta
                best_candidate_val = 2.0 * n * b_new.max()
                best_b = b_new
                accept = True

        if accept:
            # perform the accepted move
            a[i] -= delta
            a[best_j] += delta
            b = best_b
            cur_val = best_candidate_val
            no_improve = 0
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[HillAnneal {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ---- stagnation handling ----
        if no_improve > IMPROVE_LIMIT:
            # perturb the current best and restart from there
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            s = a.sum()
            if s == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillAnneal {iteration:7d}] perturbing after stagnation")
        iteration += 1

    return best_a

# Main entry point

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # problem size – larger n gives a better theoretical bound
    n = 12000

    # ------------------------------------------------------------------
    # Create a pool of diverse initial points
    # ------------------------------------------------------------------
    def _make_initial():
        # 1. Uniform
        yield np.full(n, 1.0 / n)
        # 2. Dirichlet with low concentration (spiky)
        yield np.random.dirichlet(np.full(n, 0.5))
        # 3. Dirichlet with concentration 1 (moderately spread)
        yield np.random.dirichlet(np.full(n, 1.0))
        # 4. Gaussian‑like bump
        x = np.arange(n, dtype=float)
        mu = n / 2.0
        sigma = n / 6.0
        g = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        g /= g.sum()
        yield g
        # 5. If a previous construction exists, interpolate it
        try:
            old = np.asarray(height_sequence_1, dtype=float)   # type: ignore
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, old)
            interp = np.maximum(interp, 0.0)
            interp /= interp.sum()
            yield interp
        except Exception:
            pass

    best_a = None
    best_val = float('inf')
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – run a few Adam optimisations from the different starts
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    for init in _make_initial():
        if time.time() >= phase_a_deadline:
            break
        a0 = init
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        a_opt, val_opt = _adam_phase(a0, n,
                                     start=start_time,
                                     deadline=phase_a_deadline,
                                     best_a=best_a if best_a is not None else a0,
                                     best_val=best_val,
                                     target_val=TARGET)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[PhaseA] new global best = {best_val:.6f}")

    if best_a is None:
        # fallback – uniform
        best_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Phase B – annealed hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    phase_b_deadline = deadline
    best_a = _hillclimb_annealed(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=phase_b_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=TARGET)

    # ------------------------------------------------------------------
    # If time is still left, launch a few quick extra hill‑climbs
    # starting from fresh random Dirichlet points.
    # ------------------------------------------------------------------
    extra_block = 20.0  # seconds per extra run
    while time.time() + extra_block < deadline:
        block_end = min(deadline, time.time() + extra_block)
        a_start = np.random.dirichlet(np.full(n, 0.7))
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()
        a_candidate = _hillclimb_annealed(a_start, n,
                                          start=time.time(),
                                          deadline=block_end,
                                          best_a=best_a,
                                          best_val=best_val,
                                          target_val=TARGET)
        cand_val = _objective(a_candidate)
        if cand_val < best_val:
            best_val, best_a = cand_val, a_candidate.copy()
            print(f"[Extra] new global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Ensure the returned list respects the non‑negative, sum‑one convention
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s == 0:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a /= s

    return best_a.tolist()
```

This implementation expands the search space, uses multiple Adam restarts, and a more aggressive hill‑climbing procedure with simulated annealing and multi‑target moves. It stays within the 1000 s budget and returns the best non‑negative sequence found.
