"""
Propose a non‑negative step‑function (a list of floats) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the full linear convolution.
The algorithm works in two stages:

1.  A cheap optimisation on a reduced resolution (n≈1500).  We run many
    iterations of a soft‑max gradient descent with momentum, switching to an
    Adam‑style update after roughly half of the allocated time.  This stage
    discovers a good “shape’’ for the sequence.

2.  The shape is up‑sampled to the target length (n=6000) and refined again
    with the same two‑phase optimiser.  The best sequence seen during the
    whole run is returned.

Both phases keep the vector on the probability simplex
({x≥0, sum(x)=1}) using the Duchi‑et‑al projection.
Random perturbations are injected when progress stalls.
The routine respects the 1000 s budget (a 5 s safety margin) and prints
intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (sum(x)=1, x≥0).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x | x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full linear convolution using real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return a * b (full linear convolution) using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Core optimiser (works for any length n).
# Returns the best sequence seen and its objective value.
# ----------------------------------------------------------------------
def _optimise(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    phase_name: str = "full",
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector ``a`` under the objective
    2*n*max(conv(a,a)).  The algorithm uses a soft‑max gradient descent
    with momentum for the first half of the time budget and an Adam‑style
    exact sub‑gradient descent for the second half.
    """
    n = a.shape[0]
    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    M = conv.max()
    best_val = 2.0 * n * M  # sum(a) == 1 → objective = 2*n*max(conv)
    print(f"[{phase_name}] init: n={n}, score={best_val:.12f}")

    # momentum variables (phase‑1)
    eta = 0.8
    eta_min = 1e-6
    eta_decay = 0.9997
    eps = max(M * 0.5, 1e-12)
    eps_min = max(M * 1e-5, 1e-12)
    eps_decay = 0.9995
    momentum = 0.9
    vel = np.zeros_like(a)

    # Adam variables (phase‑2)
    beta1, beta2 = 0.9, 0.999
    eta0 = 0.12
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = 1.0, 1.0
    t_adam = 0

    # stagnation detector
    stagn = 0
    start = time.time()
    switch_to_adam = start + 0.5 * (deadline - start)

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) evaluate current objective
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = 2.0 * n * M

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            stagn = 0
            print(f"[{phase_name}] iter {t_adam if t_adam else 0:6d} new_best={best_val:.12f}")
        else:
            stagn += 1

        # --------------------------------------------------------------
        # 2) choose update rule
        # --------------------------------------------------------------
        if time.time() < switch_to_adam:
            # ------- soft‑max gradient descent (momentum) -------
            w = np.exp((conv - M) / eps)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

            rev_a = a[::-1]
            g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1 : n - 1 + n]

            vel = momentum * vel + (1.0 - momentum) * g
            a_new = a - eta * vel
            a = _project_simplex(a_new)

            # decay schedules
            eta = max(eta_min, eta * eta_decay)
            eps = max(eps_min, eps * eps_decay)

        else:
            # ------- exact max‑subgradient with Adam -------
            tol = 1e-14
            active = np.flatnonzero(conv >= M - tol)
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0 / active.size
            else:
                w[np.argmax(conv)] = 1.0

            rev_a = a[::-1]
            g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1 : n - 1 + n]

            t_adam += 1
            beta1_pow *= beta1
            beta2_pow *= beta2

            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

            a_candidate = a - step
            a = _project_simplex(a_candidate)

        # --------------------------------------------------------------
        # 3) occasional random perturbation if we are stuck
        # --------------------------------------------------------------
        if stagn > 4000:
            noise = rng.normal(scale=0.002, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            vel.fill(0.0)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            print(f"[{phase_name}] perturb (stagnation)")

        # occasional re‑normalisation (numerical safety)
        if (t_adam if t_adam else 0) % 5000 == 0:
            a = a / a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a non‑negative height sequence (list of floats) that (approximately)
    minimizes evaluate_sequence().  The routine runs for at most ``budget_s``
    seconds and prints progress information.
    """
    # ------------------------------------------------------------------
    # Timing & rng
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # 5 s safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Stage 1 – cheap optimisation on a reduced resolution.
    # ------------------------------------------------------------------
    n_small = 1500
    # initialise a small‑n candidate: mixture of a few classic shapes
    x = np.arange(n_small, dtype=np.float64)

    # Catalan‑type decreasing shape
    catalan = np.empty(n_small, dtype=np.float64)
    cur = 1.0
    catalan[0] = cur
    for k in range(1, n_small):
        cur = cur * (2.0 * k - 1) / (2.0 * k)
        catalan[k] = cur

    # Sine (always positive)
    sine = np.sin(np.pi * (x + 0.5) / n_small)

    # Semi‑circle
    t = 2.0 * x / (n_small - 1) - 1.0
    semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

    # Gaussian centred in the middle
    sigma = n_small / 6.0
    gauss = np.exp(-0.5 * ((x - (n_small - 1) / 2.0) / sigma) ** 2)

    # Simple triangle
    triangle = np.minimum(x, n_small - 1 - x) + 1.0

    bases = [catalan, sine, semicircle, gauss, triangle]
    weights = rng.random(len(bases))
    weights /= weights.sum()
    a_small = sum(w * b for w, b in zip(weights, bases))
    a_small = np.maximum(a_small, 0.0)
    a_small = a_small / a_small.sum()               # project onto simplex

    # run optimisation on the reduced size
    deadline_stage1 = start_time + 0.55 * (deadline - start_time)
    best_small, _ = _optimise(a_small, deadline_stage1, rng, phase_name="stage‑1")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to the target length and fine‑tune.
    # ------------------------------------------------------------------
    n_target = 6000
    # linear interpolation of the best small‑n shape
    xv_small = np.linspace(0, 1, n_small)
    xv_target = np.linspace(0, 1, n_target)
    a_target = np.interp(xv_target, xv_small, best_small)
    a_target = np.maximum(a_target, 0.0)
    a_target = a_target / a_target.sum()            # again on the simplex

    # fine‑tune on the full size
    best_full, final_score = _optimise(a_target, deadline, rng, phase_name="stage‑2")

    # ------------------------------------------------------------------
    # Convert the best sequence to the format expected by evaluate_sequence.
    # ------------------------------------------------------------------
    final_seq = [
        float(max(0.0, min(1000.0, x))) for x in best_full.tolist()
    ]

    # optional logging of the final score (may be slightly different due to
    # rounding / clipping)
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_val:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] could not compute final_score (evaluate_sequence missing)")

    return final_seq
