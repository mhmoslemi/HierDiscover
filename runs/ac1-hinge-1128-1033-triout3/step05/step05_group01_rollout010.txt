Projection algorithm: Implementation of Euclidean projection onto simplex (nonnegative and sum=1) using standard algorithm:

```
def project_simplex(v):
    # ensure sum=1, all >=0
    # returns w: sum(w)=1, w_i >= 0, min Euclidean distance to v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Algorithm:

- Initialise a probability vector a of length n (default 4000) from the
  previously found best sequence if available, otherwise uniform.
- The objective to minimise is 2·n·max(conv(a,a)),
  where conv denotes the full linear convolution.
- A projected sub‑gradient descent is performed directly on the exact
  non‑smooth objective: at each step we locate the lag k where the
  convolution attains its maximum, compute the sub‑gradient
  ∂/∂a_i (conv_k) = 2·a_{k‑i}, take a step in the negative direction,
  and project back onto the probability simplex (non‑negative, sum = 1).
- The step size is α_t = α₀ / √t with a moderate initial α₀.
  Periodically a small Gaussian perturbation is added if no improvement
  was observed for a while, helping to escape plateaus.
- After the sub‑gradient phase a short guided hill‑climbing phase
  (mass‑transfer moves) is executed for fine polishing.
- The best sequence encountered is returned.  The exact evaluation is
  only called at the end; all intermediate scores are computed from the
  convolution (which equals the value of ``evaluate_sequence`` when the
  vector is normalised to unit ℓ₁‑norm).

The routine respects the 1000 s budget and prints progress.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Remove possible tiny negative values caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Simple projection onto the probability simplex (non‑negative, sum=1)."""
    v = np.maximum(v, 0)
    total = v.sum()
    if total == 0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total

def _gradient_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry ``conv[k]`` with respect to ``a``.
    For the full autocorrelation conv = a*a we have
    ∂conv[k]/∂a_i = 2·a_{k‑i} (zero when the index is out of range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)          # inclusive
    # vectorised assignment
    i_range = np.arange(i_start, i_end + 1)
    j_range = k - i_range
    g[i_range] = 2.0 * a[j_range]
    return g

def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine ``a`` by repeatedly moving a small fraction of mass away from the
    pair that contributes most to the current maximal convolution entry.
    The move is accepted only if the exact objective improves.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.03  # start with a slightly larger fraction than before
    no_improve = 0

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # Pick two destination positions that are not the offending lag.
        # Prefer positions with small mass.
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            step_frac = max(step_frac * 0.99, 5e-5)
            no_improve = 0
            print(f"[guided] new best = {cur_val:.6f}")
        else:
            no_improve += 1
            step_frac = max(step_frac * 0.995, 5e-5)

        # occasional aggressive perturbation if stuck
        if no_improve >= 500:
            # add small random redistribution
            noise = rng.normal(scale=0.01, size=n)
            cur_a = cur_a + noise
            cur_a = _project_onto_simplex(cur_a)
            no_improve = 0

    return cur_a, cur_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The method combines:
    1. initialisation from a previously discovered sequence;
    2. a long projected sub‑gradient descent on the exact L∞ norm of the
       self‑convolution (with a decreasing 1/√t stepsize);
    3. a short guided hill‑climbing refinement.
    The function respects the time budget and prints intermediate progress.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector a (ℓ₁‑norm = 1)
    # ------------------------------------------------------------------
    n_target = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n_target:
            # Linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n_target)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n_target, 1.0 / n_target, dtype=float)

    a = np.maximum(a, 0.0)
    a = _project_onto_simplex(a)   # ensures sum = 1

    n = a.shape[0]
    conv = _full_convolution(a)
    cur_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_obj
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Projected sub‑gradient descent on the exact objective.
    # ------------------------------------------------------------------
    step0 = 0.8          # initial step length
    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1

        # compute full convolution and the current objective
        conv = _full_convolution(a)
        max_b = conv.max()
        cur_obj = 2.0 * n * max_b

            best_val = cur_obj
            best_a = a.copy()
            last_improve_iter = iteration

        k = int(ks[0])   # take the first

        a = a - alpha * g

    #    evaluate_sequence call).
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # 4) Report final value and return the sequence.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()

"""
Projected sub‑gradient optimisation of a step‑function.

The objective that ``evaluate_sequence`` computes (for a vector a with
ℓ₁‑norm = 1) is

    L(a) = 2·n·max_k (a * a)[k] ,

where ``*`` denotes the full linear convolution.  Minimising L is
equivalent to flattening the self‑convolution as much as possible.

The algorithm proceeds in three phases, respecting the 1000 s time
budget:

1. **Initialisation** – start from the previously discovered best
   sequence (``height_sequence_1`` if it exists) or from a uniform
   distribution; the vector is projected onto the probability simplex
   (non‑negative, sum = 1).

2. **Projected sub‑gradient descent** – repeatedly
   * compute the convolution,
   * locate a lag k where the maximum is attained,
   * build the sub‑gradient g_i = 2·a_{k‑i} (zero when the index is out of
     range),
   * take a step a ← a − α·g with a decreasing stepsize α_t = α₀/√t,
   * project back onto the simplex.
   Tiny random perturbations are injected if progress stalls.

   The exact objective value is obtained cheaply from the convolution
   (no call to ``evaluate_sequence`` inside the loop).

3. **Guided hill‑climbing** – a short refinement that moves a small
   fraction of mass away from the pair that contributes most to the
   current maximal convolution entry; moves are accepted only if the
   true ``evaluate_sequence`` value improves.

The best sequence found is returned as a plain Python list.  Progress
and intermediate bounds are printed for diagnostics.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative values caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.maximum(v, 0)
    total = v.sum()
    if total == 0:
        # fallback to the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total


# ----------------------------------------------------------------------
# Sub‑gradient of a single convolution entry
# ----------------------------------------------------------------------
def _gradient_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the entry conv[k] = (a * a)[k] w.r.t. a.
    For the full autocorrelation conv = a*a we have
        ∂conv[k]/∂a_i = 2·a_{k‑i}      (zero when out of range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)               # inclusive
    i_range = np.arange(i_start, i_end + 1)
    j_range = k - i_range
    g[i_range] = 2.0 * a[j_range]
    return g


# ----------------------------------------------------------------------
# Small guided hill‑climbing phase (uses the exact evaluate_sequence)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fine‑tune ``a`` by moving a modest fraction of mass away from the most
    offending convolution lag.  Only moves that strictly improve the true
    ``evaluate_sequence`` value are kept.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.03          # start with a modest fraction of mass to move
    no_improve = 0

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two destination indices that are unlikely to recreate the same lag
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            step_frac = max(step_frac * 0.99, 5e-5)
            no_improve = 0
            print(f"[guided] new best = {cur_val:.6f}")
        else:
            no_improve += 1
            step_frac = max(step_frac * 0.995, 5e-5)

        # if we are stuck for a while, inject a tiny random reshuffle
        if no_improve >= 500:
            noise = rng.normal(scale=0.01, size=n)
            cur_a = _project_onto_simplex(cur_a + noise)
            no_improve = 0

    return cur_a, cur_val
