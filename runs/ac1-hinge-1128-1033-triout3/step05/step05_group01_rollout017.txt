"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Algorithm
---------
1. **Scale** – work with a probability vector (ℓ₁‑norm = 1); the objective
   ``2·n·max(conv)`` is then independent of the overall scale.
2. **Smooth optimisation** – minimise a log‑sum‑exp surrogate of the
   max‑convolution with L‑BFGS‑B.  The continuation schedule for the
   smoothing parameter ε is fine‑grained and ends at 1e‑12.
3. **Multiple restarts** – after the primary pass, perturb the best
   solution and repeat the continuation on the tail of the ε‑schedule.
4. **Guided hill‑climbing** – repeatedly locate the lag *k* at which the
   convolution is maximal, identify the pair of indices contributing most
   to it, and move a small amount of mass from that pair to a pair of
   low‑mass indices that do not recreate the offending lag.  Several
   destination candidates are tried per iteration; the best improving
   move is kept.  The step size decays slowly and the procedure stops
   when the time budget expires.

The routine respects a 1000 s budget, prints progress, and returns the
best list of non‑negative heights found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Parameters
    ----------
    w   – soft‑max parameters (real vector of length n)
    eps – smoothing parameter (positive)
    n   – problem size (length of the sequence)

    Returns
    -------
    L, grad_w – surrogate value and its gradient w.r.t. w
    """
    # soft‑max (guarantees non‑negativity and ℓ₁‑norm = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # full autocorrelation a*a
    conv = _full_convolution(a)          # length 2·n‑1
    max_conv = conv.max()

    # log‑sum‑exp smoothing of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)         # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()          # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)        # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]       # middle n entries
    grad_a = 4.0 * n * seg[::-1]         # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

            # linear interpolation to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start

    # enforce feasibility (non‑negative, ℓ₁‑norm = 1)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    w = np.log(a0 + 1e-12)           # soft‑max parameters

    best_a = a0.copy()
    try:
        best_val = evaluate_sequence(best_a.tolist())
    except Exception:
        best_val = float("inf")
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Continuation schedule for ε (fine‑grained, ends at 1e‑12)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-2), np.log10(1e-12), num=30)

    # --------------------------------------------------------------
    # Primary optimisation pass (L‑BFGS‑B on the smooth surrogate)
    # --------------------------------------------------------------
    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # exact evaluation of the current soft‑max point
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Random restarts – repeat the continuation on perturbed starts
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0 and restart < 6:
        restart += 1
        # perturb the best soft‑max parameters
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.6, size=n)

        # use only the *tail* of the ε‑schedule for a quick fine‑tune
        for eps in eps_schedule[-12:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final guided hill‑climbing refinement (more exhaustive search)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb_improved(best_a,
                                                       best_val,
                                                       deadline,
                                                       rng)

    # --------------------------------------------------------------
    # Final report
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
