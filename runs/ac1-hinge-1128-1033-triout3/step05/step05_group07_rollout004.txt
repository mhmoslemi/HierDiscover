def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # initialization
    try:
        init_seq = height_sequence_1
    except NameError:
        init_seq = None

    if init_seq is None:
        n = 2500  # maybe bigger than 1800
        a = np.full(n, 1.0/n)
    else:
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.shape[0]

    # Normalize sum to 1
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= a_sum

    # FFT size
    size = 1 << ((3*n - 2).bit_length())

    # Best
    best_a = a.copy()
    conv = np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(a, size), size)[:2*n-1]
    best_obj = float(2.0 * n * conv.max())

    # schedules
    beta_start = 1.0
    beta_end = 200.0
    eta_start = 0.5   # step factor
    eta_end = 0.005

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)

        # annealing schedule
        beta = beta_start * (1 - progress) + beta_end * progress
        eta = eta_start * (1 - progress) + eta_end * progress

        # FFT of a
        a_fft = np.fft.rfft(a, size)

        # convolution a*a
        conv = np.fft.irfft(a_fft * a_fft, size)[:2*n-1]

        # compute current objective
        cur_max = conv.max()
        cur_obj = float(2.0 * n * cur_max)

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[{elapsed:.1f}s, iter={iteration}] new best obj = {best_obj:.6f}")

        # softmax over conv
        # use stability
        max_c = cur_max
        exp_term = np.exp(beta * (conv - max_c))
        w = exp_term / exp_term.sum()

        # compute gradient via cross-correlation of w and a
        a_rev = a[::-1]
        w_fft = np.fft.rfft(w, size)
        a_rev_fft = np.fft.rfft(a_rev, size)

        cross = np.fft.irfft(w_fft * a_rev_fft, size)[:len(w)+n-1]

        grad = 2.0 * cross[(n-1):(n-1+n)]

        # Clip gradient for stability of exp
        # Use cutoff based on eta to avoid overflow > 50
        clip_val = 50.0 / max(eta, 1e-12)  # avoid divide by zero
        grad_clipped = np.clip(grad, -clip_val, clip_val)

        # multiplicative update
        a = a * np.exp(-eta * grad_clipped)

        # renormalize to simplex
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= a_sum

        # possibly add occasional random jitter to escape

            src = np.random.randint(n)
            dst = np.random.randint(n-1)
            if dst >= src:
                dst += 1
            delta = 0.5 * a[src] * np.random.rand()
            if delta > 0:
                a_src_orig = a[src]
                a_dst_orig = a[dst]
                a[src] = a_src_orig - delta
                a[dst] = a_dst_orig + delta
                # re-normalize not needed as sum unchanged

    # final best

        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:

"""
Propose a candidate step‑function by minimising the peak of its self‑convolution.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) and
optimises a smooth approximation of the objective
    f(a) = max_k (a*a)_k .
The smooth surrogate used is the log‑sum‑exp of the convolution,
whose gradient can be written analytically as a cross‑correlation.
A multiplicative (exponentiated‑gradient) update keeps the iterates
strictly non‑negative and a final normalisation restores the simplex
constraint.  The smoothing parameter β and the step‑size η are
annealed during the run.  Occasionally a random “jitter” mass‑transfer
is performed to escape shallow plateaus.

The routine runs until the allotted time (≈1000 s) expires and returns
the best sequence encountered (as a plain ``list[float]``) which can be
fed to ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Gradient‑based exponentiated‑descent optimiser for the step‑function problem.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function stops a few seconds early
        to leave a safety margin for the evaluation harness).

    Returns
    -------
    list[float]
        The best coefficient sequence found (non‑negative, sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – the harness will terminate the process if we
    # overrun the budget.
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Initialise the coefficient vector.
    # Prefer a previously found construction if it exists.
    # ------------------------------------------------------------------
    try:                               # global variable supplied by the framework
        init_seq = height_sequence_1   # noqa: F821
    except NameError:
        init_seq = None

    if init_seq is None:
        # a modest size gives a good speed/quality trade‑off
        n = 2500
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.shape[0]

    # Normalise to a probability vector (the evaluation is homogeneous).
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a /= total

    # ------------------------------------------------------------------
    # Pre‑compute constants and FFT size.
    # ------------------------------------------------------------------
    # Size sufficient for the longest linear convolution we will need:
    #   len(w) = 2n‑1, len(a) = n  →  len(w)+len(a)‑1 = 3n‑2
    fft_size = _next_pow2(3 * n - 2)

    # Phase factor to obtain the FFT of the reversed vector without an extra FFT:
    k = np.arange(fft_size // 2 + 1)                     # frequencies for rfft
    phase_factor = np.exp(-2j * np.pi * (n - 1) * k / fft_size)

    # Best solution seen so far.
    # Compute the initial objective using the exact convolution.
    a_fft = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(a_fft * a_fft, fft_size)[: 2 * n - 1]
    cur_max = conv.max()
    best_obj = float(2.0 * n * cur_max)
    best_a = a.copy()
    print(f"[init] n={n}, objective={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Annealing schedules.
    #   β (beta) – smoothness of the max (log‑sum‑exp temperature)
    #   η (eta) – multiplicative step size for the exponentiated update
    # ------------------------------------------------------------------
    beta_start, beta_end = 5.0, 150.0          # start with a gentle smoothing
    eta_start, eta_end = 0.5, 0.01            # relatively aggressive early steps

    iteration = 0
    last_improve_iter = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        iteration += 1
        progress = (now - start_time) / (deadline - start_time)

        # Linear annealing of hyper‑parameters.
        beta = beta_start * (1.0 - progress) + beta_end * progress
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # --------------------------------------------------------------
        # 1. Convolution a * a (linear, not circular) via FFT.
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, fft_size)
        conv = np.fft.irfft(a_fft * a_fft, fft_size)[: 2 * n - 1]

        # Current objective (exact, no rounding).
        cur_max = conv.max()
        cur_obj = float(2.0 * n * cur_max)

        # Keep the best ever encountered.
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[{iteration:7d}] new best objective = {best_obj:.6f}")
            last_improve_iter = iteration

        # --------------------------------------------------------------
        # 2. Smooth max via soft‑max (log‑sum‑exp) and its gradient.
        # --------------------------------------------------------------
        # Soft‑max weights (stable computation).
        max_c = cur_max
        exp_terms = np.exp(beta * (conv - max_c))
        w = exp_terms / exp_terms.sum()                     # length 2n‑1, sums to 1

        # Compute gradient = 2 * (w ⋆ a)   (cross‑correlation).
        w_fft = np.fft.rfft(w, fft_size)
        # FFT of a reversed without an extra FFT:
        a_rev_fft = np.conj(a_fft) * phase_factor
        cross = np.fft.irfft(w_fft * a_rev_fft, fft_size)[: (2 * n - 1) + n - 1]  # length 3n‑2

        # gradient_i = 2 * cross[i + n - 1]
        grad = 2.0 * cross[(n - 1) : (n - 1 + n)]

        # --------------------------------------------------------------
        # 3. Exponentiated‑gradient (multiplicative) update on the simplex.
        # --------------------------------------------------------------
        # Clip the gradient to avoid overflow in the exponential.
        # Based on the current eta we keep the exponent in a safe range.
        clip_bound = 50.0 / max(eta, 1e-12)
        grad_clipped = np.clip(grad, -clip_bound, clip_bound)

        a = a * np.exp(-eta * grad_clipped)

        # Renormalise to the simplex (sum = 1).  This also discards any
        # rounding drift that may have accumulated.
        a_sum = a.sum()
        if a_sum == 0.0:
            # Very unlikely, but fall back to a uniform distribution.
            a = np.full(n, 1.0 / n, dtype=float)
        else:
            a /= a_sum

        # --------------------------------------------------------------
        # 4. Occasional random jitter moves (helps escape flat plateaus).
        # --------------------------------------------------------------
        if np.random.rand() < 0.015:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # Transfer up to half of the source mass.
            delta = 0.5 * a[src] * np.random.rand()
            if delta > 0.0:
                a[src] -= delta
                a[dst] += delta
                # No need to renormalise – total mass is preserved.

    # ------------------------------------------------------------------
    # Finished within the time budget.  Report the best solution found.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                         # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
