"""
Hybrid optimizer for the step‑function problem.

The objective to minimise is
    2·n·max(conv(a,a)) / (sum(a)**2)      (a ≥ 0)

Since the function is homogeneous we may normalise sum(a)=1 and minimise
    2·n·max(conv(a,a))

The algorithm works in three stages:

1.  *Adam on a smooth max* –  the soft‑max (log‑sum‑exp) provides a
    differentiable surrogate for the max.  The parameters are the
    logits of a soft‑max, guaranteeing non‑negativity and the simplex
    constraint.  Adam with a decaying learning‑rate explores the space
    quickly.

2.  *Projected‑gradient descent on the exact max* –  the sub‑gradient of
    the exact max is simply the reverse of the current maximal‑convolution
    entry.  A step is taken opposite to this gradient and the result is
    projected back onto the probability simplex (Duchi et al. 2008).

3.  *Simulated‑annealing hill‑climber* –  a stochastic mass‑transfer
    move (source → target) is proposed; it is accepted if it improves
    the objective or, with a temperature‑controlled probability,
    if it is slightly worse.  The source is bias‑chosen from the
    contributions to the few highest convolution entries, which gives
    a directed search.  After long stagnation the solution is
    perturbed and the search continues.

All three phases respect the time budget (default 1000 s).  The best
sequence found is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex (≥0, sum=1)."""
    # Duchi et al. 2008: O(n log n) sorting method
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection (should not happen)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase A – Adam optimisation of a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam on a smooth soft‑max approximation of max(conv(a,a))."""
    # logits for the soft‑max (a_i = exp(w_i) / sum exp(w))
    w = np.log(a0 + 1e-12)

    # Adam parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002    # a slightly larger start LR
    eps_start, eps_end = 5e-2, 1e-12
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 40_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on simplex
        w_shift = w - np.max(w)          # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        # exact objective for bookkeeping
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule (linear in time)
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smoothed max via log‑sum‑exp
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()           # softmax over convolution entries

        # gradient of smooth objective wrt a (see analysis)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)         # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]   # n‑length slice ∂/∂a
        grad_a = 4.0 * n * segment[::-1]      # (n,)

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter
        if np.random.rand() < 0.007:
            w += np.random.normal(scale=0.015, size=n)

        # restart on stagnation
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Projected gradient descent on the exact max
# ----------------------------------------------------------------------
def _projgrad_phase(a_start: np.ndarray,
                    n: int,
                    start: float,
                    deadline: float,
                    best_a: np.ndarray,
                    best_val: float,
                    target_val: float = 1.503) -> np.ndarray:
    """
    Simple projected‑gradient descent on the exact max objective.
    Uses the exact sub‑gradient of max(conv) and projects onto the simplex.
    """
    a = a_start.copy()
    cur_val = _objective(a)

    # learning‑rate schedule (large → small)
    lr_start, lr_end = 0.8, 0.01
    IMPROVE_LIMIT = 120_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # compute current maximal‑convolution index
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # sub‑gradient of max at k_max : g_i = 2 * a_{k_max-i}
        grad = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]

        # step size decreasing linearly
        progress = (now - start) / (deadline - start)
        lr = lr_start * (lr_end / lr_start) ** progress

        # take a step opposite to the gradient and project
        a_candidate = _project_simplex(a - lr * grad)

        cand_val = _objective(a_candidate)

        if cand_val < cur_val - 1e-14:
            a, cur_val = a_candidate, cand_val
            no_improve = 0
            if cand_val < best_val:
                best_val, best_a = cand_val, a.copy()
                print(f"[ProjGrad {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturbation to escape possible saddle points
            noise = np.random.normal(scale=0.02, size=n)
            a = best_a.copy()
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            cur_val = _objective(a)
            no_improve = 0
            print(f"[ProjGrad {iteration:7d}] perturb after stagnation")

        iteration += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – Simulated‑annealing hill‑climber (mass transfers)
# ----------------------------------------------------------------------
def _hillclimb_sa(a_start: np.ndarray,
                  n: int,
                  start: float,
                  deadline: float,
                  best_a: np.ndarray,
                  best_val: float,
                  target_val: float = 1.503) -> np.ndarray:
    """Targeted SA hill‑climber that moves a small amount of mass between
    indices while allowing occasional uphill moves."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # schedule parameters
    max_frac_start, max_frac_end = 0.55, 0.015
    T_start, T_end = 0.015, 1e-6          # temperature for SA
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    iteration = 0
    K_active = 3   # number of top convolution entries to bias the source selection

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog
        T = T_start * (1.0 - prog) + T_end * prog

        # --- active set (top K entries) ---------------------------------
        top_idxs = np.argpartition(-b, K_active)[:K_active]

        # --- source bias from contributions to the active set ------------
        contrib = np.zeros_like(a)
        for k in top_idxs:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib[mask] += a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # --- uniformly pick a different target index --------------------
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # --- amount of mass to move ------------------------------------
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # --- incremental update of the convolution (O(n)) ---------------
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms (always inside bounds)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # --- acceptance (SA) --------------------------------------------
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            # uphill acceptance probability
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[HillSA {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # big perturbation around the best known point
            a = best_a.copy()
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillSA {iteration:7d}] perturb after stagnation")

        iteration += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the given time budget.
    Returns a list of floats (Σ = 1) that (hopefully) minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a short safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a slightly larger n gives the optimiser more freedom
    # without a prohibitive computational cost.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previously found construction if
    # available, otherwise fall back to a heavy‑tailed (arcsine‑like)
    # distribution that is known to give a fairly flat autocorrelation.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        # arcsine‑like shape: f(x) ∝ 1 / sqrt(x·(1‑x))
        idx = np.arange(n, dtype=float) + 0.5
        a0 = 1.0 / np.sqrt(idx * (n + 1 - idx) + 1e-6)
        a0 = np.maximum(a0, 0.0)

    a0 /= a0.sum()                     # enforce Σa = 1
    best_a = a0.copy()
    best_val = float('inf')
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – smooth‑max Adam optimisation (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_phase(a0, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase B – projected gradient descent on the exact max (≈ 20 %)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + 0.55 * budget_s
    best_a = _projgrad_phase(best_a, n,
                             start=phase_a_deadline,
                             deadline=phase_b_deadline,
                             best_a=best_a,
                             best_val=best_val,
                             target_val=TARGET)
    best_val = _objective(best_a)       # refresh exact value

    # ------------------------------------------------------------------
    # Phase C – simulated‑annealing hill‑climber (remaining time)
    # ------------------------------------------------------------------
    best_a = _hillclimb_sa(best_a, n,
                           start=phase_b_deadline,
                           deadline=deadline,
                           best_a=best_a,
                           best_val=best_val,
                           target_val=TARGET)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
