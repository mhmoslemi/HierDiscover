def binary_split_reorder(a):
    n = len(a)
    sorted_vals = np.sort(a)[::-1]  # descending
    result = np.empty(n, dtype=float)
    positions = list(range(n))
    def recurse(pos_list, val_idx):
        if not pos_list:
            return val_idx
        mid = len(pos_list)//2
        index = pos_list[mid]
        result[index] = sorted_vals[val_idx]
        # left segment
        val_idx = recurse(pos_list[:mid], val_idx+1)
        # right segment
        val_idx = recurse(pos_list[mid+1:], val_idx)
        return val_idx
    recurse(positions, 0)
    return result

"""
Optimiser for the self–convolution bound.

The algorithm combines many restarts of an L‑BFGS optimiser on a
log‑sum‑exp smoothing of the ‑∞ norm, a collection of deterministic
reorderings (anti‑symmetric, bit‑reversal and binary‑split), a short
simulated‑annealing phase that searches directly over permutations,
and a final hill‑climbing phase that moves mass between entries.
All phases respect the 1000 s wall‑clock limit and keep the
sequence non‑negative and normalised (∑a_i = 1).  A best sequence
encountered during the run is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS wrapper (must be set before call)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence

def _next_pow2(x: int) -> int:
    """Return the next power of two greater or equal to ``x``."""
    return 1 << (x - 1).bit_length()

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # numerical noise may produce tiny negative values – clip them
    np.maximum(conv, 0, out=conv)
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate for the max‑convolution objective.

    Parameters
    ----------
    w : ndarray
        Unconstrained parameters; the actual step heights are obtained
        by a soft‑max transformation ``a_i = exp(w_i)/∑exp(w)``.
    eps : float
        Smoothing parameter – smaller values approximate the true max.
    n : int
        Length of the step function.

    Returns
    -------
    L : float
        Smoothed objective value ``2·n·max(conv) + 2·n·eps·log‑sum‑exp``.
    grad_w : ndarray
        Gradient of ``L`` with respect to the parameters ``w``.
    """
    # Soft‑max – guarantees a_i ≥ 0 and ∑a_i = 1
    w_shift = w - np.max(w)          # improve numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # Self‑convolution a*a
    conv = _full_convolution(a)     # length 2·n‑1
    max_conv = conv.max()

    # Smooth approximation of the max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)    # log ∑_k exp((conv_k‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # Gradient w.r.t. a (closed‑form, see derivation in the original code)
    weights = np.exp(shifted)
    p = weights / weights.sum()     # probability distribution over k

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)          # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]        # central n entries
    grad_a = 4.0 * n * seg[::-1]

    # Chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values with the smallest ones at opposite ends.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large on left
        new_a[n - 1 - i] = sorted_vals[i]          # small on right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

def _bit_reverse_permutation(n: int) -> np.ndarray:
    """
    Return a permutation array that puts index i at the position obtained
    by reversing the binary representation of i (using ⌈log2 n⌉ bits).
    """
    bits = (n - 1).bit_length()
    rev = np.empty(n, dtype=int)
    for i in range(n):
        b = '{:0{width}b}'.format(i, width=bits)
        rev_i = int(b[::-1], 2)
        rev[i] = rev_i
    # ``rev`` may contain values ≥ n; map them back into range by modulo
    rev = rev % n
    # ``rev`` may have duplicates – turn it into a permutation by sorting
    # and taking the order of the sorted values.
    order = np.argsort(rev, kind='stable')
    return order

def _bit_reverse_reorder(a: np.ndarray) -> np.ndarray:
    """Reorder ``a`` according to the bit‑reversal permutation."""
    n = a.shape[0]
    perm = _bit_reverse_permutation(n)
    return a[perm]

def _binary_split_reorder(a: np.ndarray) -> np.ndarray:
    """
    Recursively place the largest remaining value at the centre of the
    current interval, then descend on the left and right halves.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]        # descending
    result = np.empty(n, dtype=float)

    def recurse(indices, start, stop, value_idx):
        """Fill ``result`` at positions given by ``indices[start:stop]``."""
        if start >= stop:
            return value_idx
        # Choose the median position of the current slice
        mid = (start + stop) // 2
        pos = indices[mid]
        result[pos] = sorted_vals[value_idx]
        # Recurse on left half
        value_idx = recurse(indices, start, mid, value_idx + 1)
        # Recurse on right half
        value_idx = recurse(indices, mid + 1, stop, value_idx)
        return value_idx

    all_idx = np.arange(n)
    recurse(all_idx, 0, n, 0)
    return result

def _evaluate_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective (avoids the heavy security checks
    performed by ``evaluate_sequence``).  Returns the exact same
    quantity as ``evaluate_sequence`` for a normalized vector (∑a = 1).
    """
    n = a.shape[0]
    max_conv = _full_convolution(a).max()
    # The denominator is 1 because we keep the vector normalised.
    return float(2.0 * n * max_conv)

def _perm_hill_climb(a: np.ndarray, cur_val: float,
                     deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style search over permutations of ``a``.
    Only swaps are performed – the multiset of values stays unchanged.
    The function stops when the deadline is reached.
    Returns the best permutation found and its evaluation.
    """
    cur_a = a.copy()
    cur_score = cur_val
    rng = np.random.default_rng()
    # Initial temperature – roughly the typical improvement size we expect.
    T0 = 0.1
    # Linear temperature decay
    while time.time() < deadline:
        # Linear cooling based on remaining fraction of total time for this phase.
        remaining = deadline - time.time()
        total = deadline - (deadline - (deadline - time.time()))  # dummy to just get a positive number
        # Let temperature decay to almost zero after the phase
        T = T0 * (remaining / (deadline - (deadline - remaining)))  # just a placeholder; we keep T constant for simplicity

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        new_a = cur_a.copy()
        new_a[i], new_a[j] = new_a[j], new_a[i]
        new_score = _evaluate_fast(new_a)
        if new_score < cur_score - 1e-14:
            cur_a = new_a
            cur_score = new_score
        else:
            # occasional uphill move
            if rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
                cur_a = new_a
                cur_score = new_score
    return cur_a, cur_score

def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    Local improvement by random mass transfers and swaps.
    Accept only strict improvements (except for an occasional
    simulated‑annealing step at the very end).
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    # Parameters
    max_transfer = 0.2      # up to 20 % of donor's mass
    while time.time() < deadline:
        if rng.random() < 0.15:                     # perform a pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 0.0:
                continue
            delta = cur_a[i] * max_transfer * rng.random()
            if delta <= 0.0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Normalise (the total mass is unchanged, but numerical drift may appear)
        # Not needed because we transferred mass conservatively.

        new_val = _evaluate_fast(new_a)
        if new_val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = new_val
            # optional: print progress
            print(f"[refine] improvement → {cur_val:.6f}")
    return cur_a, cur_val

_GLOBAL_EPS = None   # smoothing parameter
_GLOBAL_N = None     # length of the sequence

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate for the max‑convolution objective.
    Returns (loss, gradient) with respect to the unconstrained vector w.
    """
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()                     # soft‑max => a_i ≥ 0, Σa_i = 1

    conv = _full_convolution(a)      # a*a, length 2·n‑1
    max_conv = conv.max()

    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # Gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)           # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]          # central n entries
    grad_a = 4.0 * n * seg[::-1]

    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize``."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values on the left, small values on the right (pairwise)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]
        new_a[n - 1 - i] = sorted_vals[i]
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

def _bit_reverse_permutation(n: int) -> np.ndarray:
    """Permutation that reverses the binary representation of each index."""
    bits = (n - 1).bit_length()
    rev = np.empty(n, dtype=int)
    for i in range(n):
        b = i
        rev_i = 0
        for _ in range(bits):
            rev_i = (rev_i << 1) | (b & 1)
            b >>= 1
        rev[i] = rev_i
    # The raw reverse may exceed n‑1; we reduce modulo n and then sort.
    rev = rev % n
    order = np.argsort(rev, kind='stable')
    return order

def _bit_reverse_reorder(a: np.ndarray) -> np.ndarray:
    """Reorder a according to the bit‑reversal permutation."""
    perm = _bit_reverse_permutation(a.shape[0])
    return a[perm]

def _binary_split_reorder(a: np.ndarray) -> np.ndarray:
    """
    Recursively put the largest remaining value in the centre of the
    current interval, then continue on the left and right halves.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]      # descending
    result = np.empty(n, dtype=float)

    def recurse(indices, start, stop, val_idx):
        """Fill result at positions ``indices[start:stop]``."""
        if start >= stop:
            return val_idx
        mid = (start + stop) // 2
        pos = indices[mid]
        result[pos] = sorted_vals[val_idx]
        # left half
        val_idx = recurse(indices, start, mid, val_idx + 1)
        # right half
        val_idx = recurse(indices, mid + 1, stop, val_idx)
        return val_idx

    all_idx = np.arange(n)
    recurse(all_idx, 0, n, 0)
    return result

def _evaluate_fast(a: np.ndarray) -> float:
    """Fast evaluation of the objective (∑a = 1)."""
    n = a.shape[0]
    max_conv = _full_convolution(a).max()
    return float(2.0 * n * max_conv)

def _perm_hill_climb(a: np.ndarray, cur_val: float,
                     deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    Local search over permutations of ``a`` using random swaps.
    Accepts only strict improvements (occasional random jumps are added
    with a tiny probability to avoid getting stuck).
    """
    cur_a = a.copy()
    cur_score = cur_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        new_a = cur_a.copy()
        new_a[i], new_a[j] = new_a[j], new_a[i]
        new_score = _evaluate_fast(new_a)
        if new_score < cur_score - 1e-14:
            cur_a = new_a
            cur_score = new_score
        else:
            if rng.random() < 1e-5:      # rare uphill jump
                cur_a = new_a
                cur_score = new_score
    return cur_a, cur_score

def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    Fine‑grained local improvement by random mass transfers and swaps.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    max_transfer = 0.2                 # up to 20 % of donor's mass
    while time.time() < deadline:

            new_a[i], new_a[j] = new_a[j], new_a[i]

            cur_a = new_a
            cur_val = new_val

            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _evaluate_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best (fast) = {best_val:.6f}")

        # --------------------------------------------------------------
        # Deterministic re‑orderings – try to decorrelate large entries
        # --------------------------------------------------------------
        for fn in (_anti_symmetric_reorder,
                   _bit_reverse_reorder,
                   _binary_split_reorder):
            a_ord = fn(best_a)
            val_ord = _evaluate_fast(a_ord)
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder {fn.__name__}] improved to {best_val:.6f}")

        # --------------------------------------------------------------
        # Short permutation hill‑climb (simulated annealing)
        # --------------------------------------------------------------
        if time.time() < deadline - 15.0:
            perm_deadline = time.time() + 8.0
            a_perm, val_perm = _perm_hill_climb(best_a, best_val,
                                                perm_deadline, n)
            if val_perm < best_val:
                best_val = val_perm
                best_a = a_perm.copy()
                print(f"[perm] improved to {best_val:.6f}")

        # --------------------------------------------------------------
        # Mass‑transfer hill‑climb – fine‑tune the actual heights
        # --------------------------------------------------------------
        if time.time() < deadline - 7.0:
            refine_deadline = deadline - 5.0
            best_a, best_val = _refine_hill_climb(best_a,
                                                  best_val,
                                                  refine_deadline,
                                                  n)

    # ------------------------------------------------------------------
    # Final evaluation with the official ``evaluate_sequence`` (includes
    # the extra security clipping performed by the judge).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = best_val

    return best_a.tolist()
