"""
Optimisation of a non‑negative step‑function for the provided
`evaluate_sequence` objective.

The algorithm consists of three phases, each using a different
optimisation principle and a separate time budget (the whole run is
limited to 1000 s).

1. **L2‑flattening (Phase C)** – minimise the variance of the
   autoconvolution `b = a∗a` by gradient descent on the loss
   `∑(b‑b̄)²`.  The gradient is obtained with two FFT‑based convolutions,
   followed by Euclidean projection onto the probability simplex.

2. **L∞‑subgradient descent (Phase M)** – directly target the
   objective `max(b)`.  For the (possibly multiple) indices where
   `b` attains its maximum we compute a sub‑gradient
   `∂max(b)/∂a_i = 2·a_{k‑i}` and perform a projected sub‑gradient step.

3. **Directed hill‑climbing (Phase H)** – a fast, greedy local search.
   The index `k_max` where the convolution peaks is identified; the
   contribution of each coordinate to that peak (`a_i·a_{k_max‑i}`) is
   used to pick a source (large contribution) and a target (small
   contribution).  Small mass transfers are evaluated with an
   `O(n)` incremental update of the convolution and accepted only if
   they strictly improve the true objective.

The best vector seen during the whole run is returned as a plain
Python list; the final score is printed using the supplied
`evaluate_sequence` wrapper.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0] = 0.0
    # renormalise to eliminate numerical drift
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Score helper (the internal objective equals the public one for sum=1)
# ----------------------------------------------------------------------
def _score_from_a(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) assuming ∑a = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return 2.0 * n * b.max()

        step = max(min_step, min(base_step, step))

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # Occasional jitter to escape shallow traps
        if np.random.rand() < 0.003:
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)

        # Restart after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[L2 {iter_cnt}] restart after stagnation")
        iter_cnt += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase M – sub‑gradient descent on max(b)
# ----------------------------------------------------------------------
def _phase_subgrad_max(a: np.ndarray,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the L∞‑objective."""
    n = a.shape[0]
    base_step = 0.5
    min_step = 0.001
    iter_cnt = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[L∞ {iter_cnt:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # sub‑gradient of max(b)
        k_max = int(np.argmax(b))
        grad = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        if i_low <= i_high:
            i_range = np.arange(i_low, i_high + 1)
            j_range = k_max - i_range
            grad[i_range] = 2.0 * a[j_range]

        # step schedule – simple linear decay
        step = base_step * (1.0 - iter_cnt / 200_000.0) + min_step * (iter_cnt / 200_000.0)
        step = max(min_step, min(base_step, step))

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional random perturbation
        if np.random.rand() < 0.001:
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[L∞ {iter_cnt}] stagnation – perturbing")
        iter_cnt += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase H – directed hill climbing on the exact objective
# ----------------------------------------------------------------------
def _phase_hillclimb(a_start: np.ndarray,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """Greedy local search moving mass from high‑contribution to low‑contribution coordinates."""
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start = 0.3
    max_frac_end = 0.02
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 120_000

    while time.time() < deadline:
        # recompute peak information
        k_max = int(np.argmax(b))
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")

        # contribution of each coordinate to the peak
        contr = np.zeros(n)
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        if i_low <= i_high:
            i_range = np.arange(i_low, i_high + 1)
            j_range = k_max - i_range
            contr[i_range] = a[i_range] * a[j_range]          # raw contribution (without the factor 2)

        # source = index with maximal contribution
        src = int(np.argmax(contr))
        # target = index among the least‑contributing (bottom 20 %)
        thresh = np.percentile(contr, 20)
        low_idx = np.where(contr <= thresh)[0]
        if low_idx.size == 0:
            tgt = np.random.randint(n)
        else:
            tgt = int(np.random.choice(low_idx))

        if src == tgt or a[src] <= 1e-15:
            # fall back to a random safe move
            src = int(np.argmax(a))
            tgt = np.random.randint(n)
            if tgt == src:
                continue

        # gradually shrink allowed transfer fraction
        progress = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - (deadline - time.time())))
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of the convolution (O(n))
        b_new = b.copy()
        # cross‑terms (2·δ·a) shifted by source/target
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val:
            # accept
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] improved best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # reset to the best seen so far plus a tiny perturbation
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – random perturb")

        iteration += 1
    return best_a

# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function to minimise `evaluate_sequence`.

    The routine runs three optimisation phases (L2‑flattening,
    sub‑gradient on the max‑norm, and directed hill‑climbing) while
    respecting the supplied time budget.  The best vector found is
    returned as a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin of half a second
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good compromise.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)          # enforce Σa = 1
    best_a = a.copy()
    best_val = _score_from_a(a)

    # ------------------------------------------------------------------
    # Allocate time slices for the three phases.
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.35 * budget_s
    phase_m_deadline = phase_c_deadline + 0.35 * budget_s
    # remaining time is for hill climbing
    # ------------------------------------------------------------------

    # -------------------- Phase C – L2 flattening --------------------
    best_a, best_val = _phase_l2(a, phase_c_deadline, best_a, best_val)

    # -------------------- Phase M – sub‑gradient on max ----------------
    best_a, best_val = _phase_subgrad_max(best_a, phase_m_deadline, best_a, best_val)

    # -------------------- Phase H – directed hill‑climbing ----------
    best_a = _phase_hillclimb(best_a, deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (using the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
