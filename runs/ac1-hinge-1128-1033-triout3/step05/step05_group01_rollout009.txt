def _guided_hill_climb(a, best_val, deadline, rng):
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05
    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        if ks.size == 0:
            break
        k = int(rng.choice(ks))
        # compute contributions
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])
        # choose destination low mass indices
        low_mass_idx = np.where(cur_a < cur_a.mean())[0]
        if low_mass_idx.size >= 2:
            p, q = rng.choice(low_mass_idx, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)
        # avoid same sum
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta
        if (new_a < 0).any():
            continue
        new_val = obj_from_a(new_a)
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)
    return cur_a, cur_val

"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from a uniform vector (or from the previous
   construction if ``height_sequence_1`` is present) and normalise it so
   that its entries sum to one.  The optimisation variables are the
   unconstrained log‑soft‑max parameters ``w`` (so that ``a = softmax(w)``).

2. **Smooth optimisation** – minimise a log‑sum‑exp surrogate of the
   max‑convolution using Adam.  The smoothing parameter ε is decreased
   on a logarithmic schedule (continuation).  This quickly drives the
   distribution towards shapes with a low convolution peak while keeping
   the optimisation stable.

3. **Random restarts** – after the continuation a few perturbed copies
   of the best solution are refined again (only the finest ε‑values are
   used).  This helps to escape local minima.

4. **Guided hill‑climbing** – using the exact objective
   ``2·n·max(conv)/(sum(a)^2)`` we repeatedly identify the lag where the
   convolution attains its maximum and move a small amount of mass away
   from the most offending pair of indices towards two low‑mass
   positions.  Only improvements are accepted; the step size is
   gradually reduced.  The process runs until the time budget expires.

The whole routine respects a 1000 s budget; intermediate progress and
the best evaluation seen so far are printed.  The returned list of
heights is a plain ``list[float]`` suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Soft‑max utilities
# ----------------------------------------------------------------------
def softmax(w: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate of the max‑convolution and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    together with ∂L/∂w.
    """
    # a = softmax(w)   (Σa = 1)
    a = softmax(w)

    # full autocorrelation a*a
    conv = _full_convolution(a)          # length 2·n‑1
    max_conv = conv.max()

    # log‑sum‑exp term
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)         # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # Gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()          # shape (2·n‑1,)

    # p_rev required for the convolution a * p_rev
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)        # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]      # middle n entries
    grad_a = 4.0 * n * seg[::-1]         # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)          # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Exact objective (fast version)
# ----------------------------------------------------------------------
def obj_from_a(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` without the safety clamping.
    The sum of a is assumed to be 1 (or very close to it).
    """
    n = a.shape[0]
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed surrogate
# ----------------------------------------------------------------------
def adam_optimize_w(w0: np.ndarray, eps: float, n: int,
                    max_iter: int, lr: float,
                    beta1: float, beta2: float,
                    eps_adam: float, deadline: float):
    """
    Adam optimiser on the surrogate L_ε.
    Returns the best w encountered, the corresponding a, and its exact score.
    """
    w = w0.copy()
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    best_w = w.copy()
    best_a = softmax(w)
    best_val = obj_from_a(best_a)

    t = 0
    for i in range(max_iter):
        if time.time() >= deadline:
            break
        t += 1
        L, grad = _smooth_obj_and_grad(w, eps, n)
        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w = w - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional exact evaluation
        if (i % 10) == 0 or i == max_iter - 1:
            a = softmax(w)
            cur_val = obj_from_a(a)
            if cur_val < best_val:
                best_val = cur_val
                best_w = w.copy()
                best_a = a.copy()
    return best_w, best_a, best_val


# ----------------------------------------------------------------------
# Continuation with decreasing ε
# ----------------------------------------------------------------------
def smooth_optimisation_phase(w_start: np.ndarray,
                              eps_schedule: np.ndarray,
                              n: int,
                              deadline: float,
                              rng: np.random.Generator):
    """
    Run Adam on each ε in the schedule (continuation).  Returns the
    best (w, a, value) seen during the phase.
    """
    w = w_start.copy()
    best_w = w.copy()
    best_a = softmax(w)
    best_val = obj_from_a(best_a)

    print(f"[smooth] start value = {best_val:.6f}")

    for eps in eps_schedule:
        if time.time() >= deadline:
            break
        # a moderate number of Adam steps per ε
        w, a, val = adam_optimize_w(w, eps, n,
                                    max_iter=200,
                                    lr=0.2,
                                    beta1=0.9,
                                    beta2=0.999,
                                    eps_adam=1e-8,
                                    deadline=deadline)
        if val < best_val:
            best_val = val
            best_w = w.copy()
            best_a = a.copy()
            print(f"[smooth ε={eps:.1e}] new best = {best_val:.6f}")
    return best_w, best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy descent on the *exact* objective.  At each step we locate the
    lag yielding the maximal convolution value, take the pair (i,j)
    contributing the most to that lag and transfer a small amount of
    mass to two low‑mass positions that do not recreate the offending
    lag.  The step size shrinks over time.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05   # initial fraction of mass to move

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        if ks.size == 0:
            break
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two low‑mass destination indices
        low = np.where(cur_a < cur_a.mean())[0]
        if low.size >= 2:
            p, q = rng.choice(low, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the offending lag
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform the mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        new_val = obj_from_a(new_a)
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Random‑restart wrapper
# ----------------------------------------------------------------------
def random_restart_phase(best_a: np.ndarray,
                         best_val: float,
                         n: int,
                         eps_schedule: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_restarts: int = 6):
    """
    Perform a few random restarts.  Each restart perturbs the log‑weights,
    runs a short continuation on the smallest ε‑values and finishes with
    the guided hill‑climbing.
    """
    w_base = np.log(best_a + 1e-12)
    cur_best_a = best_a.copy()
    cur_best_val = best_val

    for r in range(max_restarts):
        if time.time() >= deadline:
            break
        # perturb the current log‑weights
        w_pert = w_base + rng.normal(scale=0.5, size=n)

        # only the fine part of the schedule is needed
        tail_eps = eps_schedule[-10:]  # the ten smallest ε

        # short continuation
        w_opt, a_opt, val_opt = smooth_optimisation_phase(
            w_pert, tail_eps, n, deadline, rng)

        # final hill‑climbing
        a_opt, val_opt = _guided_hill_climb(a_opt, val_opt, deadline, rng)

        if val_opt < cur_best_val:
            cur_best_val = val_opt
            cur_best_a = a_opt.copy()
            w_base = np.log(cur_best_a + 1e-12)
            print(f"[restart {r+1}] new best = {cur_best_val:.6f}")

    return cur_best_a, cur_best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step function under the given evaluation
    budget.  Returns a list of floats representing the heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives the optimiser more freedom.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise sequence (reuse a previous construction if available)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = np.full(n, 1.0 / n)    # uniform start

    # enforce feasibility (non‑negative, ℓ₁‑norm = 1)
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 /= s

    w0 = np.log(a0 + 1e-12)          # soft‑max parameters
    best_a = a0.copy()
    best_val = obj_from_a(best_a)
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # ε‑schedule for the continuation (logarithmic from 5e‑3 to 1e‑12)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=25)

    # ------------------------------------------------------------------
    # Primary smooth optimisation (continuation)
    # ------------------------------------------------------------------
    w_opt, a_opt, val_opt = smooth_optimisation_phase(
        w0, eps_schedule, n, deadline, rng)

    if val_opt < best_val:
        best_val = val_opt
        best_a = a_opt.copy()
        print(f"[post‑smooth] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Random restarts + short continuations
    # ------------------------------------------------------------------
    a_res, val_res = random_restart_phase(
        best_a, best_val, n, eps_schedule, deadline, rng, max_restarts=6)

    if val_res < best_val:
        best_val = val_res
        best_a = a_res.copy()
        print(f"[post‑restarts] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final guided hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)
        print(f"[final hill] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided ``evaluate_sequence`` (the
    # function clamps values but our construction stays well inside the
    # allowed range).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = best_val

    return best_a.tolist()
