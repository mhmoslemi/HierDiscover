# convolution helpers (FFT)
def full_conv(x):
    n = x.shape[0]
    out_len = 2*n-1
    size = 1 << (out_len-1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx*fx, size)[:out_len]
    return conv

def conv_fft(x, y):
    n = len(x); m = len(y)
    out_len = n+m-1
    size = 1 << (out_len-1).bit_length()
    fx = np.fft.rfft(x, size); fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx*fy, size)[:out_len]
    return conv

def objective(a):
    n = a.shape[0]
    b = full_conv(a)
    return float(2 * n * b.max())

# projection onto simplex (sum 1, nonnegative)
def project_simplex(v):
    # Following algorithm from Wang & Carreira-Perpiñán
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

"""
Multi‑phase optimizer for the max‑convolution objective.

1. Random restarts with a smooth “soft‑max” Adam optimizer on a
   differentiable approximation of the max (log‑sum‑exp).  This
   discovers diverse non‑uniform step‑functions.

2. Projected gradient descent on the exact L∞‑objective.  The sub‑gradient
   of the current maximal convolution entry is computed analytically and
   the iterate is projected onto the probability simplex after each step.

3. Focused hill‑climber that repeatedly moves a small amount of mass from
   indices that contribute most to the current maximum to random
   destinations, accepting only improving moves.  After long stagnation
   the solution is perturbed.

The whole search runs as long as the time budget allows and the best
sequence seen is returned.
"""
import time
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) under the convention Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {w≥0, Σw = 1}. """
    # implementation based on Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return a unit vector at the largest entry
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth log‑sum‑exp approximation of max(conv)."""
    w = np.log(a0 + 1e-12)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 5e-2, 1e-12
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMP_LIMIT = 20_000
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # soft‑max
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1
        # schedule
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        # smoothed max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()
        # gradient of smoothed max w.r.t a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]
        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)
        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # occasional jitter
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.015, size=n)
        if no_improve > IMP_LIMIT:
            print(f"[Adam {iteration:7d}] Stagnation – restarting")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0
        iteration += 1
    return best_a, best_val

def _pgd_phase(a_start: np.ndarray,
               n: int,
               start: float,
               deadline: float,
               best_a: np.ndarray,
               best_val: float,
               target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """Projected (sub‑)gradient descent on the exact L∞‑objective."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    iteration = 0
    no_improve = 0
    IMP_LIMIT = 100_000
    while True:
        now = time.time()
        if now >= deadline:
            break
        # locate index of current maximal convolution entry
        k = int(np.argmax(b))
        # sub‑gradient of max entry
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros(n)
        g[mask] = 2.0 * a[idx[mask]]
        # step size schedule (linear decay)
        progress = (now - start) / (deadline - start)
        eta = 0.2 * (1.0 - progress) + 0.001 * progress
        a_next = a - eta * g
        a_next = _project_simplex(a_next)
        # evaluate
        b_next = _full_convolution(a_next)
        new_max = b_next.max()
        new_val = 2.0 * n * new_max
        if new_val < cur_val - 1e-12:
            a, b, cur_max, cur_val = a_next, b_next, new_max, new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[PGD {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # occasional acceptance to escape plateaus
            if np.random.rand() < 0.001:
                a, b, cur_max, cur_val = a_next, b_next, new_max, new_val
            else:
                no_improve += 1
        if no_improve > IMP_LIMIT:
            # slight random perturbation around current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            no_improve = 0
            print(f"[PGD {iteration:7d}] perturb after stagnation")
        iteration += 1
    return best_a, best_val

def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """Targeted hill‑climbing with mass transfers (same idea as before)."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    max_frac_start, max_frac_end = 0.5, 0.02
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        # locate max entry
        k_max = int(np.argmax(b))
        # contributions to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        # source proportional to contributions
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))
        if a[i] <= 1e-14:
            continue
        # target uniformly different from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue
        # incremental update of convolution (O(n))
        b_new = b.copy()
        # add contributions of moving delta from i to j
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta
        new_max = b_new.max()
        new_val = 2.0 * n * new_max
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] perturb after stagnation")
        iteration += 1
    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """Search for a non‑negative step‑function minimizing `evaluate_sequence`.

    The procedure consists of several independent trials with different
    random initialisations.  In each trial we run:
        1. a smooth‑max Adam optimiser,
        2. a projected‑gradient refinement on the exact L∞‑objective,
        3. a targeted hill‑climber.
    The best sequence seen before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin for final evaluation

    # try a few problem sizes – the theoretical lower bound does not depend on n,
    # but smaller n makes each iteration cheaper.
    sizes = [800, 1600, 3200, 6400]   # total budget will be split among them
    # allocate time roughly equally to each size
    per_size = (deadline - start_time) / len(sizes)

    global_best_a = None
    global_best_val = float('inf')
    TARGET = 1.5   # stop early if we beat this threshold

    for n in sizes:
        size_deadline = start_time + per_size
        if size_deadline > deadline:
            size_deadline = deadline
        # initialise a0 – try a Dirichlet random vector (sum = 1)
        a0 = np.random.dirichlet(alpha=np.ones(n))
        # keep a copy for the run
        best_a_local = a0.copy()
        best_val_local = float('inf')
        # Phase A – Adam on smoothed max
        best_a_local, best_val_local = _adam_phase(a0, n,
                                                   start=start_time,
                                                   deadline=size_deadline,
                                                   best_a=best_a_local,
                                                   best_val=best_val_local,
                                                   target_val=TARGET)
        # Phase B – projected gradient descent on exact objective
        best_a_local, best_val_local = _pgd_phase(best_a_local, n,
                                                  start=size_deadline,
                                                  deadline=size_deadline + (deadline - size_deadline) * 0.3,
                                                  best_a=best_a_local,
                                                  best_val=best_val_local,
                                                  target_val=TARGET)
        # Phase C – targeted hill‑climbing
        best_a_local, best_val_local = _hillclimb_phase(best_a_local, n,
                                                        start=size_deadline,
                                                        deadline=deadline,
                                                        best_a=best_a_local,
                                                        best_val=best_val_local,
                                                        target_val=TARGET)
        # update global best
        if best_val_local < global_best_val:
            global_best_val = best_val_local
            global_best_a = best_a_local.copy()
            print(f"[Overall] new global best = {global_best_val:.6f} with n={n}")

        # early exit if we have already reached an excellent bound
        if global_best_val <= 1.01:
            break

    # final evaluation (the wrapper will clip/scale as necessary)
    try:
        final_val = evaluate_sequence(global_best_a.tolist())
        print(f"[final] reported evaluation = {final_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return global_best_a.tolist()

"""
Multi‑phase optimizer for the max‑convolution objective.

The search consists of three stages:

1. **Adam on a smooth approximation** – the max of the convolution is
   replaced by a log‑sum‑exp with a temperature that decays to zero.
   This gives a differentiable surrogate that quickly leaves the
   uniform solution.

2. **Projected (sub‑)gradient descent on the exact L∞‑objective** –
   after the first phase the index `k_max` of the current maximal
   convolution entry is known.  The sub‑gradient of this entry is
   `g_i = 2·a_{k_max-i}` (zero when the index is out of range).  A
   gradient step is taken and the result is projected back onto the
   simplex (non‑negative entries that sum to one).

3. **Targeted hill‑climbing** – mass `δ` is transferred from a source
   index that contributes heavily to `k_max` to a random target index.
   Only strictly improving moves are kept; after a long stagnation
   the solution is perturbed to escape local minima.

All phases run inside the allotted time budget.  The best sequence
encountered is finally evaluated with `evaluate_sequence` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {w≥0, Σw = 1}."""
    # Duchi et al. (2008) – O(n log n) implementation
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fall back – put unit mass on the largest coordinate
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth approximation of max(conv(a,a))."""
    w = np.log(a0 + 1e-12)                     # logits for soft‑max
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 5e-2, 1e-12
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMP_LIMIT = 20_000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- simplex ↦ a ------------------------------------------------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam  {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedule ---------------------------------------------------
        prog = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # ----- smooth max via log‑sum‑exp --------------------------------
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()

        # ----- gradient w.r.t. a -----------------------------------------
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1:2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]          # shape (n,)

        # ----- chain rule through the soft‑max ----------------------------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update ------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to escape plateaus
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.015, size=n)

        # restart if stagnated
        if no_improve > IMP_LIMIT:
            print(f"[Adam  {it:7d}] stagnation – restarting")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – projected sub‑gradient descent on the exact L∞ objective
# ----------------------------------------------------------------------
def _pgd_phase(a_start: np.ndarray,
               n: int,
               start: float,
               deadline: float,
               best_a: np.ndarray,
               best_val: float,
               target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """Projected (sub‑)gradient descent on 2·n·max(conv(a,a))."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    it = 0
    no_improve = 0
    IMP_LIMIT = 80_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        k = int(np.argmax(b))                     # index of current max
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros(n)
        g[mask] = 2.0 * a[idx[mask]]              # sub‑gradient of the max entry

        # linear decay of step size
        prog = (now - start) / (deadline - start)
        eta = 0.2 * (1.0 - prog) + 0.001 * prog

        a_next = a - eta * g
        a_next = _project_simplex(a_next)

        b_next = _full_convolution(a_next)
        new_max = b_next.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            a, b, cur_max, cur_val = a_next, b_next, new_max, new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[PGD   {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # rare uphill acceptance to avoid dead‑ends
            if np.random.rand() < 0.001:
                a, b, cur_max, cur_val = a_next, b_next, new_max, new_val
            else:
                no_improve += 1

        if no_improve > IMP_LIMIT:
            # a small perturbation around the current global best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            no_improve = 0
            print(f"[PGD   {it:7d}] perturb after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """Local search that moves a small amount of mass between coordinates."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    max_frac_start, max_frac_end = 0.5, 0.02
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        prog = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - prog) + max_frac_end * prog

        # ----- current maximal convolution entry ------------------------
        k_max = int(np.argmax(b))

        # ----- contribution of each index to that entry -----------------
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- choose a source (biased to large contributors) ---------
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # ----- choose a target uniformly, distinct from i ---------------
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # ----- amount of mass to move ----------------------------------
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the convolution (O(n)) -------------
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- accept only improving moves -------------------------------
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill  {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling --------------------------------------
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill  {it:7d}] perturb after stagnation")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 8000 gives enough degrees of freedom without
    # overwhelming the time budget.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise a – Dirichlet gives a random distribution on the simplex.
    # --------------------------------------------------------------
    a0 = np.random.dirichlet(np.ones(n))

    best_a = a0.copy()
    best_val = float('inf')
    TARGET = 1.5   # early‑exit threshold (sub‑optimal but cheap)

    # --------------------------------------------------------------
    # Phase A – smooth‑max Adam optimisation (~35 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * budget_s
    best_a, best_val = _adam_phase(a0, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # --------------------------------------------------------------
    # Phase B – projected sub‑gradient descent on the exact max (≈30 %)
    # --------------------------------------------------------------
    phase_b_deadline = start_time + 0.65 * budget_s
    best_a, best_val = _pgd_phase(best_a, n,
                                  start=phase_a_deadline,
                                  deadline=phase_b_deadline,
                                  best_a=best_a,
                                  best_val=best_val,
                                  target_val=TARGET)

    # --------------------------------------------------------------
    # Phase C – targeted hill‑climbing on the exact objective (~remaining)
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb_phase(best_a, n,
                                        start=phase_b_deadline,
                                        deadline=deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=TARGET)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
