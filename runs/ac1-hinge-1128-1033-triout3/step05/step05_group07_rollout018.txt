Define function proj_simplex(v) that projects onto simplex:

def proj_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

(This algorithm ensures sum(w)=1).

   evaluate_sequence(...).  A projected hill‑climbing algorithm is used.
   It repeatedly transfers a small amount of mass from the coefficient
   that contributes most to the current peak of the self‑convolution to
   a coefficient that contributes the least, updating the convolution
   incrementally.  The move size shrinks linearly with time; occasional
   random “jitter” moves are allowed to escape plateaus.  The best
   sequence seen so far is returned.
   """
   import time
   import numpy as np

   # Helper: fast linear convolution via real FFT
   def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
       total = x.shape[0] + y.shape[0] - 1
       size = 1 << ((total - 1).bit_length())
       fx = np.fft.rfft(x, size)
       fy = np.fft.rfft(y, size)
       conv = np.fft.irfft(fx * fy, size)[:total]
       # eliminate tiny negatives caused by round‑off
       np.maximum(conv, 0, out=conv)

           conv[2 * i_src] -= delta2
           conv[2 * j_dst] -= delta2
           conv[i_src + j_dst] += 2.0 * delta2

       # escape possible plateaus.
       # ------------------------------------------------------------------

           dst = rng.integers(n - 1)

                   cur_max = jitter_max

                       best_a = a.copy()
                       print(f"[{iteration:8d}] jitter improved best = {best_obj:.6f}")
                       last_improve_iter = iteration

                   conv[src:src + n] += 2.0 * delta_j * a_tmp

                   conv[2 * dst] -= delta2j
                   conv[src + dst] += 2.0 * delta2j

       # accumulated floating‑point drift.
       # ------------------------------------------------------------------

                    conv[dst:dst + n] -= 2.0 * delta_j * a_tmp
                    conv[2 * src] -= delta2j
                    conv[2 * dst] -= delta2j
                    conv[src + dst] += 2.0 * delta2j

        # ------------------------------------------------------------------
        # Periodic full recompute to avoid accumulated floating‑point drift.
        # ------------------------------------------------------------------
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a, a)

        # ------------------------------------------------------------------
        # Simple stagnation detection – if no improvement for a long time,
        # re‑normalise to suppress tiny numerical errors.
        # ------------------------------------------------------------------
        if iteration - last_improve_iter > 200_000:
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)
            last_improve_iter = iteration

    # ----------------------------------------------------------------------
    # Final reporting
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
