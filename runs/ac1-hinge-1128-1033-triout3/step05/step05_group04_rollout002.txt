Propose a step‑function (non‑negative sequence) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of two stages:

1️⃣  **Magnitude‑flattening (Gerchberg–Saxton style).**
    We iteratively project the sequence onto two sets:
      – the probability simplex (non‑negative, sum = 1);
      – the set of sequences whose real FFT has (approximately) constant
        magnitude.  This drives the autocorrelation (the self‑convolution)
        toward a nearly flat shape, which dramatically reduces the peak.
    The projection onto the magnitude set is performed by a single FFT,
    replacement of the magnitude by a constant (chosen to keep the energy
    unchanged) and an inverse FFT.  After each iteration the best sequence
    (measured by the true objective) is recorded.

2️⃣  **Fine‑grained polishing.**
    Starting from the best candidate of stage 1 we run a soft‑max Adam
    optimiser that directly minimises a smooth approximation of the
    “max‑convolution’’ objective.  The optimisation uses the same
    convolution routine and projects after every Adam step onto the
    simplex, staying within the feasible region.

The whole routine respects the supplied time budget (default 1000 s),
keeps the best sequence seen and finally returns it after clipping it
to the limits required by `evaluate_sequence`.  If a previous solution
is stored in the global variable `height_sequence_1` it is used as a warm
start; otherwise a random start (or a modest mixture of simple shapes)
is created.
"""

import time
import numpy as np
import math
import sys

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1}
    using the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full linear convolution using a real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

        # A simple random start – later we will improve it.
        a = rng.random(n)
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Helper: evaluate the current objective (sum is guaranteed to be 1).
    # ------------------------------------------------------------------
    def _current_score(seq):
        conv = _full_convolution_fft(seq, seq)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Keep track of the best solution seen so far.
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_val = _current_score(best_seq)
    print(f"[init] n = {n}, initial_score = {best_val:.12f}")

    # ------------------------------------------------------------------
    # ---------- Phase 1 – magnitude‑flattening (Gerchberg–Saxton) ----------
    # ------------------------------------------------------------------
    # FFT length (next power of two ≥ 2n‑1) – used for all transforms.
    L = 1 << ((2 * n - 1) - 1).bit_length()

    # Project onto constant‑magnitude spectrum.
    # The constant magnitude is chosen each iteration so that the total
    # energy (sum of squares) is preserved:  |A_k| = sqrt( sum(a_i^2) / L )
    it = 0
    stagn = 0
    phase1_deadline = deadline * 0.55          # ~55 % of budget to this phase
    while time.time() < phase1_deadline:
        it += 1

        # ----- (a) forward FFT of the zero‑padded signal -----------------
        a_pad = np.zeros(L, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)                 # complex, length L/2+1

        # ----- (b) replace magnitude by a constant --------------------
        s2 = np.dot(a, a)                       # sum of squares of a
        const_mag = math.sqrt(s2 / L)
        # keep the phase, enforce constant magnitude
        A_new = const_mag * np.exp(1j * np.angle(A))

        # ----- (c) inverse FFT to obtain a candidate time signal --------
        a_pad_new = np.fft.irfft(A_new, n=L)
        a_candidate = a_pad_new[:n]

        # ----- (d) projection onto the probability simplex -------------
        a_candidate = _project_simplex(a_candidate)

        # ----- evaluate ------------------------------------------------
        cur_val = _current_score(a_candidate)
        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_seq = a_candidate.copy()
            stagn = 0
            print(f"[phase1 it {it:6d}] new_best = {best_val:.12f}")
        else:
            stagn += 1

        a = a_candidate

        # occasional random jitter to escape plateaus
        if stagn > 2000:
            jitter = rng.normal(scale=0.006, size=n)
            a = _project_simplex(a + jitter)
            stagn = 0
            print(f"[phase1 jitter] it {it}")

    # ------------------------------------------------------------------
    # ---------- Phase 2 – soft‑max Adam polishing (same as before) ----------
    # ------------------------------------------------------------------
    # Initialise Adam state with the best sequence from phase 1.
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam hyper‑parameters
    lr = 0.08
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr_min = 1e-6
    lr_decay = 0.99997

    # Soft‑max temperature (controls the smooth approximation of max)
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    eps = max(max_conv * 0.5, 5e-12)
    eps_min = max(max_conv * 1e-6, 5e-12)
    eps_decay = 0.99992

    it = 0
    stagn = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution and current score -----------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_seq = a.copy()
            stagn = 0
            print(f"[phase2 it {it:6d}] new_best = {best_val:.12f}")
        else:
            stagn += 1

        # ----- soft‑max weighting of convolution entries ---------------
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # ----- gradient of the weighted sum (cross‑correlation) ---------
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        beta1_t = beta1 ** it
        beta2_t = beta2 ** it
        m_hat = m / (1.0 - beta1_t)
        v_hat = v / (1.0 - beta2_t)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = _project_simplex(a + step)

        # ----- schedules ------------------------------------------------
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        # ----- occasional random perturbation ---------------------------
        if stagn > 3000:
            a = _project_simplex(a + rng.normal(scale=0.004, size=n))
            stagn = 0
            lr = max(lr_min, lr * 1.3)
            print(f"[phase2 perturb] it {it}")

    # ------------------------------------------------------------------
    # Final reporting.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:
        final_score = evaluate_sequence(final_seq)
    except Exception as e:
        # In the very unlikely case evaluate_sequence is not present,
        # fall back to the internally computed score.
        final_score = 2.0 * n * max(_full_convolution_fft(best_seq, best_seq))
        print("[warning] evaluate_sequence not available:", e)

    print(f"[result] final_score = {final_score:.12f}, length = {len(final_seq)}")
    return final_seq
