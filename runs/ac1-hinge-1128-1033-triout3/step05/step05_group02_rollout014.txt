def evaluate_fast(a):
    # a assumed nonnegative
    n = a.shape[0]
    # compute convolution via FFT
    size = 1 << ((2*n-1).bit_length())
    # using rfft and irfft for real FFT.
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    # avoid division by zero
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
propose_candidate
-----------------
A hybrid optimizer for the step‑function evaluation problem.

The routine consists of three stages:

1. **Smooth surrogate optimisation** – multiple L‑BFGS restarts on a
   log‑sum‑exp approximation of the max‑convolution.  A soft‑max will
   guarantee non‑negativity and a fixed ℓ₁‑norm, while the surrogate
   objective approximates the true objective but is smooth and has a
   closed‑form gradient (see `_smooth_obj_and_grad`).

2. **Anti‑symmetric re‑ordering** – the best shape from stage 1 is
   re‑ordered by pairing large with small values (function
   `_anti_symmetric_reorder`).  This dramatically reduces the peak of
   the linear convolution without changing the multiset of heights.

3. **Simulated‑annealing permutation refinement** – a fast FFT based
   evaluator (`_eval_fast`) repeatedly proposes swaps of two positions.
   Moves are accepted if they improve the objective or with a
   temperature‑controlled probability, allowing the algorithm to escape
   local minima.  The temperature is slowly cooled; the best sequence
   found during this phase is retained.

Finally a short hill‑climbing pass (mass transfers + swaps) is performed
to polish the result.  The whole search respects a 1000 s budget and
always returns a feasible non‑negative list of floats.

The implementation uses only NumPy, SciPy and Python’s standard library.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate wrapper (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # tiny negative values can appear due to round‑off – clamp them
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the exact objective defined in ``evaluate_sequence``.
    The array ``a`` is assumed to be non‑negative and already satisfies the
    clipping limits (0 ≤ a_i ≤ 1000).  Returns the same value as
    ``evaluate_sequence(a.tolist())`` but is much faster.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε (log‑sum‑exp smoothing of the max) and its
    gradient w.r.t. the soft‑max parameters ``w``.
    """
    # ----- soft‑max (guarantees positivity and Σa = 1) -----
    w_shift = w - np.max(w)                # improve numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # ----- auto‑convolution a * a (full) -----
    conv = _fft_convolve_self(a)           # length 2·n‑1
    max_conv = conv.max()

    # ----- smooth approximation of max(conv) -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log∑ exp((conv−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (closed‑form) -----
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability distribution over k

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)   # helper defined below
    # middle n entries correspond to index range [n-1, 2n-2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Helper: convolution used inside the gradient (FFT based, same as above)
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# Bind the helper to the function (so it can be called from inside)
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristic
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce an ordering where large values are paired with small ones
    when the sequence is reversed.  This simple deterministic heuristic
    often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large to the left
        new_a[n - 1 - i] = sorted_vals[i]      # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]   # keep the middle element
    return new_a

# Simulated annealing on permutations (swap moves only)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):

    Classic SA on the permutation of ``a``.  Swaps preserve the ℓ₁‑norm.
    ``cur_val`` is the evaluation of ``a`` (computed with ``_eval_fast``).
    Returns the best sequence and its value found before ``deadline``.
    """

    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initialise temperature – a small fraction of the current value
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99995

    iter_cnt = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:

                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {iter_cnt})")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        # cooling
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        iter_cnt += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps) – same as in the
# reference implementation but uses the fast evaluator.
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # 20 % chance of a pure swap, else a mass transfer
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # leave a small safety margin
    print(f"[init] time budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – 4000 works well for the FFT‑based code
    # --------------------------------------------------------------
    n = 4000
    _global = globals()
    _global['_GLOBAL_N'] = n

    # --------------------------------------------------------------
    # Initialise from a known construction if available
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to requested size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)      # enforce non‑negativity
    a0 = a0 / a0.sum()           # normalise to ℓ₁ = 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # L‑BFGS schedule (log‑space smoothing)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-10), num=15)

    # ------------------------------------------------------------------
    # Multi‑restart L‑BFGS phase
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 30.0:   # reserve ~30 s for later phases
        restart += 1
        print(f"[restart {restart}]")
        # random soft‑max parameters (wide variance → diverse shapes)
        w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 20.0:
                break
            _global['_GLOBAL_EPS'] = eps

            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # exact evaluation of the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ----------------------------------------------------------
        # Anti‑symmetric re‑ordering of the best sequence from this restart
        # ----------------------------------------------------------
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing on the permutation of the best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("[SA] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val, deadline - 10.0)

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing phase")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final value (using the official evaluator)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required
    return best_a.tolist()
