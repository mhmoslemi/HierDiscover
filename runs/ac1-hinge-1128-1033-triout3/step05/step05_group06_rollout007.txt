def run_adam(initial_a, time_budget):
    start = time.time()
    n = len(initial_a)
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))
    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5  # large base lr
    best_score = np.inf
    best_vec = np.copy(initial_a)
    last_improve_iter = 0
    iter = 0
    # tolerance schedule parameters
    init_tol_factor = 0.1  # start with 10% tolerance
    max_iter_est = 500000  # approximate # iterations we anticipate (used for tolerance schedule)
    while time.time() - start < time_budget - 0.5:
        iter += 1
        # softmax to get vector a
        a = softmax(x)
        # compute convolution via FFT
        conv = full_convolution_fft(a)
        max_b = conv.max()
        # tolerance factor decays
        # factor = max(1e-12, init_tol_factor * max(1 - iter / max_iter_est, 0.0))
        # compute threshold = max_b * (1 - factor)
        # Instead we compute tolerance absolute
        tol_factor = init_tol_factor * max(1 - iter / max_iter_est, 0.0)
        if tol_factor > 0:
            threshold = max_b * (1 - tol_factor)
            active = np.where(conv >= threshold)[0]
        else:
            active = np.where(conv >= max_b - max_b * 1e-12 - 1e-15)[0]
        # gradient
        grad = grad_log_params(x, a, conv, max_b, active)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter)
        v_hat = v / (1.0 - beta2 ** iter)
        lr = lr0 / (np.sqrt(iter) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score

"""
search for a low‑peak auto‑convolution step function.

The evaluation is invariant to scaling, so we work with a probability vector
(sum = 1).  The objective to minimise is

    F(a) = 2·n·max_k (a * a)[k]                (n = len(a)).

We use a projected Adam optimiser on the log‑parameters (softmax) with a
sub‑gradient of the max‑convolution.  The optimiser runs for as long as the
time budget permits, keeping the best vector seen.  After Adam finishes a
tiny local‑perturbation phase tries to improve the result further.

The algorithm is deliberately kept simple (no closures, no lambda) and uses
only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    size = next_pow2(2 * n)            # zero‑padding size (>= 2n)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def score_vector(a: np.ndarray) -> float:
    """Objective value for a probability vector a (sum = 1)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. log‑parameters x (a = softmax(x)).
    active_idx contains the indices where the maximum is attained (or within
    numerical tolerance).
    """
    n = a.shape[0]
    G = np.zeros_like(a)

    # accumulate contributions from each active convolution index
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]

    # sub‑gradient for the log‑parameters (see derivation in the documentation)
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad


# ----------------------------------------------------------------------
# Parametric families (used for initial seeds)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    return a / a.sum()


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    return a / a.sum()


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimiser working directly on the log‑probabilities
# ----------------------------------------------------------------------


def run_adam(initial_a: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on the log‑parameters for at most `time_budget`
    seconds.  Returns the best probability vector seen and its score.
    """
    start = time.time()
    n = initial_a.shape[0]

    # initialise log‑parameters (avoid log(0))
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5                     # initial learning rate in log‑space

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    # constants for re‑heat
    REHEAT_AFTER = 20000          # iterations without improvement

    while time.time() - start < time_budget - 0.5:
        iteration += 1

        # ----- from log‑space to probability vector -----
        a = softmax(x)

        # ----- convolution and max -----
        conv = full_convolution_fft(a)
        max_b = conv.max()

        # ----- active set (indices that realise the maximum) -----
        # a very tiny tolerance avoids missing numerically equal maxima
        tol = max_b * 1e-12 + 1e-15
        active = np.where(conv >= max_b - tol)[0]

        # ----- gradient w.r.t. log‑parameters -----
        grad = grad_log_params(x, a, conv, max_b, active)

        # ----- Adam update (standard bias‑corrected version) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of step size (helps long runs)
        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- bookkeeping -----
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # ----- re‑heat if stuck for a long time -----
        if iteration - last_improve > REHEAT_AFTER:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

    # fall back to the initial vector if optimisation never improved
    if best_vec is None:
        best_vec = initial_a.copy()
        best_score = score_vector(best_vec)

    return best_vec, best_score

            candidates.append(exp_distribution(n, lam))

        for _ in range(5):

        # uniform distribution (baseline)

        # pick the best candidate according to the fast score
        best_score = np.inf
        best_vec = None
        for a in candidates:
            sc = score_vector(a)
            if sc < best_score:
                best_score = sc
                best_vec = a
        initial_vec = best_vec
        print(f"Deterministic seed selected (length {n}, score {best_score:.6f})")

    # ------------------------------------------------------------------
    # 3. Optimisation phase (almost the whole remaining budget)
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())

        i, j = np.random.choice(len(a), size=2, replace=False)

            print(f"Refine: improved score = {best_score:.6f}")

    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_vec)})")
    # convert to plain Python list as required by the harness
    return best_vec.tolist()
