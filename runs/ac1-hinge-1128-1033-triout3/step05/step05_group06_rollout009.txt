def run_smoothed_adam(initial_a, time_limit):
    x = np.log(np.maximum(initial_a, eps))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    lr0=0.2
    beta_cur = 5.0
    beta_max = 500.0
    beta_mult = 1.001 # each iteration
    best_score = np.inf
    best_vec = None
    iter = 0
    last_improve = 0
    start = time.time()
    n = len(initial_a)
    while time.time() - start < time_limit - 0.5:
        iter += 1
        a = softmax(x)
        conv = full_convolution_fft(a, a) # use same function for conv of same length?
        max_b = conv.max()
        # compute smoothed gradient
        # compute w
        max_shift = max_b
        # compute exponent
        exponents = beta_cur * (conv - max_shift)
        # subtract max for stability: already done
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.ones_like(w) / w.shape[0]
        else:
            w = w / w_sum
        # reversed a
        rev_a = a[::-1]
        # conv2 = full_convolution_fft(w, rev_a)
        conv2 = full_convolution_fft(w, rev_a)
        grad_a = 2.0 * conv2[n-1: n-1 + n]  # length n
        # gradient in logit space
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** iter)
        v_hat = v / (1 - beta2 ** iter)
        lr = lr0 / (np.sqrt(iter) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # update best
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iter
            print(f"Iter {iter:6d}: new best {best_score:.6f}")
        # reheat
        if iter - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0)
            v.fill(0)
            last_improve = iter
            print(f"Iter {iter:6d}: reheat")
        # update beta
        beta_cur = min(beta_max, beta_cur * beta_mult)
    return best_vec, best_score

"""
Step‑function search via smoothed max‑convolution + Adam.

We model the step heights as a probability vector `a` (a_i ≥ 0, Σa_i = 1).
The evaluation function simplifies to   score = 2·n·max_k (a*a)[k].

Instead of directly minimising the non‑smooth max, we minimise a smooth
approximation

    L_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])          (β > 0)

which converges to the true maximum as β → ∞.  The gradient of L_β
with respect to the log‑parameters `x` (a = softmax(x)) can be written
using only FFT‑based convolutions, allowing millions of cheap iterations.

A standard Adam optimiser (with a simple learning‑rate decay) moves the
parameters in the direction of this gradient.  β is increased
multiplicatively each iteration (capped at β_max) so that the
optimisation first explores a smooth landscape and later focuses on the
true peak of the auto‑convolution.

The routine:
  1. Pick a good initial probability vector (external best if available,
     otherwise a hand‑crafted “U‑shaped’’ distribution).
  2. Run the smoothed‑Adam optimiser for the whole time budget.
  3. If a little time remains, finish with a short run of the original
     hard‑max Adam (as in the reference implementation) to polish the
     solution.
  4. Return the best vector found (as a plain list of floats).

All operations are O(n log n) per iteration, so the optimiser can
perform tens of thousands of steps within the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (full linear convolution via real FFT)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of real vectors a and b using real FFT.
    Returns the full result of length len(a) + len(b) - 1.
    """
    size = a.shape[0] + b.shape[0] - 1
    # Next power‑of‑two for efficient FFT
    L = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[:size]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Objective (the value returned by evaluate_sequence for a probability vector)
# ----------------------------------------------------------------------
def score_vector(a: np.ndarray) -> float:
    """Exact score = 2·n·max_k (a*a)[k] for a probability vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for deterministic seeds (used only when no external seed)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def run_smoothed_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑probabilities of `initial_a`.
    Uses a temperature‑controlled log‑sum‑exp approximation to the
    max‑convolution.
    Returns (best_vector, best_score) found within `time_limit` seconds.
    """
    start = time.time()
    n = initial_a.shape[0]

    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.25          # base learning rate in log‑space

    # temperature schedule
    beta_cur = 5.0
    beta_max = 500.0
    beta_mul = 1.01      # multiplicative increase each iteration

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    # pre‑compute FFT lengths
    size_conv = 2 * n - 1          # a*a
    L1 = 1 << (size_conv - 1).bit_length()
    size_w_rev = 3 * n - 2        # w * rev(a)
    L2 = 1 << (size_w_rev - 1).bit_length()

    while time.time() - start < time_limit - 0.5:
        iteration += 1

        # ---------- probabilities ----------
        a = softmax(x)

        # ---------- self‑convolution ----------
        fa = np.fft.rfft(a, n=L1)
        conv = np.fft.irfft(fa * fa, n=L1)[:size_conv]
        max_b = conv.max()

        # ---------- smoothed max ----------
        # w_k ∝ exp(β·(conv_k – max_b))  (shift for numerical stability)
        exp_arg = beta_cur * (conv - max_b)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.ones_like(w) / w.shape[0]
        else:
            w = w / w_sum

        # ---------- gradient wrt a ----------
        rev_a = a[::-1]
        fw = np.fft.rfft(w, n=L2)
        frev = np.fft.rfft(rev_a, n=L2)
        conv2 = np.fft.irfft(fw * frev, n=L2)[:size_w_rev]
        # grad_a[i] = 2 * Σ_j a_j w_{i+j}
        grad_a = 2.0 * conv2[n - 1 : n - 1 + n]

        # ---------- gradient wrt log‑parameters ----------
        # dL/dx = a * (grad_a - <a,grad_a>)
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- bookkeeping ----------
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # ---------- re‑heat if stuck ----------
        if iteration - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

        # ---------- increase temperature ----------
        beta_cur = min(beta_max, beta_cur * beta_mul)

    return best_vec, best_score


# ----------------------------------------------------------------------
# Reference hard‑max Adam (identical to the one shown in the prompt)
# ----------------------------------------------------------------------
def full_convolution_fft_simple(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a convolution (length 2n‑1)."""
    n = a.shape[0]
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[:size]


def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the true max‑convolution w.r.t. log‑parameters."""
    n = a.shape[0]
    G = np.zeros_like(a)
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad


def run_hardmax_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """The classic Adam optimiser that works on the exact max (active‑set) objective."""
    start = time.time()
    n = initial_a.shape[0]
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.2

    best_score = np.inf
    best_vec = None
    last_improve = 0
    it = 0

    while time.time() - start < time_limit - 0.5:
        it += 1
        a = softmax(x)
        conv = full_convolution_fft_simple(a)
        max_b = conv.max()
        tol = max_b * 1e-12 + 1e-15
        active = np.where(conv >= max_b - tol)[0]

        grad = grad_log_params(x, a, conv, max_b, active)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = it
            print(f"Iter {it:6d} (hard‑max): new best = {best_score:.6f}")

        if it - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"Iter {it:6d}: re‑heat (hard‑max)")

    return best_vec, best_score
