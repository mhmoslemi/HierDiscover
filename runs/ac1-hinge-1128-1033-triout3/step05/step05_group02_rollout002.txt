    cur_a = a.copy()
    cur_obj = cur_val

_GLOBAL_EPS = 1e-3
_GLOBAL_N = 0


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself (length 2·len−1)
    using FFT.  The result is always non‑negative (tiny negatives are clamped).
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)   # numerical safety
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε of the original objective and its gradient
    with respect to the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)  (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log∑_k exp((conv_k−max)/ε)
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # auto‑convolution a*a
    conv = _full_convolution(a)               # length 2·n‑1
    max_conv = conv.max()

    # smoothed max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)              # log Σ exp((conv−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (see derivation in the original code)
    weights = np.exp(shifted)
    p = weights / weights.sum()               # probability vector over k
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)             # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]            # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.minimize (reads globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair largest with smallest (reverse‑order on the right half).
    This minimises the central correlation Σ a_i·a_{n‑1−i}.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large on the left
        new_a[n - 1 - i] = sorted_vals[i]          # small on the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternate_reorder(a: np.ndarray) -> np.ndarray:
    """
    Alternate large and small values across the whole sequence.
    This reduces correlations at small shifts.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]          # descending
    new_a = np.empty_like(a)
    l, r = 0, n - 1
    for i in range(n):
        if i % 2 == 0:
            new_a[i] = sorted_desc[l]
            l += 1
        else:
            new_a[i] = sorted_desc[r]
            r -= 1
    return new_a


def _compute_objective(a: np.ndarray) -> float:
    """Fast objective 2·n·max(conv) / (sum(a)²) using FFT‑based convolution."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _swap_hill_climb(a: np.ndarray,
                    cur_obj: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on the permutation of the entries.
    Only swaps are performed, so the multiset of values stays unchanged.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_obj
    rng = np.random.default_rng()
    temperature = max(cur_val * 0.05, 1e-12)  # initial temperature

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose swapping positions i and j
        new_a = cur_a.copy()
        new_a[i], new_a[j] = new_a[j], new_a[i]

        # fast evaluation
        conv = _full_convolution(new_a)
        max_b = conv.max()
        new_val = float(2 * n * max_b / (new_a.sum() ** 2))

        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / max(temperature, 1e-12)):
            cur_a = new_a
            cur_val = new_val
            temperature *= 0.9999                     # cool slowly
            if delta < -1e-12:
                print(f"[swap] improved to {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a small
    ``evaluate_sequence`` score.  The routine runs for up to ``budget_s``
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size.  Larger n generally yields a smaller bound, but we keep
    # it modest enough to stay within the time budget.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Initialise from a previously discovered construction if available,
    # otherwise start from a random Dirichlet vector.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = np.random.rand(n)

    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 0.01:
        a0 = np.ones_like(a0) / n
    else:
        a0 = a0 / a0.sum()               # normalise Σa = 1

    best_a = a0.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS schedule (smooth‑max ε values)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=15)

    restart = 0
    # Main multi‑restart loop – we leave ~30 s for the final refinement.
    while time.time() < deadline - 30.0:
        restart += 1
        print(f"[restart {restart}]")

        # Random soft‑max parameters (wide spread for diversity)
        w = np.random.randn(n) * 2.0

        # Run the whole ε‑schedule for this restart
        for eps in eps_schedule:
            if time.time() >= deadline - 20.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            # L‑BFGS‑B on the smoothed surrogate
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # Exact evaluation of the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Apply deterministic reorderings and keep any improvement.
        # ------------------------------------------------------------------
        a_anti = _anti_symmetric_reorder(best_a)
        val_anti = evaluate_sequence(a_anti.tolist())
        if val_anti < best_val:
            best_val = val_anti
            best_a = a_anti.copy()
            print(f"[anti] improved to {best_val:.6f}")

        a_alt = _alternate_reorder(best_a)
        val_alt = evaluate_sequence(a_alt.tolist())
        if val_alt < best_val:
            best_val = val_alt
            best_a = a_alt.copy()
            print(f"[alt] improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # Quick permutation annealing on the current best (5 s per restart).
        # ------------------------------------------------------------------
        cur_obj = _compute_objective(best_a)
        sub_deadline = min(deadline, time.time() + 5.0)
        best_a, cur_obj = _swap_hill_climb(best_a, cur_obj, sub_deadline)
        if cur_obj < best_val:
            best_val = cur_obj
            print(f"[swap] updated best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final intensive hill‑climbing with the remaining budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_obj = _compute_objective(best_a)
        best_a, cur_obj = _swap_hill_climb(best_a, cur_obj, deadline)
        if cur_obj < best_val:
            best_val = cur_obj
            print(f"[final hill] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final exact score (the value that will be submitted).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] evaluate_sequence score = {final_score:.6f}")

    return best_a.tolist()
