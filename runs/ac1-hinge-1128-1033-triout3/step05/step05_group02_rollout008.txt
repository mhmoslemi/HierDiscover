def simulated_annealing(a, init_val, time_limit, n):
    cur = a.copy()
    cur_val = init_val
    best = cur.copy()
    best_val = cur_val
    T0 = cur_val * 0.1  # start temp relative
    start = time.time()
    while time.time() < time_limit:
        # temperature schedule: linear cooling
        elapsed = time.time() - start
        t = elapsed / (time_limit - start)
        T = T0 * (1 - t)
        # propose move: choose between swap or mass transfer
        if rng.random() < 0.3:
            i,j = rng.integers(0, n, size=2)
            if i==j: continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            i,j = rng.integers(0, n, size=2)
            if i==j or cur[i]==0: continue
            delta = cur[i] * 0.03 * rng.random()  # up to 3%
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
        # compute objective quickly
        new_val = objective_from_a(new)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / (T + 1e-12)):
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best = cur.copy()
                best_val = cur_val
                print(f"[SA] new best {best_val:.6f}")
    return best, best_val

"""
A multi‑restart optimizer for the step‑function problem.

Strategy
--------
1. Initialise a non‑negative sequence (interpolating the previous best if
   available) and normalise it to unit L¹‑norm.
2. Run many independent L‑BFGS optimisations on a smooth log‑sum‑exp
   surrogate of the max‑convolution.  The smoothing parameter ε is reduced
   on a log‑scale down to 1e‑12.
3. After each L‑BFGS run apply cheap re‑ordering heuristics:
   * anti‑symmetric ordering (large ↔ small values at opposite ends)
   * a few random permutations.
4. Finally apply a temperature‑controlled simulated‑annealing phase that
   performs swaps and small mass‑transfer moves while accepting uphill
   steps with a decaying probability.  The objective is evaluated directly
   with an FFT‑based convolution for speed.
5. Return the best sequence found (as a list of floats).  The whole
   procedure respects the 1000 s wall‑clock budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS wrapper (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter
_GLOBAL_N = None     # current sequence length


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself (length 2·len‑1).
    Uses FFT with zero‑padding to the next power of two.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives → clip
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate of the evaluation objective and its gradient
    w.r.t. the soft‑max parameters w.

    a = softmax(w)  (guarantees a_i ≥ 0, Σa_i = 1)
    L_ε(a) = 2·n·max(conv) + 2·n·ε·log∑_k exp((conv_k‑max)/ε)
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    # self‑convolution
    conv = _full_convolution(a)
    max_conv = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()               # probability distribution over k

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)             # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]           # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.minimize (reads globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Large values are placed on the left side, small values on the right side.
    This simple heuristic often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    vals = np.sort(a)               # ascending
    res = np.empty_like(a)
    half = n // 2
    for i in range(half):
        res[i] = vals[n - 1 - i]    # large → left
        res[n - 1 - i] = vals[i]    # small → right
    if n % 2:
        res[half] = vals[half]
    return res


def _objective_from_a(a: np.ndarray) -> float:
    """
    Exact evaluation of a sequence (without the security‑check wrapper).
    Returns 2·n·max(conv) / (sum(a)²).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-8:
        return np.inf
    conv = _full_convolution(a)
    return 2.0 * n * conv.max() / (s * s)


def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         n: int):
    """
    Temperature‑controlled SA that performs swaps and tiny mass transfers.
    Accepts uphill moves with probability exp(-(Δ)/T).  Returns the best
    sequence found before the deadline.
    """
    cur = a.copy()
    best = cur.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    start = time.time()

    # initial temperature: a small fraction of the current objective
    T0 = cur_val * 0.05 + 1e-12

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * (1.0 - frac)

        # ---- propose a move ------------------------------------------------
        if rng.random() < 0.30:
            # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] == 0.0:
                continue
            # move up to 5 % of donor's mass
            delta = cur[i] * 0.05 * rng.random()
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

        new_val = _objective_from_a(new)

        # ---- Metropolis acceptance ----------------------------------------
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val
                print(f"[SA] new best = {best_val:.6f}")

    return best, best_val
