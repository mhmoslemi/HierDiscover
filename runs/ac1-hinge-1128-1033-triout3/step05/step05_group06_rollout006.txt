"""
Propose a step‑function using a smoothed max‑convolution gradient.
The algorithm works on the log‑probability (softmax) parametrisation,
optimises a smooth approximation of the max‑convolution (log‑sum‑exp)
with Adam, periodically symmetrises the vector (the optimum is known to be
symmetric) and injects random re‑heats when progress stalls.
Multiple seeds (including the previously discovered construction) are
optimised within the given time budget.  The best vector found is returned
as a plain list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution using real FFT (returns length len(a)+len(b)-1)
# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = len(a) + len(b) - 1
    # use a power‑of‑two length for speed
    pow2 = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=pow2)
    fb = np.fft.rfft(b, n=pow2)
    conv = np.fft.irfft(fa * fb, n=pow2)
    return conv[:size]

def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Convolution a with itself (auto‑convolution)."""
    return conv_fft(a, a)

# ----------------------------------------------------------------------
# Softmax (ensures non‑negativity and unit sum)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

# ----------------------------------------------------------------------
# Families of deterministic seeds (symmetric, edge‑heavy)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to the nearest edge
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Compute the objective (the value returned by evaluate_sequence)
#   f(a) = 2 * n * max_k (a*a)[k]   with sum(a)=1
# ----------------------------------------------------------------------
def score_vector(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Gradient of the smooth‑max objective w.r.t. the log‑parameters.
#   smooth_maxβ(conv) = (1/β) log Σ_k exp(β conv_k)
# ----------------------------------------------------------------------
def grad_log_params(x: np.ndarray, a: np.ndarray,
                    beta: float) -> np.ndarray:
    """
    Returns ∂/∂x  (smooth_maxβ(conv))  for a = softmax(x).

    The gradient w.r.t. a is
        g_i = 2 * Σ_k w_k * a_{k-i}
    where w_k = exp(β (conv_k - max_conv)) / Σ_j exp(β (conv_j - max_conv)).
    The log‑parameter gradient follows from the softmax Jacobian.
    """
    n = a.shape[0]

    # ----- auto‑convolution -----
    conv = full_convolution_fft(a)
    max_c = conv.max()

    # ----- smooth weights (log‑sum‑exp trick) -----
    # w ∝ exp(beta * (conv - max_c))
    exp_arg = beta * (conv - max_c)
    # avoid overflow/underflow
    w_unnorm = np.exp(exp_arg)
    w = w_unnorm / w_unnorm.sum()          # shape (2n-1,)

    # ----- gradient w.r.t a (g = 2 * w ⋆ a) -----
    a_rev = a[::-1]
    conv2 = conv_fft(w, a_rev)            # length 3n-2
    # extract the slice corresponding to g_i = 2 * conv2[n-1 : n-1+n]
    g = 2.0 * conv2[n - 1 : n - 1 + n]

    # ----- gradient w.r.t log‑parameters (softmax) -----
    # ∂L/∂x_i = a_i * ( g_i - Σ_j g_j a_j )
    mu = np.dot(g, a)
    grad_x = a * (g - mu)
    return grad_x, max_c * 2.0 * n   # also return the current (unsmoothed) score

# ----------------------------------------------------------------------
# Adam optimiser on the log‑probabilities using the smooth‑max gradient.
# ----------------------------------------------------------------------
def run_adam_smooth(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for a given starting probability vector.
    Returns the best vector found and its (true) evaluation score.
    """
    start = time.time()
    n = initial_a.shape[0]

    # initialise log‑parameters (avoid log(0))
    eps = 1e-14
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.6                       # base learning rate in log‑space

    # schedule for the smoothing parameter beta
    beta_start = 5.0
    beta_end = 200.0
    beta_ramp = 5000                # iterations over which beta grows

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    while time.time() - start < time_limit - 0.25:
        iteration += 1

        # ----- softmax -----
        a = softmax(x)

        # ----- optional symmetrisation (improves convergence) -----
        if iteration % 100 == 0:
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
            x = np.log(np.maximum(a, eps))

        # ----- current smoothing parameter -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, iteration / beta_ramp)

        # ----- gradient & true score -----
        grad, cur_score = grad_log_params(x, a, beta)

        # ----- Adam update (simple decay of the stepsize) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- bookkeeping -----
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best (true) score = {best_score:.6f}")

        # ----- re‑heat if stuck -----
        if iteration - last_improve > 4000:
            print(f"Iter {iteration:6d}: re‑heat (no improvement in 4000 iters)")
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration

    # final safeguard: ensure the returned vector is a valid probability vector
    best_vec = np.maximum(best_vec, 0.0)
    if best_vec.sum() == 0.0:
        best_vec = np.ones_like(best_vec) / n
    else:
        best_vec /= best_vec.sum()
    return best_vec, best_score
