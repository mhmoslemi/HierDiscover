Function for projection onto simplex:

def simplex_projection(v):
    """Project onto probability simplex (sum=1, v_i >=0)."""
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u > (sv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (sv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function sequence.

The algorithm is a *directed hill‑climbing* on the probability simplex.
It repeatedly moves a fraction of the coefficient that contributes most to
the current peak of the self‑convolution to a location that does **not**
contribute (or contributes very little) to that peak.  The move size is
large at the beginning of the run and slowly annealed.  Moves are accepted
only if they lower the peak of the convolution; otherwise the move is
rejected and a smaller transfer amount is tried.  Small random “jitter”
steps and occasional full re‑computation of the convolution keep the search
robust.

Compared with the reference hill‑climber, the destination index is chosen
greedily (minimal partner mass at the peak) and the transfer fraction starts
much larger (≈80 % of the source mass).  This gives much larger per‑step
reductions of the objective and typically reaches a lower bound within the
1000 s budget.

The function returns the best sequence (as a list of floats) that was found
before the time limit expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast linear convolution using real FFT (full length 2*n‑1)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """Return the linear self‑convolution x*x (length 2*len(x)-1)."""
    n = x.shape[0]
    total = 2 * n - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:total]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv

                    delta2j = delta_j * delta_j

                    conv[src:src + n] -= 2.0 * delta_j * a_tmp
                    conv[dst:dst + n] += 2.0 * delta_j * a_tmp
                    conv[2 * src]      += delta2j
                    conv[2 * dst]      += delta2j
                    conv[src + dst]    -= 2.0 * delta2j

                    a[src] -= delta_j
                    a[dst] += delta_j

                    jitter_max = conv.max()
                    if jitter_max < cur_max:
                        cur_max = jitter_max
                        cur_obj = float(2.0 * n * cur_max)
                        if cur_obj < best_obj:
                            best_obj = cur_obj
                            best_a = a.copy()
                            print(f"[{iteration:7d}] jitter improved best = {best_obj:.6f}")
                            last_improve_iter = iteration
                    else:
                        # revert jitter
                        a = a_tmp
                        conv[src:src + n] += 2.0 * delta_j * a_tmp
                        conv[dst:dst + n] -= 2.0 * delta_j * a_tmp
                        conv[2 * src]      -= delta2j
                        conv[2 * dst]      -= delta2j
                        conv[src + dst]    += 2.0 * delta2j

        # ------------------------------------------------------------------
        # Periodic full recomputation of the convolution (prevents drift)
        # ------------------------------------------------------------------
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)

        # ------------------------------------------------------------------
        # Simple stagnation detection – random reshuffle if no improvement for long
        # ------------------------------------------------------------------
        if iteration - last_improve_iter > 300_000:
            rnd = np.random.rand(n)
            a = 0.9 * a + 0.1 * rnd
            a = np.maximum(a, 0)
            a /= a.sum()
            conv = _full_convolution_fft(a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)
            last_improve_iter = iteration

    # ----------------------------------------------------------------------
    # Final evaluation (for information only)
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
