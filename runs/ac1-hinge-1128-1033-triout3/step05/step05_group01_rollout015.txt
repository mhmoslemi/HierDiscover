Pseudo:

def solve_lp(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)  # b >= 0 enforced
    constraints = []
    for k in range(2*n-1):
        # compute i range
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # building sum of f[i] * b[k-i]
        expr = 0
        for i in range(i_start, i_end+1):
            j = k - i
            expr += f[i] * b[j]
        constraints.append(expr <= M)
    # objective maximize sum(b)
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    b_opt = b.value
    return b_opt

"""
A two‑stage optimizer for the step‑function problem.

Stage 1 – L2‑smoothing:
    We minimise the squared deviation of the full autocorrelation
    from the constant value C = (∑a)²/(2n−1).  Using a soft‑max
    parametrisation a = S·softmax(w) (S = √(2n)) guarantees the
    ℓ¹‑norm and non‑negativity.  The gradient of the L2 loss can be
    written in closed form using two FFT‑based convolutions, so that
    L‑BFGS‑B can be applied efficiently.

Stage 2 – Guided hill‑climbing:
    Starting from the best L2‑solution we repeatedly locate the lag
    where the convolution is maximal, remove a small amount of mass
    from the most offending pair (i,j) and redistribute it to two
    other indices that do not recreate the offending lag.
    The move is accepted only if the exact objective
    ``evaluate_sequence`` improves.  The step‑size is annealed.

Both stages respect the 1000 s time budget and keep all entries
non‑negative.  The final list of heights is returned as a plain
Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# L2 loss (conv ≈ constant) and its gradient
# ----------------------------------------------------------------------
def _l2_loss_and_grad(w: np.ndarray, n: int, sum_target: float):
    """
    Compute
        loss = Σ_k ( (a*a)[k] - C )²,
        grad_w = ∂loss/∂w
    where a = sum_target·softmax(w) and C = sum_target²/(2n‑1).
    """
    # ---------- soft‑max (guaranteed ℓ¹‑norm) ----------
    w_shift = w - np.max(w)                # improve stability
    exp_w = np.exp(w_shift)
    s = exp_w / np.sum(exp_w)              # probabilities
    a = s * sum_target                     # actual coefficients

    # ---------- full auto‑convolution ----------
    conv = _full_convolution(a)            # length 2n‑1

    # target constant value (average of the convolution)
    C = (sum_target * sum_target) / (2 * n - 1)

    diff = conv - C
    loss = np.dot(diff, diff)              # Σ diff²

    # ---------- gradient w.r.t. a ----------
    # grad_a_i = 2 * Σ_k diff_k * a_{k‑i}
    a_rev = a[::-1]
    conv2 = _conv_fft(diff, a_rev)         # length 3n‑2
    # segment that corresponds to indices 0 … n‑1
    grad_a = 2.0 * conv2[n - 1: 2 * n - 1]

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)                # Σ_i grad_a_i a_i
    grad_w = a * (grad_a - dot / sum_target)

    return float(loss), grad_w.astype(np.float64)


class _L2Objective:
    """
    Callable that wraps _l2_loss_and_grad for a fixed n and sum_target.
    Used with scipy.optimize.minimize (L‑BFGS‑B).
    """
    def __init__(self, n: int, sum_target: float):
        self.n = n
        self.sum_target = sum_target

    def __call__(self, w: np.ndarray):
        loss, grad = _l2_loss_and_grad(w, self.n, self.sum_target)
        return loss, grad


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimisation that directly reduces the true objective.
    Moves a small amount of mass from the most offending pair (i,j)
    to two other positions (p,q) that do not reproduce the same lag.
    The step size shrinks over time (simulated‑annealing style).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.15                     # start fairly aggressive

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # all lags attaining the max (within tolerance)
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # indices that contribute to lag k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # Choose two destination indices that do NOT recreate lag k
        attempts = 0
        while attempts < 5:
            p, q = rng.integers(0, n, size=2)
            if p + q != k and p != i_max and q != j_max:
                break
            attempts += 1

        # Determine feasible transfer amount
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-16:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-16:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # Exact evaluation (scale‑invariant, works directly)
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.98, 1e-5)   # tighten a bit
        else:
            step_frac = max(step_frac * 0.995, 1e-5)

    return cur_a, cur_val

            exp_w = np.exp(w_shift)
            s = exp_w / np.sum(exp_w)
            a_opt = s * sum_target
            val = evaluate_sequence(a_opt.tolist())
            return a_opt, val
        except Exception as e:
            print("[l2_opt] exception:", e)
            return None, None

    # first optimisation run
    a_opt, val_opt = _run_l2(w)
    if a_opt is not None and val_opt < best_val:
        best_val = val_opt
        best_a = a_opt.copy()
        w = np.log(best_a + 1e-12)
        print(f"[l2] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Random restarts – increase robustness
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0 and restart < 12:
        restart += 1
        w_pert = w + rng.normal(scale=0.5, size=n)  # moderate noise
        a_opt, val_opt = _run_l2(w_pert)
        if a_opt is not None and val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            w = np.log(best_a + 1e-12)
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final guided hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # --------------------------------------------------------------
    # Final report
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
