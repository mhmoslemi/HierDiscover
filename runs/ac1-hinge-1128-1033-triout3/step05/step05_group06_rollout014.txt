def full_convolution_fft(a):
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[:2*n-1]

def cross_correlation_fft(w, a):
    n = a.shape[0]
    size = 1
    while size < len(w) + n - 1:

    start = n - 1
    end = start + n

def simplex_projection(v):
    # v is 1-D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def run_l2_opt(initial_a, time_limit):
    start = time.time()
    a = initial_a.copy()
    n = a.shape[0]
    mu = 1.0 / (2 * n - 1)
    # preallocate FFT size
    size = 1
    while size < 3 * n:
        size <<= 1
    best_score = float('inf')
    best_a = a.copy()
    best_iter = 0
    last_improve = 0
    it = 0
    # maybe compute conv once for best
    while time.time() - start < time_limit - 0.5:
        it += 1
        conv = full_convolution_fft(a)
        cur_score = 2.0 * n * conv.max()   # since sum a = 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"L2 iter {it}: new best score = {best_score:.6f}")
        # compute residual
        r = conv - mu
        # FFT of residual
        r_fft = np.fft.rfft(r, n=size)
        a_rev_fft = np.fft.rfft(a[::-1], n=size)
        cross_full = np.fft.irfft(r_fft * a_rev_fft, n=size)
        start_idx = n - 1
        grad = 4.0 * cross_full[start_idx:start_idx + n]
        # step size decay
        lr = 0.5 / (np.sqrt(it) + 1.0)
        a = a - lr * grad
        # projection onto simplex
        a = simplex_projection(a)
        # reheating
        if it - last_improve > 5000:
            a = a + np.random.randn(n) * 1e-4
            a = simplex_projection(a)
            last_improve = it
            print(f"L2 iter {it}: reheated")
    return best_a, best_score

"""
Propose a step‑function via two complementary continuous optimisers.

1️⃣  Adam on the log‑probabilities (soft‑max) – the same optimiser that was
    used to obtain the previous best bound.  It quickly finds a good
    region of the simplex.

2️⃣  Projected (simplex) gradient descent on the *L2‑variance* of the
    full convolution.
    The loss L = Σ_k ( (a∗a)[k] – μ )²   with  μ = 1/(2 n‑1)  (the
    constant value that would give the theoretical optimum 1) is smooth,
    convex in a and has gradient

        ∇L = 4·cross_corr( (a∗a) – μ , a ) ,

    where the cross‑correlation can be evaluated by a single FFT‑based
    convolution.  After each gradient step the vector is projected onto
    the probability simplex (non‑negative, sum = 1).  The step size decays
    as 0.5/√t, a simple re‑heat is performed when no improvement is seen
    for a while, and the best vector found (according to the true
    evaluation function) is kept.

Both stages respect the scaling‑invariance of the evaluation function,
work with any length and stay well within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution (real‑FFT, length 2n‑1)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using real FFT (output length 2·len(a)-1)."""
    n = a.shape[0]
    size = 2 * n                         # zero‑padding length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# Helper: probability‑simplex projection (Euclidean)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all elements are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

# ----------------------------------------------------------------------
# Soft‑max (used by the Adam optimiser)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

# ----------------------------------------------------------------------
# Parametric families for deterministic seeds
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Gradient of the max‑convolution w.r.t. log‑probability parameters
# (used by the Adam optimiser)
# ----------------------------------------------------------------------
def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    G = np.zeros_like(a)
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]
    # Sub‑gradient for the log‑parameters
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad

# ----------------------------------------------------------------------
# Adam optimiser on the log‑probabilities (same spirit as the previous
# implementation, but with a slightly broader active set for smoother
# updates).
# ----------------------------------------------------------------------
def run_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    start = time.time()
    n = initial_a.shape[0]
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.2

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    while time.time() - start < time_limit - 0.5:
        iteration += 1
        a = softmax(x)
        conv = full_convolution_fft(a)
        max_b = conv.max()
        # include near‑maxima (relative tolerance 0.1 %)
        tol = max_b * 0.001 + 1e-12
        active = np.where(conv >= max_b - tol)[0]

        grad = grad_log_params(x, a, conv, max_b, active)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cur_score = 2.0 * n * max_b           # sum(a)=1 ⇒ denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Adam iter {iteration:6d}: new best = {best_score:.6f}")

        if iteration - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Adam iter {iteration:6d}: re‑heat (stalled)")

    return best_vec, best_score


# ----------------------------------------------------------------------
# Projected gradient descent on L2 loss of the convolution.
# ----------------------------------------------------------------------
def run_l2_opt(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Minimise  Σ_k ( (a*a)[k] – μ )²  on the probability simplex.
    Returns the best vector (as numpy array) and its true evaluation score.
    """
    start = time.time()
    a = initial_a.copy()
    n = a.shape[0]
    mu = 1.0 / (2 * n - 1)

    # FFT size for cross‑correlation (full length = 3n‑2)
    size = 1
    while size < 3 * n:
        size <<= 1

    best_score = float('inf')
    best_a = a.copy()
    last_improve = 0
    it = 0

    while time.time() - start < time_limit - 0.5:
        it += 1
        conv = full_convolution_fft(a)
        cur_score = 2.0 * n * conv.max()      # sum(a)=1 ⇒ denominator = 1
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            print(f"L2 iter {it:6d}: new best = {best_score:.6f}")

        # residual of the convolution w.r.t. the uniform target μ
        r = conv - mu

        # cross‑correlation r ⋆ a (via convolution with reversed a)
        r_fft = np.fft.rfft(r, n=size)
        a_rev_fft = np.fft.rfft(a[::-1], n=size)
        cross_full = np.fft.irfft(r_fft * a_rev_fft, n=size)
        start_idx = n - 1
        grad = 4.0 * cross_full[start_idx:start_idx + n]

        # step size decays like 0.5/√t
        lr = 0.5 / (np.sqrt(it) + 1.0)
        a = a - lr * grad

        # projection onto the probability simplex
        a = simplex_projection(a)

        # re‑heat if no improvement for a long time
        if it - last_improve > 5000:
            a = a + np.random.randn(n) * 1e-4
            a = simplex_projection(a)
            last_improve = it
            print(f"L2 iter {it:6d}: re‑heat (stalled)")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Searches for a non‑negative step‑function that minimises the provided
    evaluation function.  A blend of Adam (log‑prob) and a custom L2‑based
    projected gradient descent is used.  The routine respects the supplied
    time budget and returns the best vector found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Determine problem size n
    # --------------------------------------------------------------
    default_n = 8192                      # power of two – FFT‑friendly
    external_seed = None
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        if external.sum() > 0.0:
            external /= external.sum()
            external_seed = external
            default_n = max(default_n, external.shape[0])
            print(f"External seed detected (length {external.shape[0]})")
    except Exception:
        pass

    n = default_n

    # --------------------------------------------------------------
    # Build a modest portfolio of starting points (all normalised)
    # --------------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.ones(n) / n)

    # parametric families
    for p in [0.5, 1.0, 1.5]:
        seeds.append(u_min_distribution(n, p))
        seeds.append(u_prod_distribution(n, p))
    for lam in [5e-4, 1e-3, 5e-3, 1e-2]:
        seeds.append(exp_distribution(n, lam))

    # a few Dirichlet‑like random draws
    for _ in range(4):
        r = np.random.rand(n)
        seeds.append(r / r.sum())

    # external construction (padded / trimmed to length n if needed)
    if external_seed is not None:
        if external_seed.shape[0] != n:
            pad = n - external_seed.shape[0]
            left = pad // 2
            right = pad - left
            ext = np.pad(external_seed, (left, right), mode='constant')
            ext = ext / ext.sum()
            seeds.append(ext)
        else:
            seeds.append(external_seed)

    # --------------------------------------------------------------
    # Optimisation loop – allocate time per seed dynamically
    # --------------------------------------------------------------
    best_overall_vec = None
    best_overall_score = np.inf

    remaining_time = deadline - time.time()
    per_seed_budget = remaining_time / max(1, len(seeds))

    for idx, seed_vec in enumerate(seeds):
        # normalise seed to a probability vector
        a = np.maximum(seed_vec, 0.0)
        if a.sum() == 0.0:
            a = np.ones_like(a)
        a = a / a.sum()

        # time budgeting for this seed
        time_for_this = per_seed_budget
        time_left = deadline - time.time()
        if time_left < 5.0:
            break
        if time_for_this > time_left:
            time_for_this = time_left - 2.0
        if time_for_this <= 0:
            break

        # split the budget: 40 % Adam, 60 % L2 optimiser
        t_adam = time_for_this * 0.4
        t_l2  = time_for_this - t_adam

        print(f"\n--- Seed {idx+1}/{len(seeds)}: n={n}, "
              f"budget {time_for_this:.1f}s (Adam {t_adam:.1f}s, L2 {t_l2:.1f}s) ---")

        # 1. Adam on log‑probabilities
        vec_adam, _ = run_adam(a, t_adam)

        # 2. L2‑based projected GD
        vec_l2, _ = run_l2_opt(vec_adam, t_l2)

        # evaluate with the official evaluator (handles clipping etc.)
        final_score = evaluate_sequence(vec_l2.tolist())
        if final_score < best_overall_score:
            best_overall_score = final_score
            best_overall_vec = vec_l2.copy()
            print(f"*** New global best (seed {idx+1}): score = {best_overall_score:.6f}")

        # update remaining time and per‑seed budget for the rest
        remaining_time = deadline - time.time()
        if remaining_time <= 5.0:
            break
        per_seed_budget = remaining_time / max(1, len(seeds) - idx - 1)

    # fallback (should never happen)
    if best_overall_vec is None:
        best_overall_vec = np.ones(n) / n
        best_overall_score = evaluate_sequence(best_overall_vec.tolist())

    print("\n=== Search finished ===")
    print(f"Best score found: {best_overall_score:.6f} (length {len(best_overall_vec)})")
    return best_overall_vec.tolist()
