    """Find a non‑negative sequence that minimises evaluate_sequence(...).

    The algorithm starts from the best known construction (if available)
    and applies an alternating scheme:
    1) solve a linear program that finds a direction g which respects the
       current convolution bound,
    2) mix the current sequence with the normalised direction,
    3) repeat while time permits.
    After the LP phase a short hill‑climbing phase refines the result.
    """
    import time, numpy as np
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None
    # random seed
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # ----------------------------------------------------------------------
    # initialise sequence a
    # ----------------------------------------------------------------------
    try:
        init_seq = height_sequence_1   # noqa: F821
    except NameError:
        init_seq = None

    if init_seq is None:
        n = 1800
        a = np.full(n, 1.0 / n)
    else:
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.shape[0]
        # normalise to sum = 1
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

    # ----------------------------------------------------------------------
    # bookkeeping
    # ----------------------------------------------------------------------
    conv = np.convolve(a, a)
    cur_max = conv.max()
    cur_obj = float(2.0 * n * cur_max)
    best_obj = cur_obj
    best_a = a.copy()
    print(f"[init] n={n}, objective={cur_obj:.6f}")

    # helper to build constraint matrix for LP (dense)
    # ----------------------------------------------------------------------
    def build_constraint_matrix(vec):
        """Return matrix A of shape (2n‑1, n) with A[k,i]=vec[k‑i] (0 when out of range)."""
        nn = vec.shape[0]
        rows = 2 * nn - 1
        A = np.zeros((rows, nn), dtype=np.float64)
        # Fill each column i with vec shifted down by i
        for i in range(nn):
            A[i:i+nn, i] = vec
        return A

    # ----------------------------------------------------------------------
    # main optimisation loop (LP + mixing)
    # ----------------------------------------------------------------------
    iteration = 0
    no_improve = 0
    max_no_improve = 30   # stop after this many failed LP steps

    while time.time() < deadline:
        iteration += 1
        # recompute current convolution and bound
        conv = np.convolve(a, a)
        M = conv.max()

        if linprog is None:
            # no LP support – break to hill‑climb
            break

        # Build constraint matrix A (dense)
        A = build_constraint_matrix(a)

        # Linear program: maximise sum(g) (i.e. minimise -sum(g))
        c = -np.ones(n)
        bounds = [(0, None) for _ in range(n)]
        # RHS is M for every row
        b = np.full(2 * n - 1, M)

        # Solve
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – exit loop
            print(f"[{iteration}] LP failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            print(f"[{iteration}] LP returned zero sum")
            break

        # Normalise g to have sum 1 (same as a)
        g_norm = g / sum_g

        # Pre‑compute auxiliary convolutions
        conv_ag = np.convolve(a, g_norm)
        conv_g = np.convolve(g_norm, g_norm)

        # Search a mixing factor t that reduces the peak.
        best_t = 0.0
        best_Mnew = M
        # try a small set of t values; we focus on 0<t<=0.5
        for t in np.linspace(0.05, 0.5, 10):
            # (1‑t)^2 f*f + 2t(1‑t) f*g + t^2 g*g
            mixed = (1 - t) ** 2 * conv + 2 * t * (1 - t) * conv_ag + t ** 2 * conv_g
            Mnew = mixed.max()
            if Mnew < best_Mnew:
                best_Mnew = Mnew
                best_t = t
        if best_t > 0:
            # accept mixing
            a = (1 - best_t) * a + best_t * g_norm
            cur_obj = float(2.0 * n * best_Mnew)
            if cur_obj < best_obj - 1e-8:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration}] LP mix improved best = {best_obj:.6f} (t={best_t:.3f})")
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # stop the LP phase – likely stalled
            print(f"[{iteration}] no improvement in {max_no_improve} LP steps, exiting LP phase")
            break

    # ----------------------------------------------------------------------
    # final polishing: a few thousand cheap hill‑climbing moves
    # ----------------------------------------------------------------------
    # update time
    now = time.time()
    remaining = deadline - now
    if remaining > 5.0:
        # reuse incremental update code with a small budget
        # parameters:
        moves = int(min(200000, remaining * 10000))  # rough estimate
        max_frac = 0.1
        min_frac = 0.02
        start_hill = time.time()
        # keep copy of best for this phase
        a_h = best_a.copy()
        conv_h = np.convolve(a_h, a_h)
        cur_obj_h = float(2.0 * n * conv_h.max())
        best_obj_h = cur_obj_h
        best_a_h = a_h.copy()
        for step in range(moves):
            if time.time() >= deadline:
                break
            # annealing move size
            progress = step / moves
            move_frac = max_frac * (1.0 - progress) + min_frac * progress
            # locate current peak
            k_star = int(conv_h.argmax())
            i_start = max(0, k_star - (n - 1))
            i_end = min(n - 1, k_star)
            idx = np.arange(i_start, i_end + 1)
            partner = k_star - idx
            contributions = a_h[idx] * a_h[partner]
            i_src = int(idx[contributions.argmax()])
            # pick destination
            j_dst = np.random.randint(n - 1)
            if j_dst >= i_src:
                j_dst += 1

            conv_h[j_dst:j_dst + n] += 2.0 * delta * a_old
            conv_h[2 * i_src] += delta * delta
            conv_h[2 * j_dst] += delta * delta
            conv_h[i_src + j_dst] -= 2.0 * delta * delta
            a_h[i_src] -= delta
            a_h[j_dst] += delta

                    best_obj_h = new_obj

                conv_h[j_dst:j_dst + n] -= 2.0 * delta * a_old
                conv_h[2 * i_src] -= delta * delta
                conv_h[2 * j_dst] -= delta * delta
                conv_h[i_src + j_dst] += 2.0 * delta * delta

        if best_obj_h < best_obj:

            best_a = best_a_h.copy()
            print(f"[final hill] improved best = {best_obj:.6f}")

Propose a step‑function sequence that minimizes ``evaluate_sequence``.

The search consists of two phases:

1. **LP‑mix phase** – starting from the best known construction (if
   available) we repeatedly
   * compute the current self‑convolution bound `M`,
   * solve a linear program  
     `max  Σ g_j   s.t.   (a * g)_k ≤ M ,  g ≥ 0`,
   * normalise the LP solution to a probability vector,
   * line‑search a mixing factor `t` and replace  
     `a ← (1‑t) a + t ĝ`.  
   This is the exact algorithm described in the statement and
   typically lowers the peak of the convolution in a few iterations.

2. **Hill‑climbing polish** – a short, cheap “mass‑transfer’’ hill‑climb
   (the same incremental update used in the reference code) is run for
   the remaining time to squeeze a few more improvements.

Both phases respect the simplex (`a_i ≥ 0`, `Σ a_i = 1`) and the search
stops before the 1000 s budget expires.  The best sequence found is
returned and its score is printed.
"""

import time
import numpy as np

# SciPy is used for the linear program; if unavailable we simply skip the LP phase.

    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def _build_constraint_matrix(vec: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1) × n) that encodes the
    convolution constraints `(vec * g)_k ≤ M`.  The entry A[k,i] equals
    ``vec[k‑i]`` when the index is in range, otherwise 0.
    """
    n = vec.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i is a shifted copy of ``vec``
    for i in range(n):
        A[i:i + n, i] = vec
    return A
