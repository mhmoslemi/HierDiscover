"""
search_step_function.py

Algorithm
---------

We treat the step‑function as a probability vector `a` (non‑negative, sum = 1).
For such a vector the evaluation function reduces to

    score(a) = 2·n·max_k (a * a)[k]               (n = len(a))

Our goal is to minimise the peak of the auto‑convolution.  The programme
combines two complementary optimisation phases:

1. **Projected Adam on the log‑probabilities**  
   Adam is run on several diverse initial seeds (uniform, exponential,
   power‑law, random Dirichlet, and the external seed if present).  The
   parameters are the logits `x` with `a = softmax(x)`.  The gradient of the
   peak w.r.t. `x` is obtained analytically (see `grad_log_params`).  Each
   Adam run is given a fraction of the total budget.

2. **Simulated Annealing (SA) fine‑tuning**  
   Starting from the best vector produced by Adam we perform a fast
   stochastic local search.  At each step a small amount of mass is moved
   from one index to another, the new peak is recomputed (via an FFT‑based
   convolution) and the move is accepted with the usual Metropolis rule.
   Temperature decays steadily; occasional improvements are recorded as the
   global best.

The two phases are deliberately complementary: Adam exploits first‑order
information, while SA explores the non‑convex landscape beyond the basin
captured by the gradient method.  The whole routine respects the 1000 s
time limit, prints progress, and finally returns the best vector found as
a plain Python list.

The code is self‑contained, uses only NumPy and the supplied
`evaluate_sequence` function (via an equivalent fast scorer) and works on
a single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 2 * n                      # zero‑pad to length 2n (no need for power‑of‑2)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a probability vector a (sum == 1).
    Returns the same value that `evaluate_sequence(a.tolist())` would produce
    (ignoring the clipping to [0,1000] which never triggers here).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families for deterministic seeds
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law decreasing from the edges."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """Product of distances to the two ends, raised to -p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(alpha)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.ones_like(x)
    return x / x.sum()


# ----------------------------------------------------------------------
# Softmax and its gradient helpers (used by Adam)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] with respect to the logits x,
    assuming a = softmax(x).  The result is dF/dx.
    """
    n = a.shape[0]
    G = np.zeros_like(a)

    # For each active convolution index accumulate contributions.
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]

    # Sub‑gradient w.r.t. x (chain rule through softmax)
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on logits (log‑probabilities)
# ----------------------------------------------------------------------
def run_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the logits, starting from the probability vector `initial_a`.
    Returns the best probability vector found and its score.
    """
    start = time.time()
    n = initial_a.shape[0]

    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.3                     # a slightly larger base LR than before

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    while time.time() - start < time_limit - 0.3:
        iteration += 1

        a = softmax(x)

        conv = full_convolution_fft(a)
        max_b = conv.max()

        # active set (allow a tiny tolerance)
        tol = max_b * 1e-12 + 1e-15
        active = np.where(conv >= max_b - tol)[0]

        grad = grad_log_params(x, a, conv, max_b, active)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)   # simple decay

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"[Adam] Iter {iteration:6d}: new best score = {best_score:.6f}")

        # If stuck for a while, inject a tiny random perturbation (re‑heat)
        if iteration - last_improve > 4000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[Adam] Iter {iteration:6d}: re‑heat (no improvement)")

    return best_vec, best_score

# Simulated Annealing fine‑tuning
# ----------------------------------------------------------------------
def simulated_annealing(start_a: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a small amount of mass between two entries.
    Returns the best vector seen and its corresponding score.
    """
    start = time.time()
    n = start_a.shape[0]

    cur_a = start_a.copy()
    cur_score = score_vector(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    # Temperature schedule (linear decay in log‑space)
    T0 = max(1e-6, cur_score * 0.02)        # start a few percent of the score
    Tmin = cur_score * 1e-6
    T = T0
    decay = 0.9995                           # per‑iteration multiplicative decay

    # Step size schedule (max fraction of the source entry to move)
    step_max = 0.02
    iter_cnt = 0

    while time.time() - start < time_budget:
        iter_cnt += 1

        # pick a source index with non‑negligible mass
        i = np.random.randint(n)
        if cur_a[i] < 1e-12:
            continue

        # pick a distinct target index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose a mass transfer
        delta = min(cur_a[i], step_max * np.random.rand())
        if delta < 1e-14:
            continue

        cand_a = cur_a.copy()
        cand_a[i] -= delta
        cand_a[j] += delta

        # evaluate candidate
        cand_score = score_vector(cand_a)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp((cur_score - cand_score) / max(T, 1e-12)):
            cur_a = cand_a
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_a = cand_a.copy()
                print(f"[SA] Iter {iter_cnt:7d}: new global best = {best_score:.6f}")

        # temperature decay
        T = max(Tmin, T * decay)

        # occasional step‑size reduction when temperature is low
        if T <= Tmin * 10:
            step_max = max(1e-4, step_max * 0.99)

    return best_a, best_score
