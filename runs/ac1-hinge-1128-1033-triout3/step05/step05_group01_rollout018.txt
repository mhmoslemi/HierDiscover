def project_to_simplex(v, s=1):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho = max i s.t. u_i > cssv_i / i
    rho = np.where(u > cssv / ind)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights) that (approximately) minimises
``evaluate_sequence``.  The implementation uses three ideas:

1. **Projection onto the probability simplex** – the sum of the heights is
   kept constant (we normalise to 1, any scaling does not change the score).

2. **Sub‑gradient descent on the max‑convolution** – the objective
   ``f(a)=max_k conv_k`` is non‑smooth.  For the current lag `k*` that attains
   the maximum we use the exact sub‑gradient
   ``∂conv_{k*}/∂a_i = a_i + a_{k*-i}`` (out‑of‑range indices contribute 0).
   A step opposite to this gradient is taken, followed by a projection onto the
   simplex.  The step size is adapted automatically.

3. **Guided hill‑climbing refinement** – after the sub‑gradient phase we
   repeatedly move a small amount of mass from the pair of indices that
   contributes most to the current maximum lag to two “quiet’’ positions
   (indices with the smallest possible increase of that lag).  The move is
   accepted only if the exact ``evaluate_sequence`` value improves.

The whole routine respects the 1000 s budget; intermediate scores are printed.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------
# FFT utilities (real data)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full (linear) autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


# --------------------------------------------------------------
# Projection onto the (non‑negative) simplex
# --------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i >= 0, sum(x) = s}.
    """
    if s <= 0:
        raise ValueError("Simplex radius must be positive")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find rho = max{i | u_i > cssv_i / i}
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Exact objective (identical to evaluate_sequence but cheaper)
# --------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Compute 2·n·max(conv) / (sum(a)**2).  Assumes a already non‑negative."""
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _full_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# --------------------------------------------------------------
# Sub‑gradient descent on max‑convolution
# --------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a projected sub‑gradient descent on the max‑convolution.
    Returns the best sequence seen and its objective value.
    """
    cur_a = a0.copy()
    cur_val = _objective(cur_a)
    n = cur_a.shape[0]

    # initial step size (as a fraction of the current L1 norm)
    step_frac = 0.05

    # statistics
    it = 0
    while time.time() < deadline:
        it += 1
        # full autocorrelation
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # pick one of the maxima (deterministic enough)
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # ---- sub‑gradient of conv_k w.r.t. a ----
        grad = cur_a.copy()                # the a_i term
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.intp)
        rev_idx = k - idx
        grad[idx] += cur_a[rev_idx]        # add the a_{k-i} term

        # step size – decrease slowly with the iteration count
        eta = step_frac / np.sqrt(it)

        # take the step and re‑project onto the simplex (sum = 1)
        new_a = cur_a - eta * grad
        new_a = _project_to_simplex(new_a, s=cur_a.sum())
        new_val = _objective(new_a)

        if new_val < cur_val:
            cur_a, cur_val = new_a, new_val
            # successful step – allow a slightly larger step next time
            step_frac = min(step_frac * 1.05, 0.3)
            print(f"[subgrad] iter {it:5d}  new best = {cur_val:.6f}")
        else:
            # no improvement – shrink the step
            step_frac = max(step_frac * 0.7, 1e-5)

        # occasional random perturbation to escape plateaus
        if it % 200 == 0 and rng.random() < 0.15:
            noise = rng.normal(scale=0.02, size=n)
            pert = cur_a + noise
            pert = np.maximum(pert, 0.0)
            pert = _project_to_simplex(pert, s=cur_a.sum())
            pert_val = _objective(pert)
            if pert_val < cur_val:
                cur_a, cur_val = pert, pert_val
                print(f"[subgrad] random perturbation improved to {cur_val:.6f}")

    return cur_a, cur_val


# --------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# --------------------------------------------------------------
def _guided_hill_climb(a0: np.ndarray,
                       best_val: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing that moves a small amount of mass away from the
    most offending convolution lag.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_val = best_val
    step_frac = 0.04        # initial fraction of the *pair* mass to move

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # contributions a[i] * a[k-i]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        if contrib.size == 0:
            continue

        # pair with the largest contribution to this lag
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # Choose two destination indices that increase the offending lag the
        # least.  For a candidate p the increase of conv[k] when adding δ to
        # a[p] is δ * a[k-p] (if that index exists).  We approximate this by
        # the score s[p] = a[p] + a[k-p] (both terms are non‑negative, zero when
        # out of range).  Small scores are “quiet’’ positions.
        scores = np.full(n, np.inf)
        valid = (np.arange(n) >= 0) & (np.arange(n) < n)
        # compute a[k-p] where it exists
        rev_idx = k - np.arange(n)
        mask = (rev_idx >= 0) & (rev_idx < n)
        scores[mask] = cur_a + cur_a[rev_idx[mask]]

        # pick two distinct quiet positions (avoid the offending pair)
        excl = {i_max, j_max}
        candidates = np.where(~np.isinf(scores))[0]
        candidates = np.setdiff1d(candidates, list(excl), assume_unique=True)
        if candidates.size < 2:
            # fallback – pick any two distinct positions
            p, q = rng.choice(n, size=2, replace=False)
        else:
            # smallest scores
            order = np.argsort(scores[candidates])
            p = candidates[order[0]]
            q = candidates[order[1] if candidates.size > 1 else order[0]]

        # Determine feasible mass to move
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-15:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max (the lag is even and the max comes from a diagonal term)
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-15:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        # safety – keep non‑negativity
        if (new_a < 0).any():
            continue

        new_val = _objective(new_a)

        if new_val < cur_val:
            cur_a, cur_val = new_a, new_val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = min(step_frac * 1.02, 0.2)   # enlarge a bit after success
        else:
            step_frac = max(step_frac * 0.95, 1e-5)

    return cur_a, cur_val

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights under the given time budget.

    The routine:
      * loads a previous construction if available,
      * runs a projected sub‑gradient descent,
      * finishes with a guided hill‑climbing phase,
      * returns the best list of floats found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise: reuse the best known construction if present,
    # otherwise start from the uniform distribution.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start

    # enforce feasibility (non‑negative, ℓ₁‑norm = 1)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = _objective(best_a)
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1. Sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a, deadline, rng)

    # ------------------------------------------------------------------
    # 2. Guided hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    final_score = _objective(best_a)
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    # Return a plain Python list as required by the framework
    return best_a.tolist()
