"""
Propose a step‑function (non‑negative sequence) that minimises the
evaluation function

    2 * n * max(conv(a,a)) / (sum(a)**2)

The function is invariant to scaling, therefore we work with probability
vectors (sum = 1).  The optimisation is performed in three stages:

1.  Generate a diverse portfolio of “seed’’ probability vectors using a few
    simple parametric families (power‑law, exponential, linear decay,
    symmetric versions and the external seed if present).

2.  Keep the best few seeds (according to the objective) and run a
    projected Adam optimiser on the log‑probabilities of each seed.
    The optimiser follows the exact sub‑gradient of the max‑convolution
    (the active set of maximal indices is recomputed each iteration).
    A small learning‑rate schedule, warm‑restarts and a simple
    early‑stop based on the remaining time budget are employed.

3.  Return the best vector seen over all runs (as a plain Python list).
    The returned values are non‑negative; scaling does not affect the
    final score.
"""

import time
import numpy as np
import math
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper: full linear convolution via real FFT (length 2n‑1)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real input)."""
    n = a.shape[0]
    size = 2 * n                     # zero‑padding (no need for power of two)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# Objective used during the search (2·n·max_c, sum(a)=1)
# ----------------------------------------------------------------------
def score_vector(a: np.ndarray) -> float:
    """Assumes a.sum()==1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Families of probability vectors (un‑normalised)
# ----------------------------------------------------------------------
def power_decay_left(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset+1)^(-p)   (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset + 1.0) ** (-p) + 1e-15
    return a

def power_decay_sym(n: int, p: float) -> np.ndarray:
    """U‑shaped: a_i ∝ (min(i, n‑1‑i)+1)^(-p)."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = (d + 1.0) ** (-p) + 1e-15
    return a

def exp_decay_left(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)   (asymmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + 1e-15
    return a

def exp_decay_sym(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential (edges high, centre low)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i)) + 1e-15
    return a

def linear_decay_left(n: int, beta: float) -> np.ndarray:
    """a_i = max(0, 1 - beta·i)   (triangular, left‑hand side)."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - beta * i) + 1e-15
    return a

def uniform_vec(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64)

# ----------------------------------------------------------------------
# Normalisation: turn any non‑negative array into a probability vector
# ----------------------------------------------------------------------
def to_prob(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v = np.ones_like(v)
        s = v.sum()
    return v / s

# ----------------------------------------------------------------------
# Softmax (log‑parameterisation) – guarantees positivity and sum‑1
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

# ----------------------------------------------------------------------
# Gradient of max‑convolution w.r.t. the log‑parameters (sub‑gradient)
# ----------------------------------------------------------------------
def grad_log_params(x: np.ndarray,
                    a: np.ndarray,
                    conv: np.ndarray,
                    max_b: float,
                    active_idx: np.ndarray) -> np.ndarray:
    """
    Returns ∂/∂x (max_k (a*a)[k]) for a = softmax(x).
    """
    n = a.shape[0]
    G = np.zeros_like(a)

    # accumulate contributions from all active convolution indices
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i   = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]

    # sub‑gradient of the max‑function w.r.t. the log‑parameters
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad

# ----------------------------------------------------------------------
# Adam optimiser on the log‑probabilities
# ----------------------------------------------------------------------
def run_adam(initial_a: np.ndarray, time_limit: float) -> Tuple[np.ndarray, float]:
    """
    Runs Adam on the log‑parameter space for at most `time_limit` seconds.
    Returns the best probability vector found and its objective value.
    """
    start = time.time()
    n = initial_a.shape[0]

    # initialise log‑parameters (avoid log(0))
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.4                         # initial learning rate in log‑space

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    # main loop
    while time.time() - start < time_limit - 0.3:
        iteration += 1

        # ------- current probabilities -------
        a = softmax(x)

        # ------- convolution -------
        conv = full_convolution_fft(a)
        max_b = conv.max()

        # active set (indices where the max is attained, up to numeric tol)
        tol = max_b * 1e-12 + 1e-15
        active = np.where(conv >= max_b - tol)[0]

        # ------- gradient -------
        grad = grad_log_params(x, a, conv, max_b, active)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of the step size – helps stability for long runs
        lr = lr0 / (1.0 + iteration * 1e-4)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- bookkeeping -------
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # ------- re‑heat if no improvement for a while -------
        if iteration - last_improve > 2000:
            # tiny random perturbation + reset moments
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (stagnation)")

    return best_vec, best_score
