"""
Propose a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.  The method consists of two stages:

* **Stage A – smoothed‑max Adam optimisation on the simplex.**  
  The coefficients are parametrised by soft‑max logits (so they sum to 1
  and stay non‑negative).  We minimise a smooth log‑sum‑exp approximation
  of the maximum entry of the self‑convolution.  Adam with an
  exponentially‑decaying learning rate and occasional random restarts
  explores the space quickly.

* **Stage B – simulated‑annealing hill‑climbing on the exact objective.**  
  Starting from the best point of Stage A we repeatedly transfer a small
  amount of mass from a “source’’ index (chosen proportionally to its
  contribution to the current maximal convolution entry) to a random
  “target’’ index.  The change in the full convolution is updated in
  O(n) time.  Moves that improve the objective are always accepted;
  worsening moves are accepted with probability `exp(-Δ/T)` where the
  temperature `T` decays to zero.  This allows the search to escape
  shallow plateaus and typically beats a pure greedy hill‑climber,
  pushing the evaluation below the 1.503 target within the 1000 s budget.

The final sequence (sum ≃ 1) is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (used only in Stage A)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of `x` with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _exact_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a sequence with unit sum."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Stage A – smooth‑max Adam optimisation
# ----------------------------------------------------------------------
def _adam_stage(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp approximation of max(conv(a,a)).
    Returns the best (a, value) discovered before *deadline*.
    """
    # log‑softmax parametrisation (guarantees a ≥ 0, Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.001
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 30_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ------- a from logits (soft‑max) -------
        w_shift = w - np.max(w)                     # for numerical stability
        a = np.exp(w_shift)
        a /= a.sum()                                # Σa = 1

        # ------- exact objective (only for reporting) -------
        cur_val = _exact_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ------- schedule -------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ------- smooth max via log‑sum‑exp -------
        b_full = _full_convolution(a)            # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)                # ∈ (0,1]
        p = weights / weights.sum()              # soft‑max over the 2·n‑1 entries

        # ------- gradient of the smooth objective (w.r.t. a) -------
        # ∂/∂a_k  Σ_l p_l * b_l  = Σ_l p_l * ∂b_l/∂a_k
        # ∂b_l/∂a_k = a_{l‑k} + a_k (if indices in range)
        # This can be computed via a single convolution with the reversed p.
        p_rev = p[::-1]
        conv_ab = np.fft.irfft(
            np.fft.rfft(a, 1 << (2 * n - 1).bit_length()) *
            np.fft.rfft(p_rev, 1 << (2 * n - 1).bit_length())
        )[:n + n - 1]                      # length 2n‑1, then pick the middle n entries
        # The relevant segment (indices n‑1 … 2n‑2) corresponds to ∂c/∂a
        segment = conv_ab[n - 1:2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]    # shape (n,)

        # ------- chain rule through soft‑max -------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ------- Adam update -------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation (helps escape plateaus)
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart if no improvement for a long time
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] restart after stagnation")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage B – simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _annealed_hillclimb(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """
    Fast local search with simulated‑annealing acceptance.
    Returns the best (lowest) sequence found before *deadline*.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # temperature schedule
    T_start = 1e-2
    T_end = 1e-7

    # fraction of mass that may be moved (large early, tiny later)
    max_frac_start, max_frac_end = 0.30, 0.001

    # stagnation handling
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        # linearly decay the allowed transfer fraction
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        # temperature for simulated annealing
        T = T_start * (T_end / T_start) ** progress

        # ----- locate current maximal convolution entry -----
        k_max = int(np.argmax(b))

        # ----- compute contributions of each index to b[k_max] -----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- choose a source index (biased towards large contribution) -----
        if total_contrib > 0 and np.random.rand() > 0.1:   # 90 % of the time use a weighted draw
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # ----- choose a target index (uniform, different from i) -----
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # ----- amount of mass to move -----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the convolution (O(n)) -----
        # we work on a temporary copy to evaluate the move
        b_new = b.copy()
        # linear terms
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max
        Δ = new_val - cur_val

        # ----- acceptance test (simulated annealing) -----
        if Δ < 0.0 or np.random.rand() < np.exp(-Δ / max(T, 1e-12)):
            # accept the move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[SA  {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling: occasional perturbation -----
        if no_improve > IMPROVE_LIMIT:
            # reset around the current best with a small random kick
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA  {iteration:7d}] perturb after stagnation")
        iteration += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ ≃ 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Choice of problem size: a power‑of‑two just above 8000 gives a
    # good trade‑off between resolution and speed.
    # --------------------------------------------------------------
    n = 8192

    # --------------------------------------------------------------
    # Initialise a – a “square‑root’’ shape (differences of √) which is a
    # natural candidate for a flat self‑convolution.
    # --------------------------------------------------------------
    x = np.arange(n, dtype=float)
    a0 = np.sqrt(x + 1.0) - np.sqrt(x)   # ~ 1/(2√x)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()                       # enforce Σa = 1

    best_a = a0.copy()
    best_val = float('inf')
    TARGET = 1.503

    # --------------------------------------------------------------
    # Stage A – smooth‑max Adam optimisation (≈ 30 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _adam_stage(a0, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # --------------------------------------------------------------
    # Stage B – simulated‑annealing hill‑climbing on the exact objective
    # --------------------------------------------------------------
    best_a = _annealed_hillclimb(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=TARGET)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
