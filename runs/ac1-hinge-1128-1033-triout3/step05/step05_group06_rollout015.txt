    best_score = np.inf
    best_a = None
    iter_count = 0
    last_improve_iter = 0
    warmup_end = start + 0.2 * time_limit
    beta_start = 10.0
    beta_max = 200.0

    while time.time() - start < time_limit - safety:
        iter_count += 1

        a = softmax(x)
        # full convolution via FFT

        max_b = conv.max()
        score = 2 * len(a) * max_b

            best_score = score
            best_a = a.copy()
            last_improve_iter = iter_count

        if time.time() < warmup_end:

            # compute beta increasing linearly with time
            elapsed = time.time() - start

            beta = beta_start + frac * (beta_max - beta_start)
            # compute softmax weights

            # maybe reset best? Not

"""
propose_candidate
=================

A timeout‑aware search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The algorithm works on probability vectors
(`sum = 1`) and uses a projected Adam optimiser on the log‑softmax
parameters.

Two stages are employed:

* **Warm‑up (≈10 % of the allocated time)** – a smooth approximation of
  the ``max`` term is used.  The loss is
  ``Lβ(a) = (1/β)·log( Σ_k exp(β·(a*a)[k]) )`` and its gradient is
  obtained via a cross‑correlation (FFT).  The temperature ``β`` is
  increased linearly from 5 to 200 during the warm‑up.

* **Refinement** – the exact sub‑gradient of the ``max`` term (active
  set of indices attaining the maximum) is used.  This gradient is
  computed with a cheap O(|active|·n) loop.

Multiple simple parametric families (power‑law, exponential, cosine,
sine, quadratic) are evaluated quickly, the best few are taken as
starting points and each is optimised for its allotted share of the
remaining time budget.  Re‑heating (small random perturbation of the
log‑parameters) is triggered when no improvement is seen for a large
number of iterations.

The best vector found (converted to a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution a * a using FFT.
    Returns a real array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.fft(a, size)
    conv = np.fft.ifft(fa * fa).real
    return conv[: 2 * n - 1]


def cross_correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[i] = Σ_j a[j]·w[i+j] for i = 0…len(a)-1.
    Computed via FFT.  `w` is assumed to be already normalised (Σ w = 1).
    """
    size = next_pow2(len(a) + len(w) - 1)
    fa = np.fft.fft(a, size)
    fw = np.fft.fft(w, size)
    corr = np.fft.ifft(np.conj(fa) * fw).real
    return corr[: len(a)]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def score_vector(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) for a probability vector (sum‑1)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families – used to generate diverse seeds
# ----------------------------------------------------------------------
def power_law_sym(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exponential_sym(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cosine_bump(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i / (n - 1) - 0.5)) + 1.0  # zero at ends, 2 at centre
    a = np.maximum(a, 1e-14) ** power
    a /= a.sum()
    return a


def sine_bump(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) + 1e-14
    a = np.maximum(a, 1e-14) ** power
    a /= a.sum()
    return a


def quadratic_sym(n: int, p: float = 0.8) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on log‑probabilities (probability simplex)
# ----------------------------------------------------------------------
def run_optimizer(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Optimises a probability vector `initial_a` using projected Adam.
    Returns the best vector encountered and its score.
    """
    start = time.time()
    n = initial_a.shape[0]

    # initialise log‑parameters safely
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.3          # base learning rate in log‑space

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    # warm‑up parameters (smooth max)
    warmup_end = start + 0.10 * time_limit          # 10 % of budget
    beta_start, beta_max = 5.0, 200.0

    reheat_patience = 20000      # iterations without improvement before re‑heat

    while time.time() - start < time_limit - 0.5:
        iteration += 1

        # ------- probability vector -------
        a = softmax(x)

        # ------- convolution and max -------
        conv = full_convolution_fft(a)
        conv = np.clip(conv, 0.0, None)  # numerical safety
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # ------- keep global best -------
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best = {best_score:.6f}")

        # ------- decide which gradient to use -------
        now = time.time()
        if now < warmup_end:
            # ---- smooth approximation of max (softmax) ----
            elapsed = now - start
            frac = elapsed / (warmup_end - start)
            beta = beta_start + frac * (beta_max - beta_start)

            # stable softmax weights over the convolution entries
            exps = np.exp(beta * (conv - max_b))   # ≤ 1
            w = exps / exps.sum()
            # gradient w.r.t. a via cross‑correlation
            grad_a = 2.0 * cross_correlation_fft(a, w)
        else:
            # ---- exact sub‑gradient (active set) ----
            tol = max_b * 1e-12 + 1e-15
            active = np.where(conv >= max_b - tol)[0]
            # small loop – active set is typically tiny
            G = np.zeros_like(a)
            for k in active:
                si = max(0, k - (n - 1))
                ei = min(n - 1, k)
                i_range = np.arange(si, ei + 1, dtype=np.int64)
                j_range = k - i_range
                G[i_range] += 2.0 * a[j_range]
            grad_a = G

        # ------- convert to gradient w.r.t. log‑params -------
        dot = np.dot(a, grad_a)               # scalar a·∇_a f
        grad_x = a * (grad_a - dot)           # Jacobian of softmax

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple step‑size decay
        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- re‑heat if stuck -------
        if iteration - last_improve > reheat_patience:
            x += np.random.randn(n) * 1e-4
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

    return best_vec, best_score
