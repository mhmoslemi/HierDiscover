"""
A fast two‑stage optimizer for the step‑function problem.

Stage 1 – Linear‑programming “expansion” step.
    With f fixed, we solve
        max   Σ g_i
        s.t.  f * g ≤ M   (entry‑wise) ,   g ≥ 0
    where M = max(f*f) and “*” denotes linear convolution.
    The LP is a very sparse problem (sliding‑window constraints) that can be
    built and solved in O(n²) time using SciPy’s HiGHS solver.
    The resulting g is scaled to the same ℓ₁‑norm as f and mixed with f.
    This step repeatedly reduces the peak M.

Stage 2 – High‑precision sub‑gradient descent.
    After the LP passes we refine the result with the exact max‑sub‑gradient
    (the same gradient that the reference code uses) together with an
    Adam‑style adaptive step size.  The vector is kept on the simplex
    (non‑negative, prescribed sum) via a simple renormalisation.

The two stages are run within the 1000 s budget; the best sequence seen
throughout the run is returned.  All numbers are clipped to the limits
imposed by `evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


def _make_conv_matrix(f: np.ndarray, row_idx: np.ndarray, col_idx: np.ndarray) -> sp.csr_matrix:
    """
    Construct the sparse matrix A (shape (2n‑1, n)) whose product
    with a vector g yields the linear convolution f*g.
    """
    data = np.repeat(f, f.shape[0])
    return sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * f.shape[0] - 1, f.shape[0]))

                break

    # --------------------------------------------------------------
    # Stage 2 – Adam‑style sub‑gradient descent (exact max‑sub‑gradient)
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.12

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0
    t_adam = 0
    stagn = 0

    while time.time() < deadline:
        t_adam += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _full_convolution_fft(a, a)
        M = conv.max()
        obj = M                                   # with our scaling, objective == M
        if obj < best_M - 1e-12:
            best_M = obj
            best_seq = a.copy()
            stagn = 0
            print(f"[adam iter {t_adam:6d}] new best={best_M:.12f}")
        else:
            stagn += 1

        # active set (indices attaining the maximum)
        tol = 1e-14
        active = np.flatnonzero(conv >= M - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        # sub‑gradient of max‑conv: g = 2 * rev(a) * w (restricted to n entries)
        rev_a = a[::-1]
        g_sub = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_sub
        v = beta2 * v + (1.0 - beta2) * (g_sub * g_sub)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_candidate = a - step
        # simplex projection (scale to target_sum afterwards)
        a_candidate = _project_simplex(a_candidate) * target_sum
        a = a_candidate

        if stagn > 4000:
            noise = rng.normal(scale=0.003 * target_sum, size=n)
            a = a + noise
            a = np.maximum(a, eps)
            a = a / a.sum() * target_sum
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            print(f"[adam perturb] iter {t_adam}")

        if t_adam % 5000 == 0:
            a = a / a.sum() * target_sum

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # optional logging of the true score
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")

    return final_seq
