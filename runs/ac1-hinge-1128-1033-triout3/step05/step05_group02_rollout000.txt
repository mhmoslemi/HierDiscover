    j = idx - k

    return g

def _gd_max(a, max_iters, time_deadline, n):
    best_a = a.copy()
    best_val = 2*n * _max_conv(a) # internal objective function
    a_curr = a.copy()
    for t in range(1, max_iters+1):
        if time.time() >= time_deadline:
            break
        conv = _full_convolution(a_curr)
        max_val = conv.max()
        idx = int(conv.argmax())
        obj = 2*n * max_val
        if obj < best_val:
            best_val = obj
            best_a = a_curr.copy()
            print(f"[gd] iter {t}, new best val {best_val:.6f}")
        # compute gradient
        g = np.zeros_like(a_curr)
        low = max(0, idx - (n-1))
        high = min(idx, n-1)
        if low <= high:
            ks = np.arange(low, high+1)
            js = idx - ks
            g[js] = a_curr[ks]
        # step size schedule
        eta = 0.2 / np.sqrt(t)  # 0.2 maybe.
        a_new = a_curr - eta * g
        # project onto simplex
        a_curr = _project_simplex(a_new)
    return best_a, best_val

Implementation of project onto simplex:

def _project_simplex(v):
    # shape (n,)
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # Find rho: the largest i where u_i - cssv_i/(i+1) > 0
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)

"""
Propose a candidate step‑function using projected sub‑gradient descent on the
∞‑norm of the self‑convolution (the actual objective).  The method works as
follows:

* Start from many random points on the simplex (sum = 1, a_i ≥ 0) or from the
  previously‑found construction if it exists.
* For each start run a projected sub‑gradient descent on
        f(a) = max_i (a * a)[i]
  The sub‑gradient at the maximal index k is simply the shifted copy of a:
        (∇f)_j = a_{k−j}  (if the index is inside the array, otherwise 0).
  After each gradient step the vector is projected back onto the probability
  simplex.
* After the descent, try the anti‑symmetric re‑ordering heuristic (large
  values at opposite ends) and optionally a short second descent.
* Keep the best sequence seen so far (according to the true objective
  2·n·max_conv, which equals the value returned by ``evaluate_sequence`` when
  the sum of the sequence is 1).
* Finally perform a fast hill‑climbing refinement that moves small amounts of
  mass between positions and swaps entries, accepting only strict improvements.

The whole search respects the 1000 s time budget, prints progress, and returns
the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, all top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Numerical noise may create tiny negatives – clip to 0.
    np.maximum(conv, 0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find the rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_conv(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of max_i (a*a)[i] at the index ``idx``.
    For each j we have (∇max)_j = a_{idx−j} if 0 ≤ idx−j < n, else 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    # The overlapping region of the two copies
    low = max(0, idx - (n - 1))
    high = min(idx, n - 1)
    if low <= high:
        ks = np.arange(low, high + 1)          # source indices in a
        js = idx - ks                          # destination indices
        g[js] = a[ks]
    return g


def _gd_max_convolution(
    a_start: np.ndarray,
    max_iters: int,
    deadline: float,
    n: int,
    verbose_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max_i (a*a)[i].
    Returns the best sequence (sum = 1) and its objective value
    2·n·max_conv.
    """
    a = a_start.copy()
    best_a = a.copy()
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a)
        max_val = conv.max()
        idx = int(conv.argmax())
        cur_val = 2.0 * n * max_val

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{verbose_tag}] iter {t:4d} new best = {best_val:.6f}")

        # Sub‑gradient at the current maximal index
        g = _subgradient_max_conv(a, idx)

        # Diminishing step size (η₀ = 0.2 works well for n≈4000)
        eta = 0.2 / np.sqrt(t)

        # Gradient step followed by simplex projection
        a = _project_simplex(a - eta * g)

    return best_a, best_val


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Re‑order the vector so that large values are placed at opposite ends.
    This heuristic often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)  # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large → left side
        new_a[n - 1 - i] = sorted_vals[i]      # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]        # middle element stays
    return new_a


def _refine_hill_climb(
    a: np.ndarray,
    cur_val: float,
    deadline: float,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Simple hill climbing: random swaps (20 % chance) or small mass transfers.
    Accept only strict improvements (using the true ``evaluate_sequence``).
    """
    cur_a = a.copy()
    cur_best = cur_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:
            # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()  # up to 5 % of donor
            if delta <= 0.0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Evaluate using the official function (ensures all clipping/guards)
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_best - 1e-12:
            cur_a = new_a
            cur_best = val
            print(f"[refine] new best = {cur_best:.6f}")
    return cur_a, cur_best
