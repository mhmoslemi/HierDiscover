"""
Simulated‑annealing hill‑climbing on the probability simplex.

The algorithm starts from the best known construction (``height_sequence_1``)
or from a uniform distribution.  At each step we locate the convolution
entry with maximal value, pick the coefficient that contributes most to
that peak and move a random fraction of its mass to a destination that
contributes *least* to the same peak (preferably an index that does not
overlap the peak at all).  Moves that lower the objective are always
accepted; uphill moves are accepted with a Metropolis probability that
decreases linearly with time (simulated‑annealing).  Occasional random
“jitter’’ moves improve exploration.  The convolution is updated
incrementally (O(n)) and periodically recomputed with an FFT to avoid
numerical drift.  The best sequence encountered is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: exact linear convolution via real FFT (used for periodic
# re‑synchronisation).
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (length len(x)+len(y)-1) using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negatives from rounding
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – the caller cuts off at exactly ``budget_s`` s
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise the starting vector ``a``.
    # --------------------------------------------------------------
    try:
        init_seq = height_sequence_1                # noqa: F821
    except NameError:
        init_seq = None

    if init_seq is None:
        n = 1800
        a = np.full(n, 1.0 / n)
    else:
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.shape[0]

    # Normalise to a probability vector (scaling does not affect the score)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # --------------------------------------------------------------
    # Initialise the exact convolution and bookkeeping.
    # --------------------------------------------------------------
    conv = _full_convolution_fft(a, a)          # length 2n‑1
    cur_obj = float(2.0 * n * conv.max())       # 2·n·max(b)
    best_obj = cur_obj
    best_a = a.copy()

    print(f"[init] n={n}, objective={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for the 1000 s budget)
    # --------------------------------------------------------------
    max_frac_start = 0.45          # at the beginning up to 45 % of source mass may move
    max_frac_end   = 0.015         # near the end we shrink to about 1.5 %
    min_frac       = 0.001        # absolute lower bound for the transferred fraction
    jitter_prob    = 0.018        # probability of a random jitter move each iteration
    recompute_interval = 5000     # full FFT recompute frequency
    stagnation_limit = 300_000    # re‑normalise after this many non‑improving steps

    # Temperature for simulated annealing – roughly 2 % of the initial objective,
    # linearly cooled to almost zero.
    T0 = max(cur_obj * 0.02, 1e-8)

    iteration = 0
    last_improve_iter = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        progress = (now - start_time) / (deadline - start_time)

        # ---- annealing schedule ------------------------------------------------
        move_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        move_frac = max(move_frac, min_frac)                # never drop below ``min_frac``
        T = max(T0 * (1.0 - progress), 1e-12)               # temperature → 0

        # ---- locate the current peak ------------------------------------------
        k_star = int(conv.argmax())                         # index of max(b)
        # indices that actually contribute to b[k_star]
        i_start = max(0, k_star - (n - 1))
        i_end   = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        partner = k_star - idx
        products = a[idx] * a[partner]
        i_src = int(idx[products.argmax()])                 # source index

        # ---- choose a destination that hurts the peak the least -----------------
        # Prefer an index that does *not* overlap the peak (a[k_star‑j] = 0).
        if k_star < n:          # there exists a region j > k_star
            if k_star < n - 1:
                j_dst = np.random.randint(k_star + 1, n)
            else:               # k_star == n‑1 → fallback to minimal partner
                low = max(0, k_star - (n - 1))
                high = min(n - 1, k_star)
                cand = np.arange(low, high + 1)
                # a[partner] = a[k_star‑cand]
                a_partners = a[k_star - cand]
                mask = cand != i_src
                if np.any(mask):
                    j_dst = int(cand[mask][a_partners[mask].argmin()])
                else:
                    j_dst = (i_src + 1) % n
        else:                    # k_star >= n  → region j < k_star‑(n‑1)
            max_low = k_star - (n - 1) - 1
            if max_low >= 0:
                j_dst = np.random.randint(0, max_low + 1)
            else:               # fallback as above
                low = max(0, k_star - (n - 1))
                high = min(n - 1, k_star)
                cand = np.arange(low, high + 1)
                a_partners = a[k_star - cand]
                mask = cand != i_src
                if np.any(mask):
                    j_dst = int(cand[mask][a_partners[mask].argmin()])
                else:
                    j_dst = (i_src + 1) % n

        # ---- propose a transfer ------------------------------------------------
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue

        # keep a copy of the old vector for a fast revert
        a_old = a.copy()

        # incremental update of the convolution (exact formula)
        delta2 = delta * delta
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[j_dst:j_dst + n] += 2.0 * delta * a_old
        conv[2 * i_src] += delta2
        conv[2 * j_dst] += delta2
        conv[i_src + j_dst] -= 2.0 * delta2

        # apply the mass transfer
        a[i_src] -= delta
        a[j_dst] += delta

        # ---- evaluate new objective --------------------------------------------
        new_obj = float(2.0 * n * conv.max())

        # ---- Metropolis acceptance ------------------------------------------------
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            # uphill move – accept with probability exp(‑Δ/T)
            prob = np.exp(-(new_obj - cur_obj) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration:7d}] new best = {best_obj:.6f}")
                last_improve_iter = iteration
        else:
            # revert both a and the convolution
            a[:] = a_old
            conv[i_src:i_src + n] += 2.0 * delta * a_old
            conv[j_dst:j_dst + n] -= 2.0 * delta * a_old
            conv[2 * i_src] -= delta2
            conv[2 * j_dst] -= delta2
            conv[i_src + j_dst] += 2.0 * delta2

        # ---- occasional random jitter (large, unbiased move) --------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            if a[src] > 0.0:
                delta_j = 0.5 * a[src] * np.random.rand()
                a_old_j = a.copy()
                delta2j = delta_j * delta_j
                conv[src:src + n] -= 2.0 * delta_j * a_old_j
                conv[dst:dst + n] += 2.0 * delta_j * a_old_j
                conv[2 * src] += delta2j
                conv[2 * dst] += delta2j
                conv[src + dst] -= 2.0 * delta2j
                a[src] -= delta_j
                a[dst] += delta_j

                jitter_obj = float(2.0 * n * conv.max())
                accept_j = False
                if jitter_obj < cur_obj:
                    accept_j = True
                else:
                    prob_j = np.exp(-(jitter_obj - cur_obj) / T)
                    if np.random.rand() < prob_j:
                        accept_j = True

                if accept_j:
                    cur_obj = jitter_obj
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{iteration:7d}] jitter improved best = {best_obj:.6f}")
                        last_improve_iter = iteration
                else:
                    # revert jitter
                    a[:] = a_old_j
                    conv[src:src + n] += 2.0 * delta_j * a_old_j
                    conv[dst:dst + n] -= 2.0 * delta_j * a_old_j
                    conv[2 * src] -= delta2j
                    conv[2 * dst] -= delta2j
                    conv[src + dst] += 2.0 * delta2j

        # ---- periodic full recomputation (drift mitigation) --------------------
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a, a)

        # ---- simple stagnation handling (renormalise) -------------------------
        if iteration - last_improve_iter > stagnation_limit:
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_obj = float(2.0 * n * conv.max())
            last_improve_iter = iteration

    # ----------------------------------------------------------------------
    # Final reporting
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                                    # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
