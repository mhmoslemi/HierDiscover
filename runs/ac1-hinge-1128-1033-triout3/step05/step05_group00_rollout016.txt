def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np

    start = time.time()
    deadline = start + budget_s - 2.0
    rng = np.random.default_rng(seed)

    n = 6000
    best_val = np.inf
    best_seq = None

    # define helper functions: project_simplex, full_conv

    # Phase: Random restarts with simple gradient descent (softmax approx)
    while time.time() < deadline:
        # allocate a time slice each iteration (restart) maybe 5 sec
        # compute restart_end = min(deadline, time.time() + 5.0)  # ensure at least one iteration
        # initialize a
        alpha = 10 ** rng.uniform(-2, 1)  # alpha in [0.01,10]
        a_raw = rng.gamma(alpha, 1.0, size=n)
        a = a_raw / a_raw.sum()

        # inner loop
        eta = 0.5
        eps = None
        momentum = 0.9
        vel = np.zeros_like(a)
        stagn = 0

        while time.time() < deadline:
            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            score = 2.0 * n * max_conv / (np.sum(a) ** 2)
            # update best
            if score < best_val:
                best_val = score
                best_seq = a.copy()
                stagn = 0
                print(f"[restart log] new_best={best_val:.12f}")
            else:
                stagn += 1

            # softmax weight: temperature adaptively set
            if eps is None:
                eps = max(max_conv * 0.2, 1e-12)
            else:
                eps = max(eps * 0.9999, 1e-12)

            w = np.exp((conv - max_conv) / eps)
            w_sum = w.sum()
            w = w / w_sum

            rev_a = a[::-1]
            g = 2.0 * _full_convolution_fft(rev_a, w)[n-1:n-1+n]

            # momentum step
            vel = momentum * vel + (1 - momentum) * g

            a_new = a - eta * vel
            a = _project_simplex(a_new)

            # decays
            eta = max(1e-5, eta * 0.9995)

            # periodic injection of noise if stagnate
            if stagn > 2000:
                a = a + rng.normal(scale=0.001, size=n)
                a = _project_simplex(a)
                stagn = 0
                eta = max(1e-5, eta * 2)

            # break out of inner loop if time left for restart is low
            if time.time() > deadline - 5.0:
                break

        # after inner loop we continue to next restart

    # After time out, return best_seq processed
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] Could not compute final_score")
    return final_seq

Phase C – exact max‑sub‑gradient descent with Adam (as in the reference

The routine runs for at most `budget_s` seconds, prints intermediate

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x | x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (zero‑padded to the next power of two)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return a * b (full linear convolution)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

    eps_min = max(max_conv * 1e-5, 1e-12)
    eps_decay = 0.9995
    momentum = 0.9
    vel = np.zeros_like(a)
    it = 0
    stagn = 0

    while time.time() < phaseA_end:
        it += 1

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phaseA iter {it:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # soft‑max weights over the whole convolution
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # momentum update
        vel = momentum * vel + (1.0 - momentum) * g

        # gradient step + projection
        a_new = a - eta * vel
        a = _project_simplex(a_new)

        # schedule decay
        eta = max(eta_min, eta * eta_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional perturbation
        if stagn > 2500:
            a = a + rng.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            stagn = 0
            eta = max(eta_min, eta * 1.5)
            print(f"[phaseA perturb] iteration {it}")

    # ------------------------------------------------------------------
    # Phase B – top‑K sub‑gradient descent (Adam)
    # ------------------------------------------------------------------
    phaseB_end = start_time + 0.75 * budget_s
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0
    t_adam = 0
    stagn = 0

    while time.time() < phaseB_end:
        t_adam += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phaseB iter {t_adam:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # ---- top‑K weighting -------------------------------------------------
        K = max(1, int(conv.size * 0.001))          # about 12‑15 indices
        top_idx = np.argpartition(-conv, K - 1)[:K]
        w = np.zeros_like(conv)
        w[top_idx] = 1.0 / K

        # ---- gradient (exact sub‑gradient for the selected entries) ----------
        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a = _project_simplex(a_candidate)

        # ---- occasional perturbation ----------------------------------------
        if stagn > 4000:
            a = a + rng.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            print(f"[phaseB perturb] iter {t_adam}")

    # ------------------------------------------------------------------
    # Phase C – exact max sub‑gradient descent (Adam) for final polishing
    # ------------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0
    t_adam = 0
    stagn = 0

    while time.time() < deadline:
        t_adam += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phaseC iter {t_adam:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # ---- exact active set ------------------------------------------------
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = _project_simplex(a - step)

        # ---- perturb if stuck ------------------------------------------------
        if stagn > 4000:
            a = a + rng.normal(scale=0.001, size=n)
            a = _project_simplex(a)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            print(f"[phaseC perturb] iter {t_adam}")

    # ------------------------------------------------------------------
    # Convert best sequence to the format expected by evaluate_sequence
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")
    return final_seq
