def propose_candidate(budget_s=1000, seed=0):
    import time, numpy as np
    deadline = time.time() + budget_s - 0.5
    rng = np.random.default_rng(seed)
    best_seq_global = None
    best_val_global = float('inf')
    # maybe try multiple n values
    possible_n = [4096, 5000, 8192]   # choose based on time
    for n in possible_n:
        # if time left insufficient break
        if time.time() > deadline:
            break
        # initialization:
        a = initialize_shape(n, rng)  # function returns normalized a
        # set up variables
        lr = 0.5   # initial step factor for line search scaling (affects alpha_max)
        # For each run, maybe have some iterations limited by time
        it = 0
        stagn = 0
        while time.time() < deadline:
            it += 1
            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            score = 2.0 * n * max_conv
            # update best global
            if score < best_val_global - 1e-10:
                best_val_global = score
                best_seq_global = a.copy()
                stagn = 0
                print(f"[n{n} iter{it}] new best={best_val_global:.12f}")
            else:
                stagn += 1
            # gradient (active set)
            tol = max_conv * 1e-9 + 1e-14
            mask = (conv >= max_conv - tol).astype(np.float64)
            w = mask / mask.sum()
            grad_full = _full_convolution_fft(a[::-1], w)
            grad = 2.0 * grad_full[n-1:n-1+n]
            # compute max step allowed to keep non-neg
            pos = grad > 0
            if np.any(pos):
                alphas = a[pos] / grad[pos]
                max_alpha = alphas.min()
            else:
                max_alpha = 1.0
            # candidate step fractions
            fractions = [0.9, 0.5, 0.2, 0.05, 0.01]
            improved = False
            for frac in fractions:
                alpha = frac * max_alpha
                a_candidate = a - alpha * grad
                # ensure non-negative
                a_candidate = np.maximum(a_candidate, 0.0)
                a_candidate = _project_simplex(a_candidate)
                conv_cand = _full_convolution_fft(a_candidate, a_candidate)
                max_cand = conv_cand.max()
                score_cand = 2.0 * n * max_cand
                if score_cand < score - 1e-12:

def initialize_shape(n, rng):
    x = np.arange(n)
    shapes = []
    shapes.append(np.sin(np.pi* (x+0.5)/n) ** 2) # or just sin
    t = 2*x/(n-1)-1
    shapes.append(np.sqrt(np.clip(1 - t*t,0, None))) # semicircle
    sigma = n/6.0
    shapes.append(np.exp(-0.5 * ((x - (n-1)/2)/sigma)**2))
    shapes.append(np.minimum(x, n-1-x)+1) # triangle
    # power law decreasing: a_i ∝ 1/(i+1)^p with random p in [0.5,2]
    p = rng.uniform(0.5,2.0)
    shapes.append(1.0/(np.arange(1,n+1)**p))
    # combine
    weights = rng.random(len(shapes))
    weights /= weights.sum()
    a = sum(w*s for w,s in zip(weights, shapes))
    a = np.maximum(a,0)
    a /= a.sum()
    return a
```

"""
Hybrid random‑search + sub‑gradient refinement for the step‑function problem.

The function explores a variety of parametric shapes (sine, semicircle,
Gaussian, triangle, power‑law, exponential, Kaiser, DPSS) at several
lengths, keeps the best candidate seen so far and, when a promising
candidate is found, runs a few thousand iterations of a simple
projected sub‑gradient method with a line‑search on the exact max‑convolution
constraint.  Random jitter is added when progress stalls.
Finally a short “hard‑max” polishing pass is performed and the result is
clamped to the limits required by `evaluate_sequence`.

The routine respects the given time budget (default 1000 s) and prints
intermediate scores.
"""

import time
import numpy as np
from math import ceil, log2

# Optional import – DPSS and Kaiser windows from SciPy
try:
    from scipy.signal import windows
except Exception:          # pragma: no cover
    windows = None


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the probability simplex {x ≥ 0, sum(x)=1}
    (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


def _init_mixed_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random mixture of a few classic smooth shapes."""
    x = np.arange(n, dtype=np.float64)

    # basic ingredients
    sin_shape = np.sin(np.pi * (x + 0.5) / n)
    t = 2.0 * x / (n - 1) - 1.0
    semicircle_shape = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
    sigma = n / 6.0
    gauss_shape = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
    triangle_shape = np.minimum(x, n - 1 - x) + 1.0

    bases = [sin_shape, semicircle_shape, gauss_shape, triangle_shape]
    weights = rng.random(len(bases))
    weights /= weights.sum()
    a = sum(w * b for w, b in zip(weights, bases))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _power_law_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p) (right‑hand side)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _kaiser_shape(n: int, beta: float) -> np.ndarray:
    """Kaiser window (always non‑negative)."""
    if windows is None:                     # pragma: no cover
        return _init_mixed_shape(n, np.random.default_rng())
    a = windows.kaiser(n, beta, sym=False)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _dpss_shape(n: int, NW: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) sequence – first DPSS."""
    if windows is None:                     # pragma: no cover
        return _init_mixed_shape(n, np.random.default_rng())
    # windows.dpss returns shape (Kmax, n); take the first eigenvector
    a = windows.dpss(n, NW, Kmax=1)[0]
    a = np.maximum(a, 0.0)                 # DPSS should already be ≥0
    a /= a.sum()
    return a


def _refine_local(a: np.ndarray, n: int, deadline: float,
                  rng: np.random.Generator) -> np.ndarray:
    """Simple projected sub‑gradient descent with an adaptive line‑search."""
    # we keep the best local version seen during refinement
    best_a = a.copy()
    best_score = 2.0 * n * _full_convolution_fft(a, a).max()
    it = 0
    while time.time() < deadline and it < 3000:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        # build the active mask (indices that are essentially maximal)
        tol = max_conv * 1e-9 + 1e-14
        mask = (conv >= max_conv - tol).astype(np.float64)
        w = mask / mask.sum()
        # gradient of the max (sub‑gradient)
        grad_full = _full_convolution_fft(a[::-1], w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # maximal step that keeps the iterate non‑negative (before projection)
        pos = grad > 1e-14
        if np.any(pos):
            alphas = a[pos] / grad[pos]
            max_alpha = alphas.min()
        else:
            max_alpha = 1.0

        improved = False
        # try a few decreasing fractions of the maximal step
        for frac in (0.9, 0.5, 0.2, 0.05, 0.01):
            alpha = frac * max_alpha
            cand = a - alpha * grad
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)
            cand_score = 2.0 * n * _full_convolution_fft(cand, cand).max()
            if cand_score < best_score - 1e-12:
                a = cand
                best_a = cand.copy()
                best_score = cand_score
                improved = True
                break

        if not improved:
            # occasional jitter to escape tiny plateaus
            if it % 500 == 0:
                a += rng.normal(scale=0.004, size=n)
                a = np.maximum(a, 0.0)
                a = _project_simplex(a)
    return best_a


def _hard_max_polish(a: np.ndarray, n: int, steps: int = 2000) -> np.ndarray:
    """Final polishing using the binary mask of the true maximal entries."""
    for _ in range(steps):
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        tol = max_conv * 1e-9 + 1e-14
        mask = (conv >= max_conv - tol).astype(np.float64)
        w = mask / mask.sum()
        grad_full = _full_convolution_fft(a[::-1], w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]
        # a tiny constant step works well in this regime
        a = a - 2e-5 * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
    return a

    """
    Search for a non‑negative height sequence that minimises the function
    used in `evaluate_sequence`.  The search combines random parametric
    generation, a fast line‑search sub‑gradient refinement and a final
    hard‑max polishing pass.
    """
    deadline = time.time() + budget_s - 0.5   # a small safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Global best bookkeeping
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')

    # Allow warm‑start from the global variable if present
    if 'height_sequence_1' in globals():
        try:
            start = np.asarray(height_sequence_1, dtype=np.float64)
            start = np.maximum(0.0, start)
            if start.sum() > 0.0:
                start = start / start.sum()
                n0 = start.shape[0]
                val0 = 2.0 * n0 * _full_convolution_fft(start, start).max()
                best_seq = start.copy()
                best_val = val0
                print(f"[warm‑start] n={n0}, score={best_val:.12f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Parameter spaces we are willing to explore
    # ------------------------------------------------------------------
    length_choices = [2048, 4096, 5000, 8192]
    # Order of generation strategies (avoid lambda‑closures)
    # 0 – mixed smooth shapes, 1 – power‑law, 2 – exponential,
    # 3 – Kaiser, 4 – DPSS (if SciPy available)
    strategy_count = 5

    cand_counter = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Pick a random problem size and a random generation strategy
        # ------------------------------------------------------------------
        n = int(rng.choice(length_choices))
        strat = int(rng.integers(0, strategy_count))
        if strat == 0:
            a = _init_mixed_shape(n, rng)
        elif strat == 1:
            p = rng.uniform(0.2, 3.0)               # power‑law exponent
            a = _power_law_shape(n, p)
        elif strat == 2:
            lam = rng.uniform(5e-4, 1e-2)           # exponential decay rate
            a = _exponential_shape(n, lam)
        elif strat == 3:
            beta = rng.uniform(0.5, 10.0)           # Kaiser beta
            a = _kaiser_shape(n, beta)
        else:   # DPSS – only if SciPy is available
            if windows is not None:
                NW = rng.integers(2, 7)             # typical bandwidth parameters
                a = _dpss_shape(n, float(NW))
            else:
                a = _init_mixed_shape(n, rng)

        # ------------------------------------------------------------------
        # Evaluate the raw candidate quickly (FFT based)
        # ------------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_c = conv.max()
        score = 2.0 * n * max_c

        cand_counter += 1
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            print(f"[candidate {cand_counter}] new best raw score={best_val:.12f} (n={n})")

            # ------------------------------------------------------------------
            # If we still have time, run a short local refinement on this
            # promising candidate.
            # ------------------------------------------------------------------
            if time.time() < deadline - 0.5:
                refined = _refine_local(a, n, deadline, rng)
                conv_ref = _full_convolution_fft(refined, refined)
                score_ref = 2.0 * n * conv_ref.max()
                if score_ref < best_val - 1e-12:
                    best_val = score_ref
                    best_seq = refined.copy()
                    print(f"   [refine] improved to {best_val:.12f}")

    # ------------------------------------------------------------------
    # Final polishing (hard‑max sub‑gradient) – cheap and often helps a
    # little more.
    # ------------------------------------------------------------------
    if best_seq is not None:
        n_best = best_seq.shape[0]
        best_seq = _hard_max_polish(best_seq, n_best, steps=2000)

    # ------------------------------------------------------------------
    # Convert to the format required by `evaluate_sequence` and report.
    # ------------------------------------------------------------------
    if best_seq is None:                     # fallback: uniform distribution
        n_fallback = 5000
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:
        final_score = evaluate_sequence(final_seq)
    except Exception:                       # pragma: no cover
        final_score = float('nan')
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
