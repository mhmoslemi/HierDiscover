"""
Propose a non‑negative step function that minimizes `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – try to reuse the historic best (`height_sequence_1`);
   otherwise fall back to a smooth Gaussian bump mixed with a tiny uniform
   background.  The vector is normalised to sum 1.

2. **Phase A – projected gradient descent on the true max‑convolution**.
   The objective (for a fixed ℓ₁–norm) is 2·n·max(conv(a,a)).
   We iteratively:
   * compute the full self‑convolution,
   * locate the index `k_max` where it attains its maximum,
   * evaluate the sub‑gradient of `max(conv)` w.r.t. `a`,
   * take a small step in the negative gradient direction,
   * project back onto the simplex (clip negatives and renormalise).

   The learning rate decays linearly from 0.2 to 0.001.  The best vector
   encountered is stored.

3. **Phase B – stochastic hill‑climbing with simulated annealing**.
   Starting from the best vector of Phase A we repeatedly transfer a tiny
   amount of mass `δ` from a *source* index `i` (chosen proportionally to its
   contribution to the current maximal convolution entry) to a *target*
   index `j` (uniformly at random).  The convolution is updated in O(n) by
   an exact rank‑one update; the new objective is computed and the move is
   accepted if it improves the score, or with a probability `exp(-(Δ)/T)`
   where the temperature `T` decays exponentially during the run.
   After a long streak without improvement the current best is perturbed
   by small Gaussian noise to escape stagnation.

Both phases respect the overall time budget (`budget_s`).  The best sequence
found is finally returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution x * x (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective helper (exact, assumes sum(a) == 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Gradient of the max‑convolution w.r.t. a
# ----------------------------------------------------------------------
def _grad_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of max(conv(a,a)) at the current maximal index."""
    n = a.shape[0]
    idx = np.arange(n)
    # first term: a[k_max - i]   (if the index lies in [0, n))
    j1 = k_max - idx
    mask1 = (j1 >= 0) & (j1 < n)
    grad = np.zeros_like(a)
    grad[mask1] += a[j1[mask1]]

    # second term: a[i - k_max] (if the index lies in [0, n))
    j2 = idx - k_max
    mask2 = (j2 >= 0) & (j2 < n)
    grad[mask2] += a[j2[mask2]]

    return grad


# ----------------------------------------------------------------------
# Phase A – projected gradient descent
# ----------------------------------------------------------------------
def _gradient_descent_phase(a0: np.ndarray,
                            deadline: float,
                            best_a: np.ndarray,
                            best_val: float) -> tuple[np.ndarray, float]:
    """Run projected gradient descent on max(conv) until *deadline*."""
    a = a0.copy()
    n = a.shape[0]

    # hyper‑parameters
    lr_start, lr_end = 0.20, 0.001
    start_time = time.time()
    it = 0
    no_improve = 0
    IMP_LIMIT = 150_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # convolution and objective
        b = _full_convolution(a)
        k_max = int(np.argmax(b))
        cur_max = b[k_max]
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # learning‑rate schedule (linear in time)
        progress = (now - start_time) / (deadline - start_time + 1e-12)
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # gradient of the *max* entry
        grad = _grad_max_conv(a, k_max)

        # gradient step and projection onto simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            # fallback to a random point on the simplex
            a_new = np.random.dirichlet(np.ones_like(a))
        else:
            a_new /= s

        a = a_new
        it += 1

        # occasional restart if we are stuck
        if no_improve > IMP_LIMIT:
            a = np.random.dirichlet(np.ones(n))
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – stochastic hill‑climbing with simulated annealing
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """Fine‑grained local search using mass‑transfer moves."""
    a = a_start.copy()
    n = a.shape[0]

    # pre‑compute the convolution
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # schedule parameters
    max_frac_start, max_frac_end = 0.5, 0.005
    temp_start, temp_end = best_val * 0.05, 1e-6

    start_time = time.time()
    no_improve = 0
    IMP_LIMIT = 300_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time + 1e-12)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        temperature = temp_start * (1.0 - progress) + temp_end * progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – biased toward large contributions
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # target index – uniform, ensure i != j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- Incremental convolution update (O(n)) ----
        b_new = b.copy()
        # linear terms
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms (ordered pairs)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        # evaluate new objective
        new_val = 2.0 * n * b_new.max()

        # accept move?
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            # simulated annealing acceptance
            if temperature > 0.0:
                prob = np.exp(-(new_val - cur_val) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
        else:
            no_improve += 1

        # stagnation handling – perturb the current best
        if no_improve > IMP_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.random.dirichlet(np.ones(n))
            else:
                a /= s
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a
