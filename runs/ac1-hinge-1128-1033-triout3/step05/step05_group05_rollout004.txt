"""
Hybrid optimization for the min‑max auto‑convolution problem.

The algorithm consists of three stages:

1. **L2‑flattening (gradient descent)** – minimise the L² variance of the
   full convolution `b = a*a`.  This makes the convolution flatter and gives
   a good starting shape.  After each gradient step the vector is projected
   onto the probability simplex.

2. **Targeted max‑reduction (sub‑gradient hill‑climbing)** – repeatedly
   reduces the current maximum entry of `b`.  For the index `k*` where
   `b` attains its maximum we compute the “influence” vector
   `g[i] = a[k*‑i]` (zero outside the support).  Moving a small amount of
   mass from a source `i` with large `a[i]*g[i]` to a target `j` with a
   small `g[j]` provably decreases `b[k*]`.  The convolution is updated
   incrementally in O(n) time, and the best solution seen so far is kept.

3. **Polish (random hill‑climbing)** – a short final phase that performs
   cheap random mass‑transfer moves (as in the reference implementation)
   to escape possible tiny local minima.

All steps respect the non‑negativity and simplex constraints
(`∑a = 1`).  The routine respects the 1000 s budget and returns the best
list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero; return uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase 1 – L2 gradient descent (flattening the convolution)
# ----------------------------------------------------------------------
def _l2_gradient_phase(a: np.ndarray,
                       n: int,
                       phase_deadline: float,
                       start_time: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """
    Perform gradient descent on the L2 variance of the convolution.
    Returns updated (best_a, best_val) and the (possibly changed) a vector.
    """
    avg_target = 1.0 / (2 * n - 1)  # desired constant value of b when Σa = 1
    base_step = 0.5
    min_step = 0.001
    iter_c = 0
    no_improve_c = 0
    IMPROVE_LIMIT_C = 180_000

    while time.time() < phase_deadline:
        b = _full_convolution(a)          # length 2n‑1
        cur_val = 2.0 * n * b.max()       # exact objective (∑a = 1)

        # keep the best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve_c = 0
            print(f"[L2 {iter_c:07d}] new best = {best_val:.6f}")
        else:
            no_improve_c += 1

        # residual to flat target
        r = b - avg_target

        # gradient of Σ(r_k²) w.r.t. a (see analysis)
        a_rev = a[::-1]
        conv1 = _conv_fft(r, a)          # length ≈ 3n‑2, we will slice
        conv2 = _conv_fft(r, a_rev)
        start = n - 1
        end = 2 * n - 1
        grad = 2.0 * (conv1[start:end] + conv2[start:end])

        # step size schedule (linear decay)
        progress = (time.time() - start_time) / (phase_deadline - start_time)
        step = base_step * (1.0 - progress) + min_step * progress

        # gradient step + projection
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional jitter to escape shallow traps
        if np.random.rand() < 0.003:
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)

        # restart after long stagnation
        if no_improve_c > IMPROVE_LIMIT_C:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve_c = 0
            print(f"[L2 {iter_c}] restart after stagnation")

        iter_c += 1

    return a, best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Targeted max‑reduction hill climbing
# ----------------------------------------------------------------------
def _max_hillclimb_phase(a_start: np.ndarray,
                         n: int,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Reduce the maximum entry of the convolution by moving mass from indices
    that contribute heavily to the current peak to indices that contribute
    little.  Updates the convolution incrementally in O(n) per move.
    """
    a = a_start.copy()
    b = _full_convolution(a)                # length 2·n‑1
    cur_val = 2.0 * n * b.max()
    max_frac = 0.4                         # max fraction of source mass to move
    IMPROVE_LIMIT = 150_000                # stagnation trigger
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # current peak and its “influence” vector
        k_star = int(np.argmax(b))

        # g[i] = a[k_star - i] where defined, else 0
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        idx_i = np.arange(low, high + 1, dtype=int)
        g[idx_i] = a[k_star - idx_i]

        # source selection – favour large a[i]·g[i]
        prod = a * g
        if prod.sum() > 0 and np.random.rand() < 0.9:
            i = int(np.argmax(prod))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            # nothing to move from this index – pick a random non‑zero source
            nz = np.where(a > 1e-14)[0]
            if nz.size == 0:
                break
            i = int(np.random.choice(nz))

        # target selection – smallest g (least influence on the peak)
        g_copy = g.copy()
        g_copy[i] = np.inf                     # avoid picking the source again
        j = int(np.argmin(g_copy))

        # amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept only improving moves
        if new_val < cur_val - 1e-13:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[MaxHill {iteration:07d}] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # perturb after stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[MaxHill {iteration:07d}] stagnation – perturbing")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Light random hill climbing (polish)
# ----------------------------------------------------------------------
def _random_hillclimb_phase(a_start: np.ndarray,
                            n: int,
                            deadline: float,
                            best_a: np.ndarray,
                            best_val: float) -> tuple[np.ndarray, float]:
    """
    A lightweight version of the original random hill‑climbing used as a
    final polishing step.  It moves a small random amount of mass between
    two indices and accepts the move only if the true objective improves.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    max_frac = 0.3
    IMPROVE_LIMIT = 120_000
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # random source (biased towards larger entries)
        if np.random.rand() < 0.85:
            i = int(np.argmax(a))
        else:
            i = np.random.choice(n, p=a)

        if a[i] <= 1e-14:
            continue

        # random target ≠ source
        j = np.random.randint(n)
        if j == i:
            continue

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-13:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[RandHill {iteration:07d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.008, size=n)
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[RandHill {iteration:07d}] stagnation – perturbing")
        iteration += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function (vector of heights) under the
    supplied time budget.  Returns a list of floats whose sum is 1.
    The optimisation follows the three phases described above.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin for the host

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between expressivity and
    # speed on the typical hardware of the execution environment.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise `a` – reuse a known good construction if available,
    # otherwise start from a random Dirichlet vector.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
    except Exception:
        a = np.random.rand(n)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)

    best_a = a.copy()
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Phase 1 – L2 gradient descent (≈40 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.40 * budget_s
    a, best_a, best_val = _l2_gradient_phase(
        a, n, phase1_deadline, start_time, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Phase 2 – Targeted max‑reduction hill climbing (≈45 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.85 * budget_s
    best_a, best_val = _max_hillclimb_phase(
        best_a, n, phase2_deadline, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Phase 3 – Light random hill climbing to polish the result
    # ------------------------------------------------------------------
    best_a, best_val = _random_hillclimb_phase(
        best_a, n, deadline, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
