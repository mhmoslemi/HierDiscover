"""
Propose a step‑function (list of heights) that minimizes `evaluate_sequence`.

Algorithm
---------
1. **Initialization & cheap exploration**  
   * Uniform distribution (baseline).  
   * 2000 random Dirichlet samples.  
   * Systematic grid over a two‑parameter family  
     a_i ∝ (i+1)^α·(n−i)^β (α,β ∈ [−1.5,1.5]).  
   * Simple convex combinations of the best found shape with the uniform
     vector.

   All candidates are evaluated with a fast FFT‑based convolution and the
   best one (lowest value of 2·n·max(conv(a,a))) is kept.

2. **Refinement – exponentiated‑gradient descent on the exact objective**  
   For a current vector `a` (∑a = 1, a ≥ 0) we compute the convolution
   `b = a*a` and the index `k*` where `b` attains its maximum.
   The sub‑gradient of `max(b)` w.r.t. `a` is

       ∂ max(b) / ∂ a_i = 2·a_{k*‑i}   (if 0 ≤ k*‑i < n, else 0).

   We perform an *exponentiated‑gradient* (multiplicative) update

       a_i ← a_i·exp(‑η·grad_i) ,   then renormalise to sum 1,

   with a learning‑rate `η` that decays from 1.0 to 0.001.
   Only strictly improving moves are kept; after a long stagnation
   we perturb the current best solution with small Gaussian noise and
   continue.  The loop runs until the overall time budget (≈ 1000 s) is
   exhausted.

3. **Final evaluation** – the best vector obtained is handed to the
   provided `evaluate_sequence` and returned as a plain Python list.

The method stays entirely on the simplex, exploits the exact gradient
of the true L∞‑objective and, thanks to the cheap parametric search,
starts the descent from a promising non‑uniform shape.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Exact sub‑gradient of max(conv(a,a))
# ----------------------------------------------------------------------
def _max_grad(a: np.ndarray, k_star: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at index `k_star`.
    Returns a non‑negative vector of length n.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    j = k_star - np.arange(n)               # candidate source indices
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact L∞‑objective
# ----------------------------------------------------------------------
def _eg_descent(a_start: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on `max(conv(a,a))`.
    Keeps the current vector on the simplex and returns the best solution
    seen before `deadline`.
    """
    a = a_start.copy()
    cur_val = _objective(a)

    # schedule for learning‑rate η (1.0 → 0.001)
    eta_start, eta_end = 1.0, 0.001
    stagnation_limit = 200_000
    stagnation = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress‑based learning‑rate
        prog = (now - start) / (deadline - start)
        eta = eta_start * (1.0 - prog) + eta_end * prog

        # convolution and gradient based on the current maximum entry
        b = _full_convolution(a)
        k_star = int(np.argmax(b))
        grad = _max_grad(a, k_star)

        # multiplicative update (maintains positivity)
        a_new = a * np.exp(-eta * grad)
        # avoid exact zeros that could break renormalisation
        eps = 1e-16
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_val = _objective(a_new)

        # accept only strictly improving steps
        if new_val < cur_val - 1e-14:
            a = a_new
            cur_val = new_val
            stagnation = 0

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[EG {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    return best_a, best_val
        else:
            stagnation += 1

        # if we stall for a long time, perturb the best solution
        if stagnation > stagnation_limit:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            cur_val = _objective(a)
            stagnation = 0
            print(f"[EG {it:7d}] perturbing after stagnation")

        it += 1

    return best_a, best_val

    # Small convex mixtures with the uniform vector
    for lam in np.linspace(0.0, 1.0, 11):
        a = lam * uniform + (1.0 - lam) * best_a
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Mix λ={lam:.2f}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Refinement – exponentiated‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    best_a, best_val = _eg_descent(
        a_start=best_a,
        n=n,
        start=start_time,
        deadline=deadline,
        best_a=best_a,
        best_val=best_val,
        target_val=1.503,
    )

    # ------------------------------------------------------------------
    # 3️⃣  Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
