# Helper: full linear convolution via real FFT (used many times)

def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[init] best deterministic score = {best_score:.12f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – several random restarts until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        restart_counter += 1
        # decide a problem size for this restart (range ∼4000‑6000)
        n = int(rng.integers(4000, 6000))

        # --------------------------------------------------------------
        # Initialise a (probability vector) for this restart
        # --------------------------------------------------------------
        if "height_sequence_1" in globals():
            try:
                base = np.asarray(height_sequence_1, dtype=np.float64)
                if base.shape[0] == n:
                    a = np.maximum(base, 0.0)
                else:
                    a = rng.random(n)
            except Exception:
                a = rng.random(n)
        else:
            a = rng.random(n)

        if a.sum() == 0.0:
            a = rng.random(n)

        a = a / a.sum()                       # project onto simplex
        log_a = np.log(a + 1e-30)              # log‑variables for mirror descent

        # --------------------------------------------------------------
        # Hyper‑parameters for this run
        # --------------------------------------------------------------
        eta = 0.35              # step size for the log‑update
        eta_min = 5e-6
        eta_decay = 0.9995

        eps = None              # temperature for soft‑max approximation
        eps_decay = 0.9996

        stagn = 0               # iterations without improvement

        # allocate a modest inner time budget (≈20 s or the remaining fraction)
        inner_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.7)

        # --------------------------------------------------------------
        # 2️⃣  Mirror‑descent optimisation (soft‑max → hard‑max)
        # --------------------------------------------------------------
        while time.time() < inner_deadline:
            # current convolution & score
            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            score = 2.0 * n * max_conv

            # global best update
            if score < best_score - 1e-12:
                best_score = score
                best_seq = a.copy()
                stagn = 0
                print(
                    f"[{time.time() - (deadline - budget_s):.1f}s] "
                    f"new best {best_score:.12f} (restart {restart_counter}, n={n})"
                )
            else:
                stagn += 1

            # ----- temperature schedule -----
            if eps is None:
                eps = max(max_conv * 0.5, 1e-12)
            else:
                eps = max(eps * eps_decay, max_conv * 1e-6, 1e-12)

            # ----- construct weighting vector w -----
            if eps > max_conv * 1e-6:
                # smooth soft‑max
                w = np.exp((conv - max_conv) / eps)
                w_sum = w.sum()
                if w_sum > 0.0 and np.isfinite(w_sum):
                    w /= w_sum
                else:
                    w = np.full_like(conv, 1.0 / conv.size)
            else:
                # hard‑max mask (binary)
                tol = max_conv * 1e-9 + 1e-14
                mask = (conv >= max_conv - tol).astype(np.float64)
                w_sum = mask.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w = mask / w_sum

            # ----- gradient of weighted sum w.r.t. a -----
            rev_a = a[::-1]
            grad_full = _full_convolution_fft(rev_a, w)          # length 3n‑2
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]            # slice to length n

            # ----- mirror‑descent (KL) update in log‑space -----
            log_a = log_a - eta * grad
            # re‑normalise (softmax) to stay on the simplex
            max_log = np.max(log_a)
            exp_log = np.exp(log_a - max_log)
            a = exp_log / exp_log.sum()
            # keep log_a synchronised (numerically stable)
            log_a = np.log(a + 1e-30)

            # ----- learning‑rate decay -----
            eta = max(eta_min, eta * eta_decay)

            # ----- occasional jitter to escape plateaus -----
            if stagn > 3000:
                log_a += rng.normal(scale=0.015, size=n)
                max_log = np.max(log_a)
                exp_log = np.exp(log_a - max_log)
                a = exp_log / exp_log.sum()
                eta = max(eta_min, eta * 1.2)
                stagn = 0
                print(f"[{time.time() - (deadline - budget_s):.1f}s] jitter (restart {restart_counter})")

        # --------------------------------------------------------------
        # 3️⃣  Hard‑max polishing (few thousand tiny sub‑gradient steps)
        # --------------------------------------------------------------
        a = best_seq if best_seq is not None else a
        for _ in range(2500):
            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            tol = max_conv * 1e-9 + 1e-14
            mask = (conv >= max_conv - tol).astype(np.float64)
            w = mask / mask.sum()
            grad_full = _full_convolution_fft(a[::-1], w)
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]
            a = a - 4e-6 * grad               # tiny step
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = rng.random(n)
                a = a / a.sum()
            else:
                a = a / s

        # final polish assessment
        final_score = _obj(a)
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(
                f"[{time.time() - (deadline - budget_s):.1f}s] "
                f"post‑polish best {best_score:.12f} (restart {restart_counter})"
            )

    # ------------------------------------------------------------------
    # Return the best sequence found (clamped to the limits required by
    # evaluate_sequence).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: plain uniform distribution of length 5000
        n_fallback = 5000
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    final_seq = [
        float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()
    ]
    # One final call to evaluate_sequence gives the official score
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
