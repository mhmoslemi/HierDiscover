def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

Exponential‑gradient descent on a smooth approximation of the
ℓ∞‑norm of the self‑convolution.

The evaluation function for a normalized step‑function a (∑a=1) is

# ----------------------------------------------------------------------
# Helper: linear convolution a*a (full length 2·n‑1) via FFT
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, size: int) -> np.ndarray:
    """Linear self‑convolution of a, returned length 2·len(a)‑1."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    # eliminate tiny negative values caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv

# Helper: correlation w⋆a (valid part, length n) via FFT.

def _corr_valid(w: np.ndarray, a: np.ndarray, size: int) -> np.ndarray:
    """Correlation of w with a (valid mode) – length = len(a)."""

    start = a.shape[0] - 1
    return corr_full[start : start + a.shape[0]]

        return np.full(n, 1.0 / n)
    rho = ind[cond].max()
