def init_distribution(n, typ, rng):
    x = np.arange(n, dtype=np.float64)
    if typ == 'decreasing':
        # a_i proportional to 1/(i+1)
        a = 1.0 / (x + 1.0)
    elif typ == 'exponential':
        lam = rng.uniform(0.0005, 0.005)
        a = np.exp(-lam * x)
    elif typ == 'gaussian':
        sigma = n / 5.0
        center = n / 2.0
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif typ == 'random':
        a = rng.random(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.ones(n)
    a = a / a.sum()
    return a

#!/usr/bin/env python3
"""
Search for a step‑function sequence minimizing
2 * n * max(conv(a,a)) / (sum(a)**2).

The algorithm combines a cheap parametric grid search over
simple symmetric shapes with a stochastic sub‑gradient descent
on the simplex.  Multiple independent runs with different length
and initialisation are performed until the time budget expires.
The best sequence found is returned.
"""

import time
import numpy as np

def _project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution_fft(a, b):
    """Linear convolution via real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

def _score(a):
    """Return the objective value for a (which is assumed to sum to 1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())

def _initial_shape(n, shape_type, rng):
    """Create a deterministic shape of length n and normalise it."""
    x = np.arange(n, dtype=np.float64)
    if shape_type == 'uniform':
        a = np.ones(n, dtype=np.float64)
    elif shape_type == 'semicircle':
        t = 2.0 * x / (n - 1) - 1.0
        a = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
    elif shape_type == 'sine':
        a = np.sin(np.pi * (x + 0.5) / n)
    elif shape_type == 'gaussian':
        sigma = n / 6.0
        a = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
    elif shape_type == 'power':
        # a_i = (i * (n-1-i) + eps) ^ p   with p randomly drawn
        p = rng.uniform(0.1, 0.9)
        eps = 1e-12
        a = (x * (n - 1 - x) + eps) ** p
    elif shape_type == 'random':
        a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.ones(n, dtype=np.float64)
    a = a / a.sum()
    return a

def _subgradient_update(a, step, rng):
    """One projected sub‑gradient step for the max‑convolution objective."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_val = conv.max()
    # tolerance to select all (almost) active constraints
    tol = max(max_val * 1e-9, 1e-14)
    mask = (conv >= max_val - tol).astype(np.float64)
    rev_a = a[::-1]
    grad_full = _full_convolution_fft(rev_a, mask)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    # projected sub‑gradient step
    a_new = a - step * grad
    # enforce non‑negativity and renormalise
    a_new = _project_simplex(a_new)
    # optional tiny random jitter to escape plateaus
    if rng.random() < 0.001:
        jitter = rng.normal(scale=step * 0.5, size=n)
        a_new = a_new + jitter
        a_new = _project_simplex(a_new)
    return a_new, max_val

"""
Search for a step‑function sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear self‑convolution of a non‑negative vector a.
The routine works under the 1000 s budget, performs a cheap parametric
grid‑search over a few classic shapes, then runs many short stochastic
sub‑gradient descents on various lengths and initialisations.  The best
sequence seen is finally polished with a few Adam steps (essentially a
hard‑max optimisation) and returned in the format required by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full (linear) convolution using a real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Evaluate the objective for a vector that already sums to 1.
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Generate a deterministic “nice’’ shape of length n.
# ----------------------------------------------------------------------
def _initial_shape(n: int, shape_type: str, rng: np.random.Generator) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)

    if shape_type == "uniform":
        a = np.ones(n, dtype=np.float64)

    elif shape_type == "semicircle":
        # a_i ∝ sqrt(1 - t^2),  t ∈ [-1, 1]
        t = 2.0 * x / (n - 1) - 1.0
        a = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

    elif shape_type == "sine":
        a = np.sin(np.pi * (x + 0.5) / n)

    elif shape_type == "gaussian":
        sigma = n / 6.0
        a = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

    elif shape_type == "power":
        # a_i ∝ (i * (n-1-i) + eps) ** p, with p randomly chosen
        p = rng.uniform(0.1, 0.9)
        eps = 1e-12
        a = (x * (n - 1 - x) + eps) ** p

    elif shape_type == "random":
        a = rng.random(n)

    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.ones(n, dtype=np.float64)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# One projected sub‑gradient update for the max‑convolution objective.
# ----------------------------------------------------------------------
def _subgradient_update(a: np.ndarray, step: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a single projected sub‑gradient step.
    Returns the updated vector and the current max‑convolution value.
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_val = conv.max()
    # mask of (almost) active constraints
    tol = max(max_val * 1e-9, 1e-14)
    mask = (conv >= max_val - tol).astype(np.float64)

    # gradient of the weighted sum  (cross‑correlation with the mask)
    rev_a = a[::-1]
    grad_full = _full_convolution_fft(rev_a, mask)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    # projected sub‑gradient step
    a_new = a - step * grad
    a_new = _project_simplex(a_new)

    # enforce symmetry – the optimal solution is expected to be symmetric
    a_new = (a_new + a_new[::-1]) * 0.5

    # occasional tiny jitter to help escape plateaus
    if rng.random() < 0.001:
        jitter = rng.normal(scale=step * 0.5, size=n)
        a_new = a_new + jitter
        a_new = _project_simplex(a_new)

    return a_new, max_val
