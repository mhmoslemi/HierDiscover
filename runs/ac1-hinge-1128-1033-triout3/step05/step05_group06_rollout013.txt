def solve_lp(f, M, solver='ECOS', max_iters=5000):
    n = f.shape[0]
    b = cp.Variable(n, nonneg=True)
    conv = cp.conv(b, f)  # length 2n-1
    constraints = [conv <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver, warm_start=True, max_iters=max_iters)
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    b_opt = b.value
    return np.maximum(b_opt, 0.0)  # ensure nonneg due to numerical issues
```

"""...
Implements a hybrid search: uses deterministic seeds,
projected Adam on the log‑probabilities and a
simple linear‑program refinement (idea from the paper).
The LP step solves: max Σ b_i s.t. a * b ≤ M , b ≥ 0,
where M is the current maximal auto‑convolution.
The solution is mixed with the current a to strictly
decrease the objective.
"""

import time
import numpy as np

# Convolution helper
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def score_vector(a: np.ndarray) -> float:
    # assumes sum(a) == 1
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

def softmax(x: np.ndarray) -> np.ndarray:
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    G = np.zeros_like(a)
    # vectorised version: for each active k, we add contributions.
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad

def run_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    start = time.time()
    n = initial_a.shape[0]

    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.2

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    while time.time() - start < time_limit - 0.5:
        iteration += 1
        a = softmax(x)

        conv = full_convolution_fft(a)
        max_b = conv.max()

        tol = max_b * 1e-12 + 1e-15
        active = np.where(conv >= max_b - tol)[0]

        grad = grad_log_params(x, a, conv, max_b, active)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Adam iter {iteration:6d}: new best score = {best_score:.6f}")

        if iteration - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Adam iter {iteration:6d}: re‑heat (no improvement)")

    return best_vec, best_score

# Parametric families for deterministic seeds
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# Linear‑program refinement (uses cvxpy if available)
def lp_refine(a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """Perform a few iterations of the LP refinement described in the prompt.

    Returns the best vector seen and its score.
    """
    try:
        import cvxpy as cp
    except Exception:
        print("cvxpy not available – skipping LP refinement.")
        return a, score_vector(a)

    n = a.shape[0]
    best_vec = a.copy()
    best_score = score_vector(a)
    start = time.time()
    iteration = 0

    # we keep a small maximum number of LP solves to stay within time
    MAX_ITERS = 20

    while time.time() - start < time_limit - 1.0 and iteration < MAX_ITERS:
        iteration += 1
        conv = full_convolution_fft(a)
        M = conv.max()
        # Early stop if already close to average
        avg = conv.mean()
        if M <= avg * 1.02:
            break

        # Build and solve LP: max sum(b) s.t. a * b <= M, b >= 0
        b = cp.Variable(n, nonneg=True)
        # convolution a * b (order doesn't matter)
        conv_expr = cp.conv(b, a)   # length 2n-1
        constraints = [conv_expr <= M + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-8, reltol=1e-6)
        except Exception as e:
            # fallback to SCS if ECOS fails
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=3000, verbose=False)

        if b.value is None:
            print(f"LP iteration {iteration}: solver failed, stop.")
            break

        b_opt = np.array(b.value, dtype=float)
        b_opt = np.maximum(b_opt, 0.0)
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            break
        g = b_opt / sum_b

        # Choose mixing weight – larger at the beginning, smaller later
        t = 0.5 if iteration <= 5 else 0.2

        new_a = (1.0 - t) * a + t * g
        new_a = np.maximum(new_a, 0.0)
        new_a = new_a / new_a.sum()

        new_score = score_vector(new_a)
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_vec = new_a.copy()
            print(f"LP iter {iteration}: new best score = {best_score:.6f}")

        a = new_a

    return best_vec, best_score

"""
Hybrid optimisation for the step‑function attack.

The routine builds a probability vector (non‑negative, sum = 1) and tries
to minimise the peak of its auto‑convolution, i.e.  2·n·max((a*a)[k]).
It works in three phases:

1.  Initialise from either the external seed ``height_sequence_1`` or from
    a few deterministic families (u_min, u_prod, exponential, uniform).
2.  Run projected Adam on the log‑probabilities (the code from the
    previous attempt) for a fraction of the time budget.
3.  Refine the result with a simple linear‑program (LP) step
    described in the statement:

        maximise Σ b_i   subject to   a * b ≤ M ,   b ≥ 0

    where M = max(a*a) for the current a.  The optimiser returns a
    vector b that can be scaled to a new probability distribution g.
    Mixing a with g (a ← (1‑t)·a + t·g) strictly reduces the objective.
    The LP is solved with CVXPY (fallback to Adam if CVXPY is unavailable).

The final sequence (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, returns length 2·n‑1)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def score_vector(a: np.ndarray) -> float:
    """Objective value 2·n·max((a*a)[k]) for a probability vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Softmax (log‑parameterisation)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution w.r.t. log‑parameters
# ----------------------------------------------------------------------
def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    """∂/∂x ( max_k (a*a)[k] ) for a = softmax(x)."""
    n = a.shape[0]
    G = np.zeros_like(a)

    # contributions for each active convolution index
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]

    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad

# ----------------------------------------------------------------------
# Projected Adam on the log‑probabilities (same as the previous code)
# ----------------------------------------------------------------------

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.2

    best_vec = None
    last_improve = 0
    iteration = 0

    while time.time() - start < time_limit - 0.5:
        iteration += 1
        a = softmax(x)

        conv = full_convolution_fft(a)
        max_b = conv.max()

# Parametric families for deterministic seeds
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# Linear‑program refinement (idea from the statement)
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑refinement iterations:

        maximise Σ b_i   s.t.   a * b ≤ M ,   b ≥ 0

    where M = max(a*a).  The solution b is normalised to a probability
    vector g = b / Σb and mixed with a (a ← (1‑t)a + t·g).  Returns the
    best vector seen and its score.
    """
    try:
        import cvxpy as cp
    except Exception:
        print("cvxpy not available – skipping LP refinement.")
        return a, score_vector(a)

    n = a.shape[0]
    best_vec = a.copy()
    best_score = score_vector(a)
    start = time.time()
    iteration = 0
    MAX_ITERS = 20                     # hard limit on LP solves

    while time.time() - start < time_limit - 1.0 and iteration < MAX_ITERS:
        iteration += 1
        conv = full_convolution_fft(a)
        M = conv.max()
        avg = conv.mean()
        # stop early if max is already close to the average value
        if M <= avg * 1.02:
            break

        # LP: max Σ b_i  subject to  a * b ≤ M,  b ≥ 0
        b = cp.Variable(n, nonneg=True)
        conv_expr = cp.conv(b, a)               # length 2n‑1
        constraints = [conv_expr <= M + 1e-12]   # element‑wise
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=2000, abstol=1e-8, reltol=1e-6)
        except Exception:
            prob.solve(solver=cp.SCS, warm_start=True,
                       max_iters=3000, verbose=False)

        if b.value is None:
            print(f"LP iteration {iteration}: solver failed, aborting.")
            break

        b_opt = np.array(b.value, dtype=float)
        b_opt = np.maximum(b_opt, 0.0)
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            break

        g = b_opt / sum_b

        # blending weight – larger at the beginning, smaller later
        t = 0.5 if iteration <= 5 else 0.2

        new_a = (1.0 - t) * a + t * g
        new_a = np.maximum(new_a, 0.0)
        new_a = new_a / new_a.sum()

        new_score = score_vector(new_a)
        if new_score < best_score - 1e-9:
            best_score = new_score
            best_vec = new_a.copy()
            print(f"LP iter {iteration}: new best score = {best_score:.6f}")

        a = new_a

    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the time budget and returns the best
    sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # 5 s safety margin

    # --------------------------------------------------------------
    # 1.  Initialise a probability vector (sum = 1)
    # --------------------------------------------------------------
    initial_vec = None
    n = 5000                               # fallback length

    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        if external.sum() > 0.0:
            external = external / external.sum()
            initial_vec = external
            n = external.shape[0]
            print(f"Using external seed of length {n}")
    except NameError:
        pass

    if initial_vec is None:
        # deterministic sweep over a few families
        candidates = []
        for p in [0.5, 0.8, 1.0, 1.2]:
            candidates.append(u_min_distribution(n, p))
            candidates.append(u_prod_distribution(n, p))
        for lam in [0.002, 0.005, 0.01]:
            candidates.append(exp_distribution(n, lam))
        candidates.append(np.ones(n) / n)       # uniform

        best_score = np.inf
        for a in candidates:
            sc = score_vector(a)
            if sc < best_score:
                best_score = sc
                initial_vec = a.copy()
        print(f"Deterministic seed chosen (length {n}, initial score {best_score:.6f})")
    else:
        n = initial_vec.shape[0]

    # --------------------------------------------------------------
    # 2.  Adam optimisation for a fraction of the remaining budget
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 30.0:
        adam_budget = min(remaining * 0.30, 120.0)   # at most 2 min for Adam
        print(f"Running Adam for {adam_budget:.1f}s")
        vec, adam_score = run_adam(initial_vec, adam_budget)
        vec = np.maximum(vec, 0.0)
        if vec.sum() > 0.0:
            vec = vec / vec.sum()
        current_vec = vec
        best_vec = vec
        best_score = adam_score
        now = time.time()
        remaining = deadline - now
    else:
        current_vec = initial_vec.copy()
        best_vec = current_vec.copy()
        best_score = score_vector(current_vec)

    # --------------------------------------------------------------
    # 3.  LP refinement for the rest of the time (if possible)
    # --------------------------------------------------------------
    if remaining > 10.0:
        lp_budget = remaining
        print(f"Running LP refinement for up to {lp_budget:.1f}s")
        refined_vec, refined_score = lp_refine(current_vec, lp_budget)
        if refined_score < best_score:
            best_score = refined_score
            best_vec = refined_vec

    # --------------------------------------------------------------
    # 4.  Optional final Adam pass (uses any leftover time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        print(f"Final Adam pass for {remaining:.1f}s")
        vec, sc = run_adam(best_vec, remaining)
        vec = np.maximum(vec, 0.0)
        if vec.sum() > 0.0:
            vec = vec / vec.sum()
        final_score = score_vector(vec)
        if final_score < best_score:
            best_score = final_score
            best_vec = vec

    # --------------------------------------------------------------
    # 5.  Return the best vector
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_vec)})")
    best_vec = np.maximum(best_vec, 0.0)
    return best_vec.tolist()
