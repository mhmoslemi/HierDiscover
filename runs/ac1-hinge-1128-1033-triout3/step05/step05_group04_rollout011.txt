"""
propose_candidate

Hybrid optimizer for the step‑function self‑convolution problem.

Key ideas
---------
1. **Exponentiated‑gradient (EG) descent** on a smooth soft‑max
   approximation of `max(conv(a,a))`.  The EG update automatically
   respects the simplex constraints (non‑negative, sum‑to‑one) and
   often yields faster reduction of the peak of the convolution than
   the projected‑Adam method used previously.

2. **Multiple random restarts** – each run starts from a different
   reasonable shape (uniform, random Dirichlet, a mixture of smooth
   “nice’’ kernels, an arcsine‑type U‑shape or the last best global
   sequence).  The best sequence across all runs is retained.

3. **Final polishing** – after the restarts the current best sequence
   is refined with a short Adam‑style optimisation (the same machinery
   that proved effective in the reference implementation).

The whole routine respects the 1000 s time budget, continually checks
remaining time, and prints progress.  The returned list is already
clamped to the limits imposed by ``evaluate_sequence``.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# (used only in the Adam polishing stage)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent with soft‑max temperature.
# ----------------------------------------------------------------------
def _eg_step(a: np.ndarray,
             eta: float,
             eps: float) -> (np.ndarray, float, float):
    """
    One EG update step.
    Returns the new a, updated eta and eps (both decayed internally).
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()

    # soft‑max weights centred at the current maximum
    w = np.exp((conv - max_conv) / eps)
    w_sum = w.sum()
    if w_sum > 0 and np.isfinite(w_sum):
        w = w / w_sum
    else:
        w = np.full_like(conv, 1.0 / conv.size)

    # gradient of the smooth max (cross‑correlation)
    rev_a = a[::-1]
    grad_full = _full_convolution_fft(rev_a, w)
    grad = 2.0 * grad_full[n - 1: n - 1 + n]

    # EG update (stay on the simplex automatically)
    a = a * np.exp(-eta * grad)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / total
    return a, max_conv, None  # `None` placeholder for compatibility


# ----------------------------------------------------------------------
# Candidate generator – the entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    `2 * n * max(conv(a, a)) / (sum(a) ** 2)`.  Uses many short
    exponentiated‑gradient runs with random restarts followed by a final
    Adam polishing phase.
    """
    # ------------------------------------------------------------------
    # General set‑up.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    n = 5000                     # the length used in the reference run
    best_val = np.inf
    best_seq = None

    # --------------------------------------------------------------
    # Helper to compute the objective (without the clipping overhead).
    # ------------------------------------------------------------------
    def _objective(a_vec: np.ndarray) -> float:
        """Return 2 * n * max(conv(a,a)) assuming sum(a)=1."""
        conv = _full_convolution_fft(a_vec, a_vec)
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # Initialisation functions.
    # --------------------------------------------------------------
    def _init_uniform() -> np.ndarray:
        return np.full(n, 1.0 / n, dtype=np.float64)

    def _init_random() -> np.ndarray:
        a = rng.random(n)
        return a / a.sum()

    def _init_mixed() -> np.ndarray:
        """Mixture of several smooth shapes (similar to the reference code)."""
        x = np.arange(n, dtype=np.float64)

        # Catalan‑like decreasing sequence
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)
            catalan[k] = cur

        sine = np.sin(np.pi * (x + 0.5) / n)
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
        triangle = np.minimum(x, n - 1 - x) + 1.0

        # Discrete arcsine‑type (U‑shaped) kernel
        arcs = 1.0 / np.sqrt((x + 0.5) * (n - 0.5 - x))

        bases = [catalan, sine, semicircle, gauss, triangle, arcs]
        ws = rng.random(len(bases))
        ws /= ws.sum()
        a = sum(w * b for w, b in zip(ws, bases))
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        return a

    # Include the previously best known sequence if it exists.
    init_fns = [_init_uniform, _init_random, _init_mixed]
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.ndim == 1 and prev.size == n:
                prev = np.clip(prev, 0.0, None)
                sum_prev = prev.sum()
                if sum_prev > 0:
                    prev /= sum_prev
                else:
                    prev = _init_random()
                init_fns.append(lambda: prev.copy())
        except Exception:
            pass  # fall back to the other inits

    # --------------------------------------------------------------
    # Main search loop – many short EG runs with random restarts.
    # --------------------------------------------------------------
    run_counter = 0
    while time.time() < deadline - 30.0:      # reserve ~30 s for final polishing
        run_counter += 1
        init = rng.choice(init_fns)
        a = init()
        # quick check of the starting value
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[run {run_counter}] init best_val={best_val:.12f}")

        # EG hyper‑parameters (tuned for fast convergence on this scale)
        eta = 0.6                     # initial learning‑rate
        eta_min = 1e-5
        eta_decay = 0.9997

        # temperature for the soft‑max approximation of the max
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        eps = max(max_conv * 0.5, 1e-12)
        eps_min = max(max_conv * 1e-6, 1e-12)
        eps_decay = 0.9997

        stagn = 0
        max_iters = 5000
        for it in range(max_iters):
            # evaluate current objective
            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            cur_val = 2.0 * n * max_conv
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = a.copy()
                stagn = 0
                print(f"[run {run_counter} it {it:04d}] new_best={best_val:.12f}")
            else:
                stagn += 1

            # stop early if both schedules are already tiny
            if eta < eta_min and eps < eps_min:
                break

            # soft‑max weighting centred at the current maximum
            w = np.exp((conv - max_conv) / eps)
            w_sum = w.sum()
            if w_sum > 0 and np.isfinite(w_sum):
                w = w / w_sum
            else:
                w = np.full_like(conv, 1.0 / conv.size)

            # gradient (cross‑correlation)
            rev_a = a[::-1]
            grad_full = _full_convolution_fft(rev_a, w)
            grad = 2.0 * grad_full[n - 1: n - 1 + n]

            # exponentiated‑gradient update
            a = a * np.exp(-eta * grad)
            tot = a.sum()
            if tot == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / tot

            # decay schedules
            eta = max(eta_min, eta * eta_decay)
            eps = max(eps_min, eps * eps_decay)

            # occasional random jitter to escape plateaus
            if stagn > 2500:
                a = a + rng.normal(scale=0.005, size=n)
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                stagn = 0
                eta = max(eta_min, eta * 1.2)
                print(f"[run {run_counter} jitter it {it}]")

    # ------------------------------------------------------------------
    # Final polishing – a short Adam optimisation (same as the reference).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Adam hyper‑parameters
    lr = 0.05
    lr_min = 1e-6
    lr_decay = 0.9999
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # initialise temperature for the soft‑max
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    eps = max(max_conv * 0.5, 1e-12)
    eps_min = max(max_conv * 1e-6, 1e-12)
    eps_decay = 0.9999

    stagn = 0
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-13:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[polish {it}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # soft‑max weights
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0 and np.isfinite(w_sum):
            w = w / w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # gradient
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected
        beta1_t = beta1 ** it
        beta2_t = beta2 ** it
        m_hat = m / (1.0 - beta1_t)
        v_hat = v / (1.0 - beta2_t)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a + step
        a = _project_simplex(a)

        # schedules
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional jitter if stuck
        if stagn > 5000:
            a = a + rng.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            stagn = 0
            lr = max(lr_min, lr * 1.2)
            print(f"[polish jitter {it}]")

    # ------------------------------------------------------------------
    # Convert to the format required by `evaluate_sequence` and report.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
