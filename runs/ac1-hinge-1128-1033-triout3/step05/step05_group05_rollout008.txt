"""
Propose a step‑function that minimises `evaluate_sequence`.

Algorithm
---------

We minimise the quantity 2·n·max(a∗a) (the true score when Σa=1) by a
two‑phase optimisation.

1. **Smooth‑max gradient descent**
   •  The objective `max(b)` (with b = a∗a) is replaced by the smooth
      approximation
          Sβ(b) = log‑sum‑exp(β·b) / β .
      Its gradient w.r.t. a is
          ∇Sβ = 2·(p ⋆ a) ,   p_i = exp(β·b_i)/Σexp(β·b)  (soft‑max weights)
      where “⋆” is the cross‑correlation.
   •  Each iteration performs two FFT‑based convolutions:
        – b = a∗a
        – g = p ⋆ a  (implemented as a convolution with the reversed a)
   •  An Adam‑style adaptive step is taken and the trial vector is
      projected onto the probability simplex (non‑negative, Σ=1).
   •  The inverse temperature β is increased exponentially from
      β₀=1e2 to β₁=2·10⁵, so the optimisation first flattens the whole
      convolution and later focuses on the highest peak.

2. **Deterministic hill‑climbing**
   •  Starting from the best point found in phase‑1 we perform cheap
      mass‑transfer moves: a small amount δ is moved from a source index i
      to a target j.  The effect on the full convolution is updated in
      O(n) time using the bilinearity of the convolution.
   •  Only moves that strictly improve the true score
          score = 2·n·max(b) / (Σa)²  (Σa stays ≈1) are accepted.
   •  After a long period without improvement the algorithm perturbs
      the current solution with small Gaussian noise and resumes the
      search.

The whole procedure respects the 1000 s wall‑clock budget.  The final
list of heights is returned; it can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` and `y` using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x≥0, Σx=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w

            continue

        # pick a distinct target
        j = np.random.randint(n)
        if j == i:
            continue

        # amount to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        # cross‑terms 2·δ·a (added at target, removed from source)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j]   += delta * delta
        b_new[2 * i]   += delta * delta
        b_new[i + j]   -= 2.0 * delta * delta

        new_score = 2.0 * n * b_new.max() / (a.sum() ** 2)   # a.sum() stays ≈1

        if new_score < cur_score:          # improvement?
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[Hill {it:8d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        if no_improve > stagnation_limit:
            # escape stagnation
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_score = 2.0 * n * b.max() / (a.sum() ** 2)
            no_improve = 0
            print(f"[Hill {it:8d}] stagnation – perturbing")

        it += 1

    return best_a

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function under a fixed time budget.
    Returns a list of floats ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the wrapper

    # --------------------------------------------------------------
    # Problem size – 8000 provides a good trade‑off between expressivity
    # and speed (FFT size stays ≤ 32768).
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise a – try to reuse a known good construction, otherwise
    # start from a random Dirichlet vector.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # Interpolate to the desired length
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)        # enforce Σa = 1, a_i ≥ 0
    best_a = a.copy()
    best_score = float('inf')

    # --------------------------------------------------------------
    # Phase C – smooth‑max gradient descent (≈55 % of the total budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.55 * budget_s
    it_c = 0
    no_improve_c = 0
    IMPROVE_STAG_C = 120_000

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)

    # β schedule (smooth‑max temperature)
    beta_min = 1e2
    beta_max = 2e5

    # Step‑size schedule (Adam learning‑rate)
    lr_start = 0.15
    lr_end   = 0.001

    while time.time() < phase_c_deadline:
        # ----- compute convolution and current score -----
        b = _full_convolution(a)                     # length 2·n‑1
        max_b = b.max()
        cur_score = 2.0 * n * max_b / (a.sum() ** 2)  # Σa≈1 → simple

        # track best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve_c = 0
            print(f"[Grad {it_c:7d}] new best = {best_score:.6f}")
        else:
            no_improve_c += 1

        # ----- smooth‑max weights -----
        # β grows exponentially with the fraction of elapsed time in this phase
        elapsed = time.time() - start_time
        phase_elapsed = phase_c_deadline - start_time
        progress = min(1.0, elapsed / phase_elapsed)
        beta = beta_min * (beta_max / beta_min) ** progress

        # soft‑max over b (shift by max for numerical stability)
        shifted = b - max_b           # ≤0, max entry becomes 0
        w = np.exp(beta * shifted)   # w_i ∈ (0,1]
        p = w / w.sum()              # probabilities, sum to 1

        # ----- gradient of smooth‑max w.r.t a -----
        a_rev = a[::-1]
        conv_corr = _conv_fft(p, a_rev)                 # length 3·n‑2
        grad = 2.0 * conv_corr[n - 1:2 * n - 1]          # slice of size n

        # ----- Adam update -----
        it_c += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it_c)
        v_hat = v / (1 - beta2 ** it_c)

        # linear decay of the base learning‑rate
        lr = lr_start * (1.0 - progress) + lr_end * progress
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a = _project_simplex(a_new)

        # occasional random jitter to avoid shallow traps
        if np.random.rand() < 0.002:
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)

        # restart if stuck for a long time
        if no_improve_c > IMPROVE_STAG_C:
            a = np.random.rand(n)
            a = _project_simplex(a)
            m.fill(0)
            v.fill(0)
            no_improve_c = 0
            print(f"[Grad {it_c}] restart after stagnation")

    # --------------------------------------------------------------
    # Phase D – deterministic hill‑climbing (uses the exact objective)
    # --------------------------------------------------------------
    best_a = _hillclimb_phase(best_a, n, deadline, best_a, best_score)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                              # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
