    best_local_a = best_a.copy()

    T_start = cur_val * 0.01   # 1% of current value maybe 0.01
    T_end = 1e-7

    IMPROVE_LIMIT = 400000

        T = T_start * (T_end / T_start) ** progress
        max_delta = max_delta_initial * (1 - progress) + max_delta_final * progress

        k_max = int(np.argmax(b))

#!/usr/bin/env python
"""
A two‑phase optimiser for the step‑function problem.

Phase A – exponentiated‑gradient on a smooth (log‑sum‑exp) approximation
         of the L∞ norm of the self‑convolution.  The vector is kept on
         the probability simplex, the temperature and learning‑rate are
         annealed, and occasional restarts avoid long stalls.

Phase B – simulated‑annealing hill‑climber that moves a small fraction of
         mass from a “source’’ index (chosen proportionally to its
         contribution to the current maximal convolution entry) to a
         uniformly sampled “target’’ index.  Moves that worsen the exact
         objective are accepted with probability exp(−Δ/T) where the
         temperature T is gradually reduced.  After a long period without
         acceptance the current best point is perturbed and the search
         continues.

Both phases share fast FFT‑based convolution helpers and keep track of
the best sequence seen.  The routine respects the 1000 s budget and
returns a plain list of non‑negative heights ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assuming the vector sums to 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase A – exponentiated‐gradient on a smooth max approximation
# ----------------------------------------------------------------------
def _eg_smooth_phase(a0: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent with KL divergence) on the
    smooth maximum `max_eps(b) = eps * log Σ exp(b/eps)`.  Returns the
    best (a, value) found before *deadline*.
    """
    a = a0.copy()
    best_local_a = best_a.copy()
    best_local_val = best_val

    # schedule parameters
    eps_start, eps_end = 0.1, 1e-9
    lr_start, lr_end = 1.0, 0.01
    stagnation_limit = 60_000
    no_improve = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # exact objective (for bookkeeping)
        cur_val = _objective(a)
        if cur_val < best_local_val:
            best_local_val, best_local_a = cur_val, a.copy()
            no_improve = 0
            print(f"[EG   ] new best = {best_local_val:.6f}")
            if best_local_val <= target_val:
                break
        else:
            no_improve += 1

        # annealing schedule
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)                 # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        # protect against overflow
        shifted = np.clip(shifted, -50, 50)
        weights = np.exp(shifted)
        p = weights / weights.sum()                    # soft distribution over k

        # gradient of the smooth max (scaled by the factor 2·n)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                 # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]           # the n entries we need
        grad = 4.0 * n * segment[::-1]                # ∇(2·n·smooth_max)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-lr * grad)
        total = a.sum()
        if total == 0.0 or not np.isfinite(total):
            # fallback to uniform if we collapsed to zero
            a = np.ones_like(a) / n
        else:
            a /= total

        # occasional random jitter to avoid exact symmetry
        if np.random.rand() < 0.01:
            a += np.random.normal(scale=1e-3, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # restart if we have been idle for a long time
        if no_improve > stagnation_limit:
            print(f"[EG   ] restart after {no_improve} idle steps")
            a = np.random.rand(n)
            a /= a.sum()
            no_improve = 0

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
# Phase B – simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _annealed_hillclimb(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """
    Local search that transfers a small amount of mass between coordinates.
    Moves that increase the exact objective are accepted with probability
    exp(−Δ/T); the temperature T cools down linearly in log‑time.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    best_local_a = best_a.copy()
    best_local_val = best_val

    # temperature schedule (start a small fraction of the current objective)
    T_start = cur_val * 0.02
    T_end = 1e-7
    # mass‑transfer schedule
    max_frac_start, max_frac_end = 0.5, 0.02

    stagnation_counter = 0
    stagnation_limit = 300_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress‑based parameters
        progress = (now - start) / (deadline - start)
        T = T_start * (T_end / T_start) ** progress
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to b[k_max] (including diagonal term)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index: biased towards large contributions
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index: uniform, distinct from source
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms due to the moved mass
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        delta_val = new_val - cur_val

        # Metropolis acceptance
        accept = False
        if delta_val < -1e-14:
            accept = True
        else:
            prob = np.exp(-delta_val / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            stagnation_counter = 0

            if new_val < best_local_val - 1e-14:
                best_local_val, best_local_a = new_val, a.copy()
                print(f"[HillA] new best = {best_local_val:.6f} (iter {iteration})")
                if best_local_val <= target_val:
                    # early exit – reached the target
                    break
        else:
            stagnation_counter += 1

        # perturb after long stagnation
        if stagnation_counter > stagnation_limit:
            print(f"[HillA] perturb after {stagnation_counter} rejects (iter {iteration})")
            a = best_local_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            stagnation_counter = 0

        iteration += 1

    return best_local_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a finer grid for flattening the
    # self‑convolution, but we must keep the per‑iteration cost reasonable.
    # ------------------------------------------------------------------
    n = 10000  # feel free to adjust; 10k ≈ 250 KB for the convolution

    # ------------------------------------------------------------------
    # Initialise a – arcsine‑like distribution (heavy at the ends, low in the centre)
    # which is known to give a relatively flat self‑convolution.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a0 = np.clip(a0, 1e-12, None)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = _objective(best_a)

    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – exponentiated‑gradient on the smoothed max (≈ 20 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.20 * (budget_s - 0.5)
    best_a, best_val = _eg_smooth_phase(a0, n,
                                        start=start_time,
                                        deadline=phase_a_deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase B – simulated‑annealing hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    best_a = _annealed_hillclimb(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=TARGET)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
