def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np

    deadline = time.time() + budget_s - 5

    rng = np.random.default_rng(seed)

    # Choose n
    n = 20000   # maybe adjust if time short

    # arcsine shape initial
    i = np.arange(n, dtype=np.float64) + 0.5
    # Avoid zeros
    a = 1.0 / np.sqrt(i * (n - i))
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2 * n * max_conv
    print(f"[init] n={n}, score={best_val}")

    # Parameters
    lr = 0.5   # step size
    decay = 0.9999
    min_lr = 1e-5
    epsilon = max(max_conv * 0.5, 1e-6) # temperature like for softmax weighting

    iter = 0
    stagn = 0

    while time.time() < deadline:
        iter += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2 * n * max_conv
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {iter}] new best {best_val:.12f}")
        else:
            stagn += 1

        # Softmax weights
        eps = max(epsilon * (0.99 ** iter), 1e-12)   # decaying
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # gradient
        rev = a[::-1]
        grad_full = _full_convolution_fft(rev, w)
        grad = 2.0 * grad_full[n-1:n-1+n]

        # Adam? we can use simple momentum
        # Let's use simple gradient descent with adaptive lr = lr / (1+0.0001*iter)
        step = - lr * grad
        a_new = a + step
        a_new = _project_simplex(a_new)
        a = a_new

        # decay lr
        lr = max(min_lr, lr * decay)

        if stagn > 5000:
            # random perturb to escape
            a = a + rng.normal(scale=0.01, size=n)
            a = _project_simplex(a)
            stagn = 0

    # After time, final polishing: subgradient steps on hard max
    a = best_seq.copy()
    for _ in range(5000):
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        # mask of approx max entries
        mask = (conv >= max_conv - 1e-12).astype(np.float64)
        w = mask / mask.sum()
        rev = a[::-1]
        grad = 2.0 * _full_convolution_fft(rev, w)[n-1:n-1+n]
        a = a - 5e-6 * grad
        a = _project_simplex(a)

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score}, length={len(final_seq)}")
    return final_seq

Propose a step‑function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

The idea is to start from a *continuous arcsine* density,
which in the limit yields an (almost) constant self‑convolution.
We initialise a mixture of this shape with the uniform distribution,
pick the best mixture by a tiny grid‑search, and then refine it with
a two‑phase optimiser:

1️⃣  Soft‑max Adam – minimise a smooth approximation of the max
    (log‑sum‑exp) while staying on the probability simplex.
2️⃣  Hard‑max polishing – switch to the true sub‑gradient of the
    max‑convolution and run a long projected‑gradient descent
    with occasional jitter.

The optimiser respects the supplied time budget (default 1000 s)
and keeps the best sequence seen so far.  The final sequence is
clamped to the limits required by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv
