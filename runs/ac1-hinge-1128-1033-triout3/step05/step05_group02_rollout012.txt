"""
Multi‑restart smoothed L‑BFGS + diverse permutation heuristics +
targeted sub‑gradient descent + final hill‑climb.

The algorithm works under a 1000 s budget and returns the best
non‑negative step‑function (list of floats) found.  Relative to the
previous baseline it adds:

* a richer set of re‑ordering tricks (alternating extremes,
  bit‑reversal, several random shuffles);
* a cheap sub‑gradient refinement that directly attacks the
  non‑smooth max‑convolution;
* more aggressive restarts (new starts are drawn both from a plain
  Gaussian and from noisy log‑samples of the current best);
* a slightly tighter ε‑schedule for the smooth surrogate.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global parameters for the L‑BFGS wrapper (required to avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a real vector with itself.
    Returns an array of length 2·len(x)‑1 using FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative values can appear due to rounding – clamp to 0
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (FFT based)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smooth surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a = softmax(w)  →  a_i ≥ 0, Σa = 1
    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k–max)/ε)

    Returns (L_ε, ∂L_ε/∂w) as required by scipy.minimize.
    """
    # ---- soft‑max (guarantees a ≥ 0, Σa = 1) ----
    w_shift = w - np.max(w)          # improve numeric stability
    a = np.exp(w_shift)
    a = a / a.sum()

    # ---- auto‑convolution a * a ----
    conv = _full_convolution(a)      # length 2·n‑1
    max_conv = conv.max()

    # ---- smooth max via log‑sum‑exp ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)     # log Σ_k exp((conv_k‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed form) ----
    weights = np.exp(shifted)        # ∝ exp((conv‑max)/ε)
    p = weights / weights.sum()      # probability vector over k

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    # compute efficiently via convolution with reversed p
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)   # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]  # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Thin wrapper needed for scipy.minimize (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Large values are placed on the left side, small values on the right,
    i.e. (a_max, …, a_min) ↔ (a_min, …, a_max).
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]   # large → left
        new_a[n - 1 - i] = sorted_vals[i]   # small → right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternating_extremes_reorder(a: np.ndarray) -> np.ndarray:
    """
    Interleaves the largest and smallest remaining values:
    [max, min, second‑max, second‑min, …].
    """
    s = np.sort(a)               # ascending
    lo, hi = 0, s.shape[0] - 1
    out = np.empty_like(a)
    for i in range(s.shape[0]):
        if i % 2 == 0:
            out[i] = s[hi]
            hi -= 1
        else:
            out[i] = s[lo]
            lo += 1
    return out


def _bitreversal_reorder(a: np.ndarray) -> np.ndarray:
    """
    Permute a according to the bit‑reversal ordering
    (as used in radix‑2 FFT).  This tends to spread large
    values across the whole index range.
    """
    n = a.shape[0]
    # work with the next power of two and then truncate
    m = 1 << (n - 1).bit_length()
    bits = int(np.log2(m))
    # bit‑reversal of 0 … m‑1
    rev = np.arange(m, dtype=np.int64)
    for b in range(bits):
        rev = ((rev & 1) << (bits - 1)) | (rev >> 1)
    # keep only the first n positions – they form a permutation of 0…n‑1
    perm = rev[:n]
    # map the sorted values onto this permutation
    sorted_vals = np.sort(a)
    out = np.empty_like(a)
    out[perm] = sorted_vals
    return out


def _subgradient_refine(a: np.ndarray, max_iters: int,
                        init_eta: float, deadline: float):
    """
    Simple sub‑gradient descent on the true objective
    (max(conv) with Σa = 1).  The step size is reduced when
    no improvement is observed.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a).max()

    eta = init_eta
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a)
        k = int(conv.argmax())                     # index of the peak
        # gradient of max(conv) w.r.t. a (sub‑gradient)
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k - idx]

        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:               # should not happen, but guard against it
            break
        a_new /= s                 # re‑normalize Σa = 1

        new_val = 2.0 * n * _full_convolution(a_new).max()
        if new_val < best_val - 1e-14:
            best_val = new_val
            best_a = a_new.copy()

        if new_val < 2.0 * n * conv.max():          # accepted improvement
            a = a_new
            # keep step size (or slightly increase)
            eta = min(eta * 1.05, init_eta)
        else:
            # no improvement → shrink step size
            eta *= 0.5
            if eta < 1e-8:
                break
    return best_a, best_val


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random mass‑transfer + random swaps, keeping Σa = 1.
    Accept only strict improvements (fast).
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # 15 % chance of a pure swap, 85 % chance of a mass transfer
        if rng.random() < 0.15:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            # Move up to 7 % of donor’s mass (a little larger than before)
            delta = cur_a[i] * 0.07 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # quick objective without the heavy evaluate_sequence call
        conv = _full_convolution(new_a)
        val = 2.0 * n * conv.max()
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            # occasional progress report
            print(f"[refine] improved to {cur_val:.6f}")

    return cur_a, cur_val

            # Convert to the actual sequence (soft‑max)
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()

            cur_val = 2.0 * n * _full_convolution(a).max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best ≈ {best_val:.6f}")

        # ------------------------------------------------------------------
        # Apply a suite of inexpensive re‑ordering heuristics
        # ------------------------------------------------------------------
        for reorder_fun, name in [
            (_anti_symmetric_reorder, "anti"),
            (_alternating_extremes_reorder, "alternating"),
            (_bitreversal_reorder, "bitrev")
        ]:
            a_ord = reorder_fun(best_a)
            val_ord = 2.0 * n * _full_convolution(a_ord).max()
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[reorder {name}] improved to {best_val:.6f}")

        # a few random shuffles (cheap)
        for _ in range(2):
            a_rand = np.random.permutation(best_a)
            val_rand = 2.0 * n * _full_convolution(a_rand).max()
            if val_rand < best_val:
                best_val = val_rand
                best_a = a_rand.copy()
                print(f"[reorder rand] improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # Sub‑gradient refinement (fast, directly attacks max(conv))
        # ------------------------------------------------------------------
        if time.time() < deadline - 20.0:
            a_sub, val_sub = _subgradient_refine(best_a,
                                                 max_iters=250,
                                                 init_eta=0.25,
                                                 deadline=deadline - 10.0)
            if val_sub < best_val:
                best_val = val_sub
                best_a = a_sub.copy()
                print(f"[subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climb refinement (mass transfers + swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val,
                                              deadline, n)

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
