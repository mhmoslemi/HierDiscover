def simplex_projection(v):
    # v may be any real numbers
    # returns w in simplex
    # using algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()

"""
Multi‑restart subgradient + targeted hill‑climbing optimizer.

The algorithm searches for a non‑negative step‑function (list of heights)
that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

Because the objective is scale‑invariant we keep Σa = 1.
For a given length `n` the plan is:

1. Generate several diverse initial vectors on the simplex
   (uniform, Gaussian, triangular, cosine, Dirichlet‑α, spectral, …).
2. For each start run a *subgradient descent* on the exact objective.
   The subgradient of `max(conv(a,a))` is given by the contributions of
   the current maximal convolution entry – a cheap O(n) operation.
   After a gradient step we project back onto the probability simplex.
   Simple step‑size schedule `η₀/√t` and occasional random perturbations
   avoid stagnation.
3. Refine the best point with a *targeted hill‑climber* very similar to
   the one used in the previous state‑of‑the‑art.  It transfers a small
   amount of mass `δ` from a source index (chosen proportionally to its
   contribution to the maximal convolution entry) to a uniformly chosen
   target index, updating the convolution in O(n).  The transfer fraction
   is linearly annealed from ~0.4 down to 0.001.
4. Keep the overall best sequence seen across all restarts and finally
   return it as a plain Python list.

The whole routine respects the given time budget (default 1000 s) and
prints intermediate best values for monitoring.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution of `a` with itself (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum x = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Initialisation helpers – a handful of diverse shapes
# ------------------------------------------------------------
def generate_initial(n: int) -> np.ndarray:
    """Return a random non‑negative vector of length n, Σa = 1."""
    types = ['uniform', 'gaussian', 'triangular', 'cosine', 'dirichlet',
             'spectral']
    typ = np.random.choice(types)

    if typ == 'uniform':
        a = np.ones(n, dtype=float)

    elif typ == 'gaussian':
        x = np.arange(n, dtype=float)
        mu = n / 2.0
        sigma = n / 6.0
        a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)

    elif typ == 'triangular':
        # symmetric triangular bump, peaks at centre
        if n % 2 == 0:
            half = n // 2
            inc = np.arange(1, half + 1)
            a = np.concatenate([inc, inc[::-1]])
        else:
            half = n // 2
            inc = np.arange(1, half + 2)
            a = np.concatenate([inc, inc[-2::-1]])

    elif typ == 'cosine':
        x = np.arange(n, dtype=float)
        a = 1.0 + np.cos(2.0 * np.pi * x / (n - 1))

    elif typ == 'dirichlet':
        # Dirichlet with concentration α < 1 yields spiky draws
        alpha = np.random.uniform(0.2, 0.8)
        a = np.random.gamma(alpha, 1.0, size=n)

    elif typ == 'spectral':
        # Random phases → real-valued sequence → take absolute value
        # to enforce non‑negativity.
        phases = np.exp(2j * np.pi * np.random.rand(n))
        y = np.fft.ifft(phases).real
        a = np.abs(y)

    else:
        a = np.random.rand(n)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ------------------------------------------------------------
# Subgradient descent on the exact max‑convolution objective
# ------------------------------------------------------------
def subgradient_descent(a0: np.ndarray,
                        n: int,
                        start_time: float,
                        deadline: float,
                        max_iter: int = 20000,
                        eta0: float = 0.5,
                        improve_limit: int = 1500) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on `2·n·max(conv(a,a))`.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    # initial objective
    b = full_convolution(a)
    best_val = 2.0 * n * b.max()

    no_improve = 0
    iter_cnt = 0

    while True:
        now = time.time()
        if now >= deadline or iter_cnt >= max_iter:
            break

        # ---- evaluate current point ----
        b = full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[SubGrad {iter_cnt:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---- sub‑gradient of max convolution ----
        k_max = int(np.argmax(b))
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        g = np.zeros_like(a)
        idx = np.arange(i_min, i_max + 1)
        g[idx] = 4.0 * n * a[k_max - idx]   # derivative of 2·n·max_b

        # ---- step size (sqrt‑decay) ----
        eta = eta0 / np.sqrt(iter_cnt + 1)

        # ---- projected update ----
        a_new = a - eta * g
        a = simplex_projection(a_new)

        # ---- occasional perturbation to escape flat regions ----
        if no_improve >= improve_limit:
            # perturb around the best seen so far
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            no_improve = 0
            print(f"[SubGrad {iter_cnt:7d}] perturbing after stagnation")

        iter_cnt += 1

    return best_a, best_val


# ------------------------------------------------------------
# Targeted hill‑climbing (exact objective)
# ------------------------------------------------------------
def hillclimb_targeted(a_start: np.ndarray,
                       n: int,
                       start_time: float,
                       deadline: float,
                       best_a_global: np.ndarray,
                       best_val_global: float,
                       target_val: float = 1.503) -> np.ndarray:
    """
    Fine‑grained local search that moves a small amount of mass from a
    “source” index to a “target” index, updating the convolution in O(n).
    The source is biased by its contribution to the current maximal entry.
    """
    a = a_start.copy()
    b = full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # annealing schedule for the transferred fraction
    max_frac_start, max_frac_end = 0.40, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0
    best_a = best_a_global.copy()
    best_val = best_val_global

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress fraction for schedule
        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the current maximum of the convolution
        k_max = int(np.argmax(b))

        # contributions of each a_i to b[k_max] (a_i * a_{k_max-i})
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – biased toward large contributions
        if total_contrib > 0 and np.random.rand() < 0.9:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            # fallback – take a random index with positive mass
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size == 0:
                i = np.random.randint(n)
            else:
                i = np.random.choice(nonzero)

        if a[i] <= 1e-14:
            continue

        # target index – uniform, forced to differ from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of convolution (O(n)) ----
        b_new = b.copy()
        # linear terms 2·δ·a added/removed
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept only strict improvements
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[HillClimb {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    # target reached – exit early
                    return best_a
        else:
            no_improve += 1

        # stagnation handling – re‑perturb around the best known point
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillClimb {iteration:7d}] perturbing after stagnation")

        iteration += 1

    return best_a


# ------------------------------------------------------------
# Main entry point for the evaluation framework
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimize a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that (approximately)
    minimizes `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a short margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem dimension – a larger n gives more freedom while FFT size
    # stays at 2^15 = 32768 for n up to ≈12000, which is cheap.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Global best across all restarts
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)   # start with uniform as a safe fallback
    best_val = float('inf')
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Helper: try to reuse the previously discovered construction if it
    # exists in the global namespace (height_sequence_1).
    # ------------------------------------------------------------------
    init_candidates = []
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.size != n:
            old = np.interp(np.linspace(0.0, 1.0, n),
                            np.linspace(0.0, 1.0, old.size), old)
        old = np.maximum(old, 0.0)
        if old.sum() < 1e-12:
            old = np.full(n, 1.0 / n)
        else:
            old /= old.sum()
        init_candidates.append(old)
    except Exception:
        pass

    # Always add a few random seeds
    for _ in range(4):
        init_candidates.append(generate_initial(n))

    # ------------------------------------------------------------------
    # Main optimisation loop – keep launching restarts while time permits.
    # ------------------------------------------------------------------
    restart_idx = 0
    while time.time() < deadline - 20.0:   # leave ~20 s for a final polish
        a0 = init_candidates[restart_idx % len(init_candidates)].copy()
        restart_idx += 1

        # ---- Phase A: projected sub‑gradient descent (exact max) ----
        sub_deadline = min(deadline, time.time() + 0.3 * (deadline - start_time))
        a_sgd, val_sgd = subgradient_descent(a0, n,
                                             start_time=start_time,
                                             deadline=sub_deadline,
                                             max_iter=20000,
                                             eta0=0.8,
                                             improve_limit=1200)

        # ---- Phase B: targeted hill‑climbing refinement ----
        a_hc = hillclimb_targeted(a_sgd, n,
                                  start_time=start_time,
                                  deadline=deadline,
                                  best_a_global=best_a,
                                  best_val_global=best_val,
                                  target_val=TARGET)

        # evaluate the candidate we just produced
        b_final = full_convolution(a_hc)
        val_final = 2.0 * n * b_final.max()
        if val_final < best_val:
            best_val = val_final
            best_a = a_hc.copy()
            print(f"[Overall] new global best = {best_val:.6f}")

        # optional early exit if we already beat the target
        if best_val <= TARGET:
            break

    # ------------------------------------------------------------------
    # Final polishing on the current global best (if a little time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # a short extra hill‑climb with a tighter transfer fraction
        best_a = hillclimb_targeted(best_a, n,
                                    start_time=start_time,
                                    deadline=deadline,
                                    best_a_global=best_a,
                                    best_val_global=best_val,
                                    target_val=TARGET)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # ensure we return a plain Python list of floats
    return best_a.tolist()
