"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation consists of three stages:

1. **Exploratory parametric search** – random piece‑wise power‑law
   families (and a few classical shapes) are generated and evaluated
   with a fast FFT based convolution.  The best candidate found is kept.

2. **Local hill‑climbing on the parameters** – starting from the best
   piece‑wise shape we perform a simple stochastic coordinate search
   (adjusting the break point and the two decay exponents) while the
   time budget permits.

3. **Fine‑grained sub‑gradient refinement** – the best sequence from
   stage 2 is refined with an Adam‑style projected sub‑gradient method
   that directly minimizes the true ‑∞‑norm (the maximum convolution
   entry).  The vector is kept on the probability simplex via the
   Duchi‑et‑al. projection.

The whole routine respects a 1000 s time budget and prints progress
information.  The final sequence is returned as a plain Python list
compatible with `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast and accurate)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective (the quantity that evaluate_sequence returns, up to scaling)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2 * n * max(conv(a,a))  (a is assumed to sum to 1)."""
    conv = _full_convolution_fft(a, a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Piece‑wise power‑law construction
# ----------------------------------------------------------------------
def _piecewise_powerlaw(n: int, B: int, p1: float, p2: float) -> np.ndarray:
    """
    Return a normalized vector of length n:
    a[i] = (i+1)^{-p1}               for i < B
    a[i] = (B+1)^{p2-p1} * (i+1)^{-p2}   for i >= B
    """
    idx = np.arange(1, n + 1, dtype=np.float64)

    # first segment
    a = np.empty(n, dtype=np.float64)
    a[:B] = idx[:B] ** (-p1)

    # continuity factor at the break point
    scale = (B + 1) ** (p2 - p1)
    a[B:] = scale * idx[B:] ** (-p2)

    # normalise to a probability distribution
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative height sequence that (approximately) minimises the
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and prints intermediate scores.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a size that is large enough to be competitive but
    # still cheap to evaluate many candidates.
    # ------------------------------------------------------------------
    n = 6000

    best_val = np.inf
    best_seq = None

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate and keep the best one seen.
    # ------------------------------------------------------------------
    def _try_candidate(a_vec: np.ndarray, source: str):
        nonlocal best_val, best_seq
        val = _objective(a_vec, n)
        if val < best_val:
            best_val = val
            best_seq = a_vec.copy()
            print(f"[{source}] new best = {best_val:.12f}")
        return val

    # ------------------------------------------------------------------
    # 1. Classical shapes (Catalan, exponential, power‑law)
    # ------------------------------------------------------------------
    # Catalan‑type decreasing shape (the exact solution of f*f = const on
    # the infinite line, truncated to length n).
    catalan = np.empty(n, dtype=np.float64)
    cur = 1.0
    for k in range(n):
        catalan[k] = cur
        cur = cur * (2.0 * (k + 1) - 1.0) / (2.0 * (k + 1))
    _try_candidate(catalan / catalan.sum(), "catalan")

    # Simple exponential families
    for alpha in np.linspace(0.4, 3.0, 15):
        L = alpha / n
        r = np.exp(-L)
        exp_shape = (1.0 - r) * (r ** np.arange(n, dtype=np.float64))
        _try_candidate(exp_shape / exp_shape.sum(), f"expα={alpha:.2f}")

    # Power‑law families
    for p in np.linspace(0.05, 0.95, 15):
        pw = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        _try_candidate(pw / pw.sum(), f"powp={p:.2f}")

    # ------------------------------------------------------------------
    # 2. Random piece‑wise power‑law search (coarse)
    # ------------------------------------------------------------------
    limit_stage2 = start_time + 0.30 * budget_s
    while time.time() < limit_stage2:
        B = rng.integers(int(0.15 * n), int(0.85 * n) + 1)
        p1 = rng.uniform(0.01, 0.50)        # slow decay at the left
        p2 = rng.uniform(0.60, 2.50)        # faster decay on the right
        a = _piecewise_powerlaw(n, B, p1, p2)
        _try_candidate(a, "rand_piece")

    # ------------------------------------------------------------------
    # 3. Local refinement of the best piece‑wise parameters
    # ------------------------------------------------------------------
    # If a piece‑wise candidate was the best, we keep its parameters;
    # otherwise we fall back to the current best_seq.
    if best_seq is None:
        # should never happen, but guard against it
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Remember the parameters of the current best piece‑wise shape
    # (very rough initialisation – will be overwritten if a better one
    # is discovered during the local search).
    cur_B = int(0.5 * n)
    cur_p1 = 0.30
    cur_p2 = 1.20

    # Try to infer parameters from best_seq (optional – simple heuristic)
    # We just keep the defaults; the local search will quickly correct them.

    step_B = max(1, int(0.10 * n))
    step_p = 0.2
    limit_stage3 = start_time + 0.45 * budget_s
    while time.time() < limit_stage3:
        # propose a neighbourhood move
        B = cur_B + rng.integers(-step_B, step_B + 1)
        B = int(np.clip(B, 1, n - 1))
        p1 = cur_p1 + rng.normal(scale=step_p)
        p2 = cur_p2 + rng.normal(scale=step_p)

        # keep parameters inside sensible bounds
        p1 = float(np.clip(p1, 0.01, 0.99))
        p2 = float(np.clip(p2, 0.51, 4.0))

        a = _piecewise_powerlaw(n, B, p1, p2)
        val = _objective(a, n)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            cur_B, cur_p1, cur_p2 = B, p1, p2
            print(f"[local] improved to {best_val:.12f} (B={B}, p1={p1:.3f}, p2={p2:.3f})")

        # shrink the neighbourhood slowly
        step_B = max(1, int(step_B * 0.97))
        step_p *= 0.97

    # ------------------------------------------------------------------
    # 4. Fine‑grained sub‑gradient optimisation (Adam on the simplex)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.08                # base learning rate
    beta1_pow = 1.0
    beta2_pow = 1.0
    stagn = 0

    while time.time() < deadline:
        # current convolution and score
        conv = _full_convolution_fft(a, a)
        max_b = conv.max()
        score = 2.0 * n * max_b

        if score < best_val - 1e-13:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[adam] new best = {best_val:.12f}")
        else:
            stagn += 1

        # active set (indices where the max is attained up to a tiny tolerance)
        tol = 1e-14
        active = np.flatnonzero(conv >= max_b - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size  # uniform weighting over the active set

        # gradient of the (smooth) max‑convolution w.r.t. a
        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update (projected onto the simplex after each step)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_candidate = a - step
        a = _project_simplex(a_candidate)

        # occasional random perturbation if we seem stuck
        if stagn > 3500:
            noise = rng.normal(scale=0.003, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            print("[adam] perturbation injected")

        # keep the vector well‑scaled every few thousand steps
        if stagn % 5000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Convert best sequence to plain Python list (clamped to [0,1000]).
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Optional logging of the final objective (using the supplied function)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] could not compute final_score (evaluate_sequence missing)")

    return final_seq
