Implementation details for convolution via FFT:

def full_convolution_fft(x, y):

    size = len(x) + len(y) -1

    N = 1 << (size-1).bit_length() # next power of 2

    fx = np.fft.rfft(x, n=N)

    fy = np.fft.rfft(y, n=N)

    fz = fx * fy

    z = np.fft.irfft(fz, n=N)

    return z[:size]

Define compute_correlation_fft(x, y):
    size = len(x) + len(y) - 1
    N = 1 << (size-1).bit_length() # next power of 2
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * np.conj(Y)
    corr = np.fft.irfft(Z, n=N)
    return corr[:size]

def project_simplex(v):
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
A fast projected‑Adam optimizer that works directly on the probability vector.
The objective to minimise is 2·n·max_k (a*a)[k] (scale‑invariant).
We smooth the max with a soft‑max (log‑sum‑exp) using a temperature β
that is increased during optimisation, which yields a differentiable surrogate.
The iterate is kept on the probability simplex via Euclidean projection
and symmetry a_i = a_{n‑1‑i} is enforced (this reduces the search space and
improves the shape of the autocorrelation).

The routine builds a small portfolio of symmetric seeds (including the
previous best construction if available) and runs the optimiser for each
seed within the remaining time budget.  The best vector found is returned
as a plain list – it can be passed to `evaluate_sequence` without further
post‑processing (the evaluation function is scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two (≥ v)
# ----------------------------------------------------------------------
def _next_pow2(v: int) -> int:
    """Return the smallest power of two ≥ v."""
    if v <= 0:
        return 1
    return 1 << ((v - 1).bit_length())


# ----------------------------------------------------------------------
# Full linear convolution using real FFT (returns length len(a)+len(b)-1)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    size = a.shape[0] + b.shape[0] - 1
    N = _next_pow2(size)
    Fa = np.fft.rfft(a, n=N)
    Fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(Fa * Fb, n=N)
    return conv[:size]


# ----------------------------------------------------------------------
# Autocorrelation a*a (full convolution)
# ----------------------------------------------------------------------
def _autocorrelation_fft(a: np.ndarray) -> np.ndarray:
    """Return full autocorrelation (a * a) using a single FFT."""
    n = a.shape[0]
    size = 2 * n - 1
    N = _next_pow2(size)
    Fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(Fa * Fa, n=N)
    return conv[:size]


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (∑x_i = 1, x_i ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond].max()
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(w, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Enforce symmetry a_i = a_{n-1-i}
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return 0.5 * (v + v[::-1])


# ----------------------------------------------------------------------
# Gradient of the smooth (soft‑max) peak of the autocorrelation
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  ( (1/β) log Σ_k exp(β·conv_k) )
    where conv = a * a (full).  The gradient is
        g_i = 2 * Σ_k w_k a_{k-i}
    with w_k = exp(β·conv_k) / Σ_j exp(β·conv_j).
    """
    n = a.shape[0]
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:          # safety, should not happen for reasonable β
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # correlation of w with a (no reversal) : g_i = 2 * Σ_j w_{i+j} a_j
    # This can be obtained from convolution of w with reversed a.
    a_rev = a[::-1]
    conv_rev = _full_convolution_fft(w, a_rev)          # length = 3n‑2
    start = n - 1
    g = 2.0 * conv_rev[start:start + n]
    return g


# ----------------------------------------------------------------------
# Optimiser (projected Adam) for a single seed
# ----------------------------------------------------------------------
def _run_optimizer(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector to minimise 2·n·max(a*a) within `time_limit` seconds.
    Returns the best vector encountered and its exact score.
    """
    start = time.time()
    n = initial_a.shape[0]

    # Adam state
    m = np.zeros_like(initial_a)
    v = np.zeros_like(initial_a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.30                     # base learning rate (in the original space)

    # temperature schedule for the smooth max
    beta = 1.0
    beta_target = 50.0
    beta_factor = 1.005            # multiplicative increase per iteration

    a = initial_a.copy()
    best_score = np.inf
    best_vec = a.copy()
    last_improve = 0
    iteration = 0

    # pre‑computed sizes for FFTs (they stay constant for a fixed n)
    size_conv = 2 * n - 1
    N_conv = _next_pow2(size_conv)           # for a*a
    size_grad = 3 * n - 2
    N_grad = _next_pow2(size_grad)           # for w * a_rev

    while time.time() - start < time_limit - 0.5:
        iteration += 1

        # ---- autocorrelation and current score ----
        Fa = np.fft.rfft(a, n=N_conv)
        conv = np.fft.irfft(Fa * Fa, n=N_conv)[:size_conv]
        max_c = conv.max()
        cur_score = 2.0 * n * max_c

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # ---- gradient of the smooth max ----
        # compute weights w and gradient G
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        a_rev = a[::-1]
        Fw = np.fft.rfft(w, n=N_grad)
        Fa_rev = np.fft.rfft(a_rev, n=N_grad)
        conv_rev = np.fft.irfft(Fw * Fa_rev, n=N_grad)[:size_grad]
        start_idx = n - 1
        G = 2.0 * conv_rev[start_idx:start_idx + n]

        # ---- Adam update in the original space ----
        m = beta1 * m + (1.0 - beta1) * G
        v = beta2 * v + (1.0 - beta2) * (G * G)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- projection & symmetry ----
        a = _project_simplex(a)
        a = _symmetrize(a)

        # ---- temperature schedule ----
        if beta < beta_target:
            beta = min(beta_target, beta * beta_factor)

        # ---- re‑heat if stuck for a long time ----
        if iteration - last_improve > 3000:
            a += np.random.randn(n) * 1e-4
            a = _project_simplex(a)
            a = _symmetrize(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

    return best_vec, best_score


# ----------------------------------------------------------------------
# Parametric families for deterministic seeds (all symmetric)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

    try:
        ext = np.array(height_sequence_1, dtype=float)
        ext = np.maximum(ext, 0.0)
        s = ext.sum()
        if s > 0.0:
            ext /= s
            ext = _symmetrize(ext)
            seeds.append(ext)
            print(f"External seed found (length {len(ext)})")
    except NameError:
        pass

    # deterministic families at several lengths
    for n in (5808, 8000, 10000):
        seeds.append(u_min_distribution(n, 0.9))
        seeds.append(u_prod_distribution(n, 0.9))
        seeds.append(exp_distribution(n, 0.01))
        seeds.append(np.ones(n) / n)                     # uniform (symmetrised automatically)

    # a few random Dirichlet vectors (forced symmetric)
    for n in (5808, 8000, 10000):
        r = np.random.rand(n)
        r = r / r.sum()
        r = _symmetrize(r)
        seeds.append(r)

    # --------------------------------------------------------------
    # 1. Optimise each seed within the remaining budget
    # --------------------------------------------------------------
    best_overall_vec = None
    best_overall_score = np.inf

    total_seeds = len(seeds)
    for idx, seed_vec in enumerate(seeds):
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # allocate roughly equal remaining time to the seeds still to process
        seeds_left = total_seeds - idx
        per_seed_budget = remaining / seeds_left
        per_seed_budget = min(per_seed_budget, 300.0)   # never waste too much on one seed

        a = np.maximum(seed_vec, 0.0)
        if a.sum() == 0.0:
            a = np.ones_like(a)
        a = a / a.sum()
        a = _project_simplex(a)
        a = _symmetrize(a)

        print(f"\n=== Optimising seed {idx + 1}/{total_seeds} "
              f"(n={len(a)}) → budget {per_seed_budget:.1f}s ===")
        vec, sc = _run_optimizer(a, per_seed_budget)
        print(f"Seed {idx + 1} best score = {sc:.6f}")

        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_vec = vec.copy()
            print(f"*** New global best = {best_overall_score:.6f}")

    # --------------------------------------------------------------
    # 2. Fallback (should never happen)
    # --------------------------------------------------------------
    if best_overall_vec is None:
        n_fallback = 5000
        best_overall_vec = np.ones(n_fallback) / n_fallback

    print("\n=== Search finished ===")
    print(f"Best score found: {best_overall_score:.6f} (length {len(best_overall_vec)})")
    return best_overall_vec.tolist()
