"""
Propose a step‑function (non‑negative sequence) that approximately minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in two stages:

1. Fast Adam optimisation on a smooth approximation of the max‑convolution.
   The sequence stays on the probability simplex (non‑negative, unit sum)
   using the Duchi‑et‑al. projection.

2. A linear‑programming refinement (if SciPy is available).  For the current
   vector f we solve

        max   sum(b)
        s.t.  conv(f, b) <= M   (M = max(conv(f,f))),
              b >= 0

   The LP often finds a vector b whose total mass exceeds that of f.
   After normalising b to unit sum we mix it with the current iterate.
   This step can strictly reduce the maximal autocorrelation and yields
   a noticeable improvement over pure gradient descent.

Both phases respect the overall time budget and keep the best sequence
seen.  The returned list complies with the security checks of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for many calls).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv
