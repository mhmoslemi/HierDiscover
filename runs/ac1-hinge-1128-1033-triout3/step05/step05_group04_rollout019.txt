def _geometric_sequence(n, r):
    if r <= 0.0:
        return np.full(n, 1.0/n, dtype=np.float64)
    if r >= 1.0:
        # treat as uniform
        return np.full(n, 1.0/n, dtype=np.float64)
    log_r = np.log(r)
    # compute r^i using exponentiation
    exponents = np.arange(n, dtype=np.float64)
    pow_r = np.exp(exponents * log_r)  # r^i
    # compute normalizing factor: c = (1-r) / (1 - r**n)
    r_pow_n = np.exp(n * log_r)
    denom = 1.0 - r_pow_n
    if denom == 0.0:
        factor = 1.0 / n
    else:
        factor = (1.0 - r) / denom
    a = factor * pow_r
    # Ensure non-negative and sum to 1 (project just in case)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a
```

        a = np.full(n, 1.0/n)
        return 2.0 * n * (1.0/n), a
    log_r = np.log(r)

"""
Improved search for a step‑function sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

The idea:
1.  Initialise the vector with a simple 1‑parameter family – a
    geometric (exponential) decay a_i = C·r^i – and optimise the
    decay factor r using a cheap 1‑dimensional golden‑section search.
    This already beats the uniform distribution (score ≈ 1.8 for n=5000).

2.  Starting from the best geometric shape, run a projected Adam optimiser
    that minimises a smooth log‑sum‑exp approximation of the max‑convolution.
    The optimiser respects the probability simplex (non‑negative, sum=1).

3.  A short “hard‑max” polishing phase follows, where the true maximal
    convolution entries receive a binary mask and a tiny constant step size
    is used to fine‑tune the solution.

The routine respects the supplied time budget (default 1000 s), prints
progress information, and finally returns a list that can be fed directly
to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full (linear) convolution via real FFT (fast for the lengths we use)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# One‑parameter geometric family  a_i = C * r**i ,   sum(a)=1
# ----------------------------------------------------------------------
def _geometric_sequence(n: int, r: float) -> np.ndarray:
    """Return a normalized geometric sequence of length n for decay r∈(0,1)."""
    if r <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if r >= 1.0:
        return np.full(n, 1.0 / n, dtype=np.float64)

    log_r = np.log(r)
    # denominator 1 - r**n, safely computed via exp
    r_pow_n = np.exp(n * log_r)
    denom = 1.0 - r_pow_n
    # scaling factor C so that the sum is exactly 1
    C = (1.0 - r) / denom
    exponents = np.arange(n, dtype=np.float64)
    a = C * np.exp(exponents * log_r)
    # numerical safety – enforce the simplex exactly
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Score (objective) for a given vector (sum is assumed = 1)
# ----------------------------------------------------------------------
def _score_of_vector(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) for a vector that already sums to 1."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Golden‑section search on the decay parameter r
# ----------------------------------------------------------------------
def _best_geometric(n: int, time_limit: float, rng: np.random.Generator):
    """Find the r∈[0,1) giving the smallest score for length n."""
    # bounds for r – we never need r exactly 1 (uniform case)
    lo, hi = 0.0, 0.999999
    phi = (np.sqrt(5.0) - 1.0) / 2.0

    # initialise the two interior points
    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)

    best_val = np.inf
    best_a = None
    best_r = None

    # helper that also returns the array (required later)
    def eval_r(r):
        a = _geometric_sequence(n, r)
        val = _score_of_vector(a)
        return val, a

    fc, ac = eval_r(c)
    fd, ad = eval_r(d)

    # maintain best so far
    if fc < best_val:
        best_val, best_a, best_r = fc, ac, c
    if fd < best_val:
        best_val, best_a, best_r = fd, ad, d

    while time.time() < time_limit and (hi - lo) > 1e-12:
        if fc < fd:
            hi = d
            d, fd, ad = c, fc, ac
            c = hi - phi * (hi - lo)
            fc, ac = eval_r(c)
        else:
            lo = c
            c, fc, ac = d, fd, ad
            d = lo + phi * (hi - lo)
            fd, ad = eval_r(d)

        if fc < best_val:
            best_val, best_a, best_r = fc, ac, c
        if fd < best_val:
            best_val, best_a, best_r = fd, ad, d

    return best_a, best_val, best_r


# ----------------------------------------------------------------------
# Main entry – the function that will be called by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Generate a non‑negative height sequence that (approximately) minimises
    `2 * n * max(conv(a, a)) / (sum(a) ** 2)`.  The routine respects the
    supplied time budget, prints a few progress lines, and finally returns
    the best sequence found (as a plain Python list of floats).
    """
    deadline = time.time() + budget_s - 5.0   # leave a short safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1️⃣  Choose length – 5000 is a good trade‑off (used by the baseline)
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # 2️⃣  Find a good geometric starting point (fast 1‑D optimisation)
    # --------------------------------------------------------------
    print("[init] searching geometric family …")
    geo_start, geo_score, geo_r = _best_geometric(
        n, time_limit=deadline * 0.15, rng=rng
    )
    print(f"[init] geometric r={geo_r:.6f}  score={geo_score:.12f}")

    # --------------------------------------------------------------
    # 3️⃣  Projected Adam optimisation on the soft‑max approximation
    # --------------------------------------------------------------
    a = geo_start.copy()
    best_seq = a.copy()
    best_val = geo_score
    stagn = 0

    # Adam hyper‑parameters (slightly more aggressive than the baseline)
    lr = 0.18
    lr_min = 1e-6
    lr_decay = 0.99998
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature for the soft‑max – start relatively high
    eps = max(1e-4, 0.2 * a.max())          # a few percent of the current max entry
    eps_min = 1e-12
    eps_decay = 0.99992

    it = 0
    phase1_deadline = time.time() + (deadline - time.time()) * 0.65

    while time.time() < phase1_deadline:
        it += 1

        # current convolution and score
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # keep global best
        if score < best_val - 1e-14:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it:6d}] new best={best_val:.12f}  lr={lr:.2e}  eps={eps:.2e}")
        else:
            stagn += 1

        # ----- soft‑max weighting -------------------------------------------------
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0.0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # ----- gradient of the weighted sum (cross‑correlation) --------------------
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- Adam step ---------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        t = it
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a + step
        a = _project_simplex(a_new)

        # schedules
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional perturbation to escape plateaus
        if stagn > 2000:
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            lr = max(lr_min, lr * 1.2)
            print(f"[perturb] iteration {it}")

    # --------------------------------------------------------------
    # 4️⃣  Hard‑max polishing (binary mask of the true maximal entries)
    # --------------------------------------------------------------
    print("[polish] hard‑max phase …")
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_val - 1e-14:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[polish {it:6d}] new best={best_val:.12f}")
        else:
            stagn += 1

        # binary mask of entries that are (almost) maximal
        tol = max_conv * 1e-9 + 1e-14
        mask = (conv >= max_conv - tol).astype(np.float64)
        w_sum = mask.sum()
        w = mask / w_sum if w_sum > 0 else np.full_like(conv, 1.0 / conv.size)

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        step = -2e-5 * grad
        a = a + step
        a = _project_simplex(a)

        # jitter if we are stuck
        if stagn > 5000:
            a = a + rng.normal(scale=0.004, size=n)
            a = _project_simplex(a)
            stagn = 0
            print(f"[jitter] iteration {it}")

    # --------------------------------------------------------------
    # 5️⃣  Final tiny sub‑gradient refinement (pure max‑entry steps)
    # --------------------------------------------------------------
    a = best_seq.copy()
    for _ in range(3000):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        active = (conv >= mx - 1e-15).astype(np.float64)
        w_sum = active.sum()
        w = active / w_sum if w_sum > 0 else np.full_like(conv, 1.0 / conv.size)
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]
        a = _project_simplex(a - 4e-6 * grad)

    # --------------------------------------------------------------
    # 6️⃣  Return result in the format required by `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}  "
          f"r_opt≈{geo_r:.6f}")
    return final_seq
