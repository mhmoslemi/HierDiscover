"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

Algorithm
---------

1. **Initialisation** – start from the known construction `height_sequence_1`
   (if present) by linear interpolation, otherwise from a random Dirichlet
   vector on the simplex (Σa = 1).

2. **Phase A – Smooth‑max gradient descent**  
   We minimise the maximum entry of the autoconvolution `b = a∗a`
   via a smooth approximation  
       `smooth_max(b) = (1/α)·log Σ exp(α·b_k)`.  
   Its gradient is `∂/∂a_i smooth_max = 2·(p∗a)_i` where  
   `p_k = exp(α·b_k) / Σ exp(α·b)`.  
   The smoothing factor `α` is increased from 1 to ≈12 during the phase.
   After each gradient step we **project** the trial vector back onto the
   probability simplex (non‑negative entries summing to 1).  A small amount
   of random jitter is added occasionally to escape shallow plateaus.

3. **Phase B – Deterministic hill‑climbing**  
   Starting from the best point found in Phase A we perform fast
   mass‑transfer moves: a small amount `δ` is moved from a source index `i`
   to a target `j`.  The full convolution can be updated in `O(n)` by
   bilinearity, and a move is accepted only if it strictly reduces the
   true objective `2·n·max(b)`.  After many useless attempts we re‑inject
   a small Gaussian perturbation and continue.

Both phases comfortably fit into the 1000 s budget (≈55 % for Phase A,
the remainder for Phase B).  The final vector is returned as a plain
list of Python floats ready for scoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {z≥0, Σz=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries zero
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    # renormalise to avoid tiny drift
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Smooth‑max gradient (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, alpha: float):
    """
    Returns:
        grad   – ∂/∂a smooth_max(b)  (same shape as a)
        b      – full convolution a∗a
        b_max  – maximum entry of b
    """
    b = _full_convolution(a)                 # length 2·n-1
    b_max = b.max()
    # p_k ∝ exp(alpha·(b_k - b_max))  – numerical stability via shift
    shifted = b - b_max
    exp_term = np.exp(alpha * shifted)
    sum_exp = exp_term.sum()
    if sum_exp == 0.0:                       # underflow – concentrate on maxima
        mask = np.isclose(b, b_max)
        exp_term = mask.astype(np.float64)
        sum_exp = exp_term.sum()
    p = exp_term / sum_exp                    # probability distribution over k

    # gradient = 2 * (p ∗ a)   (slice n‑1 … 2·n‑2)
    conv = _conv_fft(p, a)                    # length 3·n‑2
    start = a.shape[0] - 1
    grad = 2.0 * conv[start:start + a.shape[0]]
    return grad, b, b_max

        if a[i] <= 1e-14:
            continue

        # target index (different from source)
        j = np.random.randint(n)
        if j == i:
            continue

        # amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        # contribution of cross‑terms 2·δ·a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val:                      # accept improving move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # after prolonged stagnation inject a tiny random perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")

        it += 1

    return best_a

# Main entry point required by the competition framework

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a step‑function under the given time budget and returns a list
    of non‑negative floats (Σa = 1) ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between expressivity
    # and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise `a` – reuse a known construction if available.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)          # enforce Σa = 1
    best_a = a.copy()
    best_val = float('inf')

    # --------------------------------------------------------------
    # Phase A – smooth‑max gradient descent
    # --------------------------------------------------------------
    phase_a_end = start_time + 0.55 * budget_s
    iteration = 0
    alpha_max = 12.0                 # final smoothing strength
    step_start = 0.12
    step_end = 0.001

    while time.time() < phase_a_end:
        now = time.time()
        progress = (now - start_time) / (phase_a_end - start_time)  # 0 → 1
        alpha = 1.0 + (alpha_max - 1.0) * progress
        step = step_start * (1.0 - progress) + step_end * progress

        grad, b, b_max = _smooth_max_grad(a, alpha)
        cur_val = 2.0 * n * b_max

        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Grad {iteration:7d}] new best = {best_val:.6f} (α={alpha:.2f})")

        # gradient step + projection
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional random jitter to escape tiny plateaus
        if np.random.rand() < 0.002:
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)

        # very long stagnation → full restart
        if iteration % 250_000 == 0 and iteration != 0:
            a = np.random.rand(n)
            a = _project_simplex(a)
            print(f"[Grad {iteration}] restart after stagnation")

        iteration += 1

    # --------------------------------------------------------------
    # Phase B – deterministic hill‑climbing on the true objective
    # --------------------------------------------------------------
    best_a = _hillclimb_phase(best_a, n, start_time, deadline,
                              best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
