def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Determine n and initial a0
    try:
        old = np.asarray(height_sequence_1, dtype=float)
        n = len(old)
        a0 = old.copy()
    except Exception:
        n = 4000
        a0 = np.full(n, 1.0 / n, dtype=float)

    # Normalize
    a0 = np.maximum(a0, 0.0)
    total = a0.sum()
    if total < 1e-12:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / total

    best_a = a0.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # main loop
    restart = 0
    while time.time() < deadline - 10.0:
        restart += 1
        # generate start
        if rng.random() < 0.3:
            # perturb best
            noise = rng.normal(scale=0.01, size=n)
            start_a = best_a + noise
            start_a = np.clip(start_a, 0, None)
            s = start_a.sum()
            if s > 0:
                start_a = start_a / s
            else:
                start_a = np.full(n, 1.0/n)
        else:
            # Dirichlet
            alpha = rng.choice([0.2, 0.5, 1.0, 2.0])
            start_a = rng.dirichlet(np.full(n, alpha))
        cur_a = start_a
        cur_val = evaluate_sequence(cur_a.tolist())
        if cur_val < best_val:
            best_val, best_a = cur_val, cur_a.copy()
            print(f"[restart {restart}] new global best from start = {best_val:.6f}")

        # annealing parameters
        T = max(0.01 * cur_val, 0.001)
        T_min = 1e-6
        decay = 0.99995

        # move loop
        move_count = 0
        while time.time() < deadline - 5.0:
            move_count += 1
            # decide move type
            r = rng.random()
            if r < 0.6:
                # mass transfer (random)
                donor = rng.integers(0, n)
                if cur_a[donor] <= 1e-15:
                    continue
                recipient = rng.integers(0, n-1)
                if recipient >= donor:
                    recipient += 1
                delta = cur_a[donor] * rng.random() * 0.2
                if delta <= 0:
                    continue
                new_a = cur_a.copy()
                new_a[donor] -= delta
                new_a[recipient] += delta
            elif r < 0.8:
                # targeted mass transfer
                new_a = cur_a.copy()
                # compute conv and index
                conv = _full_convolution(cur_a)
                k = int(conv.argmax())
                i_min = max(0, k-(n-1))
                i_max = min(n-1, k)
                i_vals = np.arange(i_min, i_max+1)
                j_vals = k - i_vals
                products = cur_a[i_vals] * cur_a[j_vals]
                idx = int(products.argmax())
                i_selected = i_vals[idx]
                j_selected = j_vals[idx]
                donor = i_selected if cur_a[i_selected] >= cur_a[j_selected] else j_selected
                if cur_a[donor] <= 1e-15:
                    continue
                recipient = rng.integers(0, n)
                if recipient == donor:
                    recipient = (recipient+1) % n
                delta = cur_a[donor] * rng.random() * 0.2
                new_a[donor] -= delta
                new_a[recipient] += delta
            else:
                # swap
                i = rng.integers(0, n)
                j = rng.integers(0, n-1)
                if j >= i:
                    j += 1
                new_a = cur_a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]

            # evaluate new
            try:
                new_val = evaluate_sequence(new_a.tolist())
            except Exception:
                continue
            if np.isinf(new_val) or np.isnan(new_val):
                continue

            # acceptance

            if new_val < cur_val:
                accept = True
            else:
                # metropolis
                prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                cur_a = new_a
                cur_val = new_val
                # update global best
                if cur_val < best_val:
                    best_val, best_a = cur_val, cur_a.copy()
                    print(f"[restart {restart}] iter {move_count} new best = {best_val:.6f}")

            T = max(T * decay, T_min)

            # periodic reordering (every 5000 moves)
            if move_count % 5000 == 0:
                # anti-symmetric reorder
                a_ord = _anti_symmetric_reorder(cur_a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:

                    print(f"[restart {restart}] anti-symmetric reorder improved to {best_val:.6f}")
                # interleave

                if val_int < best_val:

        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:

        val_int = evaluate_sequence(a_int.tolist())
        if val_int < best_val:

"""
Optimization of a non‑negative step function with respect to the
``evaluate_sequence`` objective.

The method is a multi‑restart simulated‑annealing / hill‑climbing
procedure:

* Initialise either from the previously discovered sequence
  ``height_sequence_1`` (if present) or from a random Dirichlet draw.
* For each restart perform a stochastic local search:
    – random mass‑transfer moves,
    – targeted mass‑transfer moves that reduce the current
      convolution peak,
    – random swaps.
  Moves are accepted if they improve the objective, otherwise with a
  Metropolis probability that gradually cools.
* Periodically apply cheap deterministic reorderings
  (anti‑symmetric and high‑low interleaving) that are known to flatten
  the self‑convolution.
* After the time budget a short deterministic hill‑climb (mass transfers
  + swaps) refines the best solution.
The algorithm respects the 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to ``x``."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution ``x * x`` using FFT.
    The length of ``x`` is ``n`` and the result has length ``2*n-1``.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative values can appear because of rounding; clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce an ordering where the largest values occupy the left half
    and the smallest values the symmetric right half.  This heuristic
    often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large on left
        new_a[n - 1 - i] = sorted_vals[i]          # small on right

    if n % 2:
        new_a[half] = sorted_vals[half]            # middle stays
    return new_a


def _interleave_high_low(a: np.ndarray) -> np.ndarray:
    """
    Interleave the largest remaining value with the smallest remaining
    value (high‑low‑high‑low …).  Another cheap re‑ordering that tends
    to flatten the convolution.
    """
    n = a.shape[0]
    vals = np.sort(a)          # ascending
    lo = 0
    hi = n - 1
    new_a = np.empty_like(a)
    for idx in range(n):
        if idx % 2 == 0:
            new_a[idx] = vals[hi]
            hi -= 1
        else:
            new_a[idx] = vals[lo]
            lo += 1
    return new_a


def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Simple deterministic hill‑climb that repeatedly tries
    random mass‑transfers (up to 5 % of the donor mass) and random swaps.
    Only strict improvements are kept.  The loop stops when the time
    budget expires.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # 20 % chance of a pure swap, 80 % chance of a mass transfer
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 0.0:
                continue
            # move up to 5 % of donor's mass
            delta = cur_a[i] * 0.05 * rng.random()
            if delta <= 0.0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if not np.isfinite(val):
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence`` within the allotted time budget.

    The procedure consists of many independent stochastic restarts,
    simulated‑annealing style moves, periodic deterministic re‑orderings,
    and a final hill‑climb refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine problem size and an initial sequence
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` is a global possibly set by a previous run
        old = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = int(old.shape[0])
        a0 = old.copy()
    except Exception:
        n = 4000
        a0 = np.full(n, 1.0 / n, dtype=float)

    # enforce feasibility required by ``evaluate_sequence``
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s < 1e-12:
        a0 = np.full(n, 1.0 / n, dtype=float)
    else:
        a0 = a0 / s

    best_a = a0.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Main stochastic‑annealing loop (multiple restarts)
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 15.0:   # leave a few seconds for final refinement
        restart += 1
        print(f"[restart {restart}] starting")

        # --- initialise this restart -------------------------------------------------
        if rng.random() < 0.25:
            # start from a noisy version of the current global best
            noise = rng.normal(scale=0.02, size=n)
            start_a = best_a + noise
            start_a = np.clip(start_a, 0.0, None)
            total = start_a.sum()
            if total > 0:
                start_a = start_a / total
            else:
                start_a = np.full(n, 1.0 / n, dtype=float)
        else:
            # random Dirichlet start – provides both spiky and smooth draws
            alpha = rng.choice([0.2, 0.5, 1.0, 2.0])
            start_a = rng.dirichlet(np.full(n, alpha))
        cur_a = start_a.copy()
        cur_val = evaluate_sequence(cur_a.tolist())
        if cur_val < best_val:
            best_val, best_a = cur_val, cur_a.copy()
            print(f"[restart {restart}] new global best from start = {best_val:.6f}")

        # --- annealing parameters ---------------------------------------------------
        T = max(0.02 * cur_val, 1e-3)    # initial temperature
        T_min = 1e-6
        decay = 0.99994                 # exponential cooling factor

        move_counter = 0
        # -------------------------------------------------------------------------
        # local search loop
        # -------------------------------------------------------------------------
        while time.time() < deadline - 10.0:
            move_counter += 1
            r = rng.random()

            # ----------- 1) random mass transfer ----------------------------------
            if r < 0.55:
                donor = rng.integers(0, n)
                if cur_a[donor] <= 1e-15:
                    continue
                recipient = rng.integers(0, n - 1)
                if recipient >= donor:
                    recipient += 1
                delta = cur_a[donor] * rng.random() * 0.20   # up to 20 % of donor
                new_a = cur_a.copy()
                new_a[donor] -= delta
                new_a[recipient] += delta

            # ----------- 2) targeted mass transfer (focus on current peak) ------
            elif r < 0.8:
                # compute the current self‑convolution (FFT – cheap)
                conv = _full_convolution(cur_a)
                k = int(conv.argmax())
                i_min = max(0, k - (n - 1))
                i_max = min(n - 1, k)
                i_vals = np.arange(i_min, i_max + 1)
                j_vals = k - i_vals
                products = cur_a[i_vals] * cur_a[j_vals]
                idx = int(products.argmax())
                i_sel = i_vals[idx]
                j_sel = j_vals[idx]
                donor = i_sel if cur_a[i_sel] >= cur_a[j_sel] else j_sel
                if cur_a[donor] <= 1e-15:
                    continue
                recipient = rng.integers(0, n)
                if recipient == donor:
                    recipient = (recipient + 1) % n
                delta = cur_a[donor] * rng.random() * 0.20
                new_a = cur_a.copy()
                new_a[donor] -= delta
                new_a[recipient] += delta

            # ----------- 3) swap ---------------------------------------------------
            else:
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                new_a = cur_a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]

            # -------------------------------------------------------------------------
            # evaluate the proposal
            # -------------------------------------------------------------------------
            try:
                new_val = evaluate_sequence(new_a.tolist())
            except Exception:
                continue
            if not np.isfinite(new_val):
                continue

            # acceptance test (Metropolis)
            if new_val < cur_val:
                accept = True
            else:
                prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
                accept = rng.random() < prob

            if accept:
                cur_a = new_a
                cur_val = new_val
                if cur_val < best_val:
                    best_val, best_a = cur_val, cur_a.copy()
                    print(f"[restart {restart}] iter {move_counter} new best = {best_val:.6f}")

            # temperature update
            T = max(T * decay, T_min)

            # every few thousand moves try cheap deterministic reorderings
            if move_counter % 5000 == 0:
                # anti‑symmetric reorder
                a_as = _anti_symmetric_reorder(cur_a)
                val_as = evaluate_sequence(a_as.tolist())
                if val_as < best_val:
                    best_val, best_a = val_as, a_as.copy()
                    print(f"[restart {restart}] anti‑symmetric reorder improved to {best_val:.6f}")
                # high‑low interleave
                a_il = _interleave_high_low(cur_a)
                val_il = evaluate_sequence(a_il.tolist())
                if val_il < best_val:
                    best_val, best_a = val_il, a_il.copy()
                    print(f"[restart {restart}] interleave improved to {best_val:.6f}")

        # End of this restart – a final quick reordering attempt
        a_as = _anti_symmetric_reorder(cur_a)
        v_as = evaluate_sequence(a_as.tolist())
        if v_as < best_val:
            best_val, best_a = v_as, a_as.copy()
            print(f"[restart {restart}] final anti‑symmetric improved to {best_val:.6f}")
        a_il = _interleave_high_low(cur_a)
        v_il = evaluate_sequence(a_il.tolist())
        if v_il < best_val:
            best_val, best_a = v_il, a_il.copy()
            print(f"[restart {restart}] final interleave improved to {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Final deterministic hill‑climb (mass transfers + swaps)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ----------------------------------------------------------------------
    # Report the final result
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
