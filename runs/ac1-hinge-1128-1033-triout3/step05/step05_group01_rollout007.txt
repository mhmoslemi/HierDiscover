```
def project_to_simplex(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```
        i_end = min(k, n-1)
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        contrib = best_a[i_range] * best_a[j_range]
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])
        # choose random destinations with low mass

        if low.size >= 2:

        else:

        attempts = 0

            attempts += 1

            if delta < 1e-15: continue
            new_a = best_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:
            delta = best_a[i_max] * step_frac
```

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------

1. **Multiple initialisations** – uniform, random Dirichlet, a sine‑wave,
   a symmetric triangle and (if available) the previously best known
   sequence are normalised to sum = 1.

2. **Projected exponentiated‑gradient descent** –  we minimise
   ``max(conv)`` where ``conv = a * a`` (full linear convolution).
   For the current maximum lag ``k`` the sub‑gradient of
   ``max(conv)`` w.r.t. ``a_i`` is ``2·a_{k−i}``.  An exponentiated‑
   gradient update ``a_i ← a_i·exp(−η·2·a_{k−i})`` keeps the iterates
   on the probability simplex (renormalisation after each step).
   The step‑size ``η`` follows a 1/√t schedule and a tiny random jitter
   is added every 1000 iterations to escape plateaux.

3. **Guided hill‑climbing** – starting from the best point found,
   repeatedly move a small amount of mass away from the pair
   ``(i_max , j_max)`` that contributes most to the current peak
   ``max(conv)`` and add it to two low‑mass positions.  The move is
   kept only if the exact ``evaluate_sequence`` value improves.

All heavy work (convolution) is performed via an FFT implementation
(`_full_convolution`).  The routine respects a 1000 s time limit; the
best sequence found so far is returned.

The code is self‑contained, uses only NumPy and the already imported
``evaluate_sequence`` function, and complies with the required API.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative values caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (unused – kept for
# reference / possible future tweaks)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the largest index where the condition holds
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing – small deterministic mass transfers that reduce
# the *exact* objective.
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Refine ``a`` by repeatedly moving a fraction of the mass away from the
    most offending pair (i_max, j_max) to two low‑mass positions.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_best = cur_val
    step_frac = 0.05  # start moving 5 % of the responsible mass

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # (re‑compute objective directly; sum(cur_a)==1)
        cur_obj = 2.0 * n * max_b

        # locate a lag attaining the maximum
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two destination indices that (mostly) do not recreate the same lag
        low = np.where(cur_a < cur_a.mean())[0]
        if low.size >= 2:
            p, q = rng.choice(low, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # amount of mass to move
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-15:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p]      += delta
            new_a[q]      += delta
        else:  # i_max == j_max (happens for k even and i=j)
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-15:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p]     += delta

        # safety – keep non‑negativity and re‑normalise (sum is preserved up to rounding)
        if (new_a < 0).any():
            continue
        new_a = np.maximum(new_a, 0.0)
        new_a = new_a / new_a.sum()

        # evaluate the exact objective (fast direct computation)
        new_obj = 2.0 * n * _full_convolution(new_a).max()
        if new_obj < cur_best:
            cur_a = new_a
            cur_best = new_obj
            print(f"[hillclimb] new best = {cur_best:.6f}")
            step_frac = max(step_frac * 0.9, 1e-4)
        else:
            step_frac = max(step_frac * 0.99, 1e-4)

    return cur_a, cur_best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the heights of a step‑function under the evaluate_sequence
    objective using a combination of exponentiated‑gradient descent,
    several diverse initialisations and a final guided hill‑climb.
    Returns the best list of floats found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate size keeps each FFT cheap while offering
    # enough degrees of freedom to flatten the convolution.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Helper to compute the objective when the sum of a is (approximately) 1
    # ------------------------------------------------------------------
    def _objective(a_vec: np.ndarray) -> float:
        return 2.0 * n * _full_convolution(a_vec).max()

    # ------------------------------------------------------------------
    # Initialise a collection of diverse starting points (all normalised)
    # ------------------------------------------------------------------
    init_funcs = []

    # 1. Uniform distribution
    init_funcs.append(lambda: np.full(n, 1.0 / n))

    # 2. Random Dirichlet‑like (gamma(1,1))
    init_funcs.append(lambda: rng.random(n))

    # 3. Positive sine wave (smooth, zero‑free)
    init_funcs.append(lambda: np.sin(np.pi * (np.arange(n) + 0.5) / n))

    # 4. Symmetric triangle (largest in the middle)
    init_funcs.append(lambda: np.minimum(np.arange(1, n + 1),
                                         np.arange(n, 0, -1)))

    # 5. Previous best sequence, if the global variable exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a_old = np.interp(new_x, old_x, old)
        else:
            a_old = old.copy()
        init_funcs.append(lambda: a_old)
    except Exception:
        pass  # no historic data

    best_a = None
    best_val = float("inf")
    max_iters_per_start = 20000   # enough for convergence; loop will stop earlier if out of time
    step0 = 5.0                    # initial multiplicative step size (1/√t schedule)

    # ------------------------------------------------------------------
    # Main optimisation loop over the initialisers
    # ------------------------------------------------------------------
    for init_id, init in enumerate(init_funcs):
        if time.time() >= deadline:
            break

        a = init()
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[init {init_id}] new global best = {best_val:.6f}")

        # ----------- exponentiated‑gradient descent on the max‑conv ---------
        for it in range(max_iters_per_start):
            if time.time() >= deadline:
                break

            conv = _full_convolution(a)
            maxb = conv.max()
            cur_val = 2.0 * n * maxb

            # keep track of the best ever seen
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[init {init_id} it {it}] new global best = {best_val:.6f}")

            # pick a lag attaining the current maximum (break ties randomly)
            ks = np.flatnonzero(conv == maxb)
            k = int(rng.choice(ks))

            # sub‑gradient of max(conv) for this lag (g_i = 2·a_{k−i})
            g = np.zeros_like(a)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] = 2.0 * a[j_range]

            # step size (1/√t schedule)
            eta = step0 / np.sqrt(it + 1.0)

            # exponentiated‑gradient update (keeps positivity)
            a = a * np.exp(-eta * g)
            # avoid exact zeros that could stall the dynamics
            a = np.maximum(a, 1e-20)
            a = a / a.sum()

            # occasional tiny random jitter to shake out of flat plateaux
            if (it + 1) % 1000 == 0:
                jitter = rng.normal(scale=1e-3, size=n)
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = a / a.sum()

    # ------------------------------------------------------------------
    # Final refinement with a deterministic hill‑climb (if time permits)
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final report (the evaluation framework will call evaluate_sequence
    # again, but we print the value here for debugging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
