"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

by running several short restarts with diverse initial shapes.
Each restart uses a soft‑max (temperature) gradient together with an
Adam‑style update on the probability simplex.  In addition to the
gradient of the (smoothed) maximum we add a small “flatness” penalty
that pushes the self‑convolution towards a constant (its average
value 1/(2n‑1)).  The best vector seen over all restarts is returned.
The routine respects a 1000 s time‑budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the probability simplex (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (length = len(a)+len(b)-1)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Gradient of a linear combination of convolution entries:
# for a weight vector w (same length as full convolution) we have
#   d/da sum_k w_k (a*a)_k = 2 * rev(a) * w   (restricted to n components)
# ----------------------------------------------------------------------
def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    rev_a = a[::-1]
    full = _full_convolution_fft(rev_a, w)          # length 3n-2 when w has length 2n-1
    n = a.shape[0]
    return 2.0 * full[n - 1:n - 1 + n]               # slice the needed n components


# ----------------------------------------------------------------------
# Initial‑shape generators
# ----------------------------------------------------------------------
def _init_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.power(i + 1.0, p)
    a /= a.sum()
    return a


def _init_exponential(n: int, scale: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / scale)
    a /= a.sum()
    return a


def _init_double_bump(n: int, w: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[:w] = 1.0
    a[-w:] = 1.0
    a /= a.sum()
    return a


def _init_random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a non‑negative height sequence that (approximately) minimizes the
    evaluation function.  The algorithm runs for at most `budget_s` seconds,
    explores several initialisations and uses an Adam optimiser with a
    flatness regulariser.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin
    rng = np.random.default_rng(seed)

    # Choose a size that still fits in a single 16384‑point FFT for the self‑convolution.
    n = 8192

    # ------------------------------------------------------------------
    # Build a list of diverse starting vectors.
    # ------------------------------------------------------------------
    init_vectors = []

    # power‑law shapes (p around 0.5, the interesting exponent)
    for p in (0.45, 0.5, 0.55, 0.6):
        init_vectors.append(_init_power_law(n, p))

    # exponential decays with a few scales
    for scale in (n / 20.0, n / 10.0, n / 5.0):
        init_vectors.append(_init_exponential(n, scale))

    # two small blocks at the two ends
    w = max(1, n // 80)
    init_vectors.append(_init_double_bump(n, w))

    # a random point on the simplex
    init_vectors.append(_init_random_simplex(n, rng))

    # optionally reuse a previously found construction
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.shape[0] != n:
                if prev.shape[0] > n:
                    prev = prev[:n]
                else:
                    pad = np.zeros(n - prev.shape[0], dtype=np.float64)
                    prev = np.concatenate([prev, pad])
            prev = np.maximum(0.0, prev)
            if prev.sum() == 0.0:
                prev = rng.random(n)
            prev = prev / prev.sum()
            init_vectors.append(prev)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Global best tracking.
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None

    # Time allocated to each restart (even split of the remaining budget)
    total_remaining = deadline - time.time()
    per_restart = total_remaining / max(1, len(init_vectors))

    # ------------------------------------------------------------------
    # Optimisation routine for a single start vector.
    # ------------------------------------------------------------------
    def _optimise_one(start_a: np.ndarray, time_limit: float):
        nonlocal best_val, best_seq
        a = start_a.copy()
        n_local = a.shape[0]

        # Adam parameters
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        eta0 = 0.12

        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow, beta2_pow = 1.0, 1.0
        step = 0

        # soft‑max temperature for the max term
        eps_soft = max(1e-6, a.max() ** 2)
        eps_decay = 0.9995

        # weight of the flatness regulariser
        lam = 0.2
        lam_decay = 0.9997

        start = time.time()
        while time.time() < start + time_limit:
            step += 1

            # ---- current objective -------------------------------------------------
            conv = _full_convolution_fft(a, a)          # length 2n‑1
            max_conv = conv.max()
            score = 2.0 * n_local * max_conv

            if score < best_val - 1e-12:
                best_val = score
                best_seq = a.copy()
                print(f"[global best] new best={best_val:.12f}")

            # ---- soft‑max weights (smooth max) ------------------------------------
            w = np.exp((conv - max_conv) / eps_soft)
            w_sum = w.sum()
            if not np.isfinite(w_sum) or w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

            # ---- gradient of the (soft) max term ----------------------------------
            g_max = _grad_from_weights(a, w)

            # ---- flatness regulariser (encourage uniform convolution) ----------
            target = 1.0 / (2 * n_local - 1)            # average of conv when sum(a)=1
            err = conv - target                         # signed error vector
            # gradient of ½||err||² is 4 * rev(a) * err (see comment in analysis)
            g_flat = 4.0 * _grad_from_weights(a, err)

            # ---- combine gradients -------------------------------------------------
            g = g_max + lam * g_flat

            # ---- Adam update -------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            beta1_pow *= beta1
            beta2_pow *= beta2

            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            step_vec = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

            a_candidate = a - step_vec
            a = _project_simplex(a_candidate)

            # ---- scheduling --------------------------------------------------------
            eps_soft = max(eps_soft * eps_decay, 1e-12)
            lam = max(lam * lam_decay, 0.0)

            # ---- occasional perturbation to escape stagnation ----------------------
            if step % 3000 == 0:
                noise = rng.normal(scale=0.003, size=n_local)
                a = _project_simplex(a + noise)
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow = 1.0
                beta2_pow = 1.0

            # ---- renormalise every few thousand steps -------------------------------
            if step % 4000 == 0:
                a = a / a.sum()

        # end of this restart
        return

    # ------------------------------------------------------------------
    # Run the restarts sequentially, respecting the overall deadline.
    # ------------------------------------------------------------------
    for idx, vec in enumerate(init_vectors):
        now = time.time()
        time_left = deadline - now
        if time_left <= 0.0:
            break
        # give each restart at most `per_restart` seconds
        allocated = min(per_restart, time_left)
        print(f"[restart {idx + 1}/{len(init_vectors)}] allocated {allocated:.1f}s")
        _optimise_one(vec, allocated)

    # ------------------------------------------------------------------
    # Fallback in the (very unlikely) case that no sequence was ever stored.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Convert to plain Python floats, clip to the evaluator's limits and log.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:

        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")

    return final_seq
