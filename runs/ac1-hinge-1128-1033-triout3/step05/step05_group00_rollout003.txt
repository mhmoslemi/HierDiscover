    2 * n * max(conv(a, a)) / (sum(a) ** 2)

1.  A pool of cheap random initialisations (uniform, power‑law,
    triangular, Gaussian‑mixture, etc.).
2.  For each initialisation a short projected Adam optimisation that
    directly minimises the maximum entry of the convolution.
3.  Many independent restarts are run until the time budget expires;
    the best sequence seen is returned.

All auxiliary routines are defined at top level (no closures) and
the code respects the 1000 s budget.

import time

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real‑valued).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

# Random initial shape generator (cheap, diverse).
# ----------------------------------------------------------------------
def _init_shape(rng: np.random.Generator, n: int) -> np.ndarray:
    """Create a random non‑negative vector of length n and normalise it."""
    shape_type = rng.integers(0, 6)
    x = np.arange(n, dtype=float)

    if shape_type == 0:
        # uniform
        a = np.ones(n)
    elif shape_type == 1:
        # power‑law decreasing / increasing (p may be negative)
        p = rng.uniform(-2.5, 2.5)
        a = (x + 1) ** p
    elif shape_type == 2:
        # mirrored power‑law (large values near the right end)
        p = rng.uniform(-2.5, 2.5)
        a = (n - x) ** p
    elif shape_type == 3:
        # symmetric triangular (tent)
        a = np.minimum(x + 1, n - x)
    elif shape_type == 4:
        # mixture of a few classic smooth shapes
        sigma = rng.uniform(n / 12.0, n / 4.0)
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

        sine = np.sin(np.pi * (x + 0.5) / n) + 0.1

        triangle = np.minimum(x, n - 1 - x) + 1.0

        bases = [gauss, semicircle, sine, triangle]
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a = sum(w * b for w, b in zip(weights, bases))
    else:
        # random mask‑sparse vector
        mask = rng.choice([0, 1], size=n, p=[0.5, 0.5])
        a = rng.random(n) * mask
        if a.sum() == 0.0:
            a = rng.random(n)

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.ones(n)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimiser for a single start vector.
# ----------------------------------------------------------------------
def _run_adam_opt(a0: np.ndarray,
                  n: int,
                  max_seconds: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a short projected‑Adam optimisation starting from ``a0``.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.30                # base learning rate
    t = 0
    best_val = np.inf
    best_seq = a.copy()
    stagn = 0
    start = time.time()
    deadline = start + max_seconds

    while time.time() < deadline:
        t += 1
        # convolution and objective
        conv = _full_convolution_fft(a, a)                     # length 2n-1
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val - 1e-14:
            best_val = val
            best_seq = a.copy()
            stagn = 0
        else:
            stagn += 1

        # --- exact max sub‑gradient (average over active entries) ---
        tol = 1e-14
        active = np.flatnonzero(conv >= max_c - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        else:                               # fallback – should never happen
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        grad = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a = _project_simplex(a_candidate)

        # occasional random perturbation if we are stuck
        if stagn > 2500:
            noise = rng.normal(scale=0.004, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a non‑negative height sequence that (approximately) minimises
    the evaluation function.  The routine spends at most ``budget_s`` seconds,
    prints progress information and finally returns the best sequence it has
    discovered.
    """
    start_time = time.time()
    # leave a small safety margin for the final conversion / printout
    deadline = start_time + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # Problem size – you may change this, 6000 works well on a single CPU.
    n = 6000

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_val = np.inf
    best_seq = None

    # Helper to evaluate a vector quickly (no clipping, sum already = 1)
    def _objective(vec: np.ndarray) -> float:
        conv = _full_convolution_fft(vec, vec)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Seed the search with a few deterministic candidates (including the
    # sequence from a previous run if it exists).
    # ------------------------------------------------------------------
    initial_candidates = []

    # 1. previous solution, if available
    if 'height_sequence_1' in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=float)
            if arr.shape[0] != n:
                if arr.shape[0] > n:
                    arr = arr[:n]
                else:
                    arr = np.pad(arr, (0, n - arr.shape[0]), constant_values=0.0)
            arr = np.maximum(arr, 0.0)
            if arr.sum() > 0.0:
                initial_candidates.append(arr / arr.sum())
        except Exception:
            pass

    # 2. a plain uniform vector
    initial_candidates.append(np.ones(n) / n)

    # 3. a few random shapes right away (helps to find a decent baseline)
    for _ in range(4):
        initial_candidates.append(_init_shape(rng, n))

    # Evaluate the cheap starters
    for cand in initial_candidates:
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[init] value={best_val:.12f}")

    # ------------------------------------------------------------------
    # Multi‑restart optimisation loop.
    # ------------------------------------------------------------------
    run_id = 0
    while time.time() < deadline - 10.0:          # keep a few seconds for finalisation
        run_id += 1
        # allocate a modest slice of remaining time to the current run
        remaining = deadline - time.time()
        run_budget = min(25.0, remaining * 0.25)    # at most 25 s per restart

        # choose a random start – 80 % fresh, 20 % a perturbed current best
        if best_seq is not None and rng.random() < 0.2:
            a0 = best_seq.copy()
            a0 += rng.normal(scale=0.008, size=n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = _init_shape(rng, n)

        # run the local optimiser
        cand_seq, cand_val = _run_adam_opt(a0, n, run_budget, rng)

        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq.copy()
            print(f"[run {run_id:03d}] new global best={best_val:.12f}")

    # ------------------------------------------------------------------
    # Convert the best numpy array into the list format expected by
    # ``evaluate_sequence`` (clamped to the allowed numeric range).
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely – fall back to a uniform vector
        best_seq = np.ones(n) / n

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # optional logging of the final score (if evaluate_sequence is available)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")

    return final_seq
