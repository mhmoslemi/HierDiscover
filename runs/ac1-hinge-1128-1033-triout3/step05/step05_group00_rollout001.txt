def build_conv_matrix(f):
    n = f.shape[0]
    # Build rows as outer sum of arange
    i = np.arange(n)[:, None]  # shape n,1

"""
Propose a step‑function (non‑negative sequence) that approximately minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

by an **alternating linear‑programming (ALP) scheme**.

The idea follows the reference Frank‑Wolfe style iteration:

1.  Normalise the current vector `f` so that  Σ f = √(2 n).  
    With this normalisation the objective equals the maximal
    convolution entry `M = max(f*f)`.

2.  Solve the linear programme  

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   for all k = 0 … 2n‑2
                    b_j ≥ 0

    The constraint matrix is just the (full) convolution operator
    written in sparse form.  The solution `b` has a larger ℓ₁‑norm
    than `f` while respecting the same maximal convolution bound.

3.  Rescale `b` to the same ℓ₁‑norm as `f` (i.e. Σ b = √(2 n)),
    obtain `g`, and replace

        f ← (1‑t)·f + t·g      (t ∈ (0,1]).

    Because both `f` and `g` satisfy the bound `M`, the new vector
    never increases `M`.  Repeating the step drives `M` down.

The routine runs a number of ALP iterations (limited by the
remaining CPU time) and keeps the best vector seen.  If progress
stalls a small random perturbation is injected to avoid flat
plateaus.  Finally the best sequence is returned as a plain Python
list; it is automatically clipped to the range accepted by
`evaluate_sequence`.

The implementation uses only NumPy and SciPy (``scipy.optimize.linprog``
with the HiGHS solver and a sparse matrix representation of the
convolution operator).  The whole search respects the 1000 s budget.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper: builds the (2n‑1) × n convolution matrix for a given vector f.
# The matrix does NOT depend on the values of f, only on its length n,
# therefore the index arrays are built once and reused.
# ----------------------------------------------------------------------
def _conv_matrix_indices(n: int):
    """
    Returns (row_idx, col_idx) suitable for constructing the convolution
    matrix A where A @ b == np.convolve(f, b) (full) for any f of length n.
    """
    # outer sum of the row and column indices gives the convolution index.
    i = np.arange(n, dtype=np.int32)[:, None]   # shape (n,1)
    j = np.arange(n, dtype=np.int32)[None, :]   # shape (1,n)
    rows = (i + j).ravel()                     # length n*n
    cols = np.tile(np.arange(n, dtype=np.int32), n)  # repeated n times
    return rows, cols


def _build_conv_matrix(f: np.ndarray, rows: np.ndarray, cols: np.ndarray) -> sp.csr_matrix:
    """
    Construct the sparse matrix A (size (2n‑1) × n) representing
    the linear map b ↦ conv(f, b).  ``rows`` and ``cols`` are the index
    arrays produced by ``_conv_matrix_indices``.
    """
    data = np.tile(f, len(f))                     # each column receives the full vector f
    return sp.coo_matrix((data, (rows, cols)),
                         shape=(2 * len(f) - 1, len(f))).tocsr()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Searches for a low‑score step‑function within the given time budget.
    Returns the best sequence found as a list of plain Python floats.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # a tiny safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size.
    # We keep n moderate (≈ 1500–2000) so that the linear programme is
    # fast while still offering enough degrees of freedom.
    # ------------------------------------------------------------------
    n = 1700
    target_sum = np.sqrt(2.0 * n)                        # Σ f = √(2n)

    # ------------------------------------------------------------------
    # Initialise f.
    # If a previously discovered sequence exists, down‑sample it.
    # Otherwise start from a random smooth shape.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.shape[0] >= n:
                # simple block‑averaging down‑sampling
                block = prev.shape[0] // n
                trimmed = prev[:block * n]
                f = trimmed.reshape(n, block).mean(axis=1)
            else:
                pad = np.zeros(n - prev.shape[0], dtype=np.float64)
                f = np.concatenate([prev, pad])
        except Exception:
            f = None
    else:
        f = None

    if f is None or np.all(f == 0):
        # a smooth “semi‑circle” shape – good generic start
        x = np.linspace(-1.0, 1.0, n)
        f = np.sqrt(np.clip(1.0 - x * x, 0.0, None))

    # Normalise to the required ℓ₁ norm.
    f = np.maximum(f, 0.0)
    f = f / f.sum() * target_sum

    # Pre‑compute index arrays for the convolution matrix (they never change).
    rows_idx, cols_idx = _conv_matrix_indices(n)

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution discovered so far.
    # ------------------------------------------------------------------
    conv_f = np.convolve(f, f)
    M = conv_f.max()                         # with the chosen normalisation this equals the score
    best_f = f.copy()
    best_M = M
    print(f"[init] n={n}, Σf={f.sum():.6f}, M={M:.12f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – alternating linear programme + mixing.
    # ------------------------------------------------------------------
    iter_no = 0
    stagn = 0                # counts iterations without improvement
    t_mix = 0.5              # initial mixing weight, will decay slowly
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Build the constraint matrix for the current f.
        # ------------------------------------------------------------------
        A = _build_conv_matrix(f, rows_idx, cols_idx)

        # ------------------------------------------------------------------
        # Linear programme:
        # maximise 1ᵀb  (equivalently minimise -1ᵀb)
        # subject to A @ b ≤ M and b ≥ 0.
        # ------------------------------------------------------------------
        c = -np.ones(n)                                 # linprog performs minimisation
        b_ub = np.full(2 * n - 1, M)                    # RHS of the inequality constraints
        bounds = [(0.0, None) for _ in range(n)]

        # Solve – HiGHS is the default method for recent SciPy versions.
        res = opt.linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})

        if not res.success:
            # If the LP fails we fall back to a tiny random perturbation and continue.
            print(f"[iter {iter_no}] LP failed («{res.message}») – applying a perturbation.")
            f += rng.normal(scale=0.001 * target_sum, size=n)
            f = np.maximum(f, 0.0)
            f = f / f.sum() * target_sum
            stagn += 1
            continue

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            # Degenerate solution – break out.
            print(f"[iter {iter_no}] LP returned zero vector – terminating.")
            break

        # ------------------------------------------------------------------
        # Rescale b to have the same ℓ₁‑norm as f (Σ = √(2n)).
        # ------------------------------------------------------------------
        g = b * (target_sum / sum_b)

        # ------------------------------------------------------------------
        # Convex mixing: f ← (1‑t) f + t g.
        # The mixing weight slowly decays to keep the updates stable.
        # ------------------------------------------------------------------
        f_new = (1.0 - t_mix) * f + t_mix * g
        # Numerical safety: enforce non‑negativity and exact normalisation.
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum() * target_sum

        # ------------------------------------------------------------------
        # Evaluate the new maximal convolution entry.
        # ------------------------------------------------------------------
        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()

        # By construction M_new ≤ M (modulo floating‑point noise).  Keep the best.
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f_new.copy()
            stagn = 0
            print(f"[iter {iter_no:5d}] new best M={best_M:.12f}")
        else:
            stagn += 1

        # Prepare for next iteration.
        f = f_new
        M = M_new
        iter_no += 1
        t_mix = max(0.02, t_mix * 0.995)          # gentle decay

        # ------------------------------------------------------------------
        # If we have been stuck for a while, inject a small random jitter.
        # ------------------------------------------------------------------
        if stagn >= 30:
            jitter = rng.normal(scale=0.0005 * target_sum, size=n)
            f += jitter
            f = np.maximum(f, 0.0)
            f = f / f.sum() * target_sum
            stagn = 0
            print(f"[iter {iter_no}] jitter applied (stagnation).")

        # ------------------------------------------------------------------
        # Periodic sanity check – avoid exceeding the time budget.
        # ------------------------------------------------------------------
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Convert the best vector to a plain Python list, clip to the range
    # accepted by ``evaluate_sequence`` and report the final score.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_f.tolist()]

    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        # ``evaluate_sequence`` should be available in the execution environment.
        print("[result] Unable to compute final_score (evaluate_sequence missing).")

    return final_seq
