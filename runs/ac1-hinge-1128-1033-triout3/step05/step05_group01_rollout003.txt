Pseudo:

def simplex_proj(v):
    # v is any vector; project onto {w >=0, sum w =1}
    # Using sort and find threshold theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    """Project vector v onto the probability simplex {x >=0, sum x =1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho = max i s.t. u_i > cssv_i / i
    condition = u - cssv / ind > 0
    if not condition.any():
        # all non-positive
        # return uniform distribution?
        return np.full(n, 1.0/n)
    rho = ind[condition].max()
    theta = cssv[condition].sum() / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_descent(a, deadline, rng):
    n = a.shape[0]
    cur_conv = _full_convolution(a)
    cur_max = cur_conv.max()
    cur_val = 2 * n * cur_max
    iteration = 0
    no_improve = 0
    # parameters
    step_factor = 1.0
    while time.time() < deadline and no_improve < 2000:
        # compute subgradient for kmax
        k = int(np.flatnonzero(np.isclose(cur_conv, cur_max, atol=1e-15))[0])
        # compute g
        g = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        # set gradient
        g[i_range] = 2 * a[j_range]  # for each i in i_range, derivative = 2 * a_j
        # compute step size parameter based on norm
        norm_g2 = np.dot(g, g)
        if norm_g2 == 0:
            break
        # base step
        eta = step_factor * cur_max / (norm_g2 + 1e-12)
        # try line search: 
        improved = False
        for _ in range(10):
            a_candidate = a - eta * g
            # project onto simplex
            a_candidate = project_to_simplex(a_candidate)
            conv_candidate = _full_convolution(a_candidate)
            max_candidate = conv_candidate.max()
            if max_candidate < cur_max - 1e-15:  # improvement
                a = a_candidate
                cur_conv = conv_candidate
                cur_max = max_candidate
                cur_val = 2 * n * cur_max

            step_factor = max(step_factor * 0.7, 1e-4)
            no_improve += 1
        iteration += 1
    return a, cur_val

``evaluate_sequence``.  

The algorithm works in three stages:

1. **Initialisation** – load the best known sequence (global
   ``height_sequence_1``) or start from a uniform distribution.
   The vector is normalised to have ℓ₁‑norm equal to 1.

2. **Projected sub‑gradient descent** – the objective

3. **Final verification** – the best vector found is evaluated with
   the supplied ``evaluate_sequence`` function and returned.

# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        # v is all non‑positive – return the uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
