"""
Propose a non‑negative sequence that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using three stages:

1.  Exponentiated‑gradient descent with a soft‑max weighting of the
    convolution entries.  The temperature is annealed aggressively,
    the step size decays, and the iterate is projected onto the
    probability simplex after each multiplicative update.

2.  Adam‑style sub‑gradient descent on the true ‑∞‑norm (active set
    weighting).  This refines the solution obtained by the first stage.

3.  Simple hill‑climbing: repeatedly move a tiny amount of mass from a
    random index to another and keep the change only if it strictly
    improves the objective.  This local search can capture the tiny
    improvements needed to beat the previous best (≈1.5115).

All three stages respect the 1000 s budget, log progress, and finally
return the best sequence seen (clipped to the range required by
`evaluate_sequence`).  The best sequence is stored in the global
`height_sequence_1` variable for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Duchi et al. (2008) – O(n) simplex projection."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b, full length, using a power‑of‑2 FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

    start_time = time.time()
    # leave a small safety margin for returning the result
    deadline = start_time + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a bit larger than the previous run (still cheap)
    # ------------------------------------------------------------------
    n = 8000                     # length of the step function

    # ------------------------------------------------------------------
    # Initialisation – reuse a previous solution if available,
    # otherwise build a smooth random mixture.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] != n:
                if base.shape[0] > n:
                    base = base[:n]
                else:
                    base = np.concatenate([base,
                                            np.zeros(n - base.shape[0],
                                                     dtype=np.float64)])
            a = np.maximum(0.0, base)
            if a.sum() == 0.0:
                a = rng.random(n)
            a = a / a.sum()
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # start from a blend of a few classic shapes
        x = np.arange(n, dtype=np.float64)

        # semi‑circle (positive)
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

        # raised cosine (always positive)
        cosine = 0.5 * (1.0 + np.cos(np.pi * (x + 0.5) / n))

        # Gaussian centred in the middle
        sigma = n / 5.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

        # simple triangle
        triangle = np.minimum(x, n - 1 - x) + 1.0

        bases = [semicircle, cosine, gauss, triangle]
        w = rng.random(len(bases))
        w /= w.sum()
        a = sum(wi * bi for wi, bi in zip(w, bases))
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution seen so far
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    sum_a = a.sum()
    best_val = 2.0 * n * max_conv / (sum_a ** 2)
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient with soft‑max weighting
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.38 * budget_s          # ~38 % of the budget
    eta = 0.6                                          # step‑size for exp‑grad
    eps = max_conv * 0.8                               # temperature for soft‑max
    eps_min = max(max_conv * 1e-5, 1e-14)
    eps_decay = 0.9997
    eta_min = 1e-5
    eta_decay = 0.9999
    stagn = 0

    while time.time() < phase1_end:
        # convolution & current score
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        sum_a = a.sum()
        score = 2.0 * n * max_conv / (sum_a ** 2)

        # record best
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phase1] new_best={best_val:.12f}  eta={eta:.2e}  eps={eps:.2e}")
        else:
            stagn += 1

        # soft‑max weights over the full convolution
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the (smoothed) max‑conv
        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # multiplicative (exponentiated) update
        log_a = np.log(np.maximum(a, 1e-16))
        log_a -= eta * g
        a = np.exp(log_a)

        # project back onto the simplex
        a = _project_simplex(a)

        # decay schedules
        eta = max(eta_min, eta * eta_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional random perturbation if we are stuck
        if stagn > 3000:
            delta = rng.normal(scale=0.003, size=n)
            a = a + delta
            a = _project_simplex(a)
            stagn = 0
            eta = max(eta_min, eta * 1.2)
            print("[phase1] perturbation injected")

    # ------------------------------------------------------------------
    # Phase 2 – exact max sub‑gradient descent with Adam
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    eta0 = 0.12                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = 1.0, 1.0
    t_adam = 0
    stagn = 0

    while time.time() < deadline:
        t_adam += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        sum_a = a.sum()
        score = 2.0 * n * max_conv / (sum_a ** 2)

        # keep global best
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phase2] iter {t_adam:6d} new_best={best_val:.12f}")
        else:
            stagn += 1

        # exact max sub‑gradient – average over all active entries
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)
        a = a_candidate

        # Escape flat regions
        if stagn > 5000:
            delta = rng.normal(scale=0.0025, size=n)
            a = a + delta
            a = _project_simplex(a)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            print(f"[phase2] perturbation at iter {t_adam}")

        # occasional renormalisation guard
        if t_adam % 4000 == 0:
            a = a / a.sum()

        # ------------------------------------------------------------------
        # Tiny hill‑climbing after every 2000 Adam steps (still within budget)
        # ------------------------------------------------------------------
        if t_adam % 2000 == 0 and time.time() < deadline - 5.0:
            # try a few random mass transfers
            for _ in range(1200):
                i = rng.integers(n)
                j = rng.integers(n)
                if i == j:
                    continue
                # move a very small amount
                delta = min(a[i], 0.001) * rng.random()
                if delta <= 0:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                conv_new = _full_convolution_fft(a_new, a_new)
                max_conv_new = conv_new.max()
                score_new = 2.0 * n * max_conv_new / (a_new.sum() ** 2)
                if score_new < best_val - 1e-12:
                    a = a_new
                    best_val = score_new
                    best_seq = a.copy()
                    print(f"[local] new_best={best_val:.12f}  Δ={delta:.2e} (i={i},j={j})")
                    break   # accept the first improvement found

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits required by
    # `evaluate_sequence`).  Also expose it in the global namespace so
    # that a future run can reuse it as a warm start.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    globals()['height_sequence_1'] = final_seq   # make reusable

    try:
        # sanity check – optional, does not affect the returned value
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")

    return final_seq
