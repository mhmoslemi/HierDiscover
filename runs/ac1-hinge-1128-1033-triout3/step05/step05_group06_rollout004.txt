def local_refine(a_start, time_limit):
    start = time.time()
    a = a_start.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = score_vector(a)
    # schedule parameters
    delta_initial = 0.01  # up to 1% mass per move
    delta_final = 0.0002
    temp_initial = best_score * 0.001
    temp_final = best_score * 1e-7
    iteration = 0
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        delta = delta_initial * (1 - t_frac) + delta_final * t_frac
        temp = temp_initial * (1 - t_frac) + temp_final * t_frac

        # compute convolution, find max index
        conv = full_convolution_fft(a)
        max_b = conv.max()
        k_max = int(conv.argmax())

        # compute product contributions
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k_max - i_range
        w = a[i_range] * a[j_range]

        # choose source and destination
        src_idx = i_range[w.argmax()]
        # destination: choose index with minimal product or random remote
        # use minimal w as dst proxy, but if src==dst or w.min() is zero
        dst_candidates = np.where(w == w.min())[0]
        # choose random among minimal
        dst_local_idx = i_range[np.random.choice(dst_candidates)]
        # ensure src != dst
        if src_idx == dst_local_idx:
            # pick random other index
            dst_local_idx = np.random.randint(0, n)

        # compute amount to move
        max_move = min(a[src_idx], delta)
        if max_move <= 0:
            continue
        move = max_move * np.random.rand()
        if move <= 0:
            continue

        a_new = a.copy()
        a_new[src_idx] -= move
        a_new[dst_local_idx] += move

        # compute new score
        new_score = score_vector(a_new)

        # acceptance
        if new_score < best_score:
            a = a_new
            best_score = new_score
            best_a = a_new.copy()
        else:
            # probabilistic acceptance
            if temp > 0 and np.random.rand() < np.exp((best_score - new_score) / temp):
                a = a_new
        iteration += 1
    return best_a, best_score

"""
Hybrid optimisation for the “max auto‑convolution” step‑function problem.

The evaluation function is scale‑invariant, therefore we work with probability
vectors (∑a_i = 1, a_i ≥ 0) and minimise

    F(a) = 2·n·max_k (a * a)[k] .

Two complementary phases are used:

1. **Projected Adam** on the log‑probabilities (softmax) – a fast first‑order
   method that follows the sub‑gradient of the active convolution peaks.

2. **Targeted local refinement** – a cheap simulated‑annealing style search.
   At each iteration the index k where the auto‑convolution attains its
   maximum is identified; mass is shifted from the index that contributes most
   to that peak towards the index that contributes least.  The move size and
   acceptance temperature are continuously cooled, allowing fine‑grained
   improvements beyond the Adam phase.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, length 2·n‑1)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear (full) convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 2 * n                     # zero‑padding length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]        # keep the true length 2n‑1


def score_vector(a: np.ndarray) -> float:
    """
    Objective value for a probability vector `a` (∑a = 1).
    Equivalent to evaluate_sequence([...]) but much faster.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families – useful for deterministic seeds
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Softmax (projection onto the probability simplex)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Gradient of the max‑convolution w.r.t. log‑probabilities (sub‑gradient)
# ----------------------------------------------------------------------
def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)[k]   with respect to the log‑parameters `x`,
    where a = softmax(x).  The derivation follows the comment in the
    original implementation.
    """
    n = a.shape[0]
    G = np.zeros_like(a)

    # Accumulate contributions for each active convolution index
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]

    # Sub‑gradient for the log‑parameters
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad


# ----------------------------------------------------------------------
# Projected Adam optimiser on the log‑probabilities
# ----------------------------------------------------------------------
def run_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Runs Adam (in the log‑space) for at most `time_limit` seconds.
    Returns the best probability vector seen and its score.
    """
    start = time.time()
    n = initial_a.shape[0]

    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.2

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    # main loop ---------------------------------------------------------
    while time.time() - start < time_limit - 0.5:
        iteration += 1

        # softmax → probability vector
        a = softmax(x)

        # convolution and active set
        conv = full_convolution_fft(a)
        max_b = conv.max()
        tol = max_b * 1e-12 + 1e-15
        active = np.where(conv >= max_b - tol)[0]

        # gradient
        grad = grad_log_params(x, a, conv, max_b, active)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # bookkeeping
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Adam iter {iteration:6d}: new best = {best_score:.6f}")

        # re‑heat if stuck
        if iteration - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Adam iter {iteration:6d}: re‑heat (no improvement)")

    # safety – return something sensible even if nothing improved
    if best_vec is None:
        best_vec = softmax(x)
        best_score = 2.0 * n * full_convolution_fft(best_vec).max()
    return best_vec, best_score


# ----------------------------------------------------------------------
# Targeted local refinement (simulated annealing / hill climbing)
# ----------------------------------------------------------------------
def local_refine(start_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Small‑scale, temperature‑controlled search that moves probability mass
    away from the indices contributing most to the current maximal convolution
    value.  Runs for at most `time_limit` seconds.
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_score = score_vector(a)

    # schedule parameters
    delta_start = 0.01      # up to 1 % of total mass per move initially
    delta_end = 0.0002      # final move size
    temp_start = best_score * 0.001
    temp_end = best_score * 1e-7

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        elapsed = time.time() - start
        frac = elapsed / time_limit

        # linearly cool move size and temperature
        delta = delta_start * (1 - frac) + delta_end * frac
        temp = temp_start * (1 - frac) + temp_end * frac

        # current convolution & its maximal index
        conv = full_convolution_fft(a)
        max_b = conv.max()
        k_max = int(conv.argmax())

        # contribution of each index to the current peak
        i0 = max(0, k_max - (n - 1))
        i1 = min(k_max, n - 1)
        i_range = np.arange(i0, i1 + 1, dtype=np.int64)
        j_range = k_max - i_range
        contrib = a[i_range] * a[j_range]

        # source = index with biggest contribution
        src_local = i_range[np.argmax(contrib)]
        # destination = index with smallest contribution (may be the same)
        dst_local = i_range[np.argmin(contrib)]

        # if source and destination coincide, pick a random other index
        if src_local == dst_local:
            dst_local = np.random.randint(0, n)

        # amount of mass to relocate
        max_move = min(a[src_local], delta)
        if max_move <= 0:
            continue
        move = max_move * np.random.rand()
        if move <= 0:
            continue

        a_new = a.copy()
        a_new[src_local] -= move
        a_new[dst_local] += move

        new_score = score_vector(a_new)
        # accept if better, otherwise according to Metropolis rule
        if new_score < best_score:
            a = a_new
            best_a = a_new.copy()
            best_score = new_score
        else:
            # probabilistic uphill move
            if temp > 0 and np.random.rand() < np.exp((best_score - new_score) / temp):
                a = a_new

    return best_a, best_score

        best_sc = np.inf

            if sc < best_sc:

        external_vec = internal_vec

    # --------------------------------------------------------------

    print(f"\n=== Search completed ===")

    # Return as plain Python list
    return final_vec.tolist()
