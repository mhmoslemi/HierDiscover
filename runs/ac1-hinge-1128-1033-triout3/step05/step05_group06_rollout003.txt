"""
Propose a step‑function using a smoothed max‑convolution objective.

The evaluation is invariant to scaling, so we work with a probability vector
a (a_i ≥ 0, Σ a_i = 1).  The true objective is
    f(a) = 2·n·max_k (a * a)[k] .
Direct sub‑gradient descent (as in the previous solution) can get stuck,
because the max operator discards information from near‑maximal entries.
We therefore optimise a *smooth* approximation of the max:
    g(a) = 2·n·∑_k w_k·(a*a)[k] ,
with w_k = softmax(α·(a*a)[k]).  When α is small the landscape is smooth
and easy to explore, while for large α it approaches the true max.
The algorithm consists of three phases:

1. **Parameter sweep:**  Scan a few analytic families (power‑law,
   product‑law, exponential) over a grid of shape parameters to obtain
   strong deterministic seeds.

2. **Smoothed Adam optimisation:**  Starting from each seed we run Adam
   on the log‑probabilities x (a = softmax(x)).  The gradient of the
   smoothed objective is computed analytically using FFT‑based convolutions.
   The smoothing parameter α is increased linearly from a low value
   (≈ 5) to a high value (≈ 100) as the allotted time elapses, allowing
   the optimiser to first explore globally and then refine the true max.

3. **Final local refinement:**  If time permits we perform a cheap
   stochastic hill‑climb on the best vector found.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence (as a plain Python list) that minimises the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, arbitrary lengths)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution of two real vectors using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    n = len(a) + len(b) - 1
    size = next_pow2(n)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    c = np.fft.irfft(A * B, n=size)
    return c[:n]


def full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Convolve a with itself (shortcut)."""
    return full_convolution(a, a)


# ----------------------------------------------------------------------
# Softmax (log‑parameterisation)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Objective (the exact value returned by evaluate_sequence for sum(a)=1)
# ----------------------------------------------------------------------
def true_score(a: np.ndarray) -> float:
    """2·n·max(conv) for a normalised probability vector."""
    n = a.shape[0]
    conv = full_convolution_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (used as deterministic seeds)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution objective
# ----------------------------------------------------------------------
def smoothed_grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                             alpha: float) -> np.ndarray:
    """
    Compute the gradient of 2·n·smooth_max(conv) w.r.t. log‑parameters x.
    a = softmax(x) is the underlying probability vector.
    conv = full_convolution_self(a)  (length 2n‑1)
    alpha is the smoothing factor.
    """
    n = a.shape[0]

    # ---- smooth weights w_k = softmax(alpha * conv) ----
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    # subtract max for numerical stability
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

    # ---- correlation term:  Σ_k w_k a_{k‑j} for each j ----
    a_rev = a[::-1]                           # reversed a, length n
    corr_full = full_convolution(w, a_rev)    # length 3n‑2
    # slice to obtain length‑n vector of correlations
    corr = corr_full[n - 1 : n - 1 + n]        # corr[j] = Σ_k w_k a_{k‑j}

    # ---- gradient wrt a (includes factor 4·n) ----
    grad_a = 4.0 * n * corr                    # d f / d a

    # ---- chain rule for softmax (grad wrt logits x) ----
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)               # ∂f/∂x_i = a_i (grad_a_i – Σ a·grad_a)

    return grad_x


# ----------------------------------------------------------------------
# Adam optimiser on log‑probabilities with annealed smoothing
# ----------------------------------------------------------------------
def run_smooth_adam(initial_a: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best probability vector found and its true (unsmoothed) score.
    """
    start = time.time()
    n = initial_a.shape[0]
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.25

    best_vec = None
    best_score = np.inf
    last_improve_iter = 0
    iteration = 0

    # smoothing schedule: linearly increase α from 5 to 120
    alpha_min = 5.0
    alpha_max = 120.0

    while True:
        now = time.time()
        if now - start > time_budget - 0.5:
            break
        iteration += 1

        # ---- probability vector ----
        a = softmax(x)

        # ---- convolution and true score (for bookkeeping) ----
        conv = full_convolution_self(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve_iter = iteration
            print(f"[Adam] iter {iteration:6d}: new best true score = {best_score:.6f}")

        # ---- smoothing parameter for this iteration ----
        frac = min(1.0, (now - start) / time_budget)
        alpha = alpha_min + (alpha_max - alpha_min) * frac

        # ---- gradient wrt logits ----
        grad = smoothed_grad_log_params(x, a, conv, alpha)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of base learning rate
        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- re‑heat if stuck for many iterations ----
        if iteration - last_improve_iter > 6000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve_iter = iteration
            print(f"[Adam] iter {iteration:6d}: re‑heat (no improvement)")

    # Safety fallback
    if best_vec is None:
        best_vec = np.ones(n) / n
        best_score = true_score(best_vec)

    return best_vec, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing refinement (optional)
# ----------------------------------------------------------------------
def local_refine(a: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Very cheap random neighbourhood search.
    Returns an improved vector (if any) and its true score.
    """
    start = time.time()
    n = a.shape[0]
    best = a.copy()
    best_score = true_score(best)

    while time.time() - start < time_budget:
        i = np.random.randint(n)
        # propose a small perturbation (up to ±2 % of the current value)
        delta = (np.random.rand() - 0.5) * 0.04 * best[i]
        candidate = best.copy()
        candidate[i] = max(0.0, candidate[i] + delta)
        # renormalise
        s = candidate.sum()
        if s <= 0.0:
            continue
        candidate /= s
        sc = true_score(candidate)
        if sc < best_score:
            best = candidate
            best_score = sc
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the time budget, prints progress,
    and finally returns the best vector found as a plain Python list.
    """
    # --------------------------------------------------------------
    # 0. General set‑up
    # --------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    # --------------------------------------------------------------
    # 1. Determine sequence length (5000 works well, we keep it fixed)
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # 2. Try to use an externally supplied seed (height_sequence_1)
    # --------------------------------------------------------------
    external_vec = None
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        s = external.sum()
        if s > 0.0:
            external /= s
            n = external.shape[0]
            external_vec = external
            print(f"[Init] Using external seed of length {n}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3. Deterministic parameter sweep to obtain strong seeds
    # --------------------------------------------------------------
    # We keep the length fixed to n (unless an external seed forces another n)
    param_seeds = []

    # power‑law on distance to the closest end
    p_grid = np.linspace(0.2, 3.5, 34)   # 34 values give a good spread
    best_score = np.inf
    best_vec = None
    for p in p_grid:
        vec = u_min_distribution(n, p)
        sc = true_score(vec)
        if sc < best_score:
            best_score, best_vec = sc, vec
    param_seeds.append(best_vec)
    print(f"[Sweep] u_min best p≈{p_grid[np.argmin([true_score(u_min_distribution(n, p)) for p in p_grid])]:.3f} → score {best_score:.6f}")

    # product‑law on i·(n‑i)
    best_score = np.inf
    best_vec = None
    for p in p_grid:
        vec = u_prod_distribution(n, p)
        sc = true_score(vec)
        if sc < best_score:
            best_score, best_vec = sc, vec
    param_seeds.append(best_vec)
    print(f"[Sweep] u_prod best p≈{p_grid[np.argmin([true_score(u_prod_distribution(n, p)) for p in p_grid])]:.3f} → score {best_score:.6f}")

    # exponential decay
    lam_grid = np.logspace(-4, -1, 34)   # from 1e‑4 to 1e‑1
    best_score = np.inf
    best_vec = None
    for lam in lam_grid:
        vec = exp_distribution(n, lam)
        sc = true_score(vec)
        if sc < best_score:
            best_score, best_vec = sc, vec
    param_seeds.append(best_vec)
    print(f"[Sweep] exp best λ≈{lam_grid[np.argmin([true_score(exp_distribution(n, lam)) for lam in lam_grid])]:.5f} → score {best_score:.6f}")

    # Uniform distribution (as a baseline)
    uniform = np.ones(n) / n
    param_seeds.append(uniform)

    # --------------------------------------------------------------
    # 4. Assemble the portfolio of starting points
    # --------------------------------------------------------------
    seeds = []

    # External seed (if any) is tried first
    if external_vec is not None:
        seeds.append(external_vec)

    # Deterministic family seeds
    seeds.extend(param_seeds)

    # A few random Dirichlet draws for diversification
    for _ in range(4):
        rnd = np.random.rand(n)
        seeds.append(rnd / rnd.sum())

    # A couple of handcrafted extreme shapes
    for p in [0.4, 2.2]:
        seeds.append(u_min_distribution(n, p))
        seeds.append(u_prod_distribution(n, p))
    for lam in [2e-4, 5e-3]:
        seeds.append(exp_distribution(n, lam))

    # --------------------------------------------------------------
    # 5. Optimisation loop (smooth Adam) – allocate time per seed
    # --------------------------------------------------------------
    best_overall_vec = None
    best_overall_score = np.inf

    remaining_time = deadline - time.time()
    for idx, seed_vec in enumerate(seeds):
        # Normalise seed to a proper probability vector
        a0 = np.maximum(seed_vec, 0.0)
        if a0.sum() == 0.0:
            a0 = np.ones_like(a0)
        a0 = a0 / a0.sum()

        # Time budget for this seed (share remaining time evenly)
        remaining_time = deadline - time.time()
        if remaining_time <= 2.0:
            break
        per_seed_budget = remaining_time / (len(seeds) - idx)

        print(f"\n=== Optimisation start: seed {idx+1}/{len(seeds)} (budget {per_seed_budget:.1f}s) ===")
        vec, sc = run_smooth_adam(a0, per_seed_budget * 0.85)  # keep a bit for final refinement

        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_vec = vec.copy()
            print(f"*** New global best from seed {idx+1}: score = {best_overall_score:.6f}")

        # Optional quick local refinement (uses the leftover ~15 % of per‑seed budget)
        leftover = per_seed_budget - (deadline - time.time())
        if leftover > 0.2:
            vec_ref, sc_ref = local_refine(vec, leftover)
            if sc_ref < best_overall_score:
                best_overall_score = sc_ref
                best_overall_vec = vec_ref.copy()
                print(f"*** Global best improved by local refinement: {best_overall_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final global refinement if time permits
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_overall_vec is not None:
        print(f"\n=== Final global refinement (remaining {remaining:.1f}s) ===")
        vec, sc = run_smooth_adam(best_overall_vec, remaining * 0.7)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_vec = vec.copy()
            print(f"*** Global best after final refinement: {best_overall_score:.6f}")
        # a tiny hill‑climb on the very end
        vec_ref, sc_ref = local_refine(best_overall_vec, remaining * 0.3)
        if sc_ref < best_overall_score:
            best_overall_score = sc_ref
            best_overall_vec = vec_ref.copy()
            print(f"*** Global best after final local climb: {best_overall_score:.6f}")

    # --------------------------------------------------------------
    # 7. Report and return
    # --------------------------------------------------------------
    if best_overall_vec is None:
        # fallback – uniform distribution
        best_overall_vec = np.ones(n) / n
        best_overall_score = true_score(best_overall_vec)

    print("\n=== Search finished ===")
    print(f"Best true score found: {best_overall_score:.6f} (length {len(best_overall_vec)})")
    return best_overall_vec.tolist()
