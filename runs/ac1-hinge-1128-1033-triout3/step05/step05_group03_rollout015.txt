def hillclimb_targeted_improved(a_start, n, deadline):
    a = a_start.copy()
    b = convolve(a,a)
    cur_val = 2.0*n*b.max()
    best_a = a.copy()
    best_val = cur_val
    no_improve = 0
    max_frac_start=0.5
    max_frac_end=0.01
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        # find current peak(s)
        k_max = int(b.argmax())
        # compute contributions for each i
        # contribution[i] = a[i]*a[k_max-i] if 0<=k_max-i<n else 0
        rev_idx = k_max - np.arange(n)

        contrib[mask] = a[mask] * a[rev_idx[mask]]
        total_contrib = contrib.sum()

            i = np.random.choice(n, p=source_probs)

            i = np.argmax(a)

        # Compute a_rev[i] = a[k_max - i].

import time
import numpy as np

def _full_convolution(x):
    n = x.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx*fx, size)[:out_len]
    return conv

def _conv_fft(x, y):
    n, m = x.shape[0], y.shape[0]
    out_len = n+m-1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx*fy, size)[:out_len]
    return conv

def _objective(a):
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())
```

def _hillclimb_phase(a_start, n, start, deadline, best_a, best_val, target_val=1.503):
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a_local = best_a.copy()
    best_val_local = best_val
    arange_n = np.arange(n)
    max_frac_start, max_frac_end = 0.5, 0.01  # more aggressive final
    IMPROVE_LIMIT = 300_000
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # Find current max index
        k_max = int(np.argmax(b))

        # contributions to current max
        rev_idx = k_max - arange_n
        mask = (rev_idx >= 0) & (rev_idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[rev_idx[mask]]
        total = contrib.sum()
        if total > 0:
            i = np.random.choice(n, p=contrib/total)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # target j minimising contribution to k_max
        rev_target = np.full(n, np.inf)
        rev_target[mask] = a[rev_idx[mask]]   # contributions if moved to j
        rev_target[i] = np.inf
        # also accept out-of-range indices as zero contribution
        # for indices where rev_target == inf (i.e., not mask), set to 0
        out_of_range = ~mask
        rev_target[out_of_range] = 0.0

        min_val = rev_target.min()
        # choose among all minimal indices
        candidates = np.where(rev_target == min_val)[0]
        if candidates.size > 1:
            j = np.random.choice(candidates)
        else:
            j = candidates[0]

        # delta mass
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # simulate new b
        b_new = b.copy()
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val_local:
                best_val_local = new_val
                best_a_local = a.copy()
                print(f"[HillImproved {iteration:7d}] new best = {best_val_local:.6f}")
                if best_val_local <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb
            a = best_a_local.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillImproved {iteration:7d}] perturb after stagnation")
        iteration += 1

    return best_a_local
```

"""find_step_function.py

A two‑phase optimiser for the step‑function evaluation problem.

Phase A – Adam on a smooth approximation of `max(conv(a,a))`.
          The vector is stored as soft‑max logits so that Σa = 1, a ≥ 0.
          A log‑sum‑exp with an exponentially decaying temperature ε
          provides a differentiable proxy for the max.  Adam with a
          decaying learning‑rate and occasional random jitter drives
          the search, while stagnation triggers a restart.

Phase B – Deterministic hill‑climbing on the exact objective.
          At each move a small amount of mass is taken from a source
          index that contributes heavily to the current maximal
          convolution entry and is transferred to a target index that
          contributes minimally (or not at all) to that entry.
          The convolution is updated in O(n) and only strictly
          improving moves are kept.  After a long period without
          improvement the solution is perturbed to escape local minima.

The routine `propose_candidate` respects the 1000 s time budget,
returns a non‑negative list of heights (Σ≈1) and prints the best
objective value observed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) when Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation of a log‑sum‑exp surrogate for the max."""
    # logits → soft‑max (guarantees Σa = 1, a ≥ 0)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.50, 0.001          # start large, decay to tiny
    eps_start, eps_end = 1e-1, 1e-12        # smoothing temperature

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 25_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- a from soft‑max logits ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ---- exact objective (tracking) ----
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve_cnt = 0
            print(f"[Adam   {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve_cnt += 1

        # ---- schedule for ε and learning‑rate ----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---- smooth max via log‑sum‑exp ----
        b_full = _full_convolution(a)          # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)              # safe in double precision
        p = weights / weights.sum()            # p ≈ one‑hot when eps tiny

        # ---- gradient of smooth objective w.r.t. a ----
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]     # the n entries needed
        grad_a = 4.0 * n * segment[::-1]       # shape (n,)

        # ---- back‑prop through the soft‑max ----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ---- occasional random jitter (helps escape plateaus) ----
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.005, size=n)

        # ---- restart if stagnated ----
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam   {iteration:7d}] restart after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> np.ndarray:
    """
    Local search that moves a small amount of mass from a “source”
    index (chosen proportionally to its contribution to the current
    maximal convolution entry) to a “target” index that contributes
    as little as possible to that entry.
    """
    a = a_start.copy()
    b = _full_convolution(a)                # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # parameters for the mass‑transfer moves
    max_frac_start, max_frac_end = 0.5, 0.01
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0
    arange_n = np.arange(n)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress‑based schedule for the allowed transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # ---- locate the current maximal convolution entry ----
        k_max = int(np.argmax(b))

        # ---- contributions of each index to b[k_max] ----
        rev_idx = k_max - arange_n
        mask = (rev_idx >= 0) & (rev_idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[rev_idx[mask]]
        total_contrib = contrib.sum()

        # ---- choose a source index (biased towards large contributions) ----
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # ---- choose a target index that minimises the contribution to k_max ----
        rev_target = np.full(n, np.inf)               # start with “infinite” cost
        # indices out of range give zero contribution → best possible
        rev_target[~mask] = 0.0
        rev_target[mask] = a[rev_idx[mask]]
        rev_target[i] = np.inf                         # never move to the source itself
        min_val = rev_target.min()
        cand_js = np.where(rev_target == min_val)[0]
        j = np.random.choice(cand_js) if cand_js.size > 1 else cand_js[0]

        # ---- amount of mass to move ----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental O(n) update of the convolution (no full copy) ----
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- accept only strictly improving moves ----
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill   {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ---- stagnation handling ----
        if no_improve > IMPROVE_LIMIT:
            # Re‑initialise around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill   {iteration:7d}] perturb after stagnation")
        iteration += 1

    return best_a
