"""
A simulated‑annealing / hill‑climbing optimiser for the non‑negative
step‑function problem.

* Randomly generate many sparse candidates (support 30‑500) and keep the
  best one.
* Starting from the best candidate run a long simulated‑annealing
  search that repeatedly transfers a small amount of mass from a “source”
  index (chosen proportionally to its contribution to the current
  maximal convolution entry) to a random “target” index.
* The convolution is updated in O(n) using the exact rank‑one update
  formula, so millions of moves can be performed within the 1000 s
  budget.
* A temperature schedule allows occasional non‑improving moves,
  helping to escape local minima.
* After the time limit the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution of a with itself using a pre‑computed FFT size."""
    # a is real, length n; result length 2*n-1
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * a.shape[0] - 1]
    return conv


def _objective_fft(a: np.ndarray, size_fft: int) -> float:
    """Exact value of the evaluation function (sum(a) is assumed > 0)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, size_fft)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Random candidate generation (sparse support)
# ----------------------------------------------------------------------
def _random_candidate(n: int,
                      min_k: int = 30,
                      max_k: int = 500) -> np.ndarray:
    """Generate a random non‑negative vector of length n with a small support."""
    k = np.random.randint(min_k, max_k + 1)
    idx = np.random.choice(n, k, replace=False)
    # use exponential weights – they are always positive and more diverse
    w = np.random.exponential(scale=1.0, size=k)
    w /= w.sum()                     # Σw = 1
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a

                     **kwargs) -> list[float]:

    Optimise a step‑function within the allotted time budget.
    Returns a list of non‑negative floats (any normalisation works – the
    evaluation function is scale‑invariant).

    start_time = time.time()

    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical lower bound.
    # 8000 is a good compromise between resolution and speed.
    # ------------------------------------------------------------------
    n = 8000
    size_fft = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise – try to reuse a previously discovered construction if it

        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
