    deadline = start_time + budget_s - 2.0

    # init a

        n = a.size
    # normalize sum=1

    # compute initial objective and conv
    cur_obj, conv = compute_obj_and_conv(a)
    best_obj = cur_obj
    best_a = a.copy()
    print(f"[init] n={n}, objective={cur_obj:.6f}")

    # Precompute indices for LP
    row_i = np.repeat(np.arange(n), n)  # shape n^2
    col_j = np.tile(np.arange(n), n)
    row_k = row_i + col_j
    A_shape = (2*n-1, n)

    lp_steps = 5
    for lp_iter in range(lp_steps):
        if time.time() >= deadline:
            break
        # compute M from current conv
        M = conv.max()
        # Solve LP to get b

        A = sparse.csr_matrix((data, (row_k, col_j)), shape=A_shape)

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"[LP] iteration {lp_iter}: linprog failed")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:

"""
Propose a candidate sequence that minimizes the self‑convolution peak.

The algorithm works in two phases:

1. **LP‑flattening phase** – a few global steps solve a linear program that
   maximises the total mass of a partner distribution `b` while keeping the
   convolution `a * b` under the current maximal convolution value `M`.
   The partner is mixed with the current sequence (`a ← (1‑α)a + α b̂`) to
   obtain a flatter profile.

2. **Simulated‑annealing hill‑climbing phase** – after the LP steps the
   method performs many inexpensive mass‑transfer moves.  A move transfers a
   small amount `δ` from the index contributing most to the current peak to a
   random destination.  The convolution is updated incrementally (O(n)) and
   the move is accepted if it lowers the objective or with a probability that
   follows a linear temperature schedule, allowing occasional uphill moves.
   Periodic full FFT recomputation and jitter moves prevent drift and help
   escape plateaus.

Both phases respect the allotted time (default 1000 s) and keep the best
sequence seen.  The returned sequence is a list of non‑negative floats with
sum = 1, which is invariant for the evaluation function.
"""

import time
import math

import numpy as np
from scipy import sparse
from scipy.optimize import linprog


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    np.maximum(conv, 0, out=conv)                 # numerical safety
    return conv


def _objective(a: np.ndarray, conv: np.ndarray = None) -> float:
    """Return the evaluation objective 2·n·max(a*a) for a probability vector."""
    n = a.shape[0]
    if conv is None:
        conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function stops a couple of seconds
        before the deadline to avoid a hard termination).

    Returns
    -------
    list[float]
        The best sequence found (list of non‑negative floats, sum≈1).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin so the framework does not kill us
    deadline = start_time + budget_s - 1.5

    # ------------------------------------------------------------------
    # Initialise the probability vector a
    # ------------------------------------------------------------------
    try:
        init_seq = height_sequence_1               # noqa: F821
    except NameError:
        init_seq = None

    if init_seq is None:
        n = 1800
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.size

    # Normalise to a probability distribution (scaling does not affect score)
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a /= a_sum

    # ------------------------------------------------------------------
    # Initial objective and exact convolution
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    cur_obj = _objective(a, conv)
    best_obj = cur_obj
    best_a = a.copy()
    print(f"[init] n={n}, objective={cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Pre–compute index arrays for the LP matrix (they never change)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.int32)
    row_i = np.repeat(idx, n)           # i‑index for a_i
    col_j = np.tile(idx, n)             # j‑index for b_j
    row_k = row_i + col_j                # convolution index k = i + j
    A_shape = (2 * n - 1, n)

    # ------------------------------------------------------------------
    # Phase 1 – a few global LP‑flattening steps
    # ------------------------------------------------------------------
    max_lp_steps = 5
    for lp_step in range(max_lp_steps):
        if time.time() >= deadline:
            break

        M = conv.max()                           # current peak bound
        # Build the constraint matrix A (sparse) for the LP: (a*b)_k ≤ M
        data = a[row_i]                          # a_i repeated for each j
        A = sparse.csr_matrix((data, (row_k, col_j)), shape=A_shape)

        # Linear program: maximise sum(b)  ⇔  minimise -∑b
        c = -np.ones(n)
        ub = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A, b_ub=ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(1, int((deadline - time.time())/2))})
        if not res.success:
            # LP failed – maybe the current bound is already tight
            print(f"[LP] step {lp_step}: linprog failed ({res.message})")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Normalise the partner distribution
        g = b / sum_b

        # Mix the partner into the current distribution with a moderate step size
        for alpha in (0.5, 0.3, 0.1):
            a_new = (1.0 - alpha) * a + alpha * g
            conv_new = _full_convolution_fft(a_new, a_new)
            new_obj = _objective(a_new, conv_new)
            if new_obj < cur_obj:
                a = a_new
                conv = conv_new
                cur_obj = new_obj
                print(f"[LP] step {lp_step}, α={alpha:.2f} → obj={cur_obj:.6f}")
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                break   # accept the first improvement

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing hill climbing with incremental updates
    # ------------------------------------------------------------------
    iteration = 0
    last_improve_iter = 0

    # Annealing schedule parameters
    max_frac = 0.25          # start moving up to 25 % of the source mass
    min_frac = 0.02          # end moving about 2 %
    jitter_prob = 0.015      # probability of a large random jitter move
    recompute_interval = 5000

    # Temperature schedule (linear from T0 down to a tiny final temperature)
    T0 = cur_obj * 0.02
    T_final = cur_obj * 1e-4

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        progress = (now - start_time) / (deadline - start_time)

        # ----- temperature and move‑size schedules -----
        temp = T0 * (1.0 - progress) + T_final * progress
        move_frac = max_frac * (1.0 - progress) + min_frac * progress

        # ----- identify the most critical coefficient pair -----
        k_star = int(conv.argmax())
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx_range = np.arange(i_start, i_end + 1)
        partners = k_star - idx_range
        products = a[idx_range] * a[partners]
        i_src = int(idx_range[products.argmax()])   # source index

        # ----- choose a destination index different from the source -----
        j_dst = np.random.randint(n - 1)
        if j_dst >= i_src:
            j_dst += 1

        # ----- propose a mass‑transfer move -----
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue

        a_old = a.copy()
        conv_old = conv.copy()

        delta2 = delta * delta
        # update the convolution incrementally (exact formula)
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[j_dst:j_dst + n] += 2.0 * delta * a_old
        conv[2 * i_src] += delta2
        conv[2 * j_dst] += delta2
        conv[i_src + j_dst] -= 2.0 * delta2

        a[i_src] -= delta
        a[j_dst] += delta

        new_obj = float(2.0 * n * conv.max())

        # ----- acceptance rule (downhill always, uphill with probability) -----
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            if temp > 0.0 and np.random.rand() < math.exp(-(new_obj - cur_obj) / temp):
                accept = True

        if accept:
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration:7d}] new best = {best_obj:.6f}")
                last_improve_iter = iteration
        else:
            # revert the move
            a[:] = a_old
            conv[:] = conv_old

        # ----- occasional random jitter move (large transfer) -----
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta_j = 0.5 * a[src] * np.random.rand()
            if delta_j > 0.0:
                a_tmp = a.copy()
                conv_tmp = conv.copy()
                delta2j = delta_j * delta_j

                conv_tmp[src:src + n] -= 2.0 * delta_j * a_tmp
                conv_tmp[dst:dst + n] += 2.0 * delta_j * a_tmp
                conv_tmp[2 * src] += delta2j
                conv_tmp[2 * dst] += delta2j
                conv_tmp[src + dst] -= 2.0 * delta2j

                a_tmp[src] -= delta_j
                a_tmp[dst] += delta_j

                jitter_obj = float(2.0 * n * conv_tmp.max())
                if jitter_obj < cur_obj:
                    a = a_tmp
                    conv = conv_tmp
                    cur_obj = jitter_obj
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{iteration:7d}] jitter improved best = {best_obj:.6f}")
                # otherwise discard the jitter

        # ----- periodic full recomputation to avoid drift -----
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a, a)

        # ----- stagnation handling -----
        if iteration - last_improve_iter > 200_000:
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_obj = float(2.0 * n * conv.max())
            last_improve_iter = iteration

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
