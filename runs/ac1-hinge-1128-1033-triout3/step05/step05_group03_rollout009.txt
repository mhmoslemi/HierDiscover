    for p in patterns:

"""
Multi‑restart optimiser for the protected `evaluate_sequence` objective.

The code explores several analytically motivated initial shapes (uniform,
Gaussian bump, power‑law “edge‑heavy” distributions).  For each start it
runs a short Adam phase on a smooth max‑approximation of the convolution,
followed by the exact hill‑climber from the reference solution.
All runs share the same global best – if a better candidate is found any
future run can use it as a warm‑start.  The time budget (default 1000 s)
is split automatically between the different starts; the routine stops
as soon as the target value 1.503 is reached.

The final sequence (list of non‑negative floats) is returned; it can be
passed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum of a is assumed = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) (the denominator is 1 after normalisation)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of max(conv(a,a)).
    Returns the best (a, value) seen before *deadline*.
    """
    # initialise logits (soft‑max)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.08, 0.002
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 20000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()                       # Σa = 1, a ≥ 0

        # exact objective for tracking
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve_cnt += 1

        # schedule for ε and learning‑rate
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max (log‑sum‑exp) -----
        b_full = _full_convolution(a)               # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                 # p ≈ one‑hot when eps tiny

        # ----- gradient of smooth objective w.r.t. a -----
        p_rev = p[::-1]                             # need reversed for convolution
        conv_ab = _conv_fft(a, p_rev)               # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]         # the n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.007:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stagnated
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restart after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """
    Local search that transfers a small amount of mass between coordinates.
    Only strictly improving moves are kept; after a long stagnation the
    current best is perturbed and the search continues.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # parameters
    max_frac_start, max_frac_end = 0.5, 0.02
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # scheduled allowed transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (biased to large contributors)
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # random target different from source
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept only strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] perturb after stagnation")
        iteration += 1

    return best_a

def propose_candidate(seed: int = 42,

    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list of floats (Σ > 0) that minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the final evaluation call
    overall_deadline = start_time + budget_s - 0.5

    # problem size – 8000 gives a good trade‑off between quality & speed
    n = 8000

    best_a = None
    best_val = float('inf')
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Collection of diverse starting shapes
    # ------------------------------------------------------------------
    start_configs = [
        ("uniform", lambda n: _init_uniform(n)),
        ("gaussian", lambda n: _init_gaussian(n, sigma_factor=1 / 6)),
        ("edge_p0.4", lambda n: _init_edge_heavy(n, p=0.4)),
        ("edge_p0.8", lambda n: _init_edge_heavy(n, p=0.8)),
        ("edge_p1.2", lambda n: _init_edge_heavy(n, p=1.2)),
        ("edge_p1.6", lambda n: _init_edge_heavy(n, p=1.6)),
    ]

    # ------------------------------------------------------------------
    # Iterate over the initial configurations, allocating a share of time
    # ------------------------------------------------------------------
    for name, factory in start_configs:
        # recompute remaining budget
        now = time.time()
        if now >= overall_deadline - 5.0:         # leave a few seconds for final eval
            break
        remaining = overall_deadline - now
        # give this configuration roughly an equal share of the remaining time
        # (90 % of what is left, the rest stays for later configs)
        alloc = remaining * 0.9 / len(start_configs)

        config_deadline = now + alloc
        # ----- initialise -----
        a0 = factory(n)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0)
            s = n
        a0 /= s                                          # Σ a0 = 1

        # ----- Phase A – Adam on smoothed max -----
        phase_a_deadline = now + (config_deadline - now) * 0.4
        a_opt, val_opt = _adam_phase(
            a0,
            n,
            start=start_time,
            deadline=phase_a_deadline,
            best_a=best_a if best_a is not None else a0,
            best_val=best_val,
            target_val=TARGET,
        )

        # ----- Phase B – exact hill‑climbing -----
        a_opt = _hillclimb_targeted(
            a_opt,
            n,
            start=phase_a_deadline,
            deadline=config_deadline,
            best_a=best_a if best_a is not None else a_opt,
            best_val=best_val,
            target_val=TARGET,
        )

        # ----- evaluate this run -----
        cur_val = _objective(a_opt)
        if cur_val < best_val:
            best_val, best_a = cur_val, a_opt.copy()
            print(f"[{name}] new global best = {best_val:.6f}")

        # stop early if we have already reached the target
        if best_val <= TARGET:
            print("[early exit] target reached")
            break

    # ------------------------------------------------------------------
    # Final assessment via the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
