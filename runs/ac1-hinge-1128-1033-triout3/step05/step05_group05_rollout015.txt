"""
A fast optimisation for the step‑function problem.

The programme works in three stages:

1. **Initialisation** – start from a known good construction if it exists,
   otherwise from a random Dirichlet vector (sum = 1).  The length `n`
   is chosen large enough (8000) to give the optimiser enough freedom.

2. **Smooth‑max gradient descent** – minimise a smooth approximation of the
   true objective ``2·n·max(a∗a)``.
   The approximation is the *log‑sum‑exp* of the convolution:
        Lτ(a) = (1/τ)·log Σ_k exp( τ·( (a∗a)_k ) )
   whose gradient can be written with two FFT‑convolutions.
   We anneal the temperature `τ` (starting small, ending large) and use a
   linearly decaying step size together with a simple momentum term.
   After each step the iterate is projected onto the probability simplex
   (non‑negative entries that sum to one).

3. **Targeted hill‑climbing** – from the best point found in stage 2 we
   perform fast O(n) “mass‑transfer’’ moves that strictly improve the real
   objective.  The source index is chosen among the coordinates that
   contribute most to the current peak of the convolution; the target is
   random.  Stagnation triggers a small Gaussian perturbation.

The whole routine runs for at most ``budget_s`` seconds (default 1000 s) and
returns the best list of non‑negative heights found.  The evaluation
function `evaluate_sequence` is called only once at the end to report the
final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Gradient step for smooth‑max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, tau: float) -> np.ndarray:
    """
    Returns the gradient of   Lτ(a) = (1/τ)·log Σ exp(τ·(a∗a))
    w.r.t. `a` (the gradient of the underlying max‑approximation).
    """
    n = a.shape[0]
    b = _full_convolution(a)               # length 2·n‑1
    maxb = b.max()
    # softmax weights (always ≤ 1, numerically stable)
    w = np.exp(tau * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0.0:            # extreme τ – fall back to pure max
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    conv1 = _conv_fft(w, a)
    conv2 = _conv_fft(w, a_rev)
    start = n - 1
    end = 2 * n - 1
    grad = conv1[start:end] + conv2[start:end]      # shape (n,)
    return grad


# ----------------------------------------------------------------------
# Deterministic hill‑climbing that moves mass between coordinates
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """
    Fast local search that moves a small amount of mass between coordinates
    while keeping Σa = 1.  The source index is chosen among those that
    contribute most to the current peak of the convolution.
    Returns the best sequence seen before *deadline*.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac = 0.4            # at most this fraction of source mass may be moved
    IMPROVE_LIMIT = 150_000   # trigger a random perturbation after this many failures
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # locate the current peak of the convolution
        max_idx = int(np.argmax(b))

        # compute contributions of each coordinate to that peak
        idx = np.arange(n)
        partner = max_idx - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]

        # choose a source index (prefer a large contribution)
        if np.random.rand() < 0.85:
            i = int(np.argmax(contrib))
        else:
            # occasional random weighted choice (helps escaping plateaus)
            i = np.random.choice(n, p=a)

        if a[i] <= 1e-14:
            continue

        # random target, different from source
        j = np.random.randint(n)
        if j == i:
            continue

        # amount to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        # cross‑terms (2·δ·a)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val + 1e-12 < cur_val:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:07d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # If stagnation persists, inject a small Gaussian jitter
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:07d}] stagnation – perturbing")

        it += 1

    return best_a

    beta = 0.9

    phase_start = start_time
    for frac, tau in tau_schedule:
        phase_len = grad_budget * frac
        phase_end = phase_start + phase_len

        while time.time() < phase_end:
            # full convolution and current evaluation
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            if cur_val + 1e-12 < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Grad τ={tau:.0f}] new best = {best_val:.6f}")

            # gradient of the smooth‑max objective
            grad = _smooth_max_gradient(a, tau)

            # linear step‑size decay throughout this sub‑phase
            progress = (time.time() - phase_start) / (phase_end - phase_start)
            step = step_max * (1.0 - progress) + step_min * progress

            # simple momentum update
            momentum = beta * momentum + (1.0 - beta) * grad
            a_candidate = a - step * momentum
            a = _project_simplex(a_candidate)

            # occasional tiny random jitter to escape shallow traps
            if np.random.rand() < 0.002:
                a += np.random.normal(scale=0.005, size=n)
                a = _project_simplex(a)

            # restart if we have been stuck for a long time
            # (basic heuristic – number of iterations without improvement)
            # Here we simply keep a counter via the best_val check
            #   and fall back to a fresh random start when stagnation is long.
            # (Implementation kept minimal for clarity.)

        phase_start = phase_end

    # ------------------------------------------------------------------
    # Stage 2 – targeted hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    best_a = _hillclimb_phase(best_a, deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
