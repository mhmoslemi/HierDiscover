    # implement as convolution of w and a_rev.
    n = a.shape[0]

"""
Propose a step–function that (approximately) minimises
    2·n·max_k (a*a)[k]          (a is a probability vector, sum a = 1)

The optimiser is an annealed Adam on the *softmax* parameters.  Instead of
using the raw max we smooth it with a temperature β and compute the
gradient of the smoothed objective analytically (no explicit
enumeration of the active set).  The schedule starts with a low β,
making the landscape smooth, and slowly raises β so that the final
solution respects the true peak of the auto‑convolution.
A small portfolio of diversified seeds (external best, parametric
families and random Dirichlet draws) is built, the best few are
optimised for a share of the time budget and the overall best vector
is returned.

The code runs entirely with NumPy, uses only FFTs (O(n log n) per
iteration) and respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two (used for FFT padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, real inputs)
# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution a * a (real) using FFT.
    Returns only the first 2·len(a)-1 entries.
    """
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _weighted_sum_convolution(w: np.ndarray, a_rev: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the convolution w * a_rev (real FFT) and return the full result.
    """
    fw = np.fft.rfft(w, n=size)
    fa = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(fw * fa, n=size)
    return conv


# ----------------------------------------------------------------------
# Softmax (log‑parameterisation)
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Parametric families for diversified seeds
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to the nearer end
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian bump centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring (objective value for a probability vector)
# ----------------------------------------------------------------------
def _score_vector(a: np.ndarray) -> float:
    """Return 2·n·max_k (a*a)[k] (sum(a) == 1)."""
    n = a.shape[0]
    conv = _self_convolution_fft(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Adam optimiser using the smoothed max (temperature β)
# ----------------------------------------------------------------------
def _run_annealed_adam(initial_a: np.ndarray,
                       time_limit: float,
                       beta_max: float = 200.0,
                       base_lr: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Optimise log‑softmax parameters of `initial_a` for at most `time_limit`
    seconds.  Returns the best vector seen and its raw score (2·n·max‑conv).
    """
    start = time.time()
    n = initial_a.shape[0]

    # Pre‑compute FFT sizes (powers of two)
    size_self = _next_pow2(2 * n - 1)          # for a*a
    size_weight = _next_pow2(3 * n - 2)        # for w * a_rev (length 2n‑1 + n‑1)

    # Initialise log‑parameters (tiny epsilon avoids log(0))
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    # Simple linear schedule for β (temperature)
    beta0 = 0.5
    beta = beta0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() - start < time_limit - 0.5:
        iteration += 1

        # ---- probabilities ----
        a = _softmax(x)

        # ---- self‑convolution (a*a) ----
        conv = _self_convolution_fft(a, size_self)
        max_b = conv.max()

        # ---- record best ----
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # ---- anneal temperature β ----
        # increase β slowly, but never exceed beta_max
        if beta < beta_max:
            beta = min(beta_max, beta0 + iteration * (beta_max - beta0) / 50000.0)

        # ---- soft‑max weights w_k ≈ softmax_β(conv) ----
        # shift by max_b for numerical stability (max term becomes 1)
        w_unnorm = np.exp(beta * (conv - max_b))
        w = w_unnorm / w_unnorm.sum()

        # ---- convolution w * a_rev to obtain Σ_k w_k a_{k-i} ----
        a_rev = a[::-1]
        conv_wr = _weighted_sum_convolution(w, a_rev, size_weight)
        # slice that corresponds to Σ_k w_k a_{k-i} for i=0..n‑1
        S = conv_wr[n - 1: n - 1 + n]   # length n

        # ---- gradient wrt a (derived analytically) ----
        # grad_a_i = 4·n·S_i  (because ∂f/∂a_i = 2·n·Σ_k w_k·2·a_{k-i})
        grad_a = 4.0 * n * S

        # ---- gradient wrt log‑parameters x (softmax chain rule) ----
        dot = np.dot(a, grad_a)                # scalar = Σ a_i·grad_a_i
        grad_x = a * (grad_a - dot)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of the learning rate (helps long‑run stability)
        lr = base_lr / (np.sqrt(iteration) + 1.0)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- re‑heat if stalled for a long time ----
        if iteration - last_improve > 6000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

    return best_vec, best_score

        base_seeds = []

    # --------------------------------------------------------------
    # Build a diversified pool of starting vectors
    # --------------------------------------------------------------
    seeds = list(base_seeds)          # start with external if we have it

    # uniform vector (always a decent baseline)
    seeds.append(np.ones(n) / n)

    # parametric families (grid search)
    for p in np.linspace(0.2, 2.4, 11):
        seeds.append(u_min_distribution(n, p))
        seeds.append(u_prod_distribution(n, p))
    for lam in np.logspace(-4, -1, 8):
        seeds.append(exp_distribution(n, lam))
    for sigma in np.linspace(0.2, 1.2, 11):
        seeds.append(gaussian_distribution(n, sigma))

    # a few random Dirichlet draws (different concentration)
    for alpha in [0.2, 0.5, 1.0]:
        for _ in range(3):
            r = np.random.gamma(alpha, 1.0, size=n)
            seeds.append(r / r.sum())

    # --------------------------------------------------------------
    # Evaluate all seeds quickly and keep the best few
    # --------------------------------------------------------------
    scored_seeds = []
    for a in seeds:
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.ones_like(a)
        a = a / a.sum()
        sc = _score_vector(a)
        scored_seeds.append((sc, a))

    scored_seeds.sort(key=lambda x: x[0])
    top_k = 5
    top_seeds = [a for _, a in scored_seeds[:top_k]]
    print(f"Selected {len(top_seeds)} promising seeds (best score {scored_seeds[0][0]:.6f})")

    # --------------------------------------------------------------
    # Allocate time per seed (remaining time after preparations)
    # --------------------------------------------------------------
    remaining_time = deadline - time.time()
    per_seed_budget = max(5.0, remaining_time / max(1, len(top_seeds)))

    best_overall_vec = None
    best_overall_score = np.inf

    for idx, seed_vec in enumerate(top_seeds):
        # make sure the seed is a proper probability vector
        a = np.maximum(seed_vec, 0.0)
        a = a / a.sum()

        print(f"\n=== Optimisation start: seed {idx+1}/{len(top_seeds)} "
              f"(budget {per_seed_budget:.1f}s) ===")

        vec, sc = _run_annealed_adam(a, per_seed_budget)

        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_vec = vec.copy()
            print(f"*** New global best (score = {best_overall_score:.6f})")

        # update remaining budget for the next seed
        remaining_time = deadline - time.time()
        if remaining_time <= 3.0:
            break
        per_seed_budget = remaining_time / max(1, len(top_seeds) - idx - 1)

    # --------------------------------------------------------------
    # Final report
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    if best_overall_vec is None:
        # as a fallback, return a uniform vector (guaranteed valid)
        best_overall_vec = np.ones(n) / n
        best_overall_score = _score_vector(best_overall_vec)

    print(f"Best score found: {best_overall_score:.6f} (length {len(best_overall_vec)})")
    return best_overall_vec.tolist()
