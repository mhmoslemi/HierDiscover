def subgradient_step(a, lr):
    b = np.convolve(a,a)
    k = np.argmax(b)
    grad = np.zeros_like(a)
    # compute gradient as 2 * a_rev shifted?
    # For each i, if 0 <= k - i < n: grad[i] = 2 * a[k - i]
    n = len(a)
    low = max(0, k - (n-1))
    high = min(k, n-1)
    # i in [low, high], j = k - i in valid range.
    grad[low:high+1] = 2 * a[k - np.arange(low, high+1)]
    # step in negative direction

    # project onto simplex

Project onto simplex implementation:

def simplex_projection(v):
    """Project vector v onto probability simplex (sum=1, v>=0)"""
    # reference: https://github.com/indranil1995/fast_projection_simplex
    v = np.asarray(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Improved two‑phase optimiser for the step‑function bound.

Phase A – Adam on a smooth log‑sum‑exp approximation of the max‑convolution.
The schedule is slower (more iterations) than the reference implementation,
and we occasionally inject a large random perturbation to escape plateaus.

Phase B – Directed mass‑transfer hill‑climbing.  At each step the current
max‑convolution index k* is identified; the “source’’ index i is chosen as
the one contributing most to this maximum (max a[i]·a[k*‑i]), while the
“target’’ index j is the position whose partner a[k*‑j] is smallest.
Moving a small amount of mass δ from i to j is guaranteed to decrease the
value of the current maximum (if δ>0) and the update of the full convolution
is performed in O(n) using the linear formula from the reference code.
If no improvement occurs for a while we re‑seed from the best solution seen
so far with a small Gaussian noise.

Both phases run until the time budget is exhausted; the best sequence ever
found is returned.  The algorithm works on the full size n=8000 (≈250 KB
convolution) but the logic is fully vectorised and fits comfortably into the
1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (onto the probability simplex Σa = 1, a ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, "Projection onto the probability simplex"
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (without the scaling that evaluate_sequence does)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) (sum(a) is assumed = 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smoothed approximation of max(conv(a,a)).
    Returns the best (a, value) seen before *deadline*.
    """
    # logits for a soft‑max guarantee Σa = 1, a ≥ 0
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters (more conservative than the reference)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.002
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 20_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- soft‑max to obtain a on the simplex ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()                       # Σa = 1, a ≥ 0

        # ---- exact objective for tracking ----
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- schedules for ε and learning‑rate ----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---- smooth max via log‑sum‑exp (numerically stable) ----
        b_full = _full_convolution(a)           # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()               # soft arg‑max

        # ---- gradient of smooth objective w.r.t. a ----
        p_rev = p[::-1]
        conv_ab = np.fft.irfft(
            np.fft.rfft(a) * np.fft.rfft(p_rev), n + n - 1)[:2 * n - 1]
        # extract the n entries that correspond to ∂c/∂a
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]        # shape (n,)

        # ---- chain rule through the soft‑max ----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ---- occasional large perturbation (helps escape plateaus) ----
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.05, size=n)

        # ---- restart on long stagnation ----
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – directed hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_directed(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast local search that moves a small amount of mass from the index that
    contributes most to the current maximal convolution entry to the index
    whose partner contribution is smallest.
    """
    a = a_start.copy()
    b = _full_convolution(a)                # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.02
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- current maximal convolution index ----
        k_max = int(np.argmax(b))

        # ---- compute partner values a[k_max‑i] for all i where defined ----
        # partner[i] = a[k_max - i] or 0 if out of range
        partner = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)          # inclusive
        idx = np.arange(lo, hi + 1)
        partner[idx] = a[k_max - idx]

        # ---- source: index with largest contribution a[i] * partner[i] ----
        contrib = a * partner
        source = np.argmax(contrib)     # guaranteed >=0

        if a[source] <= 1e-14:
            # nothing to move – perturb and continue
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            continue

        # ---- target: index whose partner value is minimal (and ≠ source) ----
        # we exclude the source index explicitly
        partner[source] = np.inf          # forbid choosing the source
        target = int(np.argmin(partner))

        # ---- amount of mass to transfer (progressively shrinking) ----
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        delta = a[source] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) ----
        b_new = b.copy()
        # the linear part (2*delta*a) – a is the *old* vector before the move
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure‑quadratic corrections
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- accept only improving moves ----
        if new_val < cur_val - 1e-14:
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ---- stagnation handling ----
        if no_improve > 150_000:
            # re‑seed from the best seen so far with a small perturbation
            a = best_a.copy()
            a += np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] perturbing after stagnation")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin of 0.5 s for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # problem size – 8000 works well with the FFT‑based convolution
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – a simple smooth bump (exponential decay) + uniform mix
    # ------------------------------------------------------------------
    # exponential decay shape a_i ∝ r**i  with r≈exp(-0.5/(n-1))
    t0 = 0.5
    r0 = np.exp(-t0 / (n - 1))
    a0 = r0 ** np.arange(n, dtype=float)
    a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Phase A – smooth‑max Adam optimisation (≈ 45 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_phase(a0, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=a0.copy(),
                                   best_val=float('inf'),
                                   target_val=1.503)

    # ------------------------------------------------------------------
    # Phase B – directed hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    phase_b_deadline = deadline
    best_a, best_val = _hillclimb_directed(best_a, n,
                                           start=phase_a_deadline,
                                           deadline=phase_b_deadline,
                                           best_a=best_a,
                                           best_val=best_val,
                                           target_val=1.503)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
