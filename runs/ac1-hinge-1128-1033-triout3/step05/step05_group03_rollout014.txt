def full_convolution(a):

    n = len(a)

    out_len = 2*n -1

    size = 1 << (out_len -1).bit_length()

    fx = np.fft.rfft(a, size)

    conv = np.fft.irfft(fx * fx, size)[:out_len]

    return conv

def hillclimb_targeted(a_start, n, start_time, deadline, best_a, best_val):

    a = a_start.copy()

    b = full_convolution(a)

    cur_val = 2.0 * n * b.max()

    max_frac_start = 0.5

    max_frac_end = 0.02

    no_improve = 0

    IMPROVE_LIMIT = 100000   # if many steps without improvement, random perturb

    iteration = 0

    while True:

        now = time.time()

        if now >= deadline:

            break

        # compute progress ratio

        progress = (now - start_time) / (deadline - start_time)

        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # find k_max

        k_max = int(np.argmax(b))

        # compute contributions for each index

        idx = k_max - np.arange(n)

        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)

        contrib[mask] = a[mask] * a[idx[mask]]

        total_contrib = contrib.sum()

        if total_contrib > 0:

            probs = contrib / total_contrib

            i = np.random.choice(n, p=probs)

        else:

            i = np.argmax(a)

        # choose target j such that a[k_max - j] is minimal (or zero) and different from i

        # compute idx_target = k_max - np.arange(n)

        idx_target = k_max - np.arange(n)

        # mask valid indices for target (i.e., positions where product would be zero if out of range)

        valid = (idx_target < 0) | (idx_target >= n)

        # Among valid indices, pick one uniformly

        candidates = np.where(valid)[0]

        if len(candidates) > 0:

            j = np.random.choice(candidates)

        else:

            # all indices produce nonzero product; pick index with minimal a[idx_target] (lowest product)

            # ensure idx_target in range

            # We can compute product contributions

            prod_vals = np.full(n, np.inf)

            in_range = (idx_target >= 0) & (idx_target < n)

            prod_vals[in_range] = a[idx_target[in_range]]

            # exclude i

            prod_vals[i] = np.inf

            j = int(np.argmin(prod_vals))

        # ensure j != i

        if j == i:

            # pick something else

            j = (i + 1) % n

        # compute delta

        delta = a[i] * np.random.rand() * max_frac

        if delta <= 0:

            continue

        # update b (copy)

        b_new = b.copy()

        # range updates, ensure indices within bounds of b_new (0..2n-2)

        # add effect of +delta at position j

        b_new[j:j + n] += 2.0 * delta * a

        b_new[i:i + n] -= 2.0 * delta * a

        # pure quadratic adjustments

        # note that b_new may exceed length: ensure indices in range.

        # careful: index 2*j may be > len(b)-1 -> Check.

        idx_2j = 2 * j

        if idx_2j < b_new.shape[0]:
            b_new[idx_2j] += delta * delta

        idx_2i = 2 * i

        if idx_2i < b_new.shape[0]:
            b_new[idx_2i] += delta * delta

        idx_ij = i + j

        if idx_ij < b_new.shape[0]:
            b_new[idx_ij] -= 2.0 * delta * delta

        new_max = b_new.max()

        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-14:

            a[j] += delta

            b = b_new

            if new_val < best_val:

                best_val = new_val

                best_a = a.copy()

"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.  The algorithm combines three ideas:

1. **Multiple restarts** – many short optimisation runs are performed,
   each starting from a different random seed (uniform or Gaussian bump).
2. **Phase A – Adam on a smooth max** – the objective `max(conv(a,a))`
   is approximated via a log‑sum‑exp with a decreasing temperature.
   The parameters are represented by soft‑max logits, guaranteeing
   non‑negativity and Σa = 1.  Adam with a learning‑rate schedule
   performs fast descent.
3. **Phase B – aggressive hill‑climbing** – a specialised local search
   that always moves mass from a “source” index contributing heavily to
   the current maximal convolution entry to a “target” index that does
   **not** contribute to that entry (i.e. whose partner index is out of
   range).  This guarantees a strict decrease of the objective whenever
   a move is accepted.  Stagnation triggers a small perturbation of the
   current best solution.

The whole search runs for up to the provided budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of x and y using FFT (no zero‑padding beyond needed)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assuming ∑a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a (∑a=1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth approximation of max(conv(a,a)).
    Returns the best (a, value) seen before *deadline*.
    """
    # logits for soft‑max; guarantees Σa = 1 and a_i ≥ 0
    w = np.log(a0 + 1e-14)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.30, 0.003            # relatively aggressive schedule
    eps_start, eps_end = 5e-2, 1e-12          # temperature for smooth‑max

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- soft‑max → a on the simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()                     # Σa = 1

        # ----- exact objective for bookkeeping -----
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedules -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)       # 2·n‑1 entries
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)           # safe, eps > 0
        p = weights / weights.sum()         # soft‑max over the convolution entries

        # ----- gradient of smooth objective w.r.t. a -----
        # p_rev is needed because convolution is not commutative in the
        # indexing used by _conv_fft.
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)       # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]  # the part that equals ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]    # ∂/∂a of smooth max term

        # ----- chain rule through the soft‑max -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- occasional random perturbation -----
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.02, size=n)

        # ----- restart on stagnation -----
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restart (no improvement {no_improve})")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – aggressive hill‑climbing with a guaranteed‑improvement target
# ----------------------------------------------------------------------
def _hillclimb_targeted_improved(a_start: np.ndarray,
                                 n: int,
                                 start: float,
                                 deadline: float,
                                 best_a: np.ndarray,
                                 best_val: float,
                                 target_val: float = 1.5) -> np.ndarray:
    """
    Local search that transfers a small amount of mass from a source index
    (chosen proportionally to its contribution to the current maximal
    convolution entry) to a target index that does **not** contribute to
    that entry.  This heuristic strictly reduces the objective whenever
    a move is accepted.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # parameters for the mass‑transfer
    max_frac_start, max_frac_end = 0.5, 0.02
    IMPROVE_LIMIT = 120_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress based schedule for the allowed transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # ---- locate the current maximal convolution entry ----
        k_max = int(np.argmax(b))

        # ---- contributions of each index to b[k_max] ----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- source index (biased by contribution) ----
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        # ---- target index: pick one that does NOT appear in the product for k_max ----
        # indices j for which (k_max - j) is out of range give zero contribution.
        idx_target = k_max - np.arange(n)
        valid = (idx_target < 0) | (idx_target >= n)
        candidates = np.where(valid)[0]
        if len(candidates) > 0:
            j = int(np.random.choice(candidates))
        else:
            # all partners are in‑range → choose the index with minimal partner value
            partner_vals = np.full(n, np.inf)
            in_range = (idx_target >= 0) & (idx_target < n)
            partner_vals[in_range] = a[idx_target[in_range]]
            partner_vals[i] = np.inf                    # avoid picking the source again
            j = int(np.argmin(partner_vals))

        if j == i:
            # fallback – pick a different index at random
            j = (i + 1) % n

        # ---- amount of mass to move ----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) ----
        b_new = b.copy()
        # linear term (ordered pairs)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        idx_2j = 2 * j
        if idx_2j < b_new.shape[0]:
            b_new[idx_2j] += delta * delta
        idx_2i = 2 * i
        if idx_2i < b_new.shape[0]:
            b_new[idx_2i] += delta * delta
        idx_ij = i + j
        if idx_ij < b_new.shape[0]:
            b_new[idx_ij] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- accept only strictly improving moves ----
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[HillImp {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ---- stagnation handling ----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillImp {iteration:7d}] perturbing after stagnation")
        iteration += 1

    return best_a

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ ≈ 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a larger n gives more degrees of freedom.
    # --------------------------------------------------------------
    n = 12000                     # ~250 KB for the convolution; fits easily in memory

    best_a_global = None
    best_val_global = float('inf')

    # --------------------------------------------------------------
    # Repeated short optimisation runs (restarts) until time runs out.
    # --------------------------------------------------------------
    RESTART_MIN_TIME = 12.0       # each restart needs at least a few seconds

    while True:
        now = time.time()
        if now + RESTART_MIN_TIME >= deadline:
            break

        # allocate time for the two phases of this restart
        remaining = deadline - now
        # 30 % for Adam, 60 % for hill‑climbing, 10 % left for the next restart
        t_adam = min(30.0, remaining * 0.30)
        t_hill = min(90.0, remaining * 0.60)
        deadline_adam = now + t_adam
        deadline_hill = deadline_adam + t_hill

        # ----------------------------------------------------------
        # Random initialisation (mixture of uniform and a Gaussian bump)
        # ----------------------------------------------------------
        a0 = np.random.rand(n)
        a0 /= a0.sum()

        if np.random.rand() < 0.5:
            # add a random Gaussian component
            x = np.arange(n, dtype=float)
            mu = np.random.rand() * n
            sigma = n / (4.0 + 4.0 * np.random.rand())
            bump = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
            a0 = 0.5 * a0 + 0.5 * bump
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()

        # ----------------------------------------------------------
        # Phase A – Adam optimisation on the smooth max
        # ----------------------------------------------------------
        best_a, best_val = _adam_phase(a0, n,
                                       start=now,
                                       deadline=deadline_adam,
                                       best_a=a0.copy(),
                                       best_val=float('inf'),
                                       target_val=1.5)

        # ----------------------------------------------------------
        # Phase B – aggressive hill‑climbing
        # ----------------------------------------------------------
        best_a = _hillclimb_targeted_improved(best_a, n,
                                              start=deadline_adam,
                                              deadline=deadline_hill,
                                              best_a=best_a,
                                              best_val=best_val,
                                              target_val=1.5)

        # ----------------------------------------------------------
        # Record the best candidate of this restart
        # ----------------------------------------------------------
        cur_val = _objective(best_a)
        if cur_val < best_val_global:
            best_val_global = cur_val
            best_a_global = best_a.copy()
            print(f"[Global] new global best = {best_val_global:.6f}")

        # early exit if we already beat the target
        if best_val_global <= 1.503:
            break

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    if best_a_global is None:
        # fallback – uniform distribution (should never happen)
        best_a_global = np.full(n, 1.0 / n)

    final_seq = best_a_global.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return final_seq
