        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress

"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`. The algorithm consists of:

1. **Initialisation** – up‑sample the best known construction if available,
   otherwise fall back to a smoothly decaying exponential shape.
2. **Phase A – Adam optimisation** on a smooth `log‑sum‑exp` approximation
   of the maximal self‑convolution.  This quickly moves the distribution into
   a good basin.
3. **Phase B – Guided hill‑climbing** – each move transfers a small amount of
   mass from a “source’’ index that contributes heavily to the current maximal
   convolution entry to a “target’’ index that contributes little.  The move is
   accepted only if it strictly improves (or, with a tiny probability, if a
   simulated‑annealing temperature permits).  Stagnation triggers a random
   perturbation of the current best solution.

All computation obeys the 1000 s wall‑time limit; the best solution found is
returned as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1) – returns 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of 2·n·max(conv(a,a)) (sum(a) is assumed = 1)."""
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int,
                start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smoothed approximation of max(conv(a,a))."""
    # logits for soft‑max (guarantee Σ a = 1, a ≥ 0)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001   # a slightly larger initial step
    eps_start, eps_end = 5e-1, 1e-12   # start with a very smooth max

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 25000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max to obtain a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()   # Σ a = 1, a ≥ 0

        cur_val = _objective(a, n)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve_cnt += 1

        # schedule for ε and learning‑rate
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)          # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)              # safe
        p = weights / weights.sum()            # soft assignment

        # gradient of smooth objective w.r.t a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]    # n entries
        grad_a = 4.0 * n * segment[::-1]       # shape (n,)

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stagnated
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restarting after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – guided hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_guided(a_start: np.ndarray,
                      n: int,
                      start: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> np.ndarray:
    """
    Local search that transfers a small amount of mass from a source index
    (chosen proportionally to its contribution to the current maximal
    convolution entry) to a target index that contributes *least* to that
    entry.  Accepts only strictly improving moves; occasional uphill moves
    are allowed by a simulated‑annealing temperature.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # parameters for the mass‑transfer moves
    max_frac_start, max_frac_end = 0.45, 0.02
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    iteration = 0

    # temperature schedule for occasional uphill acceptance
    T_start, T_end = 0.015, 1e-7

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress‑based schedule for the allowed transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # ---- source probabilities (biased by contribution to the max) ----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = np.random.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # ---- target probabilities (prefer indices with small impact on the max) ----
        # impact of moving mass to j on the current maximal entry is proportional
        # to a[k_max - j] (if the index exists).  We therefore weight the inverse.
        rev = np.zeros_like(a)
        rev[mask] = a[idx[mask]]
        inv_weights = 1.0 / (rev + 1e-12)
        probs_tgt = inv_weights / inv_weights.sum()
        j = np.random.choice(n, p=probs_tgt)
        if j == i:
            continue

        # ---- amount of mass to move -------------------------------------------------
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) -------------------------
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- acceptance ------------------------------------------------------------
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            # simulated annealing – allow a tiny uphill step
            T = T_start * (T_end / T_start) ** progress
            if np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[HillGuided {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ---- stagnation handling ---------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            # perturb the current best solution and restart from there
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillGuided {iteration:7d}] perturbing after stagnation")
        iteration += 1

    return best_a

"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`. The algorithm consists of:

1. **Initialisation** – up‑sample a known construction if present,
   otherwise start from a smooth exponentially‑decaying shape.

2. **Phase A – Adam optimisation** on a log‑sum‑exp (smooth‑max) of the
   self‑convolution.  This quickly finds a good region of the search space.

3. **Phase B – Guided hill‑climbing** – each move transfers a small amount
   of mass from a “source’’ index that contributes heavily to the current
   maximal convolution entry to a “target’’ index that contributes little.
   Strict improvements are kept; occasional uphill moves are accepted by a
   tiny simulated‑annealing temperature.  Stagnation triggers a random
   perturbation of the current best solution.

The best sequence found before the deadline is returned as a plain Python
list.  All operations respect the 1000 s wall‑time limit.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1) – returns 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of 2·n·max(conv(a,a)) (sum(a) is assumed = 1)."""
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int,
                start: float, deadline: float,
                best_a: np.ndarray, best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smoothed approximation of max(conv(a,a))."""
    # logits for soft‑max (guarantee Σ a = 1, a ≥ 0)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 5e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 25000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max to obtain a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()   # Σ a = 1, a ≥ 0

        cur_val = _objective(a, n)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve_cnt += 1

        # schedule for ε and learning‑rate
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smoothed max via log‑sum‑exp
        b_full = _full_convolution(a)          # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)              # safe
        p = weights / weights.sum()            # soft assignment

        # gradient of smooth objective w.r.t a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 2·n‑1
        # the useful part of the correlation is the middle n entries
        segment = conv_ab[n - 1: 2 * n - 1]
        grad_a = 4.0 * n * segment[::-1]       # shape (n,)

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stagnated
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restarting after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – guided hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_guided(a_start: np.ndarray,
                      n: int,
                      start: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> np.ndarray:
    """
    Local search that transfers a small amount of mass from a source index
    (chosen proportionally to its contribution to the current maximal
    convolution entry) to a target index that contributes *least* to that
    entry.  Accepts only strictly improving moves; occasional uphill moves
    are allowed by a simulated‑annealing temperature.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # mass‑transfer parameters
    max_frac_start, max_frac_end = 0.45, 0.02
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    iteration = 0

    # temperature schedule for occasional uphill acceptance
    T_start, T_end = 0.015, 1e-7

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress‑based schedule for the allowed transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # ---- source probabilities (biased by contribution to the max) ----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs_src = contrib / total_contrib
            i = np.random.choice(n, p=probs_src)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-14:
            continue

        # ---- target probabilities (prefer indices with small impact) ----
        rev = np.zeros_like(a)
        rev[mask] = a[idx[mask]]
        inv_weights = 1.0 / (rev + 1e-12)
        probs_tgt = inv_weights / inv_weights.sum()
        j = np.random.choice(n, p=probs_tgt)
        if j == i:
            continue

        # ---- amount of mass to move ---------------------------------------
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) -----------------
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- acceptance ----------------------------------------------------
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            T = T_start * (T_end / T_start) ** progress
            if np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[HillGuided {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ---- stagnation handling -------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.012, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillGuided {iteration:7d}] perturbing after stagnation")
        iteration += 1

    return best_a
