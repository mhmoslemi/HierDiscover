def _guided_hill_climb_v2(a, best_val, deadline, rng):
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    delta_frac = 0.1
    no_improve_counter = 0
    while time.time() < deadline:
        # compute conv once per loop
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # identify all k with near max
        ks = np.flatnonzero(conv >= max_b - 1e-12)
        k = int(rng.choice(ks))
        # compute contributions for that k
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        # select pair with largest contribution
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two destination indices that are distinct from i_max/j_max
        # pick from low mass set or random
        # compute candidate moves
        p = int(rng.integers(0, n))
        q = int(rng.integers(0, n))
        while p == i_max and q == j_max:
            p = int(rng.integers(0, n))
            q = int(rng.integers(0, n))

        # Determine delta
        delta = min(cur_a[i_max], cur_a[j_max]) * delta_frac
        if delta < 1e-12:
            break

        new_a = cur_a.copy()
        new_a[i_max] -= delta
        new_a[j_max] -= delta
        new_a[p] += delta
        new_a[q] += delta

        # clip negatives
        if (new_a < 0).any():

        val = evaluate_sequence(new_a.tolist())

            cur_a = new_a

def _adam_optimize(w, eps, max_iter, best_a, best_val, deadline, rng):
    n = w.shape[0]
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        L, grad = _smooth_obj_and_grad(w, eps, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        if t % 200 == 0:
            # evaluate exact objective
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[adam eps {eps:.1e} iter {t}] new best = {best_val:.6f}")
    return w, best_a, best_val

def propose_candidate(seed=42, budget_s=1000, **kwargs):

    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4000

    # convert to w
    w = np.log(best_a + 1e-12)

"""
propose_candidate
-----------------

Find a step‑function (a non‑negative vector a) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the full discrete convolution.
The algorithm works in three phases:

1. **Arc‑sine based start** – a symmetric distribution
   a_i ∝ 1/√((i+½)(n‑½‑i)) is very close to the continuous
   solution of the functional equation f*f = const.  It gives a
   much lower starting value than a uniform vector.

2. **Smooth optimisation** – we minimise a log‑sum‑exp surrogate
   of the ℓ∞‑norm of the autocorrelation using Adam (first‑order)
   optimisation.  The surrogate parameter ε is decreased
   geometrically (continuation) and many Adam steps are taken
   (thousands per ε).  The parameters are a soft‑max
   (`w → a = softmax(w)`) so non‑negativity and the ℓ₁‑norm are
   enforced automatically.

3. **Guided hill‑climbing** – once a good smooth optimum is
   reached we switch to the exact objective and repeatedly move a
   small amount of mass from the pair `(i, j)` that contributes
   most to the current maximal convolution value to two low‑mass
   positions.  The step size is adapted and the move is accepted
   only if the true evaluation improves.

All phases respect the 1000 s time budget; progress and the best
score so far are printed.  The final list of floats is returned.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two >= x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative values caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp surrogate) and its gradient
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smooth surrogate of  max(conv)  and its gradient w.r.t. the
    soft‑max parameters w.
    """
    # soft‑max (a lies on the simplex)
    w_shift = w - np.max(w)          # improve stability
    a = np.exp(w_shift)
    a = a / a.sum()

    # full autocorrelation a*a
    conv = _full_convolution(a)      # length 2·n‑1
    max_conv = conv.max()

    # log‑sum‑exp smoothing of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)     # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()      # shape (2·n‑1,)

    # dL/da = 4·n· (p * a) convolved with a  (see derivation in the
    # original reference).  Implemented via one FFT.
    p_rev = p[::-1]                  # reversed p
    conv_ap = _conv_fft(a, p_rev)    # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]  # middle n entries
    grad_a = 4.0 * n * seg[::-1]     # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)      # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Adam optimiser for the smoothed surrogate
# ----------------------------------------------------------------------


def _adam_optimize(w0: np.ndarray,
                   eps: float,
                   max_iter: int,
                   best_a: np.ndarray,
                   best_val: float,
                   deadline: float,
                   rng: np.random.Generator):
    """
    Perform Adam optimisation on the smooth surrogate.
    Periodically evaluate the true objective and keep the best vector.
    Returns the final parameters w, the best a found and its value.
    """
    w = w0.copy()
    n = w.shape[0]
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2                     # initial learning rate

    for t in range(1, max_iter + 1):
        # stop if the global deadline is reached
        if time.time() >= deadline:
            break

        _, grad = _smooth_obj_and_grad(w, eps, n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # every few steps evaluate the *exact* objective
        if t % 200 == 0:
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a = a / a.sum()
            try:
                val = evaluate_sequence(a.tolist())
            except Exception:                # safety
                val = float("inf")
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[adam eps {eps:.1e} iter {t:5d}] new best = {best_val:.6f}")

    return w, best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _guided_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       rng: np.random.Generator):
    """
    Greedy descent on the true objective.
    At each iteration we look at the lag k where the autocorrelation is
    maximal, pick the pair (i, j) that contributes most to it and move a
    small fraction of that mass to two low‑mass positions that do *not*
    recreate the same lag.  The step size is adapted.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05               # initial fraction of mass to move
    success_streak = 0

    while time.time() < deadline:
        # full autocorrelation
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv >= max_b - 1e-15)
        k = int(rng.choice(ks))

        # indices that form the lag k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two destination indices – pick from the lowest‑mass tail
        low_mass = np.argsort(cur_a)[: max(2, n // 100)]   # ~1 % lowest
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # ensure we do not recreate the offending lag
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # compute a feasible delta
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:  # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        # safety check
        if (new_a < 0).any():
            continue

        # evaluate the exact objective
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            success_streak += 1
            print(f"[guided] new best = {cur_val:.6f}")
            # on success we can afford a slightly larger step
            step_frac = min(step_frac * 1.02, 0.2)
            # every few successes re‑evaluate the low‑mass set
            if success_streak % 10 == 0:
                low_mass = np.argsort(cur_a)[: max(2, n // 100)]
        else:
            # on failure shrink the step size
            step_frac = max(step_frac * 0.97, 1e-4)
            success_streak = 0

    return cur_a, cur_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a step‑function for the supplied evaluation budget.
    Returns a list of floats (the heights) that yields the smallest
    value seen before the timeout.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate length keeps the search affordable.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise: try to reuse a previously found construction, fall back
    # to the arc‑sine shape.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        # arc‑sine based start (discretised 1/√((i+½)(n‑½‑i)))
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
        a0 /= a0.sum()

    # enforce feasibility explicitly
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    w = np.log(a0 + 1e-12)               # soft‑max parameters

    best_a = a0.copy()
    try:
        best_val = evaluate_sequence(best_a.tolist())
    except Exception:
        best_val = float("inf")
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Continuation on the smoothing parameter ε.
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=18)

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        # adapt the number of Adam steps to the remaining time
        remaining = deadline - time.time()
        # each step costs roughly 2 ms → allocate conservatively
        max_iter = int(min(4000, max(800, remaining / 0.002)))

        w, best_a, best_val = _adam_optimize(
            w,
            eps,
            max_iter,
            best_a,
            best_val,
            deadline,
            rng,
        )

        # a final exact evaluation for this ε (cheap, just once)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        try:
            cur_val = evaluate_sequence(a.tolist())
        except Exception:
            cur_val = float("inf")
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Random restarts – perturb the best solution and re‑optimise.
    # --------------------------------------------------------------
    num_restarts = 3
    for r in range(num_restarts):
        if time.time() >= deadline:
            break
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.3, size=n)

        # A short continuation with the smallest ε
        eps = eps_schedule[-1]
        remaining = deadline - time.time()
        max_iter = int(min(5000, max(1200, remaining / 0.002)))

        w, best_a, best_val = _adam_optimize(
            w,
            eps,
            max_iter,
            best_a,
            best_val,
            deadline,
            rng,
        )
        print(f"[restart {r+1}] best evaluation = {best_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if time.time() < deadline:

    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
