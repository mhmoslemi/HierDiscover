"""
Improved search for a step‑function minimizing

    2·n·max_k (a*a)[k]      (with ∑a_i = 1)

The algorithm runs two stages for each seed:

1.  Smooth Adam on a soft‑max parametrisation.
    The max‑convolution is replaced by a smooth “log‑sum‑exp”
    approximation with a temperature β that is slowly increased.
    The gradient is obtained by the correlation
       G_i = 2·∑_j a_j·w_{i+j}
    where w are the normalized soft‑max weights.
2.  A short refinement using the exact active‑set gradient
    (the method from the previous state‑of‑the‑art).

Multiple seeds (the external best, deterministic families and a few random
draws) share the overall 1000 s budget; the best vector found is returned.
All numbers are non‑negative; the evaluation is invariant to scaling, so the
search works with probability vectors (∑ = 1)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Distributions used for deterministic seeds
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear convolution via FFT (full length)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = next_pow2(2 * n)          # enough for linear convolution
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Gradient for the exact max‑convolution (active‑set version)
# ----------------------------------------------------------------------
def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    """∂/∂x  max_k (a*a)[k]   for a = softmax(x)."""
    n = a.shape[0]
    G = np.zeros_like(a)

    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]

    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad


# ----------------------------------------------------------------------
# Stage 1 – Smooth Adam on the log‑probabilities
# ----------------------------------------------------------------------
def run_smooth_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """Adam on a smoothed max‑convolution objective."""
    start = time.time()
    n = initial_a.shape[0]

    # initialise logits (avoid log(0))
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.3                     # base learning rate in log‑space

    # temperature schedule for the smooth max
    beta = 5.0                    # initial temperature
    beta_inc = 0.02               # additive increment per 1000 iters
    beta_max = 300.0

    size = next_pow2(3 * n)       # size ≥ 3·n for correlation a * w
    best_score = np.inf
    best_vec = None
    last_improve = 0
    it = 0

    while time.time() - start < time_limit - 0.5:
        it += 1

        # ---- probability vector ----
        a = softmax(x)

        # ---- FFT of a (shared for convolution & correlation) ----
        fa = np.fft.rfft(a, n=size)

        # ---- linear auto‑convolution a*a ----
        conv_full = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
        max_conv = conv_full.max()
        cur_score = 2.0 * n * max_conv

        # keep best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = it
            print(f"Iter {it:6d}: new best (smooth) score = {best_score:.6f}")

        # ---- temperature update (slow increase) ----
        if (it % 1000) == 0:
            beta = min(beta_max, beta + beta_inc)

        # ---- smooth weights w_k = exp(beta*(c_k-max))/Z ----
        # (c_k - max) <= 0, so values stay ≤ 1
        w_raw = np.exp(beta * (conv_full - max_conv))
        w = w_raw / w_raw.sum()

        # ---- correlation a ⋆ w  (needed for ∂/∂a) ----
        fw = np.fft.rfft(w, n=size)
        corr = np.fft.irfft(fa * np.conj(fw), n=size)[:n]   # first n entries = Σ_j a_j w_{i+j}
        grad_a = 2.0 * corr

        # ---- gradient w.r.t. logits (softmax chain rule) ----
        grad_logits = a * (grad_a - np.dot(a, grad_a))

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)          # simple decay
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- re‑heat if stuck ----
        if it - last_improve > 7000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"Iter {it:6d}: re‑heat (no improvement)")

    return best_vec, best_score


# ----------------------------------------------------------------------
# Stage 2 – Exact max‑gradient Adam (active‑set)
# ----------------------------------------------------------------------
def run_active_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """Fine‑tuning using the exact sub‑gradient of the max‑convolution."""
    start = time.time()
    n = initial_a.shape[0]

    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.2

    best_score = np.inf
    best_vec = None
    last_improve = 0
    it = 0

    while time.time() - start < time_limit - 0.5:
        it += 1

        a = softmax(x)
        conv = full_convolution_fft(a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = it
            print(f"Iter {it:6d}: new best (active) score = {best_score:.6f}")

        # active set – all indices essentially at the maximum
        tol = max_b * 1e-12 + 1e-15
        active = np.where(conv >= max_b - tol)[0]

        grad = grad_log_params(x, a, conv, max_b, active)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if it - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"Iter {it:6d}: re‑heat (active)")

    return best_vec, best_score
