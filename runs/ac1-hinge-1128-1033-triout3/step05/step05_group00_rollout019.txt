"""
Propose a non‑negative height sequence that (approximately) minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a hybrid strategy:

1.  A fast random‑parameter sweep over simple monotone families
    (exponential, power‑law, mixtures) to obtain a good starting point.
2.  A pure sub‑gradient Adam optimiser that directly minimises the
    ℓ∞‑norm of the self‑convolution while keeping the vector on the
    probability simplex (non‑negative, unit sum) via the Duchi‑et‑al.
    projection.
3.  occasional random perturbations when progress stalls.

The routine respects the 1000 s budget, prints intermediate scores and
returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (efficient for many calls).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Core optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a non‑negative height sequence that (approximately) minimises the
    evaluation function.  The algorithm runs for at most `budget_s` seconds,
    prints progress information and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # tiny safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good balance.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Initialise a candidate.
    # Prefer a previous construction if it exists; otherwise start from a
    # simple decreasing exponential (a good baseline for this problem).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] != n:
                if base.shape[0] > n:
                    base = base[:n]
                else:
                    pad = np.zeros(n - base.shape[0], dtype=np.float64)
                    base = np.concatenate([base, pad])
            a = np.maximum(0.0, base)
            if a.sum() == 0.0:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # exponential fall‑off, normalised to unit sum
        lam = rng.uniform(0.5, 3.0)
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Evaluate the initial candidate.
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv          # sum(a) == 1
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – random parametric sweep (≈30 % of the budget).
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.3 * budget_s
    while time.time() < phase1_deadline:
        typ = rng.choice(['exp', 'pow', 'mix'])
        if typ == 'exp':
            lam = rng.uniform(0.1, 5.0)
            cand = np.exp(-lam * np.arange(n, dtype=np.float64))
        elif typ == 'pow':
            p = rng.uniform(0.1, 5.0)
            # decreasing power law: (1 - i/(n-1))**p
            cand = (1.0 - np.arange(n, dtype=np.float64) / (n - 1)) ** p
        else:  # mix of uniform and exponential
            lam = rng.uniform(0.1, 5.0)
            w = rng.random()
            cand = w * np.exp(-lam * np.arange(n, dtype=np.float64)) + (1 - w)

        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand = cand / cand_sum

        conv = _full_convolution_fft(cand, cand)
        score = 2.0 * n * conv.max()
        if score < best_val - 1e-12:
            best_val = score
            best_seq = cand.copy()
            a = cand.copy()                 # seed the next phase
            print(f"[phase1] new_best={best_val:.12f} (type={typ})")

    # ------------------------------------------------------------------
    # Phase 2 – exact max‑sub‑gradient descent with Adam.
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.20                     # base learning rate for Adam

    m = np.zeros_like(a)           # first moment
    v = np.zeros_like(a)           # second moment
    t_adam = 0
    stagn = 0

    while time.time() < deadline:
        t_adam += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # record the best solution seen so far
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phase2 iter {t_adam:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # --- exact sub‑gradient of the max convolution entry ----------
        # indices where the convolution is essentially maximal
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        # gradient w.r.t. a: 2 * (rev(a) * w)  (restricted to length n)
        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        a = a_candidate

        # occasional perturbation to escape flat regions
        if stagn > 3000:
            noise = rng.normal(scale=0.003, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            print(f"[phase2 perturb] iter {t_adam}")

        # keep numbers well‑scaled every few thousand steps
        if t_adam % 5000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Convert best sequence to the form expected by evaluate_sequence.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")

    return final_seq
