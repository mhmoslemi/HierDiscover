def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    n = 4000
    best_seq = np.full(n, 1.0/n)
    best_val = evaluate_sequence(best_seq.tolist())

    start = time.time()
    deadline = start + budget_s - 0.5

    # Stage 1: Random sampling
    while time.time() < deadline - 20:
        # Random concentration parameter between 0.1 and 5 (log-uniform)
        alpha = 10**(rng.uniform(-2, 0.7))  # from 0.01 to 5
        a_raw = rng.gamma(alpha, size=n)  # gamma draws ~ Dirichlet(alpha)
        a = a_raw / a_raw.sum()
        # optionally reorder anti-sym
        a = anti_symmetric_reorder(a)
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[sample] new best {best_val:.6f}")

    # Stage 2: Simulated annealing / hill climbing from best
    cur_seq = best_seq.copy()
    cur_val = best_val
    T0 = cur_val  # initial temperature
    t_start = time.time()
    while time.time() < deadline:
        # schedule temperature dropping linearly
        elapsed = time.time() - t_start
        frac = elapsed / (deadline - t_start)
        T = T0 * (1 - frac) + 1e-6
        # propose modification
        # choose type
        if rng.random() < 0.3:
            # swap
            i, j = rng.integers(n, size=2)
            new_seq = cur_seq.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:
            # mass transfer
            i, j = rng.integers(n, size=2)
            if cur_seq[i] > 0:
                delta = cur_seq[i] * rng.random() * 0.1
                new_seq = cur_seq.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
            else:
                continue
        val = evaluate_sequence(new_seq.tolist())
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / T):
            cur_seq, cur_val = new_seq, val
            if val < best_val:
                best_val = val
                best_seq = new_seq.copy()
                print(f"[anneal] new best {best_val:.6f}")

    return best_seq.tolist()

"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.  The search runs under a 1000 s budget and
combines three ideas:

1. **Random Dirichlet sampling** with a wide range of concentration
   parameters.  For each draw we either (a) keep the random order or
   (b) apply the classic anti‑symmetric re‑ordering (large ↔ small).
   These cheap candidates already beat the uniform construction.

2. **A short simulated‑annealing / hill‑climbing phase** starting from
   the best candidate found so far.  Each step focusses on the
   convolution index where the current max occurs and moves a small
   amount of mass from a “high‑gradient’’ position to a “low‑gradient’’
   one (or performs a random swap).  Improvements are accepted
   greedily; a simple temperature schedule occasionally lets a worse
   move pass to escape plateaus.

3. **Fast objective computation** using FFT‑based full convolution.
   ``evaluate_sequence`` is called only for the final verification
   (the intermediate objective is mathematically identical).

The whole routine respects the time limit, prints progress and finally
returns the best list of heights found.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions – all top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself (length 2·len‑1)
    using FFT.  The result is always non‑negative (tiny negatives are
    clipped to 0 to avoid numerical noise).
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors – FFT implementation."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _objective_from_array(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) / (sum(a)²) for a non‑negative array.
    The same expression is used by ``evaluate_sequence`` (without the
    security checks).  Returns ``np.inf`` for degenerate inputs.
    """
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    conv = _full_convolution(a)
    max_b = conv.max()
    n = a.shape[0]
    return 2.0 * n * max_b / (sum_a ** 2)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Largest values are placed on the left side, smallest on the right,
    producing the classic “large‑small’’ pairing that lowers the centre
    of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]   # large
        new_a[n - 1 - i] = sorted_vals[i]   # small
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _greedy_reduce_max(a: np.ndarray, max_idx: int, rng: np.random.Generator) -> np.ndarray:
    """
    One greedy mass‑transfer step aimed at reducing the peak of the
    convolution at index ``max_idx``.
    ``max_idx`` is the location where the current convolution attains
    its maximum.  The gradient of that entry w.r.t. ``a`` is
    ``g_i = 2·a_{max_idx-i}`` (zero when the partner index is out of range).
    The routine moves a modest fraction of mass from a high‑gradient
    position to a low‑gradient one.
    """
    n = a.shape[0]
    # partner index for each i in the offending shift
    partner = max_idx - np.arange(n)
    valid = (partner >= 0) & (partner < n)

    # gradient proportional to partner's current weight (2 is irrelevant for ordering)
    grad = np.zeros_like(a)
    grad[valid] = a[partner[valid]]

    # donor = index with large gradient AND non‑zero mass
    candidates = np.where((grad > 0) & (a > 0))[0]
    if candidates.size == 0:
        return a
    donor = candidates[rng.integers(candidates.size)]

    # recipient = index with smallest gradient (could be zero)
    recipient = np.argmin(grad)

    if donor == recipient:
        return a

    # transfer up to 8 % of donor's mass (randomised)
    delta = a[donor] * 0.08 * rng.random()
    if delta <= 0:
        return a

    new_a = a.copy()
    new_a[donor] -= delta
    new_a[recipient] += delta
    return new_a


def _anneal_and_hill_climb(start_a: np.ndarray,
                           start_val: float,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing / hill‑climbing improvement starting from
    ``start_a``.  The loop runs until ``deadline`` and returns the best
    vector and its objective.
    """
    a = start_a.copy()
    cur_val = start_val
    rng = np.random.default_rng()
    n = a.shape[0]

    # simple linear temperature schedule
    t_start = time.time()
    initial_temp = cur_val * 0.5 + 1e-6

    while time.time() < deadline:
        elapsed = time.time() - t_start
        frac = elapsed / (deadline - t_start)
        T = max(initial_temp * (1.0 - frac), 1e-9)

        # decide move type
        if rng.random() < 0.3:
            # pure swap (re‑ordering only)
            i, j = rng.integers(0, n, size=2)
            if i != j:
                new_a = a.copy()
                new_a[i], new_a[j] = new_a[j], new_a[i]
            else:
                continue
        else:
            # mass transfer directed at the current worst convolution entry
            conv = _full_convolution(a)
            k_max = int(conv.argmax())
            new_a = _greedy_reduce_max(a, k_max, rng)

        new_val = _objective_from_array(new_a)

        # accept if better, otherwise Metropolis criterion
        if new_val < cur_val - 1e-12:
            a, cur_val = new_a, new_val
        elif rng.random() < np.exp(-(new_val - cur_val) / T):
            a, cur_val = new_a, new_val

        # keep global best (the caller will compare with its own best)
    return a, cur_val

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence`` under a 1000 s time budget.
    The method first explores many random Dirichlet‑generated candidates,
    applies either a random permutation or an anti‑symmetric ordering,
    keeps the best one, and finally refines it with a fast annealing /
    hill‑climbing loop.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – 4000 matches the earlier best
    n = 4000

    # ------------------------------------------------------------------
    # initialise with the uniform distribution (objective = 2.0)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_from_array(best_a)
    print(f"[init] n={n}, uniform objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – random Dirichlet sampling + cheap evaluation
    # ------------------------------------------------------------------
    # allocate roughly 30 % of the total budget for this exploratory phase
    exploratory_deadline = start_time + 0.30 * budget_s
    while time.time() < exploratory_deadline:
        # draw a concentration parameter α on a log‑scale (0.02 … 5)
        log_alpha = rng.uniform(-1.7, 0.7)          # α ∈ [≈0.02, ≈5]
        alpha = 10 ** log_alpha

        # Dirichlet sample via Gamma draws
        raw = rng.gamma(alpha, size=n)
        a = raw / raw.sum()

        # two simple reorderings – keep the random order or apply anti‑symmetry
        if rng.random() < 0.5:
            # random permutation (already random)
            a_perm = a
        else:
            a_perm = _anti_symmetric_reorder(a)

        # fast objective (identical to evaluate_sequence without security checks)
        val = _objective_from_array(a_perm)

        if val < best_val:
            best_val = val
            best_a = a_perm.copy()
            print(f"[sample] α={alpha:.3f}, new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – local refinement (simulated annealing / hill climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_a, final_val = _anneal_and_hill_climb(best_a, best_val, deadline)
        if final_val < best_val:
            best_val = final_val
            best_a = final_a.copy()
            print(f"[refine] post‑anneal best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # final verification (calls the supplied evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised", exc)

    return best_a.tolist()
