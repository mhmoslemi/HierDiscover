def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself.
    Length = 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    Length = len(x)+len(y)‑1.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                           # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase A – Sub‑gradient descent on the exact objective (max b)
# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Reduce max(b) by sub‑gradient steps."""
    start = time.time()
    max_step = 0.10   # initial step size
    min_step = 1e-4   # final step size
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))             # index of the current worst convolution
        cur_max = b[k]
        cur_val = 2.0 * n * cur_max       # exact objective (Σa = 1)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # gradient of max(b) w.r.t a
        grad = np.zeros_like(a)
        # valid i where k-i is inside [0,n-1]
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]   # derivative of a_mid^2

        # linear step‑size schedule
        progress = (time.time() - start) / (deadline - start)
        step = max_step * (1.0 - progress) + min_step * progress

        # gradient step + projection
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – L²‑flattening (smooth variance of b)
# ----------------------------------------------------------------------
def _l2_flatten_phase(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """Minimise Σ(b−b̄)² to flatten the convolution."""
    start = time.time()
    base_step = 0.5
    min_step = 0.001
    avg_target = 1.0 / (2 * n - 1)    # mean of b when Σa = 1
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[L2 {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        r = b - avg_target
        a_rev = a[::-1]
        conv1 = _conv_fft(r, a)          # length 3·n‑2
        conv2 = _conv_fft(r, a_rev)
        start_idx = n - 1
        end_idx = 2 * n - 1
        grad = 2.0 * (conv1[start_idx:end_idx] + conv2[start_idx:end_idx])

        # schedule
        progress = (time.time() - start) / (deadline - start)
        step = base_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        if no_improve > IMPROVE_LIMIT:
            # random restart to escape a possible plateau
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[L2 {it}] restart after stagnation")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Greedy hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """
    Local search that moves mass from the index contributing most
    to the current maximum to the index contributing the least.
    Updates the full convolution incrementally (O(n)).
    """
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000         # stagnation trigger
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # ---- contribution vector for the current worst index ----
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]   # derivative of a_mid^2

        # source = index with largest contribution (most responsible for the peak)
        source = int(np.argmax(contrib))
        # target = index with smallest contribution (ideally zero)
        target = int(np.argmin(contrib))

        if source == target:
            # fallback to a random pair if degenerate
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # nothing to move – pick a random source
            source = np.random.choice(np.where(a > 1e-14)[0])
        # amount to move (up to 30 % of the source mass)
        delta = a[source] * 0.25 * np.random.rand()
        if delta <= 0.0:
            continue

        # ---- incremental update of the full convolution ----
        b_new = b.copy()
        # cross‑terms (2·δ·a) – use slicing; safe because target/source ≤ n‑1
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept the move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional perturbation after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            # tiny Gaussian noise, then re‑project
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1
    return best_a


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (hopefully) minimise
    `evaluate_sequence`.  The routine respects the 1000 s wall‑time
    limit, combines three optimisation phases and finally reports the
    best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 offers more flexibility while staying fast.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # pylint: disable=undefined-variable
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)   # enforce Σa = 1
    best_a = a.copy()
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Split the remaining budget into three phases.
    # ------------------------------------------------------------------
    t_total = deadline - start_time
    t_subgrad = start_time + 0.30 * t_total     # 30 % for sub‑gradient
    t_l2      = start_time + 0.60 * t_total     # next 30 % for L² flattening
    t_hill    = deadline                       # the rest for greedy hill‑climb

    # ----------------------------
    # Phase A – Sub‑gradient descent
    # ----------------------------
    best_a, best_val = _subgradient_phase(
        a, n, t_subgrad, best_a, best_val
    )
    a = best_a.copy()   # continue from the best point found

    # ----------------------------
    # Phase B – L² flattening
    # ----------------------------
    best_a, best_val = _l2_flatten_phase(
        a, n, t_l2, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase C – Greedy hill climbing
    # ----------------------------
    best_a = _greedy_hillclimb(
        a, n, t_hill, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Final evaluation (the secure wrapper may clamp values)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
