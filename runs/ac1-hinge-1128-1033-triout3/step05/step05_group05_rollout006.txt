"""
Propose a step‑function (non‑negative list) that minimises
`evaluate_sequence`.  

Algorithm
---------

1. **Initialisation** – start from the supplied `height_sequence_1`
   (interpolated to the working size) or from a random Dirichlet vector.

2. **Phase G – Gradient descent on a smooth approximation of the
   max‑convolution**.  
   We minimise the log‑sum‑exp of the normalised autoconvolution
   `b = a * a`.  For a scaling factor `α` the objective is  

       f(a) = log ⎡ Σ_k exp( α·b_k / μ ) ⎤ ,   μ = 1/(2n‑1) .

   Its gradient is
       ∇f = (2α/(μ·Σ w))·conv( w , a_rev )
   where `w_k = exp(α·b_k/μ)` (shifted for numerical stability) and
   `a_rev` is the reversed vector.  After each step we project back onto
   the probability simplex (non‑negative entries summing to one).  The
   smoothing parameter `α` is increased linearly from a small start to a
   larger target, turning the smooth objective gradually into the true
   `max` as the optimisation proceeds.

3. **Phase H – Deterministic hill‑climbing on the exact objective**.  
   Starting from the best point found in Phase G we repeatedly move a
   small amount of mass from a source index to a target index, updating
   the full convolution in `O(n)` time.  Only moves that strictly lower
   the true score `2·n·max(b)/(∑a)²` are accepted.  After a long period
   without improvement we apply a small random perturbation to escape
   local minima.

The whole search respects the 1000 s budget; the gradient phase uses
≈60 % of the time and the hill‑climbing the remainder.  The function
returns the best sequence found as a plain Python list.

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex (non‑negative, unit sum)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)

    # Find the threshold index rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w

# ----------------------------------------------------------------------

                     n: int,

    """

    IMPROVE_LIMIT = 180_000
    no_improve = 0
    iteration = 0

        if np.random.rand() < 0.85:
            i = int(np.argmax(a))
        else:

            i = np.random.choice(n, p=a)

            continue

        j = np.random.randint(n)
        if j == i:
            continue

        b_new = b.copy()
        # cross‑terms 2·δ·a

                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # Random perturbation after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max() / (a.sum() ** 2)
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a

# Main entry point required by the outer harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights under the allotted time budget.
    Returns a list of non‑negative floats (unit sum) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 5000 gives a good balance between expressivity
    # and speed (FFT size stays at 16384 for the whole run).
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # Initialise `a` – reuse a known construction if present,
    # otherwise start from a random Dirichlet vector.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_n = old.shape[0]
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)            # enforce Σa = 1
    best_a = a.copy()
    best_val = float('inf')

    # --------------------------------------------------------------
    # Phase G – gradient descent on smooth max‑approximation
    # --------------------------------------------------------------
    grad_deadline = start_time + 0.60 * budget_s
    grad_iter = 0
    no_improve_c = 0
    IMPROVE_LIMIT_C = 60_000

    # step‑size schedule (linear decay)
    step_max = 1e-3
    step_min = 1e-6

    # smoothing parameter schedule
    alpha_start = 0.5
    alpha_end = 15.0

    # constants used throughout the loop
    avg_b = 1.0 / (2 * n - 1)       # sum_a == 1 ⇒ Σb = 1, so avg = 1/(2n‑1)

    while time.time() < grad_deadline:
        # Full autoconvolution (exact)
        b = _full_convolution(a)               # length 2·n‑1
        sum_a = a.sum()
        if sum_a < 1e-12:                       # safety – re‑initialise
            a = np.random.rand(n)
            a = _project_simplex(a)
            continue

        max_b = b.max()
        cur_val = 2.0 * n * max_b / (sum_a ** 2)

        # Track the best solution seen so far
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve_c = 0
            print(f"[Grad {grad_iter:7d}] new best = {best_val:.6f}")
        else:
            no_improve_c += 1

        # ----------------------------------------------------------
        # Log‑sum‑exp gradient (smooth approximation of max)
        # ----------------------------------------------------------
        # Compute the exponential weights w_k = exp(α·b_k/μ), shifted for stability
        progress = (time.time() - start_time) / (grad_deadline - start_time)
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        e = alpha * (b / avg_b)                # shape (2n‑1,)
        max_e = e.max()
        w = np.exp(e - max_e)                  # now max(w)=1
        sum_w = w.sum()
        if sum_w == 0.0:
            sum_w = 1.0                         # should not happen

        # Gradient via convolution w * a_rev
        conv_w_rev = _conv_fft(w, a[::-1])     # length 3·n‑2
        grad = (2.0 * alpha) / (avg_b * sum_w) * conv_w_rev[n - 1:2 * n - 1]

        # Linear step‑size schedule
        step = step_max * (1.0 - progress) + step_min * progress

        # Gradient step and projection back onto the simplex
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # Small random jitter occasionally (helps escape tiny plateaus)
        if np.random.rand() < 0.002:
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)

        # Restart if no improvement for a long stretch
        if no_improve_c > IMPROVE_LIMIT_C:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve_c = 0
            print(f"[Grad {grad_iter}] restart after stagnation")

        grad_iter += 1

    # --------------------------------------------------------------
    # Phase H – deterministic hill‑climbing on the exact objective
    # --------------------------------------------------------------
    best_a = _hillclimb_phase(best_a, n, deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
