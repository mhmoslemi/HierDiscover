def _alternating_extremes_reorder(a):
    n = a.shape[0]
    sorted_vals = np.sort(a)[::-1]  # descending
    new_a = np.empty_like(a)
    left = 0
    right = n-1
    toggle = True
    for val in sorted_vals:
        if toggle:
            new_a[left] = val
            left += 1
        else:
            new_a[right] = val
            right -= 1
        toggle = not toggle
    return new_a

def _bit_reversal_reorder(a):
    n = a.shape[0]
    # n must be power of two
    bits = (n-1).bit_length()
    indices = np.arange(n, dtype=np.int64)
    rev = np.zeros_like(indices)
    for i in range(bits):
        rev = (rev << 1) | (indices & 1)
        indices >>= 1
    perm = rev
    return a[perm]

def _smooth_obj_and_grad(w, eps, n):
    # compute a via softmax
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # convolution a*a using FFT
    conv = _full_convolution(a)
    max_conv = conv.max()
    # log-sum-exp smoothing
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    weights = np.exp(shifted)
    p = weights / weights.sum()

    # compute gradient_a = 4 n * (a * p) convolution ...
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev) # length len
    seg = conv_ap[n-1:2*n-1] # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

"""
Propose the best non‑negative step function under the 1000 s budget.

The search consists of three stages:

1. **Smoothed L‑BFGS optimisation** –  a soft‑max parametrisation
   guarantees non‑negativity and unit ℓ¹‑norm.  The non‑smooth max‑convolution
   is approximated by a log‑sum‑exp (parameter ε).  A schedule of decreasing ε
   values together with many random restarts (each optionally “jittered’’ from
   the current best) drives the solution toward a low peak of the
   self‑convolution.

2. **Deterministic re‑ordering** – the shape of the values is useful but the
   ordering strongly influences the max‑convolution.  After each L‑BFGS run we
   try several inexpensive permutations:
   • anti‑symmetric (large‑small mirroring),
   • alternating‑extremes (largest, smallest, 2nd‑largest, 2nd‑smallest,…),
   • bit‑reversal (spreads large entries uniformly across the index space).

   The best permutation is kept.

3. **Stochastic hill‑climbing refinement** – with the current best sequence we
   repeatedly try small mass transfers, swaps and occasional random partial
   permutations, accepting only strict improvements.  This phase finishes the
   remaining time budget.

All helper functions are defined at module level (no closures) and the routine
respectfully copes with the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate (required for the wrapper)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative values can appear due to rounding – clip them
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)  (ensures a_i ≥ 0, Σ a_i = 1)
    L_ε = 2·n·max(conv) + 2·n·ε·log∑_k exp((conv_k‑max)/ε)
    """
    # soft‑max
    w_shift = w - np.max(w)                 # numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # auto‑convolution a * a
    conv = _full_convolution(a)            # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)           # log∑_k exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a  (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()            # probability vector over k

    # g_i = 4·n·∑_k p_k·a_{k‑i}   computed via convolution with reversed p
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)          # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]         # central n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses the global eps/n)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values on the left half, small values on the right half (mirrored)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]   # large → left
        new_a[n - 1 - i] = sorted_vals[i]   # small → right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternating_extremes_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest, smallest, 2nd‑largest, 2nd‑smallest, …  in alternating
    positions.  This spreads extreme values across the whole index range.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]          # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for val in sorted_desc:
        if toggle:
            new_a[left] = val
            left += 1
        else:
            new_a[right] = val
            right -= 1
        toggle = not toggle
    return new_a


def _bit_reversal_reorder(a: np.ndarray) -> np.ndarray:
    """
    Bit‑reversal permutation (defined for n = 2^k).  This spreads large values
    uniformly, similar to low‑discrepancy sequences.
    """
    n = a.shape[0]
    bits = (n - 1).bit_length()
    rev = np.empty(n, dtype=int)
    for i in range(n):
        x = i
        r = 0
        for _ in range(bits):
            r = (r << 1) | (x & 1)
            x >>= 1
        rev[i] = r
    return a[rev]


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Stochastic hill‑climbing that mixes:
      * pure swaps of two positions,
      * mass transfers (up to 20 % of the donor’s mass),
      * occasional random partial permutation of 20 indices.
    Only strict improvements are kept (fast convergence).
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.2:
            # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = cur_a[j], cur_a[i]
        elif move_type < 0.6:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.2 * rng.random()   # up to 20 % of donor
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # random partial permutation of 20 indices
            idx = rng.choice(n, size=20, replace=False)
            perm = rng.permutation(20)
            new_a = cur_a.copy()
            new_a[idx] = cur_a[idx][perm]

        # evaluate the candidate
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val
