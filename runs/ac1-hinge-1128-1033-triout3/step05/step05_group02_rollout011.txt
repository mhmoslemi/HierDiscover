def hill_climb_simulated(a, best_val, deadline):

    cur = a.copy(); cur_val = best_val

    # schedule: temperature = T0 * (remaining_time / total_time) maybe
    while time.time() < deadline:
        t = (deadline - time.time()) / (deadline - start_time) # proportion left
        T = T0 * t # decreasing
        # pick move
        move_type = rng.random()
        if move_type < 0.3:
            # swap
            i,j = rng.integers(0,n, size=2)
            if i==j: continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            # mass transfer
            i,j = rng.integers(0,n, size=2)
            if i==j or cur[i] == 0: continue
            delta = cur[i] * 0.1 * rng.random()  # up to 10% of donor

            new[i] -= delta
            new[j] += delta
        new_val = evaluate_sequence(new.tolist())
        # accept if better or with probability exp(-(new_val-cur_val)/T)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur.copy()
                # print improvement
    return best_a, best_val

def project_simplex(v):
    # ensure non-negative and sum to 1
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative list of floats) that minimizes the
provided ``evaluate_sequence`` objective.

The algorithm combines three complementary ideas:

1. **Smooth surrogate optimisation** –  a soft‑max parametrisation of the
   probability vector (∑a_i = 1) is optimised with L‑BFGS on a
   log‑sum‑exp smoothing of the max‑convolution (the same surrogate used in
   the reference code, but with a finer ε‑schedule and a larger L‑BFGS iteration
   budget).

2. **Fast combinatorial refinement** –  after each L‑BFGS run a greedy hill‑climb
   works directly on the probability vector.  The max‑convolution and the
   objective are evaluated with an FFT‑based convolution (O(n log n)) so thousands
   of swaps / tiny mass‑transfers can be tested quickly.  A sub‑gradient descent
   on the *unsmoothed* max‑convolution (projected onto the simplex) further
   drives the peak down.

3. **Final annealing phase** –  the remaining time is spent in a simulated‑annealing
   style hill‑climb (swap / mass‑transfer moves) that accepts occasional
   uphill moves, helping escape shallow local minima.

All operations keep the vector on the probability simplex, so scaling does
not affect the final score (the objective is invariant under a global factor).
The routine respects the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the L‑BFGS wrapper (required to avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # remove tiny negatives caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors (FFT based)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate used by L‑BFGS (identical to the reference but with
# a slightly different epsilon schedule and max‑iteration count)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.

    a = softmax(w)  (guarantees a_i ≥ 0 and Σa_i = 1)
    L_ε = 2·n·max(conv) + 2·n·ε·log∑_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    # auto‑convolution a*a
    conv = _full_convolution(a)                 # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)                # log∑_k exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                 # probability over k

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)               # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]              # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals for eps,n)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

# Utility functions that operate directly on the probability vector a

def fast_full_convolution(a: np.ndarray) -> np.ndarray:
    """FFT based self‑convolution (same as _full_convolution but kept for clarity)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def fast_objective(a: np.ndarray) -> float:
    """
    Fast evaluation of the same quantity as ``evaluate_sequence`` (but without
    the clipping/policing performed inside that function).  For a probability
    vector (∑a_i=1) the objective is 2·n·max(conv).
    """
    n = a.shape[0]
    conv = fast_full_convolution(a)
    max_conv = np.max(conv)
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (sum_a ** 2))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def subgradient_descent(a0: np.ndarray,
                        max_iters: int,
                        step0: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *unsmoothed* max‑convolution.
    The sub‑gradient of max(conv) at the peak index k* is 2·a_{k*‑i}.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_val = fast_objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = fast_full_convolution(a)
        k_max = int(np.argmax(conv))

        # build sub‑gradient (vectorised)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k_max - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a = a - step * g
        a = project_simplex(a)

        new_val = fast_objective(a)
        if new_val < cur_val:
            cur_val = new_val
    return a, cur_val


def hill_climb_fast(a0: np.ndarray,
                    max_iters: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Random hill‑climbing that mixes pure swaps (20 % chance) with tiny mass
    transfers (80 %).  Only strict improvements are kept; the fast objective
    using FFT makes it possible to try many thousands of moves within the time
    budget.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_val = fast_objective(a)
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if rng.random() < 0.2:               # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                 # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.05 * rng.random()   # up to 5 % of donor
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_val = fast_objective(new_a)
        if new_val + 1e-14 < cur_val:            # strict improvement
            a = new_a
            cur_val = new_val
    return a, cur_val


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small ones at opposite ends (the simple heuristic
    used in the reference implementation).
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)            # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]
        new_a[n - 1 - i] = sorted_vals[i]

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

        # random soft‑max parameters (wide spread)
        w = np.random.randn(n) * 2.0

        # run through the whole epsilon schedule
        for eps in eps_schedule:
            if time.time() >= deadline - final_phase_seconds:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            # L‑BFGS‑B on the smoothed surrogate
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 250,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # obtain the probability vector
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()

            # cheap fast evaluation first
            cur_fast = fast_objective(a)
            # only call the expensive evaluate_sequence when we see an improvement
            if cur_fast + 1e-12 < best_fast:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    best_fast = cur_fast
                    print(f"[eps {eps:.1e}] new best evaluation = {best_val:.6f}")

        # --------------------------------------------------------------
        # Anti‑symmetric reordering (simple heuristic) and fast hill‑climb
        # --------------------------------------------------------------
        reordered = _anti_symmetric_reorder(best_a)
        reordered_fast = fast_objective(reordered)
        if reordered_fast + 1e-12 < best_fast:
            reordered_val = evaluate_sequence(reordered.tolist())
            if reordered_val < best_val:
                best_val = reordered_val
                best_a = reordered.copy()
                best_fast = reordered_fast
                print(f"[reorder] improved evaluation = {best_val:.6f}")

        # --------------------------------------------------------------
        # Fast hill‑climbing (swap + mass‑transfer) on the current best
        # --------------------------------------------------------------
        hill_deadline = deadline - final_phase_seconds
        a_hc, fast_hc = hill_climb_fast(best_a,
                                        max_iters=50_000,
                                        deadline=hill_deadline)
        if fast_hc + 1e-12 < best_fast:
            hc_val = evaluate_sequence(a_hc.tolist())
            if hc_val < best_val:
                best_val = hc_val
                best_a = a_hc.copy()
                best_fast = fast_hc
                print(f"[hill‑climb] new best = {best_val:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient descent (projected) for further polishing
        # --------------------------------------------------------------
        sg_deadline = deadline - final_phase_seconds
        a_sg, fast_sg = subgradient_descent(best_a.copy(),
                                            max_iters=500,
                                            step0=0.1,
                                            deadline=sg_deadline)
        if fast_sg + 1e-12 < best_fast:
            sg_val = evaluate_sequence(a_sg.tolist())
            if sg_val < best_val:
                best_val = sg_val
                best_a = a_sg.copy()
                best_fast = fast_sg
                print(f"[sub‑grad] improved evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final annealing / hill‑climbing phase (uses the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline
        print("[final‑phase] start annealing")
        a_final, fast_final = hill_climb_fast(best_a,
                                              max_iters=200_000,
                                              deadline=remaining)
        # final check against the true evaluator
        final_val = evaluate_sequence(a_final.tolist())
        if final_val < best_val:
            best_val = final_val
            best_a = a_final.copy()
            print(f"[final‑phase] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
