def project_simplex(v):
    """Proj onto probability simplex (nonnegative sum=1)."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max i s.t. u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Implementation details for projection onto simplex:

def project_simplex(v):
    """Project onto simplex {x: x>=0, sum x = 1}."""
    # similar to standard algorithm
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = v - theta
    w[w < 0] = 0
    return w

def project_simplex(v):
    """Projection of v onto the probability simplex."""
    # ensure v is a numpy array
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative, return uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # normalize to sum 1 (should already sum to 1 within tolerance)
    w_sum = w.sum()
    if w_sum <= 0:
        w = np.full(n, 1.0 / n)
    else:
        w /= w_sum
    return w

"""
Sub‑gradient minimisation of the maximal auto‑convolution.

The evaluation function to be minimised is

    f(a) = 2·n·max(a*a) / (sum(a)^2)                (n = len(a))

Because the objective is scale‑invariant we fix sum(a)=1 and
optimise the shape of a on the probability simplex.
For a given peak index k = argmax(a*a) the sub‑gradient of
max(a*a) w.r.t. a is 2·a_{k‑i}.  A descent direction is therefore

    d_i = -(a_{k‑i} - M)      with M = max(a*a)

which pushes mass away from entries that contribute heavily to the
largest convolution value and adds it to low‑mass entries.
The update is projected onto the simplex (non‑negative, sum=1)
by the classic O(n log n) Euclidean projection.
A simple back‑tracking line search on the step size guarantees
monotone decrease of the maximal convolution.

The algorithm consists of three phases:

1. **Initialisation** – start from the previously found best
   sequence (``height_sequence_1``) or from a uniform distribution.
2. **Projected sub‑gradient descent** – run until the time budget
   is almost exhausted, periodically evaluating the exact score.
3. **Guided hill‑climbing** – a short refinement (mass transfers)
   identical to the one used in the previous solution.

The whole routine respects a 1000 s budget; the best sequence found
so far is always returned.  All helper functions are top‑level and
no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i ≥ 0, sum w_i = 1 }.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # find the largest rho satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → return uniform simplex point
        return np.full(n, 1.0 / n, dtype=float)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    # renormalise (numerical safety)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n, dtype=float)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (same as previous submission)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small mass‑transfer hill‑climbing that accepts only improvements
    of the exact ``evaluate_sequence`` score.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05                     # initial fraction of mass to move

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two low‑mass destination indices
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the offending lag
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # compute feasible transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:                               # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val

# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Build a step‑function that (approximately) minimises ``evaluate_sequence``.
    The routine runs a projected sub‑gradient descent on the maximal
    autocorrelation, followed by a short guided hill‑climbing phase.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – keep it moderate to stay well within the budget.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise from the previous best if available, otherwise uniform.
    # --------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
    except Exception:
        a0 = np.full(n, 1.0 / n, dtype=float)

    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0.0:
        a0 = np.full(n, 1.0 / n, dtype=float)
    else:
        a0 /= s                                         # enforce sum = 1

    # --------------------------------------------------------------
    # Keep the best solution seen so far.
    # --------------------------------------------------------------
    best_a = a0.copy()
    try:
        best_val = evaluate_sequence(best_a.tolist())
    except Exception:
        best_val = float("inf")
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Projected sub‑gradient descent on max(a*a).
    # --------------------------------------------------------------
    a = a0.copy()
    step = 0.5                 # initial step size
    it = 0
    while time.time() < deadline:
        it += 1
        # full autocorrelation and its maximum
        conv = _full_convolution(a)
        max_b = conv.max()
        approx_val = 2.0 * n * max_b

        # exact evaluation every ~100 iterations
        if it % 100 == 0 or approx_val < best_val:
            try:
                exact_val = evaluate_sequence(a.tolist())
                if exact_val < best_val:
                    best_val = exact_val
                    best_a = a.copy()
                    print(f"[iter {it}] new best = {best_val:.6f}")
            except Exception:
                pass

        # pick a random index attaining the current maximum
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # ------- sub‑gradient g = a_{k-i} - max_b (zero‑mean direction) -------
        g = -max_b * np.ones(n, dtype=float)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1, dtype=np.intp)
            rev = k - idx
            g[idx] = a[rev] - max_b

        # ------- simple back‑tracking line search -------
        local_step = step
        improved = False
        for _ in range(12):
            a_candidate = a - local_step * g
            a_candidate = _project_simplex(a_candidate)
            conv_c = _full_convolution(a_candidate)
            max_c = conv_c.max()
            if max_c <= max_b + 1e-15:               # non‑increase (allow tiny tolerance)
                a = a_candidate
                max_b = max_c
                improved = True
                step = min(local_step * 1.1, 1.0)     # gently increase step after success
                break
            else:
                local_step *= 0.5                     # shrink step and retry

        if not improved:
            # No descent found – inject a tiny random perturbation
            step = max(step * 0.7, 1e-5)
            noise = rng.normal(scale=step * 0.02, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # Final guided hill‑climbing refinement (mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # --------------------------------------------------------------
    # Final report
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:            # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
