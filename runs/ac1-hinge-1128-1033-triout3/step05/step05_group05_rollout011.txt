  the probability simplex (∑a = 1).  The evaluation only depends on `max(b)`
  where `b = a ∗ a`, so we try to flatten the autoconvolution.

* **Phase 1 – Smoothed‑max gradient descent**  
  Instead of minimising the L²‑error we directly minimise a smooth
  approximation of `max(b)`:
  
      Lα(a) = (1/α)·log(∑ₖ exp(α·bₖ)) ,   b = a∗a

  Its gradient is
      ∇Lα = 2·conv(w, a_rev) ,   wₖ ∝ exp(α·bₖ)

  where `a_rev` is `a` reversed.  For a growing `α` this gradient becomes
  the true gradient of the maximum.  We perform many iterations with a
  linearly decaying step size and project back onto the simplex after each
  update.

* **Phase 2 – Targeted hill‑climbing**  
  Using the best vector from Phase 1 we run a deterministic local search.
  At each iteration we locate the current peak `p = argmax b`.  The pair
  `(i, p‑i)` that contributes most to this peak is chosen as a source
  index, a small amount of mass is moved to a randomly chosen target,
  and the convolution is updated in `O(n)` time.  Only moves that truly
  decrease the true objective `2·n·max(b)` are kept; after long
  stagnation a small random perturbation is applied.

Both phases are timed so the whole routine finishes well within the
1000 s budget.  Intermediate progress (current best evaluation) is printed
to aid debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex (non‑negative, sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase 1 – smoothed‑max gradient descent
# ----------------------------------------------------------------------
def _smooth_max_step(a: np.ndarray, alpha: float, step: float) -> np.ndarray:
    """
    Perform one gradient‑descent step on the smoothed‑max loss.
    Returns the updated vector (still on the simplex).
    """
    n = a.shape[0]

    # convolution a*a
    b = _full_convolution(a)

    # soft‑max weights (shifted for numerical stability)
    b_max = b.max()
    w = np.exp(alpha * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:                     # numerical safeguard
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # gradient = 2 * conv(w, a_rev)  (central slice of length n)
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev)          # length 3n‑2
    grad = 2.0 * conv[n - 1: n - 1 + n]

    # gradient step + projection
    a_new = a - step * grad
    a_new = _project_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Phase 2 – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _targeted_hillclimb(a_start: np.ndarray,
                        n: int,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> np.ndarray:
    """
    Local search that moves a small mass from the index contributing most
    to the current peak to a random target.  Returns the best vector seen
    before `deadline`.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac = 0.3               # at most this fraction of source mass is moved
    IMPROVE_LIMIT = 200_000      # stagnation trigger
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # locate the current peak of the convolution
        p = int(np.argmax(b))

        # compute contributions a_i * a_{p-i} for the peak
        cont = np.zeros(n, dtype=np.float64)
        i_min = max(0, p - (n - 1))
        i_max = min(n - 1, p)
        for i in range(i_min, i_max + 1):
            j = p - i
            cont[i] = a[i] * a[j]

        # source = index with largest contribution to the peak
        i_src = int(np.argmax(cont))
        if a[i_src] <= 1e-14:
            # nothing to move – pick a random source instead
            i_src = np.random.choice(n, p=a)

        # pick a target ≠ source
        j = np.random.randint(n)
        while j == i_src:
            j = np.random.randint(n)

        # amount of mass to transfer
        delta = a[i_src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        # cross‑terms 2·δ·a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept strictly improving moves
        if new_val < cur_val:
            a[i_src] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:07d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional perturbation to escape plateaus
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:07d}] stagnation – perturbing")

        iteration += 1

    return best_a

            a = _project_simplex(a)
