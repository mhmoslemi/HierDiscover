"""
Propose a step‑function sequence that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines three ideas:

1️⃣  **Warm‑start** – use the global variable ``height_sequence_1`` if it
    exists; otherwise start from a smooth sine‑shaped vector.
    The vector is projected onto the probability simplex (sum = 1).

2️⃣  **Linear‑programming push** – given the current sequence ``a`` with
    max‑convolution value ``M``, solve the LP

        maximise   1ᵀb
        subject to  conv(a, b) ≤ M    (entry‑wise)
                    b ≥ 0

    The LP seeks a distribution ``b`` that respects the same worst‑case
    convolution bound but carries more total mass (sum(b) > 1).  Scaling
    ``b`` back to sum 1 yields a candidate ``g`` that can lower the
    objective when mixed with the current ``a``.  The mixing factor ``t``
    is chosen adaptively so that the new max‑convolution never exceeds the
    old one.

3️⃣  **Fine‑grained polishing** – after a few LP steps (or when time runs
    out) perform a short projected‑gradient descent on a smooth
    ``soft‑max`` surrogate of the max‑function.  This clears remaining
    plateaus and sharpens the final score.

All heavy work (convolutions) is performed by real FFTs for speed.
The routine respects the 1000 s budget, keeps the best sequence seen,
and finally returns it in the format required by ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Duchi et al. (2008) simplex projection."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast linear convolution via real FFT (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = rfft(a, n=nfft)
    Fb = rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Solve the LP: maximise sum(b) subject to conv(a, b) <= M and b >= 0.
# Returns the optimal b (may be all‑zero if the solver fails).
# ----------------------------------------------------------------------
def _solve_lp_max_mass(a: np.ndarray, M: float):
    """Linear program for a fixed a and max‑convolution bound M.

    Returns the optimal vector b (shape n) or None on failure.
    """
    from scipy.optimize import linprog

    n = a.shape[0]
    rows = 2 * n - 1

    # Build A_ub where each column i contributes a[i] on rows i … i+n‑1.
    A_ub = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A_ub[i:i + n, i] = a[i]

    b_ub = np.full(rows, M, dtype=np.float64)

    # maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    # Use HiGHS (fast for dense problems)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30.0})
    if not res.success:
        # The caller will ignore a None result.
        return None
    return res.x


# ----------------------------------------------------------------------
# Polishing step: a few projected sub‑gradient iterations on the exact
# max‑convolution (binary mask of active entries).
# ----------------------------------------------------------------------
def _hard_max_polish(a: np.ndarray, iters: int = 3000, lr_start: float = 2e-5):
    n = a.shape[0]
    for k in range(iters):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        # binary mask of (almost) maximal entries
        tol = mx * 1e-9 + 1e-14
        mask = (conv >= mx - tol).astype(np.float64)
        w = mask / mask.sum()
        # gradient of the max‑term
        rev = a[::-1]
        grad_full = _full_convolution_fft(rev, w)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]
        a = a - lr_start * grad
        a = _project_simplex(a)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights (the step‑function)."""
    start_time = time.time()
    # keep a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 4.0
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1️⃣ Warm‑start: obtain an initial probability vector `a`.
    # --------------------------------------------------------------
    n_target = 2000                     # moderate size – memory / speed trade‑off
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # a smooth sine shape as a generic starting point
        base = np.sin(np.linspace(0, np.pi, n_target, dtype=np.float64))

    # Down‑sample / interpolate to the chosen length `n_target`
    if base.shape[0] != n_target:
        x_src = np.linspace(0, 1, base.shape[0], dtype=np.float64)
        x_tgt = np.linspace(0, 1, n_target, dtype=np.float64)
        a = np.interp(x_tgt, x_src, base)
    else:
        a = base.copy()

    a = np.maximum(0.0, a)                # ensure non‑negative
    if a.sum() == 0.0:
        a = rng.random(n_target)
    a = a / a.sum()                       # project onto the simplex
    n = a.shape[0]

    # --------------------------------------------------------------
    # Helper to evaluate the current objective (fast, no clipping)
    # --------------------------------------------------------------
    def _score(vec):
        conv = _full_convolution_fft(vec, vec)
        return 2.0 * n * conv.max()

    # keep track of the best vector seen so far
    cur_score = _score(a)
    best_vec = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, score={cur_score:.12f}")

    # --------------------------------------------------------------
    # Main optimisation loop: LP pushes followed by occasional mixing
    # --------------------------------------------------------------
    it = 0
    mix_factor = 0.3                     # start with a fairly aggressive mix
    while time.time() < deadline * 0.80:   # reserve ~20 % of time for polishing
        it += 1
        # current max‑convolution value
        conv = _full_convolution_fft(a, a)
        M = conv.max()

        # ------------------------------------------------------------------
        # Solve the LP that tries to pack more mass under the same bound M.
        # ------------------------------------------------------------------
        b = _solve_lp_max_mass(a, M)
        if b is None:
            print(f"[lp] iteration {it}: solver failed – falling back to gradient")
            break

        sum_b = b.sum()
        # If we cannot increase the total mass, stop the LP phase.
        if sum_b <= 1.0005:
            # No significant room to improve – exit the LP loop.
            print(f"[lp] iteration {it}: sum(b)≈{sum_b:.6f} (no improvement)")
            break

        # Scale b back to a probability vector (sum = 1)
        g = b / sum_b

        # ------------------------------------------------------------------
        # Evaluate the candidate `g`.  If it already improves the score we
        # accept it outright; otherwise we mix it with the current `a`.
        # ------------------------------------------------------------------
        score_g = _score(g)
        if score_g < cur_score:
            a = g
            cur_score = score_g
            print(f"[accept] iter {it:4d} improved via direct replace: score={cur_score:.12f}")
        else:
            # mix with a small factor to guarantee non‑increase
            t = mix_factor
            # ensure the mixing does not increase the max‑convolution:
            # we bound t by sqrt(M / (conv(g,g).max())) as a safety guard.
            conv_g = _full_convolution_fft(g, g)
            M_g = conv_g.max()
            if M_g > 0:
                t = min(t, np.sqrt(M / M_g) * 0.95)
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = _project_simplex(a_candidate)
            cand_score = _score(a_candidate)
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                print(f"[mix]   iter {it:4d} mixed improvement: score={cur_score:.12f}")
            else:
                # mixing failed – keep the old `a` and shrink `t`
                mix_factor = max(mix_factor * 0.7, 0.02)

        # ------------------------------------------------------------------
        # Record best ever
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[best]  iter {it:4d} new best: {best_score:.12f}")

        # ------------------------------------------------------------------
        # occasional random jitter to escape tiny plateaus
        # ------------------------------------------------------------------
        if it % 15 == 0 and rng.random() < 0.12:
            jitter = rng.normal(scale=0.005, size=n)
            a = a + jitter
            a = _project_simplex(a)
            cur_score = _score(a)    # recompute after jitter
            print(f"[jitter] iter {it}")

    # --------------------------------------------------------------
    # 3️⃣ Polishing stage: few hundred sub‑gradient steps on the exact max.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5:
        # Use a modest number of iterations that fits the remaining budget.
        iters = int(min(5000, remaining * 1000))  # roughly 1000 iter per second
        a = best_vec.copy()
        a = _hard_max_polish(a, iters=iters, lr_start=2e-5)
        final_score = _score(a)
        if final_score < best_score:
            best_score = final_score
            best_vec = a.copy()
            print(f"[polish] after hard‑max: score={best_score:.12f}")

    # --------------------------------------------------------------
    # Convert to the format required by `evaluate_sequence` and report.
    # --------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_vec.tolist()]
    # final evaluation (the host will also evaluate, this is just for logging)
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_eval:.12f}, length={len(final_seq)}")
    except Exception as exc:
        # In case `evaluate_sequence` is not in the namespace (unlikely)
        print("[result] evaluate_sequence not available – returning raw sequence")
    return final_seq
