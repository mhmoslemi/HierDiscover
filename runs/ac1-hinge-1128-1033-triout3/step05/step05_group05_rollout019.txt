"""
Propose a step‑function that (empirically) beats the previous best score.
The algorithm consists of two stages:

1.  *Focused gradient descent* – minimise a smooth approximation of
    max(conv(a,a)).  At each iteration the convolution `b = a*a` is
    computed, the `K` largest entries are selected and a soft‑max
    weighting is applied.  The gradient of the weighted sum can be
    written in closed form as a small number of shifted copies of `a`,
    which is updated in O(K·n) time.  After the gradient step the trial
    vector is projected onto the probability simplex (non‑negative,
    sum = 1).

2.  *Deterministic hill‑climbing* – starting from the best vector found
    in stage 1 we perform cheap mass‑transfer moves.  Moving a small
    amount `δ` from a source index `i` to a target `j` can be reflected
    on the full convolution `b` in O(n) time.  Only moves that strictly
    improve the true objective `2·n·max(b)` are accepted; after a long
    period without improvement a small random perturbation is applied
    to escape local minima.

Both phases respect the 1000 s time budget, keep the sequence length
moderate (`n = 5000`) and use only NumPy.  The final sequence is returned
as a list of non‑negative floats whose sum is 1, which is the scale that
`evaluate_sequence` expects (the score is invariant under scaling).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x≥0, Σx=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # find the rho value
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # return the uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0] = 0.0
    # renormalise (numerical drift)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Full (linear) convolution of a with itself (length 2·len(a)‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution x*x using FFT (output length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Gradient of a soft‑max weighted sum of the convolution
# ----------------------------------------------------------------------
def _grad_from_top_indices(a: np.ndarray, ks: np.ndarray,
                           ws: np.ndarray) -> np.ndarray:
    """
    Compute   g = 2 * Σ_{k∈ks} w_k * shift(a, k)
    where   shift(a, k)_i = a_{k-i}   if 0≤k-i<n else 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k, w in zip(ks, ws):
        if k < n:
            # indices i = 0 … k  →  a[k‑i]
            grad[: k + 1] += 2.0 * w * a[k::-1]
        else:
            # k ≥ n : i runs from k‑(n‑1) … n‑1
            start = k - (n - 1)                 # first valid i
            length = 2 * n - 1 - k              # number of valid i's
            grad[start:] += 2.0 * w * a[:length][::-1]
    return grad


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """
    Fast local search that moves a small amount of mass between coordinates
    while keeping Σa = 1. Returns the best sequence seen before *deadline*.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac = 0.4                           # up to 40 % of source mass may move
    stagnation_limit = 150_000
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # choose source – either the current largest entry or a random one
        # --------------------------------------------------------------
        if np.random.rand() < 0.85:
            i = int(np.argmax(a))
        else:
            i = np.random.choice(n, p=a)      # weighted by current mass

        if a[i] <= 1e-14:
            continue

        # choose a different target
        j = np.random.randint(n)
        if j == i:
            continue

        # amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # Incremental update of the full convolution  (O(n) work)
        # --------------------------------------------------------------
        b_new = b.copy()
        # cross‑terms  (2·δ·a) added / subtracted
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms (only three positions are affected)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept only strictly improving moves
        if new_val < cur_val:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # --------------------------------------------------------------
        # escape stagnation
        # --------------------------------------------------------------
        if no_improve > stagnation_limit:
            # re‑inject a small random perturbation around the best point
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a non‑negative step‑function under the given time budget.
    Returns a list of floats whose sum is 1 (the scaling does not affect the
    score returned by `evaluate_sequence`).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a safety margin

    # --------------------------------------------------------------
    # Problem size – 5000 gives a good trade‑off between expressivity
    # and speed (FFT size stays at 16384).
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # Initialise `a` – reuse a known good construction if available.
    # --------------------------------------------------------------
    try:
        # `height_sequence_1` may be defined in the global scope
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # Resample/interpolate to the new length
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)          # enforce Σa = 1  and a≥0
    best_a = a.copy()
    best_val = float('inf')

    # --------------------------------------------------------------
    # Phase 1 – focused gradient descent on a smooth approximation of max(b)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.60 * budget_s
    avg_b = 1.0 / (2 * n - 1)        # average value of the convolution (sum_a = 1)

    iteration = 0
    while time.time() < phase1_deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b

        # track best solution
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Grad {iteration:7d}] new best = {best_val:.6f}")

        # ----------------------------------------------------------
        # construct a soft‑max weighting over the K largest entries
        # ----------------------------------------------------------
        K = 30                               # number of top entries to keep
        if K >= b.shape[0]:
            top_idx = np.arange(b.shape[0])
        else:
            top_idx = np.argpartition(-b, K)[:K]

        top_vals = b[top_idx]
        # smoothing parameter – a fraction of the current excess over the mean
        eps = max((max_b - avg_b) * 0.35, 1e-12)

        # compute (stable) soft‑max weights
        exp_arg = (top_vals - max_b) / eps   # ≤ 0
        exp_arg = np.clip(exp_arg, -50, 0)   # avoid under/overflow
        w_raw = np.exp(exp_arg)
        w = w_raw / w_raw.sum()

        # ----------------------------------------------------------
        # gradient = 2 * Σ w_k * shift(a, k)
        # ----------------------------------------------------------
        grad = _grad_from_top_indices(a, top_idx, w)

        # ----------------------------------------------------------
        # step‑size schedule (linear decay)
        # ----------------------------------------------------------
        prog = (time.time() - start_time) / (phase1_deadline - start_time)
        step = 0.5 * (1.0 - prog) + 0.001 * prog

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional random jitter to help escape shallow plateaus
        if np.random.rand() < 0.002:
            a += np.random.normal(scale=0.015, size=n)
            a = _project_simplex(a)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – deterministic hill‑climbing on the exact objective.
    # --------------------------------------------------------------
    best_a = _hillclimb_phase(best_a, n, deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
