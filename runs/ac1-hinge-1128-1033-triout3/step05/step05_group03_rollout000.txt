    a_i ∝ (n‑i)^p , i = 0,…,n‑1

def _poly_seq(n: int, p: int) -> np.ndarray:
    """
    Return the normalised decreasing polynomial sequence (n‑i)^p for i=0..n‑1.
    """
    # (n, n‑1, … , 1) raised to the power p
    vals = np.arange(n, 0, -1, dtype=np.float64) ** p
    total = vals.sum()
    # Normalise to Σa = 1  (the objective is invariant under scaling)
    return vals / total

    best_seq = None
    best_score = float('inf')
    best_p = None

    # exponents to test – higher p gives lower bound but may suffer from
    # floating‑point underflow; we stop when the time budget is almost used.
    for p in range(1, 11):
        if time.time() >= deadline:
            break

        a = _poly_seq(n, p)
        # Clip for safety – the evaluator does this internally anyway.
        a = np.clip(a, 0.0, 1000.0)

        try:
            score = evaluate_sequence(a.tolist())
        except Exception as exc:  # pragma: no cover
            print("[error] evaluate_sequence raised:", exc)
            continue

        print(f"[p={p:2d}] score = {score:.9f}")

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            best_p = p

        # Early exit if we already beat the official target by a comfortable margin.
        if best_score <= 0.1:
            # no need to waste time testing larger exponents
            break

    if best_seq is None:
        # fallback – return a uniform distribution (should never happen)
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = evaluate_sequence(best_seq.tolist())
        best_p = 0

    print(f"[final] best exponent p = {best_p}, best score = {best_score:.9f}")

    # Return a plain Python list as required by the framework.
    return best_seq.tolist()
