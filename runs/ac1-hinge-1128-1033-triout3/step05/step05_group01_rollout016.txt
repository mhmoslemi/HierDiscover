def _guided_hill_climb_improved(a, cur_val, deadline, rng):
    n = len(a)
    step_frac = 0.05
    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))
        # compute contributions for this lag k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        # select up to 3 top indices
        if contrib.size == 0:
            continue
        # rank indices
        top_idx = np.argpartition(-contrib, min(3, contrib.size)-1)[:min(3, contrib.size)]
        # low mass indices candidates for destinations
        low_candidates = np.where(a < a.mean())[0]
        if low_candidates.size < 2:
            low_candidates = np.arange(n)
        # generate candidate moves
        best_candidate = None
        best_candidate_val = cur_val
        for idx in top_idx:
            i_max = int(i_range[idx])
            j_max = int(j_range[idx])
            # sample few destination pairs
            for _ in range(2):
                p, q = rng.choice(low_candidates, size=2, replace=False)
                attempts = 0
                while (p+q == k) and attempts < 5:
                    p, q = rng.integers(0, n, size=2)
                    attempts += 1
                # compute delta
                if i_max != j_max:
                    delta = min(a[i_max], a[j_max]) * step_frac
                    if delta <= 1e-14:
                        continue
                    new_a = a.copy()
                    new_a[i_max] -= delta
                    new_a[j_max] -= delta
                    new_a[p] += delta
                    new_a[q] += delta
                else:
                    delta = a[i_max] * step_frac
                    if delta <= 1e-14:
                        continue
                    new_a = a.copy()
                    new_a[i_max] -= delta
                    new_a[p] += delta
                if (new_a < 0).any():

"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Algorithm
---------
1. **Simple parametric warm‑start** – try a few exponential decays
   ``a_i ∝ exp(-λ·i)`` (λ on a log‑grid) and keep the best.
2. **Smooth surrogate optimisation** – minimise a log‑sum‑exp
   approximation of the max‑convolution using L‑BFGS‑B
   on the soft‑max parametrisation.  A continuation schedule for the
   smoothing parameter ε drives the solution towards the true objective.
3. **Random restarts** – perturb the best parameters and repeat the
   continuation on the finer part of the ε‑schedule.
4. **Sub‑gradient descent on the exact max‑convolution** – compute the
   sub‑gradient of the active lag and perform a projected gradient step
   onto the probability simplex (∑a=1, a≥0).  The step size is
   gradually reduced.
5. **Guided hill‑climbing** – a final short pass of targeted mass
   transfers (as in the reference code) to squeeze out any remaining
   improvement.

All steps respect the given time budget (default 1000 s).  The function
returns the best list of non‑negative heights found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)               # clean tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp surrogate) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Calculate the smooth surrogate
        Lε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)
    together with its gradient w.r.t. the soft‑max parameters w.
    """
    # soft‑max → feasible a (non‑negative, Σa = 1)
    w_shift = w - np.max(w)                # for numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # autocorrelation a*a
    conv = _full_convolution(a)            # length 2·n‑1
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    log_sum = np.logaddexp.reduce(shifted)  # stable log‑sum‑exp
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # weights for the smoothed max
    weights = np.exp(shifted)
    p = weights / weights.sum()            # shape (2·n‑1,)

    # gradient w.r.t. a
    p_rev = p[::-1]                        # reversed weights
    conv_ap = _conv_fft(a, p_rev)          # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]        # middle n entries
    grad_a = 4.0 * n * seg[::-1]           # shape (n,)

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)            # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Simple simplex projection (required for sub‑gradient steps)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # largest index where u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, cur_val: float,
                         deadline: float, rng: np.random.Generator):
    """
    Reduce the true objective by a projected sub‑gradient step on the
    active convolution lag.  The step size decays gradually.
    """
    n = a.shape[0]
    step = 0.08                     # initial step length
    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))
        s = k - (n - 1)             # shift relative to centre

        # sub‑gradient of the active lag
        d = np.zeros_like(a)
        if s >= 0:
            d[: n - s] += a[s:]
            d[s:]      += a[: n - s]
        else:
            s_abs = -s
            d[s_abs:]  += a[: n - s_abs]
            d[: n - s_abs] += a[s_abs:]

        # projected gradient step
        a_new = a - step * d
        a_new = _project_onto_simplex(a_new)

        # evaluate exact objective
        try:
            val_new = evaluate_sequence(a_new.tolist())
        except Exception:
            # highly unlikely – just skip this move
            continue

        if val_new < cur_val:
            a = a_new
            cur_val = val_new
            print(f"[subgrad] new best = {cur_val:.6f}")
            step = max(step * 0.90, 1e-6)      # accelerate shrinking after success
        else:
            step = max(step * 0.99, 1e-6)      # gentle decay on failure
    return a, cur_val

def _guided_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float, rng: np.random.Generator):
    """
    Targeted mass‑transfer moves that directly lower the exact objective.
    In each iteration a few candidate moves are tried and the best
    improvement (if any) is accepted.
    """
    n = a.shape[0]
    step_frac = 0.05            # initial fraction of the source mass to move

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions for the offending lag
        i0 = max(0, k - (n - 1))
        i1 = min(k, n - 1)
        i_range = np.arange(i0, i1 + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        if contrib.size == 0:
            continue

        # up to three source pairs with largest contribution
        n_src = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, n_src - 1)[:n_src]

        # candidate destination indices (low‑mass tail)
        low_mass = np.where(a < a.mean())[0]
        if low_mass.size < 2:
            low_mass = np.arange(n)

        best_candidate = None
        best_candidate_val = cur_val

        for src in top_idx:
            i_src = int(i_range[src])
            j_src = int(j_range[src])
            for _ in range(2):                          # two random destinations
                p, q = rng.choice(low_mass, size=2, replace=False)
                attempts = 0
                while (p + q == k) and attempts < 5:
                    p, q = rng.integers(0, n, size=2)
                    attempts += 1

                if i_src != j_src:
                    delta = min(a[i_src], a[j_src]) * step_frac
                    if delta <= 1e-14:
                        continue
                    new_a = a.copy()
                    new_a[i_src] -= delta
                    new_a[j_src] -= delta
                    new_a[p] += delta
                    new_a[q] += delta
                else:
                    delta = a[i_src] * step_frac
                    if delta <= 1e-14:
                        continue
                    new_a = a.copy()
                    new_a[i_src] -= delta
                    new_a[p] += delta

                if (new_a < 0).any():
                    continue

                try:
                    val = evaluate_sequence(new_a.tolist())
                except Exception:
                    continue

                if val < best_candidate_val:
                    best_candidate_val = val
                    best_candidate = new_a

        if best_candidate is not None:
            a = best_candidate
            cur_val = best_candidate_val
            print(f"[guided+] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.98, 1e-5)
        else:
            step_frac = max(step_frac * 0.995, 1e-5)

    return a, cur_val

            best_w = w.copy()
            print(f"[ε {eps:.1e}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Random restarts (perturb the best solution)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 10.0 and restart < 3:
        restart += 1
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.3, size=n)

        # only run the finer part of the schedule for speed
        for eps in eps_schedule[-8:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 150,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

        # evaluate this restart
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            best_w = w.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_a, best_val = _subgradient_descent(best_a, best_val,
                                                deadline, rng)

    # ------------------------------------------------------------------
    # 5️⃣ Final guided hill‑climbing refinement (few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_val = _guided_hill_climb(best_a, best_val,
                                              deadline, rng)

    # ------------------------------------------------------------------
    # Final report and return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
