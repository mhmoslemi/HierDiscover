"""
propose_candidate – a compact two‑phase optimiser for the max‑convolution ratio.

The optimiser works on a single CPU and respects a 1000 s wall‑time budget.
It proceeds through the following stages:

1. **Initialisation** – a length‑n non‑negative vector (n≈8000) is built as a
   random convex combination of several analytic “shapes” (Gaussian,
   semi‑circle, cosine, etc.) and projected onto the probability simplex
   (∑a = 1).  If a previous construction called ``height_sequence_1`` exists it
   is re‑used (truncated / padded) as a warm start.

2. **Soft‑max GD** – a fast gradient descent with temperature‑softened
   max‑convolution (soft‑max) drives the objective down quickly.  Momentum
   (0.9) and an exponential decay of the step‑size keep the iterates stable.

3. **Exact max‑sub‑gradient (Adam)** – after the temperature has cooled,
   the true sub‑gradient of the max‑entry is used together with an Adam‑style
   adaptive step‑size.  Small random perturbations are injected when progress
   stalls.

4. **Local hill‑climbing** – a cheap, pair‑wise mass‑transfer routine
   (randomly moving a tiny amount of probability mass from one index to another)
   explores the immediate neighbourhood of the best solution found so far.

Throughout the run the best sequence (and its score) is remembered.
When the time budget expires the routine returns the best sequence discovered,
clipped to the interval [0, 1000] as required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project ``v`` onto the simplex {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution via FFT (full linear convolution).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective and sub‑gradient utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return the evaluation function value for the current sequence."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    # sum(a) is 1 because we always keep it on the simplex
    return 2.0 * n * max_conv


def _max_subgradient(a: np.ndarray, eps: float = 1e-14) -> np.ndarray:
    """
    Compute an (averaged) sub‑gradient of the max‑convolution term.
    Returns g such that moving a ← a – η g decreases the max entry
    (for sufficiently small η).
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_c = conv.max()
    active = np.flatnonzero(conv >= max_c - eps)
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    else:                                   # fallback (should never happen)
        w[np.argmax(conv)] = 1.0
    rev_a = a[::-1]
    g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]
    return g


# ----------------------------------------------------------------------
# Random initial sequence generator
# ----------------------------------------------------------------------
def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length ``n`` and unit sum.
    The vector is a random convex combination of several smooth shapes.
    """
    x = np.arange(n, dtype=np.float64)

    # analytic shapes ----------------------------------------------------
    # Gaussian
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

    # Raised cosine (positive half‑cycle)
    cosine = np.cos(np.pi * (x + 0.5) / n)
    cosine = np.maximum(cosine, 0.0)

    # Semi‑circle
    t = 2.0 * x / (n - 1) - 1.0
    semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

    # Simple triangle
    triangle = np.minimum(x, n - 1 - x) + 1.0

    # Exponential decay (controlled by a random base)
    base = rng.uniform(0.9, 0.9995)
    expo = base ** x

    bases = [gauss, cosine, semicircle, triangle, expo]
    weights = rng.random(len(bases))
    weights /= weights.sum()
    a = sum(w * b for w, b in zip(weights, bases))

    # tiny random perturbation to avoid exact symmetry
    a += 0.001 * rng.random(n)

    a = np.maximum(a, 0.0)
    a = _project_simplex(a)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s budget and returns
    the best sequence discovered.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem sizes we will try (larger sizes give a lower theoretical bound)
    # ------------------------------------------------------------------
    candidate_ns = [8000, 6000, 10000]   # order matters – try the larger first

    best_val = np.inf
    best_seq = None

    # ------------------------------------------------------------------
    # Outer loop over chosen lengths
    # ------------------------------------------------------------------
    for n in candidate_ns:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a candidate (re‑use previous global if possible)
        # --------------------------------------------------------------
        if 'height_sequence_1' in globals():
            try:
                base = np.asarray(height_sequence_1, dtype=np.float64)
                if base.shape[0] != n:
                    if base.shape[0] > n:
                        base = base[:n]
                    else:
                        pad = np.zeros(n - base.shape[0], dtype=np.float64)
                        base = np.concatenate([base, pad])
                a = np.maximum(0.0, base)
                if a.sum() == 0.0:
                    a = _initial_sequence(n, rng)
                else:
                    a = _project_simplex(a)
            except Exception:
                a = _initial_sequence(n, rng)
        else:
            a = _initial_sequence(n, rng)

        # --------------------------------------------------------------
        # Keep track of the best solution seen for *this* n
        # --------------------------------------------------------------
        cur_val = _objective(a)
        best_val_n = cur_val
        best_seq_n = a.copy()
        print(f"[init] n={n}, score={cur_val:.12f}")

        # --------------------------------------------------------------
        # Phase 1 – softened max (soft‑max) gradient descent with momentum
        # --------------------------------------------------------------
        phase1_end = time.time() + 0.35 * (deadline - time.time())
        eta = 1.0                     # initial step size
        eta_min = 1e-6
        eta_decay = 0.9998

        temp = max(a.max(), 1e-6) * 5.0   # initial temperature for soft‑max
        temp_min = 1e-6
        temp_decay = 0.9997

        momentum = 0.9
        vel = np.zeros_like(a)

        stagn = 0
        it = 0

        while time.time() < phase1_end:
            it += 1

            # full convolution & current score
            conv = _full_convolution_fft(a, a)
            max_c = conv.max()
            cur_val = 2.0 * n * max_c   # sum(a) == 1

            if cur_val < best_val_n - 1e-12:
                best_val_n = cur_val
                best_seq_n = a.copy()
                stagn = 0
                print(f"[phase1 n={n} iter {it:5d}] new_best={best_val_n:.12f}  eta={eta:.2e}  temp={temp:.2e}")
            else:
                stagn += 1

            # soft‑max weights over the convolution vector
            w = np.exp((conv - max_c) / temp)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

            # gradient of the soft‑max loss
            rev_a = a[::-1]
            g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

            # momentum step
            vel = momentum * vel + (1.0 - momentum) * g

            # gradient update + projection onto the simplex
            a_new = a - eta * vel
            a = _project_simplex(a_new)

            # schedule decay
            eta = max(eta_min, eta * eta_decay)
            temp = max(temp_min, temp * temp_decay)

            # occasional random perturbation when stuck
            if stagn > 2000:
                noise = rng.normal(scale=0.003, size=n)
                a = _project_simplex(a + noise)
                stagn = 0
                eta = max(eta_min, eta * 1.2)
                print(f"[phase1 n={n} perturb] iter {it}")

            # extra re‑normalisation (numerical safety)
            if it % 5000 == 0:
                a = a / a.sum()

        # --------------------------------------------------------------
        # Phase 2 – exact max sub‑gradient descent with Adam
        # --------------------------------------------------------------
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        eta0 = 0.25                     # base learning rate for Adam

        m = np.zeros_like(a)           # first moment
        v = np.zeros_like(a)           # second moment
        beta1_pow = 1.0
        beta2_pow = 1.0
        t_adam = 0
        stagn = 0

        while time.time() < deadline:
            t_adam += 1
            beta1_pow *= beta1
            beta2_pow *= beta2

            # current objective
            conv = _full_convolution_fft(a, a)
            max_c = conv.max()
            cur_val = 2.0 * n * max_c

            if cur_val < best_val_n - 1e-12:
                best_val_n = cur_val
                best_seq_n = a.copy()
                stagn = 0
                print(f"[phase2 n={n} iter {t_adam:6d}] new_best={best_val_n:.12f}")
            else:
                stagn += 1

            # exact max sub‑gradient (averaged over all active entries)
            g = _max_subgradient(a)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

            a_candidate = a - step
            a_candidate = _project_simplex(a_candidate)

            a = a_candidate

            # occasional random kick to escape plateaus
            if stagn > 4000:
                noise = rng.normal(scale=0.002, size=n)
                a = _project_simplex(a + noise)
                stagn = 0
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow = 1.0
                beta2_pow = 1.0
                print(f"[phase2 n={n} perturb] iter {t_adam}")

            # Keep a well‑scaled (numerical safety)
            if t_adam % 4000 == 0:
                a = a / a.sum()

        # --------------------------------------------------------------
        # Local hill‑climbing refinement (very cheap moves)
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 10.0:
            # aim for roughly 2 µs per move → a few hundred thousand moves max
            max_moves = int(min(2e5, remaining * 300))   # very rough budget
            for move in range(max_moves):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                # transfer a small fraction of mass from i to j
                delta = 0.001 * a[i] * rng.random()
                if delta == 0.0:
                    continue
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # keep feasibility
                a_new = _project_simplex(a_new)

                new_val = _objective(a_new)
                if new_val < best_val_n - 1e-12:
                    best_val_n = new_val
                    best_seq_n = a_new.copy()
                    a = a_new
                else:
                    # reject: keep the old a
                    pass

        # --------------------------------------------------------------
        # Update global best across all n
        # --------------------------------------------------------------
        if best_val_n < best_val:
            best_val = best_val_n
            best_seq = best_seq_n.copy()
            print(f"[global] improved best to {best_val:.12f} with n={n}")

    # ------------------------------------------------------------------
    # Convert best sequence to the list form required by evaluate_sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback – should never happen
        best_seq = np.ones(1, dtype=np.float64)

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Optional logging of the final score
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")

    return final_seq
