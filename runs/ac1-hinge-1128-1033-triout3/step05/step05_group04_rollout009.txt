    eps_decay = 0.99995
    lr_decay = 0.99998
    best_val = float('inf')
    best_a = a.copy()
    stagn = 0
    it=0
    while time.time() < deadline:
         it+=1

"""
Propose a (near‑optimal) height sequence for the self‑convolution bound.

The optimizer works in three phases within the allotted time budget:

1️⃣  Smooth optimisation – a softened max (soft‑max with a temperature
    `eps`) is minimised by Adam on the probability simplex.  The
    temperature decays very slowly, allowing many fine‑grained updates.

2️⃣  Hard‑max polishing – once `eps` is tiny we switch to the true max
    (binary mask) and continue with a tiny constant Adam step to push the
    peak further down.

3️⃣  Random hill‑climbing – a short stochastic local‑search phase that
    moves a small amount of mass between two indices and keeps any
    improvement.  This often shaves the last few‑thousandths off the
    bound.

The routine respects a global `height_sequence_1` warm‑start if it exists,
otherwise builds a mixed “nice’’ shape.  All vectors are projected onto the
probability simplex after each update, guaranteeing non‑negativity and a
unit L¹‑norm (the score is scale‑invariant).  The best sequence found is
clamped to the limits required by `evaluate_sequence` before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real–valued inputs).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Core optimiser for a single length `n`.
# ----------------------------------------------------------------------
def _optimise_one_length(
    a: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run the three‑phase optimisation for a fixed length `n`.

    Returns the best sequence (as an array summing to 1) and its
    corresponding score 2·n·max(conv(a,a)).
    """
    n = a.shape[0]

    # ------------------------------ initialise ------------------------------
    # make sure we are on the simplex
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = rng.random(n)
    a = a / a.sum()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters (tuned for a long smooth phase)
    lr = 0.15                # initial learning‑rate
    lr_min = 1e-6
    lr_decay = 0.99997       # very slow decay
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # temperature for soft‑max (initially fairly high)
    conv = _full_convolution_fft(a, a)
    max_c = conv.max()
    eps = max(max_c * 0.7, 1e-12)          # start a bit warmer than before
    eps_min = max(max_c * 1e-8, 1e-12)    # push toward a very low temperature
    eps_decay = 0.99996

    best_val = 2.0 * n * max_c            # we keep a sum‑1 vector → denominator =1
    best_seq = a.copy()
    stagn = 0
    it = 0

    # ------------------------------------------------------------------
    # Phase 1 – smooth optimisation (soft‑max) – 85 % of the time budget.
    # ------------------------------------------------------------------
    smooth_deadline = time.time() + (deadline - time.time()) * 0.85
    while time.time() < smooth_deadline:
        it += 1

        # current convolution & score
        conv = _full_convolution_fft(a, a)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c

        # keep the best ever seen
        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_seq = a.copy()
            stagn = 0
            print(f"[smooth {it:6d}] new_best={best_val:.12f}  lr={lr:.2e}  eps={eps:.2e}")
        else:
            stagn += 1

        # soft‑max weights – a smooth approximation of the max
        w = np.exp((conv - max_c) / eps)
        w_sum = w.sum()
        if w_sum > 0.0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # gradient of the smooth max w.r.t. a
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update on a (projected onto simplex)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        beta1_t = beta1 ** it
        beta2_t = beta2 ** it
        m_hat = m / (1.0 - beta1_t)
        v_hat = v / (1.0 - beta2_t)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a + step
        a = _project_simplex(a)

        # schedules
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional random perturbation to escape possible plateaus
        if stagn > 4000:
            noise = rng.normal(scale=0.006, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            lr = max(lr_min, lr * 1.2)
            print(f"[perturb] iteration {it}")

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max polishing (binary mask of true maximal entries).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        conv = _full_convolution_fft(a, a)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c

        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_seq = a.copy()
            stagn = 0
            print(f"[polish {it:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # binary mask of (approximately) maximal entries
        tol = max_c * 1e-9 + 1e-14
        mask = (conv >= max_c - tol).astype(np.float64)
        w_sum = mask.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask / w_sum

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # tiny constant step (no momentum) – this works well when we are
        # already almost at the optimum.
        step = - 2e-5 * grad
        a = a + step
        a = _project_simplex(a)

        if stagn > 5000:
            # occasional jitter to avoid being stuck forever
            a = a + rng.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            stagn = 0
            print(f"[jitter] iteration {it}")

    # ------------------------------------------------------------------
    # Phase 3 – short stochastic hill‑climbing (swap a tiny amount of mass).
    # ------------------------------------------------------------------
    # Use the best sequence found so far as a starting point.
    a = best_seq.copy()
    cur_val = best_val
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # move a small epsilon from i to j (respecting non‑negativity)
        if a[i] <= 0.0:
            continue
        delta = rng.random() * min(a[i], 0.005)   # at most 0.5 % of a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalise (numeric drift)
        a_new /= a_new.sum()

        conv_new = _full_convolution_fft(a_new, a_new)
        new_val = 2.0 * n * conv_new.max()

        # accept if we improve the global best
        if new_val < best_val - 1e-13:
            best_val = new_val
            best_seq = a_new.copy()
            print(f"[hill {it:6d}] new_best={best_val:.12f}")

        # also accept if we improve the current point (helps random walk)
        if new_val < cur_val:
            a = a_new
            cur_val = new_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the platform.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the time budget, keeps the
    best sequence seen and finally returns it, clamped to the limits of
    `evaluate_sequence`.
    """
    deadline = time.time() + budget_s - 0.1  # small safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a length.  We keep the length of the warm‑start if it exists;
    # otherwise we use 5000, which gives a good trade‑off between quality
    # and speed.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = base.shape[0]
            a0 = np.maximum(0.0, base)
            if a0.sum() == 0.0:
                a0 = rng.random(n)
        except Exception:
            n = 5000
            a0 = None
    else:
        n = 5000
        a0 = None

    if a0 is None:
        # Build a mixed “nice’’ shape (Catalan‑like, sine, semicircle, Gaussian,
        # triangular) similar to the original script.
        x = np.arange(n, dtype=np.float64)

        # Catalan‑like decreasing sequence
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)
            catalan[k] = cur

        # assorted shapes
        sine = np.sin(np.pi * (x + 0.5) / n)
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
        triangle = np.minimum(x, n - 1 - x) + 1.0

        bases = [catalan, sine, semicircle, gauss, triangle]
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a0 = sum(w * b for w, b in zip(weights, bases))
        a0 = np.maximum(a0, 0.0)

    # ------------------------------------------------------------------
    # Optimise – we give 90 % of the remaining budget to the main optimiser,
    # the final 10 % is reserved for a quick sanity check with the official
    # `evaluate_sequence`.
    # ------------------------------------------------------------------
    optimise_deadline = time.time() + (deadline - time.time()) * 0.9
    best_seq, best_val = _optimise_one_length(a0, rng, optimise_deadline)

    # ------------------------------------------------------------------
    # Final evaluation (clamp to the limits required by `evaluate_sequence`).
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
