"""
Propose a step‑function sequence that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a simple parametric family (Beta‑shaped PDFs) as a strong
starting point and then refining it with a projected sub‑gradient
descent on the true (non‑smooth) objective.  The routine respects a
1000 s time budget, keeps the best sequence seen and finally returns
it in the format required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Duchi et al. (2008) – O(n) projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:                                   # all entries are same
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (no padding overhead)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using a real FFT (length = len(a)+len(b)-1)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Produce a symmetric Beta(α,α) density on a discrete grid and normalise.
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, n: int) -> np.ndarray:
    """Return a length‑n probability vector proportional to
    x^{α-1} (1-x)^{α-1} on the interior of (0,1)."""
    # centre points of the bins
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid exact zero at the boundaries (helps when α<1)
    eps = 1e-12
    w = (x * (1.0 - x) + eps) ** (alpha - 1.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback – uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / w_sum


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  The algorithm proceeds in three stages:

    1️⃣  Scan a simple 1‑parameter family (symmetric Beta densities) –
        additionally blend each with the uniform distribution.  This
        quickly discovers a very good shape (often beating 1.5).

    2️⃣  Refine the best shape with a projected sub‑gradient descent on the
        true ``max(conv)`` objective.  The gradient is obtained by
        differentiating the (almost) active convolution entries.
        A diminishing step‑size and occasional random jitter help escape
        plateaus.

    3️⃣  Return the best sequence encountered, clamped to the limits required
        by ``evaluate_sequence``.
    """
    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0          # small safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size (use previous solution if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = base.shape[0]
        except Exception:
            n = 5000
    else:
        n = 5000

    # ------------------------------------------------------------------
    # Pre‑compute a uniform reference (useful for mixing)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Keep track of the best solution seen so far
    # ------------------------------------------------------------------
    best_seq = uniform.copy()
    conv = _full_convolution_fft(best_seq, best_seq)
    best_val = 2.0 * n * conv.max()          # sum(best_seq)=1 → denominator = 1
    print(f"[init] n={n}, initial_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Stage 1 – Scan the Beta‑family (α ∈ (0,5]) and a few blends with uniform
    # ------------------------------------------------------------------
    alphas = np.concatenate([
        np.linspace(0.15, 3.0, 57),               # dense linear part
        np.logspace(np.log10(0.05), np.log10(5.0), 30)  # log‑scale extremes
    ])
    alphas = np.unique(alphas)                  # remove duplicates

    blend_factors = np.linspace(0.0, 1.0, 7)     # λ for λ*Beta + (1-λ)*Uniform

    for alpha in alphas:
        if time.time() > deadline:
            break
        beta_shape = _beta_shape(alpha, n)      # already normalised (sum=1)
        for lam in blend_factors:
            # convex combination keeps the sum = 1 automatically
            cand = lam * beta_shape + (1.0 - lam) * uniform
            conv = _full_convolution_fft(cand, cand)
            score = 2.0 * n * conv.max()
            if score < best_val:
                best_val = score
                best_seq = cand.copy()
                print(f"[scan] α={alpha:.4f}, λ={lam:.2f} → new_best={best_val:.12f}")

    # ------------------------------------------------------------------
    # Stage 2 – Projected sub‑gradient descent on the true objective.
    # ------------------------------------------------------------------
    a = best_seq.copy()                         # always a probability vector
    lr = 0.12                                    # initial step size
    lr_min = 1e-6
    lr_decay = 0.9995
    stagn = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- evaluate current objective ---------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_val - 1e-13:
            best_val = cur_score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it:6d}] new_best={best_val:.12f}  lr={lr:.2e}")
        else:
            stagn += 1

        # ---- sub‑gradient of the (almost) active entries ---------------
        # mask of maximal (or almost maximal) convolution entries
        tol = max_conv * 1e-9 + 1e-15
        mask = (conv >= max_conv - tol).astype(np.float64)
        w_sum = mask.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask / w_sum

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]                  # slice to size n

        # ---- projected update -------------------------------------------
        step = - lr * grad
        a_new = a + step
        a = _project_simplex(a_new)

        # ---- schedules ---------------------------------------------------
        lr = max(lr_min, lr * lr_decay)

        # ---- occasional random jitter to escape plateaus -----------------
        if stagn > 4000:
            a = a + rng.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            stagn = 0
            lr = max(lr_min, lr * 1.3)
            print(f"[jitter] iteration {it}")

        # ---- sanity renormalisation --------------------------------------
        if it % 5000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Convert to the format expected by ``evaluate_sequence`` and report.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
