"""
Propose a non‑negative step‑function (list of heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a three‑phase strategy:

1. **Projected Adam on the exact max‑subgradient** – we optimise on the
   probability simplex (sum = 1) with a standard Adam update.  The
   sub‑gradient is taken from the *worst* convolution entry (or a small
   active set), which is a more aggressive descent direction than the
   soft‑max used in the reference code.

2. **Smooth‑and‑project** – after each Adam step we apply a short
   moving‑average filter to the vector (encouraging smooth shapes that
   give flatter autocorrelation) and re‑project onto the simplex.

3. **Light simulated‑annealing refinement** – once the Adam phase has
   exhausted the time budget we perform cheap local moves that transfer a
   tiny amount of mass between two coordinates.  Acceptance follows a
   linear temperature schedule so the method can climb out of shallow
   local minima.

The algorithm automatically adapts the problem size (default ``n = 12000``)
and, when possible, starts from the previously discovered best sequence
``height_sequence_1`` (resampled to the chosen length).  All intermediate
scores are printed; the best sequence seen so far is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x | x >= 0, sum(x) = 1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative height sequence that (approximately) minimizes
    `evaluate_sequence`.  The routine respects a `budget_s` second time
    budget, prints progress and returns the best sequence found.
    """
    start_time = time.time()
    # a small safety margin – we stop a few seconds before the deadline
    deadline = start_time + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a few ten‑thousands give more freedom without being
    # prohibitively expensive.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise a candidate.
    # If a previous good sequence exists we up‑sample / down‑sample it.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            old_n = base.shape[0]
            # linear interpolation to the new size
            x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(x_new, x_old, base)
            a = np.maximum(a, 0.0)
            if a.sum() > 0.0:
                a = a / a.sum()
            else:
                a = rng.random(n)
                a = a / a.sum()
        except Exception:
            a = rng.random(n)
            a = a / a.sum()
    else:
        a = rng.random(n)
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Keep the best solution seen so far.
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv          # sum(a) == 1
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the exact max‑subgradient.
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eta = 0.2                     # initial step size
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = 1.0, 1.0
    it = 0
    phase1_end = start_time + 0.35 * budget_s

    while time.time() < phase1_end:
        # current convolution and objective value
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[phase1] iter {it:6d} new_best={best_val:.12f}")

        # sub‑gradient: focus on the (a few) entries attaining the maximum
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(active.size, 1)   # uniform weight on the active set

        # gradient of the max‑entry (cross‑correlation with the weight vector)
        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = eta * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_candidate = a - step

        # ------------------------------------------------------------------
        # Smooth‑and‑project: a tiny moving average reduces spurious spikes.
        # ------------------------------------------------------------------
        a_candidate = np.convolve(a_candidate, np.ones(3) / 3.0, mode='same')

        # simple decay of the learning rate
        eta = max(1e-5, eta * 0.9996)

        # occasional random perturbation to escape flat plateaux
        if it % 4000 == 0:

            a = _project_simplex(a + noise)
            print(f"[phase1] perturbation at iter {it}")

    # ------------------------------------------------------------------
    # Phase 2 – refined Adam with a stronger smoothing kernel.
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    while time.time() < phase2_end:
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[phase2] iter {it:6d} new_best={best_val:.12f}")

        # tight active set (still the exact maximum)
        tol = 1e-13
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(active.size, 1)

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update (same hyper‑parameters, just a smaller step)
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = eta * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_candidate = a - step

        # a slightly wider averaging filter – it favours very smooth profiles
        a_candidate = np.convolve(a_candidate, np.ones(5) / 5.0, mode='same')
        a = _project_simplex(a_candidate)

        # adapt learning rate a bit more gently
        eta = max(1e-5, eta * 0.9998)

        if it % 3000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            a = _project_simplex(a + noise)
            print(f"[phase2] perturbation at iter {it}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing style local swaps.
    # ------------------------------------------------------------------
    # Pre‑compute the current objective once for the loop.
    conv = _full_convolution_fft(a, a)
    cur_val = 2.0 * n * conv.max()
    while time.time() < deadline:
        # pick two distinct positions and try to move a tiny amount of mass
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # how much can we move? at most a fraction of the source mass
        max_delta = min(a[i], 0.01)     # 1 % of a[i] is a safe upper bound
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate the new candidate (full convolution – cheap enough for a few ks)
        conv_new = _full_convolution_fft(a_new, a_new)
        new_val = 2.0 * n * conv_new.max()

        # linear temperature schedule: high at the start of this phase,
        # drops to near‑zero at the end.
        remaining = deadline - time.time()
        total_phase3 = deadline - phase2_end
        T = max(1e-6, remaining / total_phase3)  # in [0,1]

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp((cur_val - new_val) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

            a = a_new
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_seq = a.copy()
                print(f"[anneal] iter {it} new_best={best_val:.12f}")

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found – clip to the limits required by
    # `evaluate_sequence` and convert to a plain Python list.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Optional final logging (won’t affect the returned value)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence could not be called")

    return final_seq
