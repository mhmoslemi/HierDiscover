"""
Simulated‑annealing search on the probability simplex.

The algorithm keeps an exact self‑convolution of a non‑negative vector
`a` (∑a_i = 1) and repeatedly performs mass‑transfer moves

    a_i ← a_i – δ ,   a_j ← a_j + δ          (i ≠ j)

The convolution is updated in O(n) with the analytic formula
used in the reference hill‑climber.  Moves that increase the
objective are occasionally accepted according to a linear cooling
schedule, allowing the search to escape local minima.  The best
vector seen during the run is returned.

The implementation is deliberately lightweight (no external solvers)
and works for sequences of a few thousand entries within the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast full convolution using real FFT (used for occasional re‑sync)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y (length len(x)+len(y)-1) using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # clip tiny negatives caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Simulated‑annealing / hill‑climbing hybrid that searches for a
    step‑function (non‑negative list) minimising ``evaluate_sequence``.
    Returns the best list discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the framework
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise a – use the provided best construction if available
    # --------------------------------------------------------------
    try:
        init_seq = height_sequence_1               # noqa: F821
    except NameError:
        init_seq = None

    if init_seq is not None:
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.shape[0]
        # normalise to a probability vector (scaling does not affect the score)
        s = a.sum()
        if s == 0.0:
            a.fill(1.0 / n)
        else:
            a /= s
    else:
        # modest length – large enough to give a rich shape but cheap to update
        n = 1500
        a = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Initialise the exact convolution and bookkeeping
    # --------------------------------------------------------------
    conv = _full_convolution_fft(a, a)          # length 2n‑1
    cur_max = conv.max()
    cur_obj = float(2.0 * n * cur_max)          # current objective value
    best_obj = cur_obj
    best_a = a.copy()

    print(f"[init] n={n}, initial objective={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for the 1000 s budget)
    # --------------------------------------------------------------
    max_frac = 0.30          # start moving up to 30 % of source mass
    min_frac = 0.01          # shrink to ~1 % near the end
    jitter_prob = 0.015      # probability of a large “jitter” move each iteration
    recompute_interval = 5000  # full FFT recompute to avoid drift

    # temperature schedule (objective units)
    T0 = cur_obj * 0.30          # initial temperature
    T_end = cur_obj * 1e-5       # final temperature (practically zero)

    iteration = 0
    last_improve_iter = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        progress = (now - start_time) / (deadline - start_time)

        # ----------------------------------------------------------
        # Annealing / move‑size schedules
        # ----------------------------------------------------------
        T = T0 * (1.0 - progress) + T_end * progress
        move_frac = max_frac * (1.0 - progress) + min_frac * progress

        # ----------------------------------------------------------
        # Choose source index – the coefficient that contributes most
        # to the current peak of the convolution.
        # ----------------------------------------------------------
        k_star = int(conv.argmax())
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        partner = k_star - idx
        products = a[idx] * a[partner]
        i_src = int(idx[products.argmax()])          # source index

        # ----------------------------------------------------------
        # Choose a destination index (different from i_src)
        # ----------------------------------------------------------
        j_dst = np.random.randint(n - 1)
        if j_dst >= i_src:
            j_dst += 1

        # ----------------------------------------------------------
        # Propose a small mass‑transfer move
        # ----------------------------------------------------------
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue

        a_old = a.copy()                 # we need the old vector for the update
        delta2 = delta * delta

        # incremental update of the convolution (exact)
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[j_dst:j_dst + n] += 2.0 * delta * a_old
        conv[2 * i_src] += delta2
        conv[2 * j_dst] += delta2
        conv[i_src + j_dst] -= 2.0 * delta2

        # update the coefficients
        a[i_src] -= delta
        a[j_dst] += delta

        # ----------------------------------------------------------
        # Evaluate new objective
        # ----------------------------------------------------------
        new_max = conv.max()
        new_obj = float(2.0 * n * new_max)
        diff_obj = new_obj - cur_obj

        # Acceptance rule (SA)
        # ----------------------------------------------------------
        accept = False
        if diff_obj <= 0.0:
            accept = True
        else:
            # uphill move – accept with probability exp(-Δ/T)
            if T > 0.0:
                prob = np.exp(-diff_obj / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:

            cur_max = new_max
            if cur_obj < best_obj:

                print(f"[{iteration:7d}] new best = {best_obj:.6f}")

            a[j_dst] = a_old[j_dst]

            conv[j_dst:j_dst + n] -= 2.0 * delta * a_old
            conv[2 * i_src] -= delta2
            conv[2 * j_dst] -= delta2
            conv[i_src + j_dst] += 2.0 * delta2

        # ----------------------------------------------------------
        # Occasionally perform a large “jitter” move to escape plateaus
        # ----------------------------------------------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move up to half of the source mass (larger than the regular step)
            delta_j = 0.5 * a[src] * np.random.rand()
            if delta_j > 0.0:
                a_tmp = a.copy()
                delta2_j = delta_j * delta_j
                conv[src:src + n] -= 2.0 * delta_j * a_tmp
                conv[dst:dst + n] += 2.0 * delta_j * a_tmp
                conv[2 * src] += delta2_j
                conv[2 * dst] += delta2_j
                conv[src + dst] -= 2.0 * delta2_j

                a[src] -= delta_j
                a[dst] += delta_j

                new_max_j = conv.max()
                new_obj_j = float(2.0 * n * new_max_j)
                diff_obj_j = new_obj_j - cur_obj

                accept_j = False
                if diff_obj_j <= 0.0:
                    accept_j = True
                else:
                    if T > 0.0:
                        if np.random.rand() < np.exp(-diff_obj_j / T):
                            accept_j = True

                if accept_j:
                    cur_obj = new_obj_j
                    cur_max = new_max_j
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{iteration:7d}] jitter improved best = {best_obj:.6f}")
                        last_improve_iter = iteration
                else:
                    # revert jitter
                    a = a_tmp
                    conv[src:src + n] += 2.0 * delta_j * a_tmp
                    conv[dst:dst + n] -= 2.0 * delta_j * a_tmp
                    conv[2 * src] -= delta2_j
                    conv[2 * dst] -= delta2_j
                    conv[src + dst] += 2.0 * delta2_j

        # ----------------------------------------------------------
        # Periodic full recomputation to avoid numerical drift
        # ----------------------------------------------------------
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a, a)

        # ----------------------------------------------------------
        # Simple stagnation detection – occasionally renormalise
        # ----------------------------------------------------------
        if iteration - last_improve_iter > 300_000:
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)
            last_improve_iter = iteration

    # ----------------------------------------------------------------------
    # Final reporting
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())

    except Exception as exc:                # pragma: no cover

    return best_a.tolist()
