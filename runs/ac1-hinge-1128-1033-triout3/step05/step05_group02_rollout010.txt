def lp_mix_step(a, t):
    # compute M
    conv = np.convolve(a, a)
    M = conv.max()
    # build A
    A = build_A(a)
    # solve LP
    n = len(a)
    c = -np.ones(n)  # maximize sum
    b_ub = np.full(n, M)
    # Solve using linprog
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        print("LP failed:", res.message)
        return a
    b = res.x
    # Normalize b
    sum_b = b.sum()
    if sum_b == 0:
        return a
    g = b / sum_b
    # mix
    new_a = (1 - t) * a + t * g
    # Re-normalize
    new_a = np.maximum(new_a, 0)
    new_a = new_a / new_a.sum()
    return new_a

def _binary_convolution_matrix(n):
    # Build sparse matrix B (shape (2n-1, n)) where B_{k,i}=1 if i <= k < i+n
    rows = []
    cols = []
    data = []
    for i in range(n):
        rows.extend(range(i, i + n))   # i through i+n-1
        cols.extend([i] * n)
        data.extend([1.0] * n)

This submission builds a step‑function by repeatedly solving a *linear program* that
maximises the total mass that can be placed under the *current* peak of the
self‑convolution.  The obtained vector is mixed with the current one,
re‑ordered anti‑symmetrically, and finally polished with a tiny hill‑climb
(mass‑transfer + swaps).  

The method is completely different from the L‑BFGS approach that was used
before, yet it respects the 1000 s time budget and works with any length `n`.
All helper routines are defined at top level, there are no closures or lambdas,
and the code uses only `numpy`, `scipy.sparse` and `scipy.optimize.linprog`
(which is part of the standard SciPy installation).

```python
"""
propose_candidate – construct a non‑negative step function that minimises
evaluate_sequence.

The algorithm proceeds in three stages:

1. **Initialisation** – start from a known construction (if available) or from a
   random Dirichlet sample, normalised to sum = 1.
2. **LP‑based refinement** – repeat
      • compute M = max(a∗a)
      • build the full convolution matrix C(a)   (C_{k,i}=a_i for k∈[i,i+n‑1])
      • solve  maximise Σ b_i   s.t. C(a)·b ≤ M , b≥0
      • normalise b to unit sum and mix   a ← (1‑t)·a + t·b
      • optionally anti‑symmetric reorder
   until time runs out or the peak does not improve.
3. **Local hill‑climb** – random mass transfers and swaps that strictly improve
   the objective, executed until the deadline.

The linear programme is tiny (≈ n variables, 2n‑1 constraints) and solved with
SciPy’s HiGHS backend; the matrix C(a) is built as a sparse CSR matrix on each
iteration.  The whole search finishes within the 1000 s budget and returns the
best sequence found.
"""

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: fast full convolution via FFT (real‑valued vectors)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """
    Linear convolution of x with itself (full length 2·len(x)‑1) using FFT.
    The result is clipped at zero to avoid tiny negative artefacts.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may give -1e‑16; eliminate it.
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper: build the full convolution constraint matrix.
#    For a fixed a,   (C(a)·b)_k = Σ_i a_i·b_{k-i}   with zero‑padding.
# ----------------------------------------------------------------------
def _build_convolution_matrix_full(a: np.ndarray) -> sp.csr_matrix:
    """
    Return a sparse matrix C of shape (2n‑1, n) such that C·b = a * b
    (full linear convolution).  Row k contains a_i for all i with
    i ≤ k < i+n, otherwise zero.
    """
    n = a.shape[0]
    total = n * n                     # each column i contributes n rows
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)

    idx = 0
    for i in range(n):
        # rows i, i+1, …, i+n‑1  (length n)
        rows[idx: idx + n] = np.arange(i, i + n, dtype=np.int32)
        cols[idx: idx + n] = i
        data[idx: idx + n] = a[i]
        idx += n

    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Helper: anti‑symmetric reordering (large ↔ small pairing)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Reorder a so that the largest and smallest values are placed symmetrically."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large → left side
        new_a[n - 1 - i] = sorted_vals[i]      # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]        # centre stays unchanged
    return new_a


# ----------------------------------------------------------------------
# Helper: tiny hill‑climbing refinement (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       cur_best: float,
                       deadline: float,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Randomly move a small fraction of mass between two positions or swap
    two values.  Accept only *strictly* better candidates.
    """
    cur_a = a.copy()
    cur_val = cur_best
    rng = np.random.default_rng()
    while time.time() < deadline:
        # 20 % chance of a pure swap, else a mass transfer
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            # transfer up to 5 % of donor’s mass
            delta = cur_a[i] * 0.05 * rng.random()
            if delta <= 0.0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val
```
