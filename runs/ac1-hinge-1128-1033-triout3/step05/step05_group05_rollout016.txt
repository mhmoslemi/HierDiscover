def evaluate_sequence(sequence):
  # security checks...
  sequence = [float(x) for x in sequence]
  # protect negative: sequence = [max(0, x) for x in sequence]
  # cap to 1000.0: sequence = [min(1000.0, x) for x in sequence]

"""
Propose a non‑negative step function that (approximately) minimizes
`evaluate_sequence`.  The algorithm consists of three phases:

1.  Smooth‑max Adam optimisation.  We minimise a differentiable proxy of
    `max(a∗a)` given by a soft‑max with temperature β.  β is increased from
    10 to 2·10⁴ while the learning rate decays, and the iterates are
    projected onto the probability simplex (and symmetrised).

2.  Sub‑gradient descent on the true `max` (using the current peak index)
    with a decreasing step size.

3.  Refined hill‑climbing.  Moves transfer a small amount of mass from a
    “heavy” index to a “light” one, updating the full convolution incrementally.
    The admissible fraction of moved mass shrinks as time goes on, and
    occasional random perturbations escape stagnation.

All phases respect the 1000 s budget and keep the sum of the coefficients = 1,
so the evaluation value is invariant under scaling.  The best sequence found
is returned as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0                # cumulative sum minus 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Smooth‑max gradient (soft‑max of the convolution)
# ----------------------------------------------------------------------
def grad_smooth_max(a: np.ndarray, beta: float):
    """Gradient of the β‑soft‑max of a∗a w.r.t. a."""
    n = a.shape[0]
    b = full_convolution_self(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights (stable)
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w /= w_sum
    # gradient = 2 * (w ⋆ a_rev) on the central slice
    a_rev = a[::-1]
    conv = conv_fft(w, a_rev)                       # length 3n‑2
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, b


# ----------------------------------------------------------------------
# Sub‑gradient of the true max (uses the current peak index)
# ----------------------------------------------------------------------
def subgrad_max(a: np.ndarray):
    """Sub‑gradient of max(a∗a) and the full convolution."""
    n = a.shape[0]
    b = full_convolution_self(a)
    max_b = b.max()
    k = int(np.argmax(b))                # index of the current peak
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    return g, b


# ----------------------------------------------------------------------
# Deterministic hill‑climbing with adaptive step size
# ----------------------------------------------------------------------
def hillclimb_phase(a_start: np.ndarray,
                    n: int,
                    start_time: float,
                    deadline: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass between coordinates.
    Returns the best sequence seen before *deadline*.
    """
    a = a_start.copy()
    b = full_convolution_self(a)
    cur_val = 2.0 * n * b.max()

    # adaptive parameters
    max_frac_start = 0.30
    max_frac_end = 0.001
    IMPROVE_LIMIT = 120_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear decay of the fraction of source mass we are allowed to move
        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # Choose a source index – mostly the current largest coefficient
        if np.random.rand() < 0.80:
            i = int(np.argmax(a))
        else:
            i = int(np.random.choice(n, p=a))

        if a[i] <= 1e-15:
            continue

        # Choose a target index – either greedy (based on contribution to the peak)
        # or random, to keep exploration.
        if np.random.rand() < 0.60:
            # greedy selection based on contribution to the current max convolution entry
            k = int(np.argmax(b))
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a * a[idx[mask]]
            i = int(np.argmax(contrib))                     # heavy contributor
            j = int(np.argmin(contrib))                     # light contributor
            if j == i:
                j = (i + np.random.randint(1, n)) % n
        else:
            j = np.random.randint(n)
            while j == i:
                j = np.random.randint(n)

        # Amount to transfer (fraction of the source mass)
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # Incremental update of the full convolution (O(n) work)
        b_new = b.copy()
        # linear terms: 2·δ·a shifted
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Accept only strictly improving moves
        if new_val < cur_val - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            # Record a new global best
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:07d}] new best = {best_val:.12f}")
        else:
            no_improve += 1

        # If we have been stuck for a while, inject a tiny random perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = np.maximum(a, 0.0)
            a = project_simplex(a)
            b = full_convolution_self(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:07d}] stagnation – perturbing")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights under the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 provides a good balance of expressivity and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise a – reuse a known good construction if available.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if old.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, old)
        else:
            a = old.copy()
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = a / a.sum()
    a = project_simplex(a)
    # Enforce symmetry – the optimum is expected to be symmetric.
    a = (a + a[::-1]) / 2.0

    # --------------------------------------------------------------
    # Keep track of the best solution seen so far.
    # --------------------------------------------------------------
    b = full_convolution_self(a)
    best_val = 2.0 * n * b.max()
    best_a = a.copy()
    print(f"[Init] n={n}, evaluation={best_val:.12f}")

    # --------------------------------------------------------------
    # Phase 1 – smooth‑max Adam optimisation (≈55 % of the budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    while time.time() < phase1_end:
        progress = (time.time() - start_time) / (phase1_end - start_time)
        beta = 10.0 + (20000.0 - 10.0) * progress          # temperature schedule
        lr = 0.20 * (1.0 - progress) + 0.005 * progress   # learning‑rate schedule
        grad, b = grad_smooth_max(a, beta)
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = project_simplex(a_candidate)
        a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
        a = a_candidate
        # evaluate after the step
        b = full_convolution_self(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Phase1 {it:07d}] new best = {best_val:.12f}")
        it += 1

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on the true max (≈25 % of the budget)
    # --------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s
    it2 = 0
    while time.time() < phase2_end:
        prog2 = (time.time() - phase1_end) / (phase2_end - phase1_end)
        eta = 0.05 * (1.0 - prog2) + 0.001 * prog2        # step size schedule
        g, b = subgrad_max(a)
        a_candidate = a - eta * g
        a_candidate = project_simplex(a_candidate)
        a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
        a = a_candidate
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Phase2 {it2:07d}] new best = {best_val:.12f}")
        it2 += 1

    # --------------------------------------------------------------
    # Phase 3 – refined hill‑climbing (remaining budget)
    # --------------------------------------------------------------
    best_a, best_val = hillclimb_phase(best_a, n, start_time, deadline,
                                       best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.12f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
