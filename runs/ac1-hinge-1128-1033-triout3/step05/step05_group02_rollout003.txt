"""
Propose a step‐function minimizing ``evaluate_sequence``.

The algorithm builds on the best known construction (available as
``height_sequence_1`` when present) and then applies a dedicated
gradient–descent phase that directly reduces the peak of the
self‑convolution.  The gradient of the (non‑smooth) objective
``max(conv)`` w.r.t. the vector a is, for the currently maximal lag
k*,  g_i = a_{k*‑i} (zero outside the admissible index range).  An
exponential (mirror‑descent) update

    a_i ← a_i·exp(−α·g_i)

preserves non‑negativity, after which the vector is projected back onto
the probability simplex.  The step size α is gradually reduced.
Occasionally the anti‑symmetric re‑ordering heuristic is applied,
followed by a fast hill‑climbing refinement (random swaps / mass
transfers).  The whole procedure respects the 1000 s budget and returns
the best sequence found.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negatives that may appear due to rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small ones when the sequence is reversed.
    This heuristic often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Simple hill‑climb: random swaps (20 %) and mass transfers (80 %).
    Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex (sum to 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------- smooth surrogate (log‑sum‑exp) used by L‑BFGS ------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)  (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability distribution over k

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # extract the middle n entries that correspond to the derivative
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimizes ``evaluate_sequence`` under the 1000 s time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a 5 s safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size – a moderate length gives a good balance
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise from a previously discovered construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)   # uniform start

    # enforce feasibility (non‑negative, normalise to sum=1)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    # --------------------------------------------------------------
    # Evaluate the initial candidate
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – a few short L‑BFGS runs on the smooth surrogate (optional)
    # ------------------------------------------------------------------
    # The previous run already achieved 1.512..., but a brief extra
    # L‑BFGS refinement can sometimes push the solution a little further.
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-10), num=9)

    w = np.log(best_a + 1e-12)                 # soft‑max parameters (log‑space)
    restart = 0
    while time.time() < deadline - 30.0 and restart < 8:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        # fresh randomisation of the starting point every few restarts
        if restart % 4 == 0:
            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 150,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # convert back to probabilities
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = evaluate_sequence(a_candidate.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # occasional anti‑symmetric re‑ordering
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – direct max‑lag gradient descent (core of this submission)
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    # initial step size; will be reduced slowly
    alpha0 = 0.20

    while time.time() < deadline - 10.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())
        max_val = conv[max_k]

        # gradient of the (non‑smooth) max‑conv w.r.t. a
        g = np.zeros_like(a)

        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]                     # g_i = a_{k*-i}

        # schedule step size (geometric decay)
        alpha = alpha0 * (0.99 ** (it // 200))

        # exponential mirror‑descent update
        a = a * np.exp(-alpha * g)

        # re‑project onto the simplex (non‑negative, sum = 1)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:                       # under‑flow protection
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # occasional bookkeeping
        if it % 50 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[grad iter {it}] new best = {best_val:.6f} (max conv {max_val:.8f})")
            # also try the anti‑symmetric reorder every few hundred steps
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[grad reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement (mass transfers + swaps)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final check & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
