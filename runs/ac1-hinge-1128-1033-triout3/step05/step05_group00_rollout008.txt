def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    from scipy.optimize import linprog
    from scipy.sparse import diags

    start = time.time()
    deadline = start + budget_s - 5.0

    rng = np.random.default_rng(seed)

    # Choose n
    n = 1500

    # initial a: maybe use shape mixture as earlier; we can incorporate previously known good shape if exists.
    a = None
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] >= n:
                a = base[:n]
            else:
                a = np.concatenate([base, np.zeros(n - base.shape[0])])
        except Exception:
            a = None
    if a is None:
        # random mixture of shapes
        x = np.arange(n, dtype=np.float64)
        # some shapes
        sin = np.sin(np.pi * (x + 0.5) / n) ** 2   # positive sine squared
        tri = 1.0 - np.abs((x - (n-1)/2.0) * 2.0 / (n-1))   # triangular [0,1]
        gauss = np.exp(-0.5 * ((x - (n-1)/2) / (n/5))**2)
        a = (sin + tri + gauss) / 3.0
        a = np.maximum(a, 0)
        a = a / a.sum()

    # functions
    def full_conv(u, v):
        # convolution via FFT
        size = u.shape[0] + v.shape[0] - 1
        nfft = 1 << (size - 1).bit_length()
        Fu = np.fft.rfft(u, n=nfft)
        Fv = np.fft.rfft(v, n=nfft)
        conv = np.fft.irfft(Fu * Fv, n=nfft)[:size]
        return conv

    # projection onto simplex
    def project_simplex(v):
        # same as earlier
        nlen = v.shape[0]
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(nlen) + 1
        cond = u - cssv / ind > 0
        if not np.any(cond):
            theta = cssv[-1] / nlen
        else:
            rho = np.nonzero(cond)[0][-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    # initial score
    conv_a = full_conv(a, a)
    max_conv = conv_a.max()
    best_val = 2.0 * n * max_conv
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_val:.12f}")

    iteration = 0
    t_mix = 0.5   # start mix factor
    while time.time() < deadline:
        iteration += 1

        # Note: diagonal offsets correspond to each a_i value
        # Use diags from scipy.sparse
        data = [np.full(n, val, dtype=np.float64) for val in a]
        offsets = np.arange(n)
        A = diags(data, offsets, shape=(2*n-1, n), format='csr')

"""
Propose a non‑negative step function that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a *cut‑plane LP + mixing* method, inspired by the
reference iterative LP scheme:

1.  Keep a probability vector a (non‑negative, sum = 1).
2.  Let M = max(conv(a, a)).  Build the full convolution matrix
    A (size (2n‑1) × n) whose rows encode the linear map g ↦ conv(a, g).
3.  Solve the linear program

        maximise   1ᵀ g
        subject to A g ≤ M·1,   g ≥ 0

    (scipy.optimize.linprog, method='highs').

    The solution g carries as much mass as possible while respecting the
    current bound M.
4.  Update a by a convex combination a ← (1‑t)·a + t·g (t ≈ 0.1‑0.5) and
    project back onto the probability simplex.
    The line‑search reduces t if the new maximum exceeds M.
5.  Repeat steps 2‑4 until the time budget expires.  The best sequence
    seen (lowest evaluated score) is returned.

The LP enforces *all* convolution constraints, not just the currently
active ones, so each iteration provably cannot increase the objective.
The final sequence is clipped to the range required by `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Build the (2n‑1) × n convolution matrix A with constant diagonals.
# For a fixed vector a, (A @ g) == conv(a, g) for all g.
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray):
    n = a.shape[0]
    # each diagonal i (offset i) contains the constant a[i]
    # Create an (n × n) array where row i is filled with a[i]
    data = np.tile(a[:, np.newaxis], (1, n))
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A

            )

    # --------------------------------------------------------------
    # Return the best vector seen, clipped to the range required by
    # `evaluate_sequence`.
    # --------------------------------------------------------------
    final_seq = [
        float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()
    ]  # safety clipping
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")
    return final_seq
