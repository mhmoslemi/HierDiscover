def hill_climb(a, n, deadline):
    conv = _full_convolution_fft(a, a)
    cur_obj = 2.0 * n * conv.max()
    best_obj = cur_obj
    best_a = a.copy()
    # parameters
    max_frac = 0.25
    min_frac = 0.02
    jitter_prob=0.02
    last_improve_iter = 0
    iteration = 0
    recompute_interval=5000
    # temperature
    T0 = cur_obj * 0.01
    while time.time() < deadline:
        t = (time.time() - start) / (deadline - start) # progress
        T = T0 * (1 - t)  # linear cooling
        iteration += 1
        # pass as previous algorithm
        k_star = int(conv.argmax())
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        partner = k_star - idx
        products = a[idx] * a[partner]
        i_src = int(idx[products.argmax()])
        # choose destination j_dst
        j_dst = np.random.randint(n - 1)
        if j_dst >= i_src:
            j_dst += 1
        # delta
        move_frac = max_frac * (1.0 - t) + min_frac * t
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0:
            continue
        a_old = a.copy()
        # incremental update conv similar to earlier code
        # compute a_old for update
        delta2 = delta * delta
        # shift updates
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[j_dst:j_dst + n] += 2.0 * delta * a_old
        conv[2 * i_src] += delta2
        conv[2 * j_dst] += delta2
        conv[i_src + j_dst] -= 2.0 * delta2
        a[i_src] -= delta
        a[j_dst] += delta
        new_obj = 2.0 * n * conv.max()
        # acceptance
        if new_obj < cur_obj or np.random.rand() < np.exp(-(new_obj - cur_obj) / (T + 1e-12)):
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration}] new best obj {best_obj:.6f}")
                last_improve_iter = iteration
        else:
            # revert
            a[:] = a_old
            conv[i_src:i_src + n] += 2.0 * delta * a_old
            conv[j_dst:j_dst + n] -= 2.0 * delta * a_old
            conv[2 * i_src] -= delta2
            conv[2 * j_dst] -= delta2
            conv[i_src + j_dst] += delta2
        # occasional jitter
        if np.random.rand() < jitter_prob:
            # random source-destination
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta_j = 0.5 * a[src] * np.random.rand()
            if delta_j > 0:
                # apply incremental
                a_tmp = a.copy()
                delta2j = delta_j * delta_j
                conv[src:src + n] -= 2.0 * delta_j * a_tmp
                conv[dst:dst + n] += 2.0 * delta_j * a_tmp
                conv[2 * src] += delta2j
                conv[2 * dst] += delta2j
                conv[src + dst] -= 2.0 * delta2j
                a[src] -= delta_j
                a[dst] += delta_j
                jitter_obj = 2.0 * n * conv.max()
                # accept if improved
                if jitter_obj < cur_obj:
                    cur_obj = jitter_obj
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{iteration}] jitter improved best {best_obj:.6f}")
                else:
                    # revert jitter
                    a[:] = a_tmp
                    conv[src:src + n] += 2.0 * delta_j * a_tmp
                    conv[dst:dst + n] -= 2.0 * delta_j * a_tmp
                    conv[2 * src] -= delta2j
                    conv[2 * dst] -= delta2j
                    conv[src + dst] += delta2j
        # periodic recompute
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a, a)
        # stagnation check: reset normalization occasionally
        if iteration - last_improve_iter > 200_000:
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_obj = 2.0 * n * conv.max()
            last_improve_iter = iteration
        if time.time() >= deadline:
            break
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0
    best_seq = None
    best_obj = float('inf')
    # list of candidate sizes
    candidate_ns = [1800, 2500, 4000]
    for n in candidate_ns:
        if time.time() >= deadline:
            break
        # shape search exponential
        L_opt, obj_exp = find_best_shape_exp(n)
        # shape search power
        p_opt, obj_pow = find_best_shape_pow(n)
        # choose best
        if obj_exp < obj_pow:
            lam = L_opt/(n-1)
            a0 = np.exp(lam * np.arange(n)) + np.exp(lam * (n-1 - np.arange(n)))
        else:
            p = p_opt
            idx = np.arange(n)
            a0 = (idx+1)**p + (n-idx)**p
        a0 = a0 / a0.sum()
        # hill climbing
        a_opt, obj_hc = hill_climb(a0, start_time, deadline)
        if obj_hc < best_obj:

            best_seq = a_opt.copy()
            print(f"[global] new global best obj {best_obj:.6f} for n={n}")
    # fallback if no sequence
    if best_seq is None:

        n = 1800
        best_seq = np.full(n, 1.0/n)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")

"""
Propose a step‑function sequence that minimises
    2 * n * max(conv(f,f)) / (sum(f)^2)

The algorithm works in three stages:

1.  Parameterised “U‑shaped’’ families (exponential and power) are
    scanned using a cheap ternary search.  For a given length *n* we
    build a candidate a_i = exp(L*i/(n‑1)) + exp(L*(n‑1‑i)/(n‑1))
    (convex, larger at the ends) or a_i = (i+1)^p + (n‑i)^p.
    The best parameters provide a strong starting point.

2.  Starting from the best family candidate (or the known global
    construction ``height_sequence_1`` if it exists) we run a
    hill‑climbing / simulated‑annealing routine that repeatedly
    transfers a small amount of mass from the index that contributes
    most to the current convolution peak to a random destination.
    The convolution is updated incrementally in O(n) using the exact
    quadratic formula, so millions of iterations are feasible.

3.  The best sequence found before the time budget expires is returned.
    ``evaluate_sequence`` is called once at the end for reporting.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear convolution using a real FFT (full length 2n‑1)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return linear convolution x*y (length len(x)+len(y)-1) using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Objective for a given shape (exponential family)
# ----------------------------------------------------------------------
def _exp_shape_obj(L: float, n: int) -> float:
    """Return 2*n*max(conv) for a_i ∝ exp(L*i/(n‑1)) + exp(L*(n‑1‑i)/(n‑1))."""
    lam = L / (n - 1)
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(lam * idx) + np.exp(lam * (n - 1 - idx))
    a /= a.sum()
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def _find_best_exp_shape(n: int, left: float = 0.0, right: float = 6.0,
                        iters: int = 30):
    """Ternary search for the optimal L in the exponential family."""
    for _ in range(iters):
        l1 = left + (right - left) / 3.0
        l2 = right - (right - left) / 3.0
        o1 = _exp_shape_obj(l1, n)
        o2 = _exp_shape_obj(l2, n)
        if o1 < o2:
            right = l2
        else:
            left = l1
    best_L = (left + right) / 2.0
    best_obj = _exp_shape_obj(best_L, n)
    return best_L, best_obj


# ----------------------------------------------------------------------
# Objective for a given shape (power family)
# ----------------------------------------------------------------------
def _pow_shape_obj(p: float, n: int) -> float:
    """Return 2*n*max(conv) for a_i ∝ (i+1)^p + (n-i)^p."""
    idx = np.arange(n, dtype=np.float64)
    a = (idx + 1.0) ** p + (n - idx) ** p
    a /= a.sum()
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def _find_best_pow_shape(n: int, left: float = 0.0, right: float = 6.0,
                        iters: int = 30):
    """Ternary search for the optimal exponent p in the power family."""
    for _ in range(iters):
        p1 = left + (right - left) / 3.0
        p2 = right - (right - left) / 3.0
        o1 = _pow_shape_obj(p1, n)
        o2 = _pow_shape_obj(p2, n)
        if o1 < o2:
            right = p2
        else:
            left = p1
    best_p = (left + right) / 2.0
    best_obj = _pow_shape_obj(best_p, n)
    return best_p, best_obj


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing optimiser
# ----------------------------------------------------------------------
def _hill_climb(a_initial: np.ndarray, start_time: float, deadline: float):
    """
    Perform mass‑transfer moves on the probability simplex.
    Returns best sequence (as np.ndarray) and its objective value.
    """
    a = a_initial.copy()
    n = a.shape[0]

    conv = _full_convolution_fft(a, a)               # length 2n‑1
    cur_obj = float(2.0 * n * conv.max())
    best_obj = cur_obj
    best_a = a.copy()

    # hyper‑parameters (tuned for a 1000 s budget)
    max_frac_start = 0.30
    min_frac_end = 0.02
    jitter_prob = 0.015
    recompute_interval = 5000
    iteration = 0
    last_improve = 0
    T0 = cur_obj * 0.10                # initial temperature for annealing

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        progress = (now - start_time) / (deadline - start_time)
        move_frac = max_frac_start * (1.0 - progress) + min_frac_end * progress
        T = max(T0 * (1.0 - progress), 1e-12)

        # --------------------------------------------------------------
        # Identify the index that contributes most to the current peak.
        # --------------------------------------------------------------
        k_star = int(conv.argmax())
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        partner = k_star - idx
        products = a[idx] * a[partner]
        i_src = int(idx[products.argmax()])

        # --------------------------------------------------------------
        # Choose a random destination (different from source).
        # --------------------------------------------------------------
        j_dst = np.random.randint(n - 1)
        if j_dst >= i_src:
            j_dst += 1

        # --------------------------------------------------------------
        # Propose a small mass transfer from i_src to j_dst.
        # --------------------------------------------------------------
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue

        a_old = a.copy()                       # keep full old vector for revert
        delta2 = delta * delta

        # incremental convolution update using the exact formula
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[j_dst:j_dst + n] += 2.0 * delta * a_old
        conv[2 * i_src] += delta2
        conv[2 * j_dst] += delta2
        conv[i_src + j_dst] -= 2.0 * delta2

        a[i_src] -= delta
        a[j_dst] += delta

        new_obj = float(2.0 * n * conv.max())

        # --------------------------------------------------------------
        # Simulated‑annealing acceptance rule.
        # --------------------------------------------------------------
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_obj - cur_obj) / T):
                accept = True

        if accept:
            cur_obj = new_obj
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration:9d}] new best = {best_obj:.6f}")
                last_improve = iteration
        else:
            # revert both a and conv
            a[:] = a_old
            conv[i_src:i_src + n] += 2.0 * delta * a_old
            conv[j_dst:j_dst + n] -= 2.0 * delta * a_old
            conv[2 * i_src] -= delta2
            conv[2 * j_dst] -= delta2
            conv[i_src + j_dst] += delta2

        # --------------------------------------------------------------
        # Occasional random jitter (large transfer) to escape plateaus.
        # --------------------------------------------------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta_j = 0.5 * a[src] * np.random.rand()
            if delta_j > 0.0:
                a_tmp = a.copy()
                delta2j = delta_j * delta_j
                conv[src:src + n] -= 2.0 * delta_j * a_tmp
                conv[dst:dst + n] += 2.0 * delta_j * a_tmp
                conv[2 * src] += delta2j
                conv[2 * dst] += delta2j
                conv[src + dst] -= 2.0 * delta2j

                a[src] -= delta_j
                a[dst] += delta_j

                jitter_obj = float(2.0 * n * conv.max())
                if jitter_obj < cur_obj:
                    cur_obj = jitter_obj
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{iteration:9d}] jitter improved best = {best_obj:.6f}")
                else:
                    # revert jitter
                    a = a_tmp
                    conv[src:src + n] += 2.0 * delta_j * a_tmp
                    conv[dst:dst + n] -= 2.0 * delta_j * a_tmp
                    conv[2 * src] -= delta2j
                    conv[2 * dst] -= delta2j
                    conv[src + dst] += delta2j

        # --------------------------------------------------------------
        # Periodic full recomputation to avoid numerical drift.
        # --------------------------------------------------------------
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a, a)

        # --------------------------------------------------------------
        # Simple stagnation handling.
        # --------------------------------------------------------------
        if iteration - last_improve > 200_000:
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_obj = float(2.0 * n * conv.max())
            last_improve = iteration

    return best_a, best_obj
