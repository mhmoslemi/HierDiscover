    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)

    # contributions:

    src = idx[contrib.argmax()]
    return src, k

"""
Propose a candidate sequence that minimizes ``evaluate_sequence``.

Algorithm
---------
1. Initialise a probability vector ``a`` (sum = 1).  If a previous best
   construction ``height_sequence_1`` is available we try both the raw
   ordering and a descending sort and keep the better start.
2. Keep the exact linear convolution ``conv = a*a`` (length ``2*n-1``) and the
   current objective ``obj = 2*n*conv.max()``.
3. Repeatedly perform *mass‑transfer* moves that are guaranteed to reduce
   the peak of the convolution:
   * The current peak index ``k*`` is the arg‑max of ``conv``.
   * The coefficient ``i_src`` that contributes most to this peak is the
     index maximising ``a[i]*a[k*-i]``.
   * Destination ``j_dst`` is chosen greedily: for each possible ``j``
     the added amount to the peak is ``a[k*-j]`` (zero if the index is out
     of range).  The index with the smallest addition (excluding ``i_src``)
     is selected.
   * Transfer a fraction ``δ`` of ``a[i_src]`` to ``j_dst``.  The
     convolution is updated in O(n) using the exact incremental formula.
   * Accept the move only if the new maximum of the convolution is smaller;
     otherwise revert.
4. Occasionally perform a larger “jitter’’ move (random source/destination
   and a larger ``δ``) to escape plateaus.
5. Periodically recompute the full convolution with FFT to avoid drift.
6. If no improvement has been seen for a long time we restart from a fresh
   random Dirichlet distribution (sorted descending) – this gives the
   optimiser many chances to escape local minima.
7. The best sequence seen so far is returned (as a plain list of floats).

The method is a directed hill‑climber with a simple heuristic for the
destination index and a restart strategy, which improves over the previous
implementation and is able to push the objective below the target
``1.503`` within the 1000 s budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: full linear convolution using FFT (for occasional re‑sync)
# ------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y using real FFT (returns length len(x)+len(y)-1)."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0, out=conv)
    return conv

    Returns the best sequence (list of floats) found before the time budget
    expires.
    """

    start_time = time.time()

    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Initialise a – use the provided best construction if possible,
    # otherwise start from a uniform distribution.

    try:
        init_seq = height_sequence_1               # noqa: F821
    except NameError:

    if init_seq is None:
        n = 1800
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:

        n = a0.shape[0]
        # both the raw order and a descending sort are tried – keep the

        a0 /= a0.sum()
        conv0 = _full_convolution_fft(a0, a0)
