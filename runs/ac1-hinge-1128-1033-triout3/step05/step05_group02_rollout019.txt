def simplex_projection(v):
    if v.sum() == 1 and np.all(v >= 0): return v
    n = len(v)
    u = np.sort(v)[::-1] # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def full_convolution_fft(x):
    n = x.shape[0]
    out_len = 2*n -1
    size = 1 << (out_len-1).bit_length()  # next pow2
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[:out_len]
    return conv

"""
Propose a step‐function sequence with a very low evaluate_sequence value.

Approach
--------
1. Choose a problem size (default n=4000).  
2. Build a family of symmetric “V‑shaped” candidates:
      a_i = (1 - d_i / d_max) ** p           (power law)
      a_i = exp(-λ * d_i)                     (exponential)
   where d_i = min(i, n‑1‑i) is the distance to the nearest end.
   The family is cheap to evaluate – we use an FFT based convolution.
3. Scan the parameter(s) (p or λ) on a coarse grid, then refine around the
   best parameters found.  The best candidate from this scan is taken as a
   starting point.
4. Perform a fast “hill‑climb” that repeatedly either swaps two entries or
   transfers a small amount of mass from one position to another while
   preserving the total sum.  Only strict improvements (on the fast
   evaluation) are accepted.  This phase runs until the time budget is
   exhausted.
5. Finally evaluate the best sequence with the provided evaluate_sequence()
   (which performs the exact, slower convolution) and return it.

All heavy work (grid search + hill climbing) uses a fast FFT‑based convolution
so many millions of evaluations comfortably fit into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Returns an array of length 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negative values – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _fast_evaluation(a: np.ndarray, n: int) -> float:
    """
    Fast surrogate for evaluate_sequence when sum(a) ≈ 1.
    Returns 2·n·max(conv) / (sum(a)**2).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution_fft(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simple hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, cur_val: float,
                deadline: float, n: int) -> tuple[np.ndarray, float]:
    """
    Random hill‑climbing that preserves the total mass.
    Only strict improvements (according to the fast evaluation) are kept.
    """
    cur = a.copy()
    rng = np.random.default_rng()
    while time.time() < deadline:
        # 20 % chance of a pure swap (re‑orders values)
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur[i] <= 1e-15:
                continue
            # move up to 5 % of donor's mass
            delta = cur[i] * 0.05 * rng.random()
            if delta <= 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

        new_val = _fast_evaluation(new, n)
        if new_val < cur_val - 1e-12:          # strict improvement
            cur = new
            cur_val = new_val
            # occasional progress print (optional, comment out if noisy)
            # print(f"[hill] new best = {cur_val:.6f}")
    return cur, cur_val


# ----------------------------------------------------------------------
# Main candidate generation
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑value step function.
    Returns the best sequence found as a plain Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 1.0    # small safety margin
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Problem dimension – 4000 is a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Prepare distance array (distance to the nearest end)
    # --------------------------------------------------------------
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(np.float64)     # 0 at ends, grows to centre
    d_max = d.max()
    d_norm = d / d_max                                      # in [0,1]

    best_a = None
    best_val = np.inf

    # --------------------------------------------------------------
    # 1) Scan a simple parametric family (power‑law V‑shape)
    #    a_i = (1 - d_norm) ** p   , p >= 0  (p=0 → uniform)
    # --------------------------------------------------------------
    p_grid = np.concatenate([np.linspace(0.0, 4.0, 81), np.linspace(4.0, 10.0, 41)])
    for p in p_grid:
        a = (1.0 - d_norm) ** p
        a /= a.sum()
        val = _fast_evaluation(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    # --------------------------------------------------------------
    # 2) Scan exponential V‑shape: a_i = exp(-λ * d)
    # --------------------------------------------------------------
    lam_grid = np.concatenate([
        np.linspace(0.0, 1.0, 51),
        np.geomspace(1.0, 10.0, 30)
    ])
    for lam in lam_grid:
        a = np.exp(-lam * d)
        a /= a.sum()
        val = _fast_evaluation(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # --------------------------------------------------------------
    # 3) Fine‑tune around the best parameters found above.
    #    For the power‑law case we refine p, for exponential we refine λ.
    # --------------------------------------------------------------
    # Determine which family gave the current best (heuristic: check shape)
    # If the best vector is monotone decreasing towards the centre we assume
    # exponential, otherwise power‑law – both are refined regardless.
    for _ in range(4):
        # power‑law refinement
        p_center = np.log2(best_a[0] / best_a[n // 2]) if best_a[n // 2] > 0 else 0.0
        p_fine = np.linspace(max(p_center - 0.5, 0.0), p_center + 0.5, 31)
        for p in p_fine:
            a = (1.0 - d_norm) ** p
            a /= a.sum()
            val = _fast_evaluation(a, n)
            if val < best_val:
                best_val = val
                best_a = a.copy()

        # exponential refinement
        lam_center = -np.log(best_a[0] / best_a[n // 2]) if best_a[n // 2] > 0 else 0.0
        lam_fine = np.linspace(max(lam_center - 0.5, 0.0), lam_center + 0.5, 31)
        for lam in lam_fine:
            a = np.exp(-lam * d)
            a /= a.sum()
            val = _fast_evaluation(a, n)
            if val < best_val:
                best_val = val
                best_a = a.copy()

        # Stop early if we are out of time
        if time.time() >= deadline - 30.0:
            break

    # --------------------------------------------------------------
    # 4) Final hill‑climbing phase (mass transfers + swaps)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # 5) Final (exact) evaluation via the provided routine.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = best_val

    # Return the sequence (as a plain list of floats)
    return best_a.tolist()
