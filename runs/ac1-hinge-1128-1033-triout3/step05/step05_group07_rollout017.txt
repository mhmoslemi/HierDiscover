import scipy.sparse as sp

def solve_lp(f, M):
    n = len(f)
    # Build sparse matrix A of shape (2n-1, n)
    rows = []
    cols = []
    data = []

    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # for i in i_start..i_end:
        # coefficient for variable g_j where j = k - i
        # So for each i, we add entry at (k, j) with coefficient f[i]
        # Equivalent: for each j = k-i => i = k-j
        # So we can loop over j in index range.
        # We'll adopt loop over i.
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    bounds = [(0, None)] * n
    # maximize sum(g) -> minimize -sum(g)
    c = np.full(n, -1.0)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError('LP failed')
    g = res.x
    return g

def _full_convolution_fft(x, y):
    n = len(x)
    m = len(y)
    total = n + m - 1
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Remove small negative due to floating errors
    conv[conv < 0] = 0.0
    return conv

"""
Propose a step‑function that minimises the evaluation function.

The algorithm consists of two phases.

1.  **LP‑based flattening** –  Starting from the current best sequence
    (the global variable ``height_sequence_1`` if it exists) we compute
    the current self‑convolution peak ``M``.  With ``M`` fixed we solve a
    linear program

        maximise   Σ g_i
        subject to  (a * g)_k ≤ M   for all k
                    g_i ≥ 0

    where ``a`` is the current sequence and ``*`` denotes linear
    convolution.  The LP finds a new distribution ``g`` that can carry
    *more* total mass than ``a`` without violating the bound ``M``.
    After normalising ``g`` to a probability vector we mix it with the
    old sequence:

        a ← (1‑t)·a + t·ĝ ,    t>0 small.

    For a sufficiently small mixing factor the new peak is strictly
    smaller than the old one, therefore the objective value decreases.
    We repeat this outer loop while time permits.

2.  **Local hill‑climbing** –  When the LP stage stalls we fall back to
    the fast mass‑transfer hill‑climber used previously.  It repeatedly
    moves a small amount of mass from the coefficient that contributes
    most to the current peak to a random destination, using an O(n)
    incremental update of the convolution.  Occasional “jitter’’ moves
    help to escape plateaus.  The hill‑climber runs until the time budget
    is exhausted.

The whole routine respects the 1000 s budget, prints progress and
returns the best sequence found (a list of non‑negative floats).  The
sequence is already normalised; scaling does not affect the score.
"""

# ----------------------------------------------------------------------
# Helper: exact linear convolution using real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (full length)."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.ifft(fx * fy, size).real[:total]
    # eliminate tiny negatives from round‑off
    conv[conv < 0] = 0.0
    return conv


# ----------------------------------------------------------------------
# LP stage – one iteration builds the constraint matrix and solves the LP
# ----------------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float):
    """
    Solve   maximise Σ g_i    subject to   a * g ≤ M   (component‑wise),
    g_i ≥ 0.
    Returns the optimal vector g (may be very sparse).
    """
    from scipy.optimize import linprog
    import scipy.sparse as sp

    n = a.shape[0]

    # Build the convolution matrix of a (shape (2n‑1) × n)
    # Each column j contains the vector a shifted by +j.
    A = sp.diags([a] * n, offsets=range(n), shape=(2 * n - 1, n), format="csr")

    # Objective: maximise Σ g_i  →  minimise –Σ g_i
    c = -np.ones(n)

    # Upper bound for every convolution entry
    b = np.full(2 * n - 1, M)

    # Non‑negativity bounds
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method="highs", options={"presolve": True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x
