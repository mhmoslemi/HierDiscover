"""
Search for a non‑negative step function minimizing the evaluation metric.
The strategy combines a long run of Adam optimisation on the log‑softmax
parameters (with an adaptive active‑set tolerance) and a short simulated‑annealing
post‑process that moves probability mass between coordinates.  The best vector
found during the total time budget (≤ 1000 s) is returned.
"""
import time
import numpy as np

def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def score_vector(a: np.ndarray) -> float:
    """Assumes sum(a) == 1. Returns 2 * n * max(a*a)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

def grad_log_params(x: np.ndarray, a: np.ndarray, conv: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. the log‑parameters x.
    Assumes a = softmax(x) and ``active_idx`` are the indices where the
    convolution attains (near) maximal value.
    """
    n = a.shape[0]
    G = np.zeros_like(a)

    for k in active_idx:
        # pairs (i, j) such that i + j = k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]

    # sub‑gradient w.r.t. log‑parameters
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad

def run_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in the log‑softmax space.
    The active‑set tolerance starts at 1e‑4 and decays to 1e‑12.
    Returns the best vector (as probabilities) and its score.
    """
    start = time.time()
    n = initial_a.shape[0]

    # initialise log‑parameters
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.2

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    # active‑set tolerance schedule
    rel_tol = 1e-4
    min_rel_tol = 1e-12
    rel_decay = 0.9999

    while time.time() - start < time_limit - 0.3:
        iteration += 1

        # ---- softmax
        a = softmax(x)

        # ---- convolution
        conv = full_convolution_fft(a)
        max_b = conv.max()

        # ---- active set (indices whose value is close to the maximum)
        active = np.where(conv >= max_b * (1.0 - rel_tol))[0]

        # ---- gradient on log‑parameters
        grad = grad_log_params(x, a, conv, max_b, active)

        # ---- Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- bookkeeping
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Adam iter {iteration:7d}: new best score = {best_score:.6f}")

        # ---- re‑heat if stuck for many iterations
        if iteration - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Adam iter {iteration:7d}: re‑heat (stalled)")

        # ---- decay the active‑set tolerance
        rel_tol = max(rel_tol * rel_decay, min_rel_tol)

    return best_vec, best_score

def run_simulated_annealing(start_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the probability simplex.
    Small amounts of mass are moved between two randomly chosen indices.
    The temperature decays exponentially.
    Returns the best vector seen and its score.
    """
    start = time.time()
    n = start_a.shape[0]
    cur_a = start_a.copy()
    cur_score = score_vector(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # Initial temperature – a fraction of the current objective
    T = max(cur_score * 0.05, 1e-4)
    T_min = 1e-9
    decay = 0.9995

    # Step size is a fraction of the amount taken from the donor index.
    base_step = 0.05

    iteration = 0
    while time.time() - start < time_limit - 0.2:
        iteration += 1

        # Pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)

        # Amount to transfer (never more than what i currently holds)
        eps = base_step * cur_a[i] * np.random.rand()
        if eps <= 0.0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # Numerical safety: clip tiny negatives caused by rounding
        if new_a[i] < 0.0:
            new_a[i] = 0.0

        new_score = score_vector(new_a)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"SA iter {iteration:7d}: new best score = {best_score:.6f}")

        # Temperature schedule
        T = max(T * decay, T_min)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Uses a long Adam optimisation followed by a short
    simulated‑annealing refinement.  Returns the best sequence found as a
    plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a few seconds margin

    # --------------------------------------------------------------
    # 1. Base sequence – try to use the global `height_sequence_1`
    # --------------------------------------------------------------
    n = 5000
    if 'height_sequence_1' in globals():
        try:
            external = np.array(height_sequence_1, dtype=float)
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external /= s
                n = external.shape[0]
                base_vec = external
                print(f"Using external seed of length {n}")
        except Exception:
            base_vec = None
    else:
        base_vec = None

    if base_vec is None:
        # fallback deterministic seed: a smooth, symmetric distribution
        # of the form (i+1)^(-p) + (n-i)^(-p)
        def smooth_distribution(p: float) -> np.ndarray:
            i = np.arange(n, dtype=np.float64)
            a = (i + 1) ** (-p) + (n - i) ** (-p)
            a = np.maximum(a, 1e-14)
            a /= a.sum()
            return a
        base_vec = smooth_distribution(p=1.0)
        print(f"Generated deterministic seed (length {n})")

    # --------------------------------------------------------------
    # 2. Adam optimisation (majority of the budget)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    adam_time = remaining * 0.80
    print(f"Running Adam for {adam_time:.1f}s")
    best_adam_vec, best_adam_score = run_adam(base_vec, adam_time)

    # --------------------------------------------------------------
    # 3. Simulated annealing fine‑tuning (remaining budget)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 2.0:
        sa_time = remaining * 0.95   # leave a little buffer for safety
        print(f"Running simulated annealing for {sa_time:.1f}s")
        best_sa_vec, best_sa_score = run_simulated_annealing(best_adam_vec, sa_time)
    else:
        best_sa_vec, best_sa_score = best_adam_vec, best_adam_score

    # --------------------------------------------------------------
    # 4. Optional short Adam polish (if a few seconds still left)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 1.0:
        print(f"Polishing with a short Adam run for {remaining:.1f}s")
        best_polish_vec, best_polish_score = run_adam(best_sa_vec, remaining)
        final_vec, final_score = (best_polish_vec, best_polish_score)
    else:
        final_vec, final_score = best_sa_vec, best_sa_score

    print("\n=== Search finished ===")
    print(f"Best score found: {final_score:.6f} (length {len(final_vec)})")
    # Ensure non‑negativity and normalisation before returning
    final_vec = np.maximum(final_vec, 0.0)
    s = final_vec.sum()
    if s > 0.0:
        final_vec = final_vec / s
    return final_vec.tolist()

            best_vec = a.copy()
            last_improve = iteration
            print(f"Adam iter {iteration:7d}: new best score = {best_score:.6f}")

        # ----- re‑heat if stalled -----
        if iteration - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Adam iter {iteration:7d}: re‑heat (no improvement)")

        # ----- tighten the active‑set tolerance -----
        rel_tol = max(rel_tol * rel_decay, min_rel_tol)

    return best_vec, best_score

# ----------------------------------------------------------------------
# Simulated annealing on the simplex
# ----------------------------------------------------------------------
def run_simulated_annealing(start_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑style annealing that moves a small amount of probability
    mass between two random coordinates.  Returns the best vector discovered.
    """
    start = time.time()
    n = start_a.shape[0]
    cur_a = start_a.copy()
    cur_score = score_vector(cur_a)

    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature (a fraction of the current score)
    T = max(cur_score * 0.05, 1e-6)
    T_min = 1e-12
    decay = 0.9995

    # step size relative to the donor entry
    base_step = 0.05

    iteration = 0
    while time.time() - start < time_limit - 0.2:
        iteration += 1

        i, j = np.random.choice(n, size=2, replace=False)

        # amount to transfer (never more than what i currently holds)
        eps = base_step * cur_a[i] * np.random.rand()
        if eps <= 0.0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # safety against tiny negative values caused by rounding
        if new_a[i] < 0.0:
            new_a[i] = 0.0

        new_score = score_vector(new_a)
        delta = new_score - cur_score

        # Metropolis acceptance
        if delta < 0.0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"SA iter {iteration:7d}: new best score = {best_score:.6f}")

        T = max(T * decay, T_min)

    return best_a, best_score

# ----------------------------------------------------------------------
# Main entry point required by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Searches for a non‑negative step function that minimises the evaluation
    function.  The routine works within the supplied time budget, prints
    progress and finally returns the best vector (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # --------------------------------------------------------------
    # 1. Base sequence – try to use an external construction if present
    # --------------------------------------------------------------
    n = 5000
    if 'height_sequence_1' in globals():
        try:
            external = np.array(height_sequence_1, dtype=float)
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external /= s
                n = external.shape[0]
                base_vec = external
                print(f"Using external seed of length {n}")
            else:
                base_vec = None
        except Exception:
            base_vec = None
    else:
        base_vec = None

    # --------------------------------------------------------------
    # 2. Fallback deterministic seed if needed
    # --------------------------------------------------------------
    if base_vec is None:
        # symmetric distribution (i+1)^-p + (n-i)^-p)
        def smooth_distribution(p: float) -> np.ndarray:
            i = np.arange(n, dtype=np.float64)
            a = (i + 1) ** (-p) + (n - i) ** (-p)
            a = np.maximum(a, 1e-14)
            a /= a.sum()
            return a
        base_vec = smooth_distribution(p=1.0)
        print(f"Generated deterministic seed (length {n})")

    # --------------------------------------------------------------
    # 3. Adam optimisation (major part of the budget)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    adam_time = remaining * 0.80
    print(f"Running Adam for {adam_time:.1f}s")
    best_adam_vec, best_adam_score = run_adam(base_vec, adam_time)

    # --------------------------------------------------------------
    # 4. Simulated‑annealing fine‑tuning (use what remains)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 2.0:
        sa_time = remaining * 0.95   # keep a thin buffer for the final polish
        print(f"Running simulated annealing for {sa_time:.1f}s")
        best_sa_vec, best_sa_score = run_simulated_annealing(best_adam_vec, sa_time)
    else:
        best_sa_vec, best_sa_score = best_adam_vec, best_adam_score

    # --------------------------------------------------------------
    # 5. Optional short Adam polish (if any time left)
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining > 1.0:
        print(f"Polishing with a short Adam run for {remaining:.1f}s")
        final_vec, final_score = run_adam(best_sa_vec, remaining)
    else:
        final_vec, final_score = best_sa_vec, best_sa_score

    # --------------------------------------------------------------
    # 6. Final reporting
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {final_score:.6f} (length {len(final_vec)})")

    # Ensure proper normalisation before returning
    final_vec = np.maximum(final_vec, 0.0)
    total = final_vec.sum()
    if total > 0.0:
        final_vec = final_vec / total
    return final_vec.tolist()
