"""
Propose a non‑negative step‑function (probability vector) that minimizes
evaluate_sequence.  The optimiser works in the probability simplex (softmax
parametrisation) and minimises a smooth approximation of the L∞ norm of the
auto‑convolution using a temperature (β) that is increased during the run.
Gradients are obtained by the chain rule:

    f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
    ∂f_β/∂a_i = 2·Σ_k w_k·a_{k‑i},
    w_k = exp(β·c_k) / Σ_j exp(β·c_j)   (c = a*a)

The convolution and the weighted correlation are computed with FFTs,
and Adam is used to update the log‑probabilities.
The routine runs until the time budget (≈ 1000 s) is exhausted and returns
the best vector seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution a*a using real FFT.
    `size` must be ≥ len(a) + len(a) - 1.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * a.shape[0] - 1]


def correlation_fft(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[ℓ] = Σ_i a_i·w_{i+ℓ}.
    Implemented via FFT: corr = ifft(conj(FA) * FW).
    Returns an array of length `size`; the relevant slice
    (offset = len(w)-1) gives the values for ℓ = 0 … len(a)-1.
    """
    FA = np.fft.rfft(a, n=size)
    FW = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(np.conj(FA) * FW, n=size)
    return corr


# ----------------------------------------------------------------------
# Probability utilities
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Optimiser: Adam on the log‑probabilities using a smooth max objective.
# ----------------------------------------------------------------------
def run_smooth_adam(initial_a: np.ndarray,
                    time_limit: float,
                    beta_schedule: list[float] = (5.0, 30.0, 150.0, 500.0, 2000.0),
                    steps_per_beta: int = 2500) -> tuple[np.ndarray, float]:
    """
    Optimises a probability vector `initial_a` (sum = 1) using Adam.
    The objective is a smoothed max of the auto‑convolution:
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]).

    Parameters
    ----------
    initial_a : np.ndarray
        Starting probability vector (already normalised).
    time_limit : float
        Seconds available for this optimisation run.
    beta_schedule : sequence of floats
        Sequence of β‑values; the optimiser stays at each β for
        `steps_per_beta` Adam updates before moving to the next.
    steps_per_beta : int
        Number of Adam iterations to perform at each temperature.

    Returns
    -------
    best_vec : np.ndarray
        Best probability vector encountered.
    best_score : float
        Corresponding value of the original evaluation function.
    """
    start = time.time()
    n = initial_a.shape[0]

    # log‑parameters (avoid log(0))
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam hyper‑parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.22

    # FFT size (power of two) – must accommodate a (n) and w (2n‑1)
    fft_size = _next_pow2(3 * n)

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0
    beta_idx = 0
    beta = beta_schedule[beta_idx]

    while time.time() - start < time_limit - 0.5:
        iteration += 1

        # ----- probability vector -----
        a = softmax(x)

        # ----- auto‑convolution -----
        conv = full_convolution_fft(a, fft_size)          # length 2n‑1
        max_c = conv.max()

        # ----- smooth max weights w_k ∝ exp(β·c_k) -----
        # subtract the max exponent for numerical stability
        exp_arg = beta * (conv - max_c)
        w_raw = np.exp(exp_arg)
        w = w_raw / w_raw.sum()                           # guarantees Σ w = 1

        # ----- weighted correlation G_i = 2 Σ_k w_k a_{k-i} -----
        corr_full = correlation_fft(a, w, fft_size)       # length fft_size
        offset = 2 * n - 2                                 # len(w)-1
        G = 2.0 * corr_full[offset: offset + n]            # shape (n,)

        # ----- gradient in log‑space -----
        mu = np.dot(a, G)
        grad = a * (G - mu)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- keep best candidate (original evaluation) -----
        cur_score = float(2.0 * n * max_c)   # same as evaluate_sequence for sum=1
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best = {best_score:.6f} (β={beta:.1f})")

        # ----- temperature schedule -----
        if iteration % steps_per_beta == 0:
            if beta_idx + 1 < len(beta_schedule):
                beta_idx += 1
                beta = beta_schedule[beta_idx]
                print(f"Iter {iteration:6d}: increasing β to {beta:.1f}")

        # ----- re‑heat if stuck for a long time -----
        if iteration - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

    return best_vec, best_score

# Main entry point required by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Searches for a non‑negative step‑function that minimises the provided
    evaluation function.  The routine works within the supplied time budget,
    prints progress, and finally returns the best vector found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1. Obtain an initial seed
    # --------------------------------------------------------------
    try:
        # Use the globally supplied construction if present
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        s = external.sum()
        if s > 0.0:
            external /= s
            n = external.shape[0]
            initial_vec = external
            print(f"Using external seed of length {n}")
        else:
            raise ValueError
    except Exception:
        # Fall back to a modest length uniform distribution
        n = 5000
        initial_vec = np.ones(n) / n
        print(f"No external seed – using uniform start (length {n})")

    # --------------------------------------------------------------
    # 2. Optimisation (smooth‑max Adam)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    best_overall_vec = None
    best_overall_score = np.inf

    # we may try a few different random starts – small budget for the later ones
    seeds = [initial_vec]

    # a couple of Dirichlet draws for diversification
    for _ in range(2):
        rand = np.random.rand(n)
        seeds.append(rand / rand.sum())

    # --------------------------------------------------------------
    # 3. Run optimiser for each seed, splitting the remaining budget
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(seeds):
        # Normalise seed to a proper probability vector
        a0 = np.maximum(seed_vec, 0.0)
        if a0.sum() == 0.0:
            a0 = np.ones_like(a0)
        a0 /= a0.sum()

        # Allocate time for this seed
        now = time.time()
        remaining_time = deadline - now
        if remaining_time <= 2.0:
            break

        # Give the first seed most of the budget, later seeds a fraction
        if idx == 0:
            per_seed_budget = remaining_time * 0.7
        else:
            per_seed_budget = remaining_time * 0.3 / (len(seeds) - 1)

        print(f"\n--- Optimisation start: seed {idx+1}/{len(seeds)} "
              f"(budget {per_seed_budget:.1f}s) ---")
        vec, sc = run_smooth_adam(a0, per_seed_budget)

        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_vec = vec.copy()
            print(f"*** New global best from seed {idx+1}: score = {best_overall_score:.6f}")

        # Update deadline for the next seed
        if time.time() >= deadline - 4.0:
            break

    # --------------------------------------------------------------
    # 4. Final report
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    if best_overall_vec is not None:
        print(f"Best score found: {best_overall_score:.6f} (length {len(best_overall_vec)})")
        return best_overall_vec.tolist()
    else:
        # Should never happen, but return a safe uniform vector
        print("No valid vector found – returning uniform fallback.")
        return (np.ones(n) / n).tolist()
