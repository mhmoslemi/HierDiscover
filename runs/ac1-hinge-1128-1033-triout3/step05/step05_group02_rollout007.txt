def project_onto_simplex(v):
    # Input: vector v (numpy array)
    # Output: projection onto the probability simplex {x: x>=0, sum=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n+1))[0]
    if rho.size == 0:
        # all v <= ... maybe produce zero vector?
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Subgradient optimizer for the step‑function problem.

The evaluation metric is 2·n·max(conv) when the sequence is normalized to
sum(a)=1 (the same metric used by ``evaluate_sequence``).  The algorithm
repeatedly:

1. Starts from a non‑uniform seed (the previous best construction if
   available, otherwise a random Dirichlet vector with an anti‑symmetric
   re‑ordering that pairs large and small values).
2. Runs a projected sub‑gradient descent on the true max‑convolution
   objective.  At each iteration the index k where the current convolution
   attains its maximum is identified; the sub‑gradient of that term is
   g_i = 2·a_{k‑i}.  The direction d = –g + mean(g) guarantees that the
   simplex constraint Σa_i=1 is respected to first order.  A step is taken,
   then the vector is projected back onto the probability simplex.
3. Keeps the best vector seen so far.
4. After the sub‑gradient phase a cheap hill‑climbing refinement (random
   mass transfers and swaps) is performed until the time budget expires.

The whole search respects a 1000 s wall‑clock budget and returns the best
non‑negative sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself (length 2·len‑1).
    Implemented with FFT for speed; tiny negative values from rounding are
    clipped to zero.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)      # eliminate round‑off negatives
    return conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w | w_i ≥ 0 , Σ w_i = 1 }.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a vector a that is already
    normalized to Σ a = 1.
    Returns 2·n·max(conv) (identical to ``evaluate_sequence`` up to the
    scale factor Σa² = 1).
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Large values are placed on the left half, small values on the right
    half, thus pairing large with small when the sequence is reversed.
    This heuristic usually lowers the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Simple stochastic local search that mixes tiny mass‑transfers with swaps.
    Only strict improvements (in terms of the true objective) are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # 20 % chance of a pure swap, otherwise a small mass transfer
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()   # up to 5 % of donor
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = eval_fast(new_a)
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


def subgradient_descent(a0: np.ndarray, max_iters: int,
                        time_deadline: float):
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    The vector is kept on the probability simplex.  The function returns
    the best vector seen during the run and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    # initialise best tracker
    cur_conv = _full_convolution_fft(a)
    best_val = 2.0 * n * cur_conv.max()
    best_a = a.copy()
    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # current convolution and its maximising index
        conv = _full_convolution_fft(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active term: g_i = 2·a_{k‑i}
        g = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_end >= i_start:
            idx = np.arange(i_start, i_end + 1)
            g[idx] = 2.0 * a[k - idx]

        # direction respecting Σa = 1  (projection of –g onto the hyperplane)
        mean_g = g.mean()
        d = -g + mean_g

        # diminishing step size
        alpha = 0.5 / np.sqrt(it)

        # tentative update + projection onto the simplex
        a_new = a + alpha * d
        a_new = project_onto_simplex(a_new)

        # evaluate new point
        conv_new = _full_convolution_fft(a_new)
        val_new = 2.0 * n * conv_new.max()

        # keep the best ever seen
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[sgd iter {it}] new best = {best_val:.6f}")

        a = a_new
    return best_a, best_val

def propose_candidate(budget_s: int = 1000, seed: int = 0):
    """
    Search for a step‑function with minimal ``evaluate_sequence`` value.
    Returns the best list of heights found before the budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # 1 s safety margin

    # Problem size – a few thousand entries give a good trade‑off.
    n = 5000

    # --------------------------------------------------------------
    # Initialise from a previously discovered construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        # Random Dirichlet start (exponential distribution)
        a0 = np.random.rand(n)
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    # Break the uniform symmetry early
    a0 = _anti_symmetric_reorder(a0)

    best_a = a0.copy()
    best_val = eval_fast(best_a)
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Multi‑restart projected sub‑gradient descent
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 30.0:       # keep ~30 s for final refinement
        restart += 1
        print(f"[restart {restart}]")

        if restart == 1:
            a_start = best_a.copy()
        else:
            # fresh random start for later restarts
            a_start = np.random.rand(n)
            a_start = a_start / a_start.sum()
            a_start = _anti_symmetric_reorder(a_start)

        # allow each restart roughly 20 s (adjusted by remaining time)
        sub_deadline = min(deadline - 5.0, time.time() + 20.0)
        a_opt, val_opt = subgradient_descent(a_start,
                                              max_iters=20000,
                                              time_deadline=sub_deadline)
        if val_opt < best_val - 1e-12:
            best_val = val_opt
            best_a = a_opt.copy()

    # ------------------------------------------------------------------
    if time.time() < deadline:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = np.inf

    # Return the raw list (the outer harness will handle the score)
    return best_a.tolist()
