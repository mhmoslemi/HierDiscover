def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:

    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase S – sub‑gradient descent on max(b)
# ----------------------------------------------------------------------
def _subgradient_max_phase(a_start: np.ndarray,
                           n: int,
                           deadline: float,
                           best_a: np.ndarray,
                           best_val: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the objective 2·n·max(a*a).
    Returns the best vector seen before *deadline*.
    """
    a = a_start.copy()
    iteration = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        # record best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Sub {iteration:07d}] new best = {best_val:.6f}")

        # sub‑gradient: locate the peak index
        k_max = int(np.argmax(b))

        # gradient g_i = a_{k_max - i} (zero outside range)
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] = a[k_max - idx]

        # step size: linear decay from 0.12 to 0.001
        now = time.time()
        progress = (now - (deadline - (deadline - start_time))) / (deadline - start_time)
        step = 0.12 * (1.0 - progress) + 0.001 * progress

        # projected sub‑gradient step
        a = _project_simplex(a - step * g)

        # occasional jitter to escape shallow flats
        if np.random.rand() < 0.001:
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)

        iteration += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase H – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> np.ndarray:
    """
    Deterministic local search that moves a small amount of mass from a
    source index (large contribution to the current max) to a target index
    (small contribution).  Returns the best vector seen before *deadline*.
    """
    a = a_start.copy()
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac = 0.3                               # maximal fraction of source mass to move
    IMPROVE_LIMIT = 120_000                      # stagnation trigger
    stagnation = 0
    it = 0

    while time.time() < deadline:
        # 1) locate current peak of the convolution
        k_max = int(np.argmax(b))

        # 2) compute partner values for this peak: p_i = a_{k_max - i}
        partners = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        partners[idx] = a[k_max - idx]

        # 3) contribution of each coordinate to the peak
        contrib = a * partners

        # source: among the 15 largest contributions
        src_pool = np.argpartition(-contrib, min(15, n - 1))[:15]
        i = np.random.choice(src_pool)
        if a[i] <= 1e-14:
            # nothing to move – pick another source next round
            it += 1
            continue

        # target: choose an index whose partner is small (prefer zero)
        low_thr = np.percentile(partners, 20)
        low_mask = partners <= low_thr
        low_indices = np.where(low_mask)[0]
        if low_indices.size == 0:
            j = np.random.randint(n)
        else:
            j = np.random.choice(low_indices)

        if j == i:
            it += 1
            continue

        # amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            it += 1
            continue

        # incremental O(n) update of the convolution
        b_new = b.copy()
        # 2·δ·a cross‑terms
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept strictly improving moves
        if new_val < cur_val - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            stagnation = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {it:07d}] new best = {best_val:.6f}")
        else:
            stagnation += 1

        # stagnation handling
        if stagnation > IMPROVE_LIMIT:
            # restart from the best point with a tiny random nudge
            a = best_a.copy()
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            stagnation = 0
            print(f"[Hill {it:07d}] stagnation → perturbing")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a step‑function under the given time budget.
    Returns a list of non‑negative floats (Σ ≈ 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size – 4000 gives a good compromise.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise `a`.  If a previous construction exists, interpolate it.
    # --------------------------------------------------------------
    try:
        # height_sequence_1 is expected to be defined globally by the harness
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)        # enforce Σa = 1
    best_a = a.copy()
    best_val = float('inf')

    # --------------------------------------------------------------
    # Phase C – L²‑flattening (gradient descent on Σ(b‑avg)²)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.42 * budget_s
    avg_target = 1.0 / (2 * n - 1)          # flat target value of b
    iter_c = 0
    while time.time() < phase_c_deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_a = a.copy()
            print(f"[L2 {iter_c:07d}] new best = {best_val:.6f}")

        # residual to flat target
        r = b - avg_target

        # gradient of Σ(r_k²) w.r.t a
        a_rev = a[::-1]
        conv1 = _conv_fft(r, a)
        conv2 = _conv_fft(r, a_rev)
        start = n - 1
        end = 2 * n - 1
        grad = 2.0 * (conv1[start:end] + conv2[start:end])

        # step size schedule – a bit more aggressive at the beginning
        now = time.time()
        progress = (now - start_time) / (phase_c_deadline - start_time)
        step = 0.55 * (1.0 - progress) + 0.004 * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional jitter
        if np.random.rand() < 0.004:
            a += np.random.normal(scale=0.008, size=n)
            a = _project_simplex(a)

        iter_c += 1

    # --------------------------------------------------------------
    # Phase S – Sub‑gradient descent on the true objective (max(b))
    # --------------------------------------------------------------
    phase_s_deadline = start_time + 0.65 * budget_s
    best_a, best_val = _subgradient_max_phase(best_a, n, phase_s_deadline,
                                              best_a, best_val)

    # --------------------------------------------------------------
    # Phase H – Targeted hill‑climbing on the exact objective
    # --------------------------------------------------------------
    best_a = _hillclimb_targeted(best_a, n, deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation using the provided secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
