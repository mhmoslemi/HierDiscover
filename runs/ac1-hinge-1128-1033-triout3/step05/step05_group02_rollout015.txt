"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s):
1. Choose a modest size (default 8000).  Larger n makes the
   theoretical lower bound 2·n/(2·n‑1) ≈ 1, so we aim for <1.503.
2. Initialise from a known construction (``height_sequence_1``) if
   available – otherwise start with a uniform distribution.
3. Run many independent L‑BFGS optimisations on a smooth
   surrogate of the peak of the self‑convolution.
   – The surrogate is
       L_ε(a)=2·n·max(a*a)+2·n·ε·log‑sum‑exp((a*a‑max)/ε)
   – A log‑sum‑exp with a decreasing schedule of ε (≈ 5·10⁻³ → 1·10⁻⁹)
     yields ever tighter approximations to the true max.
   – Parameters are expressed via a soft‑max (guarantees a_i≥0 and
     Σa_i=1); the gradient is derived analytically for efficiency.
4. After each L‑BFGS run apply a few cheap permutations of the
   resulting vector:

   * interleaved high/low,
   * a random shuffle.
   Keep the best permutation.

   * Compute the full convolution; locate the lag k where the
     convolution attains its maximum.

     peak (i.e. a_i · a_j with i+j=k).

   The loop runs until the time budget expires.
6. Return the best sequence observed.

The method combines a smooth optimisation (fast, gradient‑based)
with combinatorial re‑ordering and a focused local search that directly
reduces the offending convolution peak, typically pushing the metric
below the 1.503 target within the allotted time.

_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len‑1) via FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two vectors via FFT (full length)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate objective L_ε and its gradient w.r.t. the
    soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    where conv = a * a (full linear convolution).
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # convolution a*a
    conv = _full_convolution(a)               # length 2·n‑1
    max_conv = conv.max()

    # smooth max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)              # log ∑ exp(...)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (see derivation in the comment block above)
    weights = np.exp(shifted)                 # ∝ exp((conv‑max)/ε)
    p = weights / weights.sum()               # probability distribution over k

    # convolution a * reversed(p)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)            # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]           # middle n entries
    grad_a = 4.0 * n * seg[::-1]             # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper for scipy.optimize that supplies the current eps and n."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _reverse_sorted(a: np.ndarray) -> np.ndarray:
    """Return a copy where values are sorted descending (largest ↔ smallest)."""
    return np.sort(a)[::-1]


def _interleaved_high_low(a: np.ndarray) -> np.ndarray:
    """
    Interleave the sorted values high‑low:  max, min, 2nd‑max, 2nd‑min, …
    This creates an alternating pattern that often reduces peaks in
    the self‑convolution.
    """
    vals = np.sort(a)[::-1]          # descending
    out = np.empty_like(vals)
    l, r = 0, len(vals) - 1
    idx = 0
    while l <= r:
        out[idx] = vals[l]
        idx += 1
        l += 1
        if l <= r:
            out[idx] = vals[r]
            idx += 1
            r -= 1
    return out

                if val < cur_val - 1e-12:
                    cur_a, cur_val = cand, val
                    print(f"[swap] new best = {cur_val:.6f}")
                continue

        # Compute the full convolution and locate the problematic lag
        conv = _full_convolution(cur_a)
        k = int(np.argmax(conv))

        # Contributions of each index to this lag
        idxs = np.arange(n)
        partner = k - idxs
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=cur_a.dtype)
        valid = idxs[mask]
        contrib[valid] = cur_a[valid] * cur_a[partner[mask]]

        # Choose donor = index with largest contribution (and positive mass)
        donor = int(np.argmax(contrib))
        # Its partner index (the other side of the product)
        partner_of_donor = k - donor

        # Choose a recipient with the smallest current height (avoid donor/partner)
        candidate_mask = np.ones(n, dtype=bool)
        candidate_mask[donor] = False
        candidate_mask[partner_of_donor] = False
        # pick the smallest entry among the allowed ones
        if not candidate_mask.any():
            continue
        # replace disallowed entries with +inf so argmin works
        tmp = cur_a.copy()
        tmp[~candidate_mask] = np.inf
        recipient = int(np.argmin(tmp))

        # Transfer a small fraction from donor to recipient (max 5 % of donor)
        delta = cur_a[donor] * 0.05 * rng.random()
        if delta <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[donor] -= delta
        new_a[recipient] += delta

        # Ensure non‑negativity (should hold by construction)
        if new_a[donor] < -1e-12:
            continue

        # Evaluate
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:
            cur_a, cur_val = new_a, val
            print(f"[targeted] improved to {cur_val:.6f}")
    return cur_a, cur_val


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Search for the step‑function with the minimal value returned by
    ``evaluate_sequence``.  The routine respects the time budget,
    runs many L‑BFGS restarts on a smoothed surrogate, applies several
    cheap permutations, and finishes with a targeted hill‑climb.
    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Reserve about 30 s for the final refinement
    deadline = start_time + budget_s - 30.0

    # ------------------------------------------------------------------
    # Problem size – a bigger n gives a lower theoretical bound.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from an existing construction if possible.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = np.full(n, 1.0 / n)  # uniform start

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()            # enforce Σa = 1
    best_a = a0.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS schedule (log‑space smoothing)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=13)

    restart = 0
    # Main multi‑restart loop – stop early enough to keep refinement time
    while time.time() < deadline - 15.0:
        restart += 1
        print(f"[restart {restart}]")

        # Random soft‑max parameters (broad distribution for diversity)
        w = np.random.randn(n) * 2.0

        # Run the full epsilon schedule for this restart
        for eps in eps_schedule:
            if time.time() >= deadline - 10.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            # L‑BFGS‑B on the smoothed objective
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 200,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # Exact evaluation of the current point
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a = a / a.sum()
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # --------------------------------------------------------------
        # Try a few simple permutations of the current best vector.
        # --------------------------------------------------------------
        # 1) reverse‑sorted (largest ↔ smallest)
        a_rev = _reverse_sorted(best_a)
        val_rev = evaluate_sequence(a_rev.tolist())
        if val_rev < best_val:
            best_val, best_a = val_rev, a_rev.copy()
            print(f"[perm rev] improved to {best_val:.6f}")

        # 2) interleaved high/low pattern
        a_int = _interleaved_high_low(best_a)
        val_int = evaluate_sequence(a_int.tolist())
        if val_int < best_val:
            best_val, best_a = val_int, a_int.copy()
            print(f"[perm inter] improved to {best_val:.6f}")

        # 3) a random shuffle (occasionally useful)
        a_shuf = best_a.copy()
        np.random.shuffle(a_shuf)
        val_shuf = evaluate_sequence(a_shuf.tolist())
        if val_shuf < best_val:
            best_val, best_a = val_shuf, a_shuf.copy()
            print(f"[perm shuffle] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final targeted refinement (mass transfers around the dominant lag)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] starting targeted hill‑climb")
        best_a, best_val = _targeted_refine(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
