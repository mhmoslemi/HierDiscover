"""
Smooth‑max Adam with continuation and multi‑restart.

We minimise the scale‑invariant quantity

    F(a) = 2·n·max_k (a * a)[k]       (∑ a_i = 1, a_i ≥ 0)

by optimising the log‑probabilities x (softmax(x)=a).  Instead of the
nondifferentiable max we use a smooth approximation

    fβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k]),      β ↑ 1→∞

and perform Adam on x.  β is increased gradually (continuation) so the
optimiser first sees a smooth landscape and later focuses on the true
max‑objective.  Several random seeds (including the previous best
construction, Dirichlet draws and handcrafted families) are explored;
the best vector found in the allotted budget is returned.

The algorithm runs entirely with NumPy (FFT‑based convolutions) and
fits comfortably into the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: generic full linear convolution via real FFT
# ----------------------------------------------------------------------
def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution a * b (real FFT, length = len(a)+len(b)-1)."""
    size = len(a) + len(b) - 1
    # next power of two for speed (optional)
    nfft = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(fa * fb, n=nfft)
    return conv[:size]


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0:
        # fall back to uniform if overflow killed everything
        return np.full_like(x, 1.0 / len(x))
    return e / s


def score_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective value 2·n·max(conv) (sum of a is 1)."""
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Parametric families (used as seeds)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def run_smooth_adam(init_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Optimises a probability vector `init_a` (∑=1, a≥0) using Adam on the
    log‑probabilities with a smoothly‑approximated max in the objective.
    Returns the best vector encountered and its (exact) score.
    """
    start = time.time()
    n = init_a.shape[0]

    # initialise log‑parameters (protect against log(0))
    eps = 1e-14
    x = np.log(np.maximum(init_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.3                         # base stepsize in log‑space

    # continuation on β (smoothness)
    beta = 1.0
    beta_max = 2000.0
    beta_factor = 1.02
    beta_update_every = 2000        # iterations

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    while time.time() - start < time_limit - 0.2:
        iteration += 1

        # ------- probabilities -------
        a = softmax(x)

        # ------- self‑convolution -------
        conv = convolve_fft(a, a)           # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ------- smooth‑max weights -------
        # w_i ∝ exp(β·conv_i)
        scaled = beta * conv
        max_scaled = np.max(scaled)
        w_raw = np.exp(scaled - max_scaled)   # subtract max for stability
        w = w_raw / w_raw.sum()               # ∑ w_i = 1

        # ------- gradient w.r.t. a (smooth version) -------
        a_rev = a[::-1]
        conv_rev = convolve_fft(a_rev, w)    # length n + (2n‑1) -1 = 3n‑2
        # extract the n elements that correspond to G_i = 2·Σ_k w_k·a_{k‑i}
        start_idx = n - 1
        G = 2.0 * conv_rev[start_idx:start_idx + n]

        # ------- chain rule through softmax -------
        grad_a = G
        avg = np.dot(a, grad_a)               # scalar = Σ a_i ∂f/∂a_i
        grad_x = a * (grad_a - avg)           # ∂f/∂x_i

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of learning‑rate (helps stability for long runs)
        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- bookkeeping -------
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # ------- re‑heat if stuck -------
        if iteration - last_improve > 5000:
            # inject a tiny random perturbation and reset moments
            noise = np.random.randn(n) * 1e-5
            x += noise
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

        # ------- increase β (smoothness) -------
        if iteration % beta_update_every == 0:
            old_beta = beta
            beta = min(beta * beta_factor, beta_max)
            if beta > old_beta:
                print(f"Iter {iteration:6d}: increase β to {beta:.1f}")

    return best_vec, best_score

    if best_overall_vec is None:
