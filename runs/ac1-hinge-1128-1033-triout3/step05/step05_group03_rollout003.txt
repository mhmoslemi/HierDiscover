def simulated_annealing(a, n, start, deadline, best_a, best_val):
    b = full_convolution(a)
    cur_val = 2*n * b.max()
    iteration = 0
    # schedules:
    max_frac_start = 0.5
    max_frac_end = 0.02
    T_start = 0.01
    T_end = 1e-6

    while time < deadline:
        progress = (time - start) / (deadline - start)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress
        T = T_start * (1 - progress) + T_end * progress # linear decreasing
        # locate max index
        k_max = int(np.argmax(b))
        # compute contributions to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            p = contrib / total_contrib
            i = np.random.choice(n, p=p)
        else:
            i = np.random.choice(np.where(a > 1e-12)[0])  # any positive mass
        if a[i] <= 1e-15:
            continue
        # choose target j != i
        j = np.random.randint(n-1)
        if j >= i: j += 1
        # delta
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0:
            continue
        # update b_new
        b_new = b.copy()
        # shift contributions
        b_new[j:j+n] += 2.0*delta*a
        b_new[i:i+n] -= 2.0*delta*a
        # quadratic terms
        b_new[2*j] += delta*delta
        b_new[2*i] += delta*delta
        b_new[i+j] -= 2.0*delta*delta
        new_val = 2.0*n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # accept with probability exp(-(new_val - cur_val)/T)
            if np.random.rand() < np.exp(-(new_val - cur_val)/T):
                accept = True
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA {iteration}] new best = {best_val:.6f}")
                if best_val <= target_val: break
        else:
            # optionally track rejections
            pass

        # stagnation handling: if many iterations without improvement, perturb.
        # Not necessary as SA allows occasional acceptance.
        iteration += 1

    return best_a
```

"""
Non‑negative step function optimizer.

The algorithm proceeds in three stages within the allotted time:

1. **Adam on a smooth‑max surrogate** – the vector a is represented by soft‑max
   logits w (ensuring Σa = 1, a ≥ 0).  A smooth approximation of
   `max(conv(a,a))` is built with a log‑sum‑exp temperature ε that is
   exponentially reduced.  Adam updates the logits, occasional random
   perturbations and restarts help escape local minima.

2. **Simulated‑annealing (SA) refinement** – starting from the best Adam
   point we perform mass‑transfer moves (δ taken from a “source’’ index i
   and added to a “target’’ index j).  The move is accepted if it improves
   the exact objective, otherwise with probability exp(‑Δ/T) where the
   temperature T decays linearly.  The maximal step size `max_frac`
   (fraction of the source mass that can be moved) also decays, allowing
   coarse exploration early on and fine tuning later.

3. **Pure improving hill‑climb** – a last pass that only accepts strictly
   improving moves, further polishing the solution.

All stages keep the sum of the coefficients equal to 1 and respect the
non‑negativity constraint.  The best sequence seen during the whole run
is returned as a plain Python list.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a sum‑normalised vector."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max and its gradient (for Adam)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float) -> Tuple[float, np.ndarray]:
    """
    Compute the smooth objective 2·n·smooth_max(conv(a,a)) and its gradient
    w.r.t. the logits `w`.  The current a = softmax(w) (Σa = 1, a ≥ 0).
    """
    # a = softmax(w)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    n = a.shape[0]

    # full self‑convolution
    b = _full_convolution(a)

    # smooth maximum via log‑sum‑exp
    max_b = b.max()
    # weights for the log‑sum‑exp (avoid overflow)
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    sum_weights = weights.sum()
    smooth_max = max_b + eps * np.log(sum_weights)

    # gradient of smooth_max w.r.t. a:
    #   ∂smooth_max/∂b_i = weights_i / sum_weights = p_i
    p = weights / sum_weights
    # conv(p_rev, a) will give ∂smooth_max/∂a (up to factor 2)
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                     # length 2·n‑1
    segment = conv_ab[n - 1: 2 * n - 1]               # n entries relevant for ∂/∂a
    grad_a = 4.0 * n * segment[::-1]                 # gradient of 2·n·smooth_max w.r.t. a

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    obj = 2.0 * n * smooth_max
    return obj, grad_w


# ----------------------------------------------------------------------
# Phase A – Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max objective.
    Returns the best (a, value) observed before *deadline*.
    """
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)            # initialise logits

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 25000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # schedule for ε and learning rate
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # objective and gradient at current w
        obj, grad_w = _smooth_obj_and_grad(w, eps)

        # evaluate exact objective (for tracking)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        exact_val = _objective_exact(a)

        if exact_val < best_val:
            best_val, best_a = exact_val, a.copy()
            no_improve = 0
            print(f"[Adam   {iteration:7d}] exact best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.01, size=n)

        # restart on stagnation
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam   {iteration:7d}] restarting after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Simulated‑annealing refinement on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray,
                         start: float,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float,
                         target_val: float = 1.503) -> Tuple[np.ndarray, float]:
    """
    Mass‑transfer simulated‑annealing on the exact objective.
    Only moves that keep the total mass (=1) are considered.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # SA schedules
    max_frac_start, max_frac_end = 0.5, 0.02
    T_start, T_end = 0.02, 1e-5

    iteration = 0
    last_improve_iter = 0
    IMPROVE_LIMIT = 250000

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = T_start * (1.0 - progress) + T_end * progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (biased by contribution)
        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index (uniform, different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                last_improve_iter = iteration
                print(f"[SA     {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            # No acceptance – just continue
            pass

        # stagnation handling: if we haven't improved for a long while,
        # re‑inject a small random perturbation around the best seen.
        if iteration - last_improve_iter > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            last_improve_iter = iteration
            print(f"[SA     {iteration:7d}] perturbing after stagnation")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – final pure hill‑climb (strictly improving moves only)
# ----------------------------------------------------------------------
def _final_hillclimb(a_start: np.ndarray,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> Tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing that only accepts strictly improving moves.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.02, 0.002
    IMPROVE_LIMIT = 300000
    no_improve = 0
    iteration = 0
    last_improve_iter = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # pick a random target j ≠ i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-14:        # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            last_improve_iter = iteration
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[Hill   {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – small random jitter around best
        if iteration - last_improve_iter > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            last_improve_iter = iteration
            print(f"[Hill   {iteration:7d}] perturbing after stagnation")
        iteration += 1

    return best_a, best_val
