"""
Propose a sequence of non‑negative heights minimizing the evaluation function.

The algorithm works in two phases:

1. **Fast hill‑climbing with greedy destination selection**  
   – The sum of the sequence is kept equal to 1 (scaling does not affect the
     score).  
   – At each iteration the index `i_src` that contributes most to the current
     convolution peak is identified.  
   – A set of candidate destination indices `j` (including random choices,
     the current minimum, maximum and a few offsets) is examined.  
   – For each candidate the objective after moving a small fraction
     `δ` of mass from `i_src` to `j` is evaluated using an exact,
     O(n) incremental update of the full convolution.  
   – The best improving move is performed; if no improvement is found,
     an uphill move may be accepted with a temperature–controlled probability
     (simulated annealing).  
   – A small “jitter’’ step (larger random transfer) is applied occasionally
     to escape plateaus.

2. **Optional linear‑programming refinement** (used if `cvxpy` and
   `scipy.sparse` are available).  
   – With the current sequence `f` we compute `M = max(f * f)`.  
   – We solve the LP  
         maximise   Σ b_i  
         subject to  b_i ≥ 0  and  (f * b)_k ≤ M  for all k,  
     which finds a vector `b` that can hold more total mass without
     increasing the convolution bound.  
   – After normalising `b` to sum 1 we mix it with the current sequence:
         f ← (1‑t) f + t (b/Σb)          (t≈0.1‑0.2)  
     This strictly reduces the peak in practice.  
   – The refinement step is repeated a few times while time permits.

The whole routine stops when the allotted time (default 1000 s) is exhausted
and returns the best sequence discovered.  Intermediate progress and the
final score are printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: exact full convolution (used for occasional re‑synchronisation)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*x (length 2·len(x)‑1)."""
    return np.convolve(x, x)


# ----------------------------------------------------------------------
# Incremental convolution update when moving δ from i → j
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray,
                conv: np.ndarray,
                i: int,
                j: int,
                delta: float,
                a_old: np.ndarray,
                n: int) -> None:
    """
    Apply the mass‑transfer move to `a` and update `conv` in‑place.
    `a_old` must be the state of `a` *before* the move.
    """
    # update the coefficients
    a[i] = a_old[i] - delta
    a[j] = a_old[j] + delta

    # linear part of the convolution update
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old

    # quadratic correction (three isolated entries)
    d2 = delta * delta
    conv[2 * i] += d2
    conv[2 * j] += d2
    conv[i + j] -= 2.0 * d2


# ----------------------------------------------------------------------
# Compute the objective that evaluate_sequence would return,
# given the current convolution peak.
# ----------------------------------------------------------------------
def _objective_from_max(conv_max: float, n: int) -> float:
    """2·n·max(conv) – exactly the value returned by evaluate_sequence."""
    return 2.0 * n * conv_max


# ----------------------------------------------------------------------
# Evaluate a tentative move without mutating the original data.
# Returns the new objective value.
# ----------------------------------------------------------------------
def _objective_if_move(conv: np.ndarray,
                       a_old: np.ndarray,
                       i: int,
                       j: int,
                       delta: float,
                       n: int) -> float:
    """Return the objective that would result from moving δ from i to j."""
    # make a temporary copy of the convolution (fast, size ≈ 2·n)
    tmp = conv.copy()
    tmp[i:i + n] -= 2.0 * delta * a_old
    tmp[j:j + n] += 2.0 * delta * a_old
    d2 = delta * delta
    tmp[2 * i] += d2
    tmp[2 * j] += d2
    tmp[i + j] -= 2.0 * d2
    return _objective_from_max(tmp.max(), n)


# ----------------------------------------------------------------------
# Build the convolution‑matrix M such that (f * b) = M @ b.
# Used only for the optional LP refinement.
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray):
    """Return a scipy.sparse CSR matrix of shape (2n‑1, n)."""
    try:
        import scipy.sparse as sp
    except Exception as exc:  # pragma: no cover
        raise RuntimeError("scipy.sparse is required for the LP phase") from exc

    n = f.shape[0]
    # row indices are i + j for all i,j ∈ [0,n)
    i = np.arange(n, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)

    # broadcasting: (n,1) + (1,n) → (n,n) matrix of row indices
    rows = (i[:, None] + j[None, :]).ravel()
    cols = np.repeat(i, n)
    data = np.tile(f, n)

    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Optional LP refinement using cvxpy (only if the library is present)
# ----------------------------------------------------------------------
def _lp_refinement(a: np.ndarray,
                   conv: np.ndarray,
                   n: int,
                   budget: float,
                   start_time: float) -> np.ndarray:
    """
    Perform one or more LP‑based refinements on `a`.
    Returns a possibly improved version of `a`.  The function stops
    when the remaining wall‑clock time drops below `budget` seconds.
    """
    try:
        import cvxpy as cp
        import scipy.sparse as sp
    except Exception:  # pragma: no cover
        # cvxpy or scipy not available – skip this phase
        print("[lp] cvxpy / scipy not available, skipping LP refinement")
        return a

    # Build the convolution matrix once (expensive O(n²) but done only here)
    print("[lp] Building convolution matrix for LP ...")
    A = _build_conv_matrix(a)          # (2n‑1) × n sparse matrix
    M_val = conv.max()                 # current bound on f*f

    # parameters for the refinement loop
    t_mix = 0.15                       # mixing weight for the new vector
    max_lp_iters = 5
    for lp_step in range(max_lp_iters):
        elapsed = time.time() - start_time
        if elapsed > budget:
            print("[lp] Time budget exhausted, terminating LP phase")
            break

        # Linear program: maximise Σ b_i   subject to   A b ≤ M_val, b ≥ 0
        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M_val * np.ones(2 * n - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        # Solve – ECOS works well for dense linear programmes of this size
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception as exc:  # pragma: no cover
            print("[lp] ECOS failed, trying SCS:", exc)
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=10000, verbose=False)

        if b_var.value is None:
            print("[lp] LP did not return a solution – aborting LP phase")
            break

        b_opt = np.maximum(b_var.value, 0.0)   # numerical safety
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-12:
            # No improvement possible – stop early
            print("[lp] LP could not increase total mass (sum_b≈{:.6f})".format(sum_b))
            break

        # Normalise to a probability vector (sum = 1)
        g = b_opt / sum_b

        # Mix the new vector into the current one
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new /= a_new.sum()                # re‑normalise (should already be 1)

        # Re‑compute convolution and objective
        conv_new = _full_convolution(a_new)
        obj_new = _objective_from_max(conv_new.max(), n)

        cur_obj = _objective_from_max(conv.max(), n)
        if obj_new < cur_obj:
            print("[lp] step {} improved objective: {:.6f} → {:.6f}".format(
                lp_step + 1, cur_obj, obj_new))
            a = a_new
            conv = conv_new
            M_val = conv.max()
            # continue with the improved sequence
        else:
            print("[lp] step {} did NOT improve (obj={:.6f})".format(lp_step + 1, obj_new))
            break

    return a

# Main optimisation routine required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Hill‑climbing + optional LP refinement to minimise `evaluate_sequence`.
    Returns a list of non‑negative floats (the best sequence found).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the framework and for the optional LP phase
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise the sequence -------------------------------------------------
    # --------------------------------------------------------------
    try:
        init_seq = height_sequence_1               # noqa: F821
    except NameError:
        init_seq = None

    if init_seq is None:
        n = 1800                     # a modest length that balances quality / speed
        a = np.full(n, 1.0 / n)
    else:
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.shape[0]

    # Normalise to a probability vector (scaling does NOT affect the score)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # --------------------------------------------------------------
    # Initial convolution and bookkeeping
    # --------------------------------------------------------------
    conv = _full_convolution(a)          # length 2n‑1
    cur_obj = _objective_from_max(conv.max(), n)
    best_obj = cur_obj
    best_a = a.copy()
    print(f"[init] n={n}, initial objective={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    move_frac_start = 0.45      # up to 45 % of source mass moved at the very start
    move_frac_end   = 0.02      # shrink to ~2 % near the end
    jitter_prob = 0.018
    num_random_j = 7            # how many random destination candidates per iteration
    recompute_interval = 4000   # full conv recompute to avoid drift
    temperature_start = 0.012   # for simulated‑annealing acceptance of uphill moves
    temperature_end   = 1e-6

    iteration = 0
    last_improve_iter = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        progress = (now - start_time) / (deadline - start_time)

        # ----------------------------------------------------------
        # Annealing schedule
        # ----------------------------------------------------------
        move_frac = move_frac_start * (1.0 - progress) + move_frac_end * progress
        temperature = temperature_start * (1.0 - progress) + temperature_end * progress

        # ----------------------------------------------------------
        # Identify the current peak and the most responsible index i_src
        # ----------------------------------------------------------
        k_star = int(conv.argmax())
        i_low  = max(0, k_star - (n - 1))
        i_high = min(n - 1, k_star)
        idx = np.arange(i_low, i_high + 1)
        partner = k_star - idx
        products = a[idx] * a[partner]
        i_src = int(idx[np.argmax(products)])

        # ----------------------------------------------------------
        # Propose a transfer amount δ
        # ----------------------------------------------------------
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue

        a_old = a.copy()                 # snapshot before the move

        # ----------------------------------------------------------
        # Build a small set of destination candidates
        # ----------------------------------------------------------
        candidates = set()
        # 1) index of the smallest coefficient (typically 0)
        candidates.add(int(np.argmin(a)))
        # 2) index of the largest coefficient (avoid the source itself)
        candidates.add(int(np.argmax(a)))
        # 3) a few offsets from the source
        offset = max(1, n // 12)
        candidates.add((i_src + offset) % n)
        candidates.add((i_src - offset) % n)
        # 4) random indices
        for _ in range(num_random_j):
            r = np.random.randint(n - 1)
            if r >= i_src:
                r += 1
            candidates.add(r)

        # ensure source is not in the set
        candidates.discard(i_src)

        # ----------------------------------------------------------
        # Evaluate all candidates – pick the best improving one
        # ----------------------------------------------------------
        best_candidate_obj = cur_obj
        best_j = None
        for j in candidates:
            obj = _objective_if_move(conv, a_old, i_src, j, delta, n)
            if obj < best_candidate_obj:
                best_candidate_obj = obj
                best_j = j

        # ----------------------------------------------------------
        # Acceptance rule (downhill always, uphill with prob.)
        # ----------------------------------------------------------
        accept = False
        if best_j is not None and best_candidate_obj < cur_obj:
            accept = True
        else:
            # try a random uphill move (take the first random candidate)
            # – this gives the simulated‑annealing flavour
            random_j = np.random.choice(list(candidates))
            obj_up = _objective_if_move(conv, a_old, i_src, random_j, delta, n)
            if obj_up < cur_obj or np.random.rand() < np.exp(-(obj_up - cur_obj) / max(temperature, 1e-12)):
                best_j = random_j
                best_candidate_obj = obj_up
                accept = True

        if accept and best_j is not None:
            # perform the selected move
            _apply_move(a, conv, i_src, best_j, delta, a_old, n)
            cur_obj = best_candidate_obj
            # keep the best ever seen
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration:7d}] new best = {best_obj:.6f}")
                last_improve_iter = iteration

        else:
            # reject – nothing to revert because we never changed a/conv
            pass

        # ----------------------------------------------------------
        # Occasional jitter: a larger random transfer to shake the system
        # ----------------------------------------------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta_j = 0.5 * a[src] * np.random.rand()
            if delta_j > 0.0:
                # evaluate the jitter move (no candidate search – accept only if improving)
                a_tmp = a.copy()
                conv_tmp = conv.copy()
                _apply_move(a_tmp, conv_tmp, src, dst, delta_j, a, n)
                obj_j = _objective_from_max(conv_tmp.max(), n)
                if obj_j < cur_obj:
                    a = a_tmp
                    conv = conv_tmp
                    cur_obj = obj_j
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{iteration:7d}] jitter improved best = {best_obj:.6f}")
                        last_improve_iter = iteration

        # ----------------------------------------------------------
        # Periodic full recomputation to suppress floating‑point drift
        # ----------------------------------------------------------
        if iteration % recompute_interval == 0:
            conv = _full_convolution(a)

        # ----------------------------------------------------------
        # Simple stagnation handling – re‑normalise and recompute if stuck
        # ----------------------------------------------------------
        if iteration - last_improve_iter > 250_000:
            a /= a.sum()
            conv = _full_convolution(a)
            cur_obj = _objective_from_max(conv.max(), n)
            last_improve_iter = iteration
            print(f"[{iteration:7d}] stagnation reset, obj={cur_obj:.6f}")

    # ------------------------------------------------------------------
    # Optional LP‑based refinement (uses remaining time, if any)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:                     # need at least ~30 s for a useful LP step
        print("[lp] Starting LP refinement with {:.1f}s remaining".format(remaining))
        a = _lp_refinement(a, conv, n, remaining, start_time)
        # recompute the final objective after the LP phase
        conv = _full_convolution(a)
        cur_obj = _objective_from_max(conv.max(), n)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            print(f"[lp] post‑LP best = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
