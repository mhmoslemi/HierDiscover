"""
Propose a candidate step‑function sequence that minimizes the evaluation
function defined in the competition.  The algorithm is a directed
hill‑climbing / simulated‑annealing search on the probability simplex.
At each iteration the index that contributes the most to the current
peak of the self‑convolution is identified and a small amount of its
mass is transferred to a location whose partner value (the coefficient
that pairs with it in the peak sum) is minimal.  The move is accepted
if it improves the objective, otherwise it may still be accepted with a
probability that decreases according to a linear temperature schedule.
Occasional jitter moves and periodic full recomputation / symmetrisation
help escape plateaus.  The routine runs for the allotted time budget
(≈ 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution using FFT (real input)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y using real FFT, result length len(x)+len(y)-1."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()    # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    evaluate_sequence(...).  Returns the best list of floats found before
    the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the surrounding framework
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise a – start from a known construction if available,
    # otherwise use a modest uniform distribution.
    # --------------------------------------------------------------
    try:
        init_seq = height_sequence_1          # noqa: F821
    except NameError:
        init_seq = None

    if init_seq is None:
        n = 2000                                 # a reasonable size for fast loops
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.shape[0]

    # Normalise to a probability vector (scaling does not affect the score)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a /= total

    # --------------------------------------------------------------
    # Initialise exact convolution and bookkeeping
    # --------------------------------------------------------------
    conv = _full_convolution_fft(a, a)           # length 2n‑1
    cur_max = conv.max()
    cur_obj = float(2.0 * n * cur_max)           # current objective = 2*n*max(conv)
    best_obj = cur_obj
    best_a = a.copy()

    print(f"[init] n={n}, initial objective={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for a 1000 s budget)
    # --------------------------------------------------------------
    max_frac = 0.20          # early‑stage max fraction of source mass to move
    min_frac = 0.005         # late‑stage min fraction
    jitter_prob = 0.02       # probability of a random “jitter” move per iteration
    recompute_interval = 3000   # full FFT recompute + symmetrisation interval
    temperature0 = cur_max * 0.05   # initial temperature for simulated annealing

    iteration = 0
    last_improve_iter = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        progress = (now - start_time) / (deadline - start_time)

        # ----------------------------------------------------------
        # Linear annealing of the transfer fraction
        # ----------------------------------------------------------
        move_frac = max_frac * (1.0 - progress) + min_frac * progress

        # ----------------------------------------------------------
        # Identify the current peak of the convolution and the most
        # responsible coefficient pair.
        # ----------------------------------------------------------
        k_star = int(conv.argmax())                           # position of the peak
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)                   # indices that actually appear in the sum
        partner = k_star - idx
        contrib = a[idx] * a[partner]                         # contribution of each pair to the peak
        i_src = int(idx[contrib.argmax()])                    # source index (largest contribution)

        # ----------------------------------------------------------
        # Compute partner values for *all* indices w.r.t. the current peak.
        # partner_all[j] = a[k_star - j] when the partner lies inside the array,
        # otherwise 0.  This tells us how much moving mass to j would add to the
        # current peak.
        # ----------------------------------------------------------
        partner_all = np.zeros(n, dtype=float)
        # the range of j for which the partner index is valid is 0 … k_star (clipped to n‑1)
        limit = min(k_star + 1, n)
        partner_all[:limit] = a[k_star - np.arange(limit)]

        # ----------------------------------------------------------
        # Choose a destination index with the smallest partner value
        # (ideally zero) and different from the source.
        # ----------------------------------------------------------
        candidates = np.where(np.arange(n) != i_src)[0]
        cand_vals = partner_all[candidates]
        min_val = cand_vals.min()
        dst_candidates = candidates[cand_vals == min_val]
        i_dst = int(np.random.choice(dst_candidates))

        # ----------------------------------------------------------
        # Propose a mass‑transfer move from i_src to i_dst.
        # ----------------------------------------------------------
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue

        a_old = a.copy()                         # keep a copy for the incremental update & possible revert

        # Incremental convolution update according to the exact formula:
        delta2 = delta * delta
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[i_dst:i_dst + n] += 2.0 * delta * a_old
        conv[2 * i_src] += delta2
        conv[2 * i_dst] += delta2
        conv[i_src + i_dst] -= 2.0 * delta2

        # Update the coefficients (sum is preserved exactly)
        a[i_src] -= delta
        a[i_dst] += delta

        # ----------------------------------------------------------
        # Evaluate the new objective
        # ----------------------------------------------------------
        new_max = conv.max()
        new_obj = float(2.0 * n * new_max)

        # ----------------------------------------------------------
        # Acceptance rule: downhill always accepted, uphill accepted
        # with probability decreasing with a linear temperature schedule.
        # ----------------------------------------------------------
        temperature = temperature0 * (1.0 - progress) + 1e-12
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_obj - cur_obj) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_obj = new_obj
            cur_max = new_max
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration:7d}] new best = {best_obj:.6f}")
                last_improve_iter = iteration
        else:
            # Revert the move
            a[:] = a_old
            conv[i_src:i_src + n] += 2.0 * delta * a_old
            conv[i_dst:i_dst + n] -= 2.0 * delta * a_old
            conv[2 * i_src] -= delta2
            conv[2 * i_dst] -= delta2
            conv[i_src + i_dst] += 2.0 * delta2

        # ----------------------------------------------------------
        # Occasional random “jitter” move to help escape plateaus.
        # ----------------------------------------------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta_j = 0.5 * a[src] * np.random.rand()
            if delta_j > 0.0:
                a_tmp = a.copy()
                delta2j = delta_j * delta_j
                conv[src:src + n] -= 2.0 * delta_j * a_tmp
                conv[dst:dst + n] += 2.0 * delta_j * a_tmp
                conv[2 * src] += delta2j
                conv[2 * dst] += delta2j
                conv[src + dst] -= 2.0 * delta2j

                a[src] -= delta_j
                a[dst] += delta_j

                jitter_max = conv.max()
                jitter_obj = float(2.0 * n * jitter_max)
                temperature_j = temperature0 * (1.0 - progress) + 1e-12
                accept_j = False
                if jitter_obj < cur_obj:
                    accept_j = True
                else:
                    prob_j = np.exp(-(jitter_obj - cur_obj) / temperature_j)
                    if np.random.rand() < prob_j:
                        accept_j = True

                if accept_j:
                    cur_obj = jitter_obj
                    cur_max = jitter_max
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{iteration:7d}] jitter improved best = {best_obj:.6f}")
                        last_improve_iter = iteration
                else:
                    # revert jitter
                    a[:] = a_tmp
                    conv[src:src + n] += 2.0 * delta_j * a_tmp
                    conv[dst:dst + n] -= 2.0 * delta_j * a_tmp
                    conv[2 * src] -= delta2j
                    conv[2 * dst] -= delta2j
                    conv[src + dst] += 2.0 * delta2j

        # ----------------------------------------------------------
        # Periodic full recomputation (and optional symmetrisation) to
        # avoid floating‑point drift.
        # ----------------------------------------------------------
        if iteration % recompute_interval == 0:
            # Enforce symmetry – the optimal solution is believed to be
            # symmetric, and symmetrising reduces noise.
            a = (a + a[::-1]) * 0.5
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)

        # ----------------------------------------------------------
        # Simple stagnation detection – apply a tiny random perturbation
        # if no improvement for a long time.
        # ----------------------------------------------------------
        if iteration - last_improve_iter > 500_000:
            a = np.maximum(a + 1e-4 * np.random.rand(n), 0.0)
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)
            last_improve_iter = iteration

    # ------------------------------------------------------------------
    # Final evaluation via the supplied evaluate_sequence() function.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
