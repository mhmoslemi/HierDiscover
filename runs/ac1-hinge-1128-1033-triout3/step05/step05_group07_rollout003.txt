  3. Chooses a destination index j for the transferred mass such that
     a_{k*‑j} is minimal – typically outside the range where a_{k*‑j} is defined,
     which directly reduces the peak.
  4. Transfers a fraction of a_i to a_j (δ) and updates the convolution
     incrementally (O(n) slice updates).
  5. Accepts the move if it improves the objective, otherwise accepts it
     with a temperature‑dependent probability (simulated annealing).
  6. Occasionally performs a larger “jitter’’ move and periodically
     recomputes the full convolution via FFT to avoid drift.
  7. Detects stagnation; if no improvement for many iterations it perturbs
     the best vector with a small random component.

The loop runs until the allotted time budget expires (≈1000 s) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast full convolution using real FFT (for occasional re‑sync)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (length len(x)+len(y)-1) using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Simulated‑annealing hill‑climbing for a step‑function sequence that
    minimises `evaluate_sequence`.  Starts from the globally available
    `height_sequence_1` if present, otherwise from a uniform distribution.
    Returns a list of non‑negative floats (the best sequence discovered).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin for the framework
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise a – use the provided best construction or a uniform start
    # --------------------------------------------------------------
    try:
        init_seq = height_sequence_1   # noqa: F821
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.shape[0]
    except NameError:
        n = 1800
        a = np.full(n, 1.0 / n)

    # Normalise to a probability vector (scaling does not affect the score)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # --------------------------------------------------------------
    # Initialise the exact convolution and bookkeeping
    # --------------------------------------------------------------
    conv = _full_convolution_fft(a, a)          # length 2n‑1
    cur_max = conv.max()
    cur_obj = float(2.0 * n * cur_max)          # current objective value
    best_obj = cur_obj
    best_a = a.copy()

    print(f"[init] n={n}, objective={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for the 1000 s budget)
    # --------------------------------------------------------------
    max_frac = 0.30        # start moving up to 30 % of source mass
    min_frac = 0.02        # shrink to ~2 % near the end
    jitter_prob = 0.02
    recompute_interval = 5000  # full FFT recompute to avoid drift
    T_start = 0.02         # initial temperature for simulated annealing
    T_end = 0.0

    iteration = 0
    last_best_iter = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        progress = (now - start_time) / (deadline - start_time)

        # ----------------------------------------------------------
        # Schedule move size (linear annealing)
        # ----------------------------------------------------------
        move_frac = max_frac * (1.0 - progress) + min_frac * progress
        temperature = T_start * (1.0 - progress) + T_end * progress

        # ----------------------------------------------------------
        # Determine the current peak of the convolution and the most
        # responsible coefficient pair.
        # ----------------------------------------------------------
        k_star = int(conv.argmax())

        # indices i that can contribute to k_star
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        partner = k_star - idx
        products = a[idx] * a[partner]
        i_src = int(idx[products.argmax()])      # source index (largest contribution)

        # ----------------------------------------------------------
        # Choose a destination index j that gives the smallest a_{k*-j}
        # (ideally a_{k*-j}=0, i.e. j > k_star or j < k_star-(n-1)).
        # ----------------------------------------------------------
        # Prefer indices with partner out of bounds → a_{k*-j}=0.
        if k_star + 1 <= n - 1:
            candidates = np.arange(k_star + 1, n, dtype=int)
        elif k_star - (n - 1) - 1 >= 0:
            candidates = np.arange(0, k_star - (n - 1), dtype=int)
        else:
            candidates = np.array([], dtype=int)

        # Ensure we do not pick the same index as the source.
        if candidates.size > 0:
            cand = candidates[candidates != i_src]
            if cand.size == 0:
                # fallback: pick any other index at random
                j_dst = np.random.randint(n - 1)
                if j_dst >= i_src:
                    j_dst += 1
            else:
                j_dst = int(np.random.choice(cand))
        else:
            # No zero‑partner candidate; pick index that minimises a_{k*-j}.
            # Work inside the range where partner is defined.
            cand_idx = idx[idx != i_src]
            cand_partner = k_star - cand_idx
            vals = a[cand_partner]
            if vals.size == 0:
                # fall back to a random index
                j_dst = np.random.randint(n - 1)
                if j_dst >= i_src:
                    j_dst += 1
            else:
                j_dst = int(cand_idx[vals.argmin()])

        # ----------------------------------------------------------
        # Propose a move: transfer a fraction of a[i_src] to a[j_dst]
        # ----------------------------------------------------------
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue

        a_old = a.copy()          # keep a full copy for possible revert

        # Incremental update of the convolution (exact formula)
        delta2 = delta * delta
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[j_dst:j_dst + n] += 2.0 * delta * a_old
        conv[2 * i_src] += delta2
        conv[2 * j_dst] += delta2
        conv[i_src + j_dst] -= 2.0 * delta2

        # Update the coefficients
        a[i_src] -= delta
        a[j_dst] += delta

        # ----------------------------------------------------------
        # Evaluate new objective
        # ----------------------------------------------------------
        new_max = conv.max()
        new_obj = float(2.0 * n * new_max)

        # ----------------------------------------------------------
        # Acceptance rule – downhill moves always accepted;
        # uphill moves accepted with probability exp(-(Δ)/T).
        # ----------------------------------------------------------
        accept = False
        if new_obj < cur_obj:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_obj - cur_obj) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_obj = new_obj
            cur_max = new_max
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration:8d}] new best = {best_obj:.6f}")
                last_best_iter = iteration
        else:
            # reject – revert both a and the convolution
            a[:] = a_old
            conv[i_src:i_src + n] += 2.0 * delta * a_old
            conv[j_dst:j_dst + n] -= 2.0 * delta * a_old
            conv[2 * i_src] -= delta2
            conv[2 * j_dst] -= delta2
            conv[i_src + j_dst] += 2.0 * delta2

        # ----------------------------------------------------------
        # Occasionally perform a random jitter move (a larger transfer)
        # ----------------------------------------------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta_j = 0.5 * a[src] * np.random.rand()
            if delta_j > 0.0:
                a_tmp = a.copy()
                delta2j = delta_j * delta_j
                conv[src:src + n] -= 2.0 * delta_j * a_tmp
                conv[dst:dst + n] += 2.0 * delta_j * a_tmp
                conv[2 * src] += delta2j
                conv[2 * dst] += delta2j
                conv[src + dst] -= 2.0 * delta2j

                a[src] -= delta_j
                a[dst] += delta_j

                jitter_obj = float(2.0 * n * conv.max())
                if jitter_obj < cur_obj or (temperature > 0.0 and
                                            np.random.rand() < np.exp(-(jitter_obj - cur_obj) / temperature)):
                    cur_obj = jitter_obj
                    cur_max = conv.max()
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{iteration:8d}] jitter improved best = {best_obj:.6f}")
                        last_best_iter = iteration
                else:
                    # revert jitter
                    a = a_tmp
                    conv[src:src + n] += 2.0 * delta_j * a_tmp
                    conv[dst:dst + n] -= 2.0 * delta_j * a_tmp
                    conv[2 * src] -= delta2j
                    conv[2 * dst] -= delta2j
                    conv[src + dst] += 2.0 * delta2j

        # ----------------------------------------------------------
        # Periodically recompute the convolution from scratch to avoid drift
        # ----------------------------------------------------------
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a, a)

        # ----------------------------------------------------------
        # Simple stagnation detection – apply a random perturbation if stuck
        # ----------------------------------------------------------
        if iteration - last_best_iter > 300_000:
            # blend the best known vector with a small random component
            alpha = 0.15
            noise = np.abs(np.random.randn(n))
            a = best_a * (1.0 - alpha) + noise * alpha
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_obj = float(2.0 * n * conv.max())
            last_best_iter = iteration
            print(f"[{iteration:8d}] perturbation applied, current obj={cur_obj:.6f}")

    # ----------------------------------------------------------------------
    # Final reporting
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
```
