def proj_simplex(v):
    # returns projection onto simplex {x >=0, sum x = 1}
    # Input: v array (any)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Algorithm
---------
1. Load a previously found construction (global ``height_sequence_1``) if
   available, otherwise start from a uniform distribution.  The vector is
   normalised so that its ℓ₁‑norm equals 1 (the objective is scale‑invariant).

2. **Sub‑gradient descent on the true objective.**  
   The objective for a fixed ℓ₁‑norm is  2·n·max(conv) .  At each iteration
   we compute the full autocorrelation, find the lag(s) where the
   maximum is attained and build a sub‑gradient (the sum of the gradients
   of the active lag‑convolutions).  A line‑search along the projected
   negative sub‑gradient is performed; the new point is projected onto the
   probability simplex (non‑negative, sums to 1).  This stage continues
   until the time budget is exhausted.

3. **Guided hill‑climbing refinement.**  
   After the sub‑gradient phase we apply a fast greedy mass‑transfer
   routine (the one used in the previous state‑of‑the‑art).  It moves a
   small fraction of mass away from the most offending pair of indices and
   deposits it at low‑mass locations, accepting only improvements of the
   exact ``evaluate_sequence`` value.

4. Return the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # remove tiny negative artefacts
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (onto {x ≥ 0, Σx = 1})
# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the last index where u_i > cssv_i / i
    cond = u > cssv / ind
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climbing that moves a small amount of mass away from the
    most offending lag.  Stops when the time budget is exhausted."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05  # initial fraction of mass to move

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        if ks.size == 0:
            break
        k = int(rng.choice(ks))

        # contribution a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # Choose two destination indices that are unlikely to recreate lag k
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:  # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val

def subgradient_descent(a: np.ndarray, n: int,
                        deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform projected sub‑gradient descent on 2·n·max(conv) with the ℓ₁‑norm fixed.
    Returns the best vector found and its exact evaluation.
    """
    cur_a = a.copy()
    # initialise best values using the exact evaluator
    try:
        best_val = evaluate_sequence(cur_a.tolist())
    except Exception:
        best_val = float("inf")
    best_a = cur_a.copy()
    step = 0.1          # initial step length for the line‑search
    min_step = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        cur_obj = 2.0 * n * max_b               # cheap objective (exact when Σa=1)

        # ----- active set (lags attaining the maximum) -----
        # use a tight relative tolerance
        tol = max(1e-15, max_b * 1e-12)
        active = np.where(conv >= max_b - tol)[0]

        # ----- sub‑gradient of the max‑convolution -----
        g = np.zeros_like(cur_a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            if i_start > i_end:
                continue
            j_start = k - i_end
            j_end = k - i_start
            # a segment reversed to align with i indices
            seg = cur_a[j_start:j_end + 1][::-1]
            g[i_start:i_end + 1] += 2.0 * seg
        # average over the active set (still a valid sub‑gradient)
        if active.size > 1:
            g /= float(active.size)

        # ----- line‑search on the projected direction -----
        improved = False
        alpha = step
        while alpha >= min_step:
            a_candidate = cur_a - alpha * g
            a_candidate = proj_simplex(a_candidate)
            conv_cand = _full_convolution(a_candidate)
            max_cand = conv_cand.max()
            obj_cand = 2.0 * n * max_cand
            if obj_cand < cur_obj - 1e-15:
                improved = True
                break
            alpha *= 0.5

        if improved:
            cur_a = a_candidate
            cur_obj = obj_cand
            # evaluate exactly and possibly update global best
            try:
                exact_val = evaluate_sequence(cur_a.tolist())
            except Exception:
                exact_val = float("inf")
            if exact_val < best_val:
                best_val = exact_val
                best_a = cur_a.copy()
                print(f"[sgd] new best = {best_val:.6f}")
            # gently increase step size after a success
            step = min(step * 1.05, 1.0)
        else:
            # no improvement – shrink step size
            step = max(step * 0.5, min_step)
            # occasional random jitter to escape plateaus
            if it % 100 == 0:
                perturb = rng.normal(scale=0.01, size=n)
                a_pert = cur_a + perturb
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = proj_simplex(a_pert)
                conv_pert = _full_convolution(a_pert)
                max_pert = conv_pert.max()
                obj_pert = 2.0 * n * max_pert
                if obj_pert < cur_obj:
                    cur_a = a_pert
                    cur_obj = obj_pert
                    print(f"[sgd] jitter improved to {obj_pert:.6f}")

        # simple progress output every 200 iterations
        if it % 200 == 0:
            print(f"[sgd] iter {it}, current obj ≈ {cur_obj:.6f}")

        # safety break if time is almost up
        if time.time() >= deadline - 0.1:
            break

    # final exact evaluation of the best vector found
    try:
        best_val = evaluate_sequence(best_a.tolist())
    except Exception:
        pass
    return best_a, best_val

            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start

    # enforce feasibility (non‑negative, ℓ₁‑norm = 1)
    a0 = np.maximum(a0, 0.0)
    a0_sum = a0.sum()
    if a0_sum == 0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 /= a0_sum

    # ---- evaluate the initial sequence ----
    try:
        init_val = evaluate_sequence(a0.tolist())
    except Exception:
        init_val = float("inf")
    print(f"[init] initial evaluation = {init_val:.6f}")

    # ---- sub‑gradient descent phase ----
    best_a, best_val = subgradient_descent(a0, n, deadline, rng)

    # ---- final guided hill‑climbing refinement (if time permits) ----
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ---- final report ----
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
