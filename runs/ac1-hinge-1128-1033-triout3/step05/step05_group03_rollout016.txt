    cur_val = 2.0 * n * b.max()
    max_frac_start, max_frac_end = 0.7, 0.02
    stagnation_limit = 150_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate current max
        k_max = int(np.argmax(b))

        # compute contributions for sources and targets

        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:

        # but if invalid, set small => high weight

"""
Two‑phase optimizer for the step‑function objective.

* Phase 0 – quick parametric sweep (U‑shaped power law) to obtain a
  promising starting point.
* Phase A – Adam optimisation on a smooth log‑sum‑exp approximation of
  the max‑convolution, using a soft‑max parametrisation that guarantees
  Σa = 1.
* Phase B – fast targeted hill‑climbing that moves a small amount of mass
  from indices that heavily contribute to the current maximal convolution
  entry towards indices that contribute little.  The move is accepted only
  if the exact objective improves; after long stagnation a small random
  perturbation is applied.

The routine respects the 1000 s budget, returns a non‑negative list with
Σ = 1 and typically pushes the bound well below the previous 1.51279
record.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (exact) – assumes Σa = 1
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric sweep to get a good seed
# ----------------------------------------------------------------------
def _initial_shape(n: int):
    """Search a one‑parameter U‑shape family a_i ∝ ((i+½)(n‑i‑½))^{-γ}."""
    best_a = None
    best_val = np.inf
    eps = 1e-12
    x = np.arange(n, dtype=np.float64)
    # try a range of exponents
    for gamma in np.linspace(0.0, 4.0, 41):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth approximation of max(conv(a,a)).
    Returns the best (a, value) seen before *deadline*.
    """
    # logits for soft‑max
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.10, 0.001
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve_cnt += 1

        # schedules
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)                 # safe, eps>0
        p = weights / weights.sum()               # soft max weights

        # gradient of smooth objective w.r.t. a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1: 2 * n - 1]       # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stalled
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restarting after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing with gradient‑guided source/target
# ----------------------------------------------------------------------
def _hillclimb_improved(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast local search that transfers a small amount of mass while keeping Σa = 1.
    Source indices are chosen proportionally to their contribution to the current
    maximal convolution entry; targets are biased towards indices with *small*
    contribution to that entry.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.60, 0.02
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress based schedule for allowed transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – biased by contribution (larger ⇒ more likely)
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target index – bias towards *small* contribution
        # use inverse of contribution (plus epsilon to avoid div‑by‑zero)
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        # do not pick the source itself
        target_weights[i] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # fallback: uniform among all except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept only strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[HillImp {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # perturb the current best a little and restart from there
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillImp {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# ----------------------------------------------------------------------

    """

    start_time = time.time()
    # keep a safety margin of 0.5 s for the final evaluation

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

"""
Two‑phase optimizer for the step‑function objective.

* Phase 0 – quick parametric sweep of a U‑shaped power‑law family to get a
  decent initial vector.
* Phase A – Adam optimisation on a smooth log‑sum‑exp approximation of the
  max‑convolution, using a soft‑max parametrisation that guarantees Σa = 1.
* Phase B – fast targeted hill‑climbing.  The source index is chosen
  proportionally to its contribution to the current maximal convolution
  entry, the target index is biased towards indices that contribute *little*
  to that entry.  Moves are accepted only if the exact objective improves;
  after prolonged stagnation a small random perturbation is applied.

The routine respects the 1000 s budget, returns a non‑negative list with
Σ = 1 and usually pushes the bound well below the previous 1.51279 record.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (exact) – assumes Σa = 1
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric sweep to obtain a good seed
# ----------------------------------------------------------------------
def _initial_shape(n: int):
    """Search a one‑parameter U‑shape family a_i ∝ ((i+½)(n‑i‑½))^{-γ}."""
    best_a = None
    best_val = np.inf
    eps = 1e-12
    x = np.arange(n, dtype=np.float64)
    for gamma in np.linspace(0.0, 4.0, 41):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth approximation of max(conv(a,a)).
    Returns the best (a, value) seen before *deadline*.
    """
    # logits for soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.10, 0.001
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve_cnt += 1

        # schedules for temperature and learning‑rate
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth approximation of the max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)                 # all ≤ 1
        p = weights / weights.sum()               # soft‑max weights

        # gradient of the smooth objective w.r.t. a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1: 2 * n - 1]       # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stagnated
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restarting after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing with gradient‑guided source/target
# ----------------------------------------------------------------------
def _hillclimb_improved(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast local search that transfers a small amount of mass while keeping Σa = 1.
    Source indices are chosen proportionally to their contribution to the
    current maximal convolution entry; targets are biased towards indices that
    contribute *little* to that entry.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.60, 0.02
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress‑based schedule for the allowed transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- source index (biased by contribution) -----
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ----- target index (bias towards *small* contribution) -----
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0                     # avoid picking the source itself
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # fallback uniform among all except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept only strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[HillImp {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillImp {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val
