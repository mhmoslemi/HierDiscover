def hillclimb_max(a, best_a, best_val, n, deadline):
    a = a.copy()
    b = full_convolution(a)
    cur_val = 2*n*b.max()
    while time.time() < deadline:
        # compute t* = argmax index of b
        t_star = np.argmax(b) # maybe multiple, pick first
        # compute contributions g_i
        g = np.zeros(n)
        low = max(0, t_star - (n-1))
        high = min(n-1, t_star) # i range that contributes
        # For i in low..high, g[i] = a[t_star - i]
        # we can fill via slice
        g[low:high+1] = a[t_star - np.arange(low, high+1)]
        # Compute product p_i
        p = a * g
        # source index: max p (or max a*g)
        i_source = int(np.argmax(p))
        # target index: choose index where g is minimal (preferably low a)
        # maybe pick j = argmin (g + epsilon*a) to avoid moving to location that may increase max
        j_target = int(np.argmin(g + a)) # combine
        if i_source == j_target:
            # choose random j that is not source
            j_target = (i_source + 1) % n
        # delta:
        delta = a[i_source] * np.random.rand() * 0.2 # smaller fraction
        if delta <= 0:
            continue
        # compute new b via O(n) update
        b_new = b.copy()
        # subtract
        b_new[i_source:i_source + n] -= 2.0 * delta * a
        # add
        b_new[j_target:j_target + n] += 2.0 * delta * a
        # delta^2 terms:
        b_new[2 * i_source] += delta * delta
        b_new[2 * j_target] += delta * delta
        b_new[i_source + j_target] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val:
            # accept
            a[i_source] -= delta
            a[j_target] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[HillMax {iteration}] new best = {best_val:.6f}")
        else:
            # maybe accept with probability? but we may skip
            pass
        # update iteration, maybe random perturb after stagnation
        # Keep track of streak...
    return best_a, best_val

"""
Propose a step‐function that minimises `evaluate_sequence`.

Algorithm
---------

We treat the problem as minimising `max(b)` where `b = a * a` (full
autoconvolution) under the constraints `a_i ≥ 0`, `∑ a_i = 1`.
The objective is `2·n·max(b)`.  Three complementary phases are used,
all running within the 1000 s budget:

1. **L2 flattening** – minimise the variance of `b` (i.e. ∑(b‑µ)²) by
   projected gradient descent.  This quickly drives the convolution
   towards a flat shape.

2. **Smoothed ‑∞ descent** – minimise a smooth approximation of the
   infinity norm, `L(β) = log ∑ exp(β·b) / β`.  The gradient of `L`
   can be written as a single FFT‑based convolution, so each iteration
   costs only two FFTs.  The temperature `β` is increased gradually,
   focusing the optimisation on the true maximum.

3. **Targeted hill‑climbing** – a fast local search that moves a tiny
   amount of mass from a coordinate that contributes heavily to the
   current peak of `b` to a coordinate that contributes little.
   The convolution is updated in O(n) time, and the best solution is
   kept throughout.

All updates are projected onto the probability simplex (non‑negative,
unit sum).  The best sequence found is returned as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (real FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself, length 2·len(x)‑1."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    # renormalise numerically (should already sum to 1)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase 1 – L2 variance flattening
# ----------------------------------------------------------------------
def _phase_l2(a: np.ndarray,
              n: int,
              deadline: float,
              best_a: np.ndarray,
              best_val: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on Σ(b‑μ)² (μ = average of b)."""
    avg_target = 1.0 / (2 * n - 1)               # µ for unit sum
    base_step = 0.5
    min_step = 0.001
    no_improve = 0
    IMPROVE_LIMIT = 120_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[L2   ] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # residual to flat target
        r = b - avg_target
        a_rev = a[::-1]
        conv1 = _conv_fft(r, a)        # length 3n‑2
        conv2 = _conv_fft(r, a_rev)
        start, end = n - 1, 2 * n - 1
        grad = 2.0 * (conv1[start:end] + conv2[start:end])

        # linear step schedule
        total = deadline - (deadline - (deadline - time.time()))   # dummy to keep type
        elapsed = time.time() - (deadline - (deadline - time.time()))  # not used
        # more conveniently: compute progress based on remaining time
        # (simple linear decay from start of phase)
        # We'll just use a fixed schedule here:
        step = base_step * (1.0 - (time.time() - (deadline - (deadline - time.time()))) /
                           (deadline - (deadline - time.time()))) + min_step * (
                (time.time() - (deadline - (deadline - time.time()))) /
                (deadline - (deadline - time.time())))

        # gradient step + simplex projection
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional jitter
        if np.random.rand() < 0.004:
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)

        # restart if stuck
        if no_improve > IMPROVE_LIMIT:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print("[L2   ] restart after stagnation")
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Smoothed infinity‑norm descent
# ----------------------------------------------------------------------
def _phase_smooth_max(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """Minimise (log‑sum‑exp)/β, i.e. a smooth approximation of max(b)."""
    beta = 5.0               # starting temperature
    beta_max = 60.0
    beta_mul = 1.04          # multiplicative increase
    step_base = 0.4
    step_min = 0.001

    no_improve = 0
    IMPROVE_LIMIT = 80_000

    iteration = 0
    while time.time() < deadline:
        # full convolution and current objective
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Smooth] new best = {best_val:.6f} (β={beta:.2f})")
        else:
            no_improve += 1

        # smooth weight vector w_i ∝ exp(β·b_i)
        max_b = b.max()
        # shift to avoid overflow
        shifted = beta * (b - max_b)
        exps = np.exp(shifted)
        w = exps / exps.sum()          # length 2n‑1, sums to 1

        # gradient = 2 * conv(w, a_rev) on the central n entries
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1:2 * n - 1]

        # step size – linear decay over this phase
        phase_elapsed = time.time() - (deadline - (deadline - time.time()))
        phase_total = deadline - (deadline - time.time())
        progress = phase_elapsed / (phase_total if phase_total > 0 else 1.0)
        step = step_base * (1.0 - progress) + step_min * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional increasing of β
        if iteration % 100 == 0 and beta < beta_max:
            beta = min(beta * beta_mul, beta_max)

        # small random jitter to escape plateaus
        if np.random.rand() < 0.003:
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)

        # restart if no improvement for long
        if no_improve > IMPROVE_LIMIT:
            a = np.random.rand(n)
            a = _project_simplex(a)
            beta = 5.0
            no_improve = 0
            print("[Smooth] restart after stagnation")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Targeted hill‑climbing on the exact max‑objective
# ----------------------------------------------------------------------
def _phase_hillclimb(a: np.ndarray,
                     n: int,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> tuple[np.ndarray, float]:
    """Move small mass from indices heavily contributing to the peak."""
    cur_val = 2.0 * n * _full_convolution(a).max()
    b = _full_convolution(a)          # keep the current convolution
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while time.time() < deadline:
        # locate current peak of the convolution
        t_star = int(np.argmax(b))

        # contribution of each coordinate to this peak:
        # grad_i = 2 * a[t_star - i]  (if index valid)
        grad_contrib = np.zeros(n, dtype=np.float64)
        lo = max(0, t_star - (n - 1))
        hi = min(n - 1, t_star)
        # indices lo … hi inclusive
        idx = np.arange(lo, hi + 1, dtype=int)
        grad_contrib[idx] = 2.0 * a[t_star - idx]

        # source: pick index where a_i * grad_contrib_i is largest
        src_score = a * grad_contrib
        i_src = int(np.argmax(src_score))

        # target: pick index with the smallest contribution (preferring low a)
        target_score = grad_contrib + 0.2 * a   # break ties towards small a
        i_tgt = int(np.argmin(target_score))
        if i_tgt == i_src:
            # if tie, choose a random different index
            i_tgt = (i_src + 1) % n

        if a[i_src] <= 1e-14:
            # nothing to move, pick a random source
            i_src = np.random.choice(np.where(a > 1e-14)[0])
            if a[i_src] <= 1e-14:
                continue

        # amount to move – a small fraction of the source mass
        delta = a[i_src] * np.random.rand() * 0.15
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        # adding the contribution of the incoming mass
        b_new[i_tgt:i_tgt + n] += 2.0 * delta * a
        # removing the contribution of the outgoing mass
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * i_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + i_tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val:
            # accept move
            a[i_src] -= delta
            a[i_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill  ] new best = {best_val:.6f} (iter {iteration})")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb to escape flat region
            a = best_a.copy()
            a += np.random.normal(scale=0.008, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill  ] stagnation – perturb (iter {iteration})")

        iteration += 1
    return best_a, best_val
