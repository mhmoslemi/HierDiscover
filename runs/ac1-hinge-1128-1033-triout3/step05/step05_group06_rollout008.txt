    start = time.time()
    n = len(a)
    while time.time() < time_limit:
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # pick delta as fraction of smaller of a[i], a[j]
        delta = a[i] * np.random.rand() * 0.1  # up to 10% of a[i]
        if a[i] - delta < 0:
            delta = a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # recompute score

        if sc < best_score:
            best_score = sc
            a = a_new
        # else maybe accept with prob depending on temperature
    return a, best_score

   start = time.time()
   n = len(initial_a)
   size = 1 << ((3*n - 2).bit_length())
   a = initial_a.copy()
   best_score = evaluate_obj(a)
   best_vec = a.copy()
   iter = 0
   last_improve = 0
   eta0 = 1.0
   beta0 = 5.0
   beta_max = 500.0
   while time.time() - start < time_limit - 0.2:
       iter += 1
       # compute FFT of a
       a_fft = np.fft.rfft(a, n=size)
       conv = np.fft.irfft(a_fft * a_fft, n=size)[:2*n-1]
       max_b = conv.max()
       cur_score = 2 * n * max_b
       if cur_score < best_score - 1e-12:
           best_score = cur_score
           best_vec = a.copy()
           last_improve = iter
           print(f"Iter {iter:6d}: new best {best_score:.6f}")
       # schedule beta
       beta = beta_max * (1 - np.exp(-iter/20000.0)) + beta0 * np.exp(-iter/20000.0)
       # compute softmax weights
       log_w = beta * conv
       log_w -= np.max(log_w)
       w = np.exp(log_w)
       w_sum = w.sum()
       if w_sum == 0:
           w = np.ones_like(w) / w.shape[0]
       else:
           w = w / w_sum
       # correlation to get gradient
       w_fft = np.fft.rfft(w, n=size)
       corr = np.fft.irfft(w_fft * np.conj(a_fft), n=size)
       grad = 2.0 * corr[n-1:n-1+n]
       # learning rate schedule
       eta = eta0 / np.sqrt(iter)
       # exponentiated gradient update
       a = a * np.exp(-eta * grad)
       # avoid all zeros

       a = a / a_sum

"""
Exponential‑gradient / annealed‑softmax optimizer for the max‑auto‑convolution
problem.

We minimise  f(a) = 2·n·max(a∗a)   (sum(a)=1, a_i≥0) .
The optimisation works on the probability simplex using an exponentiated‑gradient
update.  The “max’’ is replaced by a smooth soft‑max with a temperature β that is
slowly increased, providing a well‑behaved gradient that gradually focuses on
the true peak.  After the main EG phase a short hill‑climbing (mass transfer)
phase refines the solution.

The routine tolerates an optional external seed (height_sequence_1) and falls
back to a small portfolio of handcrafted distributions and random Dirichlet
vectors.  It partitions the available time budget among the seeds, keeping the
best vector found overall.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (real FFT, length 2·n‑1).
    a is assumed to be a 1‑D real array with sum(a)=1.
    """
    n = a.shape[0]
    size = next_pow2(2 * n)                # enough for a*a (zero‑padding)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def eval_obj(a: np.ndarray) -> float:
    """
    Objective value 2·n·max(a*a)  (the scaling performed by evaluate_sequence
    when the sum of a equals 1).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (used for deterministic seeds)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (soft‑max of the convolution)
# ----------------------------------------------------------------------
def run_exponential_gradient(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Optimises the probability vector `initial_a` within `time_limit` seconds.
    Returns the best vector seen and its objective value.
    """
    start = time.time()
    n = initial_a.shape[0]

    # FFT size for both the self‑convolution (length 2n‑1) and the correlation
    # needed for the gradient (length 3n‑2).  A power of two simplifies FFT calls.
    size = next_pow2(3 * n)

    a = np.clip(initial_a, 1e-20, None)   # avoid exact zeros
    a /= a.sum()                          # ensure sum = 1

    best_score = eval_obj(a)
    best_vec = a.copy()
    last_improve = 0
    iteration = 0

    # learning‑rate & temperature schedules
    eta0 = 1.0               # base step size in log‑space
    beta_start = 5.0
    beta_end = 500.0
    beta_ramp_iter = 20000   # iterations over which β grows

    while time.time() - start < time_limit - 0.2:
        iteration += 1

        # ----- 1) self‑convolution (a*a) -----
        a_fft = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(a_fft * a_fft, n=size)[: 2 * n - 1]
        max_b = conv.max()

        # ----- 2) (soft)max weights -----
        # β is increased slowly – this gives a smooth gradient early on.
        beta = beta_start + (beta_end - beta_start) * min(1.0, iteration / beta_ramp_iter)
        log_w = beta * conv
        log_w -= np.max(log_w)                     # for numerical stability
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- 3) gradient of the smoothed max w.r.t. a -----
        # ∂/∂a_i  Σ_k w_k (a*a)[k] = 2 Σ_j w_{i+j} a_j
        w_fft = np.fft.rfft(w, n=size)
        corr = np.fft.irfft(w_fft * np.conj(a_fft), n=size)
        # correlation slice giving Σ_j w_{i+j} a_j for i = 0..n‑1
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # ----- 4) step size (decay with √iteration) -----
        eta = eta0 / np.sqrt(iteration)

        # ----- 5) exponentiated‑gradient update (ensures a_i ≥ 0) -----
        a = a * np.exp(-eta * grad)

        # re‑normalise to stay on the simplex
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ----- 6) bookkeeping -----
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best = {best_score:.6f}")

        # ----- 7) re‑heat if we have not improved for a while -----
        if iteration - last_improve > 5000:
            # tiny random jitter + reset of the moments (here we have none)
            a = a * np.exp(1e-3 * (np.random.rand(n) - 0.5))
            a /= a.sum()
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

    return best_vec, best_score


# ----------------------------------------------------------------------
# Small hill‑climbing refinement (mass transfer between two coordinates)
# ----------------------------------------------------------------------
def hill_climb(a_start: np.ndarray, start_score: float, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search that moves a tiny amount of probability mass
    from one index to another, accepting only improving moves.
    """
    start = time.time()
    a = a_start.copy()
    best_score = start_score
    best_vec = a.copy()
    n = a.shape[0]
    iter_no = 0

    # maximum fraction of the source mass that may be moved in one trial
    max_frac = 0.05

    while time.time() - start < time_limit - 0.05:
        iter_no += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        src = a[i]
        if src == 0.0:
            continue
        delta = src * max_frac * np.random.rand()
        # apply the move
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # evaluate
        sc = eval_obj(a_candidate)
        if sc < best_score:
            best_score = sc
            best_vec = a_candidate.copy()
            a = a_candidate
            if iter_no % 5000 == 0:
                print(f"Hill iter {iter_no:6d}: improved to {best_score:.6f}")

    return best_vec, best_score
