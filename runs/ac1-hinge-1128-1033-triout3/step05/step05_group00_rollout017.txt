"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a two‑stage strategy:

1.  A cheap parametric sweep over simple symmetric families
    (U‑shaped, flat, and mixtures with the uniform distribution) to
    obtain a decent starting vector.
2.  A precise Adam‑style sub‑gradient descent on the exact
    max‑convolution entry (single‑active‑set sub‑gradient).  The
    iterate is projected onto the probability simplex after every step.
    Random perturbations are injected when progress stalls.

The routine respects the 1000 s budget, prints intermediate scores,
and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for our lengths).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Scoring helper – the quantity we try to minimise.
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a, a)) (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    return 2.0 * n * max_conv

    # p near 0 gives the uniform distribution; p<0 creates a mild U‑shape.
    # We also try mixtures with the flat uniform distribution.
    prod = (np.arange(n, dtype=np.float64) + 0.5) * (n - np.arange(n, dtype=np.float64) - 0.5)
    eps = 1e-12                         # avoid zero when p<0

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    p_values = np.linspace(-0.35, 0.35, 71)   # fine grid around 0
    mix_weights = [0.0, 0.2, 0.5, 0.8, 1.0]   # weight of the shaped component

    for p in p_values:
        # base shape (symmetric, non‑negative)
        base = np.power(prod + eps, p)
        s = base.sum()
        if s == 0.0:
            continue
        base /= s

        for w in mix_weights:
            if w == 0.0:
                cand = base
            elif w == 1.0:
                cand = uniform
            else:
                cand = w * base + (1.0 - w) * uniform
                cand /= cand.sum()

            val = _score_of(cand)
            if val < best_val - 1e-12:
                best_val = val
                best_seq = cand.copy()
                print(f"[sweep] p={p: .3f} mix={w:.2f}  score={val:.12f}")

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2. ADAM‑STYLE SUB‑GRADIENT REFINEMENT -------------------------------
    # --------------------------------------------------------------
    a = best_seq.copy()
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.15                     # base step size

    m = np.zeros_like(a)           # first moment estimate
    v = np.zeros_like(a)           # second moment estimate
    beta1_pow = 1.0
    beta2_pow = 1.0
    t_iter = 0
    stagn = 0

    while time.time() < deadline:
        t_iter += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # current score
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # keep the best overall
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[adam iter {t_iter:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # --- exact sub‑gradient (single active entry) -----------------
        argmax_idx = int(np.argmax(conv))
        w = np.zeros_like(conv)
        w[argmax_idx] = 1.0

        rev_a = a[::-1]
        # gradient of max(conv) w.r.t. a: 2 * rev_a * w   (restricted to length n)
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        a = a_candidate

        # occasional random shake‑up if we have not improved for a while
        if stagn > 3000:
            noise = rng.normal(scale=0.003, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            print(f"[adam perturb] iter {t_iter}")

        # keep numerical sanity
        if t_iter % 5000 == 0:
            a = a / a.sum()

    # --------------------------------------------------------------
    # Return the best sequence seen (clamped to the limits required by
    # `evaluate_sequence`).  The final score is printed for reference.
    # --------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")

    return final_seq
