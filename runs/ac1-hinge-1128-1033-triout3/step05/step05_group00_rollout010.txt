"""
A two‑stage optimizer for the step‑function evaluation problem.

Stage 0 – cheap shape search:
    A one‑parameter family a_i ∝ ((i+1)*(n-i))**p  (p∈[-1,0]) is scanned.
    The best‑scoring shape is taken as the start point.

Stage 1 – soft‑max gradient descent with momentum.
    The objective 2·n·max(conv(a,a)) is smoothed with a
    temperature ε and the gradient is taken w.r.t. the soft‑max
    weights.  After each step the vector is projected onto the
    probability simplex (Duchi‑et al. 2008).

Stage 2 – exact sub‑gradient descent with Adam‑style adaptation.
    The true max‑entry is used (averaged over all active entries)
    and an Adam update is performed on the projected vector.

Both phases keep the best sequence seen so far.
Random perturbations are injected when progress stalls.
The whole routine respects a 1000 s time budget and returns the
best sequence (clipped to [0, 1000]) that it found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Full linear convolution via real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    # next power of two for speed
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

    beta1_pow, beta2_pow = 1.0, 1.0
    t_adam = 0
    stagn = 0

    while time.time() < deadline:
        t_adam += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phase2 iter {t_adam:5d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # active set of indices where the max is attained (tiny tolerance)
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = _project_simplex(a - step)
        a = a_candidate

        # perturb if progress stalls
        if stagn > 4000:
            noise = rng.normal(scale=0.003, size=n)
            a = _project_simplex(a + noise)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            stagn = 0
            print(f"[phase2 perturb] iter {t_adam}")

        if t_adam % 5000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Optional tiny smoothing pass (may improve flatness a bit).
    # ------------------------------------------------------------------
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    def _smooth(x):
        return np.convolve(x, kernel, mode='same')
    cur_score = best_val
    for _ in range(3):
        a_smooth = _smooth(best_seq)
        a_smooth = np.maximum(0.0, a_smooth)
        a_smooth /= a_smooth.sum()
        tmp_score = 2.0 * n * _full_convolution_fft(a_smooth, a_smooth).max()
        if tmp_score < cur_score - 1e-12:
            best_seq = a_smooth
            cur_score = tmp_score
            print(f"[smooth] improved to {cur_score:.12f}")
        else:
            break

    # ------------------------------------------------------------------
    # Convert to the list format required by ``evaluate_sequence``.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Log final score (if the evaluator is available)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] final_score could not be computed (evaluate_sequence missing)")

    return final_seq
