    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[final] best_score={best_val:.12f}, final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq

"""
propose_candidate

Optimises a non‑negative step‑function (a list of heights) to minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the full linear self‑convolution.  The algorithm is a
multi‑restart Adam optimiser that employs a temperature‑controlled soft‑max
approximation of the maximum, a hard‑max polishing phase and a final
sub‑gradient refinement.  It works on up to ~20 000 coefficients, respects
the provided time budget, and returns the best sequence found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast full convolution via real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using a real FFT (zero‑padded)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# One optimisation run (Adam + soft‑max + polishing).
# ----------------------------------------------------------------------
def _run_optim(seed: int, time_budget: float) -> tuple[np.ndarray, float]:

    Perform a single optimisation run bounded by ``time_budget`` seconds.
    Returns (best_sequence, best_score) where best_score = 2*n*max(conv).
    """
    rng = np.random.default_rng(seed)
    # --------------------------------------------------------------
    # Choose a length. 20 000 is a good compromise between speed and bound.
    # --------------------------------------------------------------
    target_n = 20000

    # Warm‑start from the global variable if it exists.

        if base.shape[0] == target_n:
            a = np.maximum(0.0, base)

            new_x = np.linspace(0.0, 1.0, target_n)
            a = np.interp(new_x, old_x, base)
            a = np.maximum(0.0, a)

        a = None

    if a is None:
        n = target_n
        x = np.arange(n, dtype=np.float64)

        # (approximately) Catalan decreasing sequence
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)
            catalan[k] = cur

        sine = np.sin(np.pi * (x + 0.5) / n)
        t = 2.0 * x / (n - 1) - 1.0

    n = a.shape[0]

    conv = _full_convolution_fft(a, a)

    lr_min = 1e-6
    lr_decay = 0.99996
    beta1 = 0.9

    deadline = start + time_budget - 2.0          # safety margin
    phase1_end = start + 0.70 * time_budget      # ~70 % for smooth optimisation

    # ------------------------------------------------------------------
    # Phase 1 – smooth optimisation with Adam + soft‑max.
    # ------------------------------------------------------------------
    while time.time() < phase1_end:
        it += 1

        # current convolution and score
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # update best
        if score < best_score - 1e-15:
            best_score = score
            best_seq = a.copy()
            stagn = 0

    # ------------------------------------------------------------------

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative height sequence that (approximately) minimises
    ``2 * n * max(conv(a, a)) / (sum(a) ** 2)``.
    The function respects the supplied time budget (seconds) and returns
    the best sequence found (as a list of Python floats).
    """
    start_time = time.time()
    overall_deadline = start_time + budget_s - 5.0   # safety margin
    best_global_seq = None
    best_global_score = float("inf")
    run_idx = 0

    # We allow several independent runs (different seeds) as long as time permits.
    while time.time() < overall_deadline - 2.0:
        remaining = overall_deadline - time.time()
        # allocate at most 250 s per run (or whatever is left)
        run_budget = min(250.0, remaining)
        if run_budget < 10.0:   # not enough time for a meaningful run
            break
        seq, val = _run_optim(seed + run_idx, run_budget)
        if val < best_global_score:
            best_global_score = val
            best_global_seq = seq.copy()
            print(f"[global] new overall best (run {run_idx}) = {best_global_score:.12f}")
        run_idx += 1

    # If for any reason we never managed a run, fall back to a simple uniform vector.
    if best_global_seq is None:
        n = 5000
        best_global_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Convert to the format required by `evaluate_sequence` and report.
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_global_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
