def _mass_transfer_climb(a, best_val, deadline, rng):
    n = len(a)
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.2
    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # identify all pairs (i,j) that produce max_b (maybe many)
        ks = np.where(conv == max_b)[0]
        k = ks[0]  # pick one
        # compute contributions
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        # choose donor indices that have largest contrib
        sorted_idx = np.argsort(-contrib) # descending
        donors = sorted_idx[:2]  # top 2
        i1, i2 = i_range[donors[0]], i_range[donors[1]]

        low = np.where(cur_a < np.percentile(cur_a, 20))[0]

    n = a.shape[0]
    cur_a = a.copy()
    cur_val = cur_best_val
    step_frac = 0.2
    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()

        k_candidates = np.flatnonzero(conv == max_b)
        k = int(rng.choice(k_candidates))
        # compute contributions
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range

"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

The algorithm consists of three phases:

1. **Smooth optimisation** –  The non‑smooth objective
   ``max(conv)`` is replaced by a log‑sum‑exp surrogate.
   The surrogate is minimised with L‑BFGS‑B while the smoothing
   parameter ε is decreased (continuation).  The parameters are kept
   in a soft‑max representation, guaranteeing non‑negativity and a
   fixed ℓ₁‑norm.

2. **Random restarts** –  Several perturbed restarts of the
   continuation are performed; this helps to escape local minima.

3. **Enhanced hill‑climbing** –  After the smooth phase we repeatedly
   locate the lag ``k`` where the convolution attains its maximum,
   identify the pairs that contribute most to that lag and move a
   fraction of their mass to low‑mass positions.  The step size is
   adapted and the move is accepted only if the exact ``evaluate_sequence``
   value (computed via the fast surrogate) improves.

The routine respects the given time budget (default 1000 s) and prints
progress information.  The final list of heights is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation of a real vector (length 2·len‑1).
    Uses FFT for speed and clips tiny negative artefacts.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv-max)/ε)

    together with its gradient w.r.t. the soft‑max parameters ``w``.
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    # full auto‑convolution   (length 2·n‑1)
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()          # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)       # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]      # middle n entries
    grad_a = 4.0 * n * seg[::-1]        # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Hill‑climbing mass‑transfer refinement
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray, best_val: float,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce the exact objective by repeatedly moving a small amount of mass
    away from the lag where the convolution attains its maximum.
    Multiple high‑contributing pairs are processed at once.
    The algorithm stops when the time budget expires.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.2                     # initial fraction of mass to move

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # pick one of the (possibly several) maximal lags
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        # select the top‑few contributing pairs
        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]

        donors = []                      # list of (i, j) tuples (j may == i)
        for idx in top_idx:
            i = int(i_range[idx])
            j = int(j_range[idx])
            donors.append((i, j))

        # destination positions – pick from low‑mass tail
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        # ensure destinations are distinct from donors
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                # single index contributes a_i^2
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue

        # renormalise to keep the ℓ₁‑norm equal to 1 (scale‑invariant)
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue
        new_a /= s

        # exact objective (sum_a = 1 ⇒ obj = 2·n·max_b)
        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            # modestly increase step size after success
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            # shrink step size when no improvement
            step_frac = max(step_frac * 0.95, 1e-5)

    return cur_a, cur_val
