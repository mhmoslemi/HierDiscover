       else:

       v = beta2 * v + (1.0 - beta2) * (g * g)

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The key observation is that a simple exponential decay a_i ∝ exp(-λ i) (with the
vector normalised to sum = 1) already yields a very small objective.  The
algorithm therefore

1. determines the problem size `n` (re‑uses the previous solution if it exists);
2. evaluates a dense grid of exponential families a_i = exp(-λ i) with
   λ = c / n for many values of `c`;
3. keeps the best candidate seen;
4. runs a short Adam‑style refinement (exact sub‑gradient of the max‑convolution)
   starting from that candidate while staying on the probability simplex;
5. returns the best sequence (clamped to the range required by `evaluate_sequence`).

The whole routine respects the 1000 s budget and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for large vectors).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Compute the evaluation function (without the final clamp).
# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return 2*n*max(conv(a,a)) / (sum(a)**2).  Vector `a` need not be normalised."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (sum_a ** 2))


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative height sequence that (approximately) minimizes the
    evaluation function.  The routine runs for at most `budget_s` seconds,
    prints intermediate scores and stores the best sequence globally as
    ``height_sequence_1`` for potential reuse.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
        a0 = np.maximum(0.0, base)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
        a0 = a0 / a0.sum()
    else:
        n = 12000                     # a comfortable size for the time budget
        a0 = None

    # --------------------------------------------------------------
    # Build a list of candidate vectors (exponential families + a few
    # generic shapes for safety).  All candidates are normalised to sum = 1.
    # --------------------------------------------------------------
    candidates = []

    if a0 is not None:
        candidates.append(a0)

    x = np.arange(n, dtype=np.float64)

    # --- exponential families a_i = exp(-λ i) ---
    # we use λ = c / n with c ranging from 0.1 to 12 (dense enough)
    c_vals = np.linspace(0.1, 12.0, 120)          # 120 different families
    for c in c_vals:
        lam = c / n
        a = np.exp(-lam * x)
        a = a / a.sum()
        candidates.append(a)

    # --- power‑law families a_i ∝ (i+1)^{-p} ---
    p_vals = np.linspace(0.1, 5.0, 50)
    for p in p_vals:
        a = (x + 1.0) ** (-p)
        a = a / a.sum()
        candidates.append(a)

    # --- combined exponential × power law (few representative points) ---
    comb_params = [(0.0005, 0.5), (0.001, 1.0), (0.002, 1.5),
                   (0.003, 2.0), (0.005, 2.5)]
    for lam, p in comb_params:
        a = np.exp(-lam * x) * ((x + 1.0) ** (-p))
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        candidates.append(a)

    # --- a few classic smooth shapes (gaussian, semicircle, triangle) ---
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
    gauss = gauss / gauss.sum()
    candidates.append(gauss)

    t = 2.0 * x / (n - 1) - 1.0
    semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
    semicircle = semicircle / semicircle.sum()
    candidates.append(semicircle)

    triangle = np.minimum(x, n - 1 - x) + 1.0
    triangle = triangle / triangle.sum()
    candidates.append(triangle)

    # --------------------------------------------------------------
    # Evaluate all candidates and keep the best one.
    # --------------------------------------------------------------
    best_val = np.inf
    best_seq = None
    print(f"[init] starting search over {len(candidates)} candidates (n={n})")
    for idx, cand in enumerate(candidates):
        score = _score(cand)
        if score < best_val:
            best_val = score
            best_seq = cand.copy()
            print(f"[candidate {idx:03d}] new best score = {best_val:.12f}")

    # --------------------------------------------------------------
    # Short Adam‑style refinement (exact sub‑gradient of the max entry).
    # This is cheap because we only run a few thousand iterations.
    # --------------------------------------------------------------
    a = best_seq.copy()
    best_val_refined = best_val
    best_seq_refined = best_seq.copy()

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.2                          # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = 1.0, 1.0
    iter_adam = 0
    stagn = 0
    active_tol = 1e-14

    while time.time() < deadline:
        iter_adam += 1

        # current objective
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = float(2.0 * n * max_conv / (a.sum() ** 2))

        if score < best_val_refined - 1e-14:
            best_val_refined = score
            best_seq_refined = a.copy()
            stagn = 0
            print(f"[adam iter {iter_adam:6d}] new best = {best_val_refined:.12f}")
        else:
            stagn += 1

        # active set for the max entry
        active = np.flatnonzero(conv >= max_conv - active_tol)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        g = 2.0 * grad_full[n - 1:n - 1 + n]   # exact sub‑gradient

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a = _project_simplex(a_candidate)          # keep on the simplex

        # occasional random perturbation if we are stuck
        if stagn > 3000:
            noise = rng.normal(scale=0.002, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0
            print(f"[adam perturb] iter {iter_adam}")

        # occasional renormalisation (numerical safety)
        if iter_adam % 5000 == 0:
            a = a / a.sum()

    # --------------------------------------------------------------
    # Prepare the final answer in the format required by evaluate_sequence.
    # --------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq_refined.tolist()]

    # optional logging of the final score (if evaluate_sequence is available)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score = {final_score:.12f}, length = {len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence could not be called (not imported)")

    # expose the result for possible reuse
    globals()['height_sequence_1'] = final_seq
    return final_seq
