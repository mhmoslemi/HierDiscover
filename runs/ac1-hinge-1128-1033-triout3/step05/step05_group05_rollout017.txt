def _softmax_gradient(a, beta):
    b = _full_convolution(a)
    max_b = b.max()
    # compute w = exp(beta*(b-max_b)) to avoid overflow
    diff = b - max_b
    w = np.exp(beta * diff, dtype=np.float64)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # gradient
    conv = _conv_fft(w, a)  # length maybe 3n-2
    start = a.shape[0] - 1
    grad = 2.0 * conv[start:start + a.shape[0]]

# Helper: full self‑convolution via FFT (length 2·n‑1)

def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

# Helper: convolution of two (possibly different) vectors via FFT
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w

        # ---- L2‑gradient (flattening) ---------------------------------
        r = b - avg_target
        a_rev = a[::-1]
        conv1 = _conv_fft(r, a)               # length 3·n‑2
        conv2 = _conv_fft(r, a_rev)
        start_idx = n - 1
        end_idx = 2 * n - 1
        grad_l2 = 2.0 * (conv1[start_idx:end_idx] + conv2[start_idx:end_idx])

        # ---- soft‑max gradient (focus on max) -------------------------
        _, grad_soft = _softmax_gradient(a, beta)

        # mixing coefficient – start with mostly L2, finish with soft‑max
        mix = max(0.0, 1.0 - progress)        # 1 → 0
        grad = mix * grad_l2 + (1.0 - mix) * grad_soft

        # gradient step + projection onto simplex
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional jitter to escape shallow basins
        if np.random.rand() < 0.004:
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)

        # restart after long stagnation
        if no_improve > IMPROVE_STAGNATION:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[GD  {it}] restart after stagnation")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Deterministic hill‑climbing (same idea as the reference code)
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """
    Fast local search that transfers a tiny amount of mass between coordinates.
    The routine stops at `deadline` and returns the best sequence seen.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac = 0.35                          # max fraction of source mass to move
    IMPROVE_LIMIT = 120_000                  # stagnation trigger
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # source – usually the current largest entry (highly active)
        if np.random.rand() < 0.85:
            i = int(np.argmax(a))
        else:
            # rare random source weighted by current mass
            i = np.random.choice(n, p=a)

        if a[i] <= 1e-14:
            continue

        # pick a different target
        j = np.random.randint(n)
        if j == i:
            continue

        # amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental O(n) update of the full convolution
        b_new = b.copy()
        # cross‑terms (2·δ·a) – shifting a by the target index
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept strictly improving moves
        if new_val < cur_val:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[HC {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # after long stagnation, re‑inject a gentle random perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HC {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a


# ----------------------------------------------------------------------
# Search for a specific problem size
# ----------------------------------------------------------------------
def _search_n(n: int, time_budget: float):
    """
    Conducts a full search for length `n` within `time_budget` seconds.
    Returns the best sequence found and its evaluation score.
    """
    # allocate 70 % of the budget to gradient descent, 30 % to hill‑climbing
    grad_budget = time_budget * 0.70
    hill_budget = time_budget - grad_budget

    # ---- generate a few diverse initial guesses -----------------------
    inits = []

    # (a) arcsine‑like shape
    x = (np.arange(n) + 0.5) / n
    a_arc = 1.0 / np.sqrt(x * (1.0 - x))
    a_arc = np.maximum(a_arc, 0.0)
    a_arc /= a_arc.sum()
    inits.append(a_arc)

    # (b) symmetric power‑law (1/√i + 1/√(n‑i))
    i = np.arange(n)
    a_pow = 1.0 / np.sqrt(i + 1) + 1.0 / np.sqrt(n - i)
    a_pow = np.maximum(a_pow, 0.0)
    a_pow /= a_pow.sum()
    inits.append(a_pow)

    # (c) random Dirichlet vector
    a_rand = np.random.rand(n)
    a_rand /= a_rand.sum()
    inits.append(a_rand)

    # (d) previous construction if available
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a_interp = np.interp(new_x, old_x, old)
        a_interp = np.maximum(a_interp, 0.0)
        a_interp /= a_interp.sum()
        inits.append(a_interp)
    except Exception:
        pass

    best_seq = None
    best_score = np.inf
    start = time.time()
    # equally split the gradient budget across the initial guesses
    per_init = grad_budget / len(inits)

    for idx, a0 in enumerate(inits):
        # ensure we do not exceed the overall budget
        remaining = start + grad_budget - time.time()
        if remaining <= 0:
            break
        # give the current init whatever time is left for it (but not more than per_init)
        alloc = min(per_init, remaining)
        seq, score = _optimize_sequence(a0, alloc)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Init {idx}] new global best = {best_score:.6f}")

    # ---- Phase 2 – hill‑climbing on the best candidate ----------------
    if best_seq is not None:
        hc_start = time.time()
        deadline = hc_start + hill_budget
        best_seq = _hillclimb_phase(best_seq, n,
                                    hc_start, deadline,
                                    best_seq, best_score)
        # final score after hill‑climbing (re‑evaluate)
        try:
            best_score = evaluate_sequence(best_seq.tolist())
        except Exception:
            pass
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a step‑function under the given time budget.
    Returns a list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # small safety margin

    # We try two different lengths; the larger one usually gives a slightly
    # better bound while still fitting comfortably into the time budget.
    sizes = [4000, 8000]                         # order matters – start small
    best_seq = None
    best_val = float('inf')

    for n in sizes:
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # give each size a fair share of the remaining time
        alloc = remaining / (len(sizes) - sizes.index(n))
        seq, val = _search_n(n, alloc)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq
            print(f"[Overall] new best after size {n}: {best_val:.6f}")

    # final sanity check (the wrapper will also be called by the grader)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_seq.tolist()
