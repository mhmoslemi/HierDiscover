"""
Simulated‑annealing + greedy mass‑transfer optimisation.

We work on the probability simplex (∑a_i = 1, a_i ≥ 0) because the
evaluation function 2·n·max(a*a) is invariant under scaling.
The algorithm starts from the best known construction (``height_sequence_1``)
or from a random uniform start.  At each step we locate the convolution
peak k* and pick a source index i that contributes most to that peak.
A random destination j receives a fraction δ of the mass of i.
The move is accepted if it lowers the objective or – with a probability
that follows an exponential annealing schedule – if it raises the
objective (standard simulated‑annealing acceptance).  This allows the
search to escape local minima that trap pure hill‑climbing.

The convolution is updated incrementally in O(n) using the exact formula

    (a+δ(e_j‑e_i))*(a+δ(e_j‑e_i))
      = a*a + 2δ( a shifted by j – a shifted by i )
        + δ²(δ_{2j} – 2δ_{i+j} + δ_{2i}),

so we avoid an O(n log n) FFT at every iteration.  Every few thousand
iterations the full convolution is recomputed from scratch to wipe out
numeric drift.  The best sequence discovered before the time budget
expires is returned.

The method is deliberately stochastic and includes
* a cooling schedule (temperature → 0),
* occasional “jitter’’ moves that transfer a larger mass,
* a simple stagnation detector that re‑normalises the vector if no
  improvement occurs for a long time.

The whole search runs for the allotted ``budget_s`` seconds (default
1000 s) and prints periodic progress reports.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: exact linear convolution via FFT (used for occasional re‑sync)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y (full length) using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negatives from round‑off
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
#          (see Duchi et al., ICML 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, ∑x = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safeguard – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Simulated‑annealing search for a step‑function that minimises
    ``evaluate_sequence``.  Returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin for the framework
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise a – use the provided best construction or a random start
    # --------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the evaluation environment
        a = np.asarray(height_sequence_1, dtype=float).copy()
        n = a.shape[0]
    except NameError:
        # modest length – large enough to give a low theoretical bound
        n = 1800
        a = np.random.rand(n)

    # enforce non‑negativity and normalise to a probability vector
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # --------------------------------------------------------------
    # Exact convolution and objective bookkeeping
    # --------------------------------------------------------------
    conv = _full_convolution_fft(a, a)          # length 2n‑1
    cur_obj = float(2.0 * n * conv.max())
    best_a = a.copy()
    best_obj = cur_obj

    print(f"[init] n={n}, initial objective={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for the 1000 s budget)
    # --------------------------------------------------------------
    # annealing schedule
    T_start = cur_obj * 0.25          # initial temperature
    T_end   = 1e-6                    # final temperature

    # move size (fraction of source mass) – large at the beginning, shrinks
    move_frac_start = 0.30
    move_frac_end   = 0.02

    # how often to recompute the convolution from scratch
    recompute_every = 8000

    # jitter moves: occasionally transfer a larger chunk without the
    # acceptance test (they are only kept if they improve)
    jitter_prob   = 0.009
    jitter_frac   = 0.45

    last_improve_iter = 0
    iteration = 0

    # --------------------------------------------------------------
    # Main annealing loop
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        elapsed = now - start_time
        progress = elapsed / (deadline - start_time)   # in [0,1]

        # ----- temperature and move‑size schedules -----
        T = T_start * (1.0 - progress) + T_end * progress
        move_frac = move_frac_start * (1.0 - progress) + move_frac_end * progress

        # ----- locate the current peak of the convolution -----
        k_star = int(conv.argmax())

        # ----- source index: the coefficient that contributes most to the peak -----
        i_start = max(0, k_star - (n - 1))
        i_end   = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        partner = k_star - idx
        contrib = a[idx] * a[partner]
        i_src = int(idx[contrib.argmax()])

        # ----- random destination (different from source) -----
        j_dst = np.random.randint(n - 1)
        if j_dst >= i_src:
            j_dst += 1

        # ----- propose a mass‑transfer move -----
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue

        # keep copies for a possible revert (cheap, size ≈ 2n)
        a_old   = a.copy()
        conv_old = conv.copy()

        # ----- incremental convolution update (exact) -----
        d2 = delta * delta

        # linear part: +2δ(a shifted by j) –2δ(a shifted by i)
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[j_dst:j_dst + n] += 2.0 * delta * a_old

        # quadratic correction
        conv[2 * i_src]     += d2
        conv[2 * j_dst]     += d2
        conv[i_src + j_dst] -= 2.0 * d2

        # ----- update the coefficients -----
        a[i_src] -= delta
        a[j_dst] += delta

        # ----- evaluate new objective -----
        new_obj = float(2.0 * n * conv.max())

        # ----- Metropolis acceptance -----
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            # uphill move – accept with probability exp(-Δ/T)
            if np.random.rand() < np.exp(-(new_obj - cur_obj) / (T + 1e-12)):
                accept = True

        if accept:
            cur_obj = new_obj
            # keep the globally best sequence found so far
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration:7d}] new best = {best_obj:.6f} (T={T:.3e})")
                last_improve_iter = iteration
        else:
            # revert both a and the convolution
            a = a_old
            conv = conv_old

        # ----- occasional jitter move (large transfer, only kept if improving) -----
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            delta_j = jitter_frac * a[src] * np.random.rand()
            if delta_j > 0.0:
                a_tmp   = a.copy()
                conv_tmp = conv.copy()
                d2j = delta_j * delta_j

                conv_tmp[src:src + n] -= 2.0 * delta_j * a_tmp
                conv_tmp[dst:dst + n] += 2.0 * delta_j * a_tmp
                conv_tmp[2 * src]     += d2j
                conv_tmp[2 * dst]     += d2j
                conv_tmp[src + dst]   -= 2.0 * d2j

                a_tmp[src] -= delta_j
                a_tmp[dst] += delta_j

                jitter_obj = float(2.0 * n * conv_tmp.max())
                if jitter_obj < cur_obj:
                    # keep the jittered state
                    a = a_tmp
                    conv = conv_tmp
                    cur_obj = jitter_obj
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()
                        print(f"[{iteration:7d}] jitter improved best = {best_obj:.6f}")
                        last_improve_iter = iteration

        # ----- periodic full recomputation to avoid drift -----
        if iteration % recompute_every == 0:
            conv = _full_convolution_fft(a, a)

        # ----- stagnation detector -----
        if iteration - last_improve_iter > 250_000:
            # renormalise (helps with accumulated round‑off) and recompute
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_obj = float(2.0 * n * conv.max())
            last_improve_iter = iteration

    # ----------------------------------------------------------------------
    # Final reporting
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
