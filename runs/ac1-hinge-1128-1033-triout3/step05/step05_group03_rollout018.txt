"""
Propose a step‑function that minimizes `evaluate_sequence`.

The algorithm works in three stages:

1. **Random initialisation** – a Dirichlet draw (concentration 0.5) gives a
   diverse starting point, but we also fall back to a smooth Gaussian.
2. **Variance minimisation (Adam)** – we minimise the variance of the
   self‑convolution (which is smooth) on the probability simplex.
   The loss is  Var(a*a) = mean((a*a‑μ)²).  This spreads the mass more
   uniformly than the previous log‑sum‑exp max‑approximation and typically
   reaches a lower peak in far fewer iterations.
3. **Targeted simulated‑annealing hill‑climbing** – starting from the best
   point of stage 2 we repeatedly move a small amount of mass from an
   index that contributes most to the current maximal convolution entry
   to a random target.  Moves are accepted if they improve the exact
   objective or, with a temperature that slowly cools, with a small
   probability (allows escaping shallow local minima).  Periodic
   perturbations reset stagnation.

The whole procedure respects the 1000 s time‑budget, projects onto the
simplex after every update and returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (standard O(n) algorithm)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projects a vector onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, all‑real)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

# ----------------------------------------------------------------------
# Exact objective (sum(a) is assumed = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` when Σa = 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

# ----------------------------------------------------------------------
# Phase A – minimise variance of the convolution with Adam
# ----------------------------------------------------------------------
def _adam_variance(a0: np.ndarray,
                   n: int,
                   start: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *variance* of the self‑convolution.
    Returns the best (a, value) seen before *deadline*.
    """
    # initialise logits (soft‑max guarantees Σa = 1, a ≥ 0)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.08, 0.002
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 25_000

    while time.time() < deadline:
        # soft‑max -> current a
        w_shift = w - np.max(w)           # improve numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        # exact objective for bookkeeping
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[AdamVar] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # linear schedule for learning‑rate (fast decay)
        progress = (time.time() - start) / (deadline - start)
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- variance loss and its gradient w.r.t. a -----
        b = _full_convolution(a)                 # length 2·n‑1
        mu = b.mean()
        d = b - mu                                 # centred conv
        a_rev = a[::-1]
        # gradient of (1/L)∑(d_k)² w.r.t. a : (2/L) * conv(d, a_rev)_segment
        conv = _conv_fft(d, a_rev)                # length 3·n‑2
        seg = conv[len(b) - 1 : len(b) - 1 + n]   # n entries
        grad_a = (2.0 / len(b)) * seg

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional perturbation to avoid premature stagnation
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.005, size=n)

        # restart if nothing improves for a long while
        if no_improve > IMPROVE_LIMIT:
            print("[AdamVar] restart due to stagnation")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

    return best_a, best_val

# ----------------------------------------------------------------------
# Phase B – simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_anneal(a_start: np.ndarray,
                      n: int,
                      start: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> np.ndarray:
    """
    Local search that moves a small amount of mass between indices.
    Moves that increase the objective are occasionally accepted according
    to a cooling temperature schedule.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = _objective(a)

    # parameters
    max_frac_start, max_frac_end = 0.55, 0.02
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # temperature schedule – starts at ~5% of current best value,
        # cools exponentially to ~1e-5.
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        temperature = max(1e-5, 0.05 * (1.0 - progress) ** 3)

        # ---- locate all (near) maximal convolution entries ----
        max_b = b.max()
        k_candidates = np.where(b >= max_b - 1e-14)[0]

        # ---- compute contribution of each index to those maxima ----
        contrib = np.zeros_like(a)
        for k in k_candidates:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib[mask] += a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- pick a source index (biased by contribution) ----
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-16:
            continue

        # ---- pick a distinct target index uniformly ----
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # ---- amount of mass to move ----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) ----
        b_new = b.copy()
        # influence of +delta at position j
        b_new[j:j + n] += 2.0 * delta * a
        # influence of –delta at position i
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms (ordered pair i,i and j,j)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        # removal of the mixed i‑j term (appears twice in the slices above)
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ---- accept move ----
        improve = new_val < cur_val - 1e-14
        accept = improve or (np.random.rand() < np.exp(-(new_val - cur_val) / temperature))
        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0 if improve else no_improve + 1

            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[HillAnneal {iteration:7d}] new best = {best_val:.6f}")

            if best_val <= target_val:
                print("[HillAnneal] target reached")
                break
        else:
            no_improve += 1

        # ---- handle stagnation ----
        if no_improve > IMPROVE_LIMIT:
            # random perturbation around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective(a)
            no_improve = 0
            print(f"[HillAnneal {iteration:7d}] perturb after stagnation")

        iteration += 1

    return best_a

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a step‑function within the given time budget and returns
    a list of non‑negative heights (Σ≈1).  The procedure follows the
    three stages described in the module doc‑string.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a short safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more degrees of freedom.
    # ------------------------------------------------------------------
    n = 12000            # 12000 yields ~300 KB per vector, well within memory

    # ------------------------------------------------------------------
    # Initialise a – try to reuse an existing solution, otherwise start
    # from a random Dirichlet (α=0.5 gives a pleasantly “spiky” start).
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` is supplied by the external harness
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        # Dirichlet with concentration <1 concentrates mass while still
        # keeping many entries non‑zero – good for escaping the uniform trap.
        alpha = 0.5
        a0 = np.random.gamma(alpha, 1.0, size=n)
        a0 = np.maximum(a0, 0.0)

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()               # enforce Σa = 1

    best_a = a0.copy()
    best_val = float('inf')
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – minimise variance of the convolution with Adam
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_variance(a0, n,
                                      start=start_time,
                                      deadline=phase_a_deadline,
                                      best_a=best_a,
                                      best_val=best_val)

    # ------------------------------------------------------------------
    # Phase B – simulated‑annealing hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    best_a = _hillclimb_anneal(best_a, n,
                               start=phase_a_deadline,
                               deadline=deadline,
                               best_a=best_a,
                               best_val=best_val,
                               target_val=TARGET)

    # ------------------------------------------------------------------
    # Final evaluation (the wrapper does the clipping etc.)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        # Should never happen, but guard against unexpected errors
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
