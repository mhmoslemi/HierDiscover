def fast_evaluate(a):
    # a is numpy array floats
    # sum of a:
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    # compute convolution using FFT, result length 2*n-1
    n = a.shape[0]
    size = 1 << ((2*n-2).bit_length())  # next power of two >= 2n-1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_b = conv.max()
    # apply clipping? Not needed cause values nonnegative.
    return float(2.0 * n * max_b / (sum_a*sum_a))
```

def proj_simplex(v):
    # ensure components non-negative and sum to 1.
    # using algorithm from "Efficient Projections onto the L1-Ball for Learning in High Dimensions"
    # but sum=1, v length n.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative vector) with a low value of
``evaluate_sequence``.  

The algorithm is deliberately different from the previous L‑BFGS
approach:

*   Randomly initialise many candidates (either Dirichlet‑type or via a
    random‑phase inverse FFT – this gives “spectrally flat’’ shapes).
*   For each candidate run a cheap projected sub‑gradient descent that
    reduces the peak of the self‑convolution.  The sub‑gradient of
    ``max_k (a*a)_k`` is a weighted sum of shifted copies of ``a``; the
    update is projected onto the probability simplex (so the objective
    is scale‑invariant).
*   After the descent apply the same anti‑symmetric re‑ordering used
    before, followed by a fast hill‑climbing phase (random mass transfers
    and swaps).
*   The best vector found (checked with the official ``evaluate_sequence``
    at the end) is returned.

The implementation uses only FFT‑based convolutions (``O(n log n)``) and
a simple simplex projection, allowing thousands of iterations within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full (linear) self‑convolution a*a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # tiny negative values can appear because of round‑off – clip them
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def fast_objective(a: np.ndarray) -> float:
    """
    Compute the same value that ``evaluate_sequence`` returns, but
    using an FFT for speed.  The vector `a` must be non‑negative.
    """
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    conv = _full_convolution_fft(a)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a * sum_a))


def proj_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x >= 0, sum x = 1}.  Implements the algorithm of
    Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:               # all entries are zero
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce an ordering where the largest values are placed opposite the
    smallest ones when the sequence is read backwards.  This heuristic
    often lowers the peak of the self‑convolution.
    """
    n = a.shape[0]
    s = np.sort(a)          # ascending
    b = np.empty_like(a)

    half = n // 2
    for i in range(half):
        b[i] = s[n - 1 - i]          # large values on the left side
        b[n - 1 - i] = s[i]          # small values on the right side

    if n % 2:
        b[half] = s[half]
    return b


def hill_climb(a: np.ndarray,
               cur_val: float,
               deadline: float,
               n: int) -> tuple[np.ndarray, float]:
    """
    Simple stochastic refinement: random mass‑transfer or swap moves,
    always keeping the vector on the simplex.  Only strict improvements
    are accepted to keep the run fast.
    """
    cur_a = a.copy()
    cur_obj = cur_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # 20 % chance of a plain swap, otherwise a mass transfer
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            # move up to 5 % of donor's mass
            delta = cur_a[i] * 0.05 * rng.random()
            if delta <= 0.0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            # re‑project onto the simplex (necessary only for tiny round‑off)
            new_a = proj_onto_simplex(new_a)

        # cheap objective using the fast FFT based version
        new_val = fast_objective(new_a)
        if new_val + 1e-14 < cur_obj:       # strict improvement
            cur_a = new_a
            cur_obj = new_val
            print(f"[hill‑climb] new best = {cur_obj:.6f}")
    return cur_a, cur_obj

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a vector of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s time budget,
    uses many random restarts, a projected sub‑gradient descent on the
    peak of the self‑convolution, an anti‑symmetric re‑ordering, and a
    final hill‑climbing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Problem size – a larger n gives a lower theoretical bound but costs
    # a bit more per iteration.  8000 is a good compromise for the time‑budget.
    n = 8000

    # Initialise best solution with a uniform vector (objective = 2)
    best_a = np.full(n, 1.0 / n)
    best_val = fast_objective(best_a)
    print(f"[init] n={n}, initial fast objective = {best_val:.6f}")

    # ------- random‑initialisation helpers ---------------------------------
    def random_dirichlet():
        """Draw a Dirichlet‑like vector (exponential i.i.d.) and normalise."""
        v = np.random.exponential(scale=1.0, size=n)
        v /= v.sum()
        return v

    def random_fourier():
        """
        Build a vector from a random‑phase inverse FFT.
        The resulting real part is shifted to be non‑negative and normalised.
        """
        phases = np.random.uniform(0.0, 2.0 * np.pi, size=n)
        # Full complex spectrum (Hermitian symmetry not enforced – we take the real part)
        A = np.exp(1j * phases)
        v = np.fft.ifft(A).real
        v -= v.min()                     # make all entries >= 0
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        return v

    init_functions = [random_dirichlet, random_fourier]

    # ------- sub‑gradient descent on the max‑convolution --------------------
    def subgradient_descent(a: np.ndarray,
                            max_iters: int = 250) -> tuple[np.ndarray, float]:
        """
        Perform a projected sub‑gradient descent on
        f(a) = max_k (a*a)_k  (the denominator sum_a^2 is invariant under scaling).
        Returns the improved vector and its fast objective value.
        """
        for t in range(max_iters):
            # stop if time is about to run out
            if time.time() >= deadline - 5.0:
                break

            # compute full self‑convolution and the current peak
            conv = _full_convolution_fft(a)
            max_conv = conv.max()

            # indices where the peak is attained (within a tiny tolerance)
            tol = 1e-12
            active = np.where(conv >= max_conv - tol)[0]

            # gradient = 2 * sum_{k in active} shift(a, k)
            grad = np.zeros_like(a)
            for k in active:
                # The contribution for a given k is a reversed slice of a.
                if k < n:
                    # i = 0..k  -> a[k-i] = a[:k+1][::-1]
                    grad[:k + 1] += a[:k + 1][::-1]
                else:
                    # k >= n : i = k-(n-1) .. n-1
                    start = k - (n - 1)
                    length = 2 * n - 1 - k
                    grad[start:] += a[-length:][::-1]
            grad *= 2.0

            # simple diminishing step size
            eta = 0.7 / np.sqrt(t + 1.0)

            # gradient step + projection onto the simplex (keeps sum=1 and non‑negativity)
            a = proj_onto_simplex(a - eta * grad)

        return a, fast_objective(a)

    # ------- main multi‑restart loop ---------------------------------------
    restart = 0
    while time.time() < deadline - 12.0:          # leave a few seconds for final refine
        restart += 1
        # pick a random initialisation method
        init_fun = np.random.choice(init_functions)
        a = init_fun()

        # run a short sub‑gradient descent
        a, val = subgradient_descent(a, max_iters=180)

        # keep the best found so far
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[restart {restart}] new best (after descent) = {best_val:.6f}")

        # try the anti‑symmetric re‑ordering on the *current* best vector
        a_ord = anti_symmetric_reorder(best_a)
        val_ord = fast_objective(a_ord)
        if val_ord < best_val:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[restart {restart}] after anti‑sym reorder = {best_val:.6f}")

    # ------- final stochastic refinement ------------------------------------
    if time.time() < deadline:
        best_a, best_val = hill_climb(best_a, best_val, deadline, n)

    # ------- final verification with the official evaluator ------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence = {final_score:.6f}")
        # Use the official score as the returned value (it is the one that matters)
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
