def _smooth_obj_and_grad(w, eps, n):

    w_shift = w - np.max(w)

    shifted = (conv - max_conv) / eps
    # logsumexp
    log_sum = logsumexp(shifted)

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

The algorithm works in three stages under a 1000 s budget:

1. **Fast exploration on a small problem** – many random restarts of a
   smooth surrogate (log‑sum‑exp approximation of the max‑convolution)
   optimised with L‑BFGS‑B.  Only a few hundred L‑BFGS iterations are used
   per restart, which yields a diverse collection of promising shapes.

2. **Upscaling and refinement** – the best small‑scale shape is interpolated
   to a larger length (4000).  A longer L‑BFGS run with a finer smoothing
   schedule refines the candidate.

3. **Stochastic hill‑climbing (simulated annealing)** – random swaps
   and mass‑transfer moves are performed, accepting occasional worsening
   moves according to a linear cooling schedule.  An anti‑symmetric
   re‑ordering heuristic is applied periodically.

All convolutions are computed via FFT for speed, and a lightweight
`fast_objective` mirrors the exact ``evaluate_sequence`` computation
(except for the initial input‑validation steps, which are guaranteed to
be satisfied by construction).  The best sequence found is returned as a
list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global parameters for the L‑BFGS wrapper (avoids closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the surrogate
_GLOBAL_N = None     # current sequence length


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a vector with itself using FFT.
    Returns an array of length 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negative artefacts caused by round‑off.
    return np.maximum(conv, 0.0)


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using FFT.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return np.maximum(conv, 0.0)


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε (log‑sum‑exp approximation) and its gradient
    w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)  (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log∑_k exp((conv_k−max)/ε)

    Returns (value, gradient) where the gradient is ∂L/∂w.
    """
    # ---- soft‑max (feasible a) ----
    w_shift = w - np.max(w)            # improve numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # ---- convolution a*a ----
    conv = _full_convolution(a)        # length 2·n‑1
    max_conv = conv.max()

    # ---- smooth max via log‑sum‑exp ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)       # log∑_k exp((conv_k−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- probability weights for the smooth part ----
    weights = np.exp(shifted)
    p = weights / weights.sum()        # p_k ≥ 0, Σ p_k = 1

    # ---- gradient w.r.t. a (closed form) ----
    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)     # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]    # middle n entries
    grad_a = 4.0 * n * seg[::-1]       # shape (n,)

    # ---- chain rule through soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` – uses the global ε and n."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce a permutation where large values sit opposite small values.
    This heuristic often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)            # ascending order
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large to left side
        new_a[n - 1 - i] = sorted_vals[i]      # small to right side

    if n % 2:
        new_a[half] = sorted_vals[half]        # middle element unchanged
    return new_a


def fast_objective(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` (no input validation).
    Used internally for the many candidate evaluations.
    """
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a ** 2))


def _simulated_hill_climb(a: np.ndarray, cur_val: float,
                         deadline: float, n: int):
    """
    Simulated‑annealing style hill climbing.
    Random swaps and mass‑transfer moves keep the total mass constant.
    Accepts worsening moves with probability exp(−Δ/​T) where T cools linearly.
    """
    cur_a = a.copy()
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()
    start_time = time.time()
    # initial temperature – a modest fraction of the current objective
    T0 = max(1e-6, 0.03 * cur_val)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linear cooling: T → 0 as we approach the deadline
        T = max(1e-12, T0 * (1.0 - elapsed / (deadline - start_time)))

        # ---- propose a move -------------------------------------------------
        if rng.random() < 0.25:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                        # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            # move up to 10 % of donor's mass
            delta = cur_a[i] * 0.10 * rng.random()
            if delta <= 0.0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # ---- evaluate -------------------------------------------------------
        new_val = fast_objective(new_a)

        # ---- acceptance -----------------------------------------------------
        if new_val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_a, best_val = new_a.copy(), new_val
                print(f"[SA] new best = {best_val:.6f}")
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-(new_val - cur_val) / T):
                cur_a = new_a
                cur_val = new_val

        # occasional anti‑symmetric re‑ordering of the current solution
        if rng.random() < 0.01:
            reordered = _anti_symmetric_reorder(cur_a)
            val_reorder = fast_objective(reordered)
            if val_reorder < cur_val - 1e-12:
                cur_a, cur_val = reordered, val_reorder
                if val_reorder < best_val - 1e-12:
                    best_a, best_val = reordered.copy(), val_reorder
                    print(f"[SA] reorder improved best to {best_val:.6f}")

    return best_a, best_val
