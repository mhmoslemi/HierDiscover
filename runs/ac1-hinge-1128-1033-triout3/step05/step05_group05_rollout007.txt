   The vector is projected onto the probability simplex (Σa = 1).

   The objective  max(b) with `b = a∗a` is non‑smooth.
   We replace `max(b)` by the soft‑max
       sβ(b) = (1/β)·log Σ exp(β b_i) .
   Its gradient is
       ∇sβ(a) = 2·conv(w , a_rev)[n‑1:2·n‑1]   with   w_i = exp(β b_i)/Σexp(β b),
   where `a_rev` is `a` reversed.  Using FFT we evaluate the convolution in
   `O(n log n)`.  β is increased exponentially from 0.5 to 200,
   while the step size decays linearly.  After each step the vector is
   projected back onto the simplex.

3. **Phase B – Focused hill‑climbing on the true objective**  
   Starting from the best point found in Phase A we perform fast local moves:
   a small amount δ is moved from a *source* index `i` to a *target* index `j`.
   The source is chosen proportionally to its contribution `a_i·a_{k‑i}`
   to the current maximal convolution entry `k = argmax(b)`.  The effect on
   the full convolution can be updated in `O(n)` time using the bilinearity
   of the convolution.  Only moves that strictly lower `max(b)` are kept.
   After a long stagnation period we inject a tiny Gaussian perturbation
   and continue.

Both phases respect the time budget of 1000 s, keep the sum of the
coefficients equal to 1 (the evaluation is invariant under scaling) and
guarantee non‑negative entries.  The best sequence found is returned as a

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length = 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (length = len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector `v` onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0.0] = 0.0
    # renormalise (numerical drift)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w

# ----------------------------------------------------------------------

    keeping Σa = 1.  Returns the best sequence seen before *deadline*.

    cur_val = 2.0 * n * b.max()             # exact objective (sum(a)=1)

    max_frac = 0.5                          # max fraction of source mass to move
    IMPROVE_LIMIT = 180_000                 # stagnation trigger
    no_improve = 0
    iteration = 0

    # pre‑allocate range array for speed
    idx = np.arange(n, dtype=np.int64)

    while True:
        now = time.time()
        if now >= deadline:
            break

        # index of the current maximal convolution entry
        k = int(np.argmax(b))

        # compute contributions a_i * a_{k-i} only on the feasible slice
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        rng = slice(i_min, i_max + 1)
        i_vals = idx[rng]
        contrib = a[i_vals] * a[k - i_vals]

        # choose source index
        if contrib.sum() > 0.0 and np.random.rand() < 0.9:
            probs = contrib / contrib.sum()
            i = np.random.choice(i_vals, p=probs)
        else:
            i = np.random.choice(n, p=a)    # fallback: bias by mass

        if a[i] <= 1e-14:
            continue

        # choose target index uniformly (different from source)
        j = np.random.randint(n)
        if j == i:
            continue

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        # cross‑terms 2·δ·a shifted to j and i
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept only improving moves
        if new_val < cur_val - 1e-12:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # random perturbation after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a step‑function heights vector under the given time budget.
    Returns a list of non‑negative floats (Σ≈1) ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 6000 points give a bit more flexibility than 4000.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Initialise `a` – reuse a known construction if we have one.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)          # enforce Σa = 1, a_i ≥ 0
    best_a = a.copy()
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Phase A – smoothed‑max gradient descent
    # ------------------------------------------------------------------
    phase_a_end = start_time + 0.55 * budget_s
    iter_a = 0

    # schedules
    beta_start = 0.5
    beta_end = 200.0
    step_start = 0.5
    step_end = 0.005

    while time.time() < phase_a_end:
        # full convolution and current evaluation
        b = _full_convolution(a)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Grad {iter_a:7d}] new best = {best_val:.6f}")

        # schedule progress (0 … 1)
        prog = (time.time() - start_time) / (phase_a_end - start_time)
        # beta grows exponentially, step decays linearly
        beta = beta_start * (beta_end / beta_start) ** prog
        step_scale = step_start * (1.0 - prog) + step_end * prog

        # soft‑max weights (numerically stable)
        b_shift = b - b.max()
        w_raw = np.exp(beta * b_shift)
        w = w_raw / w_raw.sum()

        # gradient of the soft‑max objective
        a_rev = a[::-1]
        conv = _conv_fft(w, a_rev)                     # length = 3·n‑2
        grad = 2.0 * conv[n - 1:2 * n - 1]

        norm_g = np.linalg.norm(grad)
        step = step_scale / (norm_g + 1e-16) if norm_g > 0 else 0.0

        # projected gradient step
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional tiny jitter to shake out of shallow plateaus
        if np.random.rand() < 0.001:
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)

        iter_a += 1

    # ------------------------------------------------------------------
    # Phase B – deterministic hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    best_a = _hillclimb_phase(best_a, n, start_time, deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
