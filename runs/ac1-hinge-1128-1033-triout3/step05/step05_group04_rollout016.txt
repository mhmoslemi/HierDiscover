    # Helper functions defined top level

    best_seq = None

    # Define convolution via FFT _full_convolution_fft

    def compute_score(a):

        a_norm = a / a.sum()
        conv = _full_convolution_fft(a_norm, a_norm)
        return 2.0 * n * conv.max()
    
    # Warm start from global if available
    if 'height_sequence_1' in globals():
        try:
            init_a = np.asarray(height_sequence_1, dtype=np.float64)
            if init_a.size == n:

                else:

            else:

        base = base / base.sum()

    if a0 is None:

            best_seq = a.copy()

"""
Propose a step‐function sequence with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The routine combines three ideas:

1.  A very good analytical start – an exponential decay whose
    parameter is the optimum of the one‑parameter family
    a_i ∝ r**i (the solution of e**α = 1+2α).

2.  A long “soft‑max Adam’’ phase that minimises a smooth
    approximation of the max‑convolution (the usual log‑sum‑exp
    smoothing).  The temperature decays very slowly so the
    optimiser can keep making progress for many minutes.

3.  A hard‑max polishing phase that uses the exact sub‑gradient of the
    true max‑convolution and a tiny constant step‑size.  Random jitter
    is added occasionally to escape tiny plateaus.

All operations use an FFT based convolution (O(n log n)) and the
projection onto the probability simplex (Duchi et al. 2008).  The code
monitors the best sequence seen so far and returns it after the
budget (~1000 s) expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for many iterations)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

# Main optimisation routine
# ----------------------------------------------------------------------

    Search for a non‑negative height sequence (list of floats) that
    minimises `evaluate_sequence`.  The routine respects the supplied
    time budget (seconds) and returns the best sequence found.
    """

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (the previous best used n=5000, we keep the same)
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 1️⃣  Analytic exponential start (close to the optimum of the
    #      a_i ∝ r**i family).  α solves e^α = 1 + 2α  → α≈1.25643.
    # ------------------------------------------------------------------
    alpha_opt = 1.256430
    r_opt = np.exp(-alpha_opt / n)                  # r ≈ 0.9997488 for n=5000
    a = r_opt ** np.arange(n, dtype=np.float64)
    a = a / a.sum()                                 # sum = 1, non‑negative

    # If a warm‑start is available, blend it with the analytic start
    if 'height_sequence_1' in globals():
        try:
            h = np.asarray(height_sequence_1, dtype=np.float64)
            if h.shape[0] == n:
                h = np.maximum(0.0, h)
                if h.sum() > 0:
                    h = h / h.sum()
                    a = 0.5 * a + 0.5 * h
                    a = a / a.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Helper to compute the objective (sum(a) == 1 → 2·n·max(conv))
    # ------------------------------------------------------------------
    def _score(vec):
        conv = _full_convolution_fft(vec, vec)
        return 2.0 * n * conv.max()

    best_seq = a.copy()
    best_val = _score(a)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Hyper‑parameters
    lr = 0.12

    lr_decay = 0.99997          # very slow decay – we need many fine steps
    beta1, beta2 = 0.9, 0.999

    eps_decay = 0.99995               # slower than before – keep smoothing

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    stagn = 0

    # allocate ~70 % of the budget to this phase
    while time.time() < deadline * 0.70:
        it += 1

        # current convolution and score
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        # occasional renormalisation for numerical safety
        if it % 4000 == 0:

    # ------------------------------------------------------------------
    # 3️⃣  Hard‑max polishing (true sub‑gradient of the max‑convolution)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    step_sz = 2e-5
    it_polish = 0
    while time.time() < deadline:
        it_polish += 1

        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

            best_val = cur_val

            print(f"[polish {it_polish:6d}] new_best={best_val:.12f}")

        mask = (conv >= max_conv - tol).astype(np.float64)
        w_sum = mask.sum()
        if w_sum == 0.0:

        else:
            w = mask / w_sum

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        a = a - step_sz * grad
        a = _project_simplex(a)

        # slowly decay the step size
        step_sz = max(5e-7, step_sz * 0.99999)

        # random jitter if we are stuck for a long time
        if it_polish % 5000 == 0:
            a = a + rng.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            print(f"[jitter] iteration {it_polish}")

    # Final few pure sub‑gradient steps (tiny decaying step)
    # ------------------------------------------------------------------
    a = best_seq.copy()
