"""
Improved step‑function optimizer.

The routine mixes three ideas that proved useful for the
`evaluate_sequence` objective:

1️⃣  **Exponentiated gradient on the simplex** – we store the
    distribution as logits `z` (so `a = softmax(z)`).  This removes the
    need for a separate projection step and respects the positivity /
    sum‑to‑one constraints by construction.

2️⃣  **Smooth max via temperature‑controlled soft‑max** – during the
    main phase we replace `max(conv(a,a))` by a soft‑max with a slowly
    decreasing temperature `eps`.  The resulting gradient is the
    cross‑correlation of the reversed distribution with the soft‑max
    weights, which is cheap to compute using an FFT based full
    convolution.

3️⃣  **Hard‑max polishing** – once the temperature is tiny we switch to
    a true sub‑gradient step focused only on the (almost) maximal
    convolution entries, followed by a final pure sub‑gradient sweep.

The whole procedure respects a user supplied time budget (default
1000 s), keeps the best distribution seen, and returns a list suitable
for `evaluate_sequence`.  It also warm‑starts from the global variable
`height_sequence_1` if it is present.
"""

import numpy as np

# Helper: project a vector onto the probability simplex.
# (used only for safety / occasional jitter)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using a real FFT (fast and accurate).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative height sequence that (approximately) minimises
    `2 * n * max(conv(a, a)) / (sum(a) ** 2)`.
    The routine respects the supplied time budget (in seconds) and
    returns the best sequence found.
    """
    # ------------------------------------------------------------------
    # General set‑up and time limits.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a modest problem size – large enough for a good bound,
    # small enough to allow many iterations.
    # ------------------------------------------------------------------
    n = 2048                     # power of two → fast FFTs

    # ------------------------------------------------------------------
    # Warm‑start from a global variable if possible, otherwise build a
    # smooth “nice’’ shape.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.size >= n:
                a0 = base[:n].copy()
            else:
                a0 = np.concatenate([base, rng.random(n - base.size)])
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        x = np.arange(n, dtype=np.float64)
        # A mix of several smooth windows (sine, semicircle, Gaussian,
        # triangular) – well behaved in the frequency domain.
        sine = np.sin(np.pi * (x + 0.5) / n)
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
        triangle = np.minimum(x, n - 1 - x) + 1.0
        bases = [sine, semicircle, gauss, triangle]
        w = rng.random(len(bases))
        w /= w.sum()
        a0 = sum(wi * bi for wi, bi in zip(w, bases))
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
        a0 /= a0.sum()

    # ------------------------------------------------------------------
    # Work in log‑space (exponentiated gradient) to stay on the simplex.
    # ------------------------------------------------------------------
    eps0 = 1e-14
    a = np.clip(a0, eps0, None)
    a /= a.sum()
    z = np.log(a + eps0)               # logits, unconstrained

    # ------------------------------------------------------------------
    # Keep track of the best solution found.
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv          # sum(a)=1 → denominator = 1
    best_seq = a.copy()
    print(f"[init] n={n}, init_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the exponential‑gradient phase.
    # ------------------------------------------------------------------
    lr = 0.20                              # initial step size on logits
    lr_min = 1e-5
    lr_decay = 0.99992
    eps = max(max_conv * 0.5, 1e-12)        # soft‑max temperature
    eps_min = max(max_conv * 1e-6, 1e-14)
    eps_decay = 0.99993
    stagn = 0                               # iterations without improvement
    it = 0

    # ------------------------------------------------------------------
    # Phase 1 – smooth optimisation (soft‑max + exponentiated gradient).
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.78 * (deadline - start_time)

    while time.time() < phase1_deadline:
        it += 1

        # recover distribution from logits
        a = np.exp(z)
        a /= a.sum()

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # record improvement
        if score < best_val - 1e-13:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it:6d}] new_best={best_val:.12f}  lr={lr:.2e}  eps={eps:.2e}")
        else:
            stagn += 1

        # ----- soft‑max weighting of convolution entries
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # ----- gradient of the weighted sum (cross‑correlation)
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]            # slice to size n

        # ----- exponentiated‑gradient update on logits
        avg = np.dot(a, grad)
        g_z = a * (grad - avg)                               # ∇_z L
        z = z - lr * g_z
        # keep logits numerically sane
        z = np.clip(z, -50.0, 50.0)

        # ----- schedules
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        # ----- perturbation if stuck for a while
        if stagn > 2500:
            noise = rng.normal(scale=0.01, size=n)
            z += noise
            stagn = 0
            lr = max(lr_min, lr * 1.3)
            print(f"[perturb] iter {it}")

        # ----- occasional re‑normalisation (numeric safety)
        if it % 5000 == 0:
            a = np.exp(z)
            a /= a.sum()
            z = np.log(a + eps0)

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max polishing (binary mask of true maximal entries).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        a = np.exp(z)
        a /= a.sum()

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-13:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[polish {it:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # binary mask of entries attaining (or almost attaining) the max
        tol = max_conv * 1e-9 + 1e-14
        mask = (conv >= max_conv - tol).astype(float)
        w_sum = mask.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask / w_sum

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # a tiny sub‑gradient step; we stay on the simplex via re‑normalisation
        step = -2e-5 * grad
        a = a + step
        a = np.clip(a, 1e-24, None)
        a /= a.sum()
        z = np.log(a)

        # occasional jitter to escape plateaus
        if stagn > 4000:
            a = a + rng.normal(scale=0.003, size=n)
            a = np.clip(a, 1e-24, None)
            a /= a.sum()
            z = np.log(a)
            stagn = 0
            print(f"[jitter] iter {it}")

    # ------------------------------------------------------------------
    # Final pure sub‑gradient sweep (a few thousand iterations) to shave
    # off the last bits of the max‑convolution.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    for _ in range(5000):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        active = (conv >= mx - 1e-15).astype(float)
        w_sum = active.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = active / w_sum
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]
        step = 4e-6 * grad
        a = a - step
        a = np.clip(a, 0.0, None)
        a /= a.sum()

    # ------------------------------------------------------------------
    # Convert to the required list format and report the final score.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
