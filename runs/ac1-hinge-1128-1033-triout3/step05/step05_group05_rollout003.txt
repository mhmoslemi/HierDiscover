def full_convolution_fft(a):
    n = len(a)
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def conv_fft(x, y):
    n, m = len(x), len(y)
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

"""
Propose a non‑negative step‑function with a low autocorrelation peak.

Algorithm
---------

1. **Initialization** – start from the supplied `height_sequence_1` (if any)
   interpolated to a fixed length `n` (default 8000) or from a random
   Dirichlet vector.  The vector is projected onto the probability simplex
   (non‑negative, sum = 1).

2. **Phase C – Soft‑max gradient descent on the ∞‑norm of the autocorrelation**  
   For a current vector `a` we compute its full autocorrelation `b = a*a`.
   A smooth surrogate of `max(b)` is the soft‑max
   `L(β) = log Σ_k exp(β·b_k) / β`.  Its gradient can be expressed by
   two FFT‑based convolutions.  Starting with a small temperature `β`
   (smooth landscape) and annealing it up to a larger value we perform
   projected gradient steps (step‑size linearly decays).  Whenever a
   better objective `2·n·max(b)` is found it is stored.

3. **Phase D – Simulated‑annealing hill‑climbing**  
   Starting from the best point found in Phase C we repeatedly move a
   small amount `δ` from a source index to a target index.  The effect on
   the full autocorrelation can be updated in `O(n)` time using the
   bilinearity of convolution.  Moves that improve the true objective
   are always accepted; worsening moves are accepted with a Metropolis
   probability that slowly cools down.  After long stagnation a small
   random perturbation is applied to escape local minima.

Both phases respect the 1000 s time budget.  The final sequence (a list
of floats) is returned – it already satisfies the constraints required
by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (real input)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of `x` with itself (length 2·len(x)‑1) using FFT.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT (length len(x)+len(y)-1).
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # make all entries non‑negative first
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0:
        # fallback to uniform vector
        return np.full_like(v, 1.0 / v.shape[0])
    v /= s
    # the algorithm of Wang & Carreira‑Perpiñán (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Objective value (exact, using the same formula as `evaluate_sequence`)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(a*a)  (assuming Σa = 1)."""
    n = a.shape[0]
    b = _full_convolution_fft(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase C – gradient descent on a soft‑max approximation of max(b)
# ----------------------------------------------------------------------
def _softmax_gradient_phase(a_init: np.ndarray,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """
    Runs gradient descent on the soft‑max surrogate of `max(b)`.
    Returns the best vector seen and its objective.
    """
    a = a_init.copy()                     # always keeps Σa = 1
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    # hyper‑parameters
    beta_start = 0.5
    beta_end = 30.0
    base_step = 0.8
    min_step = 0.002
    max_no_improve = 120_000
    no_improve = 0

    while time.time() < deadline:
        # full autocorrelation and its maximum
        b = _full_convolution_fft(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Grad] new best {best_val:.6f}")
        else:
            no_improve += 1

        # temperature (soft‑max sharpness) linearly annealed
        progress = (time.time() - start_time) / (deadline - start_time)
        beta = beta_start + progress * (beta_end - beta_start)

        # soft‑max probabilities (stable version)
        w = np.exp(beta * (b - max_b))          # max(b) → exponent 0 → weight 1
        p = w / w.sum()

        # gradient of L = Σ p_k b_k  w.r.t. a
        a_rev = a[::-1]
        conv1 = _conv_fft(p, a)                 # length 3n‑2
        conv2 = _conv_fft(p, a_rev)             # length 3n‑2
        grad = conv1[:n] + conv2[n - 1:2 * n - 1]

        # step‑size schedule (linear decay)
        step = base_step * (1.0 - progress) + min_step * progress

        # normalise gradient to keep the step magnitude reasonable
        grad_norm = np.linalg.norm(grad) + 1e-16
        a_candidate = a - step * grad / grad_norm

        # project back onto the simplex (fast version)
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s > 0:
            a_candidate /= s
        else:
            a_candidate = np.full_like(a, 1.0 / n)

        a = a_candidate

        # occasional random jitter to escape tiny plateaus
        if np.random.rand() < 0.002:
            a += np.random.normal(scale=0.001, size=n)
            a = _project_simplex(a)

        # stagnation handling: random restart while preserving best
        if no_improve > max_no_improve:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print("[Grad] stagnation – random restart")

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – simulated‑annealing hill climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     best_a: np.ndarray,
                     best_val: float,
                     deadline: float) -> tuple[np.ndarray, float]:

    Local search that moves a small amount of mass between two coordinates.
    Accepts worsening moves with a Metropolis probability that cools down.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution_fft(a)
    cur_val = 2.0 * n * b.max()

    # simulated‑annealing schedule
    start_hill = time.time()
    temp_start = max(0.02 * cur_val, 1e-4)
    temp_end = 1e-6

    # stagnation handling
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    max_frac = 0.35                     # maximal fraction of source mass to move

    while time.time() < deadline:
        # temperature based on elapsed hill‑climb time
        progress = (time.time() - start_hill) / (deadline - start_hill)
        temp = temp_start * (1.0 - progress) + temp_end * progress

        # choose source index (biased towards the current largest entry)
        if np.random.rand() < 0.85:
            i = int(np.argmax(a))
        else:
            # probabilities proportional to a (so we are more likely to take from larger masses)
            i = np.random.choice(n, p=a / a.sum())

        if a[i] <= 1e-15:
            continue

        # choose target index ≠ i
        j = np.random.randint(n)
        if j == i:
            continue

        # amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        # cross terms 2·δ·a shifted
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability exp(-(new‑cur)/T)
            if np.random.rand() < np.exp((cur_val - new_val) / max(temp, 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] new best {best_val:.6f} at iter {iteration}")
        else:
            no_improve += 1

        # long stagnation → small random perturbation around the best found so far
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.001, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution_fft(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] stagnation – perturbation at iter {iteration}")

        iteration += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes `evaluate_sequence`.
    Returns a list of floats (Σ ≈ 1) ready for scoring.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin to guarantee we finish before the hard deadline
    deadline = start_time + budget_s - 0.4

    # --------------------------------------------------------------
    # Problem size – 8000 provides a richer search space without
    # significantly increasing FFT cost (next power of two stays 32768).
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initial vector – use the supplied construction if available,
    # otherwise start from a random Dirichlet vector.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        # interpolate to the new size
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        a0 = np.random.rand(n)

    # project onto simplex (non‑negative, sum = 1)
    a0 = _project_simplex(a0)

    best_a = a0.copy()
    best_val = _objective(best_a)

    # --------------------------------------------------------------
    # Phase C – soft‑max gradient descent (≈ 45 % of the budget)
    # --------------------------------------------------------------
    phase_c_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _softmax_gradient_phase(best_a, phase_c_deadline, start_time)

    # --------------------------------------------------------------
    # Phase D – simulated‑annealing hill climbing (remaining time)
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb_phase(best_a, best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final verification using the provided secure wrapper.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
