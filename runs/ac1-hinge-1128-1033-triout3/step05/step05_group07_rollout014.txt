def proj_simplex(v):
    # We ensure v sum = 1, v_i >= 0.
    # Implementation from Duchi et al. "Efficient Projections onto the L1-Ball for Learning in High Dimensions"
    # Actually projection onto simplex:

    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def convolve_full(x, y): # returns np.convolve(x,y) via FFT to length len(x)+len(y)-1.

    total = len(x) + len(y) - 1
    size = 1 << (total - 1).bit_length()

    fy = np.fft.rfft(y, size)

"""
Propose a candidate step‑function by aggressive peak‑flattening.

The algorithm builds on the hill‑climbing idea from the reference solution
but adds a small look‑ahead: for the index that contributes most to the
current convolution peak we evaluate several random destination indices
and pick the one that yields the largest reduction of the peak.
The convolution is kept incrementally (O(n)) and only recomputed from
scratch periodically to avoid numerical drift.  An occasional jitter move
helps to escape plateaus.  The sequence is kept on the probability simplex
(∑a_i = 1, a_i ≥ 0) because the evaluation function is scale‑invariant.
The routine runs for the allotted time budget (≈1000 s) and returns the
best sequence found.
"""

import time
import numpy as np


def _self_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT (full length 2·len(a)‑1)."""
    n = a.shape[0]
    total = 2 * n - 1
    size = 1 << (total - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:total]
    # eliminate tiny negatives from round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        best_move = None
        best_new_obj = cur_obj
        a_old = a.copy()
        conv_old = conv.copy()

        for j_dst in cand_js:
            # random fraction of the allowed move size; ensure it is >0
            delta = move_frac * a[i_src] * np.random.rand()
            if delta <= 0.0:
                continue

            # copy current convolution and apply the exact update for this delta
            conv_tmp = conv_old.copy()
            # linear part: shift contributions
            conv_tmp[i_src:i_src + n] -= 2.0 * delta * a_old
            conv_tmp[j_dst:j_dst + n] += 2.0 * delta * a_old
            # quadratic corrections
            delta2 = delta * delta
            conv_tmp[2 * i_src] += delta2
            conv_tmp[2 * j_dst] += delta2
            conv_tmp[i_src + j_dst] -= 2.0 * delta2

            new_max = conv_tmp.max()
            new_obj = float(2.0 * n * new_max)

            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_move = (j_dst, delta, conv_tmp)

        # --------------------------------------------------------------
        # Accept the best improving move (if any)
        # --------------------------------------------------------------
        if best_move is not None and best_new_obj < cur_obj - 1e-12:
            j_dst, delta, conv_new = best_move
            a[i_src] -= delta
            a[j_dst] += delta
            conv = conv_new
            cur_obj = best_new_obj
            cur_max = conv.max()

            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration:7d}] new best = {best_obj:.6f}")
                last_improve_iter = iteration
        else:
            # ----------------------------------------------------------
            # No improving move found – try an occasional “jitter” step
            # ----------------------------------------------------------
            if np.random.rand() < jitter_prob:
                src = np.random.randint(n)
                dst = np.random.randint(n - 1)
                if dst >= src:
                    dst += 1
                delta_j = 0.5 * a[src] * np.random.rand()
                if delta_j > 0.0:
                    # incremental update on a copy (for easy revert)
                    conv_tmp = conv.copy()
                    conv_tmp[src:src + n] -= 2.0 * delta_j * a
                    conv_tmp[dst:dst + n] += 2.0 * delta_j * a
                    d2 = delta_j * delta_j
                    conv_tmp[2 * src] += d2
                    conv_tmp[2 * dst] += d2
                    conv_tmp[src + dst] -= 2.0 * d2

                    new_max = conv_tmp.max()
                    new_obj = float(2.0 * n * new_max)

                    if new_obj < cur_obj:
                        a[src] -= delta_j
                        a[dst] += delta_j
                        conv = conv_tmp
                        cur_obj = new_obj
                        cur_max = conv.max()
                        if cur_obj < best_obj:
                            best_obj = cur_obj
                            best_a = a.copy()
                            print(f"[{iteration:7d}] jitter improved best = {best_obj:.6f}")
                            last_improve_iter = iteration

        # --------------------------------------------------------------
        # Periodic full recompute to avoid floating‑point drift
        # --------------------------------------------------------------
        if iteration % recompute_interval == 0:
            conv = _self_convolution_fft(a)

        # --------------------------------------------------------------
        # Simple stagnation detection – if no improvement for a long while
        # re‑normalise (which also forces a cheap full recompute).
        # --------------------------------------------------------------
        if iteration - last_improve_iter > 200_000:
            a /= a.sum()
            conv = _self_convolution_fft(a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)
            last_improve_iter = iteration
            print(f"[{iteration:7d}] re‑normalised after stagnation, obj={cur_obj:.6f}")

    # -----------------------------------------------------------------
    # Final reporting
    # -----------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
