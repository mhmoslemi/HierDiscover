def project_onto_simplex(v):
    # Implementation from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Simulated‑annealing mass‑transfer search.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0).
At each iteration it identifies the index `k*` where the self‑convolution
a*a attains its maximum and the pair (i, k*‑i) that contributes most to
that peak.  A fraction of the mass of the source index *i* is moved to a
destination *j* that contributes the *least* to the current peak (often
outside the support, i.e. a zero contribution).  The move is accepted
if it lowers the objective; otherwise it may be accepted with a small
probability governed by a linear temperature schedule (simulated
annealing).  Occasional random “jitter” moves help to escape plateaus.
The convolution is updated incrementally in O(n) per move; a full FFT
re‑computation is performed periodically to avoid drift.

The search runs for the allotted time budget (default 1000 s) and returns
the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast exact convolution via real FFT (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return linear convolution x*y (length len(x)+len(y)-1) using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()        # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Clip tiny negatives that may appear due to round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Simulated‑annealing / targeted mass‑transfer optimisation to minimise
    evaluate_sequence(...). Starts from ``height_sequence_1`` if it exists,
    otherwise from a uniform distribution of length 1800.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the framework
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Initialise a – use the provided best construction or a uniform start
    # --------------------------------------------------------------
    try:
        init_seq = height_sequence_1               # noqa: F821
    except NameError:
        init_seq = None

    if init_seq is None:
        n = 1800
        a = np.full(n, 1.0 / n)
    else:
        a = np.asarray(init_seq, dtype=float).copy()
        n = a.shape[0]
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Initialise the exact convolution and bookkeeping
    # --------------------------------------------------------------
    conv = _full_convolution_fft(a, a)          # length 2n‑1
    cur_max = conv.max()
    cur_obj = float(2.0 * n * cur_max)          # objective for sum(a)=1
    best_obj = cur_obj
    best_a = a.copy()

    print(f"[init] n={n}, initial objective={cur_obj:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    max_frac = 0.25          # up‑to‑25 % of source mass moved initially
    min_frac = 0.03          # shrink to ~3 % near the end
    jitter_prob = 0.025      # more frequent random jitter
    recompute_interval = 5000
    temperature0 = cur_obj * 0.02   # initial temperature for uphill moves

    iteration = 0
    last_improve_iter = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1
        progress = (now - start_time) / (deadline - start_time)

        # ----------------------------------------------------------
        # Linear annealing for move size and temperature
        # ----------------------------------------------------------
        move_frac = max_frac * (1.0 - progress) + min_frac * progress
        temperature = temperature0 * (1.0 - progress)

        # ----------------------------------------------------------
        # Locate the current peak of the convolution and the most
        # responsible coefficient pair.
        # ----------------------------------------------------------
        k_star = int(conv.argmax())
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        idx = np.arange(i_start, i_end + 1)
        partner = k_star - idx
        products = a[idx] * a[partner]
        i_src = int(idx[products.argmax()])          # source index

        # ----------------------------------------------------------
        # Choose a destination index that contributes the least to the
        # current peak (often an out‑of‑range partner, i.e. zero).
        # ----------------------------------------------------------
        contrib = np.zeros(n)
        # valid destinations are those where k_star - j is inside [0,n‑1]
        low = max(0, k_star - (n - 1))
        high = min(n - 1, k_star)
        idx2 = np.arange(low, high + 1)
        partner2 = k_star - idx2
        contrib[idx2] = a[partner2]          # contribution of each j to the peak
        contrib[i_src] = np.inf               # never move back to the source
        j_dst = int(np.argmin(contrib))

        # ----------------------------------------------------------
        # Propose a move: transfer a fraction of a[i_src] to a[j_dst]
        # ----------------------------------------------------------
        delta = move_frac * a[i_src] * np.random.rand()
        if delta <= 0.0:
            continue
        a_old = a.copy()
        delta2 = delta * delta

        # ---- Incremental update of the convolution (exact) ----
        conv[i_src:i_src + n] -= 2.0 * delta * a_old
        conv[j_dst:j_dst + n] += 2.0 * delta * a_old
        conv[2 * i_src] += delta2
        conv[2 * j_dst] += delta2
        conv[i_src + j_dst] -= 2.0 * delta2

        # ---- Update the coefficients ----
        a[i_src] -= delta
        a[j_dst] += delta

        # ---- Evaluate new objective ----
        new_max = conv.max()
        new_obj = float(2.0 * n * new_max)

        # ---- Acceptance rule (downhill + occasional uphill) ----
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            if temperature > 0.0:
                prob = math.exp(-(new_obj - cur_obj) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_obj = new_obj
            cur_max = new_max
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[{iteration:7d}] new best = {best_obj:.6f}")
                last_improve_iter = iteration
        else:
            # ---- Revert the move ----
            a[:] = a_old
            conv[i_src:i_src + n] += 2.0 * delta * a_old
            conv[j_dst:j_dst + n] -= 2.0 * delta * a_old
            conv[2 * i_src] -= delta2
            conv[2 * j_dst] -= delta2
            conv[i_src + j_dst] += 2.0 * delta2

        # ----------------------------------------------------------
        # Occasionally perform a random “jitter” move to shake the system.
        # ----------------------------------------------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            if src != dst:
                delta_j = 0.5 * a[src] * np.random.rand()
                if delta_j > 0.0:
                    a_tmp = a.copy()
                    delta2j = delta_j * delta_j
                    conv[src:src + n] -= 2.0 * delta_j * a_tmp
                    conv[dst:dst + n] += 2.0 * delta_j * a_tmp
                    conv[2 * src] += delta2j
                    conv[2 * dst] += delta2j
                    conv[src + dst] -= 2.0 * delta2j
                    a[src] -= delta_j
                    a[dst] += delta_j

                    jitter_max = conv.max()
                    jitter_obj = float(2.0 * n * jitter_max)

                    accept_j = False
                    if jitter_obj < cur_obj:
                        accept_j = True
                    else:
                        if temperature > 0.0:
                            prob_j = math.exp(-(jitter_obj - cur_obj) / temperature)
                            if np.random.rand() < prob_j:
                                accept_j = True

                    if accept_j:
                        cur_obj = jitter_obj
                        cur_max = jitter_max
                        if cur_obj < best_obj:
                            best_obj = cur_obj
                            best_a = a.copy()
                            print(f"[{iteration:7d}] jitter improved best = {best_obj:.6f}")
                            last_improve_iter = iteration
                    else:
                        # revert the jitter
                        a = a_tmp
                        conv[src:src + n] += 2.0 * delta_j * a_tmp
                        conv[dst:dst + n] -= 2.0 * delta_j * a_tmp
                        conv[2 * src] -= delta2j
                        conv[2 * dst] -= delta2j
                        conv[src + dst] += 2.0 * delta2j

        # ----------------------------------------------------------
        # Periodic full recomputation to avoid numeric drift.
        # ----------------------------------------------------------
        if iteration % recompute_interval == 0:
            conv = _full_convolution_fft(a, a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)

        # ----------------------------------------------------------
        # Simple stagnation detection – if no improvement for a long time
        # re‑normalise and recompute.
        # ----------------------------------------------------------
        if iteration - last_improve_iter > 250_000:
            a /= a.sum()
            conv = _full_convolution_fft(a, a)
            cur_max = conv.max()
            cur_obj = float(2.0 * n * cur_max)
            last_improve_iter = iteration

    # ----------------------------------------------------------------------
    # Final reporting
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
