        conv = _full_convolution_fft(y, y)
        max_c = conv.max()

"""
Propose a step‑function (non‑negative sequence) that approximately minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of two ideas:

1.  **Pattern search** – the target length `n` is written as a repetition of a
    shorter pattern `y` (`n = R·p`).  For a fixed pattern length `p` the
    objective becomes `2·p·max(conv(y,y))`.  A fast gradient‑descent /
    Adam optimisation is run on the small pattern (sizes from 100 up to
    3000, all divisors of `n`).  The best pattern found over all `p` values
    is lifted to a full‑length candidate.

2.  **Fine‑tuning** – the best full‑length candidate is refined with a short
    Adam run on the original 6000‑dimensional problem (exact max‑sub‑gradient
    and projection onto the probability simplex).

All updates respect the 1000 s budget, print progress information and finally
return the best sequence seen (clipped to the limits required by
`evaluate_sequence`).  The code is self‑contained and uses only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Linear convolution via real FFT (full linear convolution).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

# The objective for a pattern is 2 * p * max(conv(y, y)).
# ----------------------------------------------------------------------
def _optimize_pattern(p: int, budget_s: float, rng: np.random.Generator) -> tuple:
    """
    Return a tuple (best_y, best_score) where best_y is a non‑negative
    vector of length p with sum 1, and best_score = 2 * p * max(conv(y,y)).
    The routine runs for at most `budget_s` seconds.
    """
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ----- initial point -------------------------------------------------
    y = rng.random(p)
    y = y / y.sum()                     # onto simplex

    # ----- bookkeeping ---------------------------------------------------
    conv = _full_convolution_fft(y, y)
    max_c = conv.max()
    best_score = 2.0 * p * max_c
    best_y = y.copy()
    it = 0
    stagn = 0

    # ----- Phase 1 – soft‑max (smooth approximation of max) -------------
    phase1_end = start + 0.45 * budget_s
    eta = 0.8
    eta_min = 1e-6
    eta_decay = 0.9995
    eps = max(max_c * 0.5, 1e-12)
    eps_min = max(max_c * 1e-5, 1e-12)
    eps_decay = 0.9995
    momentum = 0.9
    vel = np.zeros_like(y)

    while time.time() < phase1_end:
        it += 1
        conv = _full_convolution_fft(y, y)
        max_c = conv.max()
        score = 2.0 * p * max_c

        if score < best_score - 1e-15:
            best_score = score
            best_y = y.copy()
            stagn = 0
            print(f"[p={p} phase1 iter {it:6d}] new_best={best_score:.12f}")
        else:
            stagn += 1

        # soft‑max weights
        w = np.exp((conv - max_c) / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient (cross‑correlation of rev(y) with w)
        rev_y = y[::-1]
        tmp = _full_convolution_fft(rev_y, w)
        g = 2.0 * tmp[p - 1:p - 1 + p]

        # momentum step
        vel = momentum * vel + (1.0 - momentum) * g

        # projected gradient step
        y = _project_simplex(y - eta * vel)

        # decay schedules
        eta = max(eta_min, eta * eta_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional perturbation when stuck
        if stagn > 2500:
            noise = rng.normal(scale=0.008, size=p)
            y = _project_simplex(y + noise)
            stagn = 0
            eta = max(eta_min, eta * 1.2)
            print(f"[p={p} phase1 perturb] iter {it}")

        # safety renormalisation
        if it % 4000 == 0:
            y = y / y.sum()

    # ----- Phase 2 – exact max sub‑gradient with Adam adaptation ---------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.12
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1_pow = 1.0
    beta2_pow = 1.0
    t_adam = 0
    stagn = 0

    while time.time() < deadline:
        t_adam += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _full_convolution_fft(y, y)
        max_c = conv.max()
        score = 2.0 * p * max_c

        if score < best_score - 1e-15:
            best_score = score
            best_y = y.copy()
            stagn = 0
            print(f"[p={p} phase2 iter {t_adam:6d}] new_best={best_score:.12f}")
        else:
            stagn += 1

        # exact max sub‑gradient (average over all active entries)
        tol = 1e-14
        active = np.flatnonzero(conv >= max_c - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        rev_y = y[::-1]
        tmp = _full_convolution_fft(rev_y, w)
        g = 2.0 * tmp[p - 1:p - 1 + p]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        y_candidate = y - step
        y_candidate = _project_simplex(y_candidate)

        y = y_candidate

        # perturb if stuck
        if stagn > 3000:
            noise = rng.normal(scale=0.004, size=p)
            y = _project_simplex(y + noise)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            print(f"[p={p} phase2 perturb] iter {t_adam}")

        if t_adam % 5000 == 0:
            y = y / y.sum()

    # final sanity recomputation
    conv = _full_convolution_fft(best_y, best_y)
    best_score = 2.0 * p * conv.max()
    return best_y, best_score

# Main entry point: propose the candidate sequence.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative height sequence that (approximately) minimizes the
    evaluation function.  The routine respects the `budget_s` time limit
    and prints intermediate scores.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep the original size used in earlier runs.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Baseline: if a previous good sequence exists, initialise from it.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if "height_sequence_1" in globals():
        try:
            seq = np.asarray(height_sequence_1, dtype=np.float64)
            if seq.shape[0] != n:
                if seq.shape[0] > n:
                    seq = seq[:n]
                else:
                    pad = np.zeros(n - seq.shape[0], dtype=np.float64)
                    seq = np.concatenate([seq, pad])
            seq = np.maximum(0.0, seq)
            seq = seq / seq.sum()
            conv = _full_convolution_fft(seq, seq)
            best_val = 2.0 * n * conv.max()
            best_seq = seq.copy()
            print(f"[init] using previous sequence, score={best_val:.12f}")
        except Exception:
            pass

    if best_seq is None:
        # fallback – uniform vector
        best_seq = np.ones(n, dtype=np.float64) / n
        best_val = 2.0 * n * (1.0 / n)   # = 2.0
        print("[init] uniform start, score=2.000000000000")

    # ------------------------------------------------------------------
    # Pattern search – try several divisors of n.
    # ------------------------------------------------------------------
    # All divisors of 6000 that are not too small/large.
    candidate_ps = [
        100, 120, 150, 200, 250, 300, 400,
        500, 600, 750, 1000, 1200, 1500,
        2000, 3000, 6000,
    ]
    candidate_ps = [p for p in candidate_ps if n % p == 0]

    total_patterns = len(candidate_ps)
    for idx, p in enumerate(candidate_ps):
        now = time.time()
        remaining = deadline - now
        # Reserve a small buffer for the remaining patterns + final fine‑tune.
        reserve = max(10.0 * (total_patterns - idx - 1), 30.0)
        if remaining <= reserve:
            print("[info] insufficient time for further pattern optimisation")
            break
        # Allocate up to 80 % of the free time to this pattern.
        alloc = min(80.0, remaining - reserve)
        print(f"[info] optimise pattern p={p} – allocated {alloc:.1f}s")
        y_opt, score_opt = _optimize_pattern(p, alloc, rng)
        print(f"[result] pattern p={p} best_score={score_opt:.12f}")

        if score_opt < best_val:
            best_val = score_opt
            best_p = p
            # lift pattern to full length
            repeats = n // p
            full = np.tile(y_opt, repeats)
            full = full / full.sum()               # project onto simplex
            best_seq = full.copy()
            print(f"[update] new global best via pattern p={p}, score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Fine‑tune the full‑length candidate (if time permits).
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 15.0:
        print(f"[fine‑tune] running Adam on full vector for ~{time_left:.1f}s")
        a = best_seq.copy()
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        eta0 = 0.08
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow = 1.0
        beta2_pow = 1.0
        t_adam = 0
        stagn = 0

        while time.time() < deadline:
            t_adam += 1
            beta1_pow *= beta1
            beta2_pow *= beta2

            conv = _full_convolution_fft(a, a)
            max_c = conv.max()
            score = 2.0 * n * max_c
            if score < best_val - 1e-12:
                best_val = score
                best_seq = a.copy()
                stagn = 0
                print(f"[fine‑tune iter {t_adam:6d}] new_best={best_val:.12f}")
            else:
                stagn += 1

            # exact max sub‑gradient
            tol = 1e-14
            active = np.flatnonzero(conv >= max_c - tol)
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0 / active.size
            else:
                w[np.argmax(conv)] = 1.0

            rev_a = a[::-1]
            tmp = _full_convolution_fft(rev_a, w)
            g = 2.0 * tmp[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)
            step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

            a_candidate = a - step
            a = _project_simplex(a_candidate)

            if stagn > 2000:
                noise = rng.normal(scale=0.003, size=n)
                a = _project_simplex(a + noise)
                stagn = 0
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow = 1.0
                beta2_pow = 1.0
                print(f"[fine‑tune perturb] iter {t_adam}")

    # ------------------------------------------------------------------
    # Convert to the form required by evaluate_sequence and report.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[final] could not compute final score (evaluate_sequence missing)")

    return final_seq
