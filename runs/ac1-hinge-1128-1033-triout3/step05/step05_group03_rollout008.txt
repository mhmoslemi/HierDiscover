    v = np.maximum(v, 0)
    if v.sum() == 0:
        return v
    # Use sorting algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _annealing_phase(a_start, n, start_time, deadline, best_a, best_val, target_val=1.503):
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    # define temperature schedule: T_start = 1e-3 maybe, T_end = 1e-7
    while time.time() < deadline:
        # Compute progress
        progress = (time.time() - start_time) / (deadline - start_time)
        T = max(1e-7, (1 - progress) * 1e-3)  # linear decay
        # Choose move like hillclimb but with delta variable
        # choose source index i proportionally to its contribution to current max
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))
        # Choose target j uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # Choose delta: small fraction
        delta = a[i] * np.random.rand() * 0.01  # up to 1% of a[i]
        if delta <= 0:
            continue
        # compute new convolution via incremental update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta
        new_val = 2.0 * n * b_new.max()
        # acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
    return best_a, best_val

"""
Search for a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.

The method consists of three stages:

1.   A quick parametric sweep (uniform, linear, quadratic, sine, exponential)
     to obtain a decent starting point.
2.   Smooth‑max Adam optimisation on the simplex (soft‑max logits) to push the
     maximum of the self‑convolution down.
3.   A targeted hill‑climber that moves tiny amounts of mass from indices that
     contribute most to the current maximal convolution entry to indices that
     contribute little.  After the hill‑climber a short simulated‑annealing
     phase accepts occasional worsening moves to escape local minima.
All phases respect the time budget (default 1000 s) and keep the sequence
non‑negative with unit sum, which makes the objective scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parameteric initialisation (uniform / linear / quadratic / sine / expo)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # linear decreasing
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic decreasing
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # sine shape (half‑wave)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # exponential family
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth approximation of max(conv(a,a))."""
    w = np.log(a0 + 1e-12)                 # soft‑max logits, Σa = 1

    # hyper‑parameters (tuned for speed / stability)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.08, 0.002
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 25000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- soft‑max to obtain a on the simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()                       # Σa = 1, a ≥ 0

        # ----- exact objective for tracking -----
        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- temperature and learning‑rate schedule -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp -----
        b_full = _full_convolution(a)          # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()            # soft max over positions

        # ----- gradient of smooth objective w.r.t. a -----
        # (∂/∂a) 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]    # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # ----- chain rule through the soft‑max -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation of logits
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stagnated
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Fast local search that transfers a small amount of mass between indices."""
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index (biased towards large contributions)
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – 30 % of the time choose a location that has a
        # *small* contribution to the maximal entry, otherwise uniform.
        if np.random.rand() < 0.30:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            # inverse weighting (favor small values)
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta          # self‑term at j
        b_new[2 * i] += delta * delta          # self‑term at i (removed mass)
        b_new[i + j] -= 2.0 * delta * delta    # cross term

        new_val = 2.0 * n * b_new.max()

        # accept only strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – perturb and restart from the best seen
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – short simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Metropolis‑style search that occasionally accepts worsening moves."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # temperature schedule (linear decay)
    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 100_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source index – same bias as hill‑climber
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # simple uniform target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer (≤ 0.5 % of source mass)
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # update convolution incrementally
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # small random shake to possibly leave a plateau
            noise = np.random.normal(scale=0.003, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand entries gives a good balance.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise a – try to reuse a known construction if available,
    # otherwise start from a cheap parametric sweep.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    # --------------------------------------------------------------
    # Phase A – Adam optimisation on a smooth version of max(conv)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + (deadline - start_time) * 0.30
    best_a, best_val = _adam_phase(a0, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # --------------------------------------------------------------
    # Phase B – targeted hill‑climbing on the exact objective
    # --------------------------------------------------------------
    phase_b_deadline = start_time + (deadline - start_time) * 0.80
    best_a = _hillclimb_targeted(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=phase_b_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=TARGET)

    # --------------------------------------------------------------
    # Phase C – short simulated annealing to escape possible plateaus
    # --------------------------------------------------------------
    best_a, best_val = _annealing_phase(best_a, n,
                                        start=phase_b_deadline,
                                        deadline=deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=TARGET)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
