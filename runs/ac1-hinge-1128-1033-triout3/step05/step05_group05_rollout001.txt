"""
propose_candidate
=================

A two‑phase optimiser for ``evaluate_sequence``.

* Phase C – Smoothed‑max gradient descent.
  We minimise a differentiable approximation of the objective
  ``max(a∗a)`` using the soft‑max (log‑sum‑exp) with an annealed
  temperature τ.  The gradient is obtained via one FFT convolution per
  iteration and the iterate is projected onto the probability simplex.
  The step size and τ are varied linearly / exponentially with time.

* Phase D – Targeted hill‑climbing.
  Starting from the best point found in Phase C we repeatedly move a
  small amount of mass from a coefficient that strongly contributes to
  the current maximal convolution entry to a coefficient that contributes
  little.  The convolution is updated in O(n) time, only the changed
  region is recomputed, and improvements are kept.  After a long streak
  without progress a tiny random perturbation is applied.

Both phases respect the time budget (default 1000 s) and the sequence is
returned as a plain list of non‑negative floats (∑a = 1, the scale is
irrelevant for the score).  The implementation uses only NumPy and the
provided ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions (real valued)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()    # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project ``v`` onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase C – Smoothed‑max gradient descent
# ----------------------------------------------------------------------
def _gradient_phase(a: np.ndarray,
                    n: int,
                    deadline: float,
                    start_time: float,
                    best_a: np.ndarray,
                    best_val: float) -> tuple[np.ndarray, float]:
    """
    Perform gradient descent on a smooth approximation of max(a∗a).

    Parameters
    ----------
    a, n      : current vector and its length
    deadline  : absolute time when this phase must stop
    start_time: beginning of the whole optimisation
    best_a, best_val : current best solution

    Returns
    -------
    (best_a, best_val) after the phase.
    """
    # hyper‑parameters (tuned empirically)
    step_start = 0.7
    step_end = 0.001
    tau_start = 5.0          # low temperature → smooth loss
    tau_end = 2000.0         # high temperature → approximates max
    max_jitter_prob = 0.003
    jitter_scale = 0.008
    iter_cnt = 0

    while time.time() < deadline:
        # full self‑convolution
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Grad {iter_cnt:7d}] new best = {best_val:.6f}")

        # elapsed fraction for schedules
        progress = (time.time() - start_time) / (deadline - start_time)
        # linear step decay, exponential temperature increase
        step = step_start * (1.0 - progress) + step_end * progress
        tau = tau_start * (tau_end / tau_start) ** progress

        # smooth‑max weights (soft‑max centred at the current maximum)
        shift = tau * (b - max_b)               # ≤ 0
        weights = np.exp(shift)
        weights_sum = weights.sum()
        if weights_sum == 0.0:                  # numerical safety
            weights = np.full_like(weights, 1.0 / weights.size)
        else:
            weights /= weights_sum                # soft‑max (probability vector)

        # gradient of the smooth approximation
        a_rev = a[::-1]
        conv = _conv_fft(weights, a_rev)       # length 3·n‑2
        grad = 2.0 * conv[n - 1::-1]           # reversed slice gives length n

        # gradient step + projection
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional random jitter to escape plateaus
        if np.random.rand() < max_jitter_prob:
            a += np.random.normal(scale=jitter_scale, size=n)
            a = _project_simplex(a)

        iter_cnt += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _targeted_hillclimb(a_start: np.ndarray,
                        n: int,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> np.ndarray:
    """
    Move small amounts of mass from coefficients that contribute most
    to the current maximal convolution entry to coefficients that
    contribute little.  Updates are performed in O(n) time.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    max_frac = 0.35            # maximal fraction of source mass to move
    IMPROVE_LIMIT = 120_000    # steps without improvement → perturb
    stagnation = 0
    it = 0

    # pre‑compute indexes used for fast slice updates
    two_n_minus_one = 2 * n - 1

    while True:
        now = time.time()
        if now >= deadline:
            break

        # locate the (one) maximal convolution entry
        kmax = int(np.argmax(b))

        # indices that actually appear in b[kmax]
        i_min = max(0, kmax - (n - 1))
        i_max = min(n - 1, kmax)
        idx = np.arange(i_min, i_max + 1)
        # contribution a[i] * a[kmax-i] (ignoring factor 2)
        contrib = a[idx] * a[kmax - idx]

        # source index: either the biggest contributor or a random heavy one
        if np.random.rand() < 0.85:
            src = idx[np.argmax(contrib)]
        else:
            # weighted random (more likely to pick a larger contribution)
            prob = contrib / contrib.sum() if contrib.sum() > 0 else None
            src = np.random.choice(idx, p=prob)

        # target index – prefer a location that barely participates in b[kmax]
        # we pick an index where a is relatively small
        low_candidates = np.where(a < a.mean())[0]
        if low_candidates.size == 0:
            tgt = np.random.randint(n)
        else:
            tgt = np.random.choice(low_candidates)

        if tgt == src:
            continue

        # amount of mass to move
        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental convolution update -----
        # (make a copy because we need the original a for the update)
        b_new = b.copy()

        # linear cross‑terms 2*δ*a shifted by src and tgt
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a

        # pure δ² terms
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept only strict improvements
        if new_val < cur_val - 1e-15:
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            stagnation = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            stagnation += 1

        # periodic random shake‑up after long stagnation
        if stagnation > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            stagnation = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")

        it += 1

    return best_a
