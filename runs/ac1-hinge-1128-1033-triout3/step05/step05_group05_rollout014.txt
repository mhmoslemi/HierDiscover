"""
Optimising a non‑negative step function for `evaluate_sequence`.

The programme consists of three stages:

1. **Initialisation** – reuse a known construction if it exists,
   otherwise start from a random point on the probability simplex.

2. **Smoothed‑max gradient descent** – we minimise a smooth approximation of
   the Ⓛ∞‑norm of the autocorrelation  (the peak of `a*a`).
   The weight vector for the soft‑max is `w_i = exp((b_i‑M)/T)`,
   where `b = a*a` and `M = max(b)`.  A linearly decaying step size
   and an exponentially decaying temperature `T` guide the optimisation.
   After each step the iterate is projected back onto the simplex.

3. **Targeted hill‑climbing** – the exact objective
   `2·n·max(a*a)` is improved by moving a small amount of mass from an
   index that most contributes to the current peak to a “safer’’ index.
   The convolution is updated in `O(n)` time, and a stagnation counter
   triggers occasional random perturbations.

Both phases respect the 1000 s time budget (a short safety margin is kept).
The best sequence seen is finally evaluated with the provided
`evaluate_sequence` wrapper and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0.0] = 0.0
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return 2.0 * n * b.max()


def _gradient_max(a: np.ndarray, T: float) -> np.ndarray:
    """
    Gradient of the temperature‑smoothed max:
        L_T(a) = T·log Σ exp(b_k / T) ,  b = a*a .
    Returns ∂L_T/∂a (vector of length n).
    """
    n = a.shape[0]
    b = _full_convolution(a)                      # length 2·n‑1
    M = b.max()
    # numerically stable soft‑max
    shifted = (b - M) / T
    # avoid overflow/underflow
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.shape[0])

    # gradient = 2·conv(w, a_rev) restricted to the middle n entries
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev)                    # length 3·n‑2
    grad = 2.0 * conv[n - 1 : 2 * n - 1]
    return grad


# ----------------------------------------------------------------------
# Phase‑C – smoothed‑max gradient descent (simple step‑size schedule)
# ----------------------------------------------------------------------
def _smoothed_max_descent(a_start: np.ndarray,
                          budget_s: float,
                          start_time: float,
                          phase_frac: float = 0.55) -> tuple[np.ndarray, float]:
    """
    Runs gradient descent on the smoothed max objective.
    Returns best (a, value) observed before the phase deadline.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_val = float("inf")

    phase_deadline = start_time + phase_frac * budget_s
    # temperature schedule: from T0 down to T_end (exponential)
    T0 = 0.02                     # relatively smooth start
    T_end = 1e-6
    # step‑size schedule (linear decay)
    step_start = 0.30
    step_end = 0.005

    while time.time() < phase_deadline:
        # current objective
        cur_val = _objective(a)
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Grad] new best = {best_val:.6f}")

        # schedule parameters
        elapsed = (time.time() - start_time) / (phase_deadline - start_time)
        T = T0 * (T_end / T0) ** elapsed          # exponential decay
        step = step_start * (1.0 - elapsed) + step_end * elapsed

        # compute gradient of the smoothed max
        grad = _gradient_max(a, T)

        # gradient step + simplex projection
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional tiny jitter to shake out flat regions
        if np.random.rand() < 0.002:
            a += np.random.normal(scale=0.008, size=n)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑D – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _targeted_hillclimb(a_start: np.ndarray,
                        n: int,
                        start_time: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> np.ndarray:
    """
    Fast local search that moves mass away from indices that contribute most
    to the current peak of the autocorrelation.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac = 0.35                # max proportion of source mass to move
    IMPROVE_LIMIT = 120_000        # stagnation trigger
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # recompute peak location and a cheap contribution map
        peak = int(np.argmax(b))
        # contribution of each index to the peak (ordered pairs counted once)
        contrib = np.zeros(n, dtype=np.float64)
        lo = max(0, peak - (n - 1))
        hi = min(n - 1, peak)       # i ≤ peak‑i ⇒ i ≤ peak/2, but we include all
        for i in range(lo, hi + 1):
            j = peak - i
            if j < 0 or j >= n:
                continue
            if i == j:
                contrib[i] = a[i] * a[i]          # diagonal term
            else:
                contrib[i] = a[i] * a[j]           # each unordered pair counted once

        # choose source: index with largest contribution (biased)
        if np.random.rand() < 0.85:
            src = int(np.argmax(contrib))
        else:
            # fall back to the globally largest entry
            src = int(np.argmax(a))

        if a[src] <= 1e-15:
            continue

        # choose a target that is *different* and has small contribution
        # (pick the index with minimal contribution among a random subset)
        cand = np.random.choice(n, size=min(20, n), replace=False)
        cand = cand[cand != src]
        if cand.size == 0:
            continue
        tgt = int(cand[np.argmin(contrib[cand])])

        # amount to move
        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        # ordered cross‑terms: each ordered pair (src, k) and (k, src) changes
        b_new[tgt:tgt + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        # pure δ² terms
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept only strict improvements
        if new_val < cur_val - 1e-12:
            a[src] -= delta
            a[tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill] new best = {best_val:.6f}  (it {it})")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # perturb around the best known point
            a = best_a.copy()
            a += np.random.normal(scale=0.012, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] stagnation – random perturbation (it {it})")
        it += 1

    return best_a
