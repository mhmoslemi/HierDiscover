def fft_convolve_real(x, y):
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx*fy, n=size)
    return conv[:len(x)+len(y)-1]

Thus compute conv_rev = fft_convolve_real(a_rev, w). Slicing conv_rev[n-1: n-1 + n] yields gradient.

def compute_gradient(a, conv, t):

"""
Propose a step‑function by minimising the peak of its auto‑convolution.
The core optimiser is an Adam optimiser on the log‑probabilities (softmax
parameterisation) but the gradient of the objective is *smoothed*:
instead of a sub‑gradient of the exact max we use the softmax of
t·(a∗a) as a weighting of all convolution coefficients.
The temperature t is increased during optimisation, so the algorithm
starts with a gentle, almost‑averaging gradient and ends with a
nearly‑hard max‑gradient.  Multiple random seeds are tried and the
best vector found (with respect to the original evaluation function)
is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, zero‑padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a (real FFT, returns length 2·len(a)-1).
    """
    n = a.shape[0]
    size = _next_pow2(2 * n)                    # enough for convolution
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def fft_convolve_real(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolution of two real vectors using FFT.
    Returns the full linear convolution (len(x)+len(y)-1).
    """
    size = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Simple stable softmax (used for the temperature weighting)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Gradient of the *smoothed* objective  f_t(a) = log-sum-exp(t·(a*a))/t
# ----------------------------------------------------------------------
def grad_a_from_weight(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∂/∂a_i  Σ_k w_k (a*a)[k]   (w is a probability vector, sum w = 1).
    The derivative equals 2·Σ_j a_j·w_{i+j}.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv = fft_convolve_real(a_rev, w)           # length n + (2n‑1) – 1 = 3n‑2
    # slice the part that corresponds to indices i = 0 … n‑1
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on log‑probabilities with a *temperature schedule*
# ----------------------------------------------------------------------
def run_adam_smooth(
    init_a: np.ndarray,
    time_limit: float,
    *,
    t_start: float = 1.0,
    t_factor: float = 1.02,
    t_max: float = 80.0,
    lr0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑probabilities of a (softmax) vector.
    The gradient is computed from a smoothed version of the max‑convolution.
    Returns the best vector seen and its (exact) evaluation score.
    """
    start = time.time()
    n = init_a.shape[0]

    # initialise log‑parameters (avoid log(0))
    eps = 1e-15
    x = np.log(np.maximum(init_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-12

    best_score = np.inf
    best_vec = None
    last_improve = 0
    it = 0
    t = t_start

    while time.time() - start < time_limit - 0.3:
        it += 1
        # ------- current distribution (softmax) -------
        a = softmax(x)

        # ------- exact score (no smoothing) -------
        conv = full_convolution_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = it
            print(f"Iter {it:6d}: new best score = {best_score:.6f}")

        # ------- temperature update -------
        t = min(t * t_factor, t_max)

        # ------- smoothed weighting of convolution entries -------
        # w_k ∝ exp(t * conv_k)
        tconv = t * conv
        w = np.exp(tconv - np.max(tconv))
        w /= np.sum(w)                     # w is a probability vector, length 2n‑1

        # ------- gradient wrt a (smooth) -------
        grad_a = grad_a_from_weight(a, w)   # shape (n,)

        # ------- transform to gradient wrt log‑parameters (mirror‑descent) -------
        dot = np.dot(a, grad_a)             # scalar ⟨a,∇_a f⟩
        grad = a * (grad_a - dot)           # ∂f/∂x_i = a_i (∂f/∂a_i – ⟨a,∇a f⟩)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple decay of learning rate (helps long runs)
        lr = lr0 / (np.sqrt(it) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- re‑heat if stuck for a long while -------
        if it - last_improve > 8000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = it
            print(f"Iter {it:6d}: re‑heat (stalled)")

    return best_vec, best_score
