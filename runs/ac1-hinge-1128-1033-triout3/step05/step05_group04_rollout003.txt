def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    if budget_s > 250:
        n = 8000
    else:
        n = 5000

    rng = np.random.default_rng(seed)

    # Starting a
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] != n:
                # Resample to n by interpolation
                base = np.interp(np.linspace(0,1, n), np.linspace(0,1, base.shape[0]), base)
            a = np.maximum(0.0, base)
            if a.sum() == 0:
                a = rng.dirichlet(np.ones(n))
            else:
                a = a / a.sum()
        except Exception:
            a = rng.dirichlet(np.ones(n))
    else:
        # build shape mixture
        x = np.arange(n, dtype=np.float64)
        # shapes
        # Catalan decreasing approximated
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)
            catalan[k] = cur
        # sine
        sine = np.sin(np.pi * (x + 0.5) / n)
        # semicircle
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
        # gaussian
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
        # triangle
        triangle = np.minimum(x, n - 1 - x) + 1.0

        bases = [catalan, sine, semicircle, gauss, triangle]
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a = sum(w * b for w, b in zip(weights, bases))
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    # define helper evaluate_score
    def score_of(arr):
        conv = _full_convolution_fft(arr, arr)
        return 2.0 * n * conv.max()

    # initial best
    best_a = a.copy()
    best_val = score_of(a)
    print(f"[init] n={n}, init_score={best_val:.12f}")

    # phase parameters
    phase1_end = start_time + 0.62 * budget_s
    phase2_end = start_time + 0.88 * budget_s

    # Adam parameters for phase1
    lr = 0.14
    lr_min = 5e-6
    lr_decay = 0.99997
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    # temperature
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    eps = max(max_conv * 0.6, 1e-12)
    eps_min = max(max_conv * 1e-6, 1e-14)
    eps_decay = 0.9999

    stagn = 0
    it = 0

    # Phase 1: softmax Adam
    while time.time() < phase1_end:
        it += 1

        # conv and max
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_val - 1e-13:
            best_val = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"[phase1 iter {it:6d}] new_best={best_val:.12f} lr={lr:.2e} eps={eps:.2e}")
        else:
            stagn += 1

        # compute softmax weights
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # gradient
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)  # size 3n-2? Actually rev_a length n, w length 2n-1, conv length n+n-1+... anyways
        # we need slice, gradient length n: indices n-1 to n-1 + n
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        beta1_t = beta1 ** it
        beta2_t = beta2 ** it
        m_hat = m / (1.0 - beta1_t)
        v_hat = v / (1.0 - beta2_t)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a + step
        a_new = _project_simplex(a_new)

        a = a_new

        # schedules
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        if stagn > 2500:
            # random perturbation to escape
            noise = rng.normal(scale=0.01, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            lr = max(lr_min, lr * 1.2)
            print(f"[perturb] iter {it}")

        if it % 3000 == 0:
            # occasional renorm for safety
            a = a / a.sum()

    # Phase 2: hard mask gradient descent
    step_size = 5e-5
    step_decay = 0.99995
    step_min = 5e-7

    while time.time() < phase2_end:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_val - 1e-13:
            best_val = cur_score
            best_a = a.copy()
            stagn = 0

        else:

        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # gradient step
        a = a - step_size * grad
        a = _project_simplex(a)

        # schedule step size
        step_size = max(step_min, step_size * step_decay)

        if stagn > 4000:
            # jitter
            a = a + rng.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            stagn = 0
            step_size = max(step_min, step_size * 1.2)
            print(f"[jitter] iter {it}")

    # Phase 3: local random swap search
    cur_a = best_a.copy()  # start from best found
    cur_score = best_val
    total_swaps = 0
    while time.time() < deadline:
        total_swaps += 1
        # pick two indices
        i, j = rng.choice(n, size=2, replace=False)
        # define delta <= cur_a[i]
        max_move = cur_a[i]
        if max_move == 0:
            continue
        # delta random proportion up to, say, 30% of mass at i
        delta = max_move * rng.random() * 0.5  # up to 0.5 fraction
        # produce candidate
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # Evaluate
        conv = _full_convolution_fft(cand, cand)
        cand_max = conv.max()
        cand_score = 2.0 * n * cand_max

        if cand_score < cur_score - 1e-14:
            cur_a = cand
            cur_score = cand_score
            best_a = cand.copy()
            best_val = cand_score
            print(f"[swap {total_swaps}] improved to {best_val:.12f}")

        # occasional re-project to avoid drift (should stay sum=1)
        if total_swaps % 10000 == 0:
            cur_a = cur_a / cur_a.sum()
            # maybe adjust step size or randomization

    print(f"[final] best_score={best_val:.12f}, length={n}")
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_a.tolist()]
    return final_seq

"""
Propose a high‑quality step‑function sequence for the evaluation metric.

The algorithm works in three phases within the 1000 s budget:

1️⃣  **Soft‑max Adam** – a smoothed version of the max‑convolution is
    minimised with Adam while the temperature is annealed.
2️⃣  **Hard‑max polishing** – as the temperature becomes tiny we switch to
    true max‑convolution gradients (binary mask) and use a decaying
    gradient‑descent step.
3️⃣  **Random local swaps** – a hill‑climbing stage that moves a small
    amount of mass from one index to another and accepts the move only
    if it improves the score.  This escapes the smooth‑gradient basin and
    often yields the final few digits of improvement.

The routine adapts the sequence length (8000 when the budget allows,
otherwise 5000) because the FFT size stays at 2¹⁴ for both, keeping the
per‑iteration cost low while giving finer resolution.  A rich mixture of
analytic shapes (Catalan, sine, semicircle, Gaussian, triangle,
inverse‑√) is used for the initial point, and the best sequence found
throughout the run is returned (clamped to the limits required by
`evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for the sizes we use).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative height sequence that (approximately) minimises
    `2 * n * max(conv(a, a)) / (sum(a) ** 2)`.
    The routine respects the supplied time budget (in seconds) and prints
    progress information.  It returns the best sequence found (as a list of
    Python floats) ready for `evaluate_sequence`.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a convenient length.
    # For n up to 8192 the FFT size stays at 2**14, so we can safely use
    # n = 8000 when we have enough time.
    # ------------------------------------------------------------------
    if budget_s > 250:
        n = 8000
    else:
        n = 5000

    # ------------------------------------------------------------------
    # Initialise a – try to use a warm‑start if available.
    # ------------------------------------------------------------------
    a = None
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] != n:
                # Resample linearly to the new length.
                old_x = np.linspace(0.0, 1.0, base.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                base = np.interp(new_x, old_x, base)
            a = np.maximum(0.0, base)
            if a.sum() == 0.0:
                a = rng.dirichlet(np.ones(n))
            else:
                a = a / a.sum()
        except Exception:
            a = None

    if a is None:
        # Build a mixture of several smooth “nice’’ shapes.
        x = np.arange(n, dtype=np.float64)

        # 1. Catalan‑like decreasing sequence.
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)
            catalan[k] = cur

        # 2. Sine.
        sine = np.sin(np.pi * (x + 0.5) / n)

        # 3. Semicircle (half‑ellipse).
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

        # 4. Gaussian.
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

        # 5. Triangle.
        triangle = np.minimum(x, n - 1 - x) + 1.0

        # 6. Inverse‑sqrt (heavy tail).
        inv_sqrt = 1.0 / np.sqrt(x + 1.0)

        # 7. Difference of square‑root (≈ density of √).
        sqrt_diff = np.sqrt(x + 1.0) - np.sqrt(x)

        bases = [catalan, sine, semicircle, gauss, triangle, inv_sqrt, sqrt_diff]
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a = sum(w * b for w, b in zip(weights, bases))
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Helper to compute the (scaled) score; sum(a) is always 1 here.
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Keep the best solution seen so far.

    best_val = _score(a)
    print(f"[init] n={n}, initial_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max Adam (≈60 % of the budget).
    # ------------------------------------------------------------------

    lr = 0.14
    lr_min = 5e-6
    lr_decay = 0.99997
    beta1 = 0.9
    beta2 = 0.999

    m = np.zeros_like(a)

    max_conv = conv.max()

    eps_decay = 0.99990
    stagn = 0
    it = 0

    while time.time() < phase1_end:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_val - 1e-13:
            best_val = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"[phase1 it {it:6d}] new_best={best_val:.12f} lr={lr:.2e} eps={eps:.2e}")
        else:
            stagn += 1

        # soft‑max weights
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # gradient of the weighted sum (cross‑correlation)
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_t = beta1 ** it
        beta2_t = beta2 ** it
        m_hat = m / (1.0 - beta1_t)
        v_hat = v / (1.0 - beta2_t)
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = _project_simplex(a + step)

        # schedules
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional random perturbation to escape plateaus
        if stagn > 2500:
            noise = rng.normal(scale=0.01, size=n)
            a = _project_simplex(a + noise)
            stagn = 0
            lr = max(lr_min, lr * 1.2)
            print(f"[perturb phase1] it {it}")

        # numerical safety
        if it % 3000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max polishing (≈30 % of the budget).
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.88 * budget_s
    step_size = 5e-5
    step_min = 5e-7
    step_decay = 0.99995
    stagn = 0

    while time.time() < phase2_end:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_val - 1e-13:
            best_val = cur_score
            best_a = a.copy()
            stagn = 0
            print(f"[phase2 it {it:6d}] new_best={best_val:.12f} step={step_size:.2e}")
        else:
            stagn += 1

        # binary mask of (almost) maximal entries
        tol = max_conv * 1e-9 + 1e-14

        mask_sum = mask.sum()

            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask / mask_sum

        # gradient for the true max
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)

        # simple gradient step with decaying step‑size
        a = _project_simplex(a - step_size * grad)
        step_size = max(step_min, step_size * step_decay)

        if stagn > 4000:
            a = _project_simplex(a + rng.normal(scale=0.003, size=n))
            stagn = 0
            step_size = max(step_min, step_size * 1.2)
            print(f"[jitter phase2] it {it}")

    # ------------------------------------------------------------------
    # Phase 3 – random local swaps (use remaining time).
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_score = best_val
    swaps = 0
    while time.time() < deadline:

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        if cur_a[i] == 0.0:

        delta = cur_a[i] * rng.random() * 0.5
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate the candidate
        cand_score = _score(cand)
        if cand_score < cur_score - 1e-14:
            cur_a = cand
            cur_score = cand_score
            best_a = cand.copy()
            best_val = cand_score
            print(f"[swap {swaps}] improved to {best_val:.12f}")

        # occasional re‑normalisation (to cure accumulated floating error)
        if swaps % 10000 == 0:
            cur_a = cur_a / cur_a.sum()

    # ------------------------------------------------------------------
    # Return the best sequence found (clamped to the limits required by
    # `evaluate_sequence`).
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_a.tolist()]
    print(f"[final] best_score={best_val:.12f}, length={len(final_seq)}")
    return final_seq
