"""
Propose a step‐function sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear convolution.  The function follows a
two‑phase projected Adam optimisation:

* **Phase 1 – soft‑max Adam**  
  Minimise a smooth surrogate of the max using the log‑sum‑exp (soft‑max)
  of the convolution entries.  The gradient is obtained by a
  cross‑correlation of the current sequence with the soft‑max weights.
  After each Adam step the vector is projected back onto the probability
  simplex.

* **Phase 2 – hard‑max polishing**  
  When the temperature becomes very small the soft‑max collapses to a
  binary mask of the (almost) maximal entries.  A simple sub‑gradient
  descent with a tiny constant step then refines the solution.

The routine respects the supplied time budget (default 1000 s), keeps the
best sequence seen so far and finally returns it after clamping to the
limits required by ``evaluate_sequence``.  If a warm‑start sequence is
available in the global variable ``height_sequence_1`` it is used,
otherwise a fresh “nice’’ shape of length 8000 is built.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()  # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)[:size]
    return conv

# ----------------------------------------------------------------------
def _optimise_for_length(

    budget_s: float,
    rng: np.random.Generator,
    warm_start: np.ndarray | None,

    Run the two‑phase Adam optimiser for a specific length ``n``.
    Returns the best sequence (already normalised to sum 1) and its
    raw score 2*n*max(conv).
    """

    # -------------------- initialise -------------------------------------------------
    if warm_start is not None and warm_start.shape[0] == n:
        a = np.maximum(0.0, warm_start.astype(np.float64))
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = rng.random(n)
            a = a / a.sum()
    else:

        x = np.arange(n, dtype=np.float64)
        sine = np.sin(np.pi * (x + 0.5) / n)
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
        bases = [sine, semicircle, gauss]
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a = sum(w * b for w, b in zip(weights, bases))
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    conv = _full_convolution_fft(a, a)

    best_max = max_conv

    lr = 0.20               # initial learning‑rate
    lr_min = 1e-6
    lr_decay = 0.99996
    beta1 = 0.90
    beta2 = 0.999

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    eps_decay = 0.99992

    it = 0

    # --------------------------------------------------------------------------
    soft_deadline = time.time() + 0.70 * (deadline - time.time())

    while time.time() < soft_deadline:
        it += 1

        # current convolution and score
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if max_conv < best_max - 1e-15:

            best_seq = a.copy()
            best_score = score

        else:
            stagn += 1

        # ----- soft‑max weighting -------------------------------------------------
        # w_i ∝ exp((conv_i - max_conv) / eps)

        w_sum = w.sum()
        if w_sum > 0.0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # ----- gradient (cross‑correlation) --------------------------------------
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)               # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]                  # slice to size n

        a_new = _project_simplex(a_new)

        a = a_new

        # ----- schedules ---------------------------------------------------------
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        # ----- occasional random perturbation ------------------------------------
        if stagn > 3000:

            a = a + noise

            stagn = 0

            print(f"[perturb] iteration {it}")

        if it % 5000 == 0:

    # Phase 2 – hard‑max polishing (≈ remaining 30 % of the budget)
    # --------------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if max_conv < best_max - 1e-15:
            best_max = max_conv
            best_seq = a.copy()
            best_score = score
            stagn = 0
            print(f"[polish {it:6d}] new_best={best_score:.12f}")
        else:
            stagn += 1

        # binary mask of almost‑maximal entries
        tol = max_conv * 1e-9 + 1e-14
        mask = (conv >= max_conv - tol).astype(np.float64)
        w_sum = mask.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask / w_sum

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # tiny constant step for fine polishing
        step = -2e-5 * grad
        a = a + step
        a = _project_simplex(a)

        if stagn > 5000:
            a = a + rng.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            stagn = 0
            print(f"[jitter] iteration {it}")

    # final best score for this length
    print(f"[result] best_score_for_n={best_score:.12f}, length={n}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a non‑negative height sequence (list of floats) that
    approximately minimises the evaluation function.  The routine works
    within the supplied time budget and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine a set of candidate lengths; larger lengths give the optimiser
    # more freedom.  We try a few different sizes and keep the best result.
    # ----------------------------------------------------------------------
    candidate_lengths = [4000, 6000, 8000]          # chosen to fit comfortably in 1 GB RAM
    best_global_score = np.inf
    best_global_seq = None

    # possible warm‑start
    warm_start = None
    if 'height_sequence_1' in globals():
        try:
            warm_start = np.asarray(height_sequence_1, dtype=np.float64)
        except Exception:
            warm_start = None

    # allocate time roughly equally among the candidates
    per_len_budget = (deadline - time.time()) / len(candidate_lengths)

    for n in candidate_lengths:
        if time.time() + per_len_budget > deadline:
            break
        seq, score = _optimise_for_length(
            n=n,
            budget_s=per_len_budget,
            rng=rng,
            warm_start=warm_start if warm_start is not None else None,
        )
        if score < best_global_score:
            best_global_score = score
            best_global_seq = seq.copy()
            print(f"[global] new overall best score={best_global_score:.12f} with n={n}")

    # ----------------------------------------------------------------------
    # Convert to the format required by `evaluate_sequence` and report.
    # ----------------------------------------------------------------------
    if best_global_seq is None:
        # fallback – return a simple uniform sequence of length 1000
        n = 1000
        best_global_seq = np.full(n, 1.0 / n)

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_global_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
