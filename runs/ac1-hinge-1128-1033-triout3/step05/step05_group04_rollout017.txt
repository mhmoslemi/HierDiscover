"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using several short Adam‑based runs from diverse starting points.
Each run consists of

* 1️⃣ a “soft‑max Adam’’ phase that replaces the non‑smooth max by a
  temperature‑controlled soft‑max, with a projected Adam update onto the
  probability simplex,
* 2️⃣ a “hard‑max polishing’’ phase that works with a binary mask of the
  currently maximal convolution entries,
* 3️⃣ an occasional random jitter to escape plateaus.

Multiple restarts are performed until the time budget (≈1000 s) is
exhausted; the best sequence seen over all restarts is returned.
The code is deliberately self‑contained, uses only numpy and the
already‑imported evaluate_sequence, and prints occasional progress
information.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast and numerically stable).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main routine – runs a single optimisation phase (soft‑max Adam + polishing).
# ----------------------------------------------------------------------
def _optimize_one_run(a_init: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Return (best_seq, best_val) obtained before `deadline`."""
    a = a_init.copy()
    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv          # sum(a)==1 ⇒ denominator = 1
    best_seq = a.copy()
    it = 0
    stagn = 0
    # --------------------------------------------------------------
    # Adam hyper‑parameters (slightly tweaked from the reference)
    # --------------------------------------------------------------
    lr = 0.10               # initial learning rate
    lr_min = 1e-6
    lr_decay = 0.99997
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature
    eps = max(max_conv * 0.6, 1e-12)
    eps_min = max(max_conv * 1e-6, 1e-12)
    eps_decay = 0.99994

    # reserve roughly two‑thirds of the remaining budget for the smooth phase
    soft_deadline = time.time() + 0.66 * (deadline - time.time())

    # --------------------------------------------------------------
    # Phase 1 – soft‑max Adam
    # --------------------------------------------------------------
    while time.time() < soft_deadline:
        it += 1

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_seq = a.copy()
            stagn = 0
            print(f"[run 1] it={it:6d}  new_best={best_val:.12f}  lr={lr:.2e}  eps={eps:.2e}")
        else:
            stagn += 1

        # soft‑max weighting of convolution entries
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0.0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # gradient of the weighted sum (cross‑correlation)
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        beta1_t = beta1 ** it
        beta2_t = beta2 ** it
        m_hat = m / (1.0 - beta1_t)
        v_hat = v / (1.0 - beta2_t)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a + step
        a = _project_simplex(a_new)

        # schedules
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional jitter to leave a plateau
        if stagn > 2500:
            noise = rng.normal(scale=0.005, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            lr = max(lr_min, lr * 1.3)
            print(f"[run 1] jitter at it={it}")

    # --------------------------------------------------------------
    # Phase 2 – hard‑max polishing (binary mask of truly maximal entries)
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-13:
            best_val = cur_val
            best_seq = a.copy()
            stagn = 0
            print(f"[run 2] it={it:6d}  new_best={best_val:.12f}")
        else:
            stagn += 1

        # binary mask of (almost) maximal entries
        tol = max_conv * 1e-9 + 1e-14
        mask = (conv >= max_conv - tol).astype(np.float64)

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, mask)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # a small constant step works well in this regime
        step = - 2e-5 * grad
        a = a + step
        a = _project_simplex(a)

        if stagn > 4000:
            a = a + rng.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            stagn = 0
            print(f"[run 2] jitter at it={it}")

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative height sequence that (approximately) minimises
    `evaluate_sequence`.  The routine respects the supplied time budget (in
    seconds) and returns the best sequence obtained.
    """
    deadline = time.time() + budget_s - 5.0          # safety margin
    global rng
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Fixed problem size – 5000 works well and keeps evaluate() fast.
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # Helper to construct a diverse initial distribution.
    # --------------------------------------------------------------
    def _make_initial():
        # if a warm‑start exists, use it (with a tiny jitter)
        if 'height_sequence_1' in globals():
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.maximum(0.0, base)
            if base.sum() == 0.0:
                base = rng.random(base.shape[0])
            base = base / base.sum()
            # add a little noise to escape the previous basin
            base = base + rng.normal(scale=0.003, size=base.shape[0])
            base = _project_simplex(base)
            return base
        # otherwise build a mixed “nice’’ shape, including an “inverse‑harmonic’’
        # component that tends to push mass towards the borders.
        x = np.arange(n, dtype=np.float64)

        # inverse‑harmonic – larger values at the edges
        inv_harm = 1.0 / (x + 1.0) + 1.0 / (n - x)
        inv_harm = np.maximum(inv_harm, 0.0)

        # sine – a smooth interior bump
        sine = np.sin(np.pi * (x + 0.5) / n)

        # triangular – rises linearly to the centre
        triangle = np.minimum(x, n - 1 - x) + 1.0

        # Gaussian – another classic bell
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

        bases = [inv_harm, sine, triangle, gauss]
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a = sum(w * (b / b.sum()) for w, b in zip(weights, bases))
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        return a

    # --------------------------------------------------------------
    # Run several short independent optimisations (re‑starts).
    # --------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float('inf')
    restart = 0

    # allocate time slices dynamically – each restart gets at least 80 s
    min_per_restart = 80.0

    while time.time() < deadline - min_per_restart:
        remaining = deadline - time.time()
        # allocate a slice: either the remaining time divided by a few restarts
        # or the minimum we guarantee to each run.
        slice_budget = max(min_per_restart, remaining / 4.0)

        # initialise
        a0 = _make_initial()
        # run the optimizer on this slice
        best_seq, best_val = _optimize_one_run(a0, n, time.time() + slice_budget)

        if best_val < best_overall_val:
            best_overall_val = best_val
            best_overall_seq = best_seq.copy()
            print(f"[global] restart={restart}  new_global_best={best_overall_val:.12f}")

        restart += 1

    # ------------------------------------------------------------------
    # If for some reason we never entered the loop (very tight budget),
    # fall back to a simple uniform sequence.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = np.ones(n) / n
        best_overall_val = 2.0 * n * (1.0 / n)   # = 2.0
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Convert to the format required by `evaluate_sequence` and report.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_overall_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
