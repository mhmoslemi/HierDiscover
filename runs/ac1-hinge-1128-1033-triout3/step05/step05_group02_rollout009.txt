"""
Efficient multi‑scale, multi‑restart optimizer for the
step‑function evaluation problem.

The algorithm consists of three phases:

1. **Fast L‑BFGS on a smooth max‑convolution surrogate**.
   The surrogate is  max(conv)  +  ε·log‑sum‑exp((conv‑max)/ε)
   where conv = a*a, a = soft‑max(w).  A geometric ε‑schedule
   (5e‑3 → 1e‑9) is used, each stage solved with L‑BFGS‑B.
   Several random restarts explore the non‑convex landscape.

2. **Heuristic re‑ordering** – an anti‑symmetric (large‑small)
   permutation that tends to decorrelate the sequence and often
   reduces the peak of its self‑convolution.

3. **Targeted hill‑climbing refinement** – random swaps,
   small mass transfers, and occasional mixing with the uniform
   distribution.  The objective is evaluated only when a cheap
   surrogate (2·n·max(conv)/sum(a)²) indicates a possible improvement,
   which saves many expensive calls to ``evaluate_sequence``.

The routine respects the 1000 s budget (a tiny safety margin is kept)
and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS wrapper (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter
_GLOBAL_N = None     # current sequence length


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of ``x`` with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # tiny negative numbers caused by floating‑point noise are clipped
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _softmax(w: np.ndarray) -> np.ndarray:
    """Stable soft‑max that enforces a_i ≥ 0 and Σa_i = 1."""
    ws = w - np.max(w)                      # improve numeric stability
    e = np.exp(ws)
    return e / e.sum()


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε = max(conv) + eps·log‑sum‑exp((conv‑max)/eps)
    together with its gradient with respect to the soft‑max parameters w.

    The returned objective is multiplied by ``2·n`` so that it matches the
    scale of ``evaluate_sequence``; the scaling does not affect the
    optimisation path.
    """
    a = _softmax(w)                     # a lies on the probability simplex
    conv = _full_convolution(a)         # length 2·n‑1
    max_conv = conv.max()

    # smoothed max (log‑sum‑exp)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * (max_conv + eps * log_sum)

    # weights for the smooth max (probability vector over shifts)
    # weight_k = exp((conv_k‑max)/eps) / Σ exp(...)
    weights = np.exp(shifted - log_sum)      # already normalised

    # gradient w.r.t. a:
    #   ∂/∂a_i  Σ_k p_k * conv_k  = 2 Σ_k p_k * a_{k‑i}
    # efficiently computed via convolution with the reversed weight vector
    p_rev = weights[::-1]                     # reverse the weight vector
    conv_ap = _conv_fft(a, p_rev)            # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]           # middle n entries
    grad_a = 2.0 * seg[::-1]                 # size n

    # chain rule through the soft‑max (Jacobian = diag(a) – a·aᵀ)
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce a permutation where the largest entries are paired with the
    smallest ones when the sequence is mirrored.  This anti‑symmetric
    ordering often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large on the left
        new_a[n - 1 - i] = sorted_vals[i]          # small on the right

    if n % 2:
        new_a[half] = sorted_vals[half]            # middle element unchanged
    return new_a


def _estimate_score(a: np.ndarray) -> float:
    """
    Quick estimate of the evaluation score using the exact formula.
    ``a`` need not be normalised; the function returns the same value
    as ``evaluate_sequence`` up to the clamping of extreme values.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:            # matches the safeguard in evaluate_sequence
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Local search that mixes three elementary moves:

    * a pure swap of two entries,
    * a small mass transfer from a donor to a receiver,
    * a convex combination with the uniform distribution.

    The candidate is accepted only if it strictly improves the
    surrogate score (which coincides with the true score for
    non‑clipped values).  The loop stops when the time budget expires.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    uniform = np.full(n, 1.0 / n)

    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.15:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move_type < 0.85:                   # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()   # up to 5 % of donor
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        else:                                     # mix with uniform
            t = rng.choice([0.01, 0.03, 0.07])       # small mixing coefficients
            new_a = (1.0 - t) * cur_a + t * uniform

        # keep the sequence feasible (non‑negative)
        new_a = np.maximum(new_a, 0.0)

        # quick surrogate evaluation – avoids an expensive call when not needed
        est = _estimate_score(new_a)
        if est < cur_val - 1e-12:
            # final precise check (protects against clipping effects)
            try:
                true_val = evaluate_sequence(new_a.tolist())
            except Exception:
                continue
            if true_val < cur_val - 1e-12:
                cur_a = new_a
                cur_val = true_val
                print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
