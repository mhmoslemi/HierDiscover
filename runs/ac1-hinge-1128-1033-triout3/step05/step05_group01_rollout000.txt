"""
Propose a non‑negative step function (list of heights) that minimises
evaluate_sequence().  The algorithm consists of four phases:

1.  Initialise a uniform (or interpolated) distribution of length n
    (4000–8000).  All vectors are projected onto the probability simplex
    (non‑negative, ℓ₁‑norm = 1) because the objective is scale invariant.

2.  Smooth optimisation:  replace the non‑smooth max‑convolution by a
    log‑sum‑exp surrogate Lₑ(a).  The surrogate is minimised with L‑BFGS‑B
    while the smoothing parameter ε is decreased (continuation).  The
    gradient is obtained analytically using FFTs.

3.  Sub‑gradient descent on the true max‑convolution.  At each iteration
    the lag k attaining the current maximum is identified; the sub‑gradient
    of convₖ w.r.t. a is 2·a shifted by k.  A step is taken opposite to the
    sub‑gradient and the resulting vector is projected back onto the simplex.
    The step size decays over time.

4.  Guided hill‑climbing / simulated‑annealing style refinement.  Mass is
    transferred away from the pair (i,j) that contributes most to the
    offending lag, and placed at two other indices that do not recreate the
    same lag.  Moves that improve the exact objective are kept; otherwise
    the step size is shrunk.

The whole routine respects the supplied time budget (default 1000 s) and
continually prints the best evaluation seen so far.  The final list of
floats is returned.
"""

    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len(x)-1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # remove tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate objective L_ε(a) and its gradient w.r.t. the soft‑max
# parameters w (so that a = softmax(w) and Σ a = 1).
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int) -> tuple[float, np.ndarray]:
    """
    Compute L_ε(a) and ∇_w L_ε(a) for a = softmax(w).  The formulation follows
    the derivation used in the previous state‑of‑the‑art code.
    """
    # ----- soft‑max (feasible a, Σ a = 1) -----
    w_shift = w - np.max(w)                 # improve stability
    a = np.exp(w_shift)
    a = a / a.sum()

    # ----- full autocorrelation a * a -----
    conv = _full_convolution(a)            # length 2·n‑1
    max_conv = conv.max()

    # ----- log‑sum‑exp smoothing -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)           # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    # p_k = exp((conv_k‑max)/ε) / Σ exp(...)
    weights = np.exp(shifted)
    p = weights / weights.sum()            # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]                        # reversed p
    conv_ap = _conv_fft(a, p_rev)          # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]        # central n entries
    grad_a = 4.0 * n * seg[::-1]           # shape (n,)

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)            # ∂L/∂w
    return float(L), grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σ x = 1}
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not cond.any():
        # all entries are non‑positive -> return the uniform vertex
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent directly on max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(a0: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a simple sub‑gradient descent on the exact objective.
    At each iteration the lag attaining the current maximum is selected,
    a step opposite to its sub‑gradient is taken and the result is projected
    onto the simplex.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_val = best_val
    step = 0.1                     # initial step size (in ℓ₁‑units)
    min_step = 1e-6

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        # all lags attaining the maximum (within a tiny tolerance)
        ks = np.flatnonzero(conv >= max_b - 1e-14)
        k = int(rng.choice(ks))

        # sub‑gradient of conv_k (2·a shifted by k)
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        grad[i_range] = 2.0 * a[k - i_range]

        # gradient step and projection
        a_new = a - step * grad
        a_new = _simplex_projection(a_new)

        # evaluate exact objective
        try:
            val = evaluate_sequence(a_new.tolist())
        except Exception:
            # unhealthy candidate – discard
            step = max(step * 0.99, min_step)
            continue

        if val < cur_val:
            a = a_new
            cur_val = val
            print(f"[subgrad] new best = {cur_val:.6f}")
            step = max(step * 0.9, min_step)       # accelerate fine‑tuning
        else:
            # no improvement – shrink step
            step = max(step * 0.99, min_step)

        # stop if step becomes tiny
        if step <= min_step:
            break

    return a, cur_val


# ----------------------------------------------------------------------
# Guided hill‑climbing (mass‑transfer) refinement
# ----------------------------------------------------------------------


def _guided_hill_climb(a0: np.ndarray,
                       best_val: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill climbing that transfers a small fraction of mass away from the
    pair most responsible for the current maximal convolution value.
    """
    n = a0.shape[0]
    a = a0.copy()
    cur_val = best_val
    step_frac = 0.08          # initial fraction of the involved masses

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        # choose a lag attaining the maximum (random tie‑break)
        ks = np.flatnonzero(conv >= max_b - 1e-15)
        k = int(rng.choice(ks))

        # find the pair (i,j) with the largest contribution a_i * a_j
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two destination indices that do NOT recreate lag k
        p, q = rng.integers(0, n, size=2)
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # compute transferable mass
        if i_max != j_max:
            delta = min(a[i_max], a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            a_candidate = a.copy()
            a_candidate[i_max] -= delta
            a_candidate[j_max] -= delta
            a_candidate[p] += delta
            a_candidate[q] += delta
        else:   # i_max == j_max (the diagonal element)
            delta = a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            a_candidate = a.copy()
            a_candidate[i_max] -= delta
            a_candidate[p] += delta

        if (a_candidate < 0).any():
            continue

        # evaluate
        try:
            val = evaluate_sequence(a_candidate.tolist())
        except Exception:
            continue

        if val < cur_val:
            a = a_candidate
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.97, 1e-4)   # slightly tighten
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence().  The routine follows the four phases described in
    the module docstring and respects the given time budget.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 gives more freedom while still being quick.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a from a previous construction if available, otherwise uniform.
    # ------------------------------------------------------------------
    old = globals().get("height_sequence_1", None)
    if isinstance(old, (list, tuple, np.ndarray)):
        old_arr = np.asarray(old, dtype=float)
        if old_arr.ndim == 1 and old_arr.size > 0:
            if old_arr.shape[0] != n:
                # linear interpolation to the required length
                old_x = np.linspace(0.0, 1.0, old_arr.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, old_arr)
            else:
                a0 = old_arr.copy()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # enforce feasibility (non‑negative, ℓ₁‑norm = 1)
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()

    # Initial evaluation
    try:
        best_val = evaluate_sequence(a0.tolist())
    except Exception:
        best_val = float("inf")
    best_a = a0.copy()
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth optimisation via L‑BFGS‑B on the log‑sum‑exp surrogate.
    # ------------------------------------------------------------------
    # soft‑max parameters
    w = np.log(best_a + 1e-12)

    # decreasing ε schedule (log‑space)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-9), num=18)

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        # L‑BFGS‑B
        res = minimize(fun_and_grad,
                       w,
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": 250,
                                "ftol": 1e-12,
                                "gtol": 1e-12,
                                "disp": False})

        w = res.x

        # evaluate the exact objective
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()
        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[smooth] eps={eps:.1e} -> new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – a few random restarts to escape possible local minima.
    # ------------------------------------------------------------------
    restarts = 0
    while time.time() < deadline - 5.0 and restarts < 3:
        restarts += 1
        # perturb the best soft‑max parameters
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.8, size=n)

        # only the finer part of the ε‑schedule
        for eps in eps_schedule[-8:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": 150,
                                    "ftol": 1e-12,
                                    "gtol": 1e-12,
                                    "disp": False})
            w = res.x

        # evaluate candidate
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a = a / a.sum()
        cur_val = evaluate_sequence(a.tolist())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restarts}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient descent directly on the true max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _subgradient_descent(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Phase 4 – guided hill‑climbing (mass transfers) for final polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
