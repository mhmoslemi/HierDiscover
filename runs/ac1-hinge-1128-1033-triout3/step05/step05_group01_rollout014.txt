"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise a non‑negative vector with unit ℓ₁‑norm.
   If a previous construction ``height_sequence_1`` exists it is interpolated
   to the working size; otherwise a uniform distribution is used.

2. **Smooth optimisation** – minimise a log‑sum‑exp surrogate of the
   maximum convolution using L‑BFGS‑B.  The surrogate is smoother than
   the original objective and a continuation schedule (ε from 5e‑3 down to
   1e‑12) is applied.  The variables are the soft‑max parameters w,
   guaranteeing feasibility automatically.

3. **Simulated‑annealing hill‑climbing** – starting from the best point
   found in step 2, repeatedly move a small amount of mass from the pair
   of indices that contributes most to the current maximal convolution lag
   to two low‑mass positions.  Moves are accepted if they improve the true
   objective, otherwise they may be accepted with a probability that
   decreases with a linear temperature schedule.  The step size adapts
   gradually, providing fine‑grained exploration until the time budget
   expires.

All heavy work (FFT based convolutions, gradient of the surrogate, etc.)
is performed with NumPy; the routine respects a 1000 s budget and prints
progress information.  The final list of heights is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
from typing import List

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len–1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # remove tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)‑1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smooth surrogate L_ε of the true objective plus gradient w.r.t. the
    soft‑max parameters w.
    """
    # soft‑max (guaranteed positive, ℓ₁‑norm = 1)
    w_shift = w - np.max(w)            # improve numerical stability
    a = np.exp(w_shift)
    a = a / a.sum()

    # full auto‑convolution a*a
    conv = _full_convolution(a)        # length 2·n‑1
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)       # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()        # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)      # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]    # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _objective_fast(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/(sum(a)²) – fast version using FFT."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _full_convolution(a)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_refine(a: np.ndarray,
                      cur_val: float,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hill‑climb with a simple simulated‑annealing schedule.
    Mass is transferred from the most offending pair to two low‑mass
    positions.  The function returns the best vector found before timeout.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    # temperature and step‑size schedules
    T0 = 0.02
    T_end = 1e-6
    step_frac = 0.08
    min_step = 5e-5

    while time.time() < deadline:
        # current convolution statistics
        conv = _full_convolution(a)
        max_b = conv.max()
        # pick one of the maximal lags (random tie‑break)
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k‑i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two destination indices from the low‑mass tail
        low = np.where(a < a.mean())[0]
        if low.size >= 2:
            p, q = rng.choice(low, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the same lag
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # decide the transferred mass
        if i_max != j_max:
            allow = min(a[i_max], a[j_max])
            delta = min(step_frac * allow, allow)
            if delta <= 0:
                continue
            new_a = a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            allow = a[i_max]
            delta = min(step_frac * allow, allow)
            if delta <= 0:
                continue
            new_a = a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # cheap evaluation
        new_val = _objective_fast(new_a)

        # linear temperature schedule
        elapsed = time.time() - (deadline - (deadline - time.time()))
        T = max(T_end, T0 * (1.0 - (time.time() - (deadline - (deadline - time.time()))) /
                             (deadline - (deadline - time.time()))))

        # acceptance rule
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[anneal] new best = {best_val:.6f}")
            # on success enlarge step a touch
            step_frac = min(step_frac * 1.02, 0.2)
        else:
            step_frac = max(step_frac * 0.995, min_step)

    return best_a, best_val

    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] exact score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
