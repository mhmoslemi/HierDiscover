Algorithm
---------

   ``max(b)`` where ``b = a∗a``.  For a temperature γ the soft‑max
   weights ``w_k = exp(γ·(b_k‑max(b))) / Σexp`` give a smooth
   approximation of the indicator of the maximal entry.

        ∇ = 2·conv_rev(a, w)   (restricted to the n central entries),

   The temperature γ is increased linearly from 0.5 to 10 so the
   optimisation gradually focuses on the true maximum.
   After each step we **project** onto the probability simplex

import time

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w

    Moves a fraction of the largest entry to the smallest entry,
    keeping Σa fixed.  Uses O(1) update of the full convolution.
    """
    a = a_start.copy()
    b = _full_convolution(a)                # length 2·n‑1
    cur_val = 2.0 * n * b.max()
    iteration = 0
    no_improve = 0

        if now >= deadline:

        else:

            continue

        j = int(np.argmin(a))

        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)      # enforce Σa = 1
    best_a = a.copy()
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase C – soft‑max (approximate max) gradient descent.
    # ------------------------------------------------------------------
    phase_c_deadline = start_time + 0.70 * budget_s   # 70 % of total budget
    avg_target = 1.0 / (2 * n - 1)                    # theoretical average of b
    max_step = 0.30
    min_step = 1e-4

    while time.time() < phase_c_deadline:
        # full autoconvolution
        b = _full_convolution(a)

        # exact objective (Σa = 1 ⇒ score = 2·n·max(b))
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Grad] new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # (i) soft‑max weights over b
        # ------------------------------------------------------------------
        max_b = b.max()
        # temperature schedule: 0.5 → 10
        progress_c = (time.time() - start_time) / (phase_c_deadline - start_time)
        gamma = 0.5 + 9.5 * progress_c

        # stable soft‑max
        w = np.exp(gamma * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # gradient of Σ w_k·b_k  w.r.t a
        grad = 2.0 * _conv_fft(a[::-1], w)[n - 1 : 2 * n - 1]

        # ------------------------------------------------------------------
        # (ii) optional L2 regularisation (helps global flatness)
        # ------------------------------------------------------------------
        r = b - avg_target
        conv1 = _conv_fft(r, a)
        conv2 = _conv_fft(r, a[::-1])
        grad_l2 = 2.0 * (conv1[n - 1 : 2 * n - 1] + conv2[n - 1 : 2 * n - 1])

        # blend the two gradients (soft‑max dominant, L2 a small stabiliser)
        grad = 0.85 * grad + 0.15 * grad_l2

        # ------------------------------------------------------------------
        # Gradient step + projection onto simplex
        # ------------------------------------------------------------------
        step = max_step * (1.0 - progress_c) + min_step * progress_c
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # occasional tiny Gaussian jitter to avoid tiny plateaus
        if np.random.rand() < 0.003:
            a += np.random.normal(scale=0.004, size=n)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Phase D – deterministic hill‑climbing on the exact objective.
    # ------------------------------------------------------------------
    best_a = _hillclimb_phase(best_a, n, deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
