def proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np, time, math

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(x):
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

def _project_simplex(v):
    # projection onto simplex {x >=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    cur_a = a.copy()

"""Step‑function optimiser.

The programme seeks a non‑negative sequence a (∑a = 1) that minimises

    E(a) = 2·n·max(a∗a)               (the evaluation function of the task)

The algorithm consists of three parts:

1. **Initialisation** – use the previously discovered best sequence (if
   available) and adapt its length; otherwise start from a uniform vector.
2. **Projected sub‑gradient descent** on the exact, non‑smooth objective.
   For the current maximal convolution lag k we have
   ∇ max(a∗a) = (a_{k−i})_i .  After a step we project back onto the
   probability simplex (non‑negative, ℓ₁‑norm = 1).  A simple
   1/√t stepsize schedule is used.
3. **Guided hill‑climbing** – a greedy local‑search that transfers a
   fraction of mass away from the pair (i,j) that contributes most to the
   current peak and puts it on two low‑mass positions.  Only moves that
   improve the true evaluation are kept.

All work respects the 1000 s time budget; the best sequence found before the
deadline is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # remove tiny numerical negatives
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (x ≥ 0, sum(x) = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, ∑x = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where u_i > (cumulative sum)/i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (exact objective)
# ----------------------------------------------------------------------
def _subgradient_optimize(a_start: np.ndarray,
                          max_iters: int,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(a∗a) by sub‑gradient steps together with
    projection onto the simplex.
    Returns the best vector seen and its exact objective value.
    """
    a = a_start.copy()
    n = a.shape[0]
    # initial exact value (no need to call evaluate_sequence)
    conv = _full_convolution(a)
    max_b = conv.max()
    best_val = 2.0 * n * max_b / (a.sum() ** 2)
    best_a = a.copy()

    step0 = 0.4  # initial stepsize – tuned empirically

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current autocorrelation and the (one of) maximising lag(s)
        conv = _full_convolution(a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-16)
        k = int(rng.choice(ks))

        # sub‑gradient of max_b w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.intp)
        g[idx] = a[k - idx]

        # stepsize decreasing like 1/√t
        step = step0 / np.sqrt(it)

        # tentative move and projection onto the simplex
        a_new = _project_simplex(a - step * g)

        # evaluate the exact objective of the candidate
        conv_new = _full_convolution(a_new)
        max_b_new = conv_new.max()
        val_new = 2.0 * n * max_b_new / (a_new.sum() ** 2)

        # accept if it improves the best ever value
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            print(f"[subgrad] iter {it:5d} new best = {best_val:.6f}")
        else:
            # accept a move that improves the *current* point (optional)
            cur_val = 2.0 * n * max_b / (a.sum() ** 2)
            if val_new < cur_val:
                a = a_new

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy local search: move a small fraction of mass away from the pair
    that contributes most to the current convolution peak and put it on
    two low‑mass positions.  Only moves that strictly improve the exact
    evaluation are kept.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05  # initial fraction of mass to transfer

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # pairs (i, j) that realise the lag k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two destination indices that are not creating the same lag
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max (a self‑pair)
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # evaluate with the provided function (exact)
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a step‑function for the given evaluation budget.
    Returns a list of non‑negative heights (the best sequence found).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # keep a small safety margin

    # ---- obtain a sensible initial vector -----------------------------
    try:
        # reused construction from a previous run (if available)
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # we work with a length of 8000 – enough degrees of freedom
        target_n = max(8000, old.shape[0] * 2)
        if old.shape[0] != target_n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, target_n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        target_n = 8000
        a0 = np.full(target_n, 1.0 / target_n, dtype=float)

    # enforce feasibility (non‑negative, ℓ₁‑norm = 1)
    a0 = np.maximum(a0, 0.0)
    if a0.sum() <= 0:
        a0 = np.full(target_n, 1.0 / target_n, dtype=float)
    else:
        a0 = a0 / a0.sum()

    # initial score
    try:
        best_val = evaluate_sequence(a0.tolist())
    except Exception:
        best_val = float("inf")
    best_a = a0.copy()
    print(f"[init] n={target_n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – projected sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    max_iters = 20000
    best_a, best_val = _subgradient_optimize(best_a, max_iters, deadline, rng)

    # ------------------------------------------------------------------
    # Stage 2 – a few random restarts to escape possible local minima
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0 and restart < 3:
        restart += 1
        # perturb the current best vector slightly
        perturb = rng.normal(scale=0.02, size=target_n)
        a_start = best_a + perturb
        a_start = np.clip(a_start, 0.0, None)
        if a_start.sum() <= 0:
            a_start = np.full(target_n, 1.0 / target_n, dtype=float)
        else:
            a_start = a_start / a_start.sum()
        best_a, best_val = _subgradient_optimize(a_start,
                                                 max_iters // 4,
                                                 deadline,
                                                 rng)
        print(f"[restart {restart}] best after restart = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – guided hill‑climbing refinement (exact evaluation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = best_val

    return best_a.tolist()
