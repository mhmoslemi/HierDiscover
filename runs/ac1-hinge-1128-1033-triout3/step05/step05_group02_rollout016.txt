def project_simplex(v):

    """Projects vector v onto the probability simplex (nonnegative, sum=1)."""

    # reference to Duchi algorithm.

   u = np.sort(v)[::-1]

   cssv = np.cumsum(u) - 1

   ind = np.arange(len(v)) + 1

   cond = u - cssv / ind > 0

   rho = ind[cond][-1]

   theta = cssv[cond][-1] / rho

   w = np.maximum(v - theta, 0)

   return w

"""
Propose a step‑function (non‑negative list of floats) that minimizes the
evaluation function

    2·n·max(conv(a,a)) / (sum(a)**2)

where conv(a,a) is the full linear convolution of a with itself.
The algorithm is a *projected sub‑gradient descent* directly on the
max‑convolution objective, combined with a few heuristic refinements:

* Multiple random restarts (including a warm start from a previously
  discovered sequence if available).
* In each restart a simple sub‑gradient of the maximal convolution entry
  is computed analytically and an additive step is taken.
* After each step the vector is projected onto the probability simplex
  (a_i ≥ 0, Σ a_i = 1) – this guarantees feasibility for ``evaluate_sequence``.
* The sub‑gradient loop runs until the time budget is almost exhausted.
* Finally the best vector is post‑processed by the anti‑symmetric reorder
  used in the reference solution and a fast hill‑climbing refinement
  (mass transfers + swaps).

The implementation uses FFT‑based convolution for speed (O(n log n) per
iteration) and a deterministic simplex projection (O(n log n)).  The
method works for the default length n = 4000 but can be changed without
modifying the rest of the code.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``x`` with itself.
    Implements a zero‑padded FFT for O(n log n) speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # clip tiny negatives that may appear due to rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` and ``y`` (FFT implementation)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Reorder ``a`` so that large values are placed opposite small values.
    This heuristic often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]   # large values on the left
        new_a[n - 1 - i] = sorted_vals[i]   # small values on the right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Very cheap local search: randomly transfer a small amount of mass
    between two positions (or swap two entries) and keep changes only
    when the true evaluation improves.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        # 20 % chance of a pure swap
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            if delta <= 0.0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:      # pragma: no cover
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x ≥ 0, Σ x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]            # sort in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Compute the exact value returned by ``evaluate_sequence`` for a
    probability vector ``a`` (sum is 1).  This avoids the slower Python
    wrapper during the inner optimisation loop.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.
    The method uses many random restarts, a projected sub‑gradient descent on
    the maximum convolution entry, and a final refinement stage.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety buffer

    # problem size – can be altered, but 4000 works well within the budget
    n = 4000

    # --------------------------------------------------------------
    # Warm start: use a previously found sequence if it exists
    # --------------------------------------------------------------
    init_a = None
    if 'height_sequence_1' in globals():
        try:
            old = np.asarray(height_sequence_1, dtype=float)
            if old.shape[0] != n:
                # linear interpolation to the required length
                old_x = np.linspace(0.0, 1.0, old.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                init_a = np.interp(new_x, old_x, old)
            else:
                init_a = old.copy()
        except Exception:   # pragma: no cover
            init_a = None

    best_a = None
    best_val = np.inf
    restart = 0

    # --------------------------------------------------------------
    # Main multi‑restart loop
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        restart += 1
        print(f"[restart {restart}]")

        # ----------------------------------------------------------
        # initialise a random feasible vector (or warm start)
        # ----------------------------------------------------------
        if restart == 1 and init_a is not None:
            a = init_a.copy()
        else:
            a = np.random.rand(n)

        # project onto the simplex Σ a_i = 1, a_i ≥ 0
        a = _project_simplex(a)

        # ----------------------------------------------------------
        # projected sub‑gradient descent on max(conv)
        # ----------------------------------------------------------
        iteration = 0
        # step‑size scaling – a modest value works well in practice
        step_scale = 0.6
        while time.time() < deadline - 2.0:
            iteration += 1

            # 1) compute full convolution and the current objective
            conv = _full_convolution(a)
            max_b = conv.max()
            cur_val = 2.0 * n * max_b / (a.sum() ** 2)  # a.sum() == 1
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[restart {restart} iter {iteration}] new best = {best_val:.6f}")

            # 2) compute a sub‑gradient of the maximal entry
            #    K = set of indices where the max is attained (within tolerance)
            tol = 1e-12
            K = np.where(conv >= max_b - tol)[0]
            g = np.zeros_like(a)

            # analytical contribution for each k ∈ K:
            #   ∂c_k/∂a_i = 2·a_{k-i}  (if the index is in range)
            for k in K:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)          # inclusive
                idx_i = np.arange(i_start, i_end + 1)
                idx_j = k - idx_i
                g[idx_i] += 2.0 * a[idx_j]

            # 3) gradient (additive) step with diminishing step size
            step = step_scale / np.sqrt(iteration)
            a = a - step * g

            # 4) project back onto the simplex
            a = _project_simplex(a)

            # occasional tiny random jitter to avoid stagnation
            if iteration % 800 == 0:
                a += np.random.normal(scale=1e-4, size=n)
                a = _project_simplex(a)

            # early exit if we are already within 1 % of the theoretical lower bound
            lower_bound = 1.0 / (2 * n - 1)          # min possible max_b
            if max_b <= lower_bound * 1.01:
                break

        # ----------------------------------------------------------
        # End of this restart – proceed to the next one
        # ----------------------------------------------------------

    # -----------------------------------------------------------------
    # Post‑processing of the best vector found so far
    # -----------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        # anti‑symmetric re‑ordering (often lowers the peak)
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _objective(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # final hill‑climbing refinement (mass transfers + swaps)
    if best_a is not None and time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # -----------------------------------------------------------------
    # Final reporting
    # -----------------------------------------------------------------
    if best_a is None:
        # fallback – uniform distribution (value = 2.0)
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a)

    final_score = _objective(best_a)
    print(f"[final] best evaluation reported = {final_score:.6f}")

    # return a plain Python list as required by the outer harness
    return best_a.tolist()
