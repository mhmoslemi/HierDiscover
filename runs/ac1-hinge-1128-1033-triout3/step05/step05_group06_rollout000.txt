def evaluate_sequence(sequence):
    # security checks, convert to float
    # protect against negative: sequence = [max(0, x) for x in sequence]
    # protect against numbers too large: sequence = [min(1000.0, x) for x in sequence]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # protect against sum too close to zero
    if sum_a < 0.01:
       return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Propose a step‑function (non‑negative vector) that minimises the
evaluate_sequence objective.

Algorithm
---------
1. **Deterministic grid search** over several sequence lengths and a
   small collection of parametric families (power‑law, product‑law,
   exponential, symmetric Beta, sine).  For each candidate we
   normalise the sum to 1, compute the fast convolution via a real FFT
   and evaluate the score ``2 * n * max(conv)`` (which equals the value
   returned by evaluate_sequence when the sum is 1).  The best vector
   from this coarse search is kept.

2. **Refinement** – the best vector becomes the seed of a projected‑Adam
   optimisation on the log‑probabilities (softmax).  The loss is the exact
   ``max_k (a*a)[k]``; a sub‑gradient is obtained from the active set of
   maximal indices.  Adam steps are performed until the remaining
   time budget is exhausted.  The optimiser occasionally “re‑heats”
   (tiny random noise) when progress stalls.

Both stages respect the 1000 s time limit.  The final vector is returned
as a plain Python list; the surrounding evaluation harness will call
``evaluate_sequence`` on it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution using real FFT (returns length 2n‑1)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself, using zero‑padding of length 2*n."""
    n = a.shape[0]
    size = 2 * n                      # zero‑padding length (no need for power of two)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# Scoring for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def score_vector(a: np.ndarray) -> float:
    """Return 2 * n * max(convolution) – the exact score used by evaluate_sequence."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Parametric families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (min(i, n‑1‑i)+1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ [(i+1)*(n‑i)]^(-p)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

def beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) on the discrete set {0,…,n‑1}.
    a_i ∝ (i+0.5)^(α‑1) * (n‑0.5‑i)^(α‑1)
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1)) * ((n + 0.5 - i) ** (alpha - 1))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

def sine_distribution(n: int) -> np.ndarray:
    """a_i ∝ sin(π * (i+1) / (n+1)).  Endpoints are zero."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

# ----------------------------------------------------------------------
# Projection: softmax (log‑parameters → probability vector)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution w.r.t. log‑probabilities
# ----------------------------------------------------------------------
def grad_log_params(x: np.ndarray, a: np.ndarray,
                    max_b: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Returns ∂/∂x (max_k (a*a)[k]) for a = softmax(x).
    """
    n = a.shape[0]
    G = np.zeros_like(a)

    # Accumulate contributions from each active convolution index
    for k in active_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_range = k - i_range
        G[i_range] += 2.0 * a[j_range]

    # Sub‑gradient for the log‑parameters (see derivation in docstring)
    grad = a * (G - 2.0 * active_idx.size * max_b)
    return grad

# ----------------------------------------------------------------------
# Adam optimiser on the log‑probabilities
# ----------------------------------------------------------------------
def run_adam(initial_a: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the log‑parameters starting from `initial_a`
    for at most `time_limit` seconds.
    Returns the best vector encountered and its score.
    """
    start = time.time()
    n = initial_a.shape[0]

    # Initialise log‑parameters (avoid log(0))
    eps = 1e-12
    x = np.log(np.maximum(initial_a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.2                     # base learning rate in log‑space

    best_score = np.inf
    best_vec = None
    last_improve = 0
    iteration = 0

    while time.time() - start < time_limit - 0.2:
        iteration += 1

        # ----- softmax -----
        a = softmax(x)

        # ----- convolution -----
        conv = full_convolution_fft(a)
        max_b = conv.max()

        # active set (indices achieving max, with a tiny tolerance)
        tol = max_b * 1e-12 + 1e-15
        active = np.where(conv >= max_b - tol)[0]

        # ----- gradient -----
        grad = grad_log_params(x, a, max_b, active)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of the stepsize (helps stability for long runs)
        lr = lr0 / (np.sqrt(iteration) + 1.0)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- bookkeeping -----
        cur_score = float(2.0 * n * max_b)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = iteration
            print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # ----- re‑heat if stuck -----
        if iteration - last_improve > 5000:
            x += np.random.randn(n) * 1e-5
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"Iter {iteration:6d}: re‑heat (no improvement)")

    return best_vec, best_score

# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence.  The routine first performs a cheap grid search
    over several parametric families, then refines the best candidate
    with projected Adam.  It respects the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin for finalisation

    # ------------------------------------------------------------------
    # 1. Coarse deterministic search over families and lengths
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf

    # sequence lengths to explore (larger n gives more freedom)
    n_candidates = [4000, 5000, 6000, 7000, 8000, 9000, 10000]

    # parameters for the families
    p_vals = [0.4, 0.7, 1.0, 1.3, 1.6, 2.0]
    lam_vals = [0.003, 0.01, 0.02]
    alpha_vals = [0.5, 0.6, 0.7, 0.8, 1.0, 1.2]

    total_checked = 0
    start_search = time.time()

    for n in n_candidates:
        # ensure we still have time left for the coarse phase
        if time.time() > deadline - 30:   # leave ~30 s for refinement
            break

        # uniform (as a baseline)
        cand = np.ones(n, dtype=np.float64) / n
        sc = score_vector(cand)
        total_checked += 1
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
            print(f"n={n:5d} uniform   score={sc:.6f}")

        # sine shape
        cand = sine_distribution(n)
        sc = score_vector(cand)
        total_checked += 1
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
            print(f"n={n:5d} sine      score={sc:.6f}")

        # families with varying parameters
        for p in p_vals:
            cand = u_min_distribution(n, p)
            sc = score_vector(cand)
            total_checked += 1
            if sc < best_score:
                best_score, best_vec = sc, cand.copy()
                print(f"n={n:5d} u_min p={p:.2f} score={sc:.6f}")

            cand = u_prod_distribution(n, p)
            sc = score_vector(cand)
            total_checked += 1
            if sc < best_score:
                best_score, best_vec = sc, cand.copy()
                print(f"n={n:5d} u_prod p={p:.2f} score={sc:.6f}")

        for lam in lam_vals:
            cand = exp_distribution(n, lam)
            sc = score_vector(cand)
            total_checked += 1
            if sc < best_score:
                best_score, best_vec = sc, cand.copy()
                print(f"n={n:5d} exp lam={lam:.4f} score={sc:.6f}")

        for alpha in alpha_vals:
            cand = beta_distribution(n, alpha)
            sc = score_vector(cand)
            total_checked += 1
            if sc < best_score:
                best_score, best_vec = sc, cand.copy()
                print(f"n={n:5d} beta a={alpha:.2f} score={sc:.6f}")

        # a few random Dirichlet draws for diversity
        for _ in range(3):
            rnd = np.random.dirichlet(np.ones(n))
            sc = score_vector(rnd)
            total_checked += 1
            if sc < best_score:
                best_score, best_vec = sc, rnd.copy()
                print(f"n={n:5d} random dirichlet score={sc:.6f}")

    print(f"\nCoarse search inspected {total_checked} candidates, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Refinement with projected Adam (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and best_vec is not None:
        print(f"\nStarting Adam refinement (budget {remaining:.1f}s)…")
        refined_vec, refined_score = run_adam(best_vec, remaining)
        if refined_score < best_score:
            best_score, best_vec = refined_score, refined_vec
            print(f"Refinement improved score to {best_score:.6f}")
        else:
            print(f"Refinement did not improve the score (still {best_score:.6f})")

    # ------------------------------------------------------------------
    # 3. Final sanity check – ensure sum is not too small after possible rounding
    # ------------------------------------------------------------------
    if best_vec is None:
        # fallback: uniform vector of length 1 (should never happen)
        best_vec = np.array([1.0])
        best_score = score_vector(best_vec)

    # Return as plain Python list
    print(f"\n=== Search finished ===\nBest final score: {best_score:.6f} (length {len(best_vec)})")
    return best_vec.tolist()
