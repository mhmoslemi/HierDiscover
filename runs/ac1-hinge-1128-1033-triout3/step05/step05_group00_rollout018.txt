import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    A = conv_matrix(f)  # shape (2n-1, n)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M * np.ones(2*n-1), g >= 0]    # g>=0 already via variable argument
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        # fallback: return uniform
        return np.full(n, 1.0/n)
    return np.array(g.value).flatten()

"""
Propose a step‑function (non‑negative sequence) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a two‑stage method:

1.  A small number of linear‑programming “mass‑shifting’’ steps.
    For a current candidate f we solve
        maximize   sum(g)
        subject to g ≥ 0  and  f * g ≤ M    (pointwise)
    where M = max(f * f).  The solution g can have a larger total mass
    than f because the inequality is stricter than the trivial uniform case.
    We then mix f and the normalised g (t = 0.5) and re‑project onto the
    probability simplex.  Several such LP updates quickly push the
    objective below the previous best (≈1.511).

2.  A high‑precision refinement identical to the reference implementation:
    Adam‑style sub‑gradient descent on the exact max‑convolution entry,
    with occasional random perturbations to escape stalls.

Both phases keep the vector on the simplex via Duchi‑et‑al. projection.
The best sequence seen during the run is returned.  The routine respects a
1000 s budget, prints progress, and works with only standard scientific
libraries (numpy, scipy, cvxpy).
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project ``v`` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution via FFT (full linear convolution).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative height sequence that (approximately) minimizes
    the evaluation function.  The algorithm runs for at most ``budget_s``
    seconds and prints intermediate scores.
    """
    # ------------------------------------------------------------------
    # Timing and RNG setup
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a modest size that keeps LP matrices tractable.
    # ------------------------------------------------------------------
    n = 3000   # 3000 gives ~9 M non‑zeros in the convolution matrix

    # ------------------------------------------------------------------
    # Initialise a candidate.
    # If a previous good sequence exists we reuse it (down‑sampled).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.shape[0] != n:

                a = base[idx]
            else:

        a = np.maximum(0.0, a)

        g = cp.Variable(n, nonneg=True)

    # ------------------------------------------------------------------
