"""
propose_candidate
-----------------

Generate a step‑function that improves on the Catalan (central‑binomial)
construction.  The algorithm works in two phases:

1. **Initialization** – build a few classic shapes (Catalan,
   semi‑circular, triangular, sinusoidal), normalise them so that
   Σa_i = √(2·n) and keep the best one as a starting point.

2. **Iterative linear‑programming refinement** – for the current sequence
   *f* compute M = max(f*f).  Solve the linear program

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k = 0 … 2·n‑2
                    g_j ≥ 0

   (the convolution is linear in *g*).  Scale the optimum *g* to have the
   same total mass as *f* and move a little towards it:
   f ← (1‑t)·f + t·g  (t ≈ 0.5).  If the evaluation score improves the
   new *f* is kept, otherwise the step size *t* is reduced.  The loop
   stops when the time budget (≈ 1000 s) is exhausted.

If SciPy is not available (or the matrix becomes too large) the routine
falls back to a simple projected gradient descent that directly minimises
a smooth approximation of the max‑convolution.

All returned coefficients are non‑negative, respect the limits of
``evaluate_sequence`` and are returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """Central‑binomial (Catalan) coefficients, length n."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    return a


def _semicircle_sequence(n: int) -> np.ndarray:
    """Sampled half‑circle: sqrt(1‑x²) on [-1,1]."""
    x = np.linspace(-1.0, 1.0, n)
    a = np.sqrt(np.maximum(0.0, 1.0 - x ** 2))
    return a


def _triangular_sequence(n: int) -> np.ndarray:
    """Symmetric triangular bump."""
    half = (n + 1) // 2
    up = np.arange(1, half + 1, dtype=float)
    if n % 2 == 0:
        down = up[::-1]
    else:
        down = up[-2::-1]
    return np.concatenate([up, down])


def _sine_sequence(n: int) -> np.ndarray:
    """Positive half‑sine on (0,π)."""
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    return a


def _normalise_to_sum(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector to the given L1 sum."""
    s = a.sum()
    if s == 0:
        return a
    return a * (target_sum / s)


def _score_of_sequence(a: np.ndarray) -> float:
    """Exact score used by evaluate_sequence (scale‑invariant)."""
    n = len(a)
    b = np.convolve(a, a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “active‑set” method from the paper)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A where A[k, j] = f_{k‑j} (0 if out of range).
    This is exactly the Toeplitz matrix that represents the convolution
    (f * g)_k = Σ_j A[k, j]·g_j.
    """
    n = len(f)
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    # scipy.linalg.toeplitz creates the proper matrix efficiently.
    from scipy.linalg import toeplitz
    return toeplitz(col, row)


def _lp_step(f: np.ndarray, t: float, target_sum: float) -> (np.ndarray, bool):
    """
    Perform one LP‑based refinement step.
    Returns (new_f, improved) where improved is True iff the score got better.
    """
    from scipy.optimize import linprog

    n = len(f)
    # current bound M = max(f*f)
    M = np.convolve(f, f).max()

    # Build the convolution matrix (2n‑1) × n.
    A = _build_conv_matrix(f)          # dense, but for n≈2000 this is OK
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # maximise Σ g_j  <=> minimise -Σ g_j
    c = -np.ones(n, dtype=float)

    # Solve LP
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return f, False

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 == 0:
        return f, False

    # Scale g to have the same L1 norm as f (which equals target_sum)
    g = g0 * (target_sum / sum_g0)

    # Convex combination
    new_f = (1.0 - t) * f + t * g

    # Evaluate
    old_score = _score_of_sequence(f)
    new_score = _score_of_sequence(new_f)

    if new_score < old_score:
        return new_f, True
    else:
        return f, False


# ----------------------------------------------------------------------
# Gradient‑descent fallback (used when SciPy is unavailable or n too big)
# ----------------------------------------------------------------------


def _gradient_step(a: np.ndarray, step: float, lam: float) -> np.ndarray:
    """
    One projected gradient step on the smooth approximation of max(b).
    a is kept with L1 = 1 (scale invariant), step>0, lam>0.
    """
    n = len(a)

    # convolution a*a
    b = np.convolve(a, a)
    maxb = b.max()

    # smooth weights (log‑sum‑exp approximation of the max)
    w = np.exp(lam * (b - maxb))

    # gradient = 2·lam·corr(w, a) (valid mode gives length n)
    grad = 2.0 * lam * np.correlate(w, a, mode="valid")

    # update, project onto non‑negative orthant and renormalise to sum=1
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new

# Main entry point
# ----------------------------------------------------------------------

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    try:

    MAX_N_FOR_LP = 2300                     # about 100 MiB for the dense matrix
