"""
Search for a low‑evaluation step‑function sequence.

Strategy
--------
1. Build a diverse set of analytic candidates (sqrt series,
   power‑law, exponential, symmetric beta) and keep the best one.
2. Refine the best candidate with an Adam‑style projected gradient descent
   on the sub‑gradient of the maximal self‑convolution.
3. When the descent stalls, perform a short simulated‑annealing walk
   (random mass transfers) to escape local minima.
4. The whole search runs under the given time budget and returns the
   best sequence (non‑negative, sum ≈ 1).  The evaluation used during the
   search is 2·n·max(a∗a), which is identical to `evaluate_sequence`
   when the sequence is normalised to sum 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families of candidate sequences
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Coefficients of (1-x)^(-1/2) truncated to length n and normalised."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


def _power_series(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** (-p)
    w /= w.sum()
    return w


def _exp_series(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam*i), normalised."""
    idx = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * idx)
    w /= w.sum()
    return w


def _beta_series(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}, normalised."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    w[w < 0] = 0.0
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a using a power‑of‑two FFT."""
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
    return conv


def _max_convolution(a: np.ndarray) -> (float, np.ndarray):
    """Return (max value, full convolution) of a∗a."""
    conv = _full_convolution(a)
    return conv.max(), conv


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_conv_subgrad(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k at the active indices.
    For each active k the contribution is g_i += 2·a_{k-i}
    where the index exists.
    The result is averaged over the active set.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    if active_idx.size > 0:
        grad /= active_idx.size
    return grad


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The algorithm combines a parametric sweep,
    Adam‑style projected gradient descent and a short simulated‑annealing
    phase.  It returns the best list found before the time budget expires.
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Length of the sequence (can be overridden via kwargs)
    # --------------------------------------------------------------
    n = int(kwargs.get("length", 4000))

    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Build a portfolio of analytic start points
    # --------------------------------------------------------------
    candidates = []

    # (a) the classic sqrt series
    candidates.append(_sqrt_series(n))

    # (b) power‑law families
    for p in np.linspace(0.3, 0.8, 11):
        candidates.append(_power_series(n, p))

    # (c) exponential families
    for lam in np.linspace(0.0005, 0.02, 10):
        candidates.append(_exp_series(n, lam))

    # (d) symmetric Beta families
    for a in np.linspace(0.2, 2.0, 10):
        candidates.append(_beta_series(n, a, a))

    # (e) the previously discovered sequence if available
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.size == n:
            candidates.append(_project_simplex(h))
        else:
            # truncate or pad with zeros to required length
            h2 = np.zeros(n, dtype=np.float64)
            m = min(h.size, n)
            h2[:m] = h[:m]
            candidates.append(_project_simplex(h2))
    except NameError:
        pass  # no global sequence provided

    # --------------------------------------------------------------
    # Evaluate the portfolio, keep the best initial point
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        # candidate is already on the simplex
        max_val, _ = _max_convolution(cand)
        score = 2.0 * n * max_val      # equal to evaluate_sequence for sum≈1
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
        print(f"[init #{idx:02d}] score={score:.6f}")

    a = best_seq.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _max_convolution(a)[0]

    print(f"[init] chosen start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam‑style projected gradient descent on the max‑conv objective
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    lr0 = 0.15
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iter_cnt = 0
    no_improve = 0
    max_no_improve = 3000

    while time.time() < deadline:
        # current convolution and the active indices
        max_val, conv = _max_convolution(a)
        active = np.flatnonzero(conv >= max_val - 1e-14)

        # sub‑gradient
        grad = _max_conv_subgrad(a, active)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iter_cnt + 1))
        v_hat = v / (1.0 - beta2 ** (iter_cnt + 1))
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a = _project_simplex(a_new)

        # occasional tiny random jitter to avoid plateaus
        if iter_cnt % 250 == 0 and iter_cnt > 0:
            jitter = np.random.randn(n) * (1e-4 / np.sqrt(iter_cnt + 1))
            a = _project_simplex(a + jitter)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{iter_cnt:6d}] new best (grad) = {best_score:.6f}")
        else:
            no_improve += 1

        # restart if stuck for a while
        if no_improve >= max_no_improve:
            a = best_seq.copy() + np.random.randn(n) * 1e-3
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{iter_cnt:6d}] restart triggered")

        iter_cnt += 1

    # --------------------------------------------------------------
    # 3️⃣  Short simulated‑annealing refinement (uses remaining time)
    # --------------------------------------------------------------
    # Remaining budget may be small; run a lightweight SA if possible.
    temperature = 0.01
    sigma = 0.002
    sa_iter = 0
    while time.time() < deadline:
        # propose a random perturbation that preserves the simplex
        delta = np.random.randn(n) * sigma
        a_prop = a + delta
        a_prop = _project_simplex(a_prop)

        max_val_prop, _ = _max_convolution(a_prop)
        prop_score = 2.0 * n * max_val_prop

        # Metropolis acceptance
        if prop_score < best_score:
            accept = True
        else:
            # probability to accept a worse move
            prob = np.exp(-(prop_score - best_score) / temperature)
            accept = np.random.rand() < prob

        if accept:
            a = a_prop
            if prop_score < best_score:
                best_score = prop_score
                best_seq = a.copy()
                print(f"[SA {sa_iter:6d}] improvement = {best_score:.6f}")

        # anneal parameters
        temperature *= 0.9995
        sigma *= 0.9995
        sa_iter += 1
        if temperature < 1e-8:
            temperature = 1e-8

    print(f"[final] best evaluation = {best_score:.6f} after {iter_cnt + sa_iter} iterations")
    return best_seq.tolist()
