"""
Propose a candidate step‑function (list of non‑negative heights).

The algorithm runs a projected Adam‑style optimizer on the simplex.
It minimises the maximum entry of the self‑convolution of the
distribution (the objective is 2·n·max(conv(p,p)) because the sum of
the heights cancels out).  A soft‑max approximation of the max is used
with an annealing temperature (parameter `alpha`) that gradually
focuses the gradient on the true maximum.  Random jitter and occasional
re‑normalisation help to escape shallow local minima.  The best vector
encountered before the time budget expires is returned.
"""

import numpy as np
import time

# --------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# --------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector v onto the simplex {x | x_i >= 0, sum x_i = 1}.
    """
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The routine runs an Adam optimiser on a
    smoothed max‑objective, stays on the probability simplex and
    returns the best sequence discovered before the time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored – kept for compatibility.

    Returns
    -------
    list[float]
        The best step‑function found (sums to 1, all entries >= 0).
    """
    rng = np.random.default_rng(seed)

    # 1. Determine the problem size.  Use the length of a known
    #    construction if it exists, otherwise fall back to a default.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name‑defined]
    except Exception:
        n = 1153  # default size used in the benchmark

    # 2. Initialise a feasible probability vector.
    # --------------------------------------------------------------
    if rng.random() < 0.3 and "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name‑defined]
        seq = np.clip(seq, 0.0, 1000.0)

        if total > 0.0:
            p = seq / total
        else:
            p = np.full(n, 1.0 / n)
    else:
        p = rng.random(n)
        p = p / p.sum()

    # --------------------------------------------------------------
    # 3. Optimisation hyper‑parameters.
    # --------------------------------------------------------------
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # Learning‑rate schedule
    lr = 0.5          # initial step size
    lr_decay = 0.9999  # gentle decay

    # Soft‑max temperature (alpha) – starts low (smooth) and grows.
    alpha = 1.0
    alpha_mul = 1.0015  # increase factor per iteration

    # Index vector needed for the correlation‑based gradient
    idx = 2 * n - 2 - np.arange(n)

    # --------------------------------------------------------------
    # 4. Book‑keeping.
    # --------------------------------------------------------------
    best_p = p.copy()
    # exact objective for sum(p)=1: 2·n·max(conv(p,p))
    best_score = 2.0 * n * np.max(np.convolve(p, p))
    print(f"Initial score: {best_score:.6f}")

    # 5. Main optimisation loop.
    # --------------------------------------------------------------

        # ---- 5.1 Convolution of p with itself (objective & soft‑max) ----
        conv = np.convolve(p, p)                # length 2n‑1

        exp_term = np.exp(alpha * (conv - max_conv))

    # --------------------------------------------------------------
    # Clip explicitly to obey the evaluate_sequence security checks.
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Finished. Best score achieved: {best_score:.6f}")
    return result
