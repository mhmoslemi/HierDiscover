"""
Propose a high‑quality step function.

The algorithm is a two‑stage optimizer:

1.  Start from a Catalan‑type (central‑binomial) sequence, scaled so that
    Σ a_i = √(2 n).  For this scaling the self‑convolution peak is ≈π/2,
    giving the classic bound 1.5708.

2.  Repeatedly improve the shape by solving a linear programme:

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k,
                    g_i ≥ 0,

   where f is the current sequence and M = max(f * f).  The LP finds a
   sequence g that adds as much L1‑mass as possible without increasing the
   convolution peak.  After rescaling g to the same L1‑norm as f we mix
   the two sequences (f ← (1−t) f + t g).  The peak M never increases,
   while the L1‑norm stays fixed, so the evaluation score 2 n·max/Σ²
   strictly improves.

   A few LP iterations (≈10–15) are performed within the time budget,
   then a tiny stochastic fine‑tuning (pairwise mass transfers) is run
   until the deadline.  The method works with `scipy.optimize.linprog`
   (HiGHS) and falls back to the simple random refinement if SciPy is
   unavailable.

The returned list contains non‑negative floats and satisfies all
constraints of `evaluate_sequence`.
"""

import time
import math
import numpy as np

# Optional imports – the LP stage is used only if SciPy is present.
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _HAS_SCIPY = True
except Exception:                     # pragma: no cover
    _HAS_SCIPY = False


def generate_catalan(n: int) -> np.ndarray:
    """Return the first *n* Catalan‐type coefficients c_k = (2k choose k)/4^k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def precompute_conv_matrix_structure(n: int):
    """
    Pre‑compute the (row, col) index pattern of the convolution matrix.
    For a fixed length *n* the pattern does not depend on the actual
    coefficients; only the data values change.
    """
    # row indices: for each i, the convolution output positions i .. i+n‑1
    rows = np.concatenate([np.arange(i, i + n, dtype=np.int32) for i in range(n)])
    # column indices: i repeated n times (one block per i)
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    return rows, cols


def build_conv_ub_matrix(f: np.ndarray, rows: np.ndarray, cols: np.ndarray):
    """Build the sparse A_ub matrix for the LP (constraints (f*g)_k ≤ M)."""
    n = len(f)
    data = np.repeat(f, n)                # each coefficient f[i] appears n times
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a problem size.  Use the length of the previously known
    #    sequence if it exists, otherwise fall back to a modest default.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:                       # pragma: no cover
        n = 800                               # a size that keeps the LP cheap

    # ------------------------------------------------------------------
    # 2. Initialise with a Catalan sequence (or a truncated existing one).
    # ------------------------------------------------------------------
    try:
        seq0 = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if len(seq0) >= n:
            a = seq0[:n].copy()
        else:                                                  # pragma: no cover
            a = np.concatenate([seq0, np.zeros(n - len(seq0), dtype=np.float64)])
    except Exception:                                         # pragma: no cover
        a = generate_catalan(n)

    # Normalise to the standard L1‑norm √(2 n); scaling does not change the score.
    sum_target = math.sqrt(2.0 * n)
    a *= sum_target / a.sum()
    f = a.copy()

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Pre‑compute the index pattern for the convolution‑matrix (used
    #    in every LP iteration).
    # ------------------------------------------------------------------
    rows_pat, cols_pat = precompute_conv_matrix_structure(n)

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (LP‑based refinement).
    # ------------------------------------------------------------------
    if _HAS_SCIPY:
        t_mix = 0.5                # mixing proportion for the first iteration
        iter_no = 0
        max_iter = 30

        while time.time() < deadline and iter_no < max_iter:
            # current peak of the self‑convolution
            conv_ff = np.convolve(f, f)
            M = conv_ff.max()

            # build the LP: maximise Σ g_i  subject to (f*g)_k ≤ M
            A_ub = build_conv_ub_matrix(f, rows_pat, cols_pat)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)                     # maximise Σ g_i
            bounds = [(0.0, None)] * n

            # give the solver a modest time limit per call
            time_left = deadline - time.time()
            if time_left <= 0:
                break
            opts = {"time_limit": max(0.0, time_left), "presolve": True}

            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method="highs", options=opts)

            if not res.success:                                   # pragma: no cover
                print(f"[LP {iter_no}] linprog failed: {res.message}")
                break

            g0 = res.x
            if g0.sum() == 0.0:                                    # pragma: no cover
                break

            # rescale g0 to have the same L1‑norm as f (i.e. √(2 n))
            g = g0 * (sum_target / g0.sum())

            # convex combination – this keeps the L1‑norm unchanged
            f_new = (1.0 - t_mix) * f + t_mix * g

            # re‑normalise numerically to avoid drift
            f_new *= sum_target / f_new.sum()

            # evaluate the candidate
            cand_score = evaluate_sequence(f_new.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = f_new.copy()
                print(f"[improve] iter {iter_no}, new best score={best_score:.6f}")

            # prepare for the next round
            f = f_new
            t_mix = max(0.1, t_mix * 0.93)          # shrink the step size gradually
            iter_no += 1
    else:                                            # pragma: no cover
        print("[info] SciPy not available – skipping LP stage.")

    # ------------------------------------------------------------------
    # 5. Tiny stochastic fine‑tuning (pairwise mass transfers).  This
    #    works irrespective of whether the LP stage was performed.
    # ------------------------------------------------------------------
    step_scale = 0.02 * (math.sqrt(2.0) / max(1.0, math.sqrt(n)))
    while time.time() < deadline:
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # propose to move a small amount δ from index j to i
        δ = (np.random.rand() - 0.5) * step_scale * best_seq[i]
        new_i = best_seq[i] + δ
        new_j = best_seq[j] - δ
        if new_i < 0.0 or new_j < 0.0:
            continue
        cand = best_seq.copy()
        cand[i] = new_i
        cand[j] = new_j
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[local] improved to {best_score:.6f}")

        # slowly shrink the step magnitude
        if np.random.rand() < 0.01:
            step_scale *= 0.95

    print(f"[final] best score={best_score:.6f}, length={len(best_seq)}")
    # enforce non‑negativity (evaluate_sequence already clamps, but we keep it clean)
    best_seq = np.maximum(best_seq, 0.0).tolist()
    return best_seq
