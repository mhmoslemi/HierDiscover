"""
Propose a candidate step‑function (list of non‑negative heights).

Algorithm
---------
1.  Initialise the sequence with the “arcsine’’ coefficients
    a_i = binom(2i,i)/4^i (the inverse‑square‑root series).  This
    already gives the classic bound π/2 ≈ 1.5708.

2.  Repeatedly improve the shape by solving a *linear program*.
    For the current sequence f we compute
        M = max( f * f )
    and build the (2n‑1)×n convolution matrix A_f such that
        (A_f @ g)[k] = (f * g)[k] .
    The LP

        maximise    Σ g_j
        subject to  A_f @ g ≤ M ,   g ≥ 0

    finds, among all non‑negative sequences g whose convolution with f
    does not exceed the present maximum M, the one with the largest
    ℓ₁‑norm.  After scaling g to have the same total mass as f we
    substitute a convex combination

        f ← (1‑t)·f + t·g

    (t starts at 0.5 and is gradually reduced).  This step never
    increases max(f*f) and therefore strictly improves the evaluation
    score 2·n·max_b / (Σa)².

3.  When the time budget is almost exhausted we perform a cheap
    stochastic “mass‑transfer’’ refinement (the same idea that was used
    in the previous baseline) to squeeze out the last possible gains.

The whole procedure runs within the allotted 1000 s, uses only one CPU
core and requires only `numpy`, `cvxpy` and the provided
`evaluate_sequence` function.
"""

import time
import numpy as np
import cvxpy as cp


def _compute_score(seq):
    """Fast evaluation of the objective (identical to evaluate_sequence
    but without the security checks)."""
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    L = 1 << ((2 * n - 1).bit_length())        # next power of two
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _arcsine_coeffs(n):
    """First n coefficients of (1‑z)^(-1/2) = Σ binom(2i,i)/4^i z^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs.tolist()


def _conv_matrix(f):
    """
    Build the (2n‑1)×n convolution matrix A_f such that A_f @ g == f * g.
    The i‑th column is the vector f placed starting at row i.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     length: int = 500) -> list[float]:
    """
    Return a step function (list of non‑negative heights) with a low
    evaluation score.  The function respects the 1000 s time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Start from the arcsine series (or from a previously
    #     discovered sequence if it exists in the global namespace).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        # use the globally supplied sequence, padded/truncated to `length`
        init = list(height_sequence_1)
        if len(init) < length:
            # pad with arcsine tail if needed
            tail = _arcsine_coeffs(length - len(init))
            init = init + tail
        else:
            init = init[:length]
        seq = np.asarray(init, dtype=np.float64)
    else:
        seq = np.asarray(_arcsine_coeffs(length), dtype=np.float64)

    best_seq = seq.copy()
    best_score = _compute_score(best_seq.tolist())
    print(f"[init] length={len(seq)}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop – linear‑programming based mass
    #     redistribution.
    # ------------------------------------------------------------------
    it = 0
    t = 0.5                      # initial mixing parameter

    while time.time() < deadline:
        it += 1
        n = len(seq)
        f = seq
        # current maximal convolution value
        M = np.max(np.convolve(f, f))

        # build the convolution matrix A_f
        A = _conv_matrix(f)

        # linear program: maximise sum(g)  s.t.  A@g <= M , g >= 0
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iters=5000,
                       abstol=1e-9, reltol=1e-9, feastol=1e-9)
        except cp.error.SolverError:
            print("[lp] solver error – aborting LP phase")
            break

        if g.value is None:
            print("[lp] no solution – aborting")
            break

        g0 = np.maximum(g.value, 0.0)            # numerical safety
        sum_f = np.sum(f)
        sum_g0 = np.sum(g0)
        if sum_g0 == 0.0:
            print("[lp] zero‑mass solution – aborting")
            break

        # rescale g0 so that it has the same total mass as f
        scale = sum_f / sum_g0
        g_scaled = g0 * scale

        # try to mix f with g_scaled; reduce t if the mixture would
        # raise the peak of the self‑convolution.
        t_try = t
        while t_try > 1e-4:
            f_new = (1.0 - t_try) * f + t_try * g_scaled
            M_new = np.max(np.convolve(f_new, f_new))
            if M_new <= M + 1e-12:
                break
            t_try *= 0.5
        if t_try <= 1e-4:
            # unable to obtain a non‑increasing max – stop the loop
            print("[lp] mixing did not improve max – exiting")
            break

        # accept the new sequence
        seq = f_new
        cur_score = _compute_score(seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] it={it}, t={t_try:.4f}, score={best_score:.6f}")

        # more conservative
        if it % 20 == 0:
            t = max(0.05, t * 0.9)

        # early termination if time is almost up
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3.  Final cheap stochastic refinement (mass transfer moves).
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    while time.time() < deadline:
        i, j = np.random.randint(len(seq), size=2)
        if i == j:
            continue
        src = seq[i]
        if src <= 0.0:
            continue
        # move a tiny random fraction of mass from i to j
        delta = np.random.exponential(scale=1e-3) * src
        if delta > src:
            continue
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        cur_score = _compute_score(new_seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"[final‑improve] score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f}, iterations={it}")
    # return a plain Python list as required by the evaluator
    return best_seq.tolist()
