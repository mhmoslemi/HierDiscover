"""
Hybrid stochastic–gradient search for a low‑value step sequence.

The algorithm starts from the classic arcsine (inverse‑square‑root) series,
which already gives the bound ≈π/2.  It then repeatedly performs two kinds
of moves while the time budget lasts:

1. **Large random mass transfers** – move a random amount of “mass”
   from an early (large) coefficient to a far‑away (small) coefficient.
   This tends to lower the peak of the self‑convolution.

2. **Soft‑max gradient descent** – approximate the max convolution by a
   smooth soft‑max (parameter β) and take a gradient step that reduces the
   surrogate while keeping the total sum fixed (the score is scale‑invariant,
   so we keep the sum equal to 1).  The temperature β is increased slowly,
   turning the surrogate into the true max and driving the sequence towards a
   flatter convolution profile.

Both move types are accepted only when they improve (or very rarely
worsen) the true evaluation score.  The best sequence seen is returned.
"""

import numpy as np
import time

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------

def _arcsine_coeffs(n: int):
    """First `n` coefficients of (1‑z)^(-1/2)."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _full_convolve(a: np.ndarray, b: np.ndarray):
    """Convolution of two real 1‑D arrays using FFT (full length)."""
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: n + m - 1]


def _fast_score(seq: np.ndarray):
    """Fast evaluation of the objective 2 n max(conv)/(sum)^2."""
    n = seq.shape[0]
    conv = _full_convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ------------------------------------------------------------
# Main candidate generator
# ------------------------------------------------------------

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     length: int = 3000):
    """
    Return a list of non‑negative heights for the step function.
    A hybrid stochastic / gradient search is run for `budget_s` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ----- 1. start from the arcsine series (normalized to sum = 1)
    seq = _arcsine_coeffs(length).astype(np.float64)
    seq /= np.sum(seq)               # keep the sum fixed (scale‑invariant)
    best_seq = seq.copy()
    best_score = _fast_score(best_seq)
    cur_score = best_score
    print(f"[init] length={length}, score={best_score:.6f}")

    # parameters for the two move families
    early_cut = max(1, length // 10)          # source index domain
    late_cut = max(1, length // 2)            # target index domain
    beta = 2.0                                # soft‑max sharpness
    beta_inc = (30.0 - beta) / (budget_s * 100)  # raise β gradually
    lr = 0.02                                 # base learning rate for gradient steps
    it = 0

    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------
        # 1) occasionally perform a soft‑max gradient step
        # ----------------------------------------------------
        if np.random.rand() < 0.2:
            # compute full convolution once
            conv_full = _full_convolve(seq, seq)          # length 2n‑1
            max_conv = np.max(conv_full)

            # soft‑max weights approximating the max
            exp_arg = beta * (conv_full - max_conv)       # shift for stability
            w = np.exp(exp_arg)
            w_sum = np.sum(w)
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the soft‑max (G) term
            grad_g = 2.0 * _full_convolve(seq, w)[:length]

            # gradient of the whole objective  (since sum(seq)=1 -> d(-2logS)= -2)
            grad = grad_g - 2.0

            # take a step, clip to non‑negative and renormalise
            step = lr * grad
            seq_new = seq - step
            seq_new = np.clip(seq_new, 0.0, None)
            s_new = np.sum(seq_new)
            if s_new > 0:
                seq_new /= s_new

            cand_score = _fast_score(seq_new)
            # accept if improves (or with tiny probability to escape)
            if cand_score < cur_score or np.random.rand() < 0.001:
                seq = seq_new
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = seq.copy()
                    print(f"[improve‑grad] it={it}, score={best_score:.6f}")

            # increase β slowly so the surrogate becomes sharper
            beta = min(beta + beta_inc, 30.0)

        # ----------------------------------------------------
        # 2) otherwise do a large random mass‑transfer move
        # ----------------------------------------------------
        else:
            i = np.random.randint(early_cut)            # early (large) index
            j = np.random.randint(late_cut, length)    # far (small) index
            if i == j:
                continue

            src = seq[i]
            if src <= 0.0:
                continue

            # transfer a random fraction (up to 50 %) of the source mass
            delta = np.random.rand() * 0.5 * src
            if delta <= 0.0:
                continue

            seq_new = seq.copy()
            seq_new[i] -= delta
            seq_new[j] += delta

            # keep the sum exactly constant (renormalise to avoid drift)
            seq_new /= np.sum(seq_new)

            cand_score = _fast_score(seq_new)
            # accept improvement or occasional uphill move
            if cand_score < cur_score or np.random.rand() < 0.001:
                seq = seq_new
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = seq.copy()
                    print(f"[improve‑move] it={it}, score={best_score:.6f}")

        # simple early‑stop if we already beat the target bound
        if best_score < 1.503:
            print("[target achieved]")
            break

    print(f"[final] best score={best_score:.6f}")
    # Return a plain Python list (evaluate_sequence expects a list)
    return best_seq.tolist()
