"""
Propose a step‑function by minimizing the L∞‑norm of its self‑convolution.

Algorithm
---------
* Initialise with the best known sequence (`height_sequence_1`) if it
  exists, otherwise use a symmetric U‑shaped distribution.
* Enforce exact symmetry: a[i] = a[n‑1‑i] (the objective is invariant
  under reversal, so the optimum can be assumed symmetric).
* Optimise on the probability simplex (non‑negative, sum = 1) by a
  projected (stochastic) gradient descent.
  – The target function is  f(a) = 2·n·max(conv(a,a)).
  – We replace the hard max by a smooth soft‑max:
        Mβ(a) = (1/β)·log Σₖ exp(β·(a∗a)ₖ)
    whose gradient w.r.t. a is
        ∇Mβ = 2·(w ∗ a_rev)   with   wₖ ∝ exp(β·(a∗a)ₖ).
  – β is increased slowly (β≈5 → 200) so the optimisation first follows
    a smooth landscape and later focuses on the true maximum.
* At each iteration we either
    – take a gradient step (projected back onto the simplex and re‑symmetrised), or
    – perform a random “mass‑transfer” move to help escape shallow minima.
* Simulated‑annealing acceptance (temperature decays) allows occasional
  uphill moves.
* The loop runs until the time budget (≈1000 s) expires, keeping the
  best sequence observed.

The returned list is a plain Python list of floats; the external
evaluator will apply its own clipping and normalisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w >= 0, sum(w)=1 }.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape”: a[i] = (min(i, n‑1‑i)+1)^(-p).
    Returns a normalised probability vector (sum = 1).
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def linear_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear (non‑circular) convolution a*a using FFT.
    Returns a 1‑D array of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given weights w (same length as conv = a*a) corresponding to the
    soft‑max derivative, compute the gradient
        g_i = 2 * Σ_k w_k * a_{k-i}
    efficiently via a single convolution (FFT).
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]

    # convolution length = len(w) + n - 1 = 3*n - 2
    L = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A_rev, n=L)[: len(w) + n - 1]

    # The required slice starts at index n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def enforce_symmetry(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric: a = (a + a[::-1]) / 2."""
    a_sym = (a + a[::-1]) * 0.5
    # Re‑project onto the simplex to restore sum = 1
    return simplex_projection(a_sym)

    # --------------------------------------------------------------

    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
