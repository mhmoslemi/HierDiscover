"""
Exponential‑gradient descent on the max‑convolution objective,
augmented with occasional random jitter and a final linear‑program
refinement (the same LP used by the published “alternating LP” method).

The objective for a probability vector p (∑p=1) is

    M(p) = 2·n·max_k (p * p)_k .

We minimise M by performing mirror‑descent on the simplex (exponential
updates) using a sub‑gradient of max_k(p*p)_k.  The sub‑gradient is
obtained from the active shifts where the convolution attains its
maximum.  Random perturbations prevent the iterate from staying at the
symmetric uniform fixed point.  When time permits we solve a linear
program

    maximise   ∑ g_j
    subject to  p * g ≤ max(p * p)     (component‑wise)
                g ≥ 0

to obtain a direction g that respects the current bound.  Mixing a
small fraction of g into p (convex combination) provably cannot increase
the bound, and typically lowers it further.  The best sequence found
before the time budget expires is returned.

All numbers are non‑negative and automatically satisfy the clipping
performed by `evaluate_sequence`.
"""

import time
import numpy as np

try:
    # SciPy is allowed; we use it only for the final LP refinement.
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:          # pragma: no cover
    _SCIPY_AVAILABLE = False


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _active_gradient(p: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k(p * p)_k.
    It averages the gradients over all active (maximal) shifts.
    """
    n = p.shape[0]
    max_val = conv.max()
    # use a tight tolerance – values are exact floats from np.convolve
    active = np.isclose(conv, max_val, atol=1e-15, rtol=0.0)

    # Build the weighting vector w (length 2n‑1) – uniform over active indices
    w = np.zeros(2 * n - 1, dtype=np.float64)
    cnt = active.sum()
    if cnt == 0:                     # should never happen
        return np.zeros_like(p)
    w[active] = 1.0 / cnt

    # Correlation p ⊛ w (equivalently convolution with reversed w)
    w_rev = w[::-1]
    corr = np.convolve(p, w_rev)          # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)        # map to gradient indices
    grad = 2.0 * corr[idx]                # sub‑gradient (size n)
    return grad


def _lp_refinement(p: np.ndarray, cur_M: float, mix_factor: float = 0.1) -> np.ndarray:
    """
    One linear‑program refinement step.
    Solve: maximise   ∑ g_j
          subject to  p * g ≤ cur_M (component‑wise)
                      g ≥ 0

    Return the convex combination (1‑mix_factor)*p + mix_factor*g,
    renormalised to a probability vector.
    """
    if not _SCIPY_AVAILABLE:
        return p

    n = p.shape[0]
    # Build the constraint matrix A_ub where each row corresponds
    # to a convolution entry (p * g)_k ≤ cur_M.
    # Row k contains the coefficients p_{k‑j} for each variable g_j.
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = p  # shift p by j positions

    b = np.full(2 * n - 1, cur_M, dtype=np.float64)

    # linprog minimises c⋅x, so we negate to maximise ∑g_j.
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        # LP failed – return the original vector.
        return p

    g = res.x
    g_sum = g.sum()
    if g_sum > 0:
        g = g / g_sum
    else:
        g = np.full(n, 1.0 / n)

    # convex combination and renormalise
    p_new = (1.0 - mix_factor) * p + mix_factor * g
    p_new = p_new / p_new.sum()
    return p_new

    time limit is returned.

            log_p -= np.max(log_p)
            p = np.exp(log_p)
            p /= p.sum()

            # ---- random jitter to break symmetry ---------------------------
            if iteration % jitter_every == 0:
                noise = rng.normal(scale=jitter_std, size=n)
                p = p + noise
                p = np.clip(p, 0.0, None)
                total = p.sum()
                if total > 0:
                    p /= total
                else:
                    p = rng.random(n)
                    p /= p.sum()

            # ---- step‑size schedule -----------------------------------------
            eta *= eta_decay

            # ---- stagnation detection ---------------------------------------
            if iteration - last_improve_iter > max_no_improve:
                # restart with a fresh random start
                break

        # If we reached the target inside the inner loop, we can stop early.
        if best_score <= 1.505:
            break

    # ------------------------------------------------------------------
    # Final linear‑program refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # recompute the bound for the best sequence
        conv_best = np.convolve(best_seq, best_seq)
        cur_M = conv_best.max()
        # Perform a single LP refinement and mix a small fraction back.
        refined = _lp_refinement(best_seq, cur_M, mix_factor=final_mix)

        # evaluate the refined candidate
        conv_ref = np.convolve(refined, refined)
        refined_score = 2.0 * n * conv_ref.max()
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined
            print(f"LP refinement improved M to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Clip to the range enforced by `evaluate_sequence` (mainly for safety)
    result = np.clip(best_seq, 0.0, 1000.0).tolist()
    print(f"Final best M = {best_score:.6f} with length {n}")
    return result
