optimizer on a smooth approximation of the max‑operator.

The original objective is
    M(p) = 2·n·max_k (p * p)_k      with   p_i ≥ 0 , Σ p_i = 1 .
The max is replaced by a log‑sum‑exp with a temperature α,
which is slowly increased so that the optimisation first sees a smooth
landscape and then focuses on the true peak.

The probability vector p is represented as a soft‑max of free
log‑it parameters z (p_i = exp(z_i)/Σexp(z)).  This guarantees feasibility
without an explicit projection.  Adam supplies adaptive step‑sizes and
momentum, leading to faster convergence than plain projected gradient
descent.  Several random restarts (including a Gaussian‑shaped start and,

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _smooth_max_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the soft‑max weights w_k = exp(α·conv_k) / Σ exp(α·conv_k),
    computed in a numerically stable way.
    """
    z = alpha * conv
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def _conv_self(p: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of p (length 2·n‑1)."""
    return np.convolve(p, p)

    """Exact objective value 2·n·max(p⋆p)."""
    n = p.shape[0]
    conv = _conv_self(p)
    return 2.0 * n * np.max(conv)

            # ---- convolution and smooth‑max weights ----
            conv = _conv_self(p)                 # length 2n‑1
            w = _smooth_max_weights(conv, alpha)  # same length

            # ---- gradient w.r.t. p (derivative of smooth max) ----
            # grad_p = 2 * (p ⊛ w)  where ⊛ denotes correlation.
            w_rev = w[::-1]
            corr = np.convolve(p, w_rev)          # length 3n‑2
            idx = 2 * n - 2 - np.arange(n)        # pick the needed slice
            grad_p = 2.0 * corr[idx]

            # ---- gradient w.r.t. logits z (mirror‑descent / KL geometry) ----
            dot = np.dot(p, grad_p)
            grad_z = p * (grad_p - dot)

            # ---- optional gradient clipping (stabilise Adam) ----
            # Clip to avoid occasional gigantic steps.
            grad_z = np.clip(grad_z, -1e2, 1e2)

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            z -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- schedule updates ----
            lr *= lr_decay
            alpha = min(alpha * alpha_mul, alpha_max)

            # ----------------------------------------------------------
            # Evaluate exact objective (no smoothing) and keep the best.
            # ----------------------------------------------------------
            cur_score = _max_convolution_score(p)
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                now = time.time()
                print(f"[{now:.1f}] restart {restart:02d} iter {it:05d} "
                      f"new best M = {best_score:.6f}")

                # Early termination if we have already reached the target.
                if best_score <= 1.503:
                    deadline = now   # force outer loop to stop
                    break

            # ----------------------------------------------------------
            # Occasionally perturb logits to escape shallow local minima.
            # ----------------------------------------------------------
            if it % 500 == 0:
                noise = rng.normal(scale=0.02, size=n)
                z += noise

        # ------------------------------------------------------------------
        # If we already hit the target, stop all restarts.
        # ------------------------------------------------------------------
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 5. Return the best sequence (clip to the evaluation routine's limits)
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    return result
