def shape_power(n, p):

    a = (i+1)**(-p) + (n-i)**(-p)
    return a / a.sum()

def shape_exp(n, lam):
    dist = np.minimum(np.arange(n), n-1 - np.arange(n))
    a = np.exp(-lam * dist)
    return a / a.sum()

def shape_arcsine(n):
    i = np.arange(1, n+1)
    a = 1.0 / np.sqrt(i * (n+1 - i))

def shape_triangular(n, K):
    a = np.zeros(n)
    # left side
    a[:K] = (K - np.arange(K)) / K
    # right side
    a[-K:] = (K - np.arange(K)[::-1]) / K # symmetrical
    # double count middle if overlap? For large K > n/2, need careful.
    # If 2*K > n, overlapping region will be double counted.
    # We'll limit K <= n//2.
    return a / a.sum()

"""
Propose a low‑evaluation step‑function.

The algorithm consists of three stages:

1. **Parametric sweep** – Search several simple families (power‑law,
   exponential, arcsine, triangular) on a few candidate lengths.
   Convolution maxima are obtained with an FFT (O(n log n)), so thousands
   of candidates are evaluated quickly.

2. **Refinement by simulated‑annealing** – Starting from the best
   candidate of the sweep we run a fast O(n) annealing loop that moves
   a small mass `δ` from a source index (the entry that most contributes
   to the current peak of the self‑convolution) to a destination index
   that contributes the least to that peak.  The temperature schedule is
   linear, the move size is limited to 50 % of the source mass and the
   Metropolis acceptance rule is used.

3. **Final check** – The best sequence found is fed to the provided
   `evaluate_sequence` function for safety and then returned.

The whole routine respects the time budget (`budget_s`) – it stops
cleanly before the deadline and returns the best sequence discovered so
far.
"""

import numpy as np
import time
import math
import random

# ----------------------------------------------------------------------
# Helper 1: fast convolution max (FFT based, O(n log n))
# ----------------------------------------------------------------------
def max_convolution_fft(a: np.ndarray) -> float:
    """Return max(self‑convolution) of a (assumed sum(a)=1)."""
    n = a.shape[0]
    # size of FFT – next power of two ≥ 2n‑1
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Helper 2: objective for a (sum is 1) → 2·n·max(conv)
# ----------------------------------------------------------------------
def objective(a: np.ndarray) -> float:
    return 2.0 * a.shape[0] * max_convolution_fft(a)


# ----------------------------------------------------------------------
# Shape families – each returns a non‑negative vector (not yet normalised)
# ----------------------------------------------------------------------
def shape_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    return a


def shape_exp(n: int, lam: float) -> np.ndarray:
    # distance to the nearest end
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-lam * d)
    return a


def shape_arcsine(n: int) -> np.ndarray:
    # discrete analogue of f(x)=1/√(x(1‑x)) on {1,…,n}
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    return a


def shape_triangular(n: int, k_frac: float) -> np.ndarray:
    # linear decay from each end, support width = k_frac·n (≤ n//2)
    K = int(k_frac * n)
    if K < 1:
        K = 1
    if K * 2 > n:
        K = n // 2
    a = np.zeros(n, dtype=float)
    # left ramp
    a[:K] = (K - np.arange(K, dtype=float)) / K
    # right ramp
    a[-K:] = (K - np.arange(K, dtype=float)[::-1]) / K
    return a


# ----------------------------------------------------------------------
# Fast O(n) convolution update when moving δ from i → j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Move `delta` from a[i] to a[j] (i ≠ j).  Return (new_a, new_conv).
    a is 1‑D of length n, sum(a)=1.
    conv = a*a (length 2n‑1).
    The update touches O(n) entries only.
    """
    n = a.shape[0]

    a_i = a[i]
    a_j = a[j]
    di = -delta
    dj = +delta

    # new vector
    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    # fast convolution update
    conv_new = conv.copy()

    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]                # all other positions
    a_q = a[q]

    conv_new[i + q] += 2.0 * di * a_q
    conv_new[j + q] += 2.0 * dj * a_q

    # self‑pair contributions
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Targeted move – source = element that most contributes to the current
# convolution peak, destination = element that contributes the least.
# ----------------------------------------------------------------------
def targeted_move_improved(a: np.ndarray, conv: np.ndarray):
    n = a.shape[0]
    k = int(np.argmax(conv))                     # index of the current peak

    # feasible i such that both i and k‑i lie in the vector
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_candidates = np.arange(i_min, i_max + 1, dtype=int)
    j_candidates = k - i_candidates

    # contribution of each unordered pair to the peak
    contrib = a[i_candidates] * a[j_candidates]
    src = int(i_candidates[np.argmax(contrib)])

    # Choose a destination that would add as little as possible to the peak.
    # For every candidate destination d we look at a partner p = k-d;
    # if p is inside the domain the product a[p] is the amount that would be added.
    # We pick a destination with minimal partner weight (randomised among the
    # best few to retain some exploration).
    all_idx = np.arange(n, dtype=int)
    mask = all_idx != src
    candidates = all_idx[mask]
    p = k - candidates
    partner_weight = np.where((p >= 0) & (p < n), a[p], 0.0)

    # pick among the 10 smallest partner weights (or fewer if n is small)
    order = np.argsort(partner_weight)
    top = min(10, len(order))
    dst = int(candidates[order[np.random.randint(top)]])

    return src, dst

    best_seq = a.copy()

    T_start = 0.30
