"""
Propose a low‑evaluation step sequence.

Algorithm overview
-------------------
1. Determine a working length `n`.  If a previous construction
   (`height_sequence_1`) exists we reuse its length, otherwise we take
   `n = 2000`.

2. **Parametric warm‑start** – scan two simple one‑parameter families
   (power‑law and exponential edge‑heavy shapes).  For each choice we
   build a non‑negative vector, normalise it to sum 1 and evaluate
   `2·n·max(conv(a,a))` via an FFT‑based convolution.  The best
   candidate becomes the starting point for the subsequent optimisation.

3. **Exponentiated‑gradient descent** – for a large fraction of the time
   budget we repeatedly perform a mirror‑descent step on the current
   maximal convolution entry.
   *  Find the index `k` where the convolution is maximal.
   *  The gradient of `max conv` w.r.t. `a_i` is `2·a_{k‑i}` (zero if the
      partner index is out of range).  We multiply the current vector by
      `exp(-η·gradient)` and renormalise onto the simplex.
   *  The step size `η` is decreased linearly from ~0.5 to ~0.001.

   The best vector seen during this phase is memorised.

4. **Directed simulated annealing** – the remainder of the time budget
   is spent on a fast SA that always moves mass from the index that
   contributes most to the current peak toward an index whose partner
   value is *minimal* (i.e. moving the mass does not increase the peak).
   The move is performed with the `apply_move` routine that updates the
   full convolution in O(n) time.  A linear temperature schedule
   (`T∈[1e‑2,1e‑5]`) governs occasional uphill moves.

5. Return the best sequence (as a plain Python list).  A final call to
   `evaluate_sequence` is performed for safety and printing.

All helper functions are defined at top‑level; the code uses only NumPy
and the standard library and respects the 1000 s time limit.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Fast convolution (full, length 2n‑1) via FFT – used in the gradient phase
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    # Next power of two ≥ 2n for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    # Trim to the exact length 2n‑1
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Fast convolution update when moving a mass δ from index i to j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector (sum = 1).
    conv : np.ndarray, shape (2n‑1,)
        Full convolution a*a.
    i, j : int
        Distinct source / destination indices, 0 ≤ i,j < n.
    delta : float
        Amount moved (0 < delta ≤ a[i]).

    Returns
    -------
    a_new : np.ndarray
    conv_new : np.ndarray
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta
    dj = +delta

    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    conv_new = conv.copy()

    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]                # all other positions
    a_q = a[q]

    # contributions of the changed entries to all sums
    conv_new[i + q] += 2.0 * di * a_q    # (i, q) and (q, i)
    conv_new[j + q] += 2.0 * dj * a_q    # (j, q) and (q, j)

    # self‑pair contributions (i,i) and (j,j)
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Parametric families for a quick warm‑start
# ----------------------------------------------------------------------
def power_law_shape(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape a_i ∝ (i+1)^{-p} + (n‑i)^{-p}."""
    idx = np.arange(n)
    left = (idx + 1) ** (-p)
    right = (n - idx) ** (-p)
    a = left + right
    # Guard against all‑zero (p very large) – fall back to uniform
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def exponential_shape(n: int, lam: float) -> np.ndarray:
    """Edge‑heavy shape a_i ∝ exp(-lam·i) + exp(-lam·(n-1‑i))."""
    idx = np.arange(n)
    left = np.exp(-lam * idx)
    right = np.exp(-lam * (n - 1 - idx))
    a = left + right
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a

# Main entry point required by the competition framework

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a step‑function (non‑negative vector) that minimises
    `evaluate_sequence`.  The function respects the 1000 s time budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin (2 s) for the final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose the length n.  Re‑use previous construction if possible.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]

        n = 2000

    # ------------------------------------------------------------------
    # Initial vector – either the previous one (normalised) or a simple

    try:
        a = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]

        a /= a.sum()

        a = 1.0 - np.sin(np.pi * x / (n - 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()

    # current convolution and objective
    conv = full_convolution(a)
    cur_val = float(2.0 * n * np.max(conv))
    best_a = a.copy()
    best_val = cur_val

    print(f"[init] n={n}, initial evaluation={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – quick scan of simple parametric families
    # ------------------------------------------------------------------
    def try_candidate(candidate):

        conv_c = full_convolution(candidate)
        val = float(2.0 * n * np.max(conv_c))
        if val < best_val:
            best_val = val
            best_a = candidate.copy()
            print(f"[param] new best = {best_val:.6f}")

    # power‑law sweep
    for p in np.linspace(0.0, 4.0, 41):

    # exponential sweep
    for lam in np.linspace(0.0, 5.0, 51):

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent (mirror descent on the

    # ------------------------------------------------------------------
    total_budget = deadline - start_time
    phase1_end = start_time + 0.60 * total_budget
    iteration = 0
    while time.time() < phase1_end:
        # Linear schedule for the step size η (from 0.5 → 0.001)
        progress = (time.time() - start_time) / (phase1_end - start_time)
        eta = 0.5 * (1.0 - progress) + 0.001 * progress

        # Identify the convolution peak
        k = int(np.argmax(conv))

        # Gradient of max‑conv w.r.t. a_i : g_i = 2·a_{k‑i} (zero if out of bounds)
        idx = np.arange(n)
        partner = k - idx
        valid = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[valid] = a[partner[valid]]

        # Mirror‑descent (exponentiated gradient) step
        a_new = a * np.exp(-eta * g)
        sum_new = a_new.sum()
        if sum_new == 0.0:               # numerical safety – re‑initialise uniformly
            a_new = np.full(n, 1.0 / n)
            sum_new = 1.0
        a_new /= sum_new

        # Full recomputation of the convolution (FFT, cheap)
        conv_new = full_convolution(a_new)
        new_val = float(2.0 * n * np.max(conv_new))

        # Accept if better, otherwise allow a tiny chance of uphill moves
        if new_val < cur_val or np.random.rand() < 0.003:

            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[grad {iteration:6d}] new best = {best_val:.6f}")

        iteration += 1
        # occasional full recomputation is already done each step

    # ------------------------------------------------------------------
    # Phase 2 – directed simulated annealing using fast O(n) updates.
    # ------------------------------------------------------------------
    # Reset temperature schedule (linear from 0.01 → 1e‑5)
    T_start = 0.01
    T_end = 1e-5
    # Ensure we have an up‑to‑date convolution for the SA phase
    conv = full_convolution(a)  # recompute from the last a
    cur_val = float(2.0 * n * np.max(conv))
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    iteration = 0
    while time.time() < deadline:
        now = time.time()
        # linear temperature schedule over the whole remaining time
        progress = (now - start_time) / total_budget
        T = T_start * (1.0 - progress) + T_end * progress

        # --------------------------------------------------------------
        # Directed move: source = index contributing most to the peak,
        # destination = index whose partner value is minimal (does not
        # increase the current peak).
        # --------------------------------------------------------------
        k = int(np.argmax(conv))

        # feasible i‑range for the current peak
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_candidates = np.arange(i_min, i_max + 1, dtype=int)
        j_candidates = k - i_candidates

        # product contributions a_i * a_j
        contrib = a[i_candidates] * a[j_candidates]
        src = int(i_candidates[np.argmax(contrib)])

        # choose destination that minimises a_{k-d} (zero if out of range)
        all_idx = np.arange(n)
        partner_all = k - all_idx
        valid_all = (partner_all >= 0) & (partner_all < n)
        partner_vals = np.zeros_like(a)
        partner_vals[valid_all] = a[partner_all[valid_all]]
        # exclude the source index
        partner_vals[src] = np.inf
        dst = int(np.argmin(partner_vals))

        # --------------------------------------------------------------
        # Perform the move (small fraction of the source mass)
        # --------------------------------------------------------------
        max_delta = a[src] * 0.5  # move at most 50 % of the source mass
        if max_delta < 1e-12:
            # source almost empty – fall back to a random move
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            max_delta = a[src] * 0.5
            if max_delta < 1e-12:
                continue

        delta = np.random.rand() * max_delta
        a_new, conv_new = apply_move(a, conv, src, dst, delta)
        new_val = float(2.0 * n * np.max(conv_new))

        # acceptance test (simulated annealing)
        improve = new_val < cur_val
        if improve or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA {iteration:6d}] new best = {best_val:.6f}")

        iteration += 1

        # Every few thousand iterations recompute the convolution from scratch
        # to avoid any accumulated numerical drift.
        if iteration % 4000 == 0:
            conv = full_convolution(a)

    # ------------------------------------------------------------------
    # Final sanity check with the provided evaluate_sequence
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
