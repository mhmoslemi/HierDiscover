    start = time.time()

    # small perturbation

    a = np.maximum(a, 1e-12)
    a /= a.sum()

    best_a = a.copy()
    best_val = float('inf')
    iteration = 0

    while time.time() < deadline and iteration < 100:
        # compute conv and current M and evaluation
        conv = np.convolve(a, a)
        M = conv.max()
        cur_eval = float(2.0 * n * M)  # sum=1

        # keep best
        if cur_eval < best_val:
            best_val = cur_eval
            best_a = a.copy()
            print(f"[{iteration}] new best eval {best_val:.6f}")

        # if near optimum reachable (lower bound close) we may break
        # lower bound = 2.0 * n / (2 * n - 1) (as sum=1) = 2.0 * n / (2 * n - 1)
        lower_bound = 2.0 * n / (2 * n - 1)
        if best_val - lower_bound < 1e-5:
            break

        # Build A matrix for constraints A @ b <= M
        rows = 2 * n - 1
        cols = n
        A = np.zeros((rows, cols))
        # fill Toeplitz with a in each column
        # For each column i, rows i to i+n-1 get a
        for i in range(n):
            A[i:i+n, i] = a

        # linprog solves min c^T x subject to A_ub @ x <= b_ub, x >= 0
        c = -np.ones(cols)  # maximize sum b
        b_ub = np.full(rows, M)

        # Solve LP
        # Linprog can be heavy; we set options to reduce output
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * cols, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})

        if not res.success:
            print(f"[{iteration}] LP failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print(f"[{iteration}] LP returned zero sum")
            break
        # Scale b to sum 1
        b = b / sum_b

        # Mix a and b
        t = 0.5  # mixing fraction
        a = (1 - t) * a + t * b
        # Ensure positivity
        a = np.maximum(a, 1e-14)
        a /= a.sum()   # renormalise

        iteration += 1

    # final best evaluation using evaluate_sequence
    try:
        final_val = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation from evaluate_sequence = {final_val:.6f}")

        print("[final] evaluate_sequence error:", e)
    return best_a.tolist()

"""
LP‑guided iterative improvement for the step‑function minimisation problem.

The objective used by evaluate_sequence is 
    2·n·max(conv(a,a)) / (sum(a)**2) .

Because the expression is scale‑invariant we may keep the vector normalised
to sum = 1 and simply minimise  n·max(conv).  Starting from a strictly
positive “edge‑heavy’’ distribution we repeatedly:

1. Compute M = max(conv(a,a)).
2. Solve the linear programme
       maximise   Σ b_j
       subject to Σ_i a_i·b_{k-i} ≤ M   (for every convolution index k)
                   b_j ≥ 0
   This finds a direction b that does not increase the current worst
   self‑convolution value.
3. Rescale b so that Σ b_j = 1, mix a←(1‑t)·a + t·b (t≈0.5) and keep the
   best vector seen so far.

The LP is solved with SciPy’s HiGHS interior‑point method; the constraint
matrix is a Toeplitz matrix built from the current a.  The loop stops
when the time budget is exhausted, when no improvement is observed for
several iterations, or when the theoretical lower bound 2·n/(2·n‑1) is
reached.

If SciPy is unavailable the function falls back to a simple simulated‑annealing
search (the original reference implementation) so that a feasible answer is
always returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: build the convolution matrix A such that (A @ b)[k] = (a * b)[k]
# ----------------------------------------------------------------------
def build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return a dense matrix A of shape (2·n‑1, n) with
        A[k, j] = a[k‑j] if 0 ≤ k‑j < n else 0.
    The matrix satisfies A @ b == np.convolve(a, b).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes a on rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = a
    return A


# ----------------------------------------------------------------------
# Helper: solve the linear programme for a given a and bound M.
# ----------------------------------------------------------------------
def lp_maximise_sum_b(a: np.ndarray, M: float):
    """
    Solve:
        maximise   Σ b_j
        subject to (a * b)[k] ≤ M   for all k
                   b_j ≥ 0
    Returns the optimal b (may be empty if LP fails).
    """
    try:
        from scipy.optimize import linprog
    except Exception:          # pragma: no cover – SciPy not installed
        return None

    n = a.shape[0]
    A = build_convolution_matrix(a)          # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises; maximise Σ b  ⇔ minimise -Σ b
    c = -np.ones(n, dtype=np.float64)

    # time limit for this LP is the remaining budget (seconds)
    # This is enforced by the HiGHS solver via the “time_limit” option.
    # If not supported the option is ignored – still safe.
    # The caller should ensure the overall time budget is respected.
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method='highs',
        options={"presolve": True},
    )
    if not res.success or res.x is None:
        return None
    return res.x


# ----------------------------------------------------------------------
# Helper: simple simulated‑annealing fallback (original reference code)
# ----------------------------------------------------------------------
def simulated_annealing_fallback(seed: int, budget_s: int, n: int):
    """Very compact version of the earlier SA that works when SciPy is missing."""
    np.random.seed(seed)
    a = np.full(n, 1.0 / n)
    # add a tiny random perturbation to break symmetry
    a += 0.001 * np.random.rand(n)
    a = np.maximum(a, 1e-12)
    a /= a.sum()

    conv = np.convolve(a, a)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    deadline = start + budget_s - 2.0
    while time.time() < deadline:
        # propose a random move
        i, j = np.random.choice(n, 2, replace=False)
        max_delta = a[i] * 0.3
        if max_delta < 1e-12:
            continue
        delta = np.random.rand() * max_delta
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()
        conv_new = np.convolve(a_new, a_new)
        new_val = 2.0 * n * conv_new.max()
        if new_val < cur_val or np.random.rand() < 0.001:
            a, cur_val = a_new, new_val
            if new_val < best_val:
                best_val, best_a = new_val, a_new.copy()
    return best_a, best_val
