propose_candidate
-----------------

Gradient‐based refinement of a symmetric U‑shaped step function.

* Initialise with several deterministic U‑shapes (different lengths and
  power‑law exponents) and keep the best seed.
* From the best seed run a smooth‑max gradient descent on the objective
  ``2·n·max(conv(a,a))``.
  The smooth max is  m_α(a)= (1/α)·log∑_k exp(α·(a*a)[k]) .
  Its gradient can be written as a convolution, which we evaluate with
  FFTs (O(n log n) per iteration).  Momentum, a slowly increasing α and
  a decaying step size give a fast, stable descent.
* After each update we enforce non‑negativity, symmetry and the simplex
  constraint (∑a_i=1) with the classic O(n log n) simplex projection.
* The loop stops when the 1000 s budget is exhausted; the best sequence
  seen is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w ≥ 0 , sum(w) = 1 }.
    Implementation of the O(n log n) algorithm from
    Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape” with power‑law decay from the edges.
    a[i] ∝ (min(i, n‑1‑i) + 1)^{‑p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)                # distance to the nearest edge
    a = (d + 1) ** (-p) + 1e-12                # tiny epsilon avoids exact zeros
    a /= a.sum()
    return a


def _next_pow2(x: int) -> int:
    """Small utility – round up to the next power of two."""
    return 1 << ((x - 1).bit_length())


def _self_convolution_fft(a: np.ndarray, N: int) -> np.ndarray:
    """Return a*a (full convolution) using real FFT of length N."""
    A = np.fft.rfft(a, N)
    conv = np.fft.irfft(A * A, N)[: 2 * a.size - 1]
    return conv


def _weighted_convolution_fft(w: np.ndarray, a_rev: np.ndarray, N: int) -> np.ndarray:
    """
    Compute (w * a_rev) where w has length 2n‑1 and a_rev has length n.
    Returns the full convolution (length 3n‑2).  The caller extracts the
    relevant n‑element slice later.
    """
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(W * A, N)[: (w.size + a_rev.size - 1)]
    return conv
