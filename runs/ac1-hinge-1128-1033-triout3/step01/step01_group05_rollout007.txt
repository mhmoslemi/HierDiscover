import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: central–binomial (constant‑convolution) coefficients
# ----------------------------------------------------------------------
def generate_binomial_prefix(max_n: int) -> np.ndarray:
    """
    Return the first `max_n` coefficients of the sequence
        b_k = C(2k, k) / 4**k .
    The recurrence  b_{k+1} = b_k * (2k+1) / (2(k+1))  is used.
    """
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, max_n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


# ----------------------------------------------------------------------
# Helper: build the convolution‑matrix needed for the linear program
# ----------------------------------------------------------------------
def build_conv_constraint_matrix(vec: np.ndarray) -> np.ndarray:
    """
    For a fixed vector `vec` of length n, construct the (2n‑1) × n matrix A
    such that  (A @ g)[k] = (vec * g)[k]  (the linear part of the convolution).
    This matrix is used as the inequality A @ g ≤ M in the LP.
    """
    n = vec.shape[0]
    # first column : vec followed by n‑1 zeros
    col = np.concatenate([vec, np.zeros(n - 1, dtype=np.float64)])
    # first row  : vec[0] followed by n‑1 zeros
    row = np.concatenate([vec[:1], np.zeros(n - 1, dtype=np.float64)])
    return toeplitz(col, row)

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) that strives to
    minimise

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The algorithm:
      1. start from the analytically optimal binomial prefix,
         normalised to sum = sqrt(2n);
      2. repeatedly solve the linear programme
            maximise   Σ g_j
            subject to   (f * g) ≤ M   and   g_j ≥ 0
         where M = max(f * f) for the current f;
      3. rescale the LP solution to the standard L1‑norm and mix it with
         the current f (convex combination).  The mixing weight is
         adaptively tuned; each successful step strictly non‑increases
         the evaluation score.
      4. occasional tiny random perturbations are tried to escape plateaus.

    """
