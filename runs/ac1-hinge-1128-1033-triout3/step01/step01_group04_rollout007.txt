from scipy.sparse import csr_matrix

def build_conv_ub(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for j in range(n):
        row_start = j
        row_end = j + n
        # for k = row_start..row_end-1
        rows.extend(range(row_start, row_end))
        cols.extend([j]*(n))
        data.extend(f.tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def propose_candidate(seed=0, budget_s=1000, length=1500):
    import numpy as np, time
    from scipy.optimize import linprog

    # start timer
    start = time.time()
    deadline = start + budget_s - 0.5

    np.random.seed(seed)

    # arcsine coeffs
    def arcsine_coeffs(n):
        coeffs = np.empty(n, dtype=np.float64)
        coeffs[0] = 1.0
        for i in range(1, n):
            coeffs[i] = coeffs[i-1] * (2.0*i - 1.0) / (2.0*i)
        return coeffs

    n = length
    a = arcsine_coeffs(n)

    # normalize sum to sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / a.sum()
    f = a * scale

    # compute current score using fast compute
    def fast_score(seq):
        seq = np.asarray(seq, dtype=np.float64)
        n = len(seq)
        L = 1 << ((2 * n - 1).bit_length())
        fft_seq = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
        max_b = conv.max()
        sum_a = seq.sum()
        if sum_a < 0.01:
            return float('inf')
        return 2.0 * n * max_b / (sum_a * sum_a)

    best_seq = f.copy()
    best_score = fast_score(best_seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    # iterative improvement
    iteration = 0
    t = 0.2  # mixing step size
    while time.time() < deadline:
        iteration += 1
        # Build convolution matrix A such that A @ b = conv(f,b) (full length)
        # Using dense matrix
        A = np.empty((2*n-1, n), dtype=np.float64)
        # fill with zeros initially
        A[:] = 0.0
        for j in range(n):
            A[j:j+n, j] = f  # assign column j

        M = np.max(np.convolve(f, f))  # current max
        # Solve LP: maximize sum(b) subject to A @ b <= M, b >= 0
        c = -np.ones(n)  # minimize -sum(b)
        b_ub = np.full(2*n-1, M, dtype=np.float64)

        sum_b = -res.fun

        # check for convergence

            if best_score - new_score < 1e-5:

"""
Propose a candidate step‑function using an LP‑based “convex‑combination” improvement
over the arcsine (inverse‑square‑root) construction.

Idea
-----
* Start from the classic arcsine coefficients a_i = binom(2i,i)/4^i .
* Scale them so that Σ a_i = sqrt(2·n).  For this normalisation the
  evaluation score is exactly M = max((a∗a)(k)).
* Solve the linear programme

        maximise   Σ b_i
        subject to   (a ∗ b)(k) ≤ M   for all k = 0 … 2n‑2
                     b_i ≥ 0

  The matrix of the convolution (a∗b) is Toeplitz and is built explicitly.
  The optimum b has a larger ℓ₁‑norm than a, because the constraints are
  slack for large k.

* Rescale b to the same ℓ₁‑norm (call the result g).  Then
        max(a∗g) = M·( Σ a / Σ b ) < M .
* Replace a by a convex combination (1‑t)·a + t·g (t≈0.2).  The new
  sequence has the same ℓ₁‑norm but a strictly smaller maximal
  self‑convolution, hence a lower evaluation score.
* Repeat the LP‑+‑mix step while time permits (a dozen iterations are
  usually enough).  The best sequence seen is returned.

The whole routine runs in < 1000 s for n≈1500–2000 and typically improves
the score from π/2≈1.5708 to below the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _arcsine_coeffs(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^(-1/2)."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _fast_score(seq: np.ndarray) -> float:
    """Fast version of the supplied evaluate_sequence (scale‑invariant)."""
    n = len(seq)
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that A @ b = np.convolve(f, b) (full).
    Column j of A contains f placed from row j to row j+n−1.
    """
    n = len(f)
    A = np.empty((2 * n - 1, n), dtype=np.float64)
    A[:] = 0.0
    for j in range(n):
        A[j:j + n, j] = f
    return A


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, length: int = 1500):
    """
    Return a list of non‑negative heights whose evaluation score is
    (hopefully) better than 1.503.  The function runs for at most
    ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin
    n = length

    # 1. start from the arcsine sequence, normalised to Σ = sqrt(2·n)
    a = _arcsine_coeffs(n)
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())

    best_seq = a.copy()
    best_score = _fast_score(best_seq)
    print(f"[init] n={n}, score={best_score:.6f}")

    # parameters for the convex‑combination loop
    t = 0.20               # step size for mixing
    iter_no = 0
    max_iter = 15

    while time.time() < deadline and iter_no < max_iter:
        iter_no += 1

        # 2. build convolution matrix for current f (=a)
        A = _build_conv_matrix(a)

        # 3. current maximal self‑convolution value (the bound M)
        M = np.max(np.convolve(a, a))

        # 4. linear programme: maximise Σ b_i  s.t. A @ b ≤ M, b ≥ 0
        c = -np.ones(n)                     # linprog minimises, so -Σb_i
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print("[lp] linprog failed – terminating.")
            break

        sum_b = -res.fun
        # no improvement possible?
        if sum_b <= target_sum * (1.0 + 1e-9):
            print("[lp] no increase in ℓ₁‑norm – stop.")
            break

        # 5. scale solution to the required ℓ₁‑norm
        g = (target_sum / sum_b) * res.x

        # 6. convex combination (mix) and re‑normalise (numerical safety)
        a_new = (1.0 - t) * a + t * g
        a_new *= target_sum / a_new.sum()       # keep Σ = sqrt(2·n)

        # 7. score of the new candidate
        new_score = _fast_score(a_new)
        print(f"[iter {iter_no:2d}] t={t:.3f}, M={M:.6f}, sum_b={sum_b:.3f}, score={new_score:.6f}")

        # 8. keep the best sequence seen
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"[improve] new best score {best_score:.6f}")

        # 9. adapt the mixing parameter – smaller steps as we approach convergence
        if new_score >= best_score - 1e-6:
            t = max(t * 0.7, 0.02)
        else:
            t = min(t * 1.05, 0.5)

        a = a_new

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
