Search for a step‑function heights vector with minimal evaluation score.

The routine uses an Adam‑optimised projected gradient descent on a
soft‑max approximation of the max‑convolution, combined with random
re‑starts and occasional Gaussian perturbations.  The current best
vector (in probability simplex) is kept and returned as a plain list

import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

"""
Search for a step‑function (list of non‑negative heights) that minimises

    M = 2·n·max(conv(a,a)) / (sum(a)**2)

The evaluation is scale‑invariant, therefore we keep the vector on the
probability simplex (non‑negative, sum = 1) and minimise
    2·n·max(conv(p,p)).
The algorithm consists of several independent Adam‑optimised projected
gradient runs.  Each run starts from a different seed (random or the
previous best construction).  The objective is smoothed with a
log‑sum‑exp (soft‑max) and the smoothing parameter is slowly increased.
Occasional Gaussian jitter is added to escape plateaus.  The best vector
found before the time budget expires is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x | x_i >= 0, sum x_i = 1}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)           # 1‑based index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Normalise a raw non‑negative vector to a probability distribution
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s
