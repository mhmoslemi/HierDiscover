def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, n+1) > 0)[0][-1]

def project_to_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho
    rho_arr = u + (1 - cssv) / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) with a low evaluation:
2·n·max(conv(a,a)) / (sum(a)^2).

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1).  If a previous construction
   (`height_sequence_1`) is available it is used, otherwise an edge‑heavy
   sinusoidal shape is generated.

2. Gradient‑descent phase (≈80 % of the time budget):
   * Use a smooth approximation of the max‑operator:
       m_α(a) = (1/α)·log Σ_k exp(α·conv_k)
   * Its gradient is  g = 2·corr(w, a)  where
       w_k = exp(α·conv_k) / Σ_j exp(α·conv_j)      (soft‑max weights)
   * Perform Adam updates on `a`, then project back onto the probability
     simplex (non‑negative, sum = 1).  The smoothing parameter α grows
     linearly from 5 to 200, so the optimisation starts smooth and ends
     sharply focused on the true maximum convolution entry.
   * Every few hundred iterations the true evaluation (via `evaluate_sequence`)
     is recorded; the best vector is kept.

3. Short simulated‑annealing refinement (remaining time):
   * Fast O(n) convolution updates (`apply_move`) allow many cheap moves.
   * Moves are either random or “targeted’’ – the pair that contributes most
     to the current maximal convolution entry is chosen as source.
   * A standard Metropolis acceptance rule with a linear temperature schedule
     is used; improvements are always accepted.
   * The best vector seen during this phase updates the global best.

4. Return the best sequence found (as a plain list of floats).
"""

import numpy as np
import time
import math

# ------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x) = 1}
# ------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Fast convolution update when moving a mass δ from i to j
# ------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].
    Both arrays are copied; the caller should replace the originals.
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta
    dj = +delta

    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    conv_new = conv.copy()

    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]
    a_q = a[q]

    # contributions of i and j with all other indices
    conv_new[i + q] += 2.0 * di * a_q
    conv_new[j + q] += 2.0 * dj * a_q

    # self‑pair contributions
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ------------------------------------------------------------
# Choose a source index most responsible for the current peak
# ------------------------------------------------------------
def targeted_move(a: np.ndarray, conv: np.ndarray):
    """
    Return (src, dst) where src is the index that most contributes
    to the current maximal convolution entry and dst is a random
    destination different from src.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))               # location of the current peak

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_candidates = np.arange(i_min, i_max + 1, dtype=int)
    j_candidates = k - i_candidates

    contrib = a[i_candidates] * a[j_candidates]
    src = int(i_candidates[np.argmax(contrib)])

    # random destination distinct from src
    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1
    return src, dst

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # ------------------------------------------------------------------

    # ------------------------------------------------------------
    # Gradient‑descent phase (Adam + soft‑max approximation)
    # ------------------------------------------------------------
    # Adam moments

    beta2 = 0.999

        conv = np.convolve(a, a)               # length 2n‑1

        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        if now >= deadline:

        else:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

        max_delta = a[src] * 0.5
        if max_delta < 1e-12:

    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_a.tolist()
