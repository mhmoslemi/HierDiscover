from scipy.optimize import linprog

def solve_lp_max_sum(f, M):
    L = len(f)
    # Build constraint matrix A_ub (size (2L-1) x L)
    # For each k, we need sum_i f_i * b_{k-i} <= M
    # Equivalent to sum_{j} f_{k-j} * b_j <= M, with f_{something}=0 if out of range.

    rows = []
    for k in range(2*L-1):
        row = np.zeros(L)
        # j from max(0, k-(L-1)) to min(L-1, k)
        i_start = max(0, k-(L-1))
        i_end = min(L-1, k)
        # b_j = b_{j}
        # term f_{k - j}
        for j in range(i_start, i_end+1):
            row[j] = f[k - j]
        rows.append(row)
    A_ub = np.array(rows)
    b_ub = np.full(2*L-1, M)
    c = -np.ones(L)  # maximize sum => minimize -sum
    # bounds for each variable b_j >= 0
    bounds = [(0, None)] * L
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback to zero vector
        return np.zeros_like(f)
    return res.x
```

"""Search for a step sequence with evaluation < 1.503.

Strategy:
1. Start from the analytic (1‑x)^{-1/2} coefficients (the “square‑root”
   series).  This sequence is known to give the asymptotic bound π/2.
2. Apply a Frank‑Wolfe like iteration:  for the current sequence `f`
   compute `M = max(f*f)`.  Solve the linear program

       maximise   sum b_j
       subject to (f * b)_k ≤ M  for all k,
                 b_j ≥ 0

   The LP finds a direction `g` that can be mixed with `f` while keeping
   the maximal convolution ≤ M.  Updating `f ← (1‑t) f + t g` with a
   small step size strictly reduces the objective.
3. After a few LP steps the sequence is refined with a projected
   gradient descent (the method used previously) to fine‑tune the
   coefficients.
The whole routine runs under the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^{-1/2}, normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (0.5 + k - 1) / k   # α = 1/2
    w /= w.sum()
    return w

def _max_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a*a)_k using an FFT (O(n log n))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _build_constraint_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the matrix A such that (A @ b)_k = (f * b)_k,
    i.e. A[k, j] = f[k - j] when the index is valid, otherwise 0.
    The matrix has shape (2*len(f)-1, len(f)) and is very sparse.
    """
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    idx = 0
    for j in range(n):
        rows[idx : idx + n] = np.arange(j, j + n)          # row indices
        cols[idx : idx + n] = j                           # column index repeated
        data[idx : idx + n] = f[j]                        # same coefficient in the whole column
        idx += n
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

def _lp_step(f: np.ndarray, M: float, step: float) -> np.ndarray:
    """
    Perform one Frank‑Wolfe style iteration.
    Returns the updated sequence (f ← (1‑step) f + step·g) where g is the
    normalised solution of the linear program.
    """
    n = f.shape[0]
    A = _build_constraint_matrix(f)               # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)              # maximise sum(b) == minimise -sum(b)
    bounds = [(0, None)] * n

    # Solve the LP with HiGHS; it accepts a sparse A_ub.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        # fallback – keep the current sequence unchanged
        return f.copy()

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        g = f.copy()
    else:
        g = b / sum_b                               # normalise to sum = 1

    # convex combination
    f_new = (1.0 - step) * f + step * g
    # re‑project onto the simplex to guard against tiny numerical drift
    f_new = _project_simplex(f_new)
    return f_new

def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected‑gradient step on the objective 2·n·max(a*a).
    The gradient is taken only on the active (maximal) convolution indices.
    """
    n = a.shape[0]
    # compute convolution via FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = conv.max()
    max_idx = np.flatnonzero(conv >= max_val - 1e-14)

    # sub‑gradient of the max‑convolution
    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    if len(max_idx) > 1:
        grad /= len(max_idx)

    a_new = a - step * grad
    a_new = _project_simplex(a_new)
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises `evaluate_sequence`.
    The routine first runs a few LP‑based Frank‑Wolfe steps starting from
    the analytic square‑root series, followed by a projected gradient descent
    to polish the result.  It respects the 1000 s time budget and returns the
    best sequence found.
    """
    np.random.seed(seed)
    n = kwargs.get("length", 5808)        # length of the step function
    deadline = time.time() + budget_s - 0.5

    # start from the analytic (1‑x)^{-1/2} series
    f = _sqrt_series(n)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, evaluation={best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Frank‑Wolfe / LP based improvement
    # --------------------------------------------------------------
    # Use a modest number of LP iterations; each iteration costs O(n²)
    # but stays well within the time budget for n≈2000–3000.
    # For very large n we reduce the dimension for the LP phase.
    # Here we adapt automatically: if n>2500 we work on a down‑sampled
    # copy, later up‑sample back to the original size.
    if n > 2500:
        # down‑sample to about 2000 points (preserve shape by linear interpolation)
        small_n = 2000
        x_small = np.linspace(0, 1, small_n)
        x_big = np.linspace(0, 1, n)
        f_small = np.interp(x_small, x_big, f)
        f_small = _project_simplex(f_small)   # ensure sum=1
        working_f = f_small
        working_n = small_n
    else:
        working_f = f
        working_n = n

    # LP iterations
    lp_iters = 0
    step_lp = 0.4          # relatively aggressive mixing factor
    while time.time() < deadline and lp_iters < 15:
        M = _max_convolution(working_f)
        working_f = _lp_step(working_f, M, step_lp)
        # evaluation on the *original* length (upsample if needed)
        if n != working_n:
            # up‑sample back to original size for evaluation
            x_small = np.linspace(0, 1, working_n)
            x_big = np.linspace(0, 1, n)
            candidate = np.interp(x_big, x_small, working_f)
            candidate = _project_simplex(candidate)
        else:
            candidate = working_f.copy()
        cur_score = evaluate_sequence(candidate.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = candidate.copy()
            print(f"[LP {lp_iters:2d}] improved evaluation = {cur_score:.6f}")
        else:
            print(f"[LP {lp_iters:2d}] evaluation = {cur_score:.6f}")
        # gradually reduce the mixing step to stabilise
        step_lp = max(step_lp * 0.85, 0.05)
        lp_iters += 1

    # --------------------------------------------------------------
    # 2) Projected gradient descent fine‑tuning
    # --------------------------------------------------------------
    # Use the best sequence found so far as a fresh start.
    a = best_seq.copy()
    step0 = 0.15
    decay = 5e-5
    iter_cnt = 0
    no_improve = 0

    while time.time() < deadline:
        a = _gradient_step(a, step0 / (1.0 + decay * iter_cnt))
        # occasional tiny jitter to leave plateaus
        if iter_cnt % 500 == 0 and iter_cnt > 0:
            jitter = np.random.randn(n) * (1e-5 / np.sqrt(iter_cnt + 1))
            a = _project_simplex(a + jitter)

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[GD {iter_cnt:5d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        if no_improve >= 3000:
            # restart from the current best with a small random perturbation
            a = best_seq.copy()
            a = _project_simplex(a + np.random.randn(n) * 1e-3)
            no_improve = 0
            print(f"[GD {iter_cnt:5d}] restart triggered")
        iter_cnt += 1

    print(f"[final] best evaluation = {best_score:.6f} after {iter_cnt} gradient steps")
    return best_seq.tolist()

   sequence observed.

The algorithm makes use of SciPy’s HiGHS linear‑programming solver and
NumPy’s FFT for fast convolutions.  It works for any length; for very
large lengths the LP phase is carried out on a down‑sampled copy and the
result is interpolated back to the original size.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# --------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Analytic square‑root series (α = ½)
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (0.5 + k - 1) / k   # recurrence for (1‑x)^‑½
    w /= w.sum()
    return w


# --------------------------------------------------------------
# Maximal value of the self‑convolution (FFT, O(n log n))
# --------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# --------------------------------------------------------------
# Build the (f * ·) matrix for the LP constraints
# --------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Construct A ∈ ℝ^{(2n‑1)×n} with
        A[k, j] = f[k‑j]   if 0 ≤ k‑j < n,
                 0          otherwise,
    i.e. (A @ b)_k = (f * b)_k.
    """
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    idx = 0
    for j in range(n):
        rows[idx: idx + n] = np.arange(j, j + n)    # rows i = j … j+n‑1
        cols[idx: idx + n] = j                     # column fixed
        data[idx: idx + n] = f[j]                  # same coefficient in the whole column
        idx += n
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# --------------------------------------------------------------
# One Frank‑Wolfe / LP step
# --------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, step: float) -> np.ndarray:
    """
    Solve
        maximise Σ b_j
        subject to (f * b)_k ≤ M   ∀k,
                   b_j ≥ 0.
    Return the convex combination (1‑step)·f + step·g where g = b / Σb.
    """
    n = f.shape[0]
    A = _build_constraint_matrix(f)                     # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                    # maximise Σb  ⇔ minimise -Σb
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options={"presolve": True})
    if not res.success:
        # fallback – keep the current sequence unchanged
        return f.copy()

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        g = f.copy()
    else:
        g = b / sum_b                                     # normalise to sum = 1

    f_new = (1.0 - step) * f + step * g
    return _project_simplex(f_new)


# --------------------------------------------------------------
# Projected gradient step on the original objective
# --------------------------------------------------------------
def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected‑gradient update for
        F(a) = 2·n·max_k (a*a)_k
    The sub‑gradient is computed only on the active (maximal) indices.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = conv.max()
    max_idx = np.flatnonzero(conv >= max_val - 1e-14)

    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    if len(max_idx) > 1:
        grad /= len(max_idx)

    a_new = a - step * grad
    return _project_simplex(a_new)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative height sequence that minimises the supplied
    `evaluate_sequence` function.  The routine follows the strategy
    described in the module docstring and respects the given time budget.
    """
    np.random.seed(seed)
    n = kwargs.get("length", 5808)          # requested length of the step function
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 0. initialise with the analytic square‑root series
    # ------------------------------------------------------------------
    f_full = _sqrt_series(n)
    best_seq = f_full.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, evaluation={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Frank‑Wolfe / LP phase (optional down‑sampling for very large n)
    # ------------------------------------------------------------------
    # If n is large we solve the LP on a down‑sampled copy to keep the
    # memory footprint reasonable.
    if n > 2500:
        # down‑sample to ≈2000 points, preserving the shape by linear interpolation
        small_n = 2000
        x_big = np.linspace(0.0, 1.0, n)
        x_small = np.linspace(0.0, 1.0, small_n)
        f = np.interp(x_small, x_big, f_full)
        f = _project_simplex(f)                 # ensure exact sum = 1
        working_n = small_n
    else:
        f = f_full.copy()
        working_n = n

    lp_iter = 0
    step_lp = 0.4                               # initial mixing factor
    while time.time() < deadline and lp_iter < 15:
        M = _max_convolution(f)
        f = _lp_step(f, M, step_lp)

        # evaluate the (up‑sampled) candidate on the original size
        if working_n != n:
            candidate = np.interp(x_big, x_small, f)
            candidate = _project_simplex(candidate)
        else:
            candidate = f.copy()

        cur_score = evaluate_sequence(candidate.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = candidate.copy()
            print(f"[LP {lp_iter:02d}] improvement → {cur_score:.6f}")
        else:
            print(f"[LP {lp_iter:02d}] score = {cur_score:.6f}")

        # gradually shrink the mixing step to stabilise the process
        step_lp = max(step_lp * 0.85, 0.05)
        lp_iter += 1

    # ------------------------------------------------------------------
    # 2. Projected gradient descent fine‑tuning on the full size
    # ------------------------------------------------------------------
    a = best_seq.copy()
    step0 = 0.15
    decay = 5e-5
    iter_cnt = 0
    no_improve = 0

    while time.time() < deadline:
        step = step0 / (1.0 + decay * iter_cnt)
        a = _gradient_step(a, step)

        # occasional tiny random jitter to escape flat plateaus
        if iter_cnt % 500 == 0 and iter_cnt > 0:
            jitter = np.random.randn(n) * (1e-5 / np.sqrt(iter_cnt + 1))
            a = _project_simplex(a + jitter)

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[GD {iter_cnt:05d}] new best = {cur_score:.6f}")
        else:
            no_improve += 1

        # simple restart if we have been stuck for a while
        if no_improve >= 3000:
            a = best_seq.copy()
            a = _project_simplex(a + np.random.randn(n) * 1e-3)
            no_improve = 0
            print(f"[GD {iter_cnt:05d}] restart triggered")

        iter_cnt += 1

    print(f"[final] best evaluation = {best_score:.6f} after {iter_cnt} GD steps")
    return best_seq.tolist()
