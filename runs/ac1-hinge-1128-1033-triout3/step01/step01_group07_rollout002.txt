"""
Enhanced core‑tail construction for the step‑function attack.

The idea is classical: a short “core’’ following the optimal
a_i = 1/√i has an (almost) constant self‑convolution M ≈ π.
If we attach a long constant “tail’’ of height δ, the sum of the
coefficients grows while the maximum of the convolution does not
increase as long as

    δ·S_core  ≤  M                (cross term)
    T·δ²      ≤  M                (tail self‑convolution)

where S_core = Σ a_i and T is the tail length.
Choosing δ = M / S_core and T = ⌊S_core² / M⌋ satisfies both
constraints with equality, giving a total ℓ¹‑norm 2·S_core and a
max‑convolution equal to M.  For the 1/√i core M ≈ π, this yields the
theoretical bound

    score = 2·n·M / (Σa)²  ≈  M/8 + ½  ≈ 0.89 .

The optimiser enumerates several core lengths, computes the exact
M using an FFT‑based convolution, builds the optimal tail, evaluates
the full sequence, and keeps the best one.  A short simulated‑annealing
phase prunes any remaining local irregularities.

The function ``propose_candidate`` returns the best sequence found
within the allotted time (≈1000 s).  All values are non‑negative,
well below the 1000‑cap of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – required by the specification)
# ----------------------------------------------------------------------


def _conv_max(seq: np.ndarray) -> float:
    """
    Return the maximum value of the linear convolution of *seq* with itself.
    An FFT of size a power‑of‑two is used for speed and numerical stability.
    """
    n = seq.shape[0]
    # size ≥ 2·n‑1, power of two
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a given sequence.
    Mirrors the logic of ``evaluate_sequence`` but is faster for bulk testing.
    """
    total = float(seq.sum())
    if total < 0.01:
        return np.inf
    max_conv = _conv_max(seq)
    n = seq.shape[0]
    return 2.0 * n * max_conv / (total * total)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Construct a step‑function sequence using the optimal 1/√i core
    and a constant tail.  The core length is swept, the tail is set
    analytically (δ = M / S_core, T = ⌊S_core² / M⌋), and a short
    simulated‑annealing refinement explores neighbouring configurations.
    The best sequence found before the time limit is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0) optional baseline from a previously discovered construction
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    if "height_sequence_1" in globals():
        try:
            cand = np.array(globals()["height_sequence_1"], dtype=np.float64)
            val = _score(cand)
            best_seq, best_val = cand.copy(), val
            print(f"[baseline] height_sequence_1 score = {best_val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) Core‑tail grid search
    # ------------------------------------------------------------------
    # We limit the total length to keep the FFT fast.
    MAX_TOTAL_LEN = 20000

    # Core lengths to try – dense in the interesting region.
    core_lengths = np.unique(
        np.linspace(200, 8000, num=150, dtype=int)
    ).astype(int)

    for C in core_lengths:
        if time.time() > deadline:
            break
        if C < 2:
            continue

        # ---- core ----------------------------------------------------
        core = 1.0 / np.sqrt(np.arange(1, C + 1, dtype=np.float64))
        M = _conv_max(core)               # max self‑convolution of the core
        S = float(core.sum())             # ℓ¹‑norm of the core

        # ---- tail size bound (T ≤ S² / M) ----------------------------
        T_opt = int(S * S / M)

        #   T·δ² ≤ M         (tail self‑convolution)
        # The safe choice is the minimum of the two, with a tiny safety factor.
        delta_cross = M / S
        delta_tail = np.sqrt(M / T_opt)
        delta = min(delta_cross, delta_tail) * 0.999  # slight under‑estimate

        tail = np.full(T_opt, delta, dtype=np.float64)
        seq = np.concatenate([core, tail])

        val = _score(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(
                f"[grid] core={C}, tail={T_opt}, δ={delta:.3e} → score={best_val:.6f}"
            )

    # ------------------------------------------------------------------
    # 2) Light simulated‑annealing refinement around the current best
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fall back to a simple uniform sequence if everything failed.
        n = 8000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _score(best_seq)

    cur_seq = best_seq.copy()
    cur_val = best_val
    start = time.time()
    # Initial temperature proportional to the current score
    T0 = max(1e-4, 0.05 * best_val)

    while time.time() < deadline:
        n = cur_seq.shape[0]

        # Small random move: transfer a tiny amount from one index to another.
        i, j = np.random.randint(n, size=2)
        if i == j or cur_seq[i] <= 0:
            continue
        amount = cur_seq[i] * np.random.rand() * 0.005   # ≤0.5 % move
        new_seq = cur_seq.copy()
        new_seq[i] -= amount
        new_seq[j] += amount

        # Ensure non‑negativity (the move cannot create negative numbers)
        if new_seq[i] < 0:
            continue

        new_val = _score(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T0):
            cur_seq, cur_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
                print(f"[SA] improved → score={best_val:.6f}")

        # Linear cooling
        elapsed = time.time() - start
        T0 = max(1e-9, T0 * (1.0 - elapsed / budget_s))

    print(f"final best score = {best_val:.6f}")
    # Return a plain Python list as required by the evaluation harness.
    return best_seq.tolist()
