def _max_convolution(seq):
    seq_arr = np.asarray(seq, dtype=np.float64)
    n = len(seq_arr)
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq_arr, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)
    return np.max(conv[:2*n-1])

"""
Propose a better step‑function sequence.

The algorithm starts from the classic arcsine (inverse‑square‑root) series,
which gives a convolution peak M = 1 and a score of π/2 ≈ 1.5708.
To push the peak below 1 we iterate the following two‑phase scheme
(ideas taken from the LP‑based “active‑set” method described in the
statement):

1. **Linear‑programming improvement**  
   With the current sequence `f` (length n) and its maximal convolution
   value `M = max(f * f)`, solve the LP

        maximise   Σ b_i
        subject to (f * b)_k ≤ M   for all k = 0 … 2n‑2
                  b_i ≥ 0

   The feasible region contains the current `f` (take b = f).  The
   optimiser pushes mass towards indices where `f_i` is small, thereby
   increasing the total sum Σb while keeping the mixed convolution
   `f * b` bounded by the same peak `M`.

2. **Convex combination**  
   Scale the LP solution `b` to the original total sum
   `S = Σ f_i`   →   `g = (S / Σb) * b`.  
   For a step size `t∈(0,1]` consider `f' = (1‑t) f + t g`.  Because
   `f * g ≤ M` (by construction) and `t` can be chosen small enough,
   the new peak `M' = max(f' * f')` never exceeds `M`.  If the
   resulting score improves we accept the update and repeat.

The loop runs until the time budget is exhausted (or no further
improvement is possible).  After the LP phase a cheap stochastic
mass‑transfer refinement is performed to squeeze a little more out of
the current best sequence.

All operations respect the non‑negativity and clipping rules of
`evaluate_sequence`.  The function returns the best list found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _arcsine_coeffs(n: int):
    """First `n` coefficients of (1‑z)^(-1/2) = Σ binom(2i,i)/4^i z^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _max_convolution(seq: np.ndarray) -> float:
    """Return max(f * f) using a real FFT (fast for modest n)."""
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    return float(np.max(conv))


def _score(seq: np.ndarray) -> float:
    """Exact evaluation used by the competition (no clipping needed)."""
    n = seq.shape[0]
    max_b = _max_convolution(seq)
    s = float(seq.sum())
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _build_conv_matrix(f: np.ndarray):
    """
    Build the Toeplitz matrix A of size (2n‑1)×n such that A @ b = f * b.
    Row `k` (0‑based) contains f_j exactly when 0 ≤ k‑j < n.
    """
    n = f.shape[0]
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    # each column j is f shifted down by j
    for j in range(n):
        A[j : j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     init_len: int = 400) -> list[float]:
    """
    Search for a low‑score non‑negative step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    init_len : int
        Length of the initial arcsine sequence (trade‑off between speed
        and quality; 400–600 is usually a good choice).

    Returns
    -------
    list[float]
        The best sequence found before the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1.  Deterministic start – arcsine series (already very good)
    # --------------------------------------------------------------
    f = _arcsine_coeffs(init_len).astype(np.float64)
    n = f.shape[0]
    max_f = _max_convolution(f)
    best_seq = f.copy()
    best_score = _score(best_seq)
    print(f"[init] n={n}, max={max_f:.6f}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  LP‑based improvement loop
    # --------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # 2a) build convolution matrix for the current f
        A = _build_conv_matrix(f)               # shape (2n‑1, n)
        K = A.shape[0]
        M = max_f                               # current peak

        # 2b) linear program: maximise sum(b)  s.t. A @ b ≤ M , b ≥ 0
        c = -np.ones(n)                         # linprog minimises → -sum(b)
        b_ub = M * np.ones(K)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[lp] iteration {iteration} – LP failed, stopping.")
            break

        b = res.x
        sum_b = b.sum()
        sum_f = f.sum()

        # No increase in total mass ⇒ cannot improve further
        if sum_b <= sum_f * (1.0 + 1e-12):
            print(f"[lp] iteration {iteration} – no sum increase, stopping.")
            break

        # 2c) scale to keep total sum unchanged
        scale = sum_f / sum_b
        g = b * scale

        # 2d) try convex combinations with decreasing step size
        t = 0.5
        improved = False
        while t >= 1e-3:
            candidate = (1.0 - t) * f + t * g
            cand_max = _max_convolution(candidate)

            # accept only if the peak does not grow (monotone) and score improves
            if cand_max <= M + 1e-12:
                cand_score = _score(candidate)
                if cand_score < best_score - 1e-12:
                    f = candidate
                    max_f = cand_max
                    best_seq = candidate.copy()
                    best_score = cand_score
                    print(f"[lp‑step] it={iteration} t={t:.4f} "
                          f"max={max_f:.6f} score={best_score:.6f}")
                    improved = True
                    break
            t *= 0.5

        if not improved:
            # could not find a beneficial mixing – terminate LP phase
            print(f"[lp] iteration {iteration} – mixing failed, stopping.")
            break

    # --------------------------------------------------------------
    # 3.  Light stochastic refinement (mass transfer) to polish result
    # --------------------------------------------------------------
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        src = f[i]
        if src <= 0.0:
            continue
        # transfer a tiny exponentially distributed amount of mass
        delta = np.random.exponential(scale=1e-3) * src
        if delta > src:
            continue
        new_f = f.copy()
        new_f[i] -= delta
        new_f[j] += delta
        new_score = _score(new_f)
        if new_score < best_score:
            f = new_f
            best_seq = new_f.copy()
            best_score = new_score
            max_f = _max_convolution(f)   # keep the peak up‑to‑date
            print(f"[refine] improved score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
