    p0 = w / w.sum()

    if rng.random() < 0.3:
        p = p0
    else:

        p = p/p.sum()
    p = (p + p[::-1]) / 2  # symmetrize

    best_p = p.copy()
    best_score = 2.0 * n * np.max(np.convolve(p, p))
    # Record for printing

    # Adam parameters
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8

    # schedule parameters
    beta = 0.5  # smoothing (inverse of prev alpha, maybe)
    beta_mul = 1.005

    lr = 0.3  # initial learning rate for Adam (initial step size)
    t = 0

    deadline = time.time() + budget_s - 5

    while time.time() < deadline:
        t += 1
        # conv
        conv = np.convolve(p, p)

        # compute softmax weights for smoothing
        # scale conv to improve numerical stability
        max_conv = conv.max()
        # using scaled conv values
        z = beta * (conv - max_conv)  # scaled; large negative values -> exp small
        # compute softmax w over length 2n-1
        exp_z = np.exp(z)
        w = exp_z / exp_z.sum()  # shape (2n-1,)

        # correlation of p with reversed w
        w_rev = w[::-1]
        # correlation length: 3n-2; we want indices idx = 2*n-2 - np.arange(n)
        corr = np.convolve(p, w_rev)  # length 3n-2

        idx = 2*n - 2 - np.arange(n)
        grad = 2.0 * corr[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad

  does not change the quality of a solution.

* The optimisation problem is non‑convex.  We use a *continuation*
  strategy: start with a very smooth approximation of the max‑operator
  (soft‑max with a low temperature) and gradually tighten it.  The
  gradient of the smoothed objective can be expressed by a single
  correlation, which we evaluate with ``np.convolve``.

* To stay inside the simplex we apply the Euclidean projection of
  Duchi et al. after every update.  The update itself is performed with
  the Adam optimiser (adaptive step sizes, momentum) – this is much
  more robust than a plain gradient step.

* All iterates are forced to be symmetric (a_i = a_{n‑1‑i}); symmetry
  is optimal for the problem and reduces the effective search space.

* Several restarts are carried out (random Dirichlet start, an
  “arcsine‑shaped” start and, if available, the best known construction
  ``height_sequence_1``).  The best sequence seen over the whole time
  budget is returned.

* After the main Adam loop a cheap local‑search phase (random mass
  transfers between two coordinates) tries to squeeze a few more
  improvements.

The routine stops a few seconds before the supplied ``budget_s`` to
avoid a hard timeout.  The returned list can be fed directly to the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project ``v`` onto {x | x_i ≥ 0 , Σ x_i = 1}."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises
        M = 2·n·max(conv(a,a)) / (sum(a)**2)

    The search runs several Adam‑optimised restarts with a smooth‑max
    continuation, keeps the iterates symmetric and projects after each
    update.  A lightweight random‑swap refinement is performed at the
    end.  The best sequence discovered before the time budget expires
    is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size  (use the known construction length when possible)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name‑defined]
    except Exception:
        n = 1153  # default length used in the benchmark

    # ------------------------------------------------------------------
    # 2. Some useful constant data
    # ------------------------------------------------------------------
    idx = 2 * n - 2 - np.arange(n)               # indices needed for the gradient
    rev_range = np.arange(n - 1, -1, -1)         # for symmetrisation

    # ------------------------------------------------------------------
    # 3. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    adam_eps = 1e-8
    base_lr = 0.5          # initial learning‑rate for Adam
    temp_start = 0.5      # start temperature (soft‑max smoothing)
    temp_mul = 1.005      # geometric increase per iteration
    jitter_interval = 2500
    jitter_scale = 0.003

    # ------------------------------------------------------------------
    # 4. Containers for the overall best solution
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 5. Function that evaluates the (unnormalised) objective
    # ------------------------------------------------------------------
    def _score(p: np.ndarray) -> float:
        """Return 2·n·max(conv(p,p)).  ``p`` is assumed to sum to 1."""
        return 2.0 * n * np.max(np.convolve(p, p))

    # ------------------------------------------------------------------
    # 6. Main loop – several independent Adam restarts
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # safety margin

    # Different ways to initialise a feasible probability vector
    init_strategies = []

    # a) the previously known construction if it exists
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name‑defined]
        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total > 0.0:
            init_strategies.append(seq / total)

    # b) an arcsine‑like “U‑shape” (heavy at the ends, light in the middle)
    i = np.arange(n, dtype=float)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    init_strategies.append(arcsine / arcsine.sum())

    # c) pure uniform start
    init_strategies.append(np.full(n, 1.0 / n))

    # d) several random Dirichlet initialisations (added later on the fly)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 6.1 Choose an initial point
        # --------------------------------------------------------------
        if rng.random() < 0.5 and init_strategies:
            p = rng.choice(init_strategies)
        else:
            p = rng.random(n)
            p = p / p.sum()

        # enforce symmetry right away
        p = (p + p[::-1]) * 0.5
        p = _simplex_projection(p)

        # Adam state
        m = np.zeros_like(p)
        v = np.zeros_like(p)

        # per‑run temperature and learning‑rate
        temp = temp_start
        lr = base_lr

        # iteration counter for this run
        t = 0

        # each restart runs until the global deadline or until a small
        # per‑run cutoff (30 s) is reached – this keeps the total number
        # of restarts reasonable.
        run_deadline = time.time() + min(30.0, deadline - time.time())

        while time.time() < run_deadline:
            t += 1

            # ----------------------------------------------------------
            # 6.2 Compute self‑convolution and the soft‑max weights
            # ----------------------------------------------------------
            conv = np.convolve(p, p)                # length 2n‑1
            conv_max = conv.max()
            # soft‑max (stabilised)
            z = temp * (conv - conv_max)
            exp_z = np.exp(z)
            w = exp_z / exp_z.sum()                 # length 2n‑1, sums to 1

            # ----------------------------------------------------------
            # 6.3 Gradient of the smoothed objective
            # ----------------------------------------------------------
            w_rev = w[::-1]
            corr = np.convolve(p, w_rev)            # length 3n‑2
            grad = 2.0 * corr[idx]                  # shape (n,)

            # ----------------------------------------------------------
            # 6.4 Adam update (projected onto the simplex)
            # ----------------------------------------------------------
            m = adam_beta1 * m + (1.0 - adam_beta1) * grad
            v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)

            m_hat = m / (1.0 - adam_beta1 ** t)
            v_hat = v / (1.0 - adam_beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            p_new = p - step

            # enforce symmetry after the gradient step
            p_new = (p_new + p_new[::-1]) * 0.5

            # projection onto the probability simplex
            p_new = _simplex_projection(p_new)

            # occasional jitter to escape shallow minima
            if t % jitter_interval == 0:
                jitter = rng.normal(scale=jitter_scale, size=n)
                p_new = _simplex_projection(p_new + jitter)

            p = p_new

            # ----------------------------------------------------------
            # 6.5 Keep the best vector seen so far
            # ----------------------------------------------------------
            cur_score = _score(p)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.copy()
                print(f"[{time.time():.2f}] Restart step {t:5d} – new global best: {best_score:.6f}")

            # ----------------------------------------------------------
            # 6.6 Parameter schedules
            # ----------------------------------------------------------
            lr *= 0.9995        # gentle decay
            temp *= temp_mul    # tighten the soft‑max

            # early exit if we already beat the target
            if best_score <= 1.503 + 1e-7:
                break

        # break outer loop early if target reached
        if best_score <= 1.503 + 1e-7:
            break

    # ------------------------------------------------------------------
    # 7. Local refinement – tiny random mass transfers
    # ------------------------------------------------------------------
    if best_seq is not None:
        p = best_seq.copy()
        cur_score = best_score
        # give the refinement about 10 % of the remaining time
        refine_deadline = time.time() + max(0.0, deadline - time.time())
        max_step = 0.02        # maximal transferred mass in one move

        while time.time() < refine_deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps = rng.random() * max_step
            eps = min(eps, p[i])          # cannot take more than we have
            candidate = p.copy()
            candidate[i] -= eps
            candidate[j] += eps

            # keep symmetry (the move is mirrored)
            candidate[i] = (candidate[i] + candidate[n - 1 - i]) * 0.5
            candidate[n - 1 - i] = candidate[i]
            candidate[j] = (candidate[j] + candidate[n - 1 - j]) * 0.5
            candidate[n - 1 - j] = candidate[j]

            candidate = _simplex_projection(candidate)

            cand_score = _score(candidate)
            if cand_score < cur_score:
                p = candidate
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = p.copy()
                print(f"[{time.time():.2f}] Refinement – new best: {best_score:.6f}")

                # reduce max_step slowly to focus the search
                max_step *= 0.99

            # stop early if we are essentially at the theoretical limit
            if best_score <= 1.503 + 1e-7:
                break

    # ------------------------------------------------------------------
    # 8. Return the best sequence (clipped to the limits enforced by the
    #    evaluator).  The values are already non‑negative and well below
    #    1000, but we apply the clip for safety.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – never happened, but keep the API contract safe
        best_seq = np.full(n, 1.0 / n)

    result = np.clip(best_seq, 0.0, 1000.0).tolist()
    # Optional: print the final objective as evaluated by the provided checker
    try:
        # the global ``evaluate_sequence`` is assumed to be available
        final_val = evaluate_sequence(result)  # type: ignore[name‑defined]
        print(f"Final returned sequence – evaluate_sequence() = {final_val:.6f}")
    except Exception:
        pass

    return result
```
