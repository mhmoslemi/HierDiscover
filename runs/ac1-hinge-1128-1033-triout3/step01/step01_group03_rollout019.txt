    a = _sqrt_series(n)

    # annealing parameters
    start_temp = 1.0
    final_temp = 1e-4
    start_eps = 0.5

        ratio = elapsed / budget_s
        temp = start_temp * (final_temp / start_temp) ** ratio
        eps = start_eps * (final_eps / start_eps) ** ratio

        delta = np.random.randn(n) * eps
        a_new = a + delta
        # project onto simplex
        a_new = _project_simplex(a_new)

        # evaluate
        conv = np.convolve(a_new, a_new)
        new_score = float(2 * n * conv.max() / (a_new.sum()**2))

        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(temp, 1e-12)):

                best_score = new_score
                best_a = a_new.copy()
                print(f"[{iter_cnt}] new best: {best_score:.6f}")

    print(f"Finished after {iter_cnt} iterations, best evaluation {best_score:.6f}")

"""
Optimization of a non‑negative step‑function via Simulated Annealing.

The objective evaluated by `evaluate_sequence` is

    F(a) = 2·n·max_k (a∗a)_k / (sum(a)^2) ,

which is scale‑invariant, so we keep the L1‑norm equal to one.
A good starting point is the analytically motivated “square‑root’’ series
(a_k ∝ binom(2k,k)/4^k).  From there a simple simulated‑annealing loop
perturbs the coefficients, projects the perturbed vector onto the probability
simplex and accepts the move according to a temperature schedule.
The algorithm keeps the best sequence observed and stops before the 1000 s
budget expires.

The implementation uses only NumPy and the supplied `evaluate_sequence`,
runs in < 1 GB memory and respects the 1‑CPU constraint.
"""
import time
import numpy as np

# --------------------------------------------------------------
# Helper: project a vector onto the probability simplex
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex {x ≥ 0, Σ x = 1} (Duchi et al., 2008)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Initialise the “square‑root’’ sequence (coefficients of (1‑x)^‑½)
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^‑½, normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# --------------------------------------------------------------
# Main candidate generator
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Simulated‑annealing search for a low‑evaluation step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function returns before this deadline.
    **kwargs : dict
        Optional ``length`` key to control the number of coefficients.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, Σ a_i = 1).
    """
    np.random.seed(seed)

    # length of the sequence – the previous best used 5808 items
    n = kwargs.get("length", 5808)

    # ------------------------------------------------------------------
    # 1.  Base solution: the analytic square‑root series
    # ------------------------------------------------------------------
    a = _sqrt_series(n)                # already normalised, sum = 1
    conv = np.convolve(a, a)
    cur_score = float(2 * n * conv.max() / (a.sum() ** 2))
    best_a = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing parameters
    # ------------------------------------------------------------------
    start_temp = 1.0          # initial temperature
    final_temp = 1e-4         # temperature at the end of the budget
    start_eps = 0.5           # initial perturbation scale
    final_eps = 1e-5          # final perturbation scale

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linearised progress in [0,1]
        progress = min(elapsed / budget_s, 1.0)

        # exponential cooling for temperature and step size
        temp = start_temp * (final_temp / start_temp) ** progress
        eps = start_eps * (final_eps / start_eps) ** progress

        # ------------------------------------------------------------------
        # 2a.  Propose a new candidate by a small Gaussian perturbation
        # ------------------------------------------------------------------
        delta = np.random.randn(n) * eps
        a_new = a + delta
        a_new = _project_simplex(a_new)           # enforce non‑negativity & sum = 1

        # ------------------------------------------------------------------
        # 2b.  Evaluate the proposal (using the same formula as evaluate_sequence)
        # ------------------------------------------------------------------
        conv_new = np.convolve(a_new, a_new)
        new_score = float(2 * n * conv_new.max() / (a_new.sum() ** 2))

        # ------------------------------------------------------------------
        # 2c.  Metropolis acceptance
        # ------------------------------------------------------------------
        delta_score = new_score - cur_score
        if delta_score <= 0 or np.random.rand() < np.exp(-delta_score / max(temp, 1e-12)):
            a = a_new
            cur_score = new_score
            # keep the best ever seen
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{it:6d}] new best: {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 3.  Final report
    # ------------------------------------------------------------------
    total_time = time.time() - start_time
    print(f"Search finished after {it} iterations, time = {total_time:.2f}s")
    print(f"Best evaluation observed: {best_score:.6f}")

    # Return a plain Python list as required by the harness
    return best_a.tolist()
