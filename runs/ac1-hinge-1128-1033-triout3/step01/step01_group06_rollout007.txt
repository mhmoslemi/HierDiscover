```
def lp_max_sum(a, M):
    n = a.shape[0]
    # convolution matrix
    col = np.concatenate([a, np.zeros(n-1, dtype=a.dtype)])
    row = np.concatenate([a[:1], np.zeros(n-1, dtype=a.dtype)])
    A = toeplitz(col, row)   # shape (2n-1, n)
    c = -np.ones(n)          # maximize sum b
    ub = np.full(2*n-1, M, dtype=a.dtype)
    # bounds: b>=0
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x
```

"""
Hybrid LP‑annealing optimizer for the step‑function problem.

* Phase 1 – Linear programming:
  For the current vector a (∑a = √(2 n)) we compute M = max(a∗a).
  We solve a linear program that maximises the total mass of a new
  vector b under the linear constraints  a∗b ≤ M (pointwise) and b ≥ 0.
  The optimum b can usually carry more mass than a without increasing the
  peak of the convolution.  After scaling b back to the required normalisation
  we mix a ← (1‑t)·a + t·b (t decreasing).  Repeating this step rapidly
  reduces the peak M.

* Phase 2 – Local refinement:
  Starting from the best vector found by the LP phase we run a short
  simulated‑annealing / fast‑move search very similar to the previous
  baseline, but now the objective is just max(a∗a) because the sum is
  already normalised.  Moves are performed with an O(n) convolution update.

The algorithm respects the 1000 s budget, prints progress and finally
returns the best coefficient list found.  The returned list consists of
non‑negative floats; the evaluation function will automatically clamp
values and compute the required score.
"""

import math
import time
from typing import List

import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution update when moving a mass δ from index i to j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].
    a – 1‑D array of length n, sum(a) = √(2 n)
    conv – convolution a*a (length 2n‑1)
    i, j – distinct indices, 0 ≤ i,j < n
    delta – 0 < delta ≤ a[i]
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta
    dj = +delta

    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    conv_new = conv.copy()

    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]
    a_q = a[q]

    # contributions of the moved entries to all other sums
    conv_new[i + q] += 2.0 * di * a_q
    conv_new[j + q] += 2.0 * dj * a_q

    # self‑pair contributions
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Linear programme: maximise sum(b)  s.t.  a∗b ≤ M  and b ≥ 0
# ----------------------------------------------------------------------
def solve_lp_max_sum(a: np.ndarray, M: float):
    """
    Solve  max 1·b  subject to  a∗b ≤ M (pointwise) ,  b ≥ 0.
    Returns the optimal b (numpy array) or None if the LP fails.
    """
    n = a.shape[0]

    # build the Toeplitz matrix that implements a∗b
    # first column = [a0,…,a_{n-1},0,…,0]   length 2n-1
    col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    # first row = [a0,0,0,…,0]   length n
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    A = toeplitz(col, row)                 # shape (2n‑1) × n

    c = -np.ones(n)                        # maximise sum(b)  →  minimise –sum(b)
    ub = np.full(2 * n - 1, M, dtype=a.dtype)
    bounds = [(0.0, None)] * n              # b ≥ 0

    # HiGHS (the default for recent SciPy) is fast for dense LPs
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')

    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Helper: normalise a vector to a prescribed sum (keeps positivity)
# ----------------------------------------------------------------------
def normalise_to_sum(vec: np.ndarray, target: float) -> np.ndarray:
    """Scale `vec` so that its L1‑norm equals `target`; keep tiny epsilon."""
    eps = 1e-12
    vec = np.maximum(vec, eps)
    return vec * (target / vec.sum())

    Returns the best coefficient list found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # --------------------------------------------------------------
    # Choose the length n and initialise the vector.
    # --------------------------------------------------------------
    try:

    except Exception:
        n = 2000
        x = np.arange(n, dtype=float)

        a = 1.0 - np.sin(np.pi * x / (n - 1))

    target_sum = math.sqrt(2.0 * n)
    a = normalise_to_sum(a, target_sum)

    a = best_a.copy()

    # temperature schedule (linear in wall‑clock time)
    T_start = 0.08
    T_end = 1e-4

    while time.time() < deadline:

        progress = (now - start_time) / (deadline - start_time)
        T = T_start * (1.0 - progress) + T_end * progress

        # --- pick a move -------------------------------------------------
        # focus on the current peak of the convolution
        k_peak = int(np.argmax(conv))

        # any i that participates in the peak (i, k_peak-i) is a candidate
        i_min = max(0, k_peak - (n - 1))
        i_max = min(n - 1, k_peak)
        i = np.random.randint(i_min, i_max + 1)

        # choose a random destination different from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to move (up to 30 % of the source mass)
        max_delta = a[i] * 0.30
        if max_delta < 1e-12:
            continue
        delta = max_delta * np.random.rand()

        # --- apply the move efficiently ------------------------------------
        a_new, conv_new = apply_move(a, conv, i, j, delta)
        new_M = float(conv_new.max())

        # --- Metropolis acceptance -----------------------------------------
        improve = new_M < cur_M
        if improve or np.random.rand() < math.exp(-(new_M - cur_M) / max(T, 1e-12)):
            a, conv, cur_M = a_new, conv_new, new_M
            iteration += 1

            if new_M < cur_best_val:
                cur_best_val = new_M
                cur_best_a = a.copy()
                print(f"[anneal] iter {iteration:07d} – new best max_conv = {new_M:.6f}")
