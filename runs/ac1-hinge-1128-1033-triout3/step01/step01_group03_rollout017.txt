"""
Improved optimizer for the step‑function evaluation problem.

The algorithm works in three stages:

1. **Analytic start** – initialise with the classic “square‑root” (1‑x)^‑½
   coefficients (they already give the bound π/2 ≈ 1.5708).

2. **Smooth‑max projected gradient descent** – instead of using only the
   active (max) convolution indices we optimise a smooth approximation of the
   max (log‑sum‑exp).  This provides a dense gradient that can move the
   solution out of the plateau where the simple sub‑gradient gets stuck.
   The update is projected onto the probability simplex (non‑negative,
   sum = 1).

3. **Random perturbations & restarts** – occasional jitter helps to escape
   shallow local minima, and a restart is triggered if no improvement has
   been seen for a while.

All operations are performed with NumPy and the FFT (O(n log n) convolution)
so that many thousands of iterations fit easily into the 1000 s budget.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al., 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic (1‑x)^{-1/2} series, normalised to sum = 1
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^{-1/2}, normalised to a probability simplex."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        # recurrence w_k = w_{k-1} * (2k‑1) / (2k)
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Fast convolution using an FFT of length power‑of‑two
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a (full convolution) using real FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(a*a) and its gradient.
    The smooth max is  (1/β)·log Σ_k exp(β·c_k) where c = a*a.
    Gradient = 2 * Σ_k w_k * a_{k‑i},  w_k = softmax(β·c_k).
    """
    # convolution
    conv = _conv_fft(a)                 # length 2n‑1
    # smooth max
    c_max = conv.max()
    exp_term = np.exp(beta * (conv - c_max))
    w = exp_term / exp_term.sum()       # softmax weights, sum to 1

    # gradient via cross‑correlation: grad_i = 2 Σ_k w_k * a_{k-i}
    a_rev = a[::-1]                     # reversed a
    m = len(w) + len(a_rev) - 1
    L = 1 << (m.bit_length())
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a_rev, n=L)
    corr = np.fft.irfft(fw * fa, n=L)   # length m
    # slice the part that corresponds to i = 0 … n‑1
    grad = 2.0 * corr[len(w) - 1 : len(w) - 1 + a.shape[0]]
    smooth_val = (c_max + np.log(exp_term.sum()) / beta)
    return smooth_val, grad


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------

                     budget_s: int = 1000,
                     **kwargs) -> list[float]:

        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds; the function must return before this.

    Returns

    list[float]

    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    a = _sqrt_series(n)                     # already lies on the simplex
    best_seq = a.copy()
    # fast local evaluation (same as evaluate_sequence after clipping)

    print(f"[init] n={n}, initial evaluation≈{best_score:.6f}")

    # optimisation hyper‑parameters
    iter_cnt = 0
    no_improve = 0
    step0 = 0.60                # initial step size (larger than before)
    decay = 5e-5                # slower decay for many iterations

    print(f"[final] best evaluation ≈ {best_score:.6f} after {iter_cnt} iterations")
    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
