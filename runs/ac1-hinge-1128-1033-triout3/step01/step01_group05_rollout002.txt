   b_k = C(2k,k)/4**k sequence (whose self‑convolution is exactly 1 on
   the first n entries).  The length n is chosen by scanning a modest
   range (≤2000) to minimise the analytic bound 2*n / (∑b_k)².
2. Repeatedly solve a linear programme:
       maximise   ∑ g_j
       subject to  (f ∗ g)[i] ≤ M  for all i               (M = max(f∗f))
                   g_j ≥ 0
   where (f ∗ g) is a linear function of g.  This finds a new
   non‑negative sequence g that can increase the total sum while keeping
   the cross‑convolution with the current f under the same bound M.
3. Rescale g to have the same ℓ₁‑norm as f and form a convex combination
       f_new = (1‑t)·f + t·g
   with a small mixing factor t (try a few values, favouring the one
   that reduces the true objective).  The mixing step never increases
   the maximal self‑convolution by much, and the objective
   2·n·max(f*f)/(∑f)² strictly decreases when the sum grows.
4. Keep the best sequence seen (according to `evaluate_sequence`) and
   continue until the time budget (≈1000 s) is exhausted.
5. Finish with a lightweight random‑perturbation hill‑climb to polish
   the result.

All numbers are kept in the allowed range [0,1000] before calling the
evaluator.  The routine returns the best list of heights found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper: generate the central‑binomial coefficients (the “constant‑conv”
# sequence) up to a given length.
# ----------------------------------------------------------------------
def generate_binomial_prefix(max_n: int):
    """Return an array of length max_n with b_k = C(2k,k)/4**k."""
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0                     # b_0 = 1
    coeffs[0] = cur
    for k in range(1, max_n):
        # recurrence: b_{k} = b_{k-1} * (2k-1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


# ----------------------------------------------------------------------
# Pre‑compute row / column indices for the convolution matrix.
# For a fixed length n, the matrix A (size (2n‑1) × n) satisfies
#   (A @ g)[i] = (f ∗ g)[i]   for any vector g.
# ----------------------------------------------------------------------
def conv_matrix_indices(n: int):
    """Return (row_idx, col_idx) for the Toeplitz‑like convolution matrix."""
    # column j contributes f[0..n‑1] at rows j .. j+n‑1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_blocks = [np.arange(j, j + n, dtype=np.int32) for j in range(n)]
    row_idx = np.concatenate(row_blocks)
    return row_idx, col_idx


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    benchmark 1.503.  The routine runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – analytical selection of a good starting length.
    # ------------------------------------------------------------------
    MAX_LEN = 2000               # a size that keeps the LP tractable
    coeffs = generate_binomial_prefix(MAX_LEN)

    cumulative = 0.0
    best_len = 1
    best_analytic = float('inf')
    for n in range(1, MAX_LEN + 1):
        cumulative += coeffs[n - 1]
        # max(conv) = 1 for the prefix, so the analytic score is 2*n / cum²
        score = 2.0 * n / (cumulative * cumulative)
        if score < best_analytic:
            best_analytic = score
            best_len = n

    # initialise f with the chosen prefix
    f = coeffs[:best_len].astype(np.float64).copy()
    # evaluate the true objective once (sanity)
    try:
        best_score = evaluate_sequence(f.tolist())
    except Exception:
        best_score = best_analytic
    best_seq = f.tolist()
    print(f"[init] length={best_len}, analytic={best_analytic:.6f}, "
          f"real={best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute the fixed sparsity pattern of the convolution matrix.
    # ------------------------------------------------------------------
    n = len(f)
    row_idx, col_idx = conv_matrix_indices(n)
    rows = 2 * n - 1

    # ------------------------------------------------------------------
    # Phase 2 – iterative LP improvement.
    # ------------------------------------------------------------------
    iteration = 0
    max_iter = 30               # generous upper bound; loop may stop earlier
    # a small set of candidate mixing factors (larger values try faster progress)
    mix_factors = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01]

    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        # current bound M = max(f ∗ f)
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))

        # Build the linear‑programming matrix A (sparse) for the constraint
        # (f ∗ g) ≤ M.  Since the sparsity pattern does not change we only
        # need to change the data vector (the f‑values repeated).
        data = np.tile(f, n)                     # length n*n
        A = csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))

        # Upper‑bound vector (same for every row)
        b_ub = np.full(rows, M, dtype=np.float64)

        # Objective: maximise sum(g)  → minimise –sum(g)
        c = -np.ones(n, dtype=np.float64)

        # Solve the LP using HiGHS (fast interior‑point method)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print(f"[lp] iteration {iteration}: LP failed, terminating.")
            break

        g = res.x
        # Normalise g to have the same ℓ₁‑norm as f (preserves the scale)
        sum_f = float(f.sum())
        sum_g = float(g.sum())
        if sum_g == 0.0:
            print("[lp] zero‑sum g, aborting.")
            break
        g = g * (sum_f / sum_g)

        # Try a few mixing factors; pick the first that yields an improvement.
        improved = False
        for t in mix_factors:
            f_candidate = (1.0 - t) * f + t * g
            # enforce the clamping required by the evaluator (0 … 1000)
            f_candidate = np.clip(f_candidate, 0.0, 1000.0)

            # quick check: avoid huge increase of the max convolution
            conv_new = np.convolve(f_candidate, f_candidate)
            M_new = float(np.max(conv_new))
            # allow a tiny increase due to numerical noise
            if M_new > M * 1.001:
                continue

            cand_score = evaluate_sequence(f_candidate.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = f_candidate.tolist()
                f = f_candidate                     # adopt as the new base
                improved = True
                print(f"[improve] it={iteration} t={t:.3f} score={cand_score:.6f}")
                break  # stop checking smaller t's

        if not improved:
            # No improvement with any mixing factor – still advance a tiny step
            # to keep the LP from stagnating.
            t = 0.01
            f = (1.0 - t) * f + t * g
            f = np.clip(f, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Phase 3 – final lightweight random hill‑climb (optional).
    # ------------------------------------------------------------------
    cur_seq = np.array(best_seq, dtype=np.float64)
    cur_score = best_score
    sigma_rel = 0.03
    no_improve = 0
    max_no_improve = 3000

    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(len(cur_seq))
        old_val = cur_seq[idx]
        delta = np.random.randn() * sigma_rel * max(old_val, 1e-12)
        new_val = max(0.0, old_val + delta)
        cand = cur_seq.copy()
        cand[idx] = new_val
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.tolist()
                print(f"[hill] new best {best_score:.6f}")
        else:
            no_improve += 1

    # final safety – ensure non‑negative values
    best_seq = [float(max(0.0, x)) for x in best_seq]
    print(f"[final] best score achieved {best_score:.6f}, length {len(best_seq)}")
    return best_seq
