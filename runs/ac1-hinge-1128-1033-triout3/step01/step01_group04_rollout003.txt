def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function (list of non‑negative heights).

Algorithm
---------
1. Initialise a probability distribution `p` from the discrete arcsine
   coefficients (the generating series of (1‑z)^‑1/2) and normalise it so
   that sum(p)=1.  This distribution yields a convolution value
   `max(conv)=π/(4·n)` and a score ≈π/2, i.e. 1.5708.

2. Gradient‑descent on the simplex:
   * For the current `p` compute the autocorrelation `c = p * p`
     (FFT‑based O(n log n)).
   * Let `k` be an index where `c` attains its maximum.
   * The gradient of the maximal entry w.r.t. `p` is
         g_i = 2·p_{k‑i}   (if 0≤k‑i<n, otherwise 0).
   * Move a small step opposite to the gradient, project the result back
     onto the probability simplex (Duchi et al. 2008) and keep the move
     only if the score improves (or with a tiny random acceptance to
     escape plateaus).  The step size slowly decays with the iteration
     counter.

3. Stochastic fine‑tuning:
   after the deterministic phase perform cheap random mass‑transfer
   moves (identical to the previous baseline) while time permits.

4. Return the best sequence found (the values themselves, not just the
   normalised probabilities – scaling does not affect the evaluation).

The whole routine respects the 1000 s budget, works for any length
(`length` – default 2000 gives a good trade‑off between speed and
solution quality) and never produces negative or overly large values
(the `evaluate_sequence` routine will clip them anyway).

"""

import time
import numpy as np

# Helper functions
# ----------------------------------------------------------------------


def _arcsine_coeffs(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^(-1/2) = Σ binom(2i,i)/4^i z^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i >= 0, sum_i x_i = 1 }.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_score(seq: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence using real FFT.
    Sequence `seq` is assumed to be a 1‑D numpy array of floats.
    """
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())   # next power of two >= 2n‑1
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)

    # --------------------------------------------------------------
    it = 0
    base_alpha = 0.12   # initial step magnitude (relative)
    decay = 0.9995      # per‑iteration decay factor

    while time.time() < deadline:
        # ----- convolution (FFT) -----
        fft_p = np.fft.rfft(p, n=L_fft)
        conv = np.fft.irfft(fft_p * fft_p, n=L_fft)[: 2 * n - 1]

        # ----- identify a maximal index -----
        k_max = int(np.argmax(conv))

        # ----- gradient of the maximal entry -----
        g = np.zeros_like(p)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        if i_start <= i_end:
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            idx_j = k_max - idx_i
            g[idx_i] = 2.0 * p[idx_j]

        # ----- direction and step size -----
        # Move opposite to the gradient.
        direction = -g
        # Normalise step so that the largest absolute change is bounded.
        max_abs = np.max(np.abs(direction))
        if max_abs == 0:
            # Gradient vanished – break early.
            break
        alpha = base_alpha / max_abs
        step = alpha * direction

        # ----- candidate via simplex projection -----
        cand = _project_simplex(p + step)
        cand_score = _compute_score(cand)

        # ----- accept if improvement -----
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            p = cand
            if it % 200 == 0:
                print(f"[grad] it={it}, score={best_score:.6f}")
        else:
            # Occasionally accept a non‑improving move to escape plateaus.
            if np.random.rand() < 0.001:
                p = cand

        # ----- step‑size decay -----
        base_alpha *= decay
        it += 1

        # Simple termination safeguard.
        if it > 200000:
            break

    # --------------------------------------------------------------
    # 3. Stochastic fine‑tuning (mass‑transfer moves) – same spirit as
    #    the baseline code but using the current best sequence as start.
    # --------------------------------------------------------------
    seq = best_seq.copy()        # will be mutated during refinement
    n = seq.shape[0]
    L_fft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        src = seq[i]
        if src <= 0.0:
            continue
        # delta = small exponential fraction of the source entry
        delta = np.random.exponential(scale=1e-3) * src
        if delta > src:
            continue

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        cur_score = _compute_score(new_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            seq = new_seq
            if int(best_score * 1e5) % 1000 == 0:
                print(f"[refine] score={best_score:.6f}")
        else:
            # tiny chance to accept a worsening move
            if np.random.rand() < 0.001:
                seq = new_seq

    print(f"[final] best score={best_score:.6f}")
    # Convert to plain Python list as required by the evaluator.
    return best_seq.tolist()
