"""
Search for a step‑function (non‑negative heights) that minimises

    M = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works on the probability simplex (∑a = 1) and
optimises the smooth approximation of the max‑operator
(soft‑max) with an Adam‑style projected gradient method.
Features:
  * initialise from the known best construction `height_sequence_1`
    when it exists, otherwise from a random Dirichlet start;
  * use a simplex projection after every update to keep feasibility;
  * increase the soft‑max temperature (`alpha`) gradually so the
    objective approaches the true infinity norm;
  * Adam‑style adaptive step‑sizes with a modest learning rate;
  * occasional random jitter and full restarts when progress stalls;
  * runs until the supplied time budget expires and returns the best
    sequence found.
"""

import time
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0 , Σ x_i = 1}.
    """
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        # Already feasible – avoid work.
        return v
    n = v.shape[0]
    # Sort v in descending order.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest rho where u[rho] > cssv[rho] / rho.
    rho_arr = np.where(u - cssv / ind > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a sequence of non‑negative heights that (approximately) minimises
    the evaluation function `evaluate_sequence`.  The method runs an
    Adam optimiser on a soft‑max approximation of the max‑convolution,
    stays on the probability simplex, and performs occasional jitter
    and restarts to escape local minima.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)                     # type: ignore[name‑defined]
        has_known = True
    except Exception:
        n = 1153                                        # default used in the benchmark
        has_known = False

    # --------------------------------------------------------------
    # 2. Initial probability vector (sum == 1)
    # --------------------------------------------------------------
    if has_known and rng.random() < 0.4:
        seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name‑defined]
        seq = np.clip(seq, 0.0, 1000.0)                    # mimic the security checks
        total = seq.sum()
        if total > 0.0:
            p = seq / total
        else:
            p = np.full(n, 1.0 / n)
    else:
        p = rng.random(n)
        p /= p.sum()

    best_p = p.copy()
    best_score = 2.0 * n * np.max(np.convolve(p, p))

    # --------------------------------------------------------------
    # 3. Adam state and hyper‑parameters
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    beta1_pow = 1.0
    beta2_pow = 1.0

    lr = 0.20                     # base learning rate for Adam
    alpha = 5.0                   # smoothing (soft‑max) factor
    alpha_mul = 1.01              # geometric increase of alpha
    jitter_interval = 500         # how often to add a tiny random perturbation
    restart_patience = 5000       # iterations without improvement before a restart
    last_improvement = 0

    # Pre‑compute the index vector needed for the gradient extraction.
    idx = 2 * n - 2 - np.arange(n)

    deadline = time.time() + budget_s - 5.0   # safety margin
    it = 0

    while time.time() < deadline:
        it += 1

        # ----------  evaluate current objective ----------
        conv = np.convolve(p, p)                 # linear convolution, length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            last_improvement = it
            print(f"Iter {it:6d} – new best: {best_score:.6f}")

        # ----------  smooth max via soft‑max ----------
        # exp(alpha * (conv - max_conv)) is numerically stable because the
        # largest entry is zero.
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()            # soft‑max weights, sum to 1

        # ----------  gradient of the smoothed objective ----------
        w_rev = w[::-1]
        corr = np.convolve(p, w_rev)             # length 3n‑2
        grad = 2.0 * corr[idx]

        # ----------  Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step
        p = _simplex_projection(p)

        # ----------  occasional jitter ----------
        if it % jitter_interval == 0:
            noise = rng.normal(scale=0.001, size=n)
            p = _simplex_projection(p + noise)

        # ----------  update schedule ----------
        alpha *= alpha_mul
        # modest decay of the base learning rate to fine‑tune later
        if it % 2000 == 0:
            lr *= 0.95

        # ----------  restart if stuck ----------
        if it - last_improvement > restart_patience:
            # Re‑initialise either from the best we have (with jitter)
            # or from a fresh random point.
            if rng.random() < 0.6:
                p = best_p.copy()
                p = _simplex_projection(p + rng.normal(scale=0.01, size=n))
            else:
                p = rng.random(n)
                p /= p.sum()
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            alpha = 5.0
            lr = 0.20
            last_improvement = it
            print(f"Iter {it:6d} – restart")

    # --------------------------------------------------------------
    # 5. Return the best sequence (list of floats)
    # --------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    final_val = 2.0 * n * np.max(np.convolve(result, result))
    print(f"Finished. Best score observed: {best_score:.6f}")
    return result
