"""
candidate generator – two non‑overlapping arcsine blocks

The classic arcsine sequence a_i = C(2i,i)/4^i satisfies
    (a * a)_k = 1   for 0 ≤ k < len(a)

Therefore a single block of length L gives evaluation
    2·L·1 / (∑a_i)^2 = π/2   (≈1.5708).

If we place **two** such blocks with a zero gap of length L
the self‑convolution of each block occupies the intervals
    [0, 2L‑2]      and   [4L, 6L‑2]
while the cross‑convolution occupies [2L, 4L‑2].
All three intervals are disjoint, so the maximal convolution
value stays 1, but the total sum of the coefficients is doubled.
The resulting evaluation is

    2·(3L)·1 / (2·∑a_i)^2   =   (3π)/8   ≈ 1.1781 ,

which is well below the target 1.503.

The function below builds this construction (the block length L can be
changed – the score is independent of L).  No stochastic search is
required; the sequence is deterministic and satisfies all constraints.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: first L coefficients of (1‑z)^(-1/2)
def _arcsine_coeffs(L: int):
    """Return the first L coefficients of (1 - z)^(-1/2)."""
    coeffs = np.empty(L, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, L):
        # recurrence a_{i} = a_{i-1} * (2i‑1) / (2i)
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs.tolist()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     block_len: int = 5000) -> list[float]:
    """
    Produce a step‑function that beats the 1.503 upper bound.

    Parameters
    ----------
    seed : int, optional
        Unused – kept for API compatibility.
    budget_s : float, optional
        Unused – the construction is deterministic and fast.
    block_len : int, optional
        Length of each arcsine block.  The total length will be
        3·block_len (block – gap – block).  Any positive value works;
        3000–5000 gives a good trade‑off between speed and score.

    Returns
    -------
    list[float]
        The concatenated sequence: [arc] + [zeros] + [arc].
    """
    # 1. build one arcsine block
    arc = _arcsine_coeffs(block_len)

    # 2. concatenate block – zero gap – block
    seq = arc + [0.0] * block_len + arc

    # Optional: a tiny random tweak can only increase the score,
    # so we keep the deterministic optimum.
    # (The function exists solely to respect the expected signature.)

    # Report the expected score (using the same fast estimator as the
    # previous solution) – useful for debugging, not required.
    try:
        # fast score computation (same as the reference implementation)
        arr = np.asarray(seq, dtype=np.float64)
        n = len(arr)
        Lfft = 1 << ((2 * n - 1).bit_length())
        fft_arr = np.fft.rfft(arr, n=Lfft)
        conv = np.fft.irfft(fft_arr * fft_arr, n=Lfft)[: 2 * n - 1]
        max_b = float(np.max(conv))
        sum_a = float(np.sum(arr))
        score = 2.0 * n * max_b / (sum_a * sum_a)
        print(f"[candidate] block_len={block_len}, total_len={n}, "
              f"max_b={max_b:.6f}, sum={sum_a:.6f}, score≈{score:.6f}")
    except Exception:      # safety – the evaluate_sequence will be used finally
        pass

    return seq
