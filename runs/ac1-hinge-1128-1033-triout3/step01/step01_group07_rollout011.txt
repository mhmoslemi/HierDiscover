"""
propose_candidate
=================

A deterministic construction that dramatically improves the bound.
We build a step function consisting of

* a *core* with the optimal 1/√i decay,
* a long *tail* of constant height.

The height of the tail is chosen as large as possible while keeping every
convolution term ≤π.  This yields three competing contributions to the
maximum convolution:

    • core * core   → ≈π,
    • core * tail   → δ·∑core,
    • tail * tail   → δ²·len(tail).

Taking  
    δ = min( sqrt(π/len(tail)),  π/∑core )
guarantees that each term is ≤π, i.e. the global maximum is π.
For a given core length the total ℓ¹‑mass
    S = ∑core + δ·len(tail)
is maximal, and the evaluation function becomes

    L = 2·n·π / S² .

The optimum occurs when the two constraints are tight,
∑core = √(π·len(tail)), which gives
    core_length ≈ π·n / (π+4)  (≈44 % of n).

The algorithm performs a cheap grid search over the core length,
computes the admissible tail height analytically, constructs the full
sequence and evaluates it with the exact `evaluate_sequence` routine.
A short local refinement (random swaps) is then applied.
The whole procedure respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast maximum of a linear convolution using an FFT of power‑of‑two.
# ----------------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Exact evaluation of the objective for a candidate sequence.
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    total = float(seq.sum())
    if total < 0.01:                 # protect against a zero sum
        return np.inf
    max_conv = _conv_max(seq)
    n = seq.shape[0]
    return 2.0 * n * max_conv / (total * total)


# ----------------------------------------------------------------------
# Main entry point used by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Build a core + tail sequence.
    The core follows a_i = 1/√(i+1) for i=0..c‑1.
    The tail height δ is the largest value that keeps
        δ·∑core ≤ π   and   δ²·len(tail) ≤ π .
    We search over the core length, pick the best candidate,
    then polish it with a few random swaps.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # keep a margin

    # ------------------------------------------------------------------
    # 0) Determine the problem size.
    # ------------------------------------------------------------------
    n = len(globals().get("height_sequence_1", [])) or 8000

    # Pre‑compute the ideal 1/√i core for the whole range.
    core_full = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    core_prefix_sum = np.cumsum(core_full)          # sum up to index i

    best_seq = None
    best_val = np.inf
    best_core_len = None
    best_delta = None

    # ------------------------------------------------------------------
    # 1) Coarse grid over core length.
    #    Step ~0.5 % of n (≈40 for n≈8000) – ~200 evaluations total.
    # ------------------------------------------------------------------
    step = max(1, n // 200)
    for core_len in range(2, n - 1, step):
        if time.time() > deadline:
            break

        sum_core = float(core_prefix_sum[core_len - 1])
        tail_len = n - core_len
        if tail_len <= 0:
            continue

        # admissible tail height from the two constraints
        delta1 = math.sqrt(math.pi / tail_len)          # tail‑tail ≤ π
        delta2 = math.pi / sum_core if sum_core > 0 else np.inf  # core‑tail ≤ π
        delta = min(delta1, delta2)

        # build the candidate
        seq = np.empty(n, dtype=np.float64)
        seq[:core_len] = core_full[:core_len]
        seq[core_len:] = delta

        val = _score(seq)

        if val < best_val:
            best_val, best_seq = val, seq.copy()
            best_core_len, best_delta = core_len, delta
            print(f"[grid] core={core_len}/{n} ({core_len/n:.3f}), "
                  f"δ={delta:.4g}, score={val:.6f}")

    # ------------------------------------------------------------------
    # 2) Fine search around the best core length (±step*5, step=1).
    # ------------------------------------------------------------------
    if best_core_len is not None:
        fine_start = max(2, best_core_len - step * 5)
        fine_end   = min(n - 1, best_core_len + step * 5)
        for core_len in range(fine_start, fine_end + 1):
            if time.time() > deadline:
                break

            sum_core = float(core_prefix_sum[core_len - 1])
            tail_len = n - core_len
            if tail_len <= 0:
                continue

            delta1 = math.sqrt(math.pi / tail_len)
            delta2 = math.pi / sum_core if sum_core > 0 else np.inf
            delta = min(delta1, delta2)

            seq = np.empty(n, dtype=np.float64)
            seq[:core_len] = core_full[:core_len]
            seq[core_len:] = delta

            val = _score(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                best_core_len, best_delta = core_len, delta

    # ------------------------------------------------------------------
    if best_seq is not None:
        cur_seq, cur_val = best_seq.copy(), best_val
        start = time.time()
        T0 = max(0.01, 0.1 * best_val)        # initial temperature

        while time.time() < deadline:

            i, j = np.random.randint(n, size=2)
            if i == j:

            # transfer a tiny fraction (≤1 %) from i to j

            amount = cur_seq[i] * np.random.rand() * 0.01
            new_seq = cur_seq.copy()
            new_seq[i] -= amount

            new_val = _score(new_seq)

            # Metropolis acceptance
            if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T0):
                cur_seq, cur_val = new_seq, new_val
                if new_val < best_val:
                    best_val, best_seq = new_val, new_seq.copy()
                    print(f"[SA] improved to {best_val:.6f}")

            # linear cooling
            elapsed = time.time() - start
            T0 = max(1e-6, T0 * (1.0 - elapsed / budget_s))

    # ------------------------------------------------------------------
    # 4) Fallback – uniform sequence (should never be needed).
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _score(best_seq)

    print(f"final best value: {best_val:.6f}")
    return best_seq.tolist()
