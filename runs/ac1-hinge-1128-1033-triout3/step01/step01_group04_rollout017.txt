def propose_candidate(seed=0, budget_s=900, length=5000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    seq = arcsine_coeffs(length)
    best_seq = seq.copy()
    best_score = _compute_score(seq)
    cur_score = best_score
    L = 1 << ((2*length - 1).bit_length())
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute conv via FFT
        ft = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(ft*ft, n=L)[: 2*length - 1]
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # compute contributions for max_idx
        if max_idx < length:
            left = seq[:max_idx+1]
            right = seq[max_idx::-1]
            start = 0
        else:
            start = max_idx - (length - 1)
            left = seq[start:]
            right = seq[max_idx - start::-1]
        prod = left * right
        rel_max = prod.argmax()
        source = start + rel_max
        # optionally also consider partner index = max_idx - source
        partner = max_idx - source
        if seq[partner] > seq[source]:
            source = partner
        # choose destination
        dest = np.random.randint(length)
        while dest == source:
            dest = np.random.randint(length)
        # transfer amount
        frac = max(0.001, 0.5 * (1 - (time.time() - start) / (budget_s - 0.5))) # decreasing fraction
        delta = seq[source] * np.random.rand() * frac
        if delta <= 0:
            continue
        new_seq = seq.copy()
        new_seq[source] -= delta
        new_seq[dest] += delta
        # ensure nonnegatives (source may become very small)
        if new_seq[source] < 0:
            continue
        # compute new score
        new_score = _compute_score(new_seq)
        # acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(1e-6, 0.1 * (1 - (time.time() - start) / (budget_s - 0.5)))):
            seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[iter {iteration}] improved: score={best_score:.6f}")
        # occasional random shake
        if iteration % 10000 == 0:
            # random perturbation
            i, j = np.random.randint(length, size=2)
            if i == j: continue
            delta = min(seq[i], seq[i] * 0.5)
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            new_score = _compute_score(new_seq)
            if new_score < cur_score:
                seq = new_seq
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()

"""
Propose a candidate step‑function using a targeted mass‑transfer local search.

The algorithm starts from the (1‑z)^(-1/2) (arcsine) coefficients,
which already give the classical bound π/2 ≈ 1.5708.
At each iteration we:
  1. Compute the full convolution by FFT and locate the index k where it
     attains its maximum.
  2. Identify the pair (i, k‑i) that contributes most to this peak.
  3. Transfer a fraction of the mass from the larger of the two indices
     to a random other index (the total sum stays constant).
  4. Accept the move if it improves the objective or with a small
     probability that decays with time (simulated‑annealing style).

The transfer size shrinks as the allotted time runs out, allowing a
coarse early exploration and fine‑grained final polishing.
The routine keeps the best sequence seen and returns it when the time
budget (default ~ 900 s) is exhausted.
"""

import numpy as np
import time


def _compute_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective using an FFT based convolution.
    Equivalent to `evaluate_sequence` but without the input‑validation
    overhead.
    """
    n = seq.size
    # length for FFT: next power of two ≥ 2*n‑1
    L = 1 << ((2 * n - 1).bit_length())
    ft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        # guard against pathological cases – should never happen here
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _arcsine_coeffs(n: int) -> np.ndarray:
    """
    First `n` coefficients of (1‑z)^(-1/2) = Σ binom(2i,i)/4^i z^i.
    The recurrence a_0 = 1; a_{i+1} = a_i * (2*i+1)/(2*(i+1)).
    """
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def propose_candidate(seed: int = 0, budget_s: float = 900.0, length: int = 5000):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises `evaluate_sequence`.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence discovered.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Total time budget in seconds.  A small safety margin is kept.
    length : int, optional
        Length of the initial arcsine sequence.  Around 5000 gives a good
        trade‑off between quality and speed.

    Returns
    -------
    list[float]
        The best sequence found before the time limit expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ---- 1. initialise from the arcsine series ---------------------------------
    seq = _arcsine_coeffs(length)
    best_seq = seq.copy()
    best_score = _compute_score(best_seq)
    cur_score = best_score

    # pre‑compute the FFT length (power of two) once
    L = 1 << ((2 * length - 1).bit_length())

    iteration = 0
    print(f"[init] length={length}, score={best_score:.6f}")

    # ---- 2. iterative mass‑transfer search --------------------------------------
    while time.time() < deadline:
        iteration += 1

        # a) full convolution and location of the current maximum
        ft = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(ft * ft, n=L)[: 2 * length - 1]
        max_val = conv.max()
        max_k = int(conv.argmax())

        # b) which pair (i, max_k‑i) contributes the most to this peak?
        if max_k < length:
            # i ranges from 0 … max_k
            left = seq[: max_k + 1]
            right = seq[max_k::-1]        # reversed slice
            start = 0
        else:
            # i ranges from max_k‑(length‑1) … length‑1
            start = max_k - (length - 1)
            left = seq[start:]
            right = seq[max_k - start::-1]
        prod = left * right                     # element‑wise products
        rel_idx = int(prod.argmax())
        src = start + rel_idx                   # absolute index of the larger contributor
        # also consider the partner index; take the larger of the two as source
        partner = max_k - src
        if 0 <= partner < length and seq[partner] > seq[src]:
            src = partner

        # c) choose a destination index (any index ≠ src)
        dst = np.random.randint(length)
        while dst == src:
            dst = np.random.randint(length)

        # d) transfer a fraction of the mass from src to dst
        # transfer size shrinks as time runs out (coarse → fine)
        elapsed = time.time() - start_time
        frac = max(0.001, 0.5 * (1.0 - elapsed / (budget_s - 0.5)))
        delta = seq[src] * np.random.rand() * frac
        if delta <= 0.0:
            continue
        new_seq = seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # e) evaluate the new candidate
        new_score = _compute_score(new_seq)

        # f) acceptance rule (greedy + simulated‑annealing)
        # temperature also decays with time
        temperature = max(1e-6, 0.05 * (1.0 - elapsed / (budget_s - 0.5)))
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / temperature)
            if np.random.rand() < prob:
                accept = True

        if accept:
            seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[iter {iteration}] improved score={best_score:.6f}")

        # occasional random shake‑up to escape possible plateaus
        if iteration % 10000 == 0:
            i, j = np.random.randint(length, size=2)
            if i != j and seq[i] > 0.0:
                shake_delta = seq[i] * 0.5
                tmp_seq = seq.copy()
                tmp_seq[i] -= shake_delta
                tmp_seq[j] += shake_delta
                shake_score = _compute_score(tmp_seq)
                if shake_score < cur_score:
                    seq = tmp_seq
                    cur_score = shake_score
                    if shake_score < best_score:
                        best_score = shake_score
                        best_seq = tmp_seq.copy()
                        print(f"[shake {iteration}] new best score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
