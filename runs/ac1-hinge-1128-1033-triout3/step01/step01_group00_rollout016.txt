    # softmax
    max_x = np.max(x)  # for numerical stability
    exp_x = np.exp(x - max_x)
    a = exp_x / np.sum(exp_x)

    # compute y = alpha * conv

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

def simplex_projection(v):
    """Project onto the probability simplex {x >=0, sum x = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def u_shaped_distribution(n, p, c=1.0):
    """Symmetric U‑shaped vector: a[i] ∝ (c + min(i, n−1−i))^{-p}."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (c + d) ** (-p) + 1e-12
    a /= a.sum()
    return a

def softmax(x):
    """Stable softmax returning a probability vector."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

def smooth_max_and_weights(a, alpha):
    """
    Return smooth approximation to max(conv(a,a)) and the
    corresponding weight vector w = softmax(alpha*conv).
    """
    conv = np.convolve(a, a)
    y = alpha * conv
    # stable log‑sum‑exp
    max_y = np.max(y)
    lse = max_y + np.log(np.sum(np.exp(y - max_y)))
    smooth = lse / alpha
    w = np.exp(y - max_y)
    w /= w.sum()
    return smooth, conv, w

def grad_smooth_wrt_a(a, w):
    """
    Gradient of the smooth max w.r.t. a:
    grad_i = 2 * Σ_k w_k * a_{k-i}.
    """
    # correlation between a and w; use convolution with reversed w
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)
    len_conv = len(w)                # = len(conv)
    grad = 2.0 * conv_rev[len_conv - 1 - np.arange(len(a))]
    return grad

def objective_and_grad(x, alpha):
    """
    Composite objective on the unconstrained variable x:
    f(x) = smooth_max(conv(softmax(x), softmax(x))) .
    Returns (value, gradient) where the gradient is ∂f/∂x.
    """
    a = softmax(x)                   # a is on the simplex
    smooth, conv, w = smooth_max_and_weights(a, alpha)
    grad_a = grad_smooth_wrt_a(a, w) # ∂f/∂a
    # chain rule for softmax: ∂a_i/∂x_j = a_i (δ_ij - a_j)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    return smooth, grad_x

def refine_sequence(a_init, alphas=None, maxiter_per=80):
    """
    Perform a continuation L‑BFGS optimization on a simplex‑restricted vector.
    a_init must already sum to 1.
    alphas : list of smoothing parameters, increasing.
    Returns the refined probability vector.
    """
    if alphas is None:
        alphas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
    # initialise x = log a (plus tiny epsilon for safety)
    eps = 1e-12
    x = np.log(a_init + eps)
    for alpha in alphas:
        res = minimize(lambda xx: objective_and_grad(xx, alpha),
                       x, method='L-BFGS-B', jac=True,
                       options={'maxiter': maxiter_per, 'disp': False})
        x = res.x
    a_opt = softmax(x)
    return a_opt

    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The procedure:
        1. construct many U‑shaped seeds (different exponents and offsets);
        2. refine each seed with a continuation L‑BFGS on a smooth max‑objective;
        3. keep the best vector seen;
        4. finish with a short simulated‑annealing style random‑mass transfer
           to polish the result.

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    n = 2000                                 # modest size for fast iterations
    alphas = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0]

        c = 10 ** np.random.uniform(-1.0, 0.5)     # roughly 0.1 .. 3
        seeds.append(u_shaped_distribution(n, p, c))

    best_score = np.inf

    for idx, a0 in enumerate(seeds):

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w ≥ 0, Σw = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """
    Symmetric U‑shaped vector:
        a[i] ∝ (c + min(i, n‑1‑i))^{-p}
    Normalised to Σa = 1.

    return a
