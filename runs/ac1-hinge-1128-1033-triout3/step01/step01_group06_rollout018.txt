"""
Improved simulated‑annealing for the step‑function minimisation problem.

Main ideas
----------
* Use the same fast O(n) convolution update as the reference implementation.
* The move that reduces the current maximal convolution entry is made deterministic:
    – src  = index i that contributes the most to the peak (max a[i]*a[k‑i]).
    – dst  = index whose counterpart a[k‑dst] is the smallest (ideally zero).
  This focuses the mass transfer on the entries that actually matter.
* A slightly hotter annealing schedule and a larger proportion of
  “targeted” moves (90 %) give faster convergence.
* After the SA loop a short deterministic hill‑climbing phase is performed
  to polish the best candidate found.
* The routine works for any length n; if a previous construction
  (``height_sequence_1``) is available its length is reused, otherwise a
  default n≈800 is used – large enough to give a good trade‑off between
  solution quality and speed.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Fast O(n) update of convolution when moving mass δ from i to j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].
    a   – 1‑D array, sum(a) = 1
    conv – a * a (linear convolution, length 2n‑1)
    i,j – distinct indices
    delta – 0 < delta ≤ a[i]
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta
    dj = +delta

    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    conv_new = conv.copy()

    # all other positions
    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]
    a_q = a[q]

    # (i,q) and (q,i)
    conv_new[i + q] += 2.0 * di * a_q
    # (j,q) and (q,j)
    conv_new[j + q] += 2.0 * dj * a_q

    # self‑pair contributions
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Deterministic move that attacks the current maximal convolution entry
# ----------------------------------------------------------------------
def best_targeted_move(a: np.ndarray, conv: np.ndarray):
    """
    Choose (src, dst) that most reduces the current peak.
    * src  – index i with largest product a[i] * a[k‑i] where k is the
             arg‑max of conv.
    * dst  – index whose counterpart a[k‑dst] is minimal (preferably zero).
    """
    n = a.shape[0]
    k = int(np.argmax(conv))

    # feasible i such that both i and k-i lie inside [0,n)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_candidates = np.arange(i_min, i_max + 1, dtype=int)
    j_candidates = k - i_candidates

    prod = a[i_candidates] * a[j_candidates]
    src = int(i_candidates[np.argmax(prod)])

    # find destination with smallest counterpart a[k-dst]
    # compute counterpart indices (may be out of range → 0)
    d = np.arange(n, dtype=int)
    d = d[d != src]                     # exclude src
    counterpart = k - d
    # mask where counterpart is inside the vector
    inside = (counterpart >= 0) & (counterpart < n)
    counterpart_val = np.empty_like(d, dtype=float)
    counterpart_val[inside] = a[counterpart[inside]]
    counterpart_val[~inside] = 0.0     # out‑of‑range → contributes nothing

    dst = int(d[np.argmin(counterpart_val)])

    return src, dst


# ----------------------------------------------------------------------
# Small deterministic polishing pass after the SA loop
# ----------------------------------------------------------------------
def polish(a: np.ndarray, conv: np.ndarray, iters: int = 20000):
    """
    Perform a simple steepest‑descent style polishing:
    repeatedly move a small fraction of mass from the worst contributing
    index to the best possible destination (always accepted if it improves).
    """
    n = a.shape[0]
    best_a = a
    best_conv = conv
    best_val = 2.0 * n * np.max(conv)

    for _ in range(iters):
        k = int(np.argmax(best_conv))
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_cand = np.arange(i_min, i_max + 1, dtype=int)
        j_cand = k - i_cand
        prod = best_a[i_cand] * best_a[j_cand]
        src = int(i_cand[np.argmax(prod)])

        # choose dst that minimises the counterpart (same as in best_targeted_move)
        d = np.arange(n, dtype=int)
        d = d[d != src]
        counterpart = k - d
        inside = (counterpart >= 0) & (counterpart < n)
        ct_val = np.empty_like(d, dtype=float)
        ct_val[inside] = best_a[counterpart[inside]]
        ct_val[~inside] = 0.0
        dst = int(d[np.argmin(ct_val)])

        if best_a[src] < 1e-12:
            continue
        delta = best_a[src] * 0.1   # modest step
        a_new, conv_new = apply_move(best_a, best_conv, src, dst, delta)
        new_val = 2.0 * n * np.max(conv_new)
        if new_val < best_val:
            best_a, best_conv, best_val = a_new, conv_new, new_val
    return best_a, best_conv, best_val

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Simulated‑annealing search with an aggressive targeted move.
    Returns the best non‑negative step sequence found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose length n (reuse previous length if possible)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 800    # a modest size that gives a good trade‑off

    # --------------------------------------------------------------
    # Initialise a “edge‑heavy’’ distribution and normalise to sum = 1
    # --------------------------------------------------------------
    x = np.arange(n, dtype=float)
    a = 1.0 - np.sin(np.pi * x / (n - 1))   # more mass near the ends
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:                       # fallback to uniform
        a = np.full(n, 1.0 / n)
    else:
        a /= total                          # now sum(a) = 1

    conv = np.convolve(a, a)                # initial convolution
    cur_val = 2.0 * n * np.max(conv)        # current objective (sum == 1)
    best_a = a.copy()
    best_val = cur_val

    print(f"[init] n={n}, initial objective={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    T_start = 0.20
    T_end   = 5e-5
    total_budget = deadline - start_time
    iteration = 0
    accept_cnt = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear temperature schedule
        progress = (now - start_time) / total_budget
        T = T_start * (1.0 - progress) + T_end * progress

        # --------------------------------------------------------------
        # Choose move – 90 % targeted, 10 % random for occasional escape
        # --------------------------------------------------------------
        if np.random.rand() < 0.90:
            src, dst = best_targeted_move(a, conv)
        else:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

        # ----------------------------------------------------------------
        # Amount of mass to move (never more than 50 % of the source)
        # ----------------------------------------------------------------
        max_delta = a[src] * 0.5
        if max_delta < 1e-12:
            continue
        delta = np.random.rand() * max_delta

        # --------------------------------------------------------------
        # Evaluate the proposed move
        # --------------------------------------------------------------
        a_new, conv_new = apply_move(a, conv, src, dst, delta)
        new_val = 2.0 * n * np.max(conv_new)

        # ----------------------------------------------------------------
        # Acceptance test (simulated annealing)
        # ----------------------------------------------------------------
        improve = new_val < cur_val
        if improve or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a, conv, cur_val = a_new, conv_new, new_val
            accept_cnt += 1

            if new_val < best_val:
                best_a, best_val = a_new.copy(), new_val
                print(f"[{iteration:7d}] new best = {best_val:.6f}")

        iteration += 1

        # re‑compute convolution occasionally to avoid accumulated rounding error
        if iteration % 4000 == 0:
            conv = np.convolve(a, a)

    # ------------------------------------------------------------------
    # Deterministic polishing phase (fast, improves the final few digits)
    # ------------------------------------------------------------------
    best_a, conv, best_val = polish(best_a, np.convolve(best_a, best_a), iters=20000)
    print(f"[polish] polished objective = {best_val:.6f}")

        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)
