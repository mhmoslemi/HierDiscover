def choose_destination(a, k, src):
    n = a.shape[0]
    indices = np.arange(n)
    # exclude src
    mask = indices != src
    # compute counterpart index = k - idx
    counterpart = k - indices
    # mask valid counterpart within [0,n-1]
    valid = (counterpart >= 0) & (counterpart < n)
    # cost = a[counterpart] where valid else 0
    cost = np.zeros_like(a)
    cost[valid] = a[counterpart[valid]]
    # also treat invalid cost as 0 (best)
    cost[~valid] = 0.0
    # we want minimal cost
    candidate = indices[mask][np.argmin(cost[mask])]
    return int(candidate)
```

def project_simplex(v):
    # v: np.ndarray
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function by a three‑phase optimisation:

1. **Sub‑gradient descent** (≈40 % of the time):
   Reduce the current maximal self‑convolution entry using the sub‑gradient
   ∂max(conv)/∂a_i = 2·a_{k−i} (k = argmax conv).  After a gradient step the
   vector is projected back onto the probability simplex.

2. **Deterministic greedy moves** (≈40 %):
   Repeatedly move a fraction of the mass from the index that contributes
   most to the current peak towards a destination that does *not* increase this
   peak (i.e. where the counterpart a_{k−dest}=0 or minimal).  Accept only
   improving moves.

3. **Simulated‑annealing fine‑tuning** (last ≈20 %):
   Classic SA with targeted moves (as in the reference code) to escape any
   remaining local minima.

All updates of the convolution after a single move are performed in O(n) by
`apply_move`.  The whole routine respects the 1000 s budget and returns the
best sequence found (scaled to sum = 1, which is scale‑invariant for the
objective).  The code works with the provided `evaluate_sequence` function.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Fast convolution update when moving a mass δ from index i to j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].
    a – 1‑D array of length n, sum(a) = 1
    conv – convolution a*a (length 2n‑1)
    i, j – distinct indices, 0 ≤ i,j < n
    delta – 0 < delta ≤ a[i]
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta            # change at i
    dj = +delta            # change at j

    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    conv_new = conv.copy()

    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]                # all other positions
    a_q = a[q]

    # contributions of the changed entries to all sums
    conv_new[i + q] += 2.0 * di * a_q    # (i, q) and (q, i)
    conv_new[j + q] += 2.0 * dj * a_q    # (j, q) and (q, j)

    # self‑pair contributions (i,i) and (j,j)
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Helper: simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w
