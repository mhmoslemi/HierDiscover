"""
Hybrid Linear‑Programming + Simulated‑Annealing optimiser.

1. Initialise a non‑negative step vector `a` (sum = 1).  If a previous
   construction `height_sequence_1` is available it is used, otherwise a
   uniform vector of length ≈1500 is created.

2. Repeatedly solve a linear program:
       maximise   Σ g_j
       subject to (a * g)[k] ≤ M      for all convolution positions k
                  g_j ≥ 0
   where M = max(a*a).  The LP finds a direction `g` that can increase the
   ℓ₁‑mass while keeping the self‑convolution bounded by the current peak.
   After scaling `g` to have sum 1 we take a convex combination
       a ← (1‑t)·a + t·g   (t≈0.2)
   which strictly reduces the peak `M`.  A few LP iterations already
   lower the evaluation considerably.

3. With the LP‑improved vector we run a fast simulated‑annealing phase.
   Moves consist of transferring a small amount of mass from a source
   index to a destination index while updating the convolution in O(n).
   The temperature is decreased linearly; accepted moves are those that
   improve the objective or are accepted with the Metropolis criterion.

4. The best vector found is finally evaluated with the provided
   `evaluate_sequence` and returned.

The implementation uses only top‑level functions, respects the 1000 s
budget, and works with `numpy`, `scipy.optimize.linprog` and
`scipy.sparse.csr_matrix`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Fast convolution update when moving a mass δ from index i to j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].

    a – 1‑D array of length n, sum(a) = 1
    conv – convolution a*a (length 2n‑1)
    i, j – distinct indices, 0 ≤ i,j < n
    delta – 0 < delta ≤ a[i]
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta
    dj = +delta

    # new vector (exact, sum stays 1)
    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    # fast convolution update
    conv_new = conv.copy()

    # all other positions q ≠ i, j
    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]
    a_q = a[q]

    # contributions of the changed entries to all sums
    conv_new[i + q] += 2.0 * di * a_q          # (i,q) and (q,i)
    conv_new[j + q] += 2.0 * dj * a_q          # (j,q) and (q,j)

    # self‑pair contributions (i,i) and (j,j)
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Targeted move focusing on the current maximum of the convolution
# ----------------------------------------------------------------------
def targeted_move(a: np.ndarray, conv: np.ndarray):
    """
    Choose a source index that most contributes to the current peak
    of the convolution and a random destination.
    Returns (src, dst).
    """
    n = a.shape[0]
    k = int(np.argmax(conv))                     # position of the current peak

    # feasible i so that both i and k-i lie inside the vector
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_candidates = np.arange(i_min, i_max + 1, dtype=int)
    j_candidates = k - i_candidates

    # contribution of each unordered pair to the peak
    contrib = a[i_candidates] * a[j_candidates]
    src = int(i_candidates[np.argmax(contrib)])

    # pick a random destination different from src
    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1
    return src, dst

        a = np.array(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        n = a.shape[0]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            raise ValueError
        a /= s
    except Exception:
        # fallback: moderate length uniform start
        n = 1500
        a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # pre‑compute the index arrays for the convolution‑matrix used in LP
    # ------------------------------------------------------------------
    col_idx = np.repeat(np.arange(n), n)
    row_idx = np.concatenate([np.arange(i, i + n) for i in range(n)])

    # ------------------------------------------------------------------
    # bookkeeping of the best vector seen so far
    # ------------------------------------------------------------------
    conv = np.convolve(a, a)
    best_val = float(2.0 * n * conv.max())
    best_a = a.copy()
    print(f"[init] n={n}, evaluation={best_val:.6f}")

    # ------------------------------------------------------------------
    # 1) Linear‑programming phase
    # ------------------------------------------------------------------
    max_lp_iters = 25
    lp_iter = 0
    no_improve_lp = 0
    T_mix = 0.20                                   # mixing weight

    while time.time() < deadline and lp_iter < max_lp_iters:
        conv = np.convolve(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        # build sparse constraint matrix A_ub (size (2n‑1) × n)
        data = np.repeat(a, n)                     # a_i repeats n times, one per column
        A_ub = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # linear programme: maximise Σ g_j   ⇔ minimise −Σ g_j
        c = -np.ones(n, dtype=float)

        try:
            res = linprog(c,
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs')
        except Exception as e:
            print(f"[lp {lp_iter}] linprog exception: {e}")
            break

        if not res.success:
            print(f"[lp {lp_iter}] linprog failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            print(f"[lp {lp_iter}] trivial solution, stop.")
            break

        # scale to keep Σg = 1  (the same normalisation used for a)
        g_scaled = g0 / sum_g0

        # convex combination step
        a = (1.0 - T_mix) * a + T_mix * g_scaled
        a /= a.sum()                               # re‑normalise (numerical safety)

        # evaluate after the update
        conv = np.convolve(a, a)
        new_val = float(2.0 * n * conv.max())

        if new_val + 1e-9 < best_val:
            best_val = new_val
            best_a = a.copy()
            no_improve_lp = 0
            print(f"[lp {lp_iter}] new best = {best_val:.6f}")
        else:
            no_improve_lp += 1

        # stop early if LP makes no progress for a few iterations
        if no_improve_lp >= 5:
            print(f"[lp] no improvement for {no_improve_lp} consecutive iterations – stop LP.")
            break

        lp_iter += 1

    # ------------------------------------------------------------------
    # 2) Local simulated‑annealing refinement
    # ------------------------------------------------------------------
    conv = np.convolve(a, a)
    cur_val = float(2.0 * n * conv.max())
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()
        print(f"[post‑lp] new best = {best_val:.6f}")

    iteration = 0
    while time.time() < deadline:
        # linear temperature schedule (very low because we are already close)
        progress = (time.time() - start_time) / (deadline - start_time)
        T = 0.01 * (1.0 - progress) + 1e-5 * progress

        # choose move type
        if np.random.rand() < 0.65:
            src, dst = targeted_move(a, conv)
        else:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

        # amount to move (up to 40 % of source mass)
        max_delta = a[src] * 0.40
        if max_delta < 1e-12:
            continue
        delta = np.random.rand() * max_delta

        a_new, conv_new = apply_move(a, conv, src, dst, delta)
        new_val = float(2.0 * n * np.max(conv_new))

        # Metropolis acceptance
        improve = new_val < cur_val
        if improve or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_val = new_val
            if new_val + 1e-9 < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA {iteration}] new best = {best_val:.6f}")

        iteration += 1

        # occasional full recomputation to avoid drift from rounding errors
        if iteration % 5000 == 0:
            conv = np.convolve(a, a)

    # ------------------------------------------------------------------
    # Final validation with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence returns {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float('inf')

    return best_a.tolist()
