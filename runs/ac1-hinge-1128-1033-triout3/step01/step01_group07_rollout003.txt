import time
import numpy as np
import cvxpy as cp

def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # choose length n
    n = 5000  # moderate
    # initial shape: inverse sqrt core + tail
    core_len = int(0.3 * n)
    core = 1.0 / np.sqrt(np.arange(1, core_len+1))
    tail_len = n - core_len
    tail = np.full(tail_len, 0.0)  # start with zeros
    f = np.concatenate([core, tail])
    # normalize sum
    target_sum = np.sqrt(2 * n)
    f = f * (target_sum / f.sum())
    
    # cvxpy参数
    f_param = cp.Parameter(n, nonneg=True)
    f_param.value = f

    # main loop
    best_f = f.copy()
    best_M = np.max(np.convolve(best_f, best_f))
    print(f"initial M = {best_M:.6f}")

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        M = np.max(np.convolve(f, f))
        # update parameter
        f_param.value = f
        M_param = cp.Parameter(nonneg=True, value=M)
        g = cp.Variable(n, nonneg=True)
        conv_fg = cp.conv(f_param, g)
        constraints = [conv_fg <= M_param]
        # solve LP: maximize sum(g)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception as e:
            print(f"LP solve failed: {e}")
            break
        if g.value is None:
            print("LP gave no solution")
            break
        g_val = np.maximum(g.value, 0.0)
        sum_g = np.sum(g_val)
        if sum_g < 1e-12:
            print("g_sum too small")
            break
        # scale g to target sum
        g_scaled = g_val * (target_sum / sum_g)

        # check self-conv of g_scaled
        max_gg = np.max(np.convolve(g_scaled, g_scaled))
        # choose mixing factor t

        if max_gg <= M * (1 + 1e-6):

"""
propose_candidate
=================

A deterministic refinement that follows the “linear‑programming + mixing”
strategy described in the problem statement.

* Start from the best known sequence (global ``height_sequence_1``) or,
  if unavailable, from a truncated 1/√i core.
* The sequence is normalised so that Σ aᵢ = √(2 n).  With this normalisation
  the evaluation function reduces to the maximum of the self‑convolution
  (the objective becomes M = maxₖ (a*a)[k]).
* At each iteration we solve the linear programme

      maximise    Σ gᵢ
      subject to  gᵢ ≥ 0
                  (f * g)[k] ≤ M   for all k = 0,…,2n‑2

  where *f* is the current sequence and *M* is its current maximum
  convolution value.  The programme pushes as much ℓ¹‑mass as possible
  into indices where the convolution with the current shape is still
  below the bound, i.e. into the “tails’’ of the step function.

* The optimiser returns a new candidate g.  After rescaling g so that it
  also satisfies Σ gᵢ = √(2 n) we mix it with the current f:

      f←(1−t)·f + t·g

  with a line‑search on *t* (starting at 0.5) that accepts the first
  mixture that strictly lowers the maximum convolution.  The process
  repeats until the time budget expires or no further improvement can
  be found.

This method systematically spreads the mass away from the high‑
convolution core, lowering the peak while keeping the ℓ¹‑norm fixed,
and often beats the simple simulated‑annealing scheme.
"""

import time
import math
import numpy as np

# Try to import cvxpy – the LP core of the algorithm.
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:          # pragma: no cover
    _CVXPY_AVAILABLE = False


def _max_convolution(seq: np.ndarray) -> float:
    """Return max value of the linear self‑convolution of seq using an FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())       # power‑of‑two size
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    return float(conv.max())


def _initial_sequence(n: int) -> np.ndarray:
    """
    Build a reasonable starting step function when no global solution is present:
    a 1/√i core followed by a zero tail, then normalise the ℓ¹‑norm.
    """
    core_len = max(2, int(0.2 * n))
    core = 1.0 / np.sqrt(np.arange(1, core_len + 1, dtype=np.float64))
    tail = np.zeros(n - core_len, dtype=np.float64)
    seq = np.concatenate([core, tail])
    target_sum = math.sqrt(2.0 * n)
    seq = seq * (target_sum / seq.sum())
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best bound.  The routine runs for at most ``budget_s`` seconds
    and logs its progress with ``print``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # leave a small safety margin

    # ------------------------------------------------------------------
    # 1.  Load a starting sequence (the best known one if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(globals()["height_sequence_1"], dtype=np.float64)
        n = len(init)
        # normalise to Σ aᵢ = √(2 n)  – this makes the objective equal to the
        # maximal self‑convolution value.
        target_sum = math.sqrt(2.0 * n)
        f = init * (target_sum / init.sum())
    else:
        n = 6000                         # a moderate size that works well with the LP
        f = _initial_sequence(n)

    # ------------------------------------------------------------------
    # 2.  Helper objects for the LP (cvxpy)
    # ------------------------------------------------------------------
    if not _CVXPY_AVAILABLE:            # pragma: no cover
        raise RuntimeError("cvxpy is required for this method but could not be imported.")

    # Parameter representing the current sequence inside the LP
    f_param = cp.Parameter(n, name="f")
    # (the parameter will be updated each iteration)

    best_seq = f.copy()
    best_M = _max_convolution(f)
    print(f"[init] n = {n}, initial max‑conv = {best_M:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    # 3.  Main refinement loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # Update parameter to the latest sequence
        f_param.value = f
        M = _max_convolution(f)                # current bound
        M_param = cp.Parameter(nonneg=True, value=M, name="M")

        # Decision variable g ≥ 0
        g = cp.Variable(n, nonneg=True, name="g")

        # Linear constraints: (f * g)[k] ≤ M  for all k
        conv_fg = cp.conv(f_param, g)          # length 2n‑1, linear in g
        constraints = [conv_fg <= M_param]

        # Objective: maximise Σ gᵢ  (i.e. increase the ℓ¹‑norm under the bound)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        # Solve the LP – ECOS is fast for medium‑sized linear programmes
        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=3000, abstol=1e-8, reltol=1e-8)
        except Exception as e:                 # pragma: no cover
            print(f"[iter {iteration}] LP failed: {e}")
            break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            print(f"[iter {iteration}] LP not solved to optimality (status={prob.status})")
            break

        g_val = np.array(g.value).flatten()
        # Clip tiny negatives that can appear due to solver tolerance
        g_val = np.maximum(g_val, 0.0)

        sum_g = g_val.sum()
        if sum_g <= 0.0:
            print(f"[iter {iteration}] LP returned zero sum – aborting.")
            break

        # --------------------------------------------------------------
        # 4.  Build a feasible mixture (mixing factor t) that lowers M
        # --------------------------------------------------------------
        # Scale g to the required ℓ¹‑norm ( √(2 n) )
        target_sum = math.sqrt(2.0 * n)
        g_scaled = g_val * (target_sum / sum_g)

        # First try a full replacement – it works if the new shape already
        # reduces the maximal convolution.
        new_M_full = _max_convolution(g_scaled)
        if new_M_full < best_M - 1e-9:
            f_new = g_scaled
            t_used = 1.0
        else:
            # Otherwise perform a line‑search on the mixing weight t.
            t = 0.5
            f_new = None
            while t > 1e-4:
                cand = (1.0 - t) * f + t * g_scaled
                cand_M = _max_convolution(cand)
                if cand_M < best_M - 1e-9:
                    f_new = cand
                    break
                t *= 0.7
            if f_new is None:                  # no improvement found
                print(f"[iter {iteration}] No improving mixture found – terminating.")
                break
            t_used = t

        # --------------------------------------------------------------
        # 5.  Accept if we have a genuine improvement
        # --------------------------------------------------------------
        cur_M = _max_convolution(f_new)
        if cur_M < best_M - 1e-12:
            best_M = cur_M
            best_seq = f_new.copy()
            f = f_new                         # continue from the improved point
            print(f"[iter {iteration}] improved M = {best_M:.6f} (t={t_used:.3f})")
        else:

            print(f"[iter {iteration}] no improvement (M stays {best_M:.6f})")

        if best_M <= 1.51:
            print("[target] reached M ≤ 1.51 – stopping early.")
            break

    print(f"[final] best max‑conv = {best_M:.6f}")
    return best_seq.tolist()
