          a[k] = a[k - 1] * (2 * k - 1) / (2 * k)

   n = len(a)
   sum_a = a.sum()
   best_seq = a.copy()
   best_score = evaluate_sequence(best_seq.tolist())
   print(f"[init] length={n}, initial score={best_score:.6f}")

   # Parameter sweep for reducing a[0]
   # consider values from current a[0] down to maybe 0.3, step 0.01
   a0_orig = a[0]
   # If a0_orig == 0, skip (unlikely). Ensure at least two elements.
   for new_a0 in np.linspace(a0_orig, max(0.0, a0_orig*0.2), num=81)[1:]:  # skip first which equals original
       scale = (sum_a - new_a0) / (sum_a - a0_orig) if (sum_a - a0_orig) != 0 else 1.0
       if scale <= 0:
           continue
       cand = a.copy()
       cand[0] = new_a0
       cand[1:] = cand[1:] * scale
       # Ensure non-negative
       cand = np.maximum(0.0, cand)
       # Renomalise to protect sum? Not needed
       try:
           cand_score = evaluate_sequence(cand.tolist())
       except Exception:
           continue
       if cand_score < best_score:
           best_score, best_seq = cand_score, cand
           print(f"[improve] a0={new_a0:.4f}, scale={scale:.4f}, new best score={best_score:.6f}")

   # Random local search around best_seq
   deadline = time.time() + budget_s - 5.0
   step_scale = 0.02 * (math.sqrt(2.0) / max(1.0, math.sqrt(n)))
   while time.time() < deadline:
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # propose moving delta from i to j
        delta = (np.random.rand() - 0.5) * step_scale
        if best_seq[i] + delta < 0:
            continue
        cand = best_seq.copy()
        cand[i] += delta
        cand[j] -= delta
        if cand[j] < 0:
            continue
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand
            print(f"[improve] random move, new best score={best_score:.6f}")

propose_candidate
-----------------
Generate a high‑quality step‑function sequence by starting from the classic
Catalan (central‑binomial) coefficients and then applying a targeted
simulated‑annealing style search.  The search repeatedly moves a small amount
of mass from an early index (where the coefficients are large and dominate the
self‑convolution) to a far‑right index (which contributes only to later
convolution positions).  This tends to lower the peak of the convolution while
preserving the total L1‑mass.  In addition a tiny isotropic random walk is
performed for fine‑grained refinement.  All moves respect the non‑negative
constraint; the objective is evaluated with the provided ``evaluate_sequence``.
The routine stops after the allotted time budget (default 1000 s) and returns
the best sequence found.
"""

    moving weight from the early, large coefficients to the far‑right tail.
    """

        # fall back to a modest size Catalan block
        n = 2000
        a = np.empty(n, dtype=float)
        a[0] = 1.0
        for k in range(1, n):
            a[k] = a[k - 1] * (2 * k - 1) / (2 * k)

    # keep a copy of the best sequence seen so far
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:               # safety net – should never happen
        best_score = float("inf")
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Search parameters.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # small safety margin
    # early indices (large coefficients) – we try to take mass from here
    early_cut = max(1, n // 10)               # first 10 % of the array
    # late indices (far tail) – we dump mass here
    late_start = max(0, n - n // 10)           # last 10 % of the array

    temperature = 1.0
    temp_decay = 0.9995
    min_temp = 1e-4

    # ------------------------------------------------------------------
    # 3. Main optimisation loop.
    #    *biased transfer*: move mass from early to late positions.
    #    *random walk*: occasional tiny perturbations of a random entry.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- biased mass transfer -------------------------------------------------
        # pick i from the early part, j from the late part
        i = np.random.randint(early_cut)
        j = np.random.randint(late_start, n)
        if i == j or best_seq[i] <= 0.0:
            continue

        # propose moving a random fraction of a[i] to a[j]
        # the fraction is uniformly drawn from (0, 0.5] to keep steps moderate
        frac = np.random.rand() * 0.5
        delta = best_seq[i] * frac

        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate candidate
        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        # ----- acceptance (simulated annealing) ------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[improve] move i={i}→j={j} δ={delta:.6f}  new best={best_score:.6f}")
        else:
            # occasional uphill moves to escape flat regions
            prob = math.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                best_seq = cand  # accept temporarily (does not become best)
        # temperature reduction
        temperature = max(min_temp, temperature * temp_decay)

        # ----- tiny isotropic refinement (once in a while) -------------------------
        if np.random.rand() < 0.01:
            idx = np.random.randint(n)
            # add a small zero‑mean perturbation (up to 2 % of typical magnitude)
            step_mag = 0.02 * (math.sqrt(2.0) / max(1.0, math.sqrt(n)))
            delta2 = (np.random.rand() - 0.5) * step_mag
            cand2 = best_seq.copy()
            cand2[idx] = max(0.0, cand2[idx] + delta2)

            # renormalise to keep the L1‑norm unchanged (objective is scale‑invariant)
            s = cand2.sum()
            if s > 0:
                cand2 *= best_seq.sum() / s
                try:
                    cand2_score = evaluate_sequence(cand2.tolist())
                except Exception:
                    continue
                if cand2_score < best_score:
                    best_score = cand2_score
                    best_seq = cand2
                    print(f"[improve] isotropic tweak idx={idx} Δ={delta2:.6f}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"[final] length={n}, best score={best_score:.6f}")
    return best_seq.tolist()
