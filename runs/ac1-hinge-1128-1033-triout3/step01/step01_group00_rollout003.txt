def auto_conv_fft(a):
    n = len(a)
    size = 1 << (2*n -1).bit_length()  # next pow2
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)
    conv = conv[:2*n-1]
    return conv

def conv_fft(a, b):
    n = len(a)
    m = len(b)
    size = 1 << (n+m-1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    conv = np.fft.irfft(A * B, size)
    return conv[:n+m-1]

"""
Propose a step‑function using a smooth‑max (soft‑max) gradient descent.
The algorithm works with a probability‑simplex representation (sum = 1,
non‑negative) and enforces symmetry a_i = a_{n-1-i}.  Starting from a set
of deterministic “U‑shaped’’ seeds (and optionally the historic
height_sequence_1) it iteratively refines the vector by:

1. Computing the self‑convolution via FFT.
2. Forming a smooth approximation of the max‑entry:
       w_k ∝ exp(β·(conv_k – max(conv)))   (β grows during optimisation)
3. Computing the exact gradient of the smooth‑max objective:
       ∇_i = 2·∑_j w_{i+j}·a_j
   (implemented efficiently as a convolution with the reversed weight
   vector, also via FFT).
4. Taking a projected gradient step onto the simplex and re‑symmetrising.
5. Occasionally perturbing the vector to escape shallow minima.

The procedure is time‑bounded (≈1000 s) and returns the best sequence
found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper distributions
# ----------------------------------------------------------------------
def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape: a[i] = (min(i,n‑1‑i)+1)^{-p}, normalised to sum = 1."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# FFT‑based convolutions (real-valued)
# ----------------------------------------------------------------------
def auto_conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the (linear) self‑convolution a*a (length 2·n‑1) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)
    return conv[: 2 * n - 1]


def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (real) via FFT, result length len(a)+len(b)-1."""
    n, m = a.shape[0], b.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    conv = np.fft.irfft(A * B, size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# Core refinement routine (soft‑max gradient descent)
# ----------------------------------------------------------------------
def _refine_vector(a: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Refine a single probability vector `a` for at most `time_budget` seconds.
    Returns the best vector found and its exact score
        score = 2·n·max(conv(a,a)).
    """
    start = time.time()
    n = a.shape[0]

    # initial score
    conv = auto_conv_fft(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    # hyper‑parameters
    beta = 1.0                # smooth‑max temperature
    max_beta = 500.0
    beta_mul = 1.03
    step = 0.10               # gradient step size
    step_mul = 0.9993

    while time.time() - start < time_budget:
        # ---------- convolution and soft‑max weights ----------
        conv = auto_conv_fft(a)
        max_c = conv.max()
        # w_k ∝ exp(β·(conv_k – max_c))  – numerically stable
        w = np.exp(beta * (conv - max_c))
        w /= w.sum()                      # now ∑ w = 1

        # ---------- gradient = 2·(w * a) (cross‑correlation) ----------
        rev_w = w[::-1]                   # reverse for cross‑correlation
        grad_full = conv_fft(a, rev_w)    # length = n + (2n‑1) – 1 = 3n‑2
        idx = 2 * n - 2 - np.arange(n)    # positions 2n‑2‑i
        g = 2.0 * grad_full[idx]

        # ---------- projected gradient step ----------
        a = a - step * g
        a = np.maximum(a, 0.0)
        a = simplex_projection(a)
        # enforce symmetry (helps to reduce the search space)
        a = 0.5 * (a + a[::-1])

        # ---------- evaluate exact score ----------
        conv_new = auto_conv_fft(a)
        score = 2.0 * n * conv_new.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"    [refine] t={time.time()-start:5.1f}s  n={n:5d}  score={score:.6f}")

        # ---------- schedule ----------
        beta = min(max_beta, beta * beta_mul)
        step *= step_mul

        # occasional random mass transfer to escape plateaus
        if np.random.rand() < 0.02:
            i, j = np.random.choice(n, 2, replace=False)
            delta = a[i] * np.random.rand() * 0.15
            a[i] -= delta
            a[j] += delta
            a = simplex_projection(a)
            a = 0.5 * (a + a[::-1])

    return best_a, best_score

    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Warm‑start from the historic solution if available
    # ------------------------------------------------------------------
    try:
        seq0 = np.array(height_sequence_1, dtype=float)
        if seq0.size > 0:
            seq0 = np.maximum(seq0, 0.0)
            if seq0.sum() > 0:
                seq0 /= seq0.sum()
                conv0 = auto_conv_fft(seq0)
                score0 = 2.0 * len(seq0) * conv0.max()
                best_seq = seq0.copy()
                best_score = score0
                print(f"Init from height_sequence_1: n={len(seq0)}  score={score0:.6f}")
    except NameError:
        pass  # no historic variable present

    # ------------------------------------------------------------------
    # 1.  Explore a few deterministic U‑shaped seeds (different lengths)
    # ------------------------------------------------------------------
    lengths = [3000, 5000, 8000, 12000]    # all cheap enough for a few FFT evals
    powers = [0.3, 0.5, 0.8, 1.0, 1.4, 1.8]

    print("=== Deterministic U‑shape exploration ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for p in powers:
            a = u_shaped_distribution(n, p)
            conv = auto_conv_fft(a)
            score = 2.0 * n * conv.max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  New seed: n={n:5d} p={p:4.1f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main stochastic refinement loop (repeat with various lengths)
    # ------------------------------------------------------------------
    print("\n=== Stochastic refinement phase ===")
    while time.time() < deadline:
        # pick a length – favour lengths that have shown good results
        n = int(np.random.choice([4000, 6000, 8000, 10000, 12000]))
        # initialise a candidate
        if best_seq is not None and len(best_seq) == n:
            a0 = best_seq.copy()
            # add a tiny random perturbation to avoid being stuck
            a0 += 0.003 * np.random.rand(n)
            a0 = np.maximum(a0, 0.0)
            a0 = simplex_projection(a0)
        else:
            # random U‑shape with a random exponent
            p = np.random.uniform(0.2, 1.8)
            a0 = u_shaped_distribution(n, p)

        # allocate a slice of the remaining budget for this refinement run
        remaining = deadline - time.time()
        if remaining < 4.0:
            break
        run_time = min(30.0, remaining)   # ≤30 s per refinement block
        cand_seq, cand_score = _refine_vector(a0, run_time)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"Global best improved: n={n:5d}  score={cand_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Fallback (should never happen, but keep it safe)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        a = np.random.rand(n)
        a = simplex_projection(a)
        best_seq = a
        print("Fallback to a random vector (very unlikely).")

    print(f"\n=== Search finished ===\nBest score found: {best_score:.6f}  (length {len(best_seq)})")
    # Return plain Python list – the external evaluator will perform its own clipping/normalisation
    return best_seq.tolist()
