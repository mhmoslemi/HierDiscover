"""
Hybrid LP + sub‑gradient optimizer for the max‑convolution problem.

The objective (for a non‑negative sequence `a` with sum = 1) is

    M = 2·n·max(conv(a, a))

i.e. minimise the ℓ∞‑norm of the self‑convolution.  The algorithm
alternates two complementary steps:

1. **Linear‑programming step** – for the current distribution `f`
   compute `M = max(conv(f,f))`.  Solve the LP

        maximise   Σ b_j
        subject to b_j ≥ 0,
                   (f * b)_k ≤ M   for all k,

   which finds a distribution `b` that spreads as much mass as possible
   while keeping the convolution under the same bound `M`.  The new
   distribution is obtained as a convex combination

        f ← (1‑t)·f + t·(b / Σb)

   which provably does not increase `M` for a sufficiently small step
   size `t`.

2. **Projected sub‑gradient descent** – using the index `k*` where the
   current convolution attains its maximum, the sub‑gradient of the
   objective is `g_i = 2·f_{k*‑i}` (zero if the index is out of range).
   A step `f ← Π_S (f – η·g)` (Π_S = simplex projection) smooths the
   distribution further.

The procedure starts from a random Dirichlet‑like vector (or from the
provided `height_sequence_1` if present) and repeats the two steps
until the time budget (≈ 1000 s) is exhausted.  The best sequence found
is returned as a plain Python list; `evaluate_sequence` will clip it to
[0, 1000] automatically.

The implementation uses only `numpy`, `scipy.optimize.linprog` (HiGHS)
and a fast Euclidean simplex projection.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project ``v`` onto the simplex {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML‑2008).
    """
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v.copy()

    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)                # 1‑based index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  The search combines a linear‑programming
    improvement step with a projected sub‑gradient descent.  The
    routine respects the provided time budget and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1. Problem size.
    # --------------------------------------------------------------
    # Use the length of the known construction if available; otherwise
    # pick a slightly larger size that still fits comfortably in memory.
    try:
        n = len(height_sequence_1)  # type: ignore[name‑defined]
    except Exception:
        n = 1500                     # a size that gives the optimizer room

    # --------------------------------------------------------------
    # 2. Initial distribution (sum == 1).
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals() and rng.random() < 0.4:
        seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name‑defined]
        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total > 0.0:
            f = seq / total
        else:
            f = rng.random(n)
            f = f / f.sum()
    else:
        # Dirichlet‑like start: exponential draws then normalise.
        f = rng.exponential(scale=1.0, size=n)
        f = f / f.sum()

    best_f = f.copy()
    best_score = 2.0 * n * np.max(np.convolve(f, f))

    # --------------------------------------------------------------
    # 3. Parameters for the iterative scheme.
    # --------------------------------------------------------------
    t = 0.15                # convex‑combination weight (small for stability)
    t_min = 0.02
    lr = 0.5                # step size for sub‑gradient phase
    lr_decay = 0.9995
    subgrad_iters = 0
    lp_iters = 0
    max_lp_iters = 25      # number of LP updates; each is cheap for the chosen n
    max_subgrad_iters = 4000

    # --------------------------------------------------------------
    # 4. Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 4.1 Linear‑programming improvement (if we have remaining LP budget)
        # ------------------------------------------------------------------
        if lp_iters < max_lp_iters:
            lp_iters += 1
            conv_f = np.convolve(f, f)
            M = conv_f.max()

            # Scale the constraints to improve numerical conditioning.
            scale = np.max(f)
            if scale == 0:
                scale = 1.0
            f_scaled = f / scale
            M_scaled = M / scale

            rows = 2 * n - 1
            # Build the (rows × n) matrix A such that (A @ b)_k = (f * b)_k.
            A = np.zeros((rows, n), dtype=float)
            # Column j contributes f_scaled to rows j … j+n‑1.
            for j in range(n):
                A[j:j + n, j] = f_scaled

            # Solve: maximise Σ b_j  ⇔ minimise –Σ b_j
            c = -np.ones(n, dtype=float)               # objective for linprog (minimise)
            b_ub = M_scaled * np.ones(rows, dtype=float)
            bounds = [(0.0, None) for _ in range(n)]

            try:
                from scipy.optimize import linprog
                # HiGHS accepts dense A_ub; a tiny time limit keeps us within budget.
                remaining = max(0.0, deadline - time.time() - 0.5)
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method='highs',
                    options={'presolve': True, 'time_limit': remaining}
                )
            except Exception as exc:
                print("Linprog failed or not available:", exc)
                break

            if not res.success:
                # If the LP cannot be solved (unlikely), fall back to gradient mode.
                print(f"LP iteration {lp_iters} failed: {res.message}")
                break

            b_opt = np.asarray(res.x, dtype=float)
            sum_b = b_opt.sum()
            if sum_b <= 0:
                # Degenerate solution – abort LP phase.
                break

            g = b_opt / sum_b                         # normalise to a probability vector
            # Convex combination (guarantees non‑increase of M for small t)
            f = (1.0 - t) * f + t * g
            f = _simplex_projection(f)                # numerical restore to simplex

            # Update best score after the LP step.
            cur_score = 2.0 * n * np.max(np.convolve(f, f))
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                print(f"[LP {lp_iters:3d}] new best M = {best_score:.6f}")

            # Reduce the step size t gradually (helps convergence).
            t = max(t_min, t * 0.95)

            # Early exit if we already beat the target.
            if best_score <= 1.503:
                break
            continue    # go to the next iteration of the outer loop

        # ------------------------------------------------------------------
        # 4.2 Projected sub‑gradient descent (soft‑max free – pure subgradient)
        # ------------------------------------------------------------------
        if subgrad_iters >= max_subgrad_iters:
            break
        subgrad_iters += 1

        conv = np.convolve(f, f)
        k_star = int(np.argmax(conv))                 # index of the maximal entry
        # Sub‑gradient: g_i = 2·f_{k*‑i} (zero outside the support)
        grad = np.zeros_like(f)
        # Compute slice of f that contributes to the gradient.
        i_low = max(0, k_star - (n - 1))
        i_high = min(k_star, n - 1)
        # Corresponding indices in grad:
        # For i in [i_low, i_high] we have grad[i] = 2·f[k_star‑i]
        grad[i_low:i_high + 1] = 2.0 * f[k_star - np.arange(i_low, i_high + 1)]

        # Gradient step + projection onto simplex
        f_new = f - lr * grad
        f = _simplex_projection(f_new)

        # Decay learning rate
        lr *= lr_decay

        # Evaluate and keep best
        cur_score = 2.0 * n * np.max(np.convolve(f, f))
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[GD {subgrad_iters:4d}] new best M = {best_score:.6f}")

        # Occasionally inject a small random perturbation to escape plateaus
        if subgrad_iters % 200 == 0:
            noise = rng.normal(scale=0.003, size=n)
            f = _simplex_projection(f + noise)

        # Dynamic reduction of the LP mixing factor as we progress
        if subgrad_iters % 500 == 0 and t > t_min:
            t = max(t_min, t * 0.9)

        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 5. Return the best sequence found (as a plain list of floats).
    # --------------------------------------------------------------
    result = np.clip(best_f, 0.0, 1000.0).tolist()
    print(f"Finished. Best M = {best_score:.6f} with length {len(result)}")
    return result
