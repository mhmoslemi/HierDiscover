"""
propose_candidate
-----------------

Gradient‑descent search on a smooth “Lp‑norm” approximation of the
max‑convolution.  Starting from the Catalan (central‑binomial) sequence,
the algorithm repeatedly:

1. computes the self‑convolution b = a*a,
2. forms the power‑weighted vector c = b**(p‑1) (with p growing from 4 to 40),
3. evaluates the correlation g_i = Σ_k c_k·a_{k‑i},
4. takes a step opposite to the gradient 2·p·g (the factor 2 accounts for the
   symmetry of the convolution),
5. projects back onto the feasible set (non‑negative and fixed L1‑norm).

The step size is chosen adaptively so that the largest change of any entry
does not exceed a small constant (≈0.01).  The procedure is stopped after
the allotted time (default 1000 s) and the best sequence seen so far is
returned.  A short stochastic refinement is performed at the end to squeeze
out a few extra points.

All numbers are kept non‑negative and are rescaled to have
∑a_i = √(2 n), which leaves the evaluation score invariant.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimizes
    evaluate_sequence.  The method uses gradient descent on a smooth
    approximation of the max‑convolution objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a moderate length (large enough to get a good constant,
    # small enough for fast iterations).  Reuse a previous solution if
    # it exists, but cap the size to keep the linear‑algebra cheap.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)
        if n > 3000:                # very long previous sequence – trim it
            n = 2000
    except NameError:
        n = 2000

    # ------------------------------------------------------------------
    # 1. Catalan (central‑binomial) coefficients – a proven good start.
    #    a_k = C(2k,k)/4^k, generated by the recurrence.
    # ------------------------------------------------------------------
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)

    # Normalise the L1‑norm to √(2 n) (the objective is scale‑invariant).
    target_sum = math.sqrt(2.0 * n)
    a *= target_sum / a.sum()
    seq = a.copy()

    # Keep the best sequence seen so far.
    best_seq = seq.copy()
    best_score = 2.0 * n * np.max(np.convolve(seq, seq)) / (target_sum ** 2)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Gradient‑descent parameters.
    # ------------------------------------------------------------------
    p = 4                     # start with a modest Lp‑exponent
    lr = 0.02                 # maximal absolute change of any entry per step
    it = 0
    stagnation = 0            # counter for rounds without improvement

    while time.time() < deadline:
        it += 1

        # ---------- convolution and current max ----------
        b = np.convolve(seq, seq)                # length 2n‑1
        max_b = b.max()
        # current evaluation (exact, no call to evaluate_sequence)
        cur_score = 2.0 * n * max_b / (target_sum ** 2)

        # ---------- increase the exponent slowly ----------
        if it % 50 == 0 and p < 40:
            p = min(p + 4, 40)
            print(f"[it {it}] increased Lp exponent to p={p}")

        # ---------- compute the gradient of Σ b_k^p ----------
        # Scale b to avoid overflow when p is large.
        scale = max_b if max_b > 0 else 1.0
        b_scaled = b / scale
        c = np.power(b_scaled, p - 1)            # c_k = (b_k/scale)^{p‑1}
        # correlation a ⊛ c   (full mode)
        corr = np.correlate(seq, c, mode='full')
        offset = len(b) - 1                      # index where lag = 0
        grad_raw = corr[offset:offset + n]       # length n
        grad = 2.0 * p * grad_raw                  # ∂ Σ b^p / ∂ a_i

        # ---------- adaptive step – limit per‑coordinate change ----------
        max_abs = np.max(np.abs(grad)) + 1e-16
        step = lr / max_abs                       # ensures |Δa_i| ≤ lr
        new_seq = seq - step * grad

        s = new_seq.sum()
        if s > 0:

        else:

            new_seq = seq.copy()

        if new_score < best_score:

            best_seq = new_seq.copy()
            stagnation = 0

        seq = new_seq

            if cand_score < best_score:

                best_seq = cand
                print(f"[refine] improved score={best_score:.6f}")

    print(f"[final] best score={best_score:.6f} after {it} main iterations")
    return best_seq.tolist()
