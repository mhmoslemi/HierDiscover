from scipy.optimize import linprog

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Iterative linear‑programming refinement of a Catalan‑type step function.
    Starts from the central‑binomial (Catalan) coefficients (scaled to
    ∑a_i = √(2n)), then repeatedly solves a linear program that maximises
    the L1‑mass under the constraint (f * g) ≤ M, where M is the current
    maximal self‑convolution. The solution is mixed with the current
    sequence, keeping the total sum unchanged, which monotonically reduces
    the evaluation score 2·n·max(f*f)/∥f∥_1².  The loop runs until the time
    budget expires and returns the best sequence found.
    """
    # ------------------------------------------------------------------
    # 1. Settings and initial sequence (Catalan coefficients).
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    np.random.seed(seed)

    # Length of the sequence: reuse previous best if present.
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000

    # Catalan / central‑binomial coefficients: a_k = product_{i=1}^k (2i‑1)/(2i)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)

    # Normalise to the invariant L1‑norm √(2n); the objective is scale‑invariant.
    target_sum = math.sqrt(2.0 * n)
    f = a * (target_sum / a.sum())

    best_seq = f.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Main optimisation loop: LP‑based mass redistribution.
    # ------------------------------------------------------------------
    t = 0.2                     # mixing factor (will be annealed)
    iteration = 0

    while time.time() < deadline:
        # Current maximal self‑convolution M
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # Build the convolution matrix A such that (f * g) = A @ g.
        rows = 2 * n - 1
        i = np.arange(rows)[:, None]          # shape (rows,1)
        j = np.arange(n)[None, :]              # shape (1,n)
        idx = i - j                            # shape (rows,n)
        mask = (idx >= 0) & (idx < n)
        # Allocate A directly; fill only the masked positions.
        A = np.zeros((rows, n), dtype=np.float64)
        A[mask] = f[idx[mask]]

        # Linear programme: maximise sum(g)  <=> minimise –sum(g)
        c = -np.ones(n, dtype=np.float64)
        b = np.full(rows, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve with HiGHS (fast dense LP solver)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})

        del A

            print(f"[lp] iter {iteration}: LP failed – {res.message}")

            idx_rand = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.01 * f[idx_rand]
            f[idx_rand] = max(0.0, f[idx_rand] + delta)

        sum_g = g.sum()
        sum_f = f.sum()    # constant = √(2n)
        if sum_g <= 0:
            print(f"[lp] iter {iteration}: zero sum returned, aborting")
            break

        # Rescale g to keep the L1‑norm unchanged.
        scale = sum_f / sum_g
        g_scaled = g * scale

        # Mix the new candidate with the current sequence.
        f = (1.0 - t) * f + t * g_scaled

        # Numerical safety – enforce non‑negativity.

            best_seq = f.copy()
            print(f"[improve] iter {iteration}, new best score={best_score:.6f}")

        # Simple annealing of the mixing factor.
        if iteration % 10 == 0 and iteration > 0:
            t = max(0.01, t * 0.9)

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {iteration} iterations, length={n}")
    return best_seq.tolist()
