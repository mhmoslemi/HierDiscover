"""
Propose a candidate step‑function (list of non‑negative heights).

The algorithm is a two‑phase projected Adam optimizer on the
smoothed (log‑sum‑exp) approximation of the max‑convolution objective.
* Phase 1 – small smoothing parameter α gives a smooth landscape and
  quickly reduces the bound.
* Phase 2 – α is increased until the soft‑max becomes almost a hard max;
  we then switch to a sub‑gradient built from the true peak positions.
A Euclidean projection onto the probability simplex keeps the iterates
feasible.  Random jitter, occasional symmetry enforcement and an
early‑stop when the target 1.503 is reached are added to improve
exploration.  The routine runs until the given time budget expires and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    Search for a step‑function (list of non‑negative heights) that
    minimises

          M = 2·n·max(conv(a,a)) / (sum(a)**2)

    using a projected Adam optimiser with a soft‑max approximation of
    the max‑operator.  The best sequence found before the time budget
    expires is returned.

    # ------------------------------------------------------------------
    # 1. Choose a few problem sizes.
    #    The benchmark construction has length 1153 – we always try it.
    #    If enough time remains we also try a larger size (2000) which
    #    gives the optimiser more freedom.

    # ------------------------------------------------------------------

    best_seq = None
    best_score = float("inf")

    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 3. Optimisation hyper‑parameters (Adam).
    # ------------------------------------------------------------------

    beta2 = 0.999
    eps = 1e-8
    lr_initial = 0.5
    lr_decay = 0.9995
    alpha_start = 1.0          # initial smoothing for soft‑max
    alpha_mul = 1.005          # increase α each iteration
    alpha_hard_switch = 200.0  # after this α acts like a hard max

    # ------------------------------------------------------------------
    # 4. Main loop over problem sizes.
    # ------------------------------------------------------------------
    for n in lengths:

        idx = 2 * n - 2 - np.arange(n)

        start_from_known = False

            try:

                if seq0.shape[0] == n:
                    start_from_known = True
            except Exception:
                start_from_known = False

        # Perform a few independent restarts for each size.
        while time.time() < deadline:
            # --------------------------------------------------------------
            # 4.1 Initialise a probability vector p (sum(p)=1).
            # --------------------------------------------------------------
            if start_from_known and rng.random() < 0.3:
                p_raw = np.clip(seq0, 0.0, 1000.0)
                total = p_raw.sum()
                if total > 0:
                    p = p_raw / total
                else:
                    p = np.full(n, 1.0 / n)
            else:

                p = rng.random(n)
                p /= p.sum()

            # Adam moment estimates.
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0

            # Optimiser schedules.
            lr = lr_initial
            alpha = alpha_start

            # Inner optimisation loop.
            while time.time() < deadline:
                t += 1

                # ----------------------------------------------------------
                # 4.2 Compute convolution and the current objective.
                # ----------------------------------------------------------
                conv = np.convolve(p, p)                # length 2n‑1
                max_conv = conv.max()
                score = 2.0 * n * max_conv               # exact M (sum(p)=1)

                # Track the best sequence.

                    best_score = score
                    best_seq = p.copy()
                    print(f"Iter {t:6d} – n={n} – new best M={best_score:.6f}")

                    # Early exit if we have reached the target bound.
                    if best_score <= 1.503:
                        print("Target bound reached – exiting.")
                        return np.clip(best_seq, 0.0, 1000.0).tolist()

                # ----------------------------------------------------------
                # 4.3 Choose gradient (soft‑max or hard max).
                # ----------------------------------------------------------
                if alpha < alpha_hard_switch:
                    # Soft‑max approximation of the max.
                    z = alpha * conv
                    z_max = np.max(z)
                    exp_z = np.exp(z - z_max)  # stabilise
                    w = exp_z / exp_z.sum()
                else:

                    # A relative tolerance of 1e‑12 is usually enough.

                # ----------------------------------------------------------
                # 4.4 Adam update + simplex projection.
                # ----------------------------------------------------------

                v = beta2 * v + (1.0 - beta2) * (grad ** 2)

                v_hat = v / (1.0 - beta2 ** t)

                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                p_next = p - step

                # Occasional random jitter to escape shallow plateaus.
                if t % 500 == 0:
                    jitter = rng.normal(scale=0.005, size=n)
                    p_next += jitter

                # Enforce symmetry every 1000 steps – often helps flatten the convolution.
                if t % 1000 == 0:
                    p_next = (p_next + p_next[::-1]) * 0.5

                # Project back onto the probability simplex.
                p = _simplex_projection(p_next)

                # ----------------------------------------------------------
                # 4.5 Schedule updates.
                # ----------------------------------------------------------
                lr *= lr_decay
                alpha *= alpha_mul

                # Minor early‑stop if we see no improvement for a long stretch.
                # (This is a cheap heuristic; we simply continue until time runs out.)

            # End of inner loop – start a new restart if time permits.
            if best_score <= 1.503:
                break

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered (clipped to the allowed range).
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution (should never happen).
        best_seq = np.full(lengths[0], 1.0 / lengths[0])
    result = np.clip(best_seq, 0.0, 1000.0).tolist()
    return result
