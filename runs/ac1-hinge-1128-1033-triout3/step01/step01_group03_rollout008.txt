"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The algorithm alternates a simple linear‑program
sub‑problem with a projected‑gradient refinement:

* For a fixed vector `a` we solve the LP

      maximise  1ᵀ b
      subject to a * b  ≤  M          (full linear convolution)
                 b ≥ 0
                 Σ b = 1

  where `M = max(a∗a)`.  The solution places the mass of `b` where the
  coefficients of `a` are small, guaranteeing `max(a∗b) ≤ M`.
  After normalising `b` to the probability simplex we replace `a←b`.
  Because `M` never increases, the objective `2·n·M` is non‑increasing.

* Every few iterations a short projected‑gradient step (with a soft‑max
  approximation of the `max`) is performed to escape possible plateaus
  and to smooth the shape.

The linear programme is solved with SciPy’s HiGHS solver (very fast for
the moderate sizes used).  If SciPy is unavailable the routine falls
back to the gradient descent method that was used in the reference
implementation.

The whole search runs for at most `budget_s` seconds and returns the
best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al., 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic start: coefficients of (1‑x)^(-1/2)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^(-1/2), normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the (soft) max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, alpha: float = 8.0) -> np.ndarray:
    """
    Gradient of  the soft‑max of the convolution a∗a.
    `alpha` controls the smoothness (larger → closer to the true max).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()
    # soft‑max weights (larger weight for larger entries)
    w = np.exp(alpha * (conv - max_val))
    w /= w.sum()

    # grad_i = 2 * Σ_k w_k a_{k-i}  (correlation of w with a)
    grad = 2.0 * np.correlate(w, a, mode="valid")
    return grad


# ----------------------------------------------------------------------
# Linear‑program sub‑problem (maximise ℓ₁ norm of b under a∗b ≤ M)
# ----------------------------------------------------------------------
def _lp_optimize(a: np.ndarray, M: float):
    """
    Solve
        maximise   Σ b_i
        subject to a∗b ≤ M   (full linear convolution)
                   b_i ≥ 0 , Σ b_i = 1
    Returns the optimiser b (already on the simplex) and a flag.
    """
    try:
        # SciPy may be missing in the execution environment – fall back.
        import scipy.optimize as opt
        import scipy.sparse as sp
    except Exception:   # pragma: no cover
        return None, False

    n = a.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    pos = 0
    for j in range(n):
        rows[pos:pos + n] = np.arange(j, j + n)   # rows j … j+n‑1
        cols[pos:pos + n] = j
        pos += n
    data = np.tile(a, n)                     # a repeated for each column
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # right‑hand side: every convolution entry must be ≤ M
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # equality: Σ b_i = 1
    A_eq = np.ones((1, n), dtype=np.float64)
    b_eq = np.array([1.0], dtype=np.float64)

    # maximise Σ b_i  →  minimise ‑Σ b_i
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # HiGHS is very fast and accepts sparse matrices
    res = opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None, False

    b = res.x
    # numerical noise → project back onto the simplex
    b = _project_simplex(b)
    return b, True


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` within the allotted time budget.

    The search uses a small linear‑program sub‑problem (see the comment
    above) and occasional projected‑gradient steps.  The best sequence
    encountered is returned.
    """
    np.random.seed(seed)
    # length of the vector – a moderate size keeps the LP cheap
    n = kwargs.get("length", 1200)

    deadline = time.time() + budget_s - 0.5   # a little safety margin
    a = _sqrt_series(n)                      # start from the analytic shape

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    iter_cnt = 0
    no_improve = 0

    # parameters for the optional gradient refinement
    grad_step = 0.15
    grad_alpha = 8.0          # soft‑max smoothing
    grad_decay = 1e-4

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) linear‑program step
        # ------------------------------------------------------------------
        conv = np.convolve(a, a)
        M = conv.max()
        b, ok = _lp_optimize(a, M)

        if ok and b is not None:
            # normalise (should already sum to 1) and evaluate
            cand = b
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                no_improve = 0
                print(f"[{iter_cnt:5d}] LP improve → {best_score:.6f}")
            else:
                no_improve += 1

            a = cand   # continue from the LP optimiser
        else:
            # LP failed – fall back to a gradient step
            no_improve += 1

        # ------------------------------------------------------------------
        # 2) occasional projected‑gradient refinement (every 30 iters)
        # ------------------------------------------------------------------
        if iter_cnt % 30 == 0:
            grad = _softmax_gradient(a, alpha=grad_alpha)
            step = grad_step / (1.0 + grad_decay * iter_cnt)
            a = _project_simplex(a - step * grad)

        # ------------------------------------------------------------------
        # 3) stagnation handling – random restart if we are stuck too long
        # ------------------------------------------------------------------
        if no_improve >= 2500:
            a = _sqrt_series(n)
            a = _project_simplex(a + np.random.randn(n) * 1e-3)
            no_improve = 0
            print(f"[{iter_cnt:5d}] restart triggered")

        iter_cnt += 1

    print(f"[final] best score = {best_score:.6f} after {iter_cnt} iterations")
    return best_seq.tolist()
