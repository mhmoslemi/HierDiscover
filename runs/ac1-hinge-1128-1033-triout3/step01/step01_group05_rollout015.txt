from scipy.optimize import linprog

def solve_lp(f):
    n = len(f)
    M = np.max(np.convolve(f, f))
    rows = 2 * n - 1
    A = np.zeros((rows, n))
    for k in range(rows):
        # i ranges over [max(0, k-(n-1)), min(k, n-1)]
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # for each i in range, column j = k - i
        # equivalently j ranges from max(0, k-(n-1))? Actually j = k - i, ranges from k-i_max to k-i_min.
        # Let's loop over i and set column j.
        for i in range(i_min, i_max + 1):
            j = k - i
            A[k, j] = f[i]
    # Solve LP
    c = -np.ones(n)  # maximize sum g
    b = np.full(rows, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    return res

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    import time
    import numpy as np
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # parameters
    N = 500  # length of the sequence
    # generate initial binomial coefficients
    coeffs = generate_binomial_prefix(N)
    f = coeffs[:N]  # already floats
    # normalise to sum 1 (scale invariant)
    f = f / np.sum(f)
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] N={N}, score={best_score:.6f}")

    # main optimisation loop
    t = 0.2  # convex combination weight
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute current bound M
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)
        # attempt LP improvement if solver available
        if linprog is not None:
            n = len(f)
            rows = 2 * n - 1
            # build constraint matrix A (rows x n)
            A = np.zeros((rows, n), dtype=np.float64)
            for k in range(rows):
                i_min = max(0, k - (n - 1))
                i_max = min(k, n - 1)
                # for each i, column j = k - i
                # vectorized: columns = np.arange(i_min, i_max+1)
                # j_vec = k - np.arange(i_min, i_max+1)
                i_vals = np.arange(i_min, i_max + 1)
                j_vals = k - i_vals
                A[k, j_vals] = f[i_vals]
            # solve LP: maximize sum(g) => minimize -sum(g)
            c = -np.ones(n)
            b_ub = np.full(rows, M)
            bounds = [(0, None)] * n
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 1.0})
            if res.success:
                g0 = res.x
                sum_g0 = np.sum(g0)
                if sum_g0 > 0:
                    # normalized version
                    g = g0 / sum_g0
                    # convex combination
                    f_new = (1 - t) * f + t * g
                    # evaluate
                    score_new = evaluate_sequence(f_new.tolist())
                    if score_new < best_score:
                        best_f = f_new.copy()
                        best_score = score_new
                        print(f"[improve-LP] it={iteration}, score={best_score:.6f}")
                    # accept update if it lowers M (or allow small upward moves)
                    conv_new = np.convolve(f_new, f_new)
                    M_new = np.max(conv_new)
                    # simple acceptance rule: accept if M decreases, else keep old f
                    if M_new < M or np.random.rand() < 0.01:
                        f = f_new
                        continue
            # if LP fails, break out to fallback hill climbing
        # fallback random perturbation hill climbing
        # choose two indices transfer
        i_src = np.random.randint(N)
        i_tgt = np.random.randint(N-1)
        if i_tgt >= i_src:
            i_tgt += 1
        # amount to move proportional to source value
        delta = np.random.rand() * 0.1 * f[i_src]
        if delta > f[i_src]:
            delta = f[i_src]
        cand = f.copy()
        cand[i_src] -= delta
        cand[i_tgt] += delta
        score_cand = evaluate_sequence(cand.tolist())
        if score_cand < best_score:
            best_f = cand.copy()
            best_score = score_cand
            print(f"[improve-HC] it={iteration}, score={best_score:.6f}")
        # accept if improved
        if score_cand < best_score * 0.999:
            f = cand
        # optionally break if no improvement after many iterations
        if iteration % 2000 == 0:
            # reduce t
            t = max(0.01, t * 0.9)
            print(f"[info] iteration {iteration}, t={t:.4f}, best={best_score:.6f}")
    # final sequence
    final_seq = [float(max(0.0, x)) for x in best_f.tolist()]
    return final_seq

"""
Propose a sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Analytic start** – use the classic “binomial” (constant‑convolution)
   coefficients `b_k = C(2k, k) / 4**k`.  After normalising to sum = 1 this
   gives the known upper bound ≈ π/2 ≈ 1.5708.

2. **Linear‑programming improvement** – for the current vector `f`
   (always normalised to sum = 1) compute `M = max(conv(f,f))`.  Solve the LP

        maximise   Σ g_j
        subject to  f * g ≤ M   (component‑wise)
                    g_j ≥ 0

   which finds a distribution `g₀` that can carry *more* total mass without
   exceeding the same convolution bound `M`.  After normalising
   `g = g₀ / Σ g₀` we replace `f ← (1‑t)·f + t·g` (the mixture also has
   sum = 1).  This strictly reduces the maximal self‑convolution after a few
   iterations.

3. **Fine‑grained stochastic hill‑climb** – a cheap random‑transfer move is
   applied repeatedly (moving a small amount of mass from one coordinate to
   another) and accepted if the true objective improves.

All steps respect the overall time budget (default 1000 s).  The function
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the constant‑convolution (binomial) coefficients
# ----------------------------------------------------------------------
def generate_binomial_prefix(max_n: int):
    """
    Returns an array `coeffs` where coeffs[k] = binom(2k, k) / 4**k  for
    0 <= k < max_n.  Computed by the recurrence  b_{k+1} = b_k * (2k+1) / (2(k+1)).
    """
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0                     # b_0
    coeffs[0] = cur
    for k in range(1, max_n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a step‑function (list of heights) with a low evaluation score.
    The method combines an analytic start, a linear‑programming based
    refinement and a final stochastic hill‑climber.  It respects the supplied
    time budget (seconds) and returns the best sequence found.
    """
    # timing
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # try to import a QP/LP solver; fall back to pure hill‑climb if unavailable
    try:
        from scipy.optimize import linprog
    except Exception:                     # pragma: no cover
        linprog = None

    # ------------------------------------------------------------------
    # Phase 1 – analytic start (binomial coefficients)
    # ------------------------------------------------------------------
    # length of the sequence – a few hundred is enough for good precision
    N = 500
    coeffs = generate_binomial_prefix(N)
    f = coeffs.astype(np.float64)          # raw (positive) coefficients
    f = f / np.sum(f)                      # normalise to sum = 1

    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] N={N}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based improvement loop
    # ------------------------------------------------------------------
    t = 0.2                                 # convex‑combination weight
    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution bound
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        # --------------------------------------------------------------
        # Try a linear‑programming step if a solver is available
        # --------------------------------------------------------------
        if linprog is not None:
            n = len(f)
            rows = 2 * n - 1

            # Build the constraint matrix A (rows × n) where
            #   (A @ g)[k] = Σ_i f_i * g_{k-i}  = (f * g)_k
            A = np.zeros((rows, n), dtype=np.float64)
            for k in range(rows):
                i_min = max(0, k - (n - 1))
                i_max = min(k, n - 1)
                i_vals = np.arange(i_min, i_max + 1)
                j_vals = k - i_vals
                A[k, j_vals] = f[i_vals]

            # maximise Σ g_j  <=> minimise -Σ g_j
            c = -np.ones(n, dtype=np.float64)
            b_ub = np.full(rows, M, dtype=np.float64)
            bounds = [(0, None)] * n

            # solve; we cap the internal solver time a little to stay safe
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={"presolve": True})

            if res.success:
                g0 = res.x
                sum_g0 = np.sum(g0)
                if sum_g0 > 0:
                    # normalise g to have sum = 1
                    g = g0 / sum_g0
                    # convex mixture
                    f_new = (1.0 - t) * f + t * g
                    # evaluate true objective (scale‑invariant, sum = 1)
                    score_new = evaluate_sequence(f_new.tolist())

                    if score_new < best_score:
                        best_f = f_new.copy()
                        best_score = score_new
                        print(f"[LP‑improve] it={it}, score={best_score:.6f}")

                    # optional acceptance criterion – keep the update if it
                    # reduces the bound M (or occasionally accept a tiny increase)
                    conv_new = np.convolve(f_new, f_new)
                    M_new = np.max(conv_new)
                    if M_new < M or np.random.rand() < 0.01:
                        f = f_new
                        continue          # go to next iteration

        # --------------------------------------------------------------
        # Phase 3 – simple stochastic hill‑climb (fallback / refinement)
        # --------------------------------------------------------------
        # randomly move a small amount of mass from one index to another
        i_src = np.random.randint(N)
        i_tgt = np.random.randint(N - 1)
        if i_tgt >= i_src:
            i_tgt += 1
        # transfer up to 10 % of the source entry
        delta = np.random.rand() * 0.10 * f[i_src]
        delta = min(delta, f[i_src])          # stay non‑negative
        cand = f.copy()
        cand[i_src] -= delta
        cand[i_tgt] += delta
        score_cand = evaluate_sequence(cand.tolist())

        if score_cand < best_score:
            best_f = cand.copy()
            best_score = score_cand
            print(f"[HC‑improve] it={it}, score={best_score:.6f}")

        # accept the candidate if it improves the current sequence
        if score_cand < evaluate_sequence(f.tolist()):
            f = cand

        # periodically shrink the mixing factor to fine‑tune
        if it % 2000 == 0:
            t = max(0.01, t * 0.9)
            print(f"[info] iteration={it}, t={t:.4f}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity – ensure non‑negativity and numeric bounds
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, x)) for x in best_f.tolist()]
    return final_seq
