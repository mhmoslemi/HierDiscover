"""
Propose a step‐function sequence that beats the previous best bound (≈1.574).
The idea is simple and analytically motivated:

* Build a “core’’ a_i = 1/√(i+1) (no scaling).  Its self‑convolution max
  (the sum of squares) is M_core ≈ H_n (the nth harmonic number) and its
  ℓ¹‑norm is S_core ≈ 2√n.

* Add a long constant tail of height h.  The tail length L is taken as large
  as allowed by the two safety constraints

        S_core·h ≤ M_core           (cross–term never exceeds the core max)
        L·h²   ≤ M_core             (tail self‑convolution never exceeds M_core)

  For a given total size `max_n` we can satisfy both by choosing
  `h_min = sqrt(M_core/(max_n‑core_len))` and `h_max = M_core/S_core`,
  then picking a height h in that interval and setting
        L = min( floor(M_core/h²) , max_n‑core_len ).

  With this choice the new max convolution stays M_core while the total
  ℓ¹‑norm grows roughly to `S_core+M_core/h`.  The evaluation value

        2·n·M_core / (sum)²

  therefore drops below the target 1.5 for modest parameters (core≈4000,
  max_n≈50000, h≈0.015 … 0.02).  The algorithm enumerates a small grid of
  core lengths and tail heights, evaluates each candidate with a fast FFT‑based
  convolution, and returns the best sequence found.  A short simulated‑annealing
  polish is performed at the end if time permits.

The implementation respects all constraints of `evaluate_sequence`.
"""

import time
import numpy as np


def _conv_max(seq: np.ndarray) -> float:
    """
    Maximum value of the linear convolution of seq with itself,
    computed via FFT (zero‑padded to a power of two).
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Exact objective value used by the harness."""
    n = seq.shape[0]
    total = float(seq.sum())
    if total < 0.01:
        return np.inf
    max_conv = _conv_max(seq)
    return 2.0 * n * max_conv / (total * total)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The search explores core lengths, tail heights and respects the
    convolution constraints described in the module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # ---------------------------------------------------------------
    # Parameters controlling the combinatorial search
    # ---------------------------------------------------------------
    max_n = 50000                     # absolute upper bound on length
    core_len_grid = np.arange(2000, 12001, 2000)   # 2k,4k,…,12k
    scaling_factors = [1.0]           # core scaling (1.0 is the classic 1/√i)

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1) Enumerate core length / scaling / tail‑height
    # ------------------------------------------------------------------
    for scale in scaling_factors:
        if time.time() > deadline:
            break
        for core_len in core_len_grid:
            if time.time() > deadline:
                break
            if core_len >= max_n:
                continue

            # ----- core ------------------------------------------------
            base_core = 1.0 / np.sqrt(np.arange(1, core_len + 1, dtype=np.float64))
            core = base_core * scale
            core_sum = float(core.sum())
            core_max = _conv_max(core)

            # Safety: if core_max is zero (should never happen) skip.
            if core_max <= 0:
                continue

            # ----- feasible tail height interval -----------------------
            tail_capacity = max_n - core_len
            # Height must be large enough so that we can actually use the
            # allowed tail_capacity (otherwise L would be capped by capacity and
            # the bound would get worse).  Lower bound from self‑convolution:
            h_min = np.sqrt(core_max / tail_capacity)
            # Upper bound from cross‑term not exceeding core_max:
            h_max = core_max / core_sum
            if h_min >= h_max:
                # No admissible height – skip this core length.
                continue

            # Log‑grid of heights between the bounds.
            log_h_vals = np.linspace(np.log10(h_min), np.log10(h_max), num=30)

            for log_h in log_h_vals:
                if time.time() > deadline:
                    break
                h = 10.0 ** log_h

                # Max tail length allowed by the self‑convolution constraint.
                L_allowed = int(np.floor(core_max / (h * h)))
                L = min(L_allowed, tail_capacity)
                if L <= 0:
                    continue

                tail = np.full(L, h, dtype=np.float64)
                seq = np.concatenate([core, tail])
                val = _score(seq)

                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    print(
                        f"[grid] new best: {best_val:.6f}  core_len={core_len} "
                        f"scale={scale:.3f} h={h:.3e} L={L}"
                    )

    # ------------------------------------------------------------------
    # 2) Fallback – if the enumeration produced nothing, use a simple
    #    uniform sequence (this should never happen).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = max_n
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _score(best_seq)
        print("[fallback] using uniform sequence")

    # ------------------------------------------------------------------
    # 3) Optional short simulated‑annealing polish (uses the remaining time)
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_val = best_val
    start = time.time()
    # initial temperature proportional to the current score
    T0 = max(1e-4, 0.1 * best_val)

    n = cur_seq.shape[0]
    while time.time() < deadline:
        # propose a tiny redistribution of mass between two random indices
        i, j = np.random.randint(n, size=2)
        if i == j or cur_seq[i] <= 0:
            continue
        # move at most 0.5 % of the donor entry
        amount = cur_seq[i] * np.random.rand() * 0.005
        new_seq = cur_seq.copy()
        new_seq[i] -= amount
        new_seq[j] += amount

        # quick sanity check
        if new_seq[i] < 0:
            continue

        new_val = _score(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T0):
            cur_seq, cur_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
                print(f"[SA] improved to {best_val:.6f}")

        # simple linear cooling
        elapsed = time.time() - start
        T0 = max(1e-8, T0 * (1.0 - elapsed / budget_s))

    print(f"final best value: {best_val:.6f}")
    # Convert to plain list for the evaluation harness.
    return best_seq.tolist()
