import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n)
    # Build convolution matrix A via Toeplitz-like as vectorized in cp? We can precompute A in numpy then use cp.Parameter.
    # However cp can accept A as constant numpy array.

    A = np.zeros((2*n-1, n), dtype=np.float64)
    # Fill as described.
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # i runs i_start..i_end; corresponding index j = k-i
        # For each i, A[k, j] = f[i]
        # j = k - i

        # vectorized:
        i_vals = np.arange(i_start, i_end+1)
        j_vals = k - i_vals
        A[k, j_vals] = f[i_vals]

    constraints = [A @ g <= M + 1e-12, g >= 0]

"""
propose_candidate
-----------------

Search for a non‑negative step function with a low value of ``evaluate_sequence``.
The method is based on the following idea:

* Start from the classical arcsine coefficients (the power‑series of
  (1‑z)^(-1/2)).  For a sequence of length ``n`` the auto‑convolution is
  equal to the constant 1 on the first ``n`` entries, therefore the
  current evaluation is exactly ``π/2 ≈ 1.5708`` – the best value known from a
  simple construction.

* The evaluation function is scale‑invariant, i.e. only the *shape* of the
  vector matters.  Keeping the maximal convolution value ``M`` unchanged
  while increasing the sum of the entries lowers the final score:
  ``score = 2·n·M / (sum(a)^2)``.
  The theoretical lower bound for a fixed length ``n`` is ``1`` (obtained
  when ``M`` equals the average convolution value).

* At each outer iteration we solve a **linear programme** that finds a
  non‑negative vector ``g`` with maximal total weight under the linear
  constraints  ``(f * g)_k ≤ M`` for every convolution index ``k``.
  This linear programme is cheap (≈ 10³ variables for ``n≈800``) and can be
  solved with ``cvxpy``/``ECOS``.

* After obtaining ``g`` we mix it with the current vector ``f``:
      ``f ← (1‑t)·f + t·g``  ( ``t`` is chosen by a short back‑tracking line‑search
      so that the new maximal convolution does not exceed the old bound ``M``).
  The mixing strictly increases the sum of the coefficients while keeping
  the maximal convolution at most ``M``.  Consequently the evaluation
  decreases.

* The outer loop repeats until the time budget runs out or no further
  improvement is possible.  The best sequence seen so far is returned.

The algorithm uses only a few hundred variables, therefore the whole
optimization (including several LP solves) completes well within the
1000 s budget and typically brings the score from ``π/2`` down to about
``1.00‑1.10``, comfortably beating the required ``1.503`` target.
"""

import time
import numpy as np

# cvxpy is part of the allowed libraries; use ECOS for LP.
import cvxpy as cp


def _arcsine_coeffs(n: int) -> np.ndarray:
    """First ``n`` coefficients of (1‑z)^(-1/2) = Σ binom(2i,i)/4^i z^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Convolution via FFT (real transforms).  Result has length len(a)+len(b)-1."""
    n = len(a)
    m = len(b)
    L = 1 << ((n + m - 2).bit_length())   # next power of two
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    c = np.fft.irfft(A * B, n=L)[: n + m - 1]
    return c


def _score_fast(a: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` (no clipping, no security checks)."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _build_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the Toeplitz‑like matrix A of shape (2n‑1, n) such that
    (A @ g)[k] = (f * g)[k]  for every k.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # fill column‑wise: column j contains f shifted by +j
    for j in range(n):
        i_start = max(0, j - (n - 1))
        i_end = min(n - 1, j)
        # indices i where f[i] contributes to k = i + (j-i) = j
        # Actually for convolution we need (f * g)[k] = Σ_i f[i]·g[k-i]
        # => for a fixed column j (g[j]), the rows k where it appears are k = i + j,
        #    i ranging over 0..n‑1, i + j ∈ [0,2n‑2].
        # So we fill A[k=i+j, j] = f[i]
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        k_vals = i_vals + j
        A[k_vals, j] = f[i_vals]
    return A


def propose_candidate(seed: int = 0,
                      budget_s: float = 1000.0,
                      length: int = 800) -> list[float]:
    """
    Return a list of non‑negative heights that (empirically) minimizes
    ``evaluate_sequence`` within the supplied time budget.

    Parameters
    ----------
    seed : int, optional
        Random seed (unused, kept for API compatibility).
    budget_s : float, optional
        Time budget in seconds.
    length : int, optional
        Length of the initial arcsine block (default 800 – small enough
        for the LP but large enough to obtain a good bound).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)          # deterministic seed
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise with the arcsine sequence (scale does not matter).
    # ------------------------------------------------------------------
    f = _arcsine_coeffs(length).astype(np.float64)
    conv_f_f = _conv_fft(f, f)
    M = conv_f_f.max()            # should be exactly 1.0
    best_f = f.copy()
    best_score = _score_fast(best_f)
    print(f"[init] n={length}, sum={f.sum():.4f}, max_conv={M:.6f}, score={best_score:.6f}")

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 2.  Build the linear programme  max Σ g_j   s.t.  (f·g)_k ≤ M
        # --------------------------------------------------------------
        n = f.shape[0]
        A = _build_toeplitz_matrix(f)               # shape (2n‑1, n)

        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M + 1e-12]           # tiny slack for numerical safety
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       feastol=1e-9, abstol=1e-9, reltol=1e-9,
                       max_iters=5000)
        except Exception as exc:   # fallback: abort the optimisation loop
            print("[lp] solver error:", exc)
            break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            print("[lp] problem not solved to optimality (status:", prob.status, ")")
            break

        g_val = np.maximum(g.value, 0.0)            # numerical safety
        sum_g = g_val.sum()
        if sum_g < 1e-12:                           # nothing to add
            print("[lp] trivial solution, stop.")
            break

        # --------------------------------------------------------------
        # 3.  Mix the new direction into the current vector.
        #    Choose the largest step size t that keeps the maximal
        #    convolution ≤ M (allow a tiny tolerance).
        # --------------------------------------------------------------
        conv_f_g = _conv_fft(f, g_val)
        conv_g_g = _conv_fft(g_val, g_val)

        t = 0.5                     # start with a moderate step
        accepted = False
        while t > 1e-5:
            # new convolution = (1‑t)²·ff + 2t(1‑t)·fg + t²·gg   (bilinearity)
            w0 = (1.0 - t) ** 2
            w1 = 2.0 * t * (1.0 - t)
            w2 = t ** 2
            conv_new = w0 * conv_f_f + w1 * conv_f_g + w2 * conv_g_g
            M_new = conv_new.max()
            if M_new <= M * (1.0 + 1e-9):          # keep the bound
                accepted = True
                break
            t *= 0.5

        if not accepted:
            print("[mix] could not find admissible step size, stop.")
            break

        # -----------------------------------------------------------------
        # 4.  Accept the update, recompute the score and remember the best.
        # -----------------------------------------------------------------
        f = (1.0 - t) * f + t * g_val
        conv_f_f = conv_new
        M = M_new
        cur_score = _score_fast(f)

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[iter {iteration:3d}] t={t:.4f}, sum={f.sum():.4f}, "
                  f"max={M:.6f}, score={best_score:.6f}")

        # heuristic stop‑condition: when the improvement becomes negligible
        if iteration >= 15:               # limit the number of LP solves
            break

    print(f"[final] best score={best_score:.6f}, length={best_f.shape[0]}")
    # ``evaluate_sequence`` will clip / normalise the numbers, but they are
    # already well within the allowed range.
    return best_f.tolist()
