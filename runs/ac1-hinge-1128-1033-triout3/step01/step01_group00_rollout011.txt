def fft_convolution(a, b):
    # compute linear convolution using FFT, returns real array.
    size = len(a) + len(b) - 1
    n = 1 << (size - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, n=n)
    fb = np.fft.rfft(b, n=n)
    conv = np.fft.irfft(fa * fb, n=n)[:size]
    return conv

"""
A gradient‑based optimiser for the self‑convolution peak.

The objective (with the sequence normalised to sum = 1) is
    score = 2·n·max(conv(a,a)) .
Thus minimising the maximal entry of the auto‑convolution is enough.
We use a smoothed “soft‑max’’ of the convolution to obtain a
differentiable surrogate,
compute its gradient via FFT‑based convolutions,
take a projected gradient step on the probability simplex,
and anneal both the smoothing temperature and a simulated‑annealing
acceptance temperature.
Randomised mass‑transfer moves are injected occasionally to escape
shallow minima.

The routine runs for at most ``budget_s`` seconds and returns the best
sequence found (as a plain list of floats).  It starts from the
previous best construction if the global variable ``height_sequence_1``
is present, otherwise from a simple U‑shaped seed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w >= 0, sum(w) = 1 }.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’ with power‑law decay towards the centre:
        a[i] = (min(i, n‑1‑i) + 1)^{-p}
    The vector is normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12           # avoid exact zeroes
    a /= a.sum()
    return a


def next_pow2(x: int) -> int:
    """small utility – smallest power of two ≥ x"""
    return 1 << ((x - 1).bit_length())


def fft_convolution(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of a and b using FFT.
    ``length`` must be a power of two ≥ len(a)+len(b)-1.
    Returns the real convolution (first len(a)+len(b)-1 entries).
    """
    fa = np.fft.rfft(a, n=length)
    fb = np.fft.rfft(b, n=length)
    conv = np.fft.irfft(fa * fb, n=length)
    return conv[: len(a) + len(b) - 1]


def max_self_convolution(a: np.ndarray, fft_len: int) -> float:
    """max_k (a * a)[k] using the same FFT length as the optimiser."""
    # self‑convolution via FFT (real signal)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def softmax(x: np.ndarray, temperature: float) -> np.ndarray:
    """
    Numerically stable softmax.
    When temperature → 0 this becomes a one‑hot at the maximal entry.
    """
    if temperature <= 0:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return w
    # shift by max to avoid overflow
    shifted = (x - np.max(x)) / temperature
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    return w


def grad_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth max‑convolution objective:
        g_i = 2 * sum_j a_j * w_{i+j}
    Computed efficiently via an FFT convolution with the reversed weight.
    """
    n = len(a)
    rev_w = w[::-1]
    conv = fft_convolution(a, rev_w, fft_len)   # length = 3n‑2
    L = len(w)                   # = 2n‑1
    idx = L - 1 - np.arange(n)   # positions L‑1‑i
    grad = 2.0 * conv[idx]
    return grad


def score_from_vector(a: np.ndarray, fft_len: int) -> float:
    """
    Exact score returned by ``evaluate_sequence`` for a vector that is
    already normalised to sum = 1 (no clipping occurs).
    """
    n = len(a)
    max_c = max_self_convolution(a, fft_len)
    return 2.0 * n * max_c

# Main optimiser

        print(f"Initialised from existing height_sequence_1 (len={len(a)})")

        p0 = 0.8

    best_score = score_from_vector(best_a, fft_len)
    print(f"Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------

    random_move_scale = 0.02

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # f) anneal parameters
        # --------------------------------------------------------------
        lr = max(lr * lr_decay, 1e-6)
        smooth_T = max(smooth_T * smooth_decay, smooth_T_min)
        SA_T = max(SA_T * SA_decay, SA_T_min)

        # occasional progress report (once per ~5 s)
        if time.time() - last_log > 5.0:
            print(
                f"Iter {iteration:6d} | best={best_score:.6f} | "
                f"lr={lr:.2e} | Tsmooth={smooth_T:.2e} | SA_T={SA_T:.2e}"
            )
            last_log = time.time()

    print("\n=== optimisation finished ===")
    print(f"Best score achieved: {best_score:.6f} (length {len(best_a)})")
    # Return as Python list – the external evaluator will apply its own clipping.
    return best_a.tolist()
