"""
Propose a step‑function (non‑negative vector) with a low score for
`evaluate_sequence`.  

The algorithm consists of three phases:

1. **Seed generation** – try a variety of simple parametric families
   (U‑shapes, exponential edge‑decay, and edge‑only uniform with a
   zero‑mass centre).  The best seed found is used as the starting point.
2. **Projected sub‑gradient descent** – repeatedly minimise the peak of
   the self‑convolution `max(conv(a,a))`.  The sub‑gradient w.r.t. the
   current maximal convolution entry `k*` is
   `g_i = 2*a_{k*-i}` (zero outside the support).  We move
   `a ← a - η g` with a step size `η` that decays like `1/√t`,
   clip negatives to zero and renormalise to stay on the probability
   simplex.  This is the exact projection required by `evaluate_sequence`
   because the function is scale‑invariant and depends only on the
   normalised vector.
3. **Escape moves** – occasional random mass‑transfer moves and a tiny
   simulated‑annealing acceptance rule help to avoid shallow local
   minima.

The whole routine respects the supplied time budget (default 1000 s)
and prints intermediate progress.  The returned list sums to 1, so the
external `evaluate_sequence` will see the same objective value as the
internal `score_from_vector` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    s = v.sum()
    if s == 0.0:
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric U‑shape: a[i] ∝ (min(i, n-1-i) + 1)^{-p}.
    The vector is normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)           # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12           # avoid exact zeros
    return simplex_normalise(a)


def exponential_distribution(n: int, r: float) -> np.ndarray:
    """
    Symmetric exponential decay from the edges:
    a[i] ∝ r**i + r**(n-1-i),  0 < r < 1.
    Returns a normalised vector.
    """
    idx = np.arange(n)
    a = r ** idx + r ** (n - 1 - idx)
    a = np.maximum(a, 1e-12)
    return simplex_normalise(a)


def zero_center_distribution(n: int, w_frac: float) -> np.ndarray:
    """
    Uniform mass on the two edge blocks, zero on a central window.
    w_frac ∈ [0, 1] is the fraction of indices forced to zero.
    """
    w = int(round(w_frac * n))
    if w <= 0:
        # pure uniform
        return np.full(n, 1.0 / n)
    start = (n - w) // 2
    mask = np.ones(n, dtype=bool)
    mask[start:start + w] = False
    a = np.zeros(n)
    a[mask] = 1.0
    return simplex_normalise(a)


def convolution_peak(a: np.ndarray) -> float:
    """Return max_k (a * a)[k] for a non‑negative vector a with sum == 1."""
    conv = np.convolve(a, a)
    return float(conv.max())


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact score returned by `evaluate_sequence` for a normalised vector
    (sum == 1).  This is 2 * n * max(conv(a,a)).
    """
    n = len(a)
    return 2.0 * n * convolution_peak(a)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence`.  Runs for at most ``budget_s`` seconds
    and returns the best sequence found (as a list of Python floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # spare a few seconds for cleanup

    # ------------------------------------------------------------------
    # 1.  Generate deterministic seeds from a few parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # lengths to try – all comfortably below 10 000 so that many
    # convolutions fit into the budget
    lengths = [5600, 6400, 7200, 8000]
    # parameter grids
    exp_rs = [0.90, 0.92, 0.94, 0.96, 0.98]
    u_powers = [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]
    centre_fracs = [0.0, 0.10, 0.20, 0.30, 0.40]

    print("=== Exploring parametric seeds ===")
    for n in lengths:
        # exponential edge‑decay
        for r in exp_rs:
            if time.time() > deadline:
                break
            a = exponential_distribution(n, r)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"exp seed   n={n:4d} r={r:.2f}  score={sc:.6f}")

        # pure U‑shapes
        for p in u_powers:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"U‑shape    n={n:4d} p={p:4.1f}  score={sc:.6f}")

        # edge‑only uniform with a zero centre
        for w in centre_fracs:
            if time.time() > deadline:
                break
            a = zero_center_distribution(n, w)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"zero‑centre n={n:4d} w={w:.2f}  score={sc:.6f}")

        # combine U‑shape with a zero centre (mask the centre)
        for p in [1.0, 2.0, 3.0]:
            for w in centre_fracs:
                if time.time() > deadline:
                    break
                base = u_shaped_distribution(n, p)
                mask = np.ones(n, dtype=bool)
                w_int = int(round(w * n))
                if w_int > 0:
                    start = (n - w_int) // 2
                    mask[start:start + w_int] = False
                a = base * mask
                a = simplex_normalise(a)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"U+mask    n={n:4d} p={p:.1f} w={w:.2f}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Use the provided `height_sequence_1` if it exists and is good
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            external = simplex_normalise(external)
            sc_ext = score_from_vector(external)
            if sc_ext < best_score:
                best_score = sc_ext
                best_seq = external.copy()
                print(f"external   n={external.size}  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback random seed (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 6000
        a = np.random.rand(n)
        best_seq = simplex_normalise(a)
        best_score = score_from_vector(best_seq)
        print(f"fallback   n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Iterative projected sub‑gradient descent with occasional random moves
    # ------------------------------------------------------------------
    current = best_seq.copy()
    current_score = best_score
    n = len(current)

    print("\n=== Starting projected sub‑gradient descent ===")
    it = 0
    # initial step size (will decay like 1/sqrt(t))
    init_step = 0.5
    # temperature for simulated‑annealing acceptance of uphill moves
    temperature = 0.2
    temp_decay = 0.9995

    while time.time() < deadline:
        it += 1

        # Convolution and maximal index
        conv = np.convolve(current, current)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # Build sub‑gradient (sum contributions if several entries tie)
        # We treat all indices within 1e-12 of the max as active.
        tol = 1e-12
        active = np.where(conv >= max_val - tol)[0]
        grad = np.zeros(n)
        for k in active:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * current[idx[mask]]

        # Diminishing step size
        step = init_step / np.sqrt(it)

        # Candidate by projected gradient step
        cand = current - step * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_normalise(cand)

        cand_score = score_from_vector(cand)

        # Acceptance rule (mostly greedy, occasional uphill)
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Iter {it:6d}: new best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Random mass‑transfer escape move (probability ≈5%)
        # ------------------------------------------------------------------
        if np.random.rand() < 0.05:
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] > 0:
                delta = current[i] * np.random.rand() * 0.3  # move up to 30 % of source mass
                esc = current.copy()
                esc[i] -= delta
                esc[j] += delta
                esc = simplex_normalise(esc)
                esc_score = score_from_vector(esc)
                # accept if improves or with a small random chance
                if esc_score < current_score or np.random.rand() < 0.1:
                    current = esc
                    current_score = esc_score
                    if esc_score < best_score:
                        best_score = esc_score
                        best_seq = esc.copy()
                        print(f"Iter {it:6d} (escape): new best = {best_score:.6f}")

        # Temperature cooling
        temperature = max(temperature * temp_decay, 1e-6)

    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # Return as plain Python list – external evaluator will clip/normalise again
    return best_seq.tolist()
