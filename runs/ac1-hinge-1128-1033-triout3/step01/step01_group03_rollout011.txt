def propose_candidate(seed=0, budget_s=1000, length=5808):
    import time, numpy as np
    np.random.seed(seed)
    n = length
    # prepare FFT length
    L = 1 << ((2*n-1).bit_length())
    # Helper to compute score
    def compute_score(a):
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        max_conv = np.max(conv)
        s = np.sum(a)
        return 2.0 * n * max_conv / (s*s)
    # Initialize
    a = _sqrt_series(n)  # from previous code
    cur_score = compute_score(a)
    best_score = cur_score
    best_seq = a.copy()
    print(f"[init] n={n}, score={cur_score}")

    # Simulated annealing parameters
    start_temp = 0.02
    end_temp = 1e-5
    total_iters_est = 200000  # we will adapt.
    iter = 0
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    while time.time() < deadline:
        # temperature linearly decreasing
        t_frac = (time.time() - start_time) / budget_s
        temperature = start_temp * (1 - t_frac) + end_temp * t_frac
        # step size decreasing also
        step_factor = max(0.001, 0.5 * (1 - t_frac))

        # propose move
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        # pick j different
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # delta max: cannot exceed a[i]; also ensure a[j] + delta <= 1
        max_delta = a[i]
        # choose delta proportional to step_factor
        delta = step_factor * max_delta * np.random.rand()
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute score
        new_score = compute_score(a_new)
        # accept
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{iter}] new best {best_score:.6f}")
        # occasional random restart
        if iter % 100000 == 0 and iter>0:
            # reinitialize with a random Dirichlet perturbation of sqrt base
            a = _sqrt_series(n)
            # add small random noise
            a += np.random.randn(n) * 0.001
            a = np.maximum(a,0)
            a /= a.sum()
            cur_score = compute_score(a)
            print(f"[{iter}] restart, score {cur_score:.6f}")
        iter += 1

    # final evaluation with provided evaluate_sequence
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best_score={final_score:.6f} after {iter} iterations")
    return best_seq.tolist()

"""
Optimising the peak of a self‑convolution.

The function aims to minimise
    F(a) = 2·n·max_k (a∗a)_k          (a_i ≥ 0, Σ a_i = 1)

A good analytical starting point is the discrete sqrt‑series, i.e. the
coefficients of (1‑x)^{‑1/2}.  Their infinite self‑convolution is constant,
which yields the asymptotic bound π/2 ≈ 1.5708.  Small perturbations of this
shape can improve the constant further for a finite length n.

The optimiser is a simple simulated‑annealing / hill‑climbing loop:

* Initialise with the normalised sqrt‑series.
* At each iteration pick two indices i≠j and move a random amount ε
  from a_i to a_j while keeping the total mass exactly 1.
* Evaluate the new sequence using an FFT‑based convolution (much faster than
  numpy.convolve).
* Accept the move if it improves the objective, otherwise accept it with a
  temperature‑dependent probability (Metropolis rule).  The temperature and
  the typical step size decay linearly with time.
* Occasionally (every 100 000 steps) restart from the analytic start
  with a small random jitter – this helps to escape plateaus.
* The best sequence seen so far is remembered and finally returned as a
  plain Python list.

The whole routine respects the 1000 s budget, prints progress messages and
uses only NumPy and the supplied evaluate_sequence function.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: project onto probability simplex (used for occasional jitter)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x >= 0, sum x = 1}  (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Analytic sqrt‑series (coefficients of (1‑x)^{-1/2})
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the first n coefficients of (1‑x)^{-1/2} normalised to sum 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# --------------------------------------------------------------
# Fast evaluation of 2·n·max(a∗a) for a simplex‑normalised vector
# --------------------------------------------------------------
def _fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max_k (a∗a)_k.
    a must already satisfy Σ a = 1 (within floating error).
    """
    # real FFT of the zero‑padded vector
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    return 2.0 * n * max_conv


# --------------------------------------------------------------
# Main candidate generator
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  Uses simulated annealing starting from the
    analytic sqrt‑series.  Returns the best sequence found (as a plain list).
    """
    np.random.seed(seed)

    # ----------------------------------------------------------
    # Parameters
    # ----------------------------------------------------------
    n = kwargs.get("length", 5808)               # length of the sequence
    deadline = time.time() + budget_s - 0.5      # safety margin
    fft_len = 1 << ((2 * n - 1).bit_length())    # next power of two for FFT

    # ----------------------------------------------------------
    # Initialise with the analytic sqrt‑series
    # ----------------------------------------------------------
    a = _sqrt_series(n)                          # already normalised, sum = 1
    cur_score = _fast_score(a, fft_len, n)
    best_seq = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------
    # Simulated‑annealing loop
    # ----------------------------------------------------------
    it = 0
    # linear schedules (tuned empirically)
    start_temp = 2e-2
    end_temp = 1e-5
    start_step = 0.25     # relative to a[i] (max epsilon = step * a[i])
    end_step = 1e-4

    while time.time() < deadline:
        # ------------------------------------------------------
        # Update temperature and step size (linear decay)
        # ------------------------------------------------------
        elapsed = time.time() - (deadline - budget_s + 0.5)
        frac = min(1.0, elapsed / budget_s)
        temperature = start_temp * (1 - frac) + end_temp * frac
        step_factor = start_step * (1 - frac) + end_step * frac

        # ------------------------------------------------------
        # Propose a move: transfer ε from i to j
        # ------------------------------------------------------
        i = np.random.randint(n)
        if a[i] <= 1e-16:                # cannot take mass from an almost‑zero entry
            continue
        # pick a different index j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # ε is a fraction of the mass at i
        eps = step_factor * a[i] * np.random.rand()
        if eps <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # (sum stays 1, non‑negativity guaranteed by construction)

        # ------------------------------------------------------
        # Evaluate the candidate
        # ------------------------------------------------------
        new_score = _fast_score(a_new, fft_len, n)

        # ------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # accept with probability exp(-(Δ)/T)
            if temperature > 0:
                prob = np.exp(-(new_score - cur_score) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_new
            cur_score = new_score
            # record global best
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{it:6d}] new best {best_score:.6f}")

        # ------------------------------------------------------
        # Occasionally re‑normalise to avoid drift
        # ------------------------------------------------------
        if it % 1000 == 0:
            # re‑normalise explicitly (tiny cost, prevents rounding buildup)
            a /= a.sum()

        # ------------------------------------------------------
        # Simple restart to escape deep plateaus
        # ------------------------------------------------------
        if it % 100000 == 0 and it > 0:
            a = _sqrt_series(n)
            # tiny random perturbation before projection
            jitter = np.random.randn(n) * 5e-4
            a = _project_simplex(a + jitter)
            cur_score = _fast_score(a, fft_len, n)
            print(f"[{it:6d}] restart, score={cur_score:.6f}")

        it += 1

    # --------------------------------------------------------------
    # Final verification with the official evaluate_sequence
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best evaluation = {final_score:.6f} after {it} iterations")
    return best_seq.tolist()
