def proj_simplex(v):
    # Implementation as per known algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
proposal: projected exponentiated‑gradient descent on a smooth Lp‑approximation
of the max‑convolution objective, followed by a fast random hill‑climbing
phase.  The algorithm works directly on the simplex (non‑negative entries,
sum=1) so the scaling invariance of `evaluate_sequence` is respected.
The Lp loss (p gradually increased) concentrates the updates on the
largest entries of the self‑convolution; an exponential update keeps the
variables non‑negative and a simple renormalisation projects back onto the
simplex.  After the gradient phase we perform many tiny mass‑transfer moves
to polish the solution.  The whole search respects the 1000 s budget and
prints progress.
"""

import time
import math
import numpy as np

def _normalize(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum=1)."""
    s = v.sum()
    if s <= 0:
        # fallback to uniform distribution if everything collapsed
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimizes
    `evaluate_sequence`.  The optimisation consists of two stages:

    1. **Exponentiated‑gradient descent** on the smoothed loss
       `L_p(a) = sum_k ( (a*a)_k / max(a*a) )^{p}`.
       The exponent `p` starts at 4 and is increased up to 64 during the run.
       The update rule is `a_i ← a_i * exp(-η * ∂L/∂a_i)`,
       followed by normalisation onto the simplex.

    2. **Random local search** (mass transfer) that repeatedly moves a small
       amount of weight from a randomly chosen index to another one.
       Every move is accepted only if it strictly improves the true score
       `2·n·max(a*a)`.

    The best sequence encountered is returned.
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # 1. Choose a starting length and initialise with the Catalan‑type
    #    coefficients (the classic 1/√k shape).  If a previous solution
    #    exists we reuse its length.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # reuse previous length if present
        a0 = np.array(height_sequence_1, dtype=float)
    except NameError:
        n = 3000                            # a comfortable default size
        a0 = np.empty(n, dtype=float)
        a0[0] = 1.0
        for k in range(1, n):
            a0[k] = a0[k - 1] * (2 * k - 1) / (2 * k)

    # Normalise to a probability vector (sum = 1)
    a = _normalize(a0)

    # --------------------------------------------------------------
    # 2. Helper to compute the objective quickly (no need to call the
    #    heavy `evaluate_sequence` each iteration).
    # --------------------------------------------------------------
    def current_score(v: np.ndarray) -> float:
        """2·n·max((v*v) convolution) – exact score when sum(v)=1."""
        b = np.convolve(v, v)
        return 2.0 * n * b.max()

    best_seq = a.copy()
    best_score = current_score(a)
    print(f"[init] n={n}, initial_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Gradient phase – exponential update on a smooth Lp‑approximation.
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0       # keep a little safety margin
    grad_phase_end = start_time + budget_s * 0.6

    base_eta = 0.20          # initial learning‑rate factor
    p_start = 4
    p_max = 64
    p_step = 2000            # increase p every `p_step` iterations

    it = 0
    while time.time() < grad_phase_end:
        # convolution and current max
        b = np.convolve(a, a)
        max_b = b.max()
        if max_b == 0.0:
            break  # degeneracy – shouldn't happen

        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[improve] iter {it:06d} → score={best_score:.6f}")

        # slowly raise the power p to focus more on the peak
        p = p_start + it // p_step
        if p > p_max:
            p = p_max

        # normalised convolution values to avoid under‑/overflow
        norm_b = b / max_b
        w = norm_b ** (p - 1)                     # shape 2·n‑1, non‑negative

        # gradient of the smoothed loss:
        #   ∂L/∂a = 2·p· (a ★ w)   where ★ is correlation.
        # Correlation can be obtained via convolution with reversed w.
        grad_full = np.convolve(a, w[::-1])       # length 3·n‑2
        grad = 2.0 * p * grad_full[2 * n - 2:3 * n - 2]   # keep exactly n entries

        # learning‑rate schedule (decays like 1/√t)
        eta = base_eta / math.sqrt(it + 1.0)

        # multiplicative (exponential) update – guarantees positivity
        a = a * np.exp(-eta * grad)

        # project back onto the simplex (simply renormalise)
        a = _normalize(a)

        it += 1

    # --------------------------------------------------------------
    # 4. Local‑search phase – tiny mass transfers that directly improve
    #    the true objective.
    # --------------------------------------------------------------
    local_start = time.time()
    a_local = best_seq.copy()   # start from the best point found so far
    # a simple schedule for the maximal transfer fraction
    max_transfer = 0.1          # up to 10 % of the source weight per move

    while time.time() < deadline:
        # randomly pick distinct source and destination indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        src_val = a_local[i]
        if src_val <= 0.0:
            continue

        # pick a transfer amount proportional to the source weight
        delta = src_val * max_transfer * np.random.rand()
        if delta <= 0.0:
            continue

        cand = a_local.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate the true score (fast because sum=1)
        cand_score = current_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a_local = cand.copy()
            print(f"[local] iter {it:06d} → score={best_score:.6f}")

        # slowly shrink the allowed transfer magnitude to fine‑tune
        if it % 5000 == 0 and max_transfer > 0.001:
            max_transfer *= 0.9
        it += 1

    # --------------------------------------------------------------
    # 5. Final report – compute the official score via `evaluate_sequence`
    #    (which also applies the security clamping) and return the best
    #    sequence as a plain Python list.
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best_score (approx)={best_score:.6f}, "
          f"evaluate_sequence={final_score:.6f}")
    return best_seq.tolist()
