def convolution_max(a):
    n = a.shape[0]
    size = 1 << (2*n-1).bit_length()  # next power of two >= 2n-1
    # compute FFT

def project_simplex(v):
    # v is 1D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def conv_max(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = A * A
    b = np.fft.irfft(B, n=size)[:2*n-1]
    return b

def _project_simplex(v):
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    # Implements the algorithm of Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """Return the linear convolution a * a using an FFT (real‑valued)."""
    n = a.shape[0]
    m = 2 * n - 1
    # choose a power‑of‑two length for speed
    sz = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, n=sz)
    B = A * A
    b = np.fft.irfft(B, n=sz)[:m]
    return b

def _objective(a):
    """Objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _conv_full(a)
    return 2.0 * n * b.max()

    """Generate a list of simple analytic candidates (normalized)."""
    cand = []

    # 1. Catalan‑type coefficients (the previous best construction)
    c = np.empty(n, dtype=float)
    c[0] = 1.0
    for k in range(1, n):
        c[k] = c[k - 1] * (2.0 * k - 1) / (2.0 * k)
    cand.append(c / c.sum())

    # 2. Uniform
    u = np.ones(n, dtype=float) / n
    cand.append(u)

    # 3. Symmetric power‑law a_i = (i+1)^{-p} + (n-i)^{-p}
    i = np.arange(n)

        cand.append(a / a.sum())

    rng = np.random.default_rng(0)
    for _ in range(5):
        a = rng.random(n)
        a = a / a.sum()
        cand.append(a)

    return cand

def _gradient_step(a, step):
    """Take one projected sub‑gradient step on max(conv(a,a))."""
    n = a.shape[0]
    b = _conv_full(a)
    k = int(np.argmax(b))

    # compute sub‑gradient of max entry
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)

    g = np.zeros_like(a)
    # entries where i != j
    diff = mask & (idx != j)
    g[diff] = 2.0 * a[j[diff]]
    # entries where i == j (the diagonal term)
    eq = mask & (idx == j)
    g[eq] = a[idx[eq]]

    # take a gradient step and project back onto the simplex
    v = a - step * g
    w = _project_simplex(v)
    return w, b.max()
