"""
propose_candidate
=================

Search for an extreme‑low value of the objective

    L(seq) = 2·n·max(conv(seq,seq)) / (∑ seq)²

by exploiting the known structure of near‑optimal step functions:
a short “core” that decays like 1/√i and a long low‑height tail.
For a core of length `L` the optimal tail height is

    δ = (∑ core_i²) / (2·∑ core_i)

and the optimal tail length (derived from the constraint that the
core–tail cross term does not exceed the core’s own maximal convolution)
is

    T = round(4·(∑ core_i)² / ∑ core_i²).

A grid search over core lengths (5…200) and a few tail‑length
multipliers quickly finds the best combination.  After the grid we
perform a tiny simulated‑annealing refinement that moves a few units
of mass inside the tail – this can shave a few percent off the score.
The whole procedure respects the 1000 s budget and always returns a
valid non‑negative list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast max of the linear convolution via FFT (power‑of‑two size)
# ----------------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Exact objective (same as evaluate_sequence but without the safety checks)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    n = seq.shape[0]
    total = float(seq.sum())
    if total < 0.01:                     # matches the guard inside evaluate_sequence
        return np.inf
    max_conv = _conv_max(seq)
    return 2.0 * n * max_conv / (total * total)


# ----------------------------------------------------------------------
# Core generator – 1/√i decay (the known asymptotically optimal shape)
# ----------------------------------------------------------------------
def _core_seq(L: int, p: float = 0.5) -> np.ndarray:
    """Return a core of length L with entries (i+1)^-p."""
    i = np.arange(1, L + 1, dtype=np.float64)
    return i ** (-p)


# ----------------------------------------------------------------------
# Build a candidate from a core by attaching the analytically optimal tail
# ----------------------------------------------------------------------
def _build_candidate(core: np.ndarray, tail_factor: float = 4.0) -> np.ndarray:
    """
    Given a core, attach a constant‑height tail.
    The tail height is  δ = (∑ core²) / (2·∑ core)
    and the tail length is   T ≈ tail_factor·(∑ core)² / ∑ core² .
    """
    S_c = core.sum()               # ℓ¹‑norm of the core
    Q_c = np.dot(core, core)       # ℓ²‑norm squared of the core
    if Q_c == 0.0:
        return core.copy()
    delta = Q_c / (2.0 * S_c)      # optimal tail height (see analysis)
    # tail length – round to the nearest integer, keep at least one element
    T = max(1, int(round(tail_factor * S_c * S_c / Q_c)))
    tail = np.full(T, delta, dtype=np.float64)
    return np.concatenate([core, tail])


# ----------------------------------------------------------------------
# Main entry point – called by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation value.
    A lightweight grid over core length, exponent and a tail‑length factor
    is performed, followed by a few refinement steps.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # spare a bit for cleanup

    # ------------------------------------------------------------------
    # 1) Grid search
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # core lengths to try – small cores are actually best for this problem
    core_lengths = np.arange(5, 201, 5)           # 5,10,15,…,200
    # decay exponents – values around 0.5 are known to be good
    exponents = [0.5, 0.45, 0.4, 0.35, 0.3]
    # factor in the tail‑length formula (the analytic optimum is 4.0)
    tail_factors = [3.5, 4.0, 4.5]

    for L in core_lengths:
        if time.time() > deadline:
            break
        for p in exponents:
            core = _core_seq(L, p)
            # skip degenerate cores (all zeros)
            if core.max() == 0.0:
                continue
            for factor in tail_factors:
                cand = _build_candidate(core, factor)
                val = _score(cand)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    print(
                        f"[grid] new best: {best_val:.6f}  L={L} p={p:.2f} factor={factor}"
                    )
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # 2) Simple refinement – move a little mass inside the tail
    # ------------------------------------------------------------------
    if best_seq is not None:
        cur_seq = best_seq.copy()
        cur_val = best_val
        # locate the start index of the tail (first element that is not part of the core)
        # we assume the core is decreasing; the tail is the trailing block of (almost) equal values.
        # A safe heuristic: the tail starts where the difference between consecutive elements
        # becomes smaller than 1e-12 (practically constant).
        diffs = np.diff(cur_seq)
        tail_start = np.argmax(diffs < 1e-12)  # first index where change is tiny
        tail_start = max(tail_start, 0)

        start_time = time.time()
        # tiny simulated‑annealing / hill‑climbing on two tail positions
        while time.time() < deadline:
            # pick two distinct indices inside the tail (if the tail is too short, fall back to any index)
            if cur_seq.shape[0] - tail_start < 2:
                i, j = np.random.randint(cur_seq.shape[0], size=2)
            else:
                i = tail_start + np.random.randint(cur_seq.shape[0] - tail_start)
                j = tail_start + np.random.randint(cur_seq.shape[0] - tail_start)
                if i == j:
                    continue
            # move a tiny fraction (≤2%) from i → j
            amount = cur_seq[i] * np.random.rand() * 0.02
            if amount <= 0.0:
                continue
            new_seq = cur_seq.copy()
            new_seq[i] -= amount
            new_seq[j] += amount
            # protect against negative values (should not happen)
            if new_seq[i] < 0.0:
                continue
            new_val = _score(new_seq)

            # accept improvements or occasional worse moves (simulated annealing)
            if new_val < cur_val or np.random.rand() < 0.001:
                cur_seq, cur_val = new_seq, new_val
                if new_val < best_val:
                    best_val, best_seq = new_val, new_seq.copy()
                    print(f"[refine] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback – if nothing succeeded, return a simple uniform sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _score(best_seq)

    print(f"final best value: {best_val:.6f}")
    # Return a plain Python list as required by the evaluation harness.
    return best_seq.tolist()
