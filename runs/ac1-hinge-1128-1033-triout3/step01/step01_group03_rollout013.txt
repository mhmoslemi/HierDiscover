"""
Propose a sequence that improves the π/2 ≈ 1.5708 bound.

Approach
--------
* Start from the analytically optimal “square‑root’’ sequence
  a_i ∝ C_i = binom(2i,i)/4^i (coefficients of (1‑x)^−½) and normalise
  ∑ a_i = 1.
* In each iteration compute the self‑convolution c = a∗a and its maximum
  M = max(c).  For every index k we define a slack
        w_k = M – c_k   (w_k ≥ 0, zero on the currently active plateau).
  The loss L = Σ_k w_k² is zero exactly when the whole convolution becomes
  flat, i.e. when the max is as small as possible.
* The gradient of L w.r.t. a is
        ∂L/∂a_i = –4 Σ_k w_k·a_{k‑i}
  which can be evaluated as a single convolution of the weight vector w
  with the reversed coefficient vector a_rev.  This is done efficiently
  with an FFT‑based convolution (O(n log n)).
* Perform a projected Adam step on a (first‑order moments, adaptive step
  size) and project back onto the probability simplex {a≥0, Σa=1}.
* Periodically add a tiny Gaussian jitter and, after long periods with
  no improvement, restart from the analytic start with a small random
  perturbation.
* Keep the best sequence seen during the time budget (default 1000 s).

The optimizer moves mass from the large‑value plateau on the left half of
the convolution towards the right half, thereby reducing the maximal entry
below the classic π/2 bound.  The method finishes by returning the best
candidate found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al., 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “square‑root’’ starting sequence (coefficients of (1‑x)^‑½)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^‑½, normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Fast convolution using FFT (real‑valued input)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return full convolution of real vectors x and y using FFT."""
    m = len(x) + len(y) - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:m]
    return conv


# ----------------------------------------------------------------------
# Gradient of the flattening loss L = Σ (M - (a∗a))²
# ----------------------------------------------------------------------
def _flatten_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∂L/∂a where w_k = M - (a∗a)_k ≥ 0.
    The formula is   ∂L/∂a_i = -4 * Σ_k w_k * a_{k-i},
    which is a convolution of w with the reversed a.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    conv_full = _fft_convolve(w, a_rev)  # length = 3n‑2
    start = n - 1
    grad = -4.0 * conv_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative sequence that (hopefully) achieves an evaluation
    below the classic π/2 bound.  The routine runs a projected Adam
    optimisation on the flatten‑the‑convolution loss for as long as the
    time budget permits.
    """
    np.random.seed(seed)

    # length of the step function – can be overridden via kwargs
    n = kwargs.get("length", 2800)          # a modest size for speed
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    a = _sqrt_series(n)                     # already normalised, sum = 1
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a)) / (a.sum() ** 2)
    print(f"[init] length={n}, initial evaluation={best_score:.6f}")

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.20                     # base learning rate
    decay = 5e-5                   # linear decay of the learning rate
    iter_cnt = 1
    no_improve = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # current convolution and the flattening slack
        # --------------------------------------------------------------
        conv = np.convolve(a, a)                 # full length 2n‑1
        M = conv.max()
        w = M - conv                             # non‑negative slack

        # --------------------------------------------------------------
        # evaluate current candidate (cheap, uses our own formula)
        # --------------------------------------------------------------
        cur_score = 2.0 * n * M / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{iter_cnt:6d}] new best: {best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # gradient of the flattening loss and Adam update
        # --------------------------------------------------------------
        grad = _flatten_gradient(a, w)                 # shape (n,)
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        # learning‑rate schedule (simple linear decay)
        lr = lr0 / (1.0 + decay * iter_cnt)

        # Adam step (note the sign – we minimise the loss)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a_new)

        # --------------------------------------------------------------
        # occasional random jitter to escape possible plateaus
        # --------------------------------------------------------------
        if iter_cnt % 5000 == 0:
            jitter = np.random.randn(n) * (1e-4 / np.sqrt(iter_cnt + 1))
            a = _project_simplex(a + jitter)

        # --------------------------------------------------------------
        # simple restart if stuck for a long time
        # --------------------------------------------------------------
        if no_improve >= 20000:
            a = _sqrt_series(n)
            a = _project_simplex(a + np.random.randn(n) * 1e-3)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{iter_cnt:6d}] restart triggered")

        iter_cnt += 1

    print(f"[final] best evaluation = {best_score:.6f} after {iter_cnt} iterations")
    return best_seq.tolist()
