"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in two stages:

1. **Analytic start** – the optimal “constant‑convolution’’ sequence
   b_k = C(2k,k) / 4**k (central binomial / 4^k) is generated and the best
   prefix length (≈3000) is taken as the initial candidate.  For this
   sequence the evaluation score equals π/2 ≈ 1.5708.

2. **Stochastic optimisation** – a simulated‑annealing / hill‑climbing loop
   that repeatedly moves a small amount of mass from a “large‑impact’’ index
   (typically in the tail, i.e. far from the start) to a “low‑impact’’ index
   (near the head).  The move preserves the total ℓ₁‑norm, so the objective
   improvement is equivalent to reducing the maximal convolution value.
   The perturbation size follows a cooling schedule, and occasionally a
   linear‑programming step (if SciPy is available) is performed to obtain a
   global direction that improves the ℓ₁‑norm under a relaxed convolution
   bound.  The best sequence seen during the allotted time (≈1000 s) is
   returned.

The whole routine respects the 1000 s budget, uses only NumPy (and SciPy
when present) and never produces negative or >1000 values – the
`evaluate_sequence` function will clamp them anyway.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: central‑binomial coefficients (the “constant‑convolution’’ base)
# ----------------------------------------------------------------------
def generate_binomial_prefix(max_n: int) -> np.ndarray:
    """Return the first ``max_n`` values of b_k = C(2k,k) / 4**k."""
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0                # b_0
    coeffs[0] = cur
    for k in range(1, max_n):
        # recurrence: b_{k} = b_{k-1} * (2k-1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs

# ----------------------------------------------------------------------
# Helper: quick evaluation of the objective (identical to evaluate_sequence
# except it skips the security clamping – all values we generate are
# already in the allowed range).
# ----------------------------------------------------------------------
def objective(seq: np.ndarray) -> float:
    """Return 2*n*max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.size
    conv = np.convolve(seq, seq)
    M = conv.max()
    S = seq.sum()
    if S < 0.01:                     # safeguard – matches the evaluator
        return np.inf
    return 2.0 * n * M / (S * S)

# ----------------------------------------------------------------------
# Optional LP step – maximise the ℓ₁‑norm of a new sequence `g`
# under a relaxed convolution bound (only used when SciPy is present).
# ----------------------------------------------------------------------
def lp_step(f: np.ndarray, bound_factor: float = 1.1):
    """Solve  max Σ g_i   s.t.  f * g ≤ bound_factor * max(f*f),  g≥0.

    Returns the optimal g (or None if the LP fails).  The LP matrix is
    dense but small enough (n ≤ 2000) to be solved quickly with the
    HiGHS backend.
    """
    try:
        from scipy.optimize import linprog
    except Exception:                     # SciPy missing – skip the step
        return None

    n = f.size
    m = 2 * n - 1
    bound = np.max(np.convolve(f, f)) * bound_factor

    # The convolution matrix A satisfies   (A @ g)[i] = (f*g)[i]
    # Row i, column k receives f[i-k] whenever 0 ≤ i-k < n.
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, np.arange(n)] = f[j]          # vectorised Toeplitz fill

    # Linear programme: minimise -Σ g_i  (i.e. maximise Σ g_i)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, bound, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    # Numerical noise may introduce tiny negatives – clip.
    g = np.maximum(g, 0.0)
    return g

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Return a list of non‑negative heights with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – analytical start (constant‑convolution coefficients)
    # ------------------------------------------------------------------
    # The analytical bound approaches π/2 ≈ 1.5708 as the length grows;
    # a length of ~2500 already gives a very tight starting point.
    N_START = 2500
    base = generate_binomial_prefix(N_START)

    seq = base.copy()                               # current state (numpy array)
    cur_score = objective(seq)
    best_seq = seq.copy()
    best_score = cur_score

    print(f"[init] length={N_START}, analytic_score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – stochastic optimisation
    # ------------------------------------------------------------------
    n = seq.size
    # temperature‑like parameters for the Metropolis acceptance rule
    temp = 0.02 * cur_score          # roughly 2 % of the score
    temp_decay = 0.9995
    # perturbation magnitude (relative standard deviation)
    sigma = 0.15                      # start with ~15 % multiplicative noise
    sigma_min = 0.005
    sigma_decay = 0.9997

    # how often we invoke the optional LP step (probability per iteration)
    lp_prob = 0.02
    lp_bound_factor = 1.08           # relax the convolution bound a little

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Occasionally try a global LP move (if SciPy is available)
        # --------------------------------------------------------------
        if np.random.rand() < lp_prob:
            g = lp_step(seq, bound_factor=lp_bound_factor)
            if g is not None:
                # Scale g so that its own max‑convolution becomes 1
                M_g = np.max(np.convolve(g, g))
                if M_g > 0:
                    g = g / np.sqrt(M_g)
                cand_score = objective(g)
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = g.copy()
                    seq = g.copy()
                    cur_score = cand_score
                    print(f"[LP‑improve #{iteration}] score={best_score:.6f}")
                    # tighten the bound a little after a successful LP step
                    lp_bound_factor = max(1.01, lp_bound_factor * 0.97)
                else:
                    # even if not better, keep the candidate as the new state
                    # with a small Metropolis probability
                    if np.random.rand() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
                        seq = g.copy()
                        cur_score = cand_score
            continue

        # --------------------------------------------------------------
        # Regular – move a small amount of mass from a “high‑impact’’
        # index (usually in the second half) to a “low‑impact’’ one
        # (usually in the first quarter).  The move keeps the total sum
        # unchanged, therefore a reduction of the maximal convolution
        # directly improves the objective.
        # --------------------------------------------------------------
        # choose a donor from the right‑hand side (large index)
        donor = np.random.randint(int(0.55 * n), n)
        if seq[donor] <= 0.0:
            continue

        # choose a receiver from the left‑hand side (small index)
        receiver = np.random.randint(0, int(0.45 * n))

        # amount to transfer – a fraction of the donor's current height
        delta = np.random.rand() * seq[donor] * sigma
        # ensure we do not exceed the hard clamp (1000) – not an issue here
        if seq[receiver] + delta > 1000.0:
            delta = 1000.0 - seq[receiver]

        cand = seq.copy()
        cand[donor] -= delta
        cand[receiver] += delta

        # keep non‑negativity (numerical safety)
        cand[donor] = max(0.0, cand[donor])
        cand[receiver] = max(0.0, cand[receiver])

        cand_score = objective(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            accept = np.random.rand() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12))

        if accept:
            seq = cand
            cur_score = cand_score
            # record global best
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[improve #{iteration}] score={best_score:.6f}")

        # --------------------------------------------------------------
        # Annealing schedule
        # --------------------------------------------------------------
        temp = max(1e-6, temp * temp_decay)
        sigma = max(sigma_min, sigma * sigma_decay)

    # ------------------------------------------------------------------
    # Final sanity – ensure non‑negative and ≤1000 (evaluate_sequence will
    # clamp anyway, but we keep the array tidy)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Return a plain Python list as required by the evaluator
    print(f"[final] best_score={best_score:.6f}, length={best_seq.size}")
    return best_seq.tolist()
