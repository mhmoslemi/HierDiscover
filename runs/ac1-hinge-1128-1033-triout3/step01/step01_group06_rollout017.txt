def project_simplex(v):
    """Project vector v onto probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def double_move(a, conv, i, j):
    # randomly choose two different destinations
    n = len(a)
    dsts = np.random.choice(np.delete(np.arange(n), [i, j]), size=2, replace=False)
    dst1, dst2 = dsts[0], dsts[1]
    # choose deltas: up to 50% mass each
    max_delta_i = a[i] * 0.5
    max_delta_j = a[j] * 0.5
    if max_delta_i < 1e-12 or max_delta_j < 1e-12:
        return a, conv  # nothing
    delta_i = np.random.rand() * max_delta_i
    delta_j = np.random.rand() * max_delta_j
    a_new, conv_new = apply_move(a, conv, i, dst1, delta_i)
    a_new, conv_new = apply_move(a_new, conv_new, j, dst2, delta_j)
    return a_new, conv_new

import cvxpy as cp

def improve_with_lp(f, M, t=0.2):
    n = len(f)
    b = cp.Variable(n)
    constraints = [b >= 0]
    # compute convolution f * b via linear expression: for each k, sum_i f[i] * b[k-i]
    # using Toeplitz matrix: conv_matrix shape (2n-1, n)
    # We'll build using scipy.linalg.toeplitz? Could build manually using np.zeros.
    # But for speed we can compute as: constraints for k: (f_rev * b) <= M, where conv linear = f_rev convolution with b (or cross-correlation). Actually conv = np.convolve(f,b) is linear in b: (conv_k) = sum_i f[i] * b[k-i] (circular w/out wrap due to zero padding). For each k we can compute indices i where 0<=i< n and 0 <= k-i < n => i ∈ [max(0, k-(n-1)), min(n-1, k)].
    # We'll loop to add constraints.
    for k in range(2*n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        if i_min <= i_max:
            coeffs = np.zeros(n)
            idx_range = np.arange(i_min, i_max+1)
            coeffs[idx_range] = f[idx_range]
            # But we need b_{k-i}, which corresponds to index j = k - i.
            # Actually b[j] where j = k - i, i range yields j range.
            # So coefficient for b[j] = f[i] where i = k-j.
            # So we can fill coefficients accordingly:
            # For i in idx_range: j = k - i
            # Set coeffs[j] = f[i].
            conv_expr = 0
            for i in idx_range:
                j = k - i
                conv_expr += f[i] * b[j]
            constraints.append(conv_expr <= M)
    # objective maximize sum(b)
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    b_opt = b.value
    if b_opt is None: return f # if fails
    sum_b = np.sum(b_opt)
    if sum_b == 0: return f
    b_scaled = b_opt * (1.0 / sum_b)  # already sum=1
    # convex combination
    f_new = (1 - t) * f + t * b_scaled
    # project onto simplex (should sum=1)
    f_new = f_new / np.sum(f_new)
    return f_new

"""
Propose a step‑function sequence with a low evaluation value.

The algorithm works in three stages:

1. **Initialisation** – several analytic shapes (edge‑heavy,
   reciprocal‑sqrt, arcsine) and the previously found best sequence are
   normalised to sum = 1.  The one with the smallest `evaluate_sequence`
   score is kept as the start point.

2. **Projected gradient descent** – we minimise the current maximal
   self‑convolution entry `max(conv)` while staying on the probability
   simplex.  For a given peak index `k` the gradient of `conv[k]` w.r.t.
   the coefficients is `∂conv[k]/∂a_i = 2·a_{k‑i}` (when the partner index
   exists).  A diminishing step size and a Euclidean projection onto the
   simplex are used.  This quickly flattens the convolution and usually
   reduces the score well below the previous 1.57379.

3. **Refinement by simulated annealing** – a fast O(n) update of the
   convolution (the `apply_move` routine) is employed.  Both single
   targeted moves and *double* moves (removing mass from the two indices
   that contribute most to the current peak) are tried.  An annealing
   schedule accepts occasional uphill moves to escape plateaus.

The total running time is limited to the provided budget (≈ 1000 s).  The
best sequence found is returned as a plain list of floats; the final
score is printed using the supplied `evaluate_sequence` function.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: project an arbitrary vector onto the probability simplex
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear convolution via FFT (exact for zero‑padded real vectors)
# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray, L: int = None) -> np.ndarray:
    """Return the linear convolution a*a using an FFT of length L."""
    n = a.shape[0]
    if L is None:
        L = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    # eliminate tiny negative artefacts
    conv_full = np.where(conv_full < 0, 0.0, conv_full)
    return conv_full


# ----------------------------------------------------------------------
# Helper: fast convolution update when moving δ from index i to j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].
    a – 1‑D array of length n, sum(a) = 1
    conv – convolution a*a (length 2n‑1)
    i, j – distinct indices, 0 ≤ i,j < n
    delta – 0 < delta ≤ a[i]
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta
    dj = +delta

    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    conv_new = conv.copy()

    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]
    a_q = a[q]

    conv_new[i + q] += 2.0 * di * a_q
    conv_new[j + q] += 2.0 * dj * a_q

    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Helper: produce a targeted move focusing on the current maximum entry
# ----------------------------------------------------------------------
def targeted_move(a: np.ndarray, conv: np.ndarray):
    """
    Choose a source index that contributes most to the current maximal
    convolution entry and a random destination different from the source.
    Returns a tuple (src, dst).
    """
    n = a.shape[0]
    k = int(np.argmax(conv))

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_candidates = np.arange(i_min, i_max + 1, dtype=int)
    j_candidates = k - i_candidates

    contrib = a[i_candidates] * a[j_candidates]
    src = int(i_candidates[np.argmax(contrib)])

    # random destination ≠ src
    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1
    return src, dst
