def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, scipy.sparse as sp, scipy.optimize as opt, time, math

    np.random.seed(seed)

    # Choose length n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000
    # Ensure n not too large e.g., cap at 3000
    n = max(500, min(n, 3000))

    # Build Catalan coefficients
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    # Normalize sum to sqrt(2n)
    target_sum = math.sqrt(2.0 * n)
    a *= target_sum / a.sum()

    # Compute initial objective
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # parameters
    t = 0.3   # mixing weight
    iter = 0
    deadline = time.time() + budget_s - 10.0
    while time.time() < deadline:
        # Compute current M
        conv = np.convolve(best_seq, best_seq)
        M = conv.max()
        # Build sparse Toeplitz matrix for convolution constraints
        # T matrix as described
        T = sp.diags(best_seq, offsets=range(n), shape=(2*n-1, n), format='csc')
        # Linear program: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n)
        b_ub = np.full(2*n - 1, M)
        # Solve LP
        res = opt.linprog(c, A_ub=T, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve':True, 'time_limit': max(0, deadline - time.time()) })
        if not res.success:
            print("[lp] failed to solve, breaking")
            break
        g = res.x
        # scale g to sum target
        sum_g = g.sum()
        if sum_g <= 0:
            print("[lp] zero sum g, break")
            break
        scale = target_sum / sum_g
        g_scaled = g * scale
        # mix to get new candidate
        new_seq = (1 - t) * best_seq + t * g_scaled
        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq
            print(f"[iter {iter}] improved score={best_score:.6f} M={M:.6f} t={t:.3f}")
        else:
            # if not improved perhaps reduce t adaptively
            t *= 0.9
        # maybe adjust t gradually

    final_score = evaluate_sequence(best_seq.tolist())

"""
propose_candidate
-----------------
A two‑phase optimizer for the convolution‑maximisation problem.

Phase 1 – deterministic improvement:
    * Initialise the sequence with the Catalan (central‑binomial) coefficients,
      rescaled so that Σaᵢ = √(2 n).  For this sequence the evaluation score equals
      the maximal self‑convolution value M ≈ π/2 ≈ 1.57.
    * Repeatedly solve a linear programme
          maximise   Σ gᵢ
          subject to  (a * g)[k] ≤ M    for all k = 0 … 2 n‑2
                     gᵢ ≥ 0
      (the convolution matrix of a is built as a sparse Toeplitz matrix).
      The LP finds a feasible sequence g that carries as much total mass as
      possible without increasing the maximal convolution value M.
    * Rescale g to the required normalisation Σgᵢ = √(2 n) and replace the
      current sequence a by a convex combination (1‑t) a + t g (t≈0.5).
      After each replacement the new maximal convolution of a is recomputed;
      only updates that strictly improve the score are kept, otherwise the
      mixing weight t is reduced.

Phase 2 – stochastic fine‑tuning:
    * With the remaining time a tiny random walk (perturb one coordinate,
      renormalise, keep the change if it improves the score) is performed.
      This cleans up any remaining irregularities.

All numbers are kept non‑negative and are automatically clipped by
`evaluate_sequence`.  The function respects the 1000 s budget and prints
intermediate scores for monitoring.
"""

import time
import math
import numpy as np

# SciPy is optional – if unavailable we fall back to pure random search.
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:          # pragma: no cover
    _SCIPY_OK = False


def _build_conv_matrix(a: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2 n‑1) × n convolution matrix T such that (T @ g) = a * g.
    T is a sparse Toeplitz matrix with constant diagonals equal to the
    entries of a.  The matrix format required by ``linprog`` is CSC.
    """
    n = a.shape[0]
    # each diagonal i (offset i) consists of the constant a[i]
    offsets = np.arange(n, dtype=int)
    data = np.tile(a, (n, 1)).astype(np.float64)   # shape (n, n)
    # `sp.diags` expects (data, offsets); data[i] is the whole diagonal i.
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return T


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The algorithm follows the two‑phase strategy
    described in the module docstring.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Choose a modest but flexible length.
    #    If a previous construction exists we reuse its length, but
    #    cap it to keep the LP tractable.
    # ------------------------------------------------------------------
    try:
        base_len = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:  # height_sequence_1 is not defined
        base_len = None

    # target length – we restrict it to at most 3000 entries to keep the LP
    # size (≈ n² non‑zeros) manageable within the time budget.
    n = 2000
    if base_len is not None and base_len > 0:
        n = min(base_len, 3000)

    # ------------------------------------------------------------------
    # 2. Initialise with the Catalan (central‑binomial) coefficients.
    # ------------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)

    # Normalise to the usual sum Σa = √(2 n); the evaluation then equals M.
    target_sum = math.sqrt(2.0 * n)
    a *= target_sum / a.sum()
    best_seq = a.copy()
    best_M = np.convolve(best_seq, best_seq).max()
    best_score = best_M                     # because Σa = √(2 n)

    print(f"[init] n={n}, initial M={best_M:.6f}")

    # ------------------------------------------------------------------
    # 3. Deterministic LP‑based refinement (if SciPy is available).
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 8.0    # leave a small safety margin
    t = 0.5                                   # initial mixing weight
    iter_cnt = 0
    max_iter = 30                             # safeguard against endless loops

    while _SCIPY_OK and time.time() < deadline and iter_cnt < max_iter:
        # recompute the current maximal convolution value M
        conv = np.convolve(best_seq, best_seq)
        M = conv.max()

        # build the (a * g) ≤ M constraints as a sparse matrix
        T = _build_conv_matrix(best_seq)

        # LP: maximise Σ gᵢ  → minimise −Σ gᵢ
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve the LP (HiGHS is the default high‑performance method)
        try:
            res = linprog(c, A_ub=T, b_ub=b_ub,
                          bounds=bounds, method='highs')
        except Exception as exc:               # pragma: no cover
            print(f"[lp] exception: {exc}")
            break

        if not res.success:
            print(f"[lp] failed ({res.message}); terminating LP phase.")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            print("[lp] got zero‑mass solution; aborting LP phase.")
            break

        # Rescale g to the required normalisation Σg = √(2 n)
        g *= target_sum / sum_g

        # Convex combination with the current best sequence
        new_seq = (1.0 - t) * best_seq + t * g

        # Evaluate the new maximal convolution (no need to call evaluate_sequence)
        new_M = np.convolve(new_seq, new_seq).max()
        new_score = new_M

        if new_score < best_score - 1e-12:         # a genuine improvement
            best_seq = new_seq
            best_score = new_score
            best_M = new_M
            print(f"[iter {iter_cnt:02d}] improved M={best_score:.6f} (t={t:.3f})")
            # keep the mixing weight, maybe even increase it a little
            t = min(0.9, t * 1.05)
        else:
            # no improvement – become more conservative
            t = max(0.05, t * 0.7)

        iter_cnt += 1

        # time‑check – break if we are running out of budget
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 4. Stochastic local search for the remaining time.
    # ------------------------------------------------------------------
    # Scale for perturbations – a few percent of the typical coefficient.
    step_scale = 0.02 * best_seq.mean()
    while time.time() < deadline:
        idx = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * step_scale
        cand = best_seq.copy()
        cand[idx] = max(0.0, cand[idx] + delta)

        # Renormalise to keep Σ = √(2 n) (the objective is homogeneous).
        s = cand.sum()
        if s <= 0.0:
            continue
        cand *= target_sum / s

        cand_M = np.convolve(cand, cand).max()
        cand_score = cand_M
        if cand_score < best_score - 1e-12:
            best_seq = cand
            best_score = cand_score
            best_M = cand_M
            print(f"[local] new best M={best_score:.6f}")

        # slowly shrink the step size
        step_scale *= 0.9999

    # ------------------------------------------------------------------
    # 5. Final verification via the provided evaluator (just for safety).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:                     # pragma: no cover
        final_score = float("inf")
    print(f"[final] best evaluation = {final_score:.6f}")

    # Return a plain Python list as required.
    return best_seq.tolist()
