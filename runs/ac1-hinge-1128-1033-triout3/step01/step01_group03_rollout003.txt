def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    np.random.seed(seed)
    n = kwargs.get('length', 2500)
    a = _sqrt_series(n)  # sum=1

    conv = np.convolve(a, a)
    cur_val = 2.0 * n * conv.max()

    best_seq = a.copy()
    best_val = cur_val

    start = time.time()
    deadline = start + budget_s - 0.3
    T0 = 0.005
    T_min = 1e-7
    max_frac0 = 0.4
    min_frac = 0.001

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 0.3)
        temperature = T0 * (1 - frac) + T_min
        max_frac = max_frac0 * (1 - frac) + min_frac

        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        while j == i:
            j = np.random.randint(0, n)

        ai = a[i]
        if ai <= 0:
            continue
        delta = ai * max_frac * np.random.rand()
        if delta <= 0:
            continue

        # compute new conv
        conv_new = conv.copy()
        # update slices
        conv_new[i:i+n] -= 2.0 * delta * a
        conv_new[j:j+n] += 2.0 * delta * a
        # delta^2 contributions
        idx2i = 2*i
        idx2j = 2*j
        idxij = i + j
        conv_new[idx2i] += delta*delta
        conv_new[idx2j] += delta*delta
        conv_new[idxij] -= 2.0 * delta*delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # acceptance
        diff = new_val - cur_val
        if diff < 0 or np.exp(-diff / temperature) > np.random.rand():
            # accept
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                # optional print
                print(f"[{int(elapsed)}s] new best {best_val:.6f}")
    # final best.
    print(f"[final] best value {best_val:.6f} after {int(time.time() - start)} s")
    return best_seq.tolist()

"""
Simulated‑Annealing optimiser for the step‑function bound.

The objective for a non‑negative vector `a` with `sum(a)=1` is

    F(a) = 2·n·max_k (a*a)_k   (where “*” is the discrete convolution)

because `evaluate_sequence` returns exactly this value (the sum is 1 and
the clamping never triggers).  The algorithm keeps a single candidate
vector on the probability simplex and repeatedly performs a tiny
mass–transfer move between two coordinates.  The resulting change of the
self‑convolution can be updated in O(n) time with a simple rank‑1
formula, so many thousands of moves can be tried per second.

A classic simulated‑annealing acceptance rule (`ΔF<0` or probabilistic
acceptance based on a temperature that cools linearly) allows the
search to escape the local minimum found by pure projected gradient
descent.  Several restarts (initialising either from the analytic
“square‑root” series or from a random Dirichlet distribution) are
performed when the search stalls.

The routine stops before the supplied time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: probability simplex projection (not needed for the SA moves,
# but kept for possible restarts)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “square‑root” start: coefficients of (1-x)^(-1/2)
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1-x)^(-1/2), normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Random Dirichlet start (uniformly random point on the simplex)
# ----------------------------------------------------------------------
def _random_simplex(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex using exponential variables."""
    x = rng.exponential(scale=1.0, size=n)
    x /= x.sum()
    return x


# ----------------------------------------------------------------------
# Objective value for a normalised vector (sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, conv: np.ndarray) -> float:
    """2·n·max(conv)   (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Simulated‑annealing move: transfer a small amount of mass from i → j
# and update the convolution incrementally.
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray,
                i: int, j: int, delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (a_new, conv_new) after moving `delta` mass from index i to j.
    The update uses the exact rank‑1 formula:
        conv_new = conv + 2·delta·(shift_j - shift_i) + delta²·Δ,
    where Δ has +1 at 2i, +1 at 2j and –2 at i+j.
    """
    n = a.shape[0]
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    conv_new = conv.copy()

    # linear part: 2·delta·(shift_j - shift_i)
    # shift_i affects conv[i : i+n]  (adds -2·delta·a)
    conv_new[i:i + n] -= 2.0 * delta * a
    conv_new[j:j + n] += 2.0 * delta * a

    # quadratic part: delta² contributions
    conv_new[2 * i] += delta * delta
    conv_new[2 * j] += delta * delta
    conv_new[i + j] -= 2.0 * delta * delta

    return a_new, conv_new


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves on
    the previous best bound (~1.57086).  The search uses simulated
    annealing with occasional restarts and runs for at most `budget_s`
    seconds.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = kwargs.get("length", 3000)          # feel free to override via kwargs
    deadline = time.time() + budget_s - 0.3   # safety margin for the harness

    # ------------------------------------------------------------------
    # initialise – first try the analytic start
    # ------------------------------------------------------------------
    a = _sqrt_series(n)          # sum = 1
    conv = np.convolve(a, a)     # length = 2n-1
    cur_obj = _objective(a, conv)

    best_seq = a.copy()
    best_obj = cur_obj

    print(f"[init] n={n}, start objective = {cur_obj:.6f}")

    # ------------------------------------------------------------------
    # annealing schedule parameters
    # ------------------------------------------------------------------
    total_time = budget_s
    start_temp = 4e-3            # temperature (in objective units)
    end_temp = 1e-7
    max_frac_start = 0.3         # maximal fraction of a[i] that may be moved
    max_frac_end = 1e-4

    # counters for stagnation detection
    iters = 0
    no_improve = 0
    RESTART_LIMIT = 8000

    while time.time() < deadline:
        # ----------------------------------------------------------------
        # temperature and step‑size schedules (linear cooling)
        # ----------------------------------------------------------------
        elapsed = time.time() - (deadline - budget_s)
        progress = min(1.0, elapsed / total_time)
        temperature = start_temp * (1.0 - progress) + end_temp
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end

        # ----------------------------------------------------------------
        # pick two distinct indices
        # ----------------------------------------------------------------
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1          # ensure j != i

        ai = a[i]
        if ai <= 0.0:
            continue        # nothing to transfer

        # ----------------------------------------------------------------
        # propose a transfer delta (always from i to j)
        # ----------------------------------------------------------------
        delta = ai * max_frac * rng.random()
        if delta <= 0.0:
            continue

        # ----------------------------------------------------------------
        # compute the updated convolution and objective
        # ----------------------------------------------------------------
        a_new, conv_new = _apply_move(a, conv, i, j, delta)
        new_obj = _objective(a_new, conv_new)

        # ----------------------------------------------------------------
        # Metropolis acceptance
        # ----------------------------------------------------------------
        diff = new_obj - cur_obj
        if diff < 0.0 or np.exp(-diff / temperature) > rng.random():
            # accept the move
            a = a_new
            conv = conv_new
            cur_obj = new_obj

            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = a.copy()
                no_improve = 0
                print(f"[{iters:6d}] new best = {best_obj:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ----------------------------------------------------------------
        # occasional random jitter to shake the system
        # ----------------------------------------------------------------
        if iters % 2500 == 0 and iters > 0:
            jitter = rng.normal(scale=1e-4, size=n)
            a = _project_simplex(a + jitter)
            conv = np.convolve(a, a)
            cur_obj = _objective(a, conv)

        # ----------------------------------------------------------------
        # restart if we have been stuck for a while
        # ----------------------------------------------------------------
        if no_improve >= RESTART_LIMIT:
            # 50 % chance to go back to analytic start, otherwise random
            if rng.random() < 0.5:
                a = _sqrt_series(n)
            else:
                a = _random_simplex(n, rng)
            conv = np.convolve(a, a)
            cur_obj = _objective(a, conv)
            no_improve = 0
            print(f"[{iters:6d}] restart (restart counter)")

        iters += 1

    # ------------------------------------------------------------------
    # final reporting
    # ------------------------------------------------------------------
    print(f"[final] best objective = {best_obj:.6f} after {iters} iterations")
    # sanity check with the supplied evaluator (optional but cheap)
    try:
        # the evaluator will clamp and re‑normalise; we already satisfy the constraints
        eval_check = evaluate_sequence(best_seq.tolist())
        if not np.isfinite(eval_check):
            print("[warning] final sequence failed internal validation")
        else:
            print(f"[validation] evaluator reports {eval_check:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[warning] evaluation failed:", exc)

    # return a plain Python list as required by the harness
    return best_seq.tolist()
