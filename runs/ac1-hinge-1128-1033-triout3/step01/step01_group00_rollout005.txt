"""
Propose a low‑score step‑function using a soft‑max‑smoothed
gradient descent on the peak of the self‑convolution, combined
with simple simulated‑annealing random mass–transfer moves.
The algorithm works on a probability simplex (sum == 1) so that
the denominator of `evaluate_sequence` is constant; the scored
objective is therefore `2 * n * max(conv(a,a))`.

Key ideas
---------

* Initialise with a symmetric “U‑shape’’ (a_i ∝ (min(i, n-1-i)+1)^‑p).
* Repeatedly improve the distribution:
    – 70 % of the time a soft‑max approximated gradient of the
      max‑convolution is taken (the temperature adapts to the
      current peak, focusing the descent on the actual maximal
      entry).
    – 30 % of the time a random mass‑transfer move (simulated‑annealing)
      is performed to escape shallow minima.
* After every move the vector is projected back onto the simplex
  (clip negatives and renormalise).  If the new candidate improves
  the score it is kept; otherwise it is accepted with a small
  probability that decays with a temperature schedule.
* The whole search runs for the given time budget (≈1000 s).  The
  best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape: a[i] ∝ (min(i,n‑1‑i)+1)^{‑p}, normalised to sum = 1."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12           # avoid exact zeros
    a /= a.sum()
    return a


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a vector with sum ≈ 1."""
    n = a.shape[0]
    # convolution (full) – np.convolve uses an efficient algorithm internally
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def simplex_renorm(v: np.ndarray) -> np.ndarray:
    """
    Very cheap “project onto the simplex’’: clip negatives then
    renormalise to sum = 1 (if the sum is >0).  For our use this
    approximation works well and is far faster than the exact O(n log n)
    algorithm.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v /= s
    else:
        # fallback to a uniform distribution
        v = np.full_like(v, 1.0 / v.size)
    return v


def softmax_weights(conv: np.ndarray, temp_factor: float) -> np.ndarray:
    """
    Compute a soft‑max distribution over the convolution entries.
    The temperature is chosen as `temp_factor * max(conv)`.
    """
    max_c = conv.max()
    # avoid division by zero – if max_c is 0 use a tiny temperature
    T = max(temp_factor * max_c, 1e-12)
    # stabilize exponentials
    exps = np.exp((conv - max_c) / T)
    return exps / exps.sum()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function runs for at most ``budget_s``
    seconds and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a base length and an initial U‑shape.
    # ------------------------------------------------------------------
    # Lengths tried.  Larger lengths give a lower theoretical bound,
    # but we keep them modest to stay inside the time budget.
    lengths = [8000, 12000, 16000]    # increasing resolution
    powers = [0.4, 0.5, 0.6, 0.8, 1.0]   # a few exponents to seed the search

    best_seq = None
    best_score = np.inf
    base_n = None

    print("=== Selecting initial seed ===")
    for n in lengths:
        for p in powers:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                base_n = n
                print(f"  seed: n={n:5d} p={p:4.2f}  score={sc:.6f}")
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2.  If a handcrafted sequence exists, try to use it as a second seed.
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            ext = np.maximum(external, 0.0)
            if ext.sum() > 0:
                ext /= ext.sum()
                sc_ext = score_from_vector(ext)
                if sc_ext < best_score:
                    best_score = sc_ext
                    best_seq = ext.copy()
                    base_n = ext.size
                    print(f"  using provided height_sequence_1  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – random simplex if nothing else succeeded.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 10000
        a = rng.random(n)
        a /= a.sum()
        best_seq = a.copy()
        best_score = score_from_vector(a)
        base_n = n
        print(f"  fallback random start  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Main optimisation loop (soft‑max gradient + random moves).
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score

    # Annealing / schedule parameters
    accept_temp = 1.0               # acceptance temperature for simulated annealing
    accept_cooling = 0.9995

    grad_temp_factor = 0.08         # multiplier for soft‑max temperature (relative to max conv)
    grad_step = 0.6                 # step size for gradient descent
    step_cooling = 0.9997

    prob_grad = 0.70                # probability to perform a gradient step

    it = 0
    last_print = time.time()

    while time.time() < deadline:
        it += 1

        n = cur.shape[0]

        # --------------------------------------------------------------
        # Decide move type
        # --------------------------------------------------------------
        if rng.random() < prob_grad:
            # ---- soft‑max gradient step ----
            conv = np.convolve(cur, cur)           # length 2n‑1
            w = softmax_weights(conv, grad_temp_factor)

            # gradient = 2 * (w convolved with reversed cur), sliced to length n
            # (w * cur_rev)[n‑1 : n‑1+n] gives Σ_k w_k * cur_{k‑j}
            g_full = np.convolve(w, cur[::-1])
            grad = 2.0 * g_full[(n - 1):(n - 1 + n)]

            cand = cur - grad_step * grad
            cand = np.maximum(cand, 0.0)
            cand = simplex_renorm(cand)
        else:
            # ---- random mass transfer (simulated annealing style) ----
            i, j = rng.choice(n, size=2, replace=False)
            if cur[i] <= 0.0:
                continue
            delta = cur[i] * rng.random() * 0.5   # move up to 50 % of source mass
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
            # renormalise (sum is already 1, but clipping may be needed)
            cand = np.maximum(cand, 0.0)
            cand = simplex_renorm(cand)

        # ---- evaluate candidate (fast internal evaluation) ----
        cand_score = score_from_vector(cand)

        # ---- acceptance criterion (simulated annealing) ----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = math.exp((cur_score - cand_score) / max(accept_temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            # keep global best
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Iter {it:7d}: new global best = {best_score:.6f}")

        # ---- schedule decay ----
        accept_temp *= accept_cooling
        grad_step *= step_cooling

        # occasional progress report (every ~2 s)
        if time.time() - last_print > 2.0:
            elapsed = time.time() - (deadline - budget_s + 5.0)
            print(f"Iter {it:7d} | cur={cur_score:.6f} | best={best_score:.6f} | n={n} | elapsed={elapsed:.1f}s")
            last_print = time.time()

    # ------------------------------------------------------------------
    # 5.  Return best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score: {best_score:.6f} (length {best_seq.shape[0]})")
    return best_seq.tolist()
