    n = len(globals().get("height_sequence_1", [])) or 8000

    best_val = np.inf

    tail_logs = np.linspace(-8, -0.5, 50)  # 50 values
    for L in core_lengths:
        core = 1.0 / np.sqrt(np.arange(1, L+1, dtype=np.float64))

            if val < best_val:

                best_seq = seq.copy()
                print(f"[grid] new best: {best_val:.6f} (L={L}, delta={delta:.2e})")

def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the linear convolution `seq * seq` using FFT."""
    n = seq.shape[0]
    # size = next power of two for the (2n‑1)-point convolution
    size = 1 << ((2 * n - 1).bit_length())
    # forward FFT, pointwise multiplication, inverse FFT
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Exact objective value for a candidate sequence."""
    total = float(seq.sum())
    if total < 0.01:                     # guard against division by zero
        return np.inf
    max_conv = _conv_max(seq)
    n = seq.shape[0]
    return 2.0 * n * max_conv / (total * total)


def _core_array(length: int, beta: float = 0.5, alpha: float = 0.0) -> np.ndarray:
    """
    Build the decreasing core:
        a_i = 1 / (i + 1 + α)^β   for i = 0 … length‑1
    """
    idx = np.arange(1, length + 1, dtype=np.float64) + alpha
    return 1.0 / (idx ** beta)


def _assemble_sequence(
    n: int, core_len: int, beta: float, alpha: float, delta: float, order: str = "core_tail"
) -> np.ndarray:
    """
    Create a full length‑`n` sequence from parameters.
    `order` can be:
        "core_tail" – core first, then the constant tail,
        "tail_core" – tail first, then the core.

        return core.copy()  # fallback – shouldn't happen in the search

    if order == "core_tail":
        seq = np.concatenate([core, tail])
    else:
        seq = np.concatenate([tail, core])
    return seq
