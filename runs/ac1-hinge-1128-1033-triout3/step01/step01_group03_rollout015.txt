def propose_candidate(seed=0, budget_s=950, length=None):
    import time, numpy as np, math

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose lengths list based on time budget
    lengths = [2000, 4000, 8000] if length is None else [length]
    best_seq = None
    best_score = float('inf')
    for n in lengths:
        # allocate part of time maybe evenly
        # compute remaining time
        remaining = deadline - time.time()
        if remaining < 5:
            break
        # allocate portion of remaining time for this length, say proportionally to 1/n (or equal)
        # For simplicity just run as much as possible for each length sequentially

        a = _sqrt_series(n)
        # add small random initial noise to avoid symmetry
        a += np.random.randn(n) * 1e-6
        a = _project_simplex(a)

        # compute initial best for this n
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[init n={n}] score {cur_score:.6f}")

        # Precompute FFT size
        size_fft = 1 << (2*n-1).bit_length()
        # SA parameters
        T0 = max(0.1, cur_score - 1.0) # maybe 0.5
        # Actually we set high T for exploration
        T0 = 0.2
        # cooling schedule: T = T0 * (1 - fraction) where fraction = (elapsed/alloc_time)
        # Determine allocated time for this n: maybe 0.8*(deadline-start)/len(lengths) etc.

        # For simplicity compute per length time budget:
        # each length get equal share of remaining total.
        alloc_time = (deadline - time.time()) / (len(lengths) - lengths.index(n))
        # but careful: we compute at start

        loop_start = time.time()
        iter_cnt = 0
        no_improve = 0

        while time.time() - loop_start < alloc_time and time.time() < deadline:
            # compute fast conv and max
            fa = np.fft.rfft(a, size_fft)
            conv = np.fft.irfft(fa * fa, size_fft)[:2*n-1]
            max_val = conv.max()
            # fast_score = 2*n*max_val
            fast_score = 2 * n * max_val

            # compute active set
            active = np.flatnonzero(conv >= max_val - 1e-12)
            grad = _max_conv_gradient(a, active)

            # temperature
            elapsed = time.time() - loop_start
            fraction = elapsed / alloc_time
            T = T0 * (1.0 - fraction)

            # propose move: random pair
            i = np.random.randint(n)
            # ensure a[i] > 0 for moving mass out
            if a[i] < 1e-12:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            # step size: scale by T and relative magnitude
            # delta max is a[i] * min(0.5, T) maybe
            delta_max = a[i] * min(0.5, T)  # at most move half of source mass
            delta = np.random.rand() * delta_max
            # create candidate
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # ensure nonnegativity (already)
            # compute fast score
            fa_new = np.fft.rfft(a_new, size_fft)
            conv_new = np.fft.irfft(fa_new * fa_new, size_fft)[:2*n-1]
            max_val_new = conv_new.max()
            fast_score_new = 2 * n * max_val_new

            # acceptance
            if fast_score_new <= fast_score or np.random.rand() < math.exp(-(fast_score_new - fast_score) / max(T, 1e-12)):
                a = a_new
                fast_score = fast_score_new
                # evaluate with evaluate_sequence occasional
                if fast_score_new < best_score - 1e-8:
                    # compute true evaluation
                    true_score = evaluate_sequence(a.tolist())
                    if true_score < best_score:
                        best_score = true_score
                        best_seq = a.copy()
                        print(f"[{n} iter {iter_cnt}] new best {best_score:.6f}")
                        no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            # occasional jitter / projection to simplex (reduce numeric drift)
            if iter_cnt % 1000 == 0:
                a = _project_simplex(a)

            # restart if stuck
            if no_improve >= 20000:
                a = _sqrt_series(n)
                a += np.random.randn(n) * 1e-3
                a = _project_simplex(a)
                no_improve = 0
                print(f"[{n} iter {iter_cnt}] restart")
            iter_cnt += 1

    # final best sequence
    print("[final] best_eval", best_score)
    return best_seq.tolist()

"""
Improved search for a step‑function that minimises the provided
`evaluate_sequence` metric.

The algorithm works in two phases:

1. **Exponentiated‑gradient refinement** – start from the analytically
   motivated “square‑root” construction, repeatedly take a sub‑gradient of
   the active convolution constraints, move in the opposite direction using
   an exponentiated‑gradient update and project back onto the probability
   simplex.  This quickly drives the evaluation down from the trivial bound
   2.0 to the theoretical limit ~π/2.

2. **Simulated‑annealing fine‑tuning** – after the gradient phase the
   iterate is close to a local optimum but still has room for improvement.
   A temperature‑controlled Metropolis walk proposes tiny mass transfers
   between two random coordinates.  The move is accepted if it lowers the
   fast FFT‑based score, otherwise it may be accepted with a probability that
   depends on the temperature.  Periodic restarts prevent long stagnation.
   The best sequence observed over the whole budget is returned.

Both phases use a fast FFT‐based evaluation (O(n log n)) and only call the
expensive `evaluate_sequence` when a new global best is found, leaving ample
time for the optimiser to explore.  The implementation respects the 1000 s
time limit, stays on a single CPU and contains no I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex using the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “square‑root” start (coefficients of (1‑x)^(-1/2))
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the normalised first n coefficients of (1‑x)^(-1/2)."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k at the active indices `max_idx`.
    For every active k we add 2*a_{k-i} to the i‑th component whenever
    0 ≤ i < n and 0 ≤ k-i < n.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    if len(max_idx) > 1:
        grad /= len(max_idx)
    return grad


# ----------------------------------------------------------------------
# Fast FFT based evaluation: 2·n·max(conv)  (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, size_fft: int) -> float:
    """Return 2·n·max(a∗a) using an FFT of length `size_fft`."""
    n = a.shape[0]
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that minimises the
    supplied `evaluate_sequence` function.

    The routine combines exponentiated‑gradient descent and a short

    """
    np.random.seed(seed)

            true_val = evaluate_sequence(candidate.tolist())
            if true_val < best_score:
                best_score = true_val

    # ------------------------------------------------------------------
    for n in lengths:

                print(f"[restart] n={n} after {iter_sa} SA steps")

            iter_sa += 1
            if time.time() >= phase_end:

    # Return the best sequence discovered (as plain Python list)
    # ------------------------------------------------------------------

        best_seq = np.full(lengths[0], 1.0 / lengths[0])
    print(f"[final] overall best evaluation = {best_score:.6f}")
