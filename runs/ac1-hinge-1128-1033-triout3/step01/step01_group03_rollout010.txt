    # a length n, p length 2n-1
    N = next_pow2(3*n)
    a_fft = np.fft.rfft(a, N)
    p_rev_fft = np.fft.rfft(p[::-1], N)
    conv_rev = np.fft.irfft(a_fft * p_rev_fft, N)
    offset = len(p) - 1
    grad = 2 * conv_rev[offset - np.arange(n)]
    return grad

def propose_candidate(seed=0, budget_s=1000, length=5808, **kwargs):
    np.random.seed(seed)
    n = length
    deadline = time.time() + budget_s - 0.5

    # Precompute for FFT size
    N = 1 << ((3*n-2).bit_length())
    # pre allocate maybe.

    # Helper evaluate fast:
    def fast_eval(a):
        # compute conv via FFT
        a_fft = np.fft.rfft(a, N)
        conv = np.fft.irfft(a_fft * a_fft, N)[:2*n-1]
        max_val = conv.max()
        return max_val, conv

    # Starting vectors list
    starts = []

    # analytic sqrt series
    starts.append(_sqrt_series(n))
    # power beta = 0.3
    for beta in [0.3, 0.5, 0.7, 1.0]:
        w = 1.0/np.power(np.arange(1, n+1, dtype=np.float64), beta)
        w /= w.sum()
        starts.append(w)

    # linear decreasing: w_i = (n-i) / sum
    lin = np.arange(n, 0, -1, dtype=np.float64)
    lin /= lin.sum()
    starts.append(lin)

    # random Dirichlet: generate 3 random draws
    for _ in range(3):
        w = np.random.dirichlet(np.ones(n))
        starts.append(w)

    best_seq = None
    best_eval = float('inf')
    best_iter = 0

    # schedule parameters
    step0 = 0.5
    min_step = 1e-5
    tol = 1e-12

    # For each start
    for s_idx, a0 in enumerate(starts):
        a = a0.copy()
        a = a / a.sum()
        # optionally jitter
        a += np.random.randn(n) * 1e-4
        a = _project_simplex(a)

        # track local best
        max_val, conv = fast_eval(a)
        cur_eval = 2 * n * max_val
        if cur_eval < best_eval:
            best_eval = cur_eval
            best_seq = a.copy()
            print(f"[init {s_idx}] new global best {best_eval:.6f}")

        no_improve = 0
        iter_local = 0
        while time.time() < deadline:
            # compute conv and max
            max_val, conv = fast_eval(a)
            # Determine active set
            active = np.flatnonzero(conv >= max_val - tol)

            # compute gradient via active set
            grad = _max_conv_gradient(a, active)

            # line search
            step = step0
            while step > min_step:
                a_candidate = _project_simplex(a - step * grad)
                max_val_new, _ = fast_eval(a_candidate)
                if max_val_new < max_val - 1e-15:  # improvement
                    a = a_candidate
                    max_val = max_val_new
                    break
                step *= 0.5
            else:
                # no improvement step; maybe random jitter or restart
                no_improve += 1
                if no_improve >= 500:
                    # restart from analytic
                    a = _sqrt_series(n)
                    a = _project_simplex(a + np.random.randn(n) * 1e-3)
                    no_improve = 0
                # also break if no step found
                continue

            cur_eval = 2 * n * max_val
            if cur_eval < best_eval - 1e-12:
                best_eval = cur_eval
                best_seq = a.copy()
                print(f"[{s_idx}:{iter_local}] new best {best_eval:.6f}")
                no_improve = 0
            else:
                no_improve += 1

            # occasional jitter
            if iter_local % 200 == 0 and iter_local > 0:
                a = _project_simplex(a + np.random.randn(n) * 1e-5)

            iter_local += 1
            # if max_val is already low enough (< threshold)
            target_val = 0.5 * (best_eval / (2*n)) # not needed

            if time.time() >= deadline:
                break

        print(f"[start {s_idx}] done after {iter_local} iters, best eval {best_eval:.6f}")

    # final result
    if best_seq is None:
        # fallback uniform
        best_seq = np.full(n, 1.0 / n)
    print(f"[final] best evaluation= {best_eval:.6f}")
    return best_seq.tolist()

import time, numpy as np

def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _sqrt_series(n):
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k-1] * (2*(k-1) + 1) / (2*k)
    w /= w.sum()
    return w

def _max_conv_gradient(a, idx):
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[i] += 2.0 * a[k - i]
    if len(idx) > 1:
        grad /= len(idx)
    return grad

def _next_pow2(x):
    return 1 << (x-1).bit_length()

import time

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008) – required after every update
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # rho = max {j | u_j - cssv_j / (j+1) > 0}
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “square‑root” start (coefficients of (1‑x)^(-1/2))
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑x)^{-1/2}
    w[0] = 1,
    w[k+1] = w[k] * (2k+1) / (2(k+1)).
    Normalised to sum = 1.
    """
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max_k (a∗a)_k at the active indices
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient for the active convolution indices."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[i] += 2.0 * a[k - i]
    if len(idx) > 1:

    return grad
