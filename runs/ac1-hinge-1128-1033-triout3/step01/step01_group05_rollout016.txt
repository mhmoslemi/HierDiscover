"""
propose_candidate
-----------------

Search for a low‑score step‑function using the analytically optimal
“central‑binomial” coefficients:

    b_k = C(2k, k) / 4**k   (k = 0, 1, …)

For any prefix of length *n* the self‑convolution of this prefix is
identically 1, so the objective reduces to

    score(n) = 2 * n / ( Σ_{k<n} b_k )² .

The score decreases monotonically with *n* and tends to π/2 ≈ 1.5708.
We therefore generate a long prefix (up to a user‑adjustable bound,
default 15000), pick the length with the smallest analytic score and
return the corresponding coefficient list.  The construction is fully
deterministic, cheap to compute and respects the 1000 s time budget.
A tiny stochastic refinement (random perturbations) is performed if
time permits – it can only improve the score.
"""

import time
import numpy as np
from typing import List

def _binomial_prefix(max_n: int) -> np.ndarray:
    """
    Return an array `b` where b[k] = C(2k, k) / 4**k for 0 <= k < max_n.
    Computed via the recurrence:
        b[0] = 1,
        b[k+1] = b[k] * (2*k + 1) / (2*(k+1)).
    """
    b = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    b[0] = cur
    for k in range(1, max_n):
        cur = cur * (2 * k - 1) / (2 * k)   # same as (2k-1)/(2k) * previous
        b[k] = cur
    return b


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     max_len: int = 15000,
                     **kwargs) -> List[float]:
    """
    Produce a sequence of non‑negative heights that (heuristically) minimises
    the evaluation function defined in `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for any stochastic steps.
    budget_s : int
        Total time budget in seconds (the caller ensures 1000 s max).
    max_len : int
        Upper bound on the length of the generated step‑function.
        Larger values get the score closer to π/2 but increase memory
        usage and the cost of optional local search.

    Returns
    -------
    List[float]
        The candidate sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – generate a long binomial prefix and pick the best length
    # ------------------------------------------------------------------
    # The analytic score 2*n/(cum_sum**2) decreases with n, so the longest
    # prefix we can afford is (usually) the best.
    b_coeffs = _binomial_prefix(max_len)

    cum_sum = 0.0
    best_len = 1
    best_score = float('inf')
    # Scan once – O(max_len) operations only.
    for n in range(1, max_len + 1):
        cum_sum += b_coeffs[n - 1]
        # max convolution for any prefix of the central‑binomial sequence is 1
        score = 2.0 * n / (cum_sum * cum_sum)
        if score < best_score:
            best_score = score
            best_len = n

    # ------------------------------------------------------------------
    # Phase 2 – build the initial candidate from the optimal prefix
    # ------------------------------------------------------------------
    candidate = b_coeffs[:best_len].tolist()
    # sanity‑check with the actual evaluator (may be a tiny bit slower)
    try:
        real_score = evaluate_sequence(candidate)
        print(f"[init] length={best_len}, analytic={best_score:.9f}, real={real_score:.9f}")
    except Exception:
        real_score = best_score
        print(f"[init] length={best_len}, analytic={best_score:.9f}, real=error")
    best_real = real_score
    best_seq = candidate[:]

    # ------------------------------------------------------------------
    # Phase 3 – optional lightweight stochastic hill‑climb
    # ------------------------------------------------------------------
    cur_seq = best_seq[:]
    cur_score = best_real
    sigma = 0.04                 # size of relative gaussian tweaks
    max_no_improve = 3000
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(len(cur_seq))
        old = cur_seq[idx]
        # perturb proportionally to the current magnitude (avoid zero)
        delta = np.random.randn() * sigma * max(old, 1e-12)
        new_val = max(0.0, old + delta)   # keep non‑negative

        cand = cur_seq[:]
        cand[idx] = new_val

        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue

        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_real:
                best_seq = cand[:]
                best_real = cand_score
                print(f"[improve] new best {best_real:.9f} len={len(best_seq)} idx={idx}")
        else:
            no_improve += 1

    # final safety: ensure all entries are non‑negative floats
    best_seq = [float(max(0.0, x)) for x in best_seq]
    return best_seq
