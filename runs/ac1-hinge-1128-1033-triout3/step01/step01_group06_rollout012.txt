"""
Propose a step‑function sequence that minimizes
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a hybrid of fast simulated‑annealing (SA) and a
linear‑programming (LP) refinement:

1. **Initialisation** – start from the previously found sequence
   (`height_sequence_1`) if it exists, otherwise generate a strongly
   edge‑heavy shape `a_i = (1‑sin(π·i/(n‑1)))**2`.  A tiny epsilon is added
   to guarantee strict positivity and the vector is normalised to sum = 1.

2. **SA phase** (≈30 % of the time budget) – uses the fast convolution
   update from the reference solution.  Random or “targeted” moves
   relocate a small amount of mass from a source index to a destination
   index.  The temperature is linearly cooled; the best vector seen so far
   is recorded.

3. **LP refinement** – for the current vector `a` we solve

        maximise   sum(b_j)
        subject to (a * b)_k ≤ M   for all k = 0…2n‑2
                    b_j ≥ 0,   b_j ≤ 1000

   where `M = max(a * a)`.  The constraints are linear because each entry
   of the cross‑convolution `a*b` is a linear form in `b`.  The LP is
   solved with `scipy.optimize.linprog` (method “highs”).  The optimal
   `b` has a larger total mass than `a`; after normalising `b` to unit
   sum we move the current solution towards it:
        a ← (1‑t)·a + t·b̂
   with a back‑tracking line search on `t` (starting from 0.5) that
   guarantees the new self‑convolution does not increase the peak `M`.
   The process is repeated until time runs out or no further improvement
   is possible.

4. **Final check** – the best sequence found is evaluated with the
   supplied `evaluate_sequence` for safety and returned.

The combination of a fast SA warm‑start and an LP that explicitly
maximises the admissible mass under the current peak bound yields a
significant reduction of the evaluation score compared with pure SA.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Fast convolution update when moving a mass δ from index i to j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].
    a – 1‑D array of length n, sum(a) = 1
    conv – convolution a*a (length 2n‑1)
    i, j – distinct indices, 0 ≤ i,j < n
    delta – 0 < delta ≤ a[i]
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta            # change at i
    dj = +delta            # change at j

    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    conv_new = conv.copy()

    # indices other than i and j
    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]                # all other positions
    a_q = a[q]

    # contributions of the changed entries to all sums
    conv_new[i + q] += 2.0 * di * a_q    # (i, q) and (q, i)
    conv_new[j + q] += 2.0 * dj * a_q    # (j, q) and (q, j)

    # self‑pair contributions (i,i) and (j,j)
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Targeted move that focuses on the current maximal convolution entry
# ----------------------------------------------------------------------
def targeted_move(a: np.ndarray, conv: np.ndarray):
    """
    Choose (src, dst) that are most responsible for the current maximal
    convolution entry and a random destination dst ≠ src.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))                     # position of the current peak

    # feasible i so that both i and k-i lie inside the vector
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_candidates = np.arange(i_min, i_max + 1, dtype=int)
    j_candidates = k - i_candidates

    # contribution of each unordered pair to the peak
    contrib = a[i_candidates] * a[j_candidates]
    src = int(i_candidates[np.argmax(contrib)])

    # pick a random destination different from src
    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1
    return src, dst


# ----------------------------------------------------------------------
# Build the convolution constraint matrix A for a given a:
#   (A @ b)[k] = (a * b)[k]   with 0‑based indexing
# ----------------------------------------------------------------------
def build_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Returns a dense matrix A of shape (2*n-1, n) such that
    (A @ b) equals the discrete convolution a * b.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes a shifted copy of a
    for j in range(n):
        A[j:j + n, j] = a
    return A
