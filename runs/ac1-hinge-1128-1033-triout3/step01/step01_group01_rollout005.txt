assistantfinal**Solution Overview**

The evaluation function  

\[
M(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

is homogeneous, therefore we may restrict ourselves to probability vectors  
\(p_i\ge 0,\;\sum p_i=1\) and minimise  

\[
\phi(p)=2\,n\;\max(p*p)\; .
\]

The maximum of a convolution is a smooth function of the *soft‑max* of the
convolution:
\[
\operatorname{softmax}_\alpha (x) = \frac{\exp(\alpha x)}{\sum\exp(\alpha x)} .
\]
For a fixed temperature \(\alpha\) the gradient of  
\(f_\alpha(p)=2n\sum_k \operatorname{softmax}_\alpha(p*p)_k\) is

\[
\nabla f_\alpha(p)= 2\,\bigl(p\star w^\text{rev}\bigr)_{2n-2-\;[\cdot]},\qquad
w_k\propto\exp\bigl(\alpha(p*p)_k\bigr),
\]

which is obtained by a single correlation (implemented with `np.convolve`).

The optimiser works in three stages

1. **Multiple restarts** – a fresh random probability vector (or the
   known construction) is created for each restart.
2. **Projected Adam on the smoothed objective** – the Adam‑style
   adaptive step is applied, the iterate is projected onto the simplex
   after every step, the temperature \(\alpha\) is slowly increased so
   that the smooth approximation becomes tighter.
3. **Hard‑max refinement** – after a few thousand Adam steps a short
   sub‑gradient descent on the *actual* maximum (no smoothing) is
   performed, again with simplex projection.

All work is bounded by the supplied time budget; the best sequence found
so far is returned.

```python
"""
Projected Adam optimisation of the soft‑max approximation of
max(p*p) on the probability simplex, followed by a hard‑max
sub‑gradient refinement.  Multiple random restarts (or a warm‑start
from the previously found construction) are performed until the
time budget expires.  The best sequence found is returned as a
plain list of floats suitable for evaluate_sequence().
"""
import time
import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Duchi et al., ICML 2008.
    """
    if np.isclose(np.sum(v), 1.0) and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        # Pre‑computed indices for the gradient formula
        idx = 2 * n - 2 - np.arange(n)   # shape (n,)

        # ----------------------------------------------------------
        # Adam optimisation on the smoothed objective
        # ----------------------------------------------------------
        max_iters = 25000
        for it in range(max_iters):
            now = time.time()
            if now >= deadline - 1.0:
                break

            # 1️⃣  convolution p*p
            conv = np.convolve(p, p)                 # length 2n‑1

            # 2️⃣  soft‑max weights (smooth approximation of max)
            z = alpha * conv
            z_max = np.max(z)
            exp_z = np.exp(z - z_max)               # stabilise
            w = exp_z / np.sum(exp_z)               # shape (2n‑1,)

            # 3️⃣  gradient of the smoothed objective
            w_rev = w[::-1]
            corr = np.convolve(p, w_rev)            # length 3n‑2
            grad = 2.0 * corr[idx]                  # shape (n,)

            # 4️⃣  Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            p = _simplex_projection(p - step)

            # 5️⃣  occasional random jitter to escape shallow minima
            if it % 500 == 0 and it > 0:
                noise = rng.normal(scale=0.004, size=n)
                p = _simplex_projection(p + noise)

            # 6️⃣  schedule updates
            lr *= lr_decay
            alpha *= alpha_mul

            # 7️⃣  current objective value
            cur_score = 2.0 * n * np.max(conv)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.copy()
                print(f"[Restart {restart}] iter {it+1:5d} – new best {best_score:.6f}")

            # early exit if we already beat the target
            if best_score <= 1.503:
                deadline = time.time()  # force termination after this loop
                break

        # ----------------------------------------------------------
        # Hard‑max sub‑gradient refinement (only if time permits)
        # ----------------------------------------------------------
        if time.time() < deadline - 2.0 and best_seq is not None:
            # use the current p (already the best at the end of Adam)
            for sub_it in range(1200):
                if time.time() >= deadline - 1.0:
                    break
                conv = np.convolve(p, p)
                k_max = int(np.argmax(conv))                # index of the current peak
                # sub‑gradient of conv(k_max) w.r.t. p:
                grad_hard = np.zeros_like(p)
                idxs = np.arange(n)
                partner = k_max - idxs
                mask = (partner >= 0) & (partner < n)
                grad_hard[mask] = 2.0 * p[partner[mask]]
                # simple step (reuse lr, but a bit smaller)
                sub_lr = lr * 0.3
                p = _simplex_projection(p - sub_lr * grad_hard)

                cur_score = 2.0 * n * np.max(conv)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = p.copy()
                    print(f"[Restart {restart}] sub‑grad {sub_it+1:4d} – improved {best_score:.6f}")

                if best_score <= 1.503:
                    deadline = time.time()  # stop as soon as we reach the target
                    break

        # -----------------------------------------------------------------
        # If the overall time budget is almost exhausted we stop launching
        # new restarts.
        # -----------------------------------------------------------------
        if time.time() >= deadline - 2.0:
            break

    # -----------------------------------------------------------------
    # Fallback – uniform distribution if nothing was ever accepted.
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 1153
        best_seq = np.full(n, 1.0 / n)

    # Clip to the limits enforced by evaluate_sequence (0 … 1000)
    result = np.clip(best_seq, 0.0, 1000.0).tolist()
    return result
```
