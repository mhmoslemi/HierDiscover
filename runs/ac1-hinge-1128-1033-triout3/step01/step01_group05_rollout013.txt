def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # start with length n = 500 (or maybe 1000)
    n = 500
    # get binomial prefix as initial f
    f = generate_binomial_prefix(n)
    # normalization: we don't need to scale because objective scale-invariant; keep as is.
    # compute initial max:
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    best_seq = f.tolist()
    best_score = evaluate_sequence(best_seq)

    t = 0.5  # maybe 0.5 blending? The paper suggests small t.
    # iterate
    while time.time() < deadline:
        # Build LP to maximize sum(g)
        # A and b
        A = []
        b = np.full(2*n-1, M, dtype=np.float64)
        for k in range(2*n-1):
            row = np.zeros(n)
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            # for i in range(i_start, i_end+1):
            #    row[i] = f[k-i]
            # Use vectorized indexing:
            # We can fill row[i_start:i_end+1] = f[k - np.arange(i_start,i_end+1)]
            idxs = np.arange(i_start, i_end+1)
            row[idxs] = f[k - idxs]
            A.append(row)
        A = np.array(A)  # shape (2n-1,n)
        # Solve LP
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g = res.x
        # Optionally, short-circuit if sum not improved
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g <= sum_f*0.999:  # not much improvement
            # consider break
            pass
        # blend
        t = 0.2
        f_new = (1 - t) * f + t * g
        # compute new max conv
        conv_f_new = np.convolve(f_new, f_new)
        M_new = conv_f_new.max()
        # If M_new < M - small epsilon: accept
        if M_new < M - 1e-8 or (M_new < M * 0.999):
            f = f_new
            M = M_new
            cur_score = evaluate_sequence(f.tolist())
            if cur_score < best_score:

                best_seq = f.tolist()
                print("[improve] score=", best_score, "n=", n, "M=", M)
        else:
            # maybe reduce t or break

            break

"""
propose_candidate
-----------------

Search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm starts from the analytically optimal binomial prefix (the
coefficients of (1‑x)^‑1/2).  It then performs a few iterations of a
linear‑programming “boost” step:

* Given the current vector `f`, compute M = max(f*f).
* Solve the LP  maximise   sum(g_i)
  subject to       f*g ≤ M   (element‑wise convolution)
                    g_i ≥ 0 .
  This adds as much mass as possible in positions that are *light* for the
  current convolution, while keeping the worst convolution value ≤ M.
* Blend the old vector with the LP solution (`f ← (1‑t)f + t g`) and keep the
  blend that gives the best (lowest) objective.  The blend factor `t` is tried
  over a short geometric schedule (0.5, 0.25, 0.125, …) so we do not raise the
  maximal convolution too much.

A few such LP‑boost iterations (normally < 30) raise the ℓ₁‑norm of the
sequence far beyond the binomial bound while keeping the maximal
self‑convolution close to its original value.  The final candidate is
returned as a plain Python list of floats.

The whole routine respects a 1000 s wall‑clock budget and falls back to the
previous simple hill‑climber if SciPy is unavailable.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def generate_binomial_prefix(max_n: int):
    """
    Central binomial coefficients divided by 4**k:
        b_k = C(2k, k) / 4**k
    They satisfy (b * b)_k = 1 for the infinite series.
    """
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, max_n):
        # recurrence  b_{k} = b_{k-1} * (2k-1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


def _objective(seq: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence() – the evaluator enforces the same
    clipping, positivity and the 0‑1000 clamp, but the generated sequences
    already respect those rules.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Return a list of non‑negative heights with a low evaluation score.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Parameters (feel free to tune)
    # ------------------------------------------------------------------
    N = 800                     # start length – fits easily into a LP
    MAX_ITER = 30              # LP boost iterations (usually enough)
    T_CANDIDATES = [0.5, 0.25, 0.125, 0.0625, 0.03125]  # blend factors

    # ------------------------------------------------------------------
    # Phase 1 – initialise with the binomial prefix (M = 1)
    # ------------------------------------------------------------------
    f = generate_binomial_prefix(N)          # NumPy array, float64
    # enforce the clamping that evaluate_sequence does (not really needed)
    f = np.clip(f, 0.0, 1000.0)

    cur_score = _objective(f)
    best_seq = f.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Attempt to import SciPy – if unavailable we fall back to a tiny
    # stochastic hill‑climber (the earlier method).
    # ------------------------------------------------------------------
    try:
        from scipy.linalg import toeplitz
        from scipy.optimize import linprog
        scipy_available = True
    except Exception:                         # pragma: no cover
        scipy_available = False

    # ------------------------------------------------------------------
    # Phase 2 – iterative LP boost (only if SciPy is present)
    # ------------------------------------------------------------------
    if scipy_available:
        for it in range(MAX_ITER):
            if time.time() >= deadline:
                break

            # current maximal self‑convolution
            conv_f = np.convolve(f, f)
            M = conv_f.max()

            # ----- build the (f * g) ≤ M constraints as a dense matrix ----------
            # Toeplitz matrix where each row k contains f_{k‑i} (or 0)
            col = np.concatenate([f, np.zeros(N - 1, dtype=np.float64)])
            row = np.concatenate([f[:1], np.zeros(N - 1, dtype=np.float64)])
            A = toeplitz(col, row)                     # shape (2N‑1, N)
            b = np.full(A.shape[0], M, dtype=np.float64)

            # ----- linear programme: maximise sum(g) ----------------------------
            c = -np.ones(N, dtype=np.float64)          # linprog minimises
            bounds = [(0.0, 1000.0)] * N                # obey the 0‑1000 clamp
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:                        # pragma: no cover
                # If the LP fails we abort the LP phase and fall back.
                print("[LP] failure at iteration", it, "msg:", res.message)
                break

            g = np.maximum(res.x, 0.0)                 # safety
            # ------------------------------------------------------------------
            # Try a few blend factors and keep the one that improves the score
            # ------------------------------------------------------------------
            improved = False
            for t in T_CANDIDATES:
                f_candidate = (1.0 - t) * f + t * g
                f_candidate = np.clip(f_candidate, 0.0, 1000.0)
                cand_score = _objective(f_candidate)

                if cand_score < best_score - 1e-12:

                    f = f_candidate
                    best_seq = f_candidate.copy()
                    best_score = cand_score
                    improved = True
                    print(f"[improve] iter={it:02d} t={t:.4f} score={best_score:.6f}")

            if not improved:
                # No blend gave an improvement – stop early.
                print("[LP] no improvement at iteration", it)
                break
    else:  # ----------------------------------------------------------------
        # SciPy not present – fall back to a very light stochastic search.
        # This reproduces essentially the “tiny stochastic hill‑climb” from the
        # previous solution.
        print("[fallback] SciPy not available – using random walk.")
        sigma_rel = 0.07
        no_improve = 0
        while time.time() < deadline and no_improve < 3000:
            idx = np.random.randint(N)

            delta = np.random.randn() * sigma_rel * max(old, 1e-12)
            new_val = max(0.0, old + delta)
            f_new = f.copy()
            f_new[idx] = new_val
            cand_score = _objective(f_new)
            if cand_score < best_score:
                f = f_new
                best_seq = f_new.copy()
                best_score = cand_score
                no_improve = 0
                print(f"[rand] new best {best_score:.6f}")
            else:
                no_improve += 1

    # ------------------------------------------------------------------
    # Final sanity – ensure no negative numbers and clamp to the allowed range
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
