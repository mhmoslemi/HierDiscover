"""
Propose a step‑function (non‑negative heights) that minimises

        M = 2·n·max(conv(a, a)) / (sum(a)**2)

The algorithm works in two phases inside the allotted time:

1. **Smoothed projected gradient descent** – a soft‑max approximation of the
   max‑operator is gradually tightened (parameter α → ∞).  The gradient of the
   smooth objective is obtained by a single correlation, then the iterate is
   projected onto the probability simplex.

2. **Exact max sub‑gradient with multiplicative (exponentiated) updates** – after
   phase‑1 the algorithm switches to a sub‑gradient of the true max.  The update
   is performed in the exponentiated‑gradient (mirror‑descent) fashion, which
   naturally respects the simplex constraints.  A tiny amount of Gaussian
   jitter is added periodically to escape shallow plateaus.

Throughout both phases the best sequence found so far is kept.  When time runs
out (≈ budget – 5 s safety margin) the best sequence is returned as a plain
Python list, already satisfying the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0 , Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        if total > 0.0:

        else:
            p = np.full(n, 1.0 / n)
    else:
        p = rng.random(n)
        p = p / p.sum()

    best_p = p.copy()

    # ------------------------------------------------------------------

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    alpha = 5.0           # start smoothing factor for soft‑max
    alpha_mul = 1.005    # speed of tightening the approximation
    lr = 0.5              # initial step size

    it = 0

        # ----- smooth approximation of max(conv) -----
        z = alpha * conv
        z_max = np.max(z)

        # ----- gradient of the smooth objective -----
