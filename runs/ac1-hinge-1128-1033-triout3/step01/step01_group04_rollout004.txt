def simplex_projection(v):
    # Project v onto simplex {x >= 0, sum x = 1}
    # Works for 1D numpy array
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _compute_score(seq):
    seq_f = np.asarray(seq,dtype=np.float64)
    n= len(seq_f)
    L = 1 << ((2*n-1).bit_length())
    fft_seq = np.fft.rfft(seq_f, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = np.max(conv)

"""
Propose a candidate step‑function (list of non‑negative heights).

Algorithm
---------
1. Choose a moderate length `n` (default 2000) and work with a
   probability vector `a` (sum = 1).  For such vectors the evaluation
   function reduces to `2 * n * max(conv(a,a))`.

2. Initialise several candidates:
   * the normalised arcsine (inverse‑square‑root) series,
   * a few random points on the simplex,
   * a family of power‑law sequences `a_i ∝ (i+shift)^{-p}` with
     randomly chosen `p` and `shift`.

   The best of these is kept as the starting point.

3. Refine the best candidate by projected gradient descent on a
   smooth approximation of the objective.
   * For a vector `a` the convolution `c = a * a` is computed by FFT.
   * The soft‑max weight vector
         `w_k ∝ exp( β (c_k – max(c)) )`
     approximates the indicator of the maximal entry.
   * The gradient of the smooth objective is
         `∇ = 2 * (w ⊛ rev(a))[n‑1 : n‑1+n]`,
     where `rev(a)` is `a` reversed and `⊛` denotes linear convolution.
   * After each gradient step the iterate is projected onto the
     probability simplex (non‑negative, sum = 1).
   * The temperature `β` is increased slowly from 5 to ~200,
     while the step size decays, to first explore broadly and then
     finely decrease the maximal convolution entry.

4. Throughout the run the best sequence (lowest true score) is stored.
   The loop stops a short time before the allocated budget.

The method combines a fast global search (randomised parametric families)
with a deterministic, constrained local optimiser that directly
targets the `max(conv)` term.  Empirically it can improve over the
plain arcsine construction, moving the evaluation below π/2 ≈ 1.5708.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _arcsine_coeffs(n: int) -> np.ndarray:
    """First `n` coefficients of (1‑z)^{-1/2}."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x >= 0, sum x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho value
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # edge case: all entries are zero or negative
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of `evaluate_sequence` for a normalised sequence
    (sum = 1).  Returns 2 * n * max(conv(seq, seq)).
    """
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    # sum(seq) is assumed to be 1, but we keep the general formula
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _make_powerlaw(n: int, p: float, shift: float) -> np.ndarray:
    """
    Return a normalised power‑law vector: a_i ∝ (i + shift)^{-p}.
    """
    idx = np.arange(n, dtype=np.float64)
    # avoid division by zero
    vec = (idx + shift) ** (-p)
    # if shift is zero the first element would be infinite for p>0,
    # so we enforce a tiny shift in that case
    if np.isinf(vec).any():
        vec = (idx + shift + 1e-12) ** (-p)
    vec_sum = np.sum(vec)
    if vec_sum == 0:
        vec = np.full(n, 1.0 / n)
    else:
        vec = vec / vec_sum
    return vec


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     length: int = 2000) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (a small safety margin is subtracted).
    length : int, optional
        Length of the candidate sequences.

    Returns
    -------
    list[float]
        The best sequence discovered before the time limit.
    """
    np.random.seed(seed)
    # safety margin: finish a bit early to avoid forced termination
    deadline = time.time() + budget_s - 0.5

    n = length
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. generate initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # (a) normalised arcsine
    arcs = _arcsine_coeffs(n).astype(np.float64)
    arcs /= arcs.sum()
    init_candidates.append(arcs)

    # (b) a few random simplex points
    for _ in range(4):
        rnd = np.random.rand(n)
        rnd /= rnd.sum()
        init_candidates.append(rnd)

    # (c) random power‑law families
    for _ in range(30):
        p = np.random.uniform(0.2, 0.9)          # exponent
        shift = np.random.uniform(0.0, 5.0)     # additive shift
        pl = _make_powerlaw(n, p, shift)
        init_candidates.append(pl)

    # Evaluate all initial candidates and keep the best
    for cand in init_candidates:
        sc = _fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[init] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. gradient‑descent refinement (projected onto simplex)
    # ------------------------------------------------------------------
    # start from the best initial point
    seq = best_seq.copy()
    # algorithmic parameters
    beta = 5.0               # initial soft‑max temperature
    beta_max = 200.0
    beta_multiplier = 1.003
    lr = 0.2                 # initial learning rate
    lr_decay = 0.9995
    min_lr = 1e-5
    iter_cnt = 0

    # pre‑allocate some arrays to avoid repeated allocations where possible
    while time.time() < deadline:
        iter_cnt += 1

        # ---- convolution via FFT (fast) ----
        L = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
        max_conv = np.max(conv)

        # current objective (sum(seq) == 1, so 2*n*max_conv)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iter_cnt} score={best_score:.6f}")

        # ---- soft‑max weights for the maximal entry ----
        # shift by max to keep exponentials ≤ 1 (numerical stability)
        w = np.exp(beta * (conv - max_conv))
        w_sum = np.sum(w)
        if w_sum == 0:
            # numerical underflow; fallback to uniform weights
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth max ----
        # we need 2 * (w ⊛ rev(seq))[n‑1 : n‑1+n]
        rev_seq = seq[::-1]
        L2 = 1 << ((w.size + rev_seq.size - 1).bit_length())
        fft_w = np.fft.rfft(w, n=L2)
        fft_rev = np.fft.rfft(rev_seq, n=L2)
        conv_wr = np.fft.irfft(fft_w * fft_rev, n=L2)
        conv_wr = conv_wr[: w.size + rev_seq.size - 1]   # length 3n‑2
        grad = 2.0 * conv_wr[n - 1 : n - 1 + n]

        # ---- gradient step + projection onto simplex ----
        seq = seq - lr * grad
        seq = _simplex_projection(seq)

        # ---- schedule updates ----
        beta = min(beta * beta_multiplier, beta_max)
        lr = max(lr * lr_decay, min_lr)

        # occasional progress report (every 5000 iterations)
        if iter_cnt % 5000 == 0:
            print(f"[status] iter={iter_cnt} beta={beta:.2f} lr={lr:.6f} "
                  f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3. final reporting
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {iter_cnt} iterations")
    # Return as a plain Python list – the evaluator expects a list of floats
    return best_seq.tolist()
