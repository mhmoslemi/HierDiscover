import numpy as np, time, math
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    # Determine n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000
    # Build Catalan sequence
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1,n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    # Scale to sum sqrt(2n)
    target_sum = math.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    
    f = a.copy()
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")
    
    deadline = time.time() + budget_s - 5.0
    t = 0.5
    iteration = 0
    while time.time() < deadline:
        # compute convolution and M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Build constraints matrix for linprog (A_ub * g <= b_ub)
        # each row k: sum_i f_i * g_{k-i} <= M
        rows = []
        cols = []
        data = []
        for k in range(2*n-1):
            i_start = max(0, k - (n-1))
            i_end = min(k, n-1)
            # i indexes f_i, g_{k-i}
            # for each i in [i_start, i_end]:
            # coefficient for variable g_{k-i} is f_i
            # So column index j = k - i
            # Add to sparse matrix
            for i in range(i_start, i_end+1):
                rows.append(k)
                cols.append(k - i)
                data.append(f[i])
        A = coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
        # bounds: g >=0
        bounds = [(0, None)] * n
        # Objective: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n)
        # RHS: M for each row
        b = np.full(2*n-1, M, dtype=np.float64)
        # Solve LP
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print("[linprog] iteration failed, stop")
            break
        g0 = res.x
        # Ensure positive
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print("[linprog] zero solution, stop")
            break
        # Scale g to same target sum
        g = g0 * (target_sum / sum_g0)
        # Combine f and g
        f = (1 - t) * f + t * g
        # Evaluate
        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[iter {iteration}] improved score={best_score:.6f}")
        # Reduce t maybe
        t = max(0.1, t * 0.95)
        iteration += 1
    return best_seq.tolist()

"""
propose_candidate
-----------------

Implements the classic alternating‑LP refinement that is used for the
“flatten the self‑convolution” problem.  We start from a Catalan (central
binomial) sequence, normalise its ℓ₁‑norm to √(2n) (so that the evaluation
score equals the maximal convolution value) and then repeatedly solve a
linear programme

    maximise   ∑ g_j
    subject to  g_j ≥ 0
                (f * g)_k ≤ M   for every k

where f is the current sequence and M = max(f * f).  The programme adds as
much mass as possible in positions that are “safe’’ – i.e. where the
convolution with the current f is still below the current worst value M.
After the LP we rescale g to have the same ℓ₁‑norm as f, take a small convex
step f ← (1‑t)·f + t·g and recompute M.  This never raises the worst
convolution value and usually shrinks it a little; after a few iterations
the maximal value drops from π/2≈1.5708 toward the target ≈1.503.

The LP constraints are a Toeplitz matrix; they are built with
`scipy.sparse.diags` and solved with the HiGHS solver.  The routine runs
as long as time permits (up to the 1000 s budget) and returns the best
sequence seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – they are defined at module level as required.
# ----------------------------------------------------------------------
def catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first n Catalan / central‑binomial coefficients:
        c_0 = 1,
        c_{k+1} = c_k * (2k+1) / (2(k+1)).
    These coefficients satisfy (c*c)_k = 1 for all k < n (in the infinite
    limit) and give the classic π/2 bound.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    return a


def conv_matrix(f: np.ndarray) -> "scipy.sparse.csr_matrix":
    """
    Build the (2n‑1) × n sparse matrix A such that
        (A @ g)[k] = (f * g)[k]   for all k.
    The matrix is Toeplitz: each diagonal d (0 ≤ d < n) consists of the
    constant value f[d].
    """
    from scipy.sparse import diags

    n = f.shape[0]
    # each diagonal d gets a constant column of length n with value f[d]
    diagonals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)               # d = row – col
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format="csr")
    return A

    expires or no further improvement is observed.
    """

    import math
    import time

    # Try to import the LP solver; if unavailable fall back to plain Catalan.
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception as exc:      # pragma: no cover
        print("SciPy not available – falling back to pure Catalan sequence.")
        # fallback: the same construction that the previous run used
        try:
            n = len(height_sequence_1)
        except NameError:
            n = 2000
        seq = catalan_sequence(n).tolist()
        return seq

    # ------------------------------------------------------------------
    # 1. Choose the length of the vector.
    #    Use an existing construction if present, otherwise a modest default.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # reuse length if a previous run gave one
    except NameError:
        n = 2000                            # default length – a good trade‑off

    # ------------------------------------------------------------------
    # 2. Initialise with the Catalan sequence and fix the ℓ₁‑norm to √(2n).
    #    With this normalisation the evaluation score is exactly the
    #    maximal self‑convolution value.
    # ------------------------------------------------------------------
    f = catalan_sequence(n)
    target_sum = math.sqrt(2.0 * n)          # the normalising constant √(2n)
    f = f * (target_sum / f.sum())           # now sum(f) == target_sum
    conv_f = np.convolve(f, f)
    M = conv_f.max()                         # current worst convolution value
    best_seq = f.copy()
    best_score = M
    print(f"[init] n={n}, initial max_convolution={M:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – alternating LP + convex update.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # small safety margin
    iteration = 0
    t = 0.3                                   # initial convex‑step size

    while time.time() < deadline:
        # 3.1 Build the constraint matrix for (f * g)_k ≤ M
        A = conv_matrix(f)                    # (2n‑1) × n sparse matrix
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # 3.2 Linear programme: maximise Σ g_j   ⇔   minimise –Σ g_j
        c = -np.ones(n, dtype=np.float64)     # objective coefficients
        bounds = [(0.0, None)] * n            # g_j ≥ 0

        # Solve the LP – HiGHS is fast and supports sparse matrices.
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0, deadline - time.time())},
        )

        if not res.success:
            # If the solver fails (rare) we abort the loop.
            print(f"[iter {iteration}] LP failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print(f"[iter {iteration}] LP returned zero vector – stop.")
            break

        # 3.3 Rescale g to have the same ℓ₁‑norm as f (the optimisation is
        #     scale‑invariant, the rescaling only keeps the objective simple).
        alpha = target_sum / sum_g0
        g = g0 * alpha

        # 3.4 Take a convex combination.  If the new max does not improve we
        #     shrink the step size a little and retry.
        improved = False
        t_cur = t
        for _ in range(5):   # try up to 5 reductions of t
            f_new = (1.0 - t_cur) * f + t_cur * g
            conv_new = np.convolve(f_new, f_new)
            M_new = conv_new.max()
            if M_new <= M:                # non‑increasing – accept
                f = f_new
                M = M_new
                improved = True
                if M < best_score:
                    best_score = M
                    best_seq = f.copy()
                    print(
                        f"[iter {iteration}] new best max={best_score:.6f} (t={t_cur:.3f})"
                    )
                break
            # otherwise reduce the step size and try again
            t_cur *= 0.5

        if not improved:
            # No improvement – shrink the global step size to be more
            # conservative in the next iteration.
            t = max(0.02, t * 0.8)

        # If we have already reached the target bound we can stop early.
        if best_score <= 1.503:
            print(f"[stop] reached target bound {best_score:.6f}")
            break

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as plain Python list of floats).
    # ------------------------------------------------------------------
    return best_seq.tolist()
