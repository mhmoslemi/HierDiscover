"""
propose_candidate
=================

Search for a step‑function (non‑negative list of heights) that minimizes the
evaluation function used in the challenge.

The algorithm has three stages:

1. **Randomised parameter search** – a compact parametrisation of a
   step‑function is used:
        * a decreasing “core”  a_i = 1/√(i + offset)
        * an optional “cut” of the first part of the core multiplied by a
          factor < 1 (to suppress the large squares that dominate the max‑convolution)
        * a long “tail” that can be constant or geometrically decaying.
   Random combinations of the parameters (core length, offset, cut length,
   cut factor, tail height, tail decay) are generated until the time budget
   is almost exhausted.  Each candidate is scored with a fast FFT‑based
   convolution.

2. **Local refinement (simulated‑annealing)** – starting from the best
   candidate found, small moves that transfer a tiny amount of mass from one
   position to another are performed.  Metropolis acceptance with a linear
   cooling schedule keeps the search focused on improvements.

3. **Return** – the best sequence discovered is converted to a plain Python
   list (the required format for ``evaluate_sequence``).

All helper functions are defined at module level and no closures or file I/O
are used.  The method explores shapes that go beyond the simple “inverse‑sqrt
core + constant tail” used previously, which gives us a good chance of
beating the 1.574 → 1.503 target.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT based convolution, exact score)
# ----------------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the linear convolution of ``seq`` with itself.

    Uses a power‑of‑two FFT length for speed.
    """
    n = seq.shape[0]
    # size = next power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Exact value of the objective function for a given sequence."""
    total = float(seq.sum())
    if total < 0.01:                # safeguard – never happens for our candidates
        return np.inf
    max_conv = _conv_max(seq)
    return 2.0 * len(seq) * max_conv / (total * total)


# ----------------------------------------------------------------------
# Sequence generator from a compact set of parameters
# ----------------------------------------------------------------------
def _build_sequence(
    n: int,
    core_len: int,
    offset: float,
    cut_len: int,
    cut_factor: float,
    tail_height: float,
    tail_decay: float,
) -> np.ndarray:
    """
    Build a candidate step‑function.

    Parameters
    ----------
    n            – total length of the sequence.
    core_len     – length of the decreasing core (first part of the sequence).
    offset       – shift added inside the square‑root (>=0).
    cut_len      – number of leading core entries that are multiplied by
                   ``cut_factor`` (0 ≤ cut_len ≤ core_len).
    cut_factor   – multiplicative factor for the cut part (0 < cut_factor ≤ 1).
    tail_height  – base height of the tail (constant if tail_decay==1).
    tail_decay   – geometric decay factor for the tail (1 → constant tail,
                   <1 → exponentially decreasing tail).

    Returns
    -------
    seq : np.ndarray of length ``n`` (non‑negative).
    """
    # ------- core -------------------------------------------------------
    idx = np.arange(1, core_len + 1, dtype=np.float64) + offset
    core = 1.0 / np.sqrt(idx)                     # 1/√(i+offset)

    if cut_len > 0:
        core[:cut_len] *= cut_factor

    # ------- tail -------------------------------------------------------
    tail_len = n - core_len
    if tail_len <= 0:
        # shouldn't happen – but return core only to avoid errors
        return core.copy()

    if tail_decay == 1.0:
        tail = np.full(tail_len, tail_height, dtype=np.float64)
    else:
        # geometric series: h, h·r, h·r², …
        tail = tail_height * (tail_decay ** np.arange(tail_len, dtype=np.float64))

    # concatenate and return
    return np.concatenate([core, tail])

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) improves the score.

    The function respects the 1000 s time budget.  It starts from a fairly
    wide random parameter sweep, then performs a short simulated‑annealing
    refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a small safety margin

    # ------------------------------------------------------------------
    # Global length of the candidate (use the provided reference length if
    # available, otherwise fall back to a reasonable default).
    # ------------------------------------------------------------------
    n = len(globals().get("height_sequence_1", []))
    if n == 0:
        n = 10000                     # a comfortable size for FFT + np.convolve

    # ------------------------------------------------------------------
    # Parameter‑space bounds

    min_core_frac = 0.02          # at least 2 % of the total length
    max_core_frac = 0.6           # we never want a core larger than 60 %
    min_log_delta = -7.0          # tail height 10⁻⁷ … 10⁰
    max_log_delta =  0.0
    decay_options = [1.0, 0.99, 0.98, 0.95, 0.90]   # 1 ⇒ constant tail

    # ------------------------------------------------------------------
    # Randomised grid search (time‑controlled)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    iter_count = 0

    while time.time() < deadline:
        # ---- core length ------------------------------------------------
        core_frac = np.random.uniform(min_core_frac, max_core_frac)
        core_len = max(2, int(core_frac * n))

        # ---- offset (float) --------------------------------------------
        offset = np.random.exponential(scale=core_len * 0.2)   # typical 0–few % of core

        # ---- cut of the first part of the core -------------------------
        cut_frac = np.random.uniform(0.0, 0.5)                  # up to half of core
        cut_len = int(cut_frac * core_len)
        cut_factor = 10 ** np.random.uniform(-3.0, 0.0)        # 0.001 … 1

        # ---- tail parameters --------------------------------------------
        log_h = np.random.uniform(min_log_delta, max_log_delta)
        tail_h = 10.0 ** log_h
        tail_decay = np.random.choice(decay_options)

        # ---- build and evaluate -----------------------------------------
        seq = _build_sequence(
            n=n,
            core_len=core_len,
            offset=offset,
            cut_len=cut_len,
            cut_factor=cut_factor,
            tail_height=tail_h,
            tail_decay=tail_decay,
        )
        val = _score(seq)

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(

                f"core={core_len} off={offset:.2f} cut={cut_len} cf={cut_factor:.3g} "
                f"δ={tail_h:.2e} decay={tail_decay}"

        iter_count += 1

        # stop early if we already have a very good value (optional)
        if best_val < 1.52 and time.time() > deadline - 30:
            # keep a few seconds for local refinement
            break

    # ------------------------------------------------------------------
    # Simple simulated‑annealing refinement around the best candidate.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform tiny sequence (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _score(best_seq)

    cur_seq = best_seq.copy()
    cur_val = best_val
    start = time.time()
    T0 = max(0.01, 0.2 * best_val)          # initial temperature

    while time.time() < deadline:
        # ------- propose a move: transfer a tiny amount between two indices -----
        i, j = np.random.randint(n, size=2)
        if i == j or cur_seq[i] <= 0.0:
            continue

        # move at most 0.5 % of the donor entry
        amount = cur_seq[i] * np.random.rand() * 0.005
        new_seq = cur_seq.copy()
        new_seq[i] -= amount
        new_seq[j] += amount

        # avoid negative entries (numerical safety)
        if new_seq[i] < 0.0:
            continue

        new_val = _score(new_seq)

        # ------- Metropolis acceptance ---------------------------------------
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T0):
            cur_seq, cur_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
                print(f"[SA]  improved to {best_val:.6f}")

        # ------- linear cooling -----------------------------------------------
        elapsed = time.time() - start
        T0 = max(1e-7, T0 * (1.0 - elapsed / budget_s))

    # ------------------------------------------------------------------
    # Return the final best sequence as a plain Python list.
    # ------------------------------------------------------------------
    print(f"final best value: {best_val:.6f}")
    return best_seq.tolist()
