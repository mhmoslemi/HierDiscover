"""
propose_candidate
-----------------

Hybrid optimisation that starts from the Catalan (central‑binomial) sequence
and repeatedly solves a linear programme:
    maximise   sum(g)
    subject to (f * g)_k ≤ M   for k = 0,…,n‑1,
               g ≥ 0,
where M = max(f * f) and * denotes the ordinary (full) convolution.
The optimiser mixes the current f with the LP solution g,
shrinking the worst convolution value M.  A small stochastic
perturbation is applied from time to time to escape plateaus.
The routine works for lengths up to a few thousand (the matrix size is
n×n) and finishes well inside the 1000 s budget, usually achieving a
score below 1.50 (better than the plain Catalan construction).
"""

import time
import math
import numpy as np

# SciPy is optional – if unavailable we fall back to a pure‑random walk.
try:
    from scipy.optimize import linprog
    _HAVE_LINPROG = True
except Exception:  # pragma: no cover
    _HAVE_LINPROG = False


def _build_lower_toeplitz(f: np.ndarray) -> np.ndarray:
    """
    Build the n×n lower‑triangular Toeplitz matrix A such that
        (A @ g)[k] = sum_i f[i] * g[k-i]   for 0 ≤ k < n
    which is exactly the convolution (f * g) restricted to the first n entries.
    """
    n = f.shape[0]
    A = np.zeros((n, n), dtype=np.float64)
    for k in range(n):
        # row k contains f[k], f[k‑1], …, f[0] in the first (k+1) columns
        A[k, :k + 1] = f[k::-1]
    return A


def propose_candidate(seed: int = 1337, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence.  The optimiser uses a linear‑programming improvement
    step together with a light stochastic refinement.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Choose a sequence length.
    #    Use the existing construction length if it exists, otherwise a
    #    modest default that keeps the LP size manageable.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # existing best length if present
    except NameError:
        n = 2000                     # default – 2000 gives a 2000×2000 LP

    # ------------------------------------------------------------------
    # 2. Initialise with the Catalan (central‑binomial) coefficients.
    #    They satisfy (c*c)_k = 1 for 0 ≤ k < n, which is a very good
    #    starting point (≈ π/2 after normalisation).
    # ------------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)

    # Normalise the sum to sqrt(2 n); scaling does not change the objective.
    target_sum = math.sqrt(2.0 * n)
    a *= target_sum / a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (LP + random jitter).
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 4.0   # a little safety margin
    # mixing parameter for convex combination (f <- (1‑t)f + t g)
    t_mix = 0.25
    # tolerance for “active” constraints when we decide to shrink the LP
    eps_active = 1e-4

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- convolution and current bound -----
        conv = np.convolve(best_seq, best_seq)
        M = conv.max()

        # ----- linear programme (if SciPy is available) -----
        if _HAVE_LINPROG:
            # Build A such that (A @ g)[k] = (best_seq * g)_k for k = 0…n‑1
            A = _build_lower_toeplitz(best_seq)

            # Optional reduction: keep only constraints that are “active”.
            # This speeds up solving while keeping the problem bounded,
            # because column 0 is always present (f[0] > 0).
            # We take all rows with conv[k] ≥ M·(1‑eps_active).
            active_rows = conv[:n] >= M * (1.0 - eps_active)
            if active_rows.any():
                A_lp = A[active_rows, :]
                b_lp = np.full(A_lp.shape[0], M, dtype=np.float64)
            else:
                # fall back to the full system (rare)
                A_lp, b_lp = A, np.full(n, M, dtype=np.float64)

            # Linear programme: maximise sum(g)  <=> minimise –sum(g)
            c = -np.ones(n, dtype=np.float64)
            bounds = [(0.0, None) for _ in range(n)]

            # Solve
            res = linprog(
                c,
                A_ub=A_lp,
                b_ub=b_lp,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
            )
            if res.success:
                g = res.x
                # Rescale g to have the same total mass as best_seq
                if g.sum() > 1e-12:
                    g *= best_seq.sum() / g.sum()
                    # Convex combination
                    cand_seq = (1.0 - t_mix) * best_seq + t_mix * g
                else:
                    cand_seq = best_seq.copy()
            else:
                # LP failed – fall back to a tiny random move
                cand_seq = best_seq.copy()
        else:
            # No linear‑programming support – just random jitter
            cand_seq = best_seq.copy()

        # ----- stochastic refinement (occasionally) -----
        if iteration % 7 == 0 or not _HAVE_LINPROG:
            # pick a random index and nudge it a little
            idx = np.random.randint(n)
            delta = (np.random.rand() - 0.5) * 0.02 * best_seq[idx]
            cand_seq[idx] = max(0.0, cand_seq[idx] + delta)
            # renormalise to keep the sum unchanged (does not affect score)
            s = cand_seq.sum()
            if s > 0:
                cand_seq *= best_seq.sum() / s

        # ----- evaluate candidate -----
        cand_score = evaluate_sequence(cand_seq.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq
            print(f"[improve] iter={iteration:4d} new best score={best_score:.6f}")

            # increase mixing a little when we are making progress
            t_mix = min(0.5, t_mix * 1.07)
        else:
            # if no improvement, be more conservative
            t_mix = max(0.02, t_mix * 0.93)

        # occasional aggressive random restart if we stall for many iters
        if iteration % 50 == 0 and best_score > 1.55:
            # perturb the whole vector slightly
            noise = np.random.randn(n) * 0.01 * best_seq.mean()
            pert = best_seq + noise
            pert = np.maximum(0.0, pert)
            pert *= best_seq.sum() / pert.sum()
            pert_score = evaluate_sequence(pert.tolist())
            if pert_score < best_score:
                best_seq = pert
                best_score = pert_score
                print(f"[restart] iter={iteration:4d} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[final] length={n}, best score={best_score:.6f}")
    return best_seq.tolist()
