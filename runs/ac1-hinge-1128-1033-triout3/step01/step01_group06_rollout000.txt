"""
Propose a candidate step function using an iterative LP‑based refinement.

The idea follows the “alternating linear programme” described in the
statement: given a current distribution f (sum = 1) we compute M = max(f*f).
We then solve a linear programme that maximises the total mass of a new
distribution b under the same L∞‑bound M on the mixed convolution f*b.
After scaling b to sum to one we convex‑combine it with f.  Repeating this
process strictly reduces the objective 2·n·max(conv) until it stabilises.
A few cheap random mass‑redistribution moves are used as fall‑backs when the
LP is ill‑conditioned (e.g. due to exact zeros in f).  The routine keeps the
best sequence found and returns it before the time budget expires.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: Build the (2n‑1) × n matrix A such that (A @ b) = f * b,
# i.e. the discrete convolution of f with b.
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return dense matrix A with A[k,i] = f[i] if 0 ≤ k‑i < n else 0."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f[i] on rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


# ----------------------------------------------------------------------
# Helper: a tiny random perturbation that respects the non‑negativity
# and normalisation constraints.
# ----------------------------------------------------------------------
def random_redistribute(a: np.ndarray, max_frac: float = 0.3) -> np.ndarray:
    """Move a random fraction (≤max_frac) of mass from a source index
    to a different destination.  Returns the new normalised vector."""
    n = a.shape[0]
    # pick a source with some mass
    possible = np.where(a > 1e-12)[0]
    if possible.size == 0:
        return a.copy()
    src = np.random.choice(possible)
    # pick a destination distinct from src
    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1
    delta = np.random.rand() * max_frac * a[src]
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    # re‑normalise (tiny rounding errors may accumulate)
    a_new /= a_new.sum()
    # enforce strict positivity to keep the LP bounded
    eps = 1e-12
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new
