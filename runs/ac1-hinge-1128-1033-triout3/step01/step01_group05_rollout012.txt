def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # Increase max length gradually while staying within time
    # The binomial prefix can be generated on the fly
    # We'll examine lengths up to maybe 20000 (or more if time permits)
    # In analytic phase we compute cumulative sum incrementally.
    MAX_LEN = 20000

    cumulative = 0.0
    best_len = 1
    best_score = float('inf')
    coeff = 1.0  # b_0
    # loop
    for n in range(1, MAX_LEN+1):
        if n == 1:
            coeff = 1.0
        else:
            k = n-1
            coeff = coeff * (2*k-1) / (2*k)   # recurrence
        cumulative += coeff
        # analytic bound: max conv = 1 for this prefix
        score = 2.0 * n / (cumulative * cumulative)
        if score < best_score:
            best_score = score
            best_len = n
    # Build the sequence for best_len
    # generate again up to best_len
    seq = np.empty(best_len, dtype=np.float64)
    cur = 1.0
    seq[0] = cur
    for i in range(1, best_len):
        cur = cur * (2*i - 1) / (2*i)
        seq[i] = cur

    # sanity evaluate
    try:
        real_score = evaluate_sequence(seq.tolist())
        print(f"[result] length={best_len} analytic={best_score:.9f} real={real_score:.9f}")
    except Exception:
        print("[result] evaluation failed")

    # Optionally fine‑tune with a very cheap local search (limit time)
    cur_seq = seq.tolist()
    cur_score = real_score if 'real_score' in locals() else best_score
    sigma_rel = 0.02
    no_improve = 0
    max_no_improve = 5000

    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(best_len)
        old = cur_seq[idx]
        delta = np.random.randn() * sigma_rel * max(old, 1e-12)
        new_val = max(0.0, old + delta)
        cand = cur_seq[:]
        cand[idx] = new_val
        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue
        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_score = cand_score
                print(f"[improve] new best {best_score:.9f}")
        else:
            no_improve += 1

    # final clamp to non‑negative
    best_seq = [float(max(0.0, x)) for x in cur_seq]
    return best_seq

"""
propose_candidate
-----------------

Return a step‑function that is (almost) optimal for the
objective 2·n·max(conv(a,a)) / (sum(a)^2).

The analytically optimal shape is the *constant‑convolution*
sequence

        b_k = C(2k,k) / 4**k   (central binomial / 4^k)

Its self‑convolution equals 1 on every index that is fully
covered, so the objective reduces to

        score(n) = 2·n / ( Σ_{k<n} b_k )² .

The sum grows like (2/√π)·√n, giving the asymptotic lower bound
π/2 ≈ 1.570796.  By scanning many lengths we can get arbitrarily
close to that bound; the longer the prefix the better.

The function below

* builds the coefficients once (recurrence b_{k+1}=b_k·(2k+1)/(2k+2)),
* scans a large range (up to MAX_LEN, chosen so that the final
  convolution can still be evaluated in a reasonable time),
* picks the prefix with the smallest *analytic* score,
* (optionally) does a very cheap stochastic hill‑climb on that
  prefix – the hill‑climb only calls ``evaluate_sequence`` a few
  thousand times, which is negligible compared with the final
  evaluation performed by the grader,
* returns the best sequence found.

The algorithm respects the 1000 s budget, uses only the public
``evaluate_sequence`` function and never writes to the filesystem.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Central‑binomial coefficients (constant‑convolution sequence)
# ----------------------------------------------------------------------
def generate_constant_convolution(max_n: int) -> np.ndarray:
    """
    Returns an array `c` where
        c[k] = C(2k, k) / 4**k ,   0 <= k < max_n.
    The recurrence   c[k+1] = c[k] * (2k+1) / (2k+2)   is used.
    """
    c = np.empty(max_n, dtype=np.float64)
    cur = 1.0                     # b_0
    c[0] = cur
    for k in range(1, max_n):
        cur = cur * (2 * k - 1) / (2 * k)      # (2k-1)/(2k) = (2(k-1)+1)/(2k)
        c[k] = cur
    return c

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Produce a non‑negative step‑function (list of heights) with a very
    low evaluation score.  The method is based on the optimal
    constant‑convolution sequence; it searches over many possible
    lengths and then performs a tiny stochastic refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ---- Phase 1: analytic scan over many lengths -------------------
    # Very long prefixes give scores arbitrarily close to π/2,
    # but the evaluator must be able to convolve the final sequence,
    # so we cap the length to a value that is still tractable.
    # Empirically a few‑tens of thousands are fast enough.
    MAX_LEN = 20000                     # feel free to increase if you like

    coeffs = generate_constant_convolution(MAX_LEN)

    cum_sum = 0.0
    best_len = 1
    best_score = float('inf')
    # The convolution of any prefix of `coeffs` has max = 1,
    # therefore the analytic score is simply 2·n / (cum_sum)² .
    for n in range(1, MAX_LEN + 1):
        cum_sum += coeffs[n - 1]
        score = 2.0 * n / (cum_sum * cum_sum)
        if score < best_score:
            best_score = score
            best_len = n

    # ---- Phase 2: build the starting candidate ----------------------
    cur_seq = coeffs[:best_len].tolist()        # plain Python list of floats

    # sanity check – evaluate once (may be a bit slow for large n)
    try:
        true_score = evaluate_sequence(cur_seq)
        print(f"[init] length={best_len}, analytic={best_score:.9f}, "
              f"real={true_score:.9f}")
        cur_score = true_score
    except Exception:   # the evaluator may be very slow for huge n; ignore
        print(f"[init] length={best_len}, analytic={best_score:.9f}")
        cur_score = best_score

    # ---- Phase 3: extremely lightweight stochastic hill‑climb -------
    # The hill‑climb works on the *already* good prefix; we only
    # perform a few thousand cheap evaluations, well within the budget.
    sigma_rel = 0.03          # relative size of Gaussian perturbations
    max_no_improve = 3000    # stop after this many non‑improving trials

    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(len(cur_seq))
        old_val = cur_seq[idx]
        # Gaussian step, keep the entry non‑negative
        delta = np.random.randn() * sigma_rel * max(old_val, 1e-12)
        new_val = max(0.0, old_val + delta)

        cand = cur_seq[:]
        cand[idx] = new_val

        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue

        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            print(f"[improve] new best {cur_score:.9f}  (len={len(cand)})")
        else:
            no_improve += 1

    # final safety: ensure all entries are non‑negative and within bounds
    best_seq = [float(max(0.0, x)) for x in cur_seq]

    return best_seq
