    cols = []

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def _compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    L = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq*fft_seq, n=L)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def _arcsine_coeffs(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0]=1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2*i - 1) / (2*i)
    return coeffs

def _build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for d in range(n):
        length = n - d
        rows.append(np.arange(d, d+length, dtype=np.int64))
        cols.append(np.arange(length, dtype=np.int64))
        data.append(np.full(length, f[d], dtype=np.float64))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

def _solve_lp_increase_sum(f, M):
    # maximize sum(g) subject to A g <= M, g>=0
    n = len(f)
    A = _build_conv_matrix(f)
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimize negative sum => maximize sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return uniform distribution
        return np.full(n, 1.0/n)
    return res.x

def propose_candidate(seed=0, budget_s=950, length=2000):
    """
    LP‑based iterative optimisation of a non‑negative step function.
    Starts from the arcsine (inverse‑square‑root) coefficients, then repeatedly
    solves a linear program that maximises the total mass while keeping the
    self‑convolution bounded by the current maximum.  The obtained vector is
    mixed with the current one, projected onto the probability simplex and the
    process repeats until the time budget expires.  The best sequence seen is
    returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # initialise with arcsine coefficients normalised to sum 1
    a = _arcsine_coeffs(length)
    p = a / np.sum(a)
    best_seq = p.copy()
    best_score = _compute_score(best_seq.tolist())
    print(f"[init] length={length}, score={best_score:.6f}")

    iteration = 0
    t = 0.2  # mixing parameter

    while time.time() < deadline:
        iteration += 1
        # compute current convolution max
        seq = p
        n = len(seq)
        L = 1 << ((2*n - 1).bit_length())
        fft_seq = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(fft_seq*fft_seq, n=L)[:2*n-1]
        M = np.max(conv)

        # solve LP to get a vector with larger ℓ1‑norm under same bound M
        g = _solve_lp_increase_sum(seq, M)

        # normalise g to be a probability vector (ℓ1 = 1)
        sum_g = np.sum(g)
        if sum_g <= 0:
            # fallback to uniform if LP failed badly
            g_norm = np.full_like(p, 1.0/len(p))
        else:
            g_norm = g / sum_g

        # convex combination
        p_new = (1 - t) * p + t * g_norm

        # optional projection onto simplex (already sums to 1)
        # ensure non‑negativity
        p_new = np.clip(p_new, 0, None)
        p_new /= np.sum(p_new)  # renormalise to avoid drift

        # evaluate
        cur_score = _compute_score(p_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p_new.copy()
            print(f"[iter {iteration}] improved score={best_score:.6f}  max={M:.6f}")

        # update current p
        p = p_new

        # occasional random perturbation to avoid stagnation
        if iteration % 10 == 0:
            # small random noise
            noise = np.random.exponential(scale=1e-4, size=n)
            p = p + 0.01 * noise
            p = np.clip(p, 0, None)
            p /= np.sum(p)

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog

def _compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    L = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq*fft_seq, n=L)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def _arcsine_coeffs(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0*i - 1.0) / (2.0*i)
    return coeffs

def _precompute_conv_structure(n):
    # returns rows, cols, offsets for the convolution matrix A where (A g)_k = Σ_i f_i g_{k-i}
    rows_list = []
    cols_list = []
    for d in range(n):
        length = n - d
        rows_list.append(np.arange(d, d+length, dtype=np.int32))
        cols_list.append(np.arange(length, dtype=np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offsets = rows - cols  # equals d, 0 <= offsets < n
    return rows, cols, offsets

def _solve_lp_increase_sum(f, M, rows, cols, offsets):
    n = len(f)
    data = f[offsets]  # shape (nnz,)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximize sum(g)
    bounds = [(0, None)] * n
    # Using HiGHS interior point method (default) which handles sparse matrices well
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        # fall back to uniform distribution in pathological case
        return np.full(n, 1.0 / n, dtype=np.float64)
    return res.x

def propose_candidate(seed=0, budget_s=950, length=2000):
    """
    LP‑based iterative refinement of a non‑negative step function.
    The algorithm starts from the arcsine coefficient vector,
    repeatedly solves a linear program that maximises the ℓ1‑norm
    while keeping the convolution with the current vector bounded
    by its current peak, mixes the result back, projects onto the
    probability simplex and repeats until the time budget expires.
    The best candidate encountered is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # initialise with arcsine coefficients normalised to sum 1
    a = _arcsine_coeffs(length)
    p = a / np.sum(a)                # probability vector
    best_seq = p.copy()
    best_score = _compute_score(best_seq.tolist())
    print(f"[init] length={length}, score={best_score:.6f}")

    rows, cols, offsets = _precompute_conv_structure(length)

    t = 0.2            # mixing proportion
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        n = len(p)
        # current convolution and its maximum
        L = 1 << ((2*n - 1).bit_length())
        fft_p = np.fft.rfft(p, n=L)
        conv = np.fft.irfft(fft_p*fft_p, n=L)[:2*n-1]
        M = np.max(conv)

        # linear programme to increase total mass under the same bound M
        g = _solve_lp_increase_sum(p, M, rows, cols, offsets)

        # normalise g to be a probability vector
        sum_g = np.sum(g)
        if sum_g <= 0:
            g_norm = np.full_like(p, 1.0 / n)
        else:
            g_norm = g / sum_g

        # convex combination
        p_new = (1 - t) * p + t * g_norm

        # make absolutely sure we stay on the simplex and stay non‑negative
        p_new = np.clip(p_new, 0.0, None)
        total = np.sum(p_new)
        if total == 0:
            # fallback to uniform
            p_new = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            p_new /= total

        cur_score = _compute_score(p_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p_new.copy()
            print(f"[iter {iteration}] improved score={best_score:.6f}  max={M:.6f}")

        p = p_new

        # occasional tiny random shake to avoid stagnation
        if iteration % 7 == 0:
            noise = np.random.exponential(scale=1e-5, size=n)
            p = p + 0.02 * noise
            p = np.clip(p, 0, None)
            p /= np.sum(p)

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def _compute_score(seq):
    """
    Fast evaluation of the objective 2 * n * max(conv) / (sum^2) using FFT.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _arcsine_coeffs(n):
    """
    First `n` coefficients of (1‑z)^(-1/2) = Σ binom(2i,i)/4^i z^i.
    Recurrence:
        a0 = 1,
        a_{i+1} = a_i * (2*i+1)/(2*(i+1))
    """
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _precompute_conv_structure(n):
    """
    Build the index structure of the Toeplitz matrix that represents
    convolution with a fixed vector `f` of length `n`:
        (f * g)[k] = Σ_i f_i * g_{k-i}
    The returned arrays allow us to create the sparse matrix quickly for any
    new `f` by just providing the data values.
    """
    rows_lst = []
    cols_lst = []
    for d in range(n):
        length = n - d
        rows_lst.append(np.arange(d, d + length, dtype=np.int32))
        cols_lst.append(np.arange(length, dtype=np.int32))
    rows = np.concatenate(rows_lst)
    cols = np.concatenate(cols_lst)
    offsets = rows - cols                     # the diagonal index d = k‑j
    return rows, cols, offsets


def _solve_lp_increase_sum(f, M, rows, cols, offsets):
    """
    Linear programme:
        maximize   Σ_j g_j
        subject to A·g ≤ M·1 ,   g ≥ 0,
    where A is the convolution matrix built from `f`.
    The LP is solved with HiGHS (via scipy.linprog) on a sparse matrix.
    """
    n = len(f)
    data = f[offsets]                         # values for the non‑zero entries
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σ g  ⇒  maximise Σ g
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        # fall back to a uniform distribution – should almost never happen
        return np.full(n, 1.0 / n, dtype=np.float64)
    return res.x


def propose_candidate(seed=0, budget_s=950, length=2000):
    """
    LP‑based iterative optimisation of a non‑negative step function.

    Starts from the arcsine coefficients (inverse‑square‑root series),
    repeatedly solves a linear programme that maximises the ℓ1‑norm while
    keeping the convolution with the current vector bounded by its current
    peak, mixes the obtained vector back, projects onto the probability
    simplex and repeats until the time budget expires.  The best sequence
    found is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # ----- 1. initial arcsine vector, normalised to ℓ1‑norm = 1
    a = _arcsine_coeffs(length)
    p = a / np.sum(a)            # probability vector, sum(p) == 1
    best_seq = p.copy()
    best_score = _compute_score(best_seq.tolist())
    print(f"[init] length={length}, score={best_score:.6f}")

    rows, cols, offsets = _precompute_conv_structure(length)

    t = 0.2                       # mixing proportion (1‑t)*p + t*g
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----- current convolution peak
        n = len(p)
        L = 1 << ((2 * n - 1).bit_length())
        fft_p = np.fft.rfft(p, n=L)
        conv = np.fft.irfft(fft_p * fft_p, n=L)[: 2 * n - 1]
        M = np.max(conv)

        # ----- linear programme: enlarge total mass under the same bound M
        g = _solve_lp_increase_sum(p, M, rows, cols, offsets)

        # ----- normalise the LP solution to a probability vector
        sum_g = np.sum(g)
        if sum_g <= 0:
            g_norm = np.full_like(p, 1.0 / n, dtype=np.float64)
        else:
            g_norm = g / sum_g

        # ----- convex combination and projection onto the simplex
        p_new = (1.0 - t) * p + t * g_norm
        p_new = np.clip(p_new, 0.0, None)
        total = np.sum(p_new)
        if total == 0:
            p_new = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            p_new /= total

        # ----- evaluate
        cur_score = _compute_score(p_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p_new.copy()
            print(f"[iter {iteration}] improved score={best_score:.6f}  max={M:.6f}")

        p = p_new

        # ----- occasional tiny random perturbation to avoid stagnation
        if iteration % 7 == 0:
            noise = np.random.exponential(scale=1e-5, size=n)
            p = p + 0.02 * noise
            p = np.clip(p, 0.0, None)
            p /= np.sum(p)

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
