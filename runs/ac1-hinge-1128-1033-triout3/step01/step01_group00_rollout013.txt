    lo = max(0, k - (n - 1))

    j_range = k - i_range
    grad[i_range] = a[j_range]
    return grad

"""
Propose a step‑function using a sophisticated Adam‑based projected gradient
descent combined with frequent stochastic perturbations.  

The algorithm
  * starts from the best known seed (the provided `height_sequence_1`
    if it exists, otherwise a simple U‑shape);
  * repeatedly evaluates the self‑convolution via an FFT (O(n log n));
  * computes a sub‑gradient of the current maximal convolution entry;
  * takes an Adam‐style update on the simplex using a projection
    (`simplex_projection`);
  * occasionally performs random mass‑transfer moves and mixes in a fresh
    Dirichlet sample to escape local minima;
  * keeps the globally best vector found during the whole run.

All helper functions are defined at module level and avoid closures.
The routine respects the supplied time budget (≈1000 s) and returns a
list of floats ready for the external `evaluate_sequence` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w ≥ 0, sum(w) = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric U‑shape: a[i] ∝ (min(i, n‑1‑i)+1)^(-p).
    Returned vector is normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14           # avoid exact zero
    a /= a.sum()
    return a


def random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet sample with concentration `alpha` (scalar)."""
    raw = np.random.gamma(alpha, 1.0, size=n)
    a = raw / raw.sum()
    return a


def next_pow_two(x: int) -> int:
    """Small utility: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_max_and_index(a: np.ndarray, fft_size: int):
    """
    Compute max(self‑convolution) and the index where it occurs
    using an FFT of length `fft_size` (≥ 2·len(a)).
    """
    fx = np.fft.rfft(a, n=fft_size)
    conv = np.fft.irfft(fx * fx, n=fft_size)[:2 * len(a) - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def subgradient_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (at shift k)
    with respect to the vector a.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    i = np.arange(lo, hi + 1)
    j = k - i
    grad[i] = a[j]
    return grad


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` for a vector already normalised
    to sum = 1 (no clipping needed).  Returns 2·n·max(conv(a,a)).
    """
    n = len(a)
    max_c, _ = conv_max_and_index(a, next_pow_two(2 * n))
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# Core optimisation routine (projected Adam + stochastic moves)
# ----------------------------------------------------------------------


def local_optimize(
    a0: np.ndarray,
    time_budget: float,
    global_best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a projected Adam optimisation on `a0` for at most
    `time_budget` seconds.  Returns the best vector found and its score.
    """
    a = a0.copy()
    n = len(a)
    fft_size = next_pow_two(2 * n)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eta = 0.5                # base learning rate
    eps = 1e-12

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_sc = score_from_vector(a)

    t = 0
    prob_random = 0.20  # start with fairly aggressive random moves
    start = time.time()
    end = start + time_budget

    while time.time() < end:
        # ---------- convolution (FFT) ----------
        max_c, k_max = conv_max_and_index(a, fft_size)
        sc = 2.0 * n * max_c

        # keep best
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            # occasional progress output (once per improvement)
            print(f"Iter {t:7d}: new best = {best_sc:.6f}")

        # ---------- gradient step ----------
        grad = subgradient_max_conv(a, k_max) * (2.0 * n)   # d(2·n·max_c)/da

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # ---------- stochastic perturbations ----------
        if np.random.rand() < prob_random:
            # simple mass‑transfer move
            donors = np.where(a_new > 1e-12)[0]
            if donors.size > 1:
                i = np.random.choice(donors)
                j = np.random.randint(n)
                delta = a_new[i] * np.random.rand() * 0.30
                a_new[i] -= delta
                a_new[j] += delta

        if np.random.rand() < 0.02:
            # occasional mixing with a fresh Dirichlet sample
            alpha = np.random.uniform(0.3, 2.5)
            rand_vec = random_dirichlet(n, alpha)
            mix = np.random.rand() * 0.15
            a_new = (1.0 - mix) * a_new + mix * rand_vec

        # ---------- projection onto simplex ----------
        a = simplex_projection(a_new)

        # slowly anneal the random‑move probability and the learning rate
        prob_random *= 0.9995
        if t % 2000 == 0:
            eta *= 0.97

        # early exit if we have already beaten a target that is very hard to improve
        if best_sc < 1.48:          # tiny safety net – not required but avoids waste
            break

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises `evaluate_sequence`.
    The routine runs for at most ``budget_s`` seconds and returns the best
    vector (as a plain list of floats) found.
    """
    np.random.seed(seed)
    safety_margin = 5.0
    deadline = time.time() + max(0.0, budget_s - safety_margin)

    # --------------------------------------------------------------
    # 1️⃣  Initialise from the supplied construction if possible.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            if external.sum() == 0.0:
                external = np.random.rand(external.size)
            external /= external.sum()
            sc_ext = score_from_vector(external)
            print(f"External seed score = {sc_ext:.6f}")
            if sc_ext < best_score:
                best_score = sc_ext
                best_seq = external.copy()
    except NameError:
        pass

    # --------------------------------------------------------------
    # 2️⃣  If no external seed, fall back to a simple U‑shape.
    # --------------------------------------------------------------
    if best_seq is None:
        n0 = 5808
        p0 = 1.0
        best_seq = u_shaped_distribution(n0, p0)
        best_score = score_from_vector(best_seq)
        print(f"Fallback U‑shape (n={n0}, p={p0}) score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Quick exploratory seeds (tiny budget) to possibly beat the start.
    # --------------------------------------------------------------
    exploratory_lengths = [3000, 4000, 5000, 5600]
    powers = [0.7, 0.9, 1.1, 1.3, 1.5]

    for n in exploratory_lengths:
        for p in powers:
            if time.time() >= deadline:
                break
            cand = u_shaped_distribution(n, p)
            sc = score_from_vector(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"Exploratory U‑seed n={n}, p={p:.2f} → {sc:.6f}")

            # one random Dirichlet per length
            alpha = np.random.uniform(0.4, 2.0)
            cand = random_dirichlet(n, alpha)
            sc = score_from_vector(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"Exploratory Dirichlet n={n}, α={alpha:.2f} → {sc:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Main optimisation phase on the best seed found so far.
    # --------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    if remaining > 1.0:
        print(f"Starting main optimisation (≈{remaining:.1f}s left) on n={len(best_seq)}")
        best_seq, best_score = local_optimize(best_seq, remaining, best_score)

    print(f"Final best score = {best_score:.6f}  (length {len(best_seq)})")
    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
