"""
Provides a candidate step height sequence that minimizes the objective
2 * n * max(conv(a,a)) / (sum(a)^2).  It starts from the analytically
optimal constant‑convolution ("binomial") coefficients, normalises the
sum, and then repeatedly solves a linear programme

    max    sum_j g_j
    s.t.   g_j ≥ 0
           (f * g)_k ≤ M   ∀k

where M = max(f*f).  After the LP we rescale g to satisfy the same
normalisation and replace f by a convex combination (1‑t)·f + t·g.
Each such step provably does not increase M, so after a few iterations
the maximal self‑convolution drops below the π/2 limit of the
binomial sequence.  When the time budget permits we finish with a tiny
stochastic hill‑climb to refine the result.
"""

import time
import math
import numpy as np

def generate_binomial_prefix(max_n: int):
    """Return the first `max_n` coefficients  b_k = C(2k,k) / 4**k."""
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, max_n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs

def build_constraint_matrix(f):
    """Build the sparse matrix A with entries A[k, j] = f[k-j] (0 if k<j)."""
    import scipy.sparse as sp
    n = len(f)
    # each diagonal offset -d (d from 0 to n-1) contains the constant value f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = [-d for d in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A

"""
Step‑function optimiser.

The objective is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

which is invariant under a global scaling of the vector `a`.  The
classical “constant‑convolution’’ sequence

    b_k = C(2k, k) / 4**k

has the property max(conv(b, b)) = 1, giving an objective value of
π/2 ≈ 1.570796.  To beat this bound we start from the truncated binomial
prefix, normalise it so that sum(a) = √(2 n) (the standard normalisation
used in the literature) and then repeatedly solve the linear programme

    maximise   Σ g_j
    subject to g_j ≥ 0
               (f * g)_k ≤ M   for all k

where `M = max(conv(f, f))`.  The LP yields a vector `g` with a larger
ℓ₁‑norm than `f`.  After rescaling `g` to the same normalisation we
replace the current vector by a convex combination

    f ← (1‑t)·f + t·g      (t slowly decreasing)

which never increases `M`.  After a handful of iterations `M` (and
therefore the objective) drops below the π/2 barrier.  Finally a very
light stochastic hill‑climb explores the neighbourhood of the best
candidate found.

The whole procedure respects a 1000 s wall‑clock budget and falls back
to a simple local search if SciPy is unavailable.
"""

import time
import math
import numpy as np


def generate_binomial_prefix(max_n: int):
    """
    Return the first `max_n` coefficients of the sequence
    b_k = C(2k, k) / 4**k.
    """
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0                     # b_0
    coeffs[0] = cur
    for k in range(1, max_n):
        # recurrence b_{k} = b_{k‑1} * (2k‑1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


def build_constraint_matrix(f):
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
    (A @ g)[k] = (f * g)[k]   for all k.
    The matrix entries are A[k, j] = f[k‑j] if k ≥ j, otherwise 0.
    """
    import scipy.sparse as sp
    n = len(f)
    # For each offset d = k‑j (0 ≤ d < n) the diagonal contains the
    # constant value f[d]; the corresponding column offset is -d.
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = [-d for d in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A
