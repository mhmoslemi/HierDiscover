```
def max_conv_fft(a):
    n = len(a)
    size = 1 << (2*n-1).bit_length()   # next power of two >= 2*n-1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A*A, n=size)[:2*n-1]
    return float(conv.max())
```

```
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, math, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = np.inf

    # Optional: try using height_sequence_1 as seed
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size>0:
            external = np.maximum(external, 0.0)
            external_sum = external.sum()
            if external_sum < 1e-12:
                external = np.ones_like(external) / len(external)
                external_sum = 1.0
            external = external / external_sum
            conv_max = max_conv_fft(external)
            score = 2 * len(external) * conv_max
            if score < best_score:
                best_score = score; best_seq = external.copy()
                print(f"Using external initial seq, score={score}")
    except NameError:
        pass

    # Stage 1: grid search over length and exponent parameters
    lengths = [4000, 6000, 8000, 10000, 12000, 16000, 20000]
    p_vals = np.linspace(0.2, 2.0, 19) # steps .1

    for n in lengths:
        if time.time()>deadline: break
        # shape families
        for p in p_vals:
            if time.time()>deadline: break
            # shape min distance
            d = np.arange(n)
            d = np.minimum(d, n-1-d)
            a = (d+1)**(-p) + 1e-12
            a /= a.sum()
            score = 2 * n * max_conv_fft(a)
            if score < best_score:
                best_score = score; best_seq = a.copy()
                print(f"Found better: n={n}, p={p:.3f}, shape=min-dist, score={score:.6f}")
            # shape sum of tails
            i = np.arange(n)
            a2 = (i+1)**(-p) + (n - i)**(-p) + 1e-12
            a2 /= a2.sum()
            score2 = 2 * n * max_conv_fft(a2)
            if score2 < best_score:
                best_score = score2; best_seq = a2.copy()
                print(f"Found better: n={n}, p={p:.3f}, shape=sum-tails, score={score2:.6f}")

    # Stage 2: local stochastic refinement (load if we have time)
    if best_seq is None:
        # fallback random
        n = 6000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = 2 * n * max_conv_fft(a)
        print(f"Fallback random: n={n}, score={best_score:.6f}")

    # Local refinement
    current = best_seq.copy()
    n = len(current)
    current_score = best_score
    temperature = 0.5
    cooling = 0.999 # slower cooling
    it = 0

    # Precompute convolution indexing for gradient: can compute grad quickly.
    while time.time() < deadline:
        it += 1
        # decide move type: random transfer or multiplicative
        if np.random.rand() < 0.5:
            # random transfer
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] == 0:
                continue
            delta = current[i] * np.random.rand() * 0.3  # up to 30% of source
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # multiplicative gradient descent
            # compute conv via FFT
            size = 1 << (2*n-1).bit_length()
            A = np.fft.rfft(current, n=size)
            conv = np.fft.irfft(A*A, n=size)[:2*n-1]
            max_val = conv.max()
            kmax = int(conv.argmax())
            # gradient
            grad = np.zeros(n)
            low = max(0, kmax - (n-1))
            high = min(n-1, kmax)
            # i runs from low to high; j = kmax - i
            # grad[i] = 2 * current[j]
            idx_i = np.arange(low, high+1)
            idx_j = kmax - idx_i
            grad[idx_i] = 2.0 * current[idx_j]

            # multiplicative update
            eta = 0.2 * temperature   # step size scales with temperature
            cand = current * np.exp(-eta * grad)
            cand_sum = cand.sum()
            if cand_sum == 0:
                cand = current
            else:
                cand = cand / cand_sum
        # evaluate
        cand_score = 2 * n * max_conv_fft(cand)
        # acceptance
        if cand_score < current_score or np.random.rand() < np.exp((current_score - cand_score) / max(temperature, 1e-12)):
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Iter {it:6d}: new best {best_score:.6f}")
        # anneal
        temperature *= cooling
        if temperature < 1e-6:
            temperature = 1e-6

    print(f"Search finished, best score {best_score:.6f}, length {len(best_seq)}")
    return best_seq.tolist()
```

"""
Propose a step‑function that minimises the given evaluation metric.

The method consists of two phases:

1. **Fast parametric sweep** –  Several simple families of symmetric
   “U‑shaped’’ distributions (power‑law and sum‑of‑tails) are generated for
   a range of lengths and exponents.  The peak of the self‑convolution is
   evaluated with an FFT‑based O(n log n) routine, which is orders of
   magnitude faster than the naïve np.convolve used by the evaluator.
   The best candidate from this sweep is kept as a warm start.

2. **Stochastic refinement on the simplex** –  Starting from the best
   candidate, a simulated‑annealing loop performs two kinds of moves:

   * a random transfer of a fraction of mass between two positions, and
   * a multiplicative (mirror‑descent) step that follows the sub‑gradient
     of the current maximal convolution entry.

   Moves are accepted if they improve the score or with a probability
   that depends on a slowly cooling temperature, allowing the search to
   escape shallow local minima.  All vectors are kept non‑negative and
   renormalised to sum to one after each move.

The routine respects a time budget (default 1000 s) and returns the best
list of floats found.  A final call to the official ``evaluate_sequence``
is performed (once) to report the exact score that will be used for
scoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def max_conv_fft(a: np.ndarray) -> float:
    """
    Compute max entry of the self‑convolution a*a using real FFT.
    The vector a is assumed to be non‑negative and summed to 1, but
    the function works for any real vector.
    """
    n = a.shape[0]
    # next power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law U‑shape:
        a[i] = (min(i, n-1-i) + 1)^{-p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12           # avoid exact zeros
    a /= a.sum()
    return a


def sum_tails_distribution(n: int, p: float) -> np.ndarray:
    """
    Sum of two power‑law tails:
        a[i] = (i+1)^{-p} + (n-i)^{-p}
    Normalised to sum = 1.
    """
    i = np.arange(n)
    a = (i + 1) ** (-p) + (n - i) ** (-p) + 1e-12
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Exponential decay from the edges:
        a[i] = exp(-alpha * min(i, n-1-i))
    Normalised to sum = 1.
    """
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-alpha * d) + 1e-12
    a /= a.sum()
    return a

# Main search routine
# ----------------------------------------------------------------------

    """
    Search for a non‑negative step function minimising `evaluate_sequence`.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0.  Try to reuse the previously provided sequence, if any.
    # ------------------------------------------------------------------
    try:

        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s < 1e-12:
                external = np.ones_like(external) / external.size
                s = 1.0
            external = external / s
            score_ext = 2.0 * len(external) * max_conv_fft(external)
            if score_ext < best_score:
                best_score = score_ext
                best_seq = external.copy()
                print(f"Seeded from height_sequence_1: score={score_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Deterministic sweep over lengths & simple parametric families.
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000, 10000, 12000, 16000, 20000]
    p_vals = np.linspace(0.2, 2.0, 19)            # 0.2,0.3,…,2.0
    alphas = np.linspace(0.0002, 0.006, 10)      # for exponential shape

    print("=== Parametric sweep ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # ----- power‑law U‑shape -----
        for p in p_vals:
            if time.time() > deadline:
                break
            a = u_shaped_distribution(n, p)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=U‑pow, p={p:4.2f}, score={sc:.6f}")

        # ----- sum‑of‑tails -----
        for p in p_vals:
            if time.time() > deadline:
                break
            a = sum_tails_distribution(n, p)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=tails, p={p:4.2f}, score={sc:.6f}")

        # ----- exponential decay -----
        for a_alpha in alphas:
            if time.time() > deadline:
                break
            a = exponential_distribution(n, a_alpha)
            sc = 2.0 * n * max_conv_fft(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"New best: n={n:5d}, shape=exp, α={a_alpha:6.4f}, score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  If nothing succeeded, fall back to a random normalised vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8000
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a
        best_score = 2.0 * n * max_conv_fft(a)
        print(f"Fallback random init: n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Local stochastic refinement on the simplex.
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = len(current)
    current_score = best_score

    temperature = 0.7            # initial temperature for SA
    cooling_rate = 0.9995        # geometric cooling
    transfer_frac = 0.4          # max fraction of source mass to move
    grad_step_base = 0.15        # base step size for multiplicative update

    it = 0
    print("\n=== Stochastic refinement ===")
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Choose move type
        # --------------------------------------------------------------
        if np.random.rand() < 0.55:
            # ---- Random mass transfer between two distinct positions ----
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] <= 0.0:
                continue
            delta = current[i] * np.random.rand() * transfer_frac
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
            # renormalise only to guard against round‑off
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
        else:
            # ---- Multiplicative mirror‑descent step (sub‑gradient) ----
            # compute current convolution to locate the peak
            size = 1 << ((2 * n - 1).bit_length())
            A = np.fft.rfft(current, n=size)
            conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
            kmax = int(conv.argmax())

            # gradient of the peak entry w.r.t. a
            grad = np.zeros_like(current)
            i_start = max(0, kmax - (n - 1))
            i_end = min(n - 1, kmax)
            idx_i = np.arange(i_start, i_end + 1)
            idx_j = kmax - idx_i
            grad[idx_i] = 2.0 * current[idx_j]

            # multiplicative update (mirror descent on the simplex)
            eta = grad_step_base * temperature
            # clamp exponent to avoid overflow
            expo = -eta * grad
            expo = np.clip(expo, -50.0, 0.0)
            cand = current * np.exp(expo)
            s = cand.sum()
            if s > 0:
                cand /= s
            else:
                cand = current.copy()

        # --------------------------------------------------------------
        # Evaluate the candidate (fast FFT based evaluation)
        # --------------------------------------------------------------
        cand_score = 2.0 * n * max_conv_fft(cand)

        # --------------------------------------------------------------
        # Simulated‑annealing acceptance
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            # update global best
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"Iter {it:7d}: new global best = {best_score:.6f}")
                # early exit if we already beat the target
                if best_score < 1.503:
                    print("Target reached – exiting early.")
                    break

        # --------------------------------------------------------------
        # Anneal temperature
        # --------------------------------------------------------------
        temperature = max(temperature * cooling_rate, 1e-6)

    # ------------------------------------------------------------------
    # 4.  Final exact evaluation (once) and report
    # ------------------------------------------------------------------
    try:
        exact_score = evaluate_sequence(best_seq.tolist())   # noqa: F821
        print(f"\nExact evaluation by evaluate_sequence: {exact_score:.6f}")
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")
        exact_score = best_score

    print("\n=== Search finished ===")
    print(f"Best score (approx.) : {best_score:.6f}")
    print(f"Best length          : {len(best_seq)}")
    return best_seq.tolist()
