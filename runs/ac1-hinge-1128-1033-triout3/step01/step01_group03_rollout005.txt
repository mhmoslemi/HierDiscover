"""
Propose a step‑height sequence that improves on the classic
π/2 ≈ 1.5708 bound.

The algorithm starts from the analytically optimal “square‑root’’
sequence (coefficients of (1‑x)^−1/2) and then performs a
projected‑gradient type optimisation on a *smooth* approximation of
the max‑convolution objective:

    F(a) = max_k (a * a)_k   (with Σ a_i = 1, a_i ≥ 0).

Instead of the non‑differentiable max we use a soft‑max
( log‑sum‑exp ) with a temperature β that grows during the run.
The gradient of the smooth objective can be written as a
convolution, so each iteration costs O(n log n) using FFT.

Updates are performed with an exponentiated‑gradient (multiplicative)
step which automatically respects non‑negativity, followed by a
renormalisation onto the probability simplex.  Small random jitter
and occasional restarts avoid stagnation.

The routine respects a time budget (default 1000 s) and returns the
best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex {x ≥ 0, Σ x = 1} (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “square‑root’’ start (coefficients of (1‑x)^(-1/2))
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑x)^(-1/2) normalised to sum to 1.
    Recurrence: w_0 = 1,  w_{k+1} = w_k * (2k+1) / (2(k+1)).
    """
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2.0 * (k - 1) + 1.0) / (2.0 * k)
    w /= w.sum()                     # now Σ w = 1
    return w


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (function must return before this expires).
    **kwargs : dict
        Optional argument ``length`` to set the sequence size.

    Returns
    -------
    list[float]
        The best sequence found (elements are ≥ 0, sum ≈ 1).
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Settings
    # ------------------------------------------------------------------
    n = kwargs.get("length", 2000)          # length of the step function
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    # initialise with the analytically motivated shape
    a = _sqrt_series(n)                     # already normalised, Σ a = 1
    best_seq = a.copy()
    # direct evaluation (avoids an extra convolution)
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    print(f"[init] length={n}, initial evaluation={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the smooth‑max optimiser
    # ------------------------------------------------------------------
    beta = 0.5          # start temperature for the soft‑max
    beta_max = 50.0     # we never need an extremely sharp max
    beta_factor = 1.012 # multiplicative increase per iteration

    eta0 = 1.0          # initial multiplicative‑gradient step size
    decay = 2e-5        # slow decay of the step size

    iter_cnt = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # current convolution and its maximum
        conv = np.convolve(a, a)                     # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv               # Σ a = 1 ⇒ denominator = 1

        # record best solution so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{iter_cnt:5d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # smooth‑max weights  w_k ∝ exp(β·(conv_k − max_conv))
        # --------------------------------------------------------------
        # shift by max to keep numbers ≤ 0 and avoid overflow
        shifted = conv - max_conv                     # ≤ 0
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:          # numerical safety
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # gradient of the smooth objective:
        #   ∇F_i = 2 * Σ_k w_k * a_{k‑i}
        #   this is a convolution of w with the reversed a
        # --------------------------------------------------------------
        grad_full = np.convolve(w, a[::-1])           # length 3n‑2
        grad = grad_full[n - 1 : n - 1 + n]           # the n relevant entries
        grad *= 2.0

        # --------------------------------------------------------------
        # multiplicative (exponentiated) gradient step + renormalisation
        # --------------------------------------------------------------
        eta = eta0 / (1.0 + decay * iter_cnt)        # decreasing step size
        a = a * np.exp(-eta * grad)                  # shrink coordinates with large grad
        a_sum = a.sum()
        if a_sum == 0.0:                             # protective fallback

        else:

        # --------------------------------------------------------------

    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] best evaluation = {best_score:.6f} after {iter_cnt} iterations")
    return best_seq.tolist()
