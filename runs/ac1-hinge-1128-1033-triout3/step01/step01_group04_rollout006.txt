import numpy as np
import time
import cvxpy as cp

def _compute_score(seq):
    # as earlier
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def _arcsine_coeffs(n):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i-1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs

def _solve_lp(f, M):
    n = len(f)
    # Build constraint matrix A of shape (2n-1, n)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # For each column i, fill rows i .. i+n-1 with f
    # i.e., A[i+i_offset, i] = f[i_offset]
    # Use broadcasting:
    # However loops okay for n up to 1000
    for i in range(n):
        A[i:i+n, i] = f
    # Set up cvxpy problem
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M, g >= 0]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)

    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    if g.value is None:

"""
Propose a candidate step‑function.

The algorithm starts from the classic arcsine (inverse‑square‑root) coefficients,
which already give a modest bound (≈ π/2).  It then repeatedly solves a small
linear program:

    maximise   Σ g_j
    subject to (f * g)_k ≤ M   for every k
                g ≥ 0

where f is the current sequence, M = max(f*f) and * denotes convolution.
Because the constraints are linear in g, the LP can be solved quickly with
CVXPY.  The resulting g carries as much total mass as possible without
raising the cross‑convolution bound M.  Mixing f and g (f ← (1‑t)f + t g)
increases the ℓ¹‑norm of the sequence while keeping the maximal convolution
value close to M, thereby reducing the evaluation metric
    2 n·max / (∑a)² .
The process is repeated until the time budget is exhausted.  Finally a short
random “mass‑transfer’’ refinement is applied.

The whole routine stays inside the given time limit and returns the best
sequence found.
"""

import time
import numpy as np
import cvxpy as cp


def _compute_score(seq):
    """Fast evaluation of the objective (same as evaluate_sequence)."""
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _arcsine_coeffs(n):
    """First n coefficients of (1‑z)^‑½ = Σ binom(2i,i)/4ⁱ zⁱ."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _conv_and_max(a):
    """Return (full) convolution a*a and its maximum entry."""
    n = len(a)
    L = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return conv, np.max(conv)


def _solve_lp(f, M):
    """
    Solve  max Σ g_j   s.t.  (f * g)_k ≤ M  (for all k) ,  g ≥ 0.
    Returns the optimal g as a NumPy array (may be all‑zeros if infeasible).
    """
    n = len(f)
    rows = 2 * n - 1

    # Build the Toeplitz‑like constraint matrix A such that A@g = f*g.
    # Column i contains f shifted by i rows.
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    # CVXPY model
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]          # each row: (f*g)_k ≤ M
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                    abstol=1e-8, reltol=1e-6, feastol=1e-8, verbose=False)
    except Exception:   # fallback if solver chokes
        return np.zeros(n, dtype=np.float64)

    if g.value is None:
        # In rare cases ECOS may fail – return a zero vector.
        return np.zeros(n, dtype=np.float64)
    return np.asarray(g.value, dtype=np.float64)


def propose_candidate(seed=0, budget_s=950.0, length=800):
    """
    Search for a non‑negative step function that minimises the supplied
    `evaluate_sequence` metric.

    Parameters
    ----------
    seed : int, optional
        Random seed for any stochastic parts.
    budget_s : float, optional
        Time budget (seconds).  The implementation stops a little before the limit.
    length : int, optional
        Length of the initial arcsine series.  600–1200 works well; larger values
        increase LP size and run time.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. starting point – either the global `height_sequence_1` if present,
    #    otherwise the arcsine series.
    # ----------------------------------------------------------------------
    try:
        init = list(height_sequence_1)          # noqa: F821  (may be undefined)
        if not isinstance(init, list) or not init:
            raise Exception
        f = np.array(init, dtype=np.float64)
        if len(f) != length:
            # truncate or pad with zeros to the requested length
            if len(f) > length:
                f = f[:length]
            else:
                f = np.concatenate([f, np.zeros(length - len(f), dtype=np.float64)])
    except Exception:
        f = _arcsine_coeffs(length)

    n = len(f)
    best_seq = f.copy()
    best_score = _compute_score(best_seq)
    print(f"[init] length={n}, start_score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2. Main optimisation loop – LP + mixing.
    # ----------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # safety margin
    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution data
        conv_f, M = _conv_and_max(f)
        cur_score = 2.0 * n * M / (f.sum() ** 2)

        # solve LP to find a mass‑rich partner g
        g = _solve_lp(f, M)
        sum_g = g.sum()
        if sum_g <= f.sum() + 1e-12:   # no improvement possible
            break

        # try a few mixing proportions, keep the best improvement
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = (1.0 - t) * f + t * g
            cand_score = _compute_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                f = cand                     # continue from this point
                improved = True
                print(f"[iter {it}] t={t:.3f}  score={best_score:.6f}")
                break
        if not improved:
            # LP did not give a useful direction – fall back to a cheap random move
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            delta = np.random.exponential(scale=1e-3) * f[i]
            if delta <= 0:
                continue
            new_f = f.copy()
            new_f[i] = max(0.0, new_f[i] - delta)
            new_f[j] += delta
            new_score = _compute_score(new_f)
            if new_score < best_score:
                best_score = new_score
                best_seq = new_f.copy()
                f = new_f
                print(f"[iter {it}] random move  score={best_score:.6f}")

        # stop after a reasonable number of LP solves to stay within the budget
        if it >= 30:
            # we have spent the bulk of the time; exit the intensive phase
            break

    # ----------------------------------------------------------------------
    # 3. Optional cheap stochastic refinement (mass‑transfer moves)
    # ----------------------------------------------------------------------
    end_deadline = time.time() + budget_s - 0.5
    while time.time() < end_deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if f[i] <= 0.0:
            continue
        delta = np.random.exponential(scale=5e-4) * f[i]
        new_f = f.copy()
        new_f[i] -= delta
        new_f[j] += delta
        new_score = _compute_score(new_f)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_f.copy()
            f = new_f
            # occasional progress report
            if it % 100 == 0:
                print(f"[refine] score={best_score:.6f}")
        it += 1

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
