import time
import numpy as np

def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Return full linear convolution of seq with itself using FFT."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv

def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of self‑convolution."""
    return float(_conv_full(seq).max())

def _score(seq: np.ndarray) -> float:
    """Exact objective value for a candidate sequence."""
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = seq.shape[0]
    max_conv = _conv_max(seq)
    return 2.0 * n * max_conv / (total * total)

    evaluation function.

    The method consists of two phases:
      1. A fairly dense grid search over a family of sequences built from
         a power‑law “core’’ a_i ∝ (i+1)^{‑α} followed by a geometric tail.
         The grid varies α, the core length, the tail height and the geometric
         decay factor and keeps the best candidate.
      2. A simulated‑annealing refinement that mixes small pairwise transfers
         with targeted moves that reduce the currently maximal convolution
         coefficient.

    The search stops after ``budget_s`` seconds and returns the best sequence
    found as a plain Python list.

import time

# ----------------------------------------------------------------------
# Helper functions (FFT based convolution, scoring, etc.)
# ----------------------------------------------------------------------
def _conv_full(seq: np.ndarray) -> np.ndarray:
    """
    Linear convolution of ``seq`` with itself using an FFT of power‑of‑two size.
    Returns the full array of length ``2*len(seq)-1``.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of ``seq``."""
    return float(_conv_full(seq).max())

    Exact value of the evaluation function for a given sequence.
    The function is scale‑invariant, so we can work with the raw numbers.
    """

        return np.inf
    n = seq.shape[0]
    max_conv = _conv_max(seq)
    return 2.0 * n * max_conv / (total * total)

            new_seq[j] += amount
        else:                                 # targeted move that attacks the maximal conv entry
            conv_full = _conv_full(cur_seq)
            max_k     = int(np.argmax(conv_full))

            # indices that participate in conv[max_k]
            i_min = max(0, max_k - (n - 1))
            i_max = min(n - 1, max_k)
            if i_min >= i_max:
                continue
            idxs      = np.arange(i_min, i_max + 1)
            partners  = max_k - idxs
            contribs  = cur_seq[idxs] * cur_seq[partners]

            donor_idx = idxs[np.argmax(contribs)]
            recipient_idx = int(np.argmin(cur_seq))

            if cur_seq[donor_idx] <= 0.0:
                continue
            # move up to 5 % of the donor entry
            amount = cur_seq[donor_idx] * np.random.rand() * 0.05
            new_seq = cur_seq.copy()
            new_seq[donor_idx] -= amount
            new_seq[recipient_idx] += amount

        # Safety: clip tiny negatives that might appear because of rounding
        new_seq = np.maximum(new_seq, 0.0)

        new_val = _score(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            cur_seq, cur_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
                print(f"[SA] new best {best_val:.6f}")

        # Linear cooling (temperature goes to zero at the end of the budget)
        elapsed = time.time() - start
        T = max(1e-8, T * (1.0 - elapsed / budget_s))

    print(f"final best value: {best_val:.6f}")
    return best_seq.tolist()
