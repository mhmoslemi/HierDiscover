"""
Propose a candidate step‑function sequence.

The algorithm starts from the analytically optimal “square‑root” (1‑x)^{-1/2}
coefficients (the best known construction giving the π/2 ≈ 1.5708 bound).
It then refines the shape using a smoothed version of the
max‑convolution objective and Adam‑type projected gradient descent.

Key ideas
---------
*   The true objective is   F(a)=2·n·max_k (a∗a)_k / (∑a)^2.
    After fixing ∑a=1 the factor 2·n·max_k (a∗a)_k is to be minimised.
*   The non‑smooth max is replaced by a smooth “log‑sum‑exp” surrogate

        F_β(a) = log( Σ_k exp(β·(a∗a)_k) ) / β ,

    whose gradient is a weighted sum of the individual convolution gradients.
    The weights are a soft‑max over the convolution values and sum to 1.
*   The gradient of a single convolution entry (a∗a)_k w.r.t. a is
    2·a_{k‑i} (zero when the index is out of range).  Hence the full gradient
    can be obtained by a correlation of the current sequence with the weight
    vector – efficiently computed with FFTs.
*   After each (Adam) update the vector is projected onto the probability
    simplex (non‑negative, sum = 1) using the O(n) algorithm of Duchi et al.
*   β is increased linearly from a small value (smooth landscape) to a large
    value (focus on the true maximum).  Small random jitters are added
    occasionally to escape plateaus.

The routine runs for at most the supplied time budget (default 1000 s) and
keeps the best sequence found.  It returns a plain list of floats,
ready to be fed into ``evaluate_sequence``.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex
# (Duchi et al., 2008)
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Initialise the “square‑root” sequence (coefficients of (1‑x)^‑1/2)
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """
    First n coefficients of (1‑x)^{-1/2}:
        w_0 = 1,
        w_{k+1} = w_k * (2k+1) / (2(k+1)).
    Normalised so that the sum equals 1.
    """
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# --------------------------------------------------------------
# Main search routine
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  Returns the best sequence found within the
    time budget.

    Parameters
    ----------
    seed : int
        Random seed for stochastic parts.
    budget_s : int
        Time budget in seconds.
    **kwargs : dict
        Optional ``length`` key to set the sequence length (default 5808).

    Returns
    -------
    list[float]
        The best sequence (elements sum to ≈1, all non‑negative).
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Settings
    # ------------------------------------------------------------------
    n = int(kwargs.get("length", 5808))          # length of the step function
    deadline = time.time() + budget_s - 0.5      # safety margin
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.12               # base learning rate
    lr_decay = 2e-5          # linear decay of the learning rate
    # Soft‑max temperature schedule
    beta_start, beta_end = 5.0, 200.0
    beta_ramp_iters = 25000   # after this many iters beta stops increasing

    # ------------------------------------------------------------------
    # FFT bookkeeping (fixed length for the whole run)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1                 # length of a∗a
    rev_len = n
    total_len = rev_len + conv_len - 1   # length of correlation (a_rev * w)
    nfft = 1 << (total_len - 1).bit_length()   # next power of two
    slice_start = n - 1
    slice_end = slice_start + n

    # ------------------------------------------------------------------
    # Initialise with the analytic square‑root shape
    # ------------------------------------------------------------------
    a = _sqrt_series(n)                 # already normalised, sum = 1
    m = np.zeros_like(a)                # first moment (Adam)
    v = np.zeros_like(a)                # second moment (Adam)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial evaluation={best_score:.6f}")

    it = 0
    no_improve = 0

    while time.time() < deadline:

        a_fft = np.fft.rfft(a, n=nfft)
        conv_full = np.fft.irfft(a_fft * a_fft, n=nfft)

        m_hat = m / (1.0 - beta1 ** t)

            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{it:5d}] restart triggered")

        it += 1

    # final verification (in case the very last candidate is best)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best evaluation = {final_score:.6f} after {it} iterations")
    return best_seq.tolist()
