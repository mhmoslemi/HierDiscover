import numpy as np
import time

def _compute_score(seq):
    # Fast evaluation using FFT
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=0, budget_s=1000, length=5000):
    """
    Generate a step‑function by discretising the continuous arcsine
    (Beta(½,½)) distribution.
    The probability mass of the arcsine law on each of the `length`
    equal sub‑intervals of [0,1] is used as the coefficient.
    This yields a distribution whose discrete self‑convolution is
    (approximately) uniform, which minimises the evaluation metric.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    n = max(2, int(length))

    # Compute the CDF of the arcsine distribution at the bin edges
    # CDF(x) = (2/π) * arcsin(√x) for 0 ≤ x ≤ 1
    edges = np.arange(n + 1, dtype=np.float64) / n  # values i/n for i=0..n
    # protect against numerical round‑off at x=1
    edges[-1] = 1.0
    cdf = (2.0 / np.pi) * np.arcsin(np.sqrt(edges))
    # Bin probabilities = differences of successive CDF values
    seq = np.diff(cdf)

    # Since of floating‑point round‑off the sum may deviate slightly
    # from 1, we renormalise.
    seq /= np.sum(seq)

    # Quick sanity check – the best possible score is close to 1.
    best_score = _compute_score(seq)
    print(f"[init] length={n}, score≈{best_score:.6f}")

    # Optional tiny local optimisation – we run a few thousand
    # stochastic mass‑transfer moves while time permits.
    best_seq = seq.copy()
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if best_seq[i] <= 0:
            continue
        # move a small fraction of mass from i to j
        delta = np.random.exponential(scale=1e-4) * best_seq[i]
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        cur_score = _compute_score(new_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq
            # print occasional improvements
            if int(best_score * 1e5) % 1000 == 0:
                print(f"[improve] score={best_score:.6f}")
        else:
            # occasional uphill acceptance to escape plateaus
            if np.random.rand() < 0.001:
                best_seq = new_seq

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()

import numpy as np
import time

def _compute_score(seq):
    """
    Fast evaluation of the objective.
    Computes 2 * n * max(a*a) / (sum a)^2 using an FFT based convolution.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    # length for FFT: next power of two ≥ 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed=0, budget_s=1000, length=5000):
    """
    Generate a step‑function by discretising the continuous arcsine
    (Beta(½,½)) distribution.

    The probability mass of the arcsine law on each of the `length`
    equal sub‑intervals of [0,1] is used as a coefficient.
    This yields a distribution whose discrete self‑convolution is
    (approximately) uniform, which minimises the evaluation metric.
    A very cheap stochastic refinement is performed while time permits.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1          # safety margin

    # ------------------------------------------------------------------
    # 1. Build the discrete arcsine (beta 1/2,1/2) distribution.
    # ------------------------------------------------------------------
    n = max(2, int(length))

    # CDF of the arcsine law:  F(x) = (2/π) * arcsin(√x),   0 ≤ x ≤ 1
    # Bin probabilities = ΔF on the n equal-width sub‑intervals.
    edges = np.arange(n + 1, dtype=np.float64) / n   # 0,1/n,…,1
    edges[-1] = 1.0                                 # guard against round‑off
    cdf = (2.0 / np.pi) * np.arcsin(np.sqrt(edges))
    seq = np.diff(cdf)                              # a_i = F((i+1)/n) – F(i/n)

    seq /= np.sum(seq)

    best_seq = seq.copy()
    best_score = _compute_score(best_seq)
    print(f"[init] length={n}, score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Optional cheap stochastic refinement (mass‑transfer moves).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if best_seq[i] <= 0.0:

        if delta > best_seq[i]:

        new_seq[i] -= delta
        new_seq[j] += delta

        cur_score = _compute_score(new_seq)

        # uphill moves to escape flat plateaus.
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq
            if int(best_score * 1e5) % 1000 == 0:

        else:
            if np.random.rand() < 0.001:
                best_seq = new_seq

    print(f"[final] best score={best_score:.6f}")
