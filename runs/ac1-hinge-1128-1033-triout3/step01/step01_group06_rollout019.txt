def targeted_move_improved(a, conv):
    n = a.shape[0]
    k = int(np.argmax(conv))
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_candidates = np.arange(i_min, i_max + 1, dtype=int)
    j_candidates = k - i_candidates
    contrib = a[i_candidates] * a[j_candidates]
    src = int(i_candidates[np.argmax(contrib)])

        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1
    else:
        dst = int(np.random.choice(dst_candidates))
    return src, dst

def lp_refine(a, conv, cur_val, best_val, best_a, n):
    M = np.max(conv)
    # Build A_ub
    # Use dense
    m = 2 * n - 1
    A_ub = np.zeros((m, n))
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # rows: for each i in [i_min, i_max], b_{k-i} column = i?
        # Actually coefficient for column j = a[k-j] if 0 <= k-j < n.
        # So we can loop over i = i_min..i_max: set j = k - i.
        for i in range(i_min, i_max + 1):
            j = k - i
            A_ub[k, j] = a[i]
    # Solve LP
    c = -np.ones(n)  # maximize sum(b)
    bounds = [(0, None)] * n
    b_vec = np.full(m, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_vec, bounds=bounds, method='highs')
    if not res.success:
        return a, conv, cur_val, best_val, best_a
    b = np.maximum(res.x, 0)
    if b.sum() <= 0:
        return a, conv, cur_val, best_val, best_a
    b_scaled = b / b.sum()  # sum to 1
    # mix
    t_mix = 0.2
    a_new = (1 - t_mix) * a + t_mix * b_scaled
    # recompute conv
    conv_new = np.convolve(a_new, a_new)
    new_val = 2.0 * n * np.max(conv_new)
    # accept if better
    if new_val < cur_val or new_val < best_val:
        a = a_new
        conv = conv_new
        cur_val = new_val
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[LP] new best {best_val:.6f}")
    return a, conv, cur_val, best_val, best_a

Algorithm
---------
1.  Use several candidate lengths (including the length of the best known
    sequence if available).  For each length a short simulated‑annealing

    executed (if SciPy is present).  The LP maximises Σb_j subject to
    b≥0 and (a*b)≤M where M is the current maximal self‑convolution.
    The solution `b` is normalised and mixed into the current vector,
    guaranteeing a non‑increasing bound.

6.  After the time budget for the current length is exhausted the best
    vector is returned.  The global best across all lengths is finally
    symmetrised (a←(a+a_rev)/2) and re‑normalised.

The function `propose_candidate` conforms to the required signature and
returns the best sequence found within the 1000 s budget.  It uses only
NumPy, SciPy (optional) and the provided `evaluate_sequence` for the
final check.
"""

import numpy as np
import time
import math
from typing import List

# optional SciPy import – used for occasional LP refinement
try:
    from scipy.optimize import linprog
    from scipy.sparse import coo_matrix
    _SCIPY_AVAILABLE = True
except Exception:       # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Fast convolution update when moving a mass δ from index i to j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current probability vector (sum = 1).
    conv : ndarray, shape (2*n-1,)
        Linear convolution a*a.
    i, j : int
        Distinct indices, 0 ≤ i, j < n.
    delta : float
        0 < delta ≤ a[i].

    Returns
    -------
    a_new, conv_new : ndarrays
        Updated vector and its convolution.
    """
    n = a.shape[0]
    di = -delta
    dj = +delta

    a_new = a.copy()
    a_new[i] = a[i] - delta
    a_new[j] = a[j] + delta

    conv_new = conv.copy()

    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]                # all other positions
    a_q = a[q]

    # contributions of the changed entries to all sums
    conv_new[i + q] += 2.0 * di * a_q    # (i, q) and (q, i)
    conv_new[j + q] += 2.0 * dj * a_q    # (j, q) and (q, j)

    # self‑pair contributions (i,i) and (j,j)
    conv_new[2 * i] += 2.0 * di * a[i] + di * di
    conv_new[2 * j] += 2.0 * dj * a[j] + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a[i] * dj + a[j] * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Targeted move: source is the biggest contributor to the current peak,
# destination is one of the smallest entries.
# ----------------------------------------------------------------------
def targeted_move(a: np.ndarray, conv: np.ndarray):
    n = a.shape[0]
    k = int(np.argmax(conv))                     # index of the current peak

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_candidates = np.arange(i_min, i_max + 1, dtype=int)
    j_candidates = k - i_candidates

    contrib = a[i_candidates] * a[j_candidates]
    src = int(i_candidates[np.argmax(contrib)])

    # pick a destination among the five smallest elements (excluding src)
    low_idx = np.argsort(a)[:5]
    low_idx = low_idx[low_idx != src]
    if low_idx.size == 0:
        # fallback to a random different index
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1
    else:
        dst = int(np.random.choice(low_idx))

    return src, dst


# ----------------------------------------------------------------------
# Tiny mixing step with the uniform distribution (helps escape plateaus)
# ----------------------------------------------------------------------
def uniform_mix_step(a: np.ndarray, n: int, temperature: float):
    """Blend a tiny fraction of the uniform vector into a."""
    # mixing strength grows a little when the temperature is high
    alpha = 0.001 * (1.0 + 5.0 * temperature)
    u = np.full(n, 1.0 / n)
    a_new = (1.0 - alpha) * a + alpha * u
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    conv_new = np.convolve(a_new, a_new)
    new_val = 2.0 * n * np.max(conv_new)
    return a_new, conv_new, new_val


# ----------------------------------------------------------------------
# Build the constraint matrix for the LP:  (a * b)[k] ≤ M   for all k
# ----------------------------------------------------------------------
def build_A_ub(a: np.ndarray):
    """Return a sparse matrix A (shape (2n‑1, n)) such that A @ b = a*b."""
    n = a.shape[0]
    m = 2 * n - 1
    rows = []
    cols = []
    data = []
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = coo_matrix((data, (rows, cols)), shape=(m, n))
    return A.tocsr()


# ----------------------------------------------------------------------
# Single LP refinement step (executed only occasionally)
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray, conv: np.ndarray, cur_val: float,
              best_val: float, best_a: np.ndarray, n: int):
    """Solve max Σb  s.t. b≥0, a*b ≤ M, mix the result into a."""
    if not _SCIPY_AVAILABLE:
        return a, conv, cur_val, best_val, best_a

    M = np.max(conv)
    A_ub = build_A_ub(a)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)               # maximize Σb  → minimize –Σb
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:            # pragma: no cover
        return a, conv, cur_val, best_val, best_a

    if not res.success:
        return a, conv, cur_val, best_val, best_a

    b = np.maximum(res.x, 0.0)
    if b.sum() == 0.0:
        return a, conv, cur_val, best_val, best_a

    # normalise b to have the same total mass as a (which is 1)
    b_norm = b / b.sum()

    # convex combination – the mixing factor can be tuned;
    # 0.2 works well in practice.
    t = 0.2
    a_new = (1.0 - t) * a + t * b_norm
    conv_new = np.convolve(a_new, a_new)
    new_val = 2.0 * n * np.max(conv_new)

    # accept the refined point if it improves (or even if not – it is
    # a cheap exploration step)
    if new_val < cur_val:
        a, conv, cur_val = a_new, conv_new, new_val
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[LP] new best value = {best_val:.6f}")
    return a, conv, cur_val, best_val, best_a


# ----------------------------------------------------------------------
# Final deterministic local improvement (greedy redistribution)
# ----------------------------------------------------------------------
def final_local_refine(a: np.ndarray, n: int, max_iter: int = 2000):
    """A few greedy moves that always lower the objective."""
    conv = np.convolve(a, a)
    cur_val = 2.0 * n * np.max(conv)

    for _ in range(max_iter):
        k = int(np.argmax(conv))
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_cand = np.arange(i_min, i_max + 1, dtype=int)
        j_cand = k - i_cand
        contrib = a[i_cand] * a[j_cand]
        src = int(i_cand[np.argmax(contrib)])
        dst = int(np.argmin(a))
        if src == dst or a[src] < 1e-15:
            break
        delta = a[src] * 0.5          # move at most half of the source mass
        a, conv = apply_move(a, conv, src, dst, delta)[:2]
        cur_val = 2.0 * n * np.max(conv)
    return a


# ----------------------------------------------------------------------
# Symmetrise a vector (a_i = a_{n-1-i}) and re‑normalise.
# ----------------------------------------------------------------------
def symmetrise(a: np.ndarray):
    a_sym = (a + a[::-1]) * 0.5
    a_sym /= a_sym.sum()
    return a_sym


# ----------------------------------------------------------------------
# Search for a single length
# ----------------------------------------------------------------------
def run_search_for_length(n: int, budget: float, rng_seed: int):
    """Return (best_sequence, best_value) for the given length."""
    np.random.seed(rng_seed)
    start = time.time()
    deadline = start + budget - 1.0   # safety margin

    # ---- initial distribution -------------------------------------------------
    # edge‑heavy shape – exponent p is randomised to increase diversity
    p = np.random.uniform(0.5, 2.0)
    x = np.arange(n, dtype=float)
    base = 1.0 - np.sin(np.pi * x / (n - 1))
    a = np.power(base, p)
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= tot                     # normalise to sum = 1

    # If a previously found global sequence of the same length exists,
    # start from it (it is already normalised).
    global height_sequence_1
    try:
        if len(height_sequence_1) == n:   # type: ignore[name-defined]
            a = np.array(height_sequence_1, dtype=float)
            a = np.maximum(a, 0.0)
            a /= a.sum()
    except Exception:
        pass

    conv = np.convolve(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val

    # ---- annealing schedule ----------------------------------------------------
    T_start = 0.30
    T_end = 1e-4
    it = 0
    accepted = 0

    while time.time() < deadline:
        now = time.time()
        progress = (now - start) / (deadline - start)
        # exponential temperature decay
        T = T_start * (T_end / T_start) ** progress

        r = np.random.rand()
        if r < 0.60:                      # targeted move
            src, dst = targeted_move(a, conv)
        elif r < 0.90:                    # random move
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
        else:                             # tiny uniform mixing
            a, conv, new_val = uniform_mix_step(a, n, T)
            if new_val < cur_val:
                cur_val = new_val
                accepted += 1
                if new_val < best_val:
                    best_val = new_val
                    best_a = a.copy()
                    print(f"[n={n} it={it}] new best (mix) = {best_val:.6f}")
            it += 1
            continue

        # magnitude of the move – a random fraction of the source mass
        max_delta = a[src] * np.random.uniform(0.3, 0.7)
        if max_delta < 1e-15:
            continue
        delta = np.random.rand() * max_delta

        a_new, conv_new = apply_move(a, conv, src, dst, delta)
        new_val = 2.0 * n * np.max(conv_new)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a, conv = a_new, conv_new
            cur_val = new_val
            accepted += 1
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[n={n} it={it}] new best = {best_val:.6f}")

        it += 1

        # occasional exact recomputation (prevents drifting errors)
        if it % 5000 == 0:
            conv = np.convolve(a, a)

        # occasional LP refinement – cheap but powerful
        if _SCIPY_AVAILABLE and it % 50000 == 0:
            a, conv, cur_val, best_val, best_a = lp_refine(
                a, conv, cur_val, best_val, best_a, n
            )

    # final greedy cleanup
    a = final_local_refine(best_a, n, max_iter=2000)

    # symmetrisation (the objective is symmetric, this never hurts)
    a = symmetrise(a)

    conv = np.convolve(a, a)
    final_val = 2.0 * n * np.max(conv)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
        print(f"[n={n}] post‑symmetrisation best = {best_val:.6f}")

    return best_a.tolist(), best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided evaluation function.  The routine uses several independent
    simulated‑annealing runs (different lengths, different random seeds)
    and occasional linear‑programming refinements.  The best sequence
    discovered before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # leave a small safety margin

    # candidate lengths – include the length of the previously known best
    n_candidates = [1500, 1742, 2000, 2500, 3000, 3500, 4000]
    try:
        n_candidates.append(len(height_sequence_1))   # type: ignore[name-defined]
    except Exception:
        pass
    n_candidates = sorted(set(n_candidates))

    best_global_seq = None
    best_global_val = float('inf')

    # allocate roughly equal budget to each length (remaining time is shared)
    for idx, n in enumerate(n_candidates):
        now = time.time()
        if now >= deadline - 20:  # keep a tiny buffer for final checks
            break
        remaining = deadline - now
        # simple equal‑share; later lengths get the leftover proportionally
        per_len_budget = remaining / (len(n_candidates) - idx)

        print(f"[main] Starting optimisation for n={n} with budget≈{per_len_budget:.1f}s")
        seq, val = run_search_for_length(n, per_len_budget, seed + n)
        print(f"[main] Finished n={n}: best evaluation = {val:.6f}")

        if val < best_global_val:
            best_global_val = val
            best_global_seq = seq
            print(f"[main] New global best (n={n}) = {best_global_val:.6f}")

    # final sanity check with the official evaluator
    try:
        final_score = evaluate_sequence(best_global_seq)
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_global_seq
