    best_f = f.copy()

    print initial.

    Then while time < deadline:

        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # solve LP
        T = build_T(f)
        c = -np.ones(n)

"""
Propose a candidate step‑function using an LP‑based alternating improvement.

The algorithm works with a probability vector `p` (non‑negative, sum = 1).
For a given `p` we compute M = max_k (p * p)_k.
We then solve the linear program

    maximise   Σ_j b_j
    subject to  (p * b)_k ≤ M   for all k = 0 .. 2n‑2
                b_j ≥ 0

(Here (p * b)_k = Σ_i p_i·b_{k‑i} is the ordinary convolution.)
The LP finds a distribution `b` that has the largest total mass while
still respecting the current peak `M`.  After normalising `b` to sum to 1
we mix it with the current vector:  p ← (1‑t)·p + t·b .
With a small step size `t` this reduces the maximal convolution entry,
hence reduces the evaluation function

    score(p) = 2·n·max(conv(p,p))

(the denominator of `evaluate_sequence` disappears because Σp = 1).

We repeat the LP / mixing loop until the time budget expires or no
improvement is observed for a while, performing several random restarts
to avoid local minima.  The best vector found is returned as a plain
list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: build the Toeplitz constraint matrix for a given vector ``p``.
# The matrix A has shape (2n‑1, n) and satisfies (p * b) = A @ b.
# ----------------------------------------------------------------------
def _build_toeplitz(p: np.ndarray) -> np.ndarray:
    n = p.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=float)
    # column j receives p placed from row j up to row j+n‑1
    for j in range(n):
        A[j:j + n, j] = p
    return A


# ----------------------------------------------------------------------
# Helper: evaluate the objective for a probability vector (sum = 1).
# The returned value is exactly what ``evaluate_sequence`` would produce.
# ----------------------------------------------------------------------
def _objective(p: np.ndarray) -> float:
    n = p.shape[0]
    conv = np.convolve(p, p)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the competition harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose the problem size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 1153                       # default length used by the benchmark

    # ------------------------------------------------------------------
    # 2. Global bookkeeping.
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_vec = None
    total_iters = 0
    restart = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation – many short random restarts.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        restart += 1

        # ----- initialise a feasible probability vector -----------------
        if "height_sequence_1" in globals() and rng.random() < 0.25:
            p = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            p = np.clip(p, 0.0, 1000.0)
            if p.sum() == 0.0:
                p = np.full(n, 1.0 / n)
            else:
                p = p / p.sum()
        else:
            # Dirichlet‑like random start (all entries positive)
            p = rng.random(n)
            p = p / p.sum()

        cur_score = _objective(p)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = p.copy()
            print(f"Restart {restart:3d} – initial score {cur_score:.6f}")

        # ----- inner loop – LP improvement steps -----------------------
        no_improve = 0                # stop inner loop after a while without gain
        inner_it = 0
        while time.time() < deadline:
            inner_it += 1
            total_iters += 1

            # current convolution peak
            conv = np.convolve(p, p)
            M = conv.max()
            cur_score = 2.0 * n * M

            # stop early if we have already reached the target
            if cur_score <= 1.503:
                best_score = cur_score
                best_vec = p.copy()
                print(f"Target reached (score {cur_score:.6f}) after {total_iters} iters "
                      f"(restart {restart}, inner {inner_it})")
                return best_vec.tolist()

            # ----- build LP constraints --------------------------------
            A = _build_toeplitz(p)                 # shape (2n‑1, n)
            b_ub = np.full(2 * n - 1, M)

            # ----- solve maximise Σ b_j subject to (p*b) ≤ M ----------
            #  (i.e. minimise –Σ b_j)
            c = -np.ones(n)                         # linear objective
            bounds = [(0.0, 1000.0) for _ in range(n)]

            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "disp": False},
            )

            if not res.success:
                # LP failed – give up this restart
                print(f"  LP failed (restart {restart}, inner {inner_it}): {res.message}")
                break

            b_opt = res.x
            sum_b = b_opt.sum()
            if sum_b <= 0.0:
                break

            # ----- scale to keep sum = 1 --------------------------------
            scale = 1.0 / sum_b
            g = b_opt * scale                       # also a probability vector

            # ----- try a few mixing parameters -------------------------
            improved = False
            for t in (0.5, 0.3, 0.2, 0.1, 0.05):
                p_new = (1.0 - t) * p + t * g
                # Numerical drift may slightly change the sum – renormalise.
                p_new = p_new / p_new.sum()
                new_score = _objective(p_new)
                if new_score < cur_score - 1e-9:
                    p = p_new
                    cur_score = new_score
                    improved = True
                    if new_score < best_score:
                        best_score = new_score
                        best_vec = p.copy()
                        print(f"Iter {total_iters:5d} – new best {best_score:.6f} (t={t})")
                    break

            if not improved:
                # No step size worked – inject a tiny random jitter and
                # continue, resetting the “no improvement” counter.
                jitter = rng.normal(scale=0.001, size=n)
                p = np.clip(p + jitter, 0.0, 1000.0)
                p = p / p.sum()
                no_improve += 1
                if no_improve >= 30:               # stagnation → restart
                    break
            else:
                no_improve = 0

        # end of inner loop – go to the next random restart
        # (unless we already hit the time limit)

    # ------------------------------------------------------------------
    # 4. Return the best vector discovered.
    # ------------------------------------------------------------------
    if best_vec is None:                     # should never happen
        best_vec = np.full(n, 1.0 / n)
    # Clip to the bounds enforced by `evaluate_sequence` for safety.
    return np.clip(best_vec, 0.0, 1000.0).tolist()
