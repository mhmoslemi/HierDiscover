def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    max_length = 200_000   # maybe adapt based on time

    # If height_sequence_1 exists, get its length; else use default
    base_seq = globals().get("height_sequence_1", [])
    n0 = len(base_seq) or 8000

    best_seq = None
    best_val = np.inf

    # Choose L values as fractions of max_length or absolute sizes
    L_vals = [int(frac * max_length) for frac in [0.001, 0.002, 0.004, 0.008, 0.016, 0.032]]
    L_vals = [max(2, l) for l in L_vals]
    L_vals = sorted(set(L_vals))

    for L in L_vals:
        if time.time() > deadline:
            break
        core = 1.0/ np.sqrt(np.arange(1, L+1, dtype=np.float64))

        S_c = core.sum()
        B_core = _conv_max(core)

        # compute maximal tail height that does not increase max
        tau_max = min(B_core / S_c, np.sqrt(B_core))
        # sweep tau factor
        for factor in np.linspace(0.2, 1.0, 9):
            tau = factor * tau_max
            if tau <= 0:
                continue
            # sweep q
            for q in [0.90,0.95,0.97,0.98,0.99,0.992,0.995,0.997,0.999,0.9995]:
                # compute maximum possible length
                max_tail_len = max_length - L
                if max_tail_len <= 0:
                    continue
                # generate tail of length max_tail_len
                tail = tau * (q ** np.arange(max_tail_len, dtype=np.float64))
                seq = np.concatenate([core, tail])
                val = _score(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    print(f"[grid] L={L}, tau={tau:.2e}, q={q:.4f}, val={val:.6f}, n={len(seq)}")
    # Additional just use tail length adaptively perhaps.

    # SA refining: similar to previous code but maybe also allow scaling of whole sequence.
    if best_seq is None:
        # fallback uniform
        best_seq = np.full(max_length, 1.0 / max_length)
        best_val = _score(best_seq)

    cur_seq = best_seq.copy()
    cur_val = best_val
    start = time.time()
    T0 = max(0.01, 0.1 * best_val)

    # Parameter to control moves: we can also randomly adjust tail height factor.
    while time.time() < deadline:
        # choose move type randomly
        if np.random.rand() < 0.5:
            # move amount between two indices
            i, j = np.random.randint(len(cur_seq), size=2)
            if cur_seq[i] <= 0:
                continue
            amount = cur_seq[i] * np.random.rand() * 0.01
            new_seq = cur_seq.copy()
            new_seq[i] -= amount
            new_seq[j] += amount
        else:
            # scale a random index up/down by up to 5%
            i = np.random.randint(len(cur_seq))
            factor = 1.0 + (np.random.rand() * 0.1 - 0.05)
            new_seq = cur_seq.copy()
            new_seq[i] = max(0.0, new_seq[i] * factor)

        new_val = _score(new_seq)
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T0):
            cur_seq, cur_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
                print(f"[SA] improved to {best_val:.6f}")

        elapsed = time.time() - start
        T0 = max(1e-6, T0 * (1.0 - elapsed / budget_s))

    print(f"final best value: {best_val:.6f}")
    return best_seq.tolist()

    conv = np.fft.irfft(fa*fa, n=size)[:2*n - 1]

"""
propose_candidate
=================

Search for a low‑evaluation step function using a two‑part construction:

* **Core** – the classic decreasing 1/√i shape.  Its self‑convolution
  is essentially constant (≈π) which yields the theoretical bound
  π/2 ≈ 1.57 for the evaluation function.

* **Tail** – a geometric (exponential) decay.  By choosing the tail
  height τ just below the safety bound τ ≤ B_core / Σ core the
  cross‑convolution never exceeds the core’s peak B_core.  The tail
  adds a large ℓ¹‑mass while keeping max(conv) unchanged, so the
  evaluation 2 n max(conv)/(∑a)² can be driven far below 1.5.

The algorithm enumerates a modest grid of core lengths, tail heights
(τ‑factors), decay rates *q* and a multiplier α that determines the
tail length T≈α/(1‑q).  Every candidate is evaluated exactly with an
FFT‑based convolution.  The best sequence found is returned; a short
simulated‑annealing phase can further polish the solution if time
allows.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions – same as in the reference implementation
# --------------------------------------------------------------

def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the linear convolution seq * seq (FFT‑based)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """Exact objective value for a given sequence."""
    n = seq.shape[0]
    total = float(seq.sum())
    if total < 0.01:
        return np.inf
    max_conv = _conv_max(seq)
    return 2.0 * n * max_conv / (total * total)

# --------------------------------------------------------------
# Main entry point required by the harness
# --------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list:
    """
    Build a step function = core (1/√i) + geometric tail.
    Perform a grid search over a few parameters and return the best
    sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5        # a small safety margin

    # Upper bound for the total length – large enough for good tails,
    # but still cheap to FFT.
    MAX_LEN = 200_000

    best_seq = None
    best_val = np.inf

    # ----------------------------------------------------------
    # 1) Grid search over simple parameter choices
    # ----------------------------------------------------------
    core_lengths = [128, 256, 512, 1024, 2048]               # size of 1/√i core
    tau_factors   = [0.5, 0.75, 1.0]                        # fraction of τ_max
    q_values      = [0.90, 0.95, 0.97, 0.98, 0.99,
                     0.992, 0.995, 0.997, 0.999, 0.9995]   # decay rates
    alphas        = [0.5, 1.0, 2.0]                         # tail‑length multiplier

    for L in core_lengths:
        if time.time() > deadline:
            break

        # core(i) = 1/√i  , i = 1 … L
        core = 1.0 / np.sqrt(np.arange(1, L + 1, dtype=np.float64))
        sum_core = float(core.sum())
        B_core   = _conv_max(core)                         # should be ≈π

        # safety bound for the tail height: τ * sum_core ≤ B_core
        tau_max = B_core / sum_core * 0.99                 # tiny safety margin

        for tf in tau_factors:
            tau = tau_max * tf
            if tau <= 0:
                continue

            for q in q_values:
                one_minus_q = 1.0 - q
                if one_minus_q <= 0:
                    continue

                for alpha in alphas:
                    # tail length ≈ α / (1‑q)
                    tail_len = int(np.ceil(alpha / one_minus_q))
                    if tail_len <= 0:
                        continue
                    if L + tail_len > MAX_LEN:
                        tail_len = MAX_LEN - L
                        if tail_len <= 0:
                            continue

                    # geometric tail: τ·q^k , k = 0 … tail_len‑1
                    tail = tau * (q ** np.arange(tail_len, dtype=np.float64))

                    seq = np.concatenate([core, tail])
                    val = _score(seq)

                    if val < best_val:
                        best_val = val
                        best_seq = seq.copy()
                        print(f"[grid] L={L}, τ={tau:.4g}, q={q:.5f}, "
                              f"α={alpha}, n={len(seq)}, val={val:.6f}")

                    if time.time() > deadline:
                        break
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ----------------------------------------------------------
    # 2) Optional short simulated‑annealing refinement
    # ----------------------------------------------------------
    if best_seq is None:                     # very unlikely, but safe fallback
        best_seq = np.full(MAX_LEN, 1.0 / MAX_LEN, dtype=np.float64)
        best_val = _score(best_seq)

    cur_seq = best_seq.copy()
    cur_val = best_val
    start = time.time()
    T0 = max(0.01, 0.1 * best_val)           # initial temperature

    while time.time() < deadline:
        # pick two indices and move a tiny amount from i → j
        i, j = np.random.randint(len(cur_seq), size=2)
        if i == j or cur_seq[i] <= 0.0:
            continue
        amount = cur_seq[i] * np.random.rand() * 0.01   # ≤1 % of donor
        new_seq = cur_seq.copy()
        new_seq[i] -= amount
        new_seq[j] += amount

        new_val = _score(new_seq)

        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T0):
            cur_seq, cur_val = new_seq, new_val
            if new_val < best_val:
                best_val, best_seq = new_val, new_seq.copy()
                print(f"[SA] improved to {best_val:.6f}")

        # simple linear cooling
        elapsed = time.time() - start
        T0 = max(1e-6, T0 * (1.0 - elapsed / budget_s))

    print(f"final best value: {best_val:.6f}")
    return best_seq.tolist()
