Algorithm:
- Work with a probability vector p (non‑negative, Σp = 1) of length n.
- The objective is 2·n·max(conv(p,p)).  Lowering max(conv) is equivalent to
  minimizing a smooth approximation of the infinity norm.
- The main loop alternates a linear‑programming step (maximise Σg subject
  to conv(p,g) ≤ M) and a line‑search to mix the current p with the LP
  solution g.  The LP enlarges the feasible mass and provably never
  increases the convolution bound, thus the objective is non‑increasing.
- When the time budget is almost exhausted a cheap projected‑gradient

import numpy as np
import time
from scipy.optimize import linprog

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    # Euclidean projection onto the probability simplex.
    n = v.shape[0]
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho_arr = u - cssv / ind
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _build_constraint_matrix(p: np.ndarray) -> np.ndarray:
    """Return matrix A (shape (2n-1, n)) such that (A @ g)[k] = (p * g)_k."""
    n = p.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # fill column j with p[j] on rows j .. j+n-1
    for j in range(n):
        A[j:j + n, j] = p[j]
    return A

def _solve_lp(p: np.ndarray, M: float) -> np.ndarray:
    """Solve max Σ g s.t. conv(p,g) ≤ M, g ≥ 0, g ≤ 1000."""
    n = p.size
    A = _build_constraint_matrix(p)
    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # minimise –Σg  ≡ maximise Σg

    return res.x

        cur = np.max(Q)
        if cur < best_val:
            best_val = cur
            best_t = t
    return best_t, best_val

"""
Algorithm
---------
We optimise a probability vector `p` (non‑negative, Σp = 1) of length `n`.
The evaluation metric is

    score(p) = 2·n·max(conv(p, p))

where `conv` is the discrete convolution.  Reducing the maximal entry of the
self‑convolution is equivalent to minimising the metric.

The search proceeds in two phases:

1. **LP‑based improvement**  
   For the current vector `p` we compute `M = max(conv(p, p))`.  The linear
   program

       maximise   Σ g_i
       subject to conv(p, g) ≤ M  (for every shift)
                 g_i ≥ 0, g_i ≤ 1000

   finds a non‑negative vector `g` that can carry more total mass than `p`
   while respecting the same convolution bound.  After normalising `g` to sum
   to one we perform a one‑dimensional line search to find the best convex
   combination `(1‑t)·p + t·g`.  This step is guaranteed never to increase
   `max(conv(·,·))` and usually produces a strict improvement.

2. **Projected‑gradient polishing**  
   When the time budget is almost exhausted a cheap projected‑gradient descent
   on a smooth approximation of the `max` operator (log‑sum‑exp) refines the
   best vector found.

Both phases keep the iterates on the probability simplex by means of the
Euclidean simplex projection.  The routine stops before the allocated time
expires and returns the best sequence (as a plain Python list) for the
provided `evaluate_sequence` function.
"""

    EPS = 1e-12

    while time.time() < deadline:
        iteration += 1
        conv_pp = np.convolve(p, p)
        M = np.max(conv_pp)

        # Solve the LP that tries to increase Σg while keeping conv(p,g) ≤ M.
        try:
            g_raw = _solve_lp(p, M)
        except RuntimeError as e:
            print(f"Iter {iteration}: LP failed – {e}")
            break

        sum_g = g_raw.sum()
        if sum_g <= 1.0 + EPS:
            print(f"Iter {iteration}: sum(g)≈{sum_g:.6f} ≤ 1 – no further gain.")
            break

        # Normalise to keep the same total mass.
        g = g_raw / sum_g

        # Find the best convex combination of p and g.
        t_opt, new_max = _best_mix_t(p, g)

        if t_opt == 0.0:
            # No improvement; inject a tiny random jitter every few steps.
            if iteration % 10 == 0:
                p += rng.normal(scale=5e-4, size=n)
                p = _simplex_projection(p)
                p /= p.sum()
            continue

        # Update the iterate.
        p = (1.0 - t_opt) * p + t_opt * g
        p = np.maximum(p, EPS)        # keep a small positivity floor
        p /= p.sum()

        new_score = 2.0 * n * new_max
        if new_score < best_score:
            best_score = new_score
            best_p = p.copy()
            print(f"Iter {iteration:5d} – new best score: {best_score:.6f}")

        # Small jitter helps escape plateaus.
        if iteration % 200 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            p = _simplex_projection(p + jitter)
            p /= p.sum()

        if best_score <= 1.503:
            print(f"Target reached (score {best_score:.6f}) after {iteration} LP steps.")
            break

    # --------------------------------------------------------------
    # Final polishing with projected‑gradient descent (if time permits).
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 2.0:
        polished_p, polished_score = _refine_by_gradient(best_p.copy(),
                                                         time_left, rng)
        if polished_score < best_score:
            best_score = polished_score
            best_p = polished_p
            print(f"Polished final score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the sequence in the format required by `evaluate_sequence`.
    # --------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    return result
