"""
Propose a sequence that improves the uniform bound (2.0) and the
square‑root bound (≈1.5708) by using a smooth‑max approximation of the
maximum convolution entry together with an Adam optimiser on the probability
simplex.

Algorithm
---------
1. Initialise a symmetric probability vector with the analytically
   motivated “square‑root” coefficients (coefficients of (1‑x)^{-1/2}),
   normalised to sum 1.

2. At each iteration
   * Compute the self‑convolution a*a via FFT.
   * Approximate the max by a soft‑max:
         w_k ∝ exp( β·(conv_k – max(conv)) )
     where β grows from a small value to a large value (continuation).
   * The gradient of the smoothed objective w.r.t. a is
         ∇ = 2·conv(w, a)               (full convolution, centred part taken)
   * Update a with the Adam rule, then project onto the probability simplex.
   * Enforce symmetry a[i]=a[n‑1‑i] (the optimum is expected to be symmetric).

3. Keep the best vector seen (according to the true `evaluate_sequence`
   function) and occasionally inject a tiny random jitter or restart from the
   analytic start if no improvement is seen for a long time.

The method works in O(n log n) per iteration thanks to FFT‑based convolutions,
allowing thousands of iterations within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Duchi et al. (2008) simplex projection."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic start: normalised coefficients of (1‑x)^(-1/2)
# ----------------------------------------------------------------------
def sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^(-1/2), normalised to sum 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolve_fft(a: np.ndarray, b: np.ndarray, out_len: int) -> np.ndarray:
    """
    Full convolution a * b using FFT. The result length is `out_len`.
    `a` and `b` are real vectors.
    """
    nfft = next_pow2(len(a) + len(b) - 1)
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    C = A * B
    conv = np.fft.irfft(C, n=nfft)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a probability vector that minimises the evaluation function.
    Parameters
    ----------
    seed : int
        Random seed.
    budget_s : int
        Time budget in seconds.
    length : int, optional
        Length of the candidate sequence (default 5808).
    Returns
    -------
    list[float]
        Best sequence found (non‑negative, sum ≈ 1).
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Configuration
    # --------------------------------------------------------------
    n = kwargs.get("length", 5808)          # dimension of the search space
    deadline = time.time() + budget_s - 0.5   # safety margin
    max_no_improve = 3000                   # steps before a restart

    # --------------------------------------------------------------
    # Adam optimiser state
    # --------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5          # initial learning rate (will decay)
    lr_decay = 5e-5    # linear decay of learning rate

    # --------------------------------------------------------------
    # Soft‑max parameters (continuation)
    # --------------------------------------------------------------
    beta_start = 2.0
    beta_end = 250.0   # final inverse temperature
    beta_inc = (beta_end - beta_start) / 50000.0   # grows roughly linearly

    # --------------------------------------------------------------
    # Initialise the candidate (symmetric sqrt series)
    # --------------------------------------------------------------
    a = sqrt_series(n)
    a = (a + a[::-1]) / 2.0                # enforce symmetry
    a = project_simplex(a)                 # just in case

    # Keep best solution seen so far
    best_seq = a.copy()
    # Manual evaluation (faster than evaluate_sequence, sum(a)=1)
    conv_init = convolve_fft(a, a, 2 * n - 1)
    best_score = 2.0 * n * conv_init.max()
    # Cross‑check with the provided function for safety
    try:
        from __main__ import evaluate_sequence  # may be already in globals
    except Exception:
        evaluate_sequence = globals().get("evaluate_sequence")
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, initial evaluation={best_score:.8f}")

    # Adam moment vectors
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Misc. counters
    iter_cnt = 0
    no_improve = 0
    beta = beta_start

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ---- convolution a*a (full) ----
        conv_len = 2 * n - 1
        conv_full = convolve_fft(a, a, conv_len)

        # ---- soft‑max weights for the max entry ----
        max_val = conv_full.max()
        # shift to avoid overflow, then normalise
        w = np.exp(beta * (conv_full - max_val))
        w_sum = w.sum()
        if w_sum == 0.0:
            # numerical safeguard – fall back to hard max
            w = np.zeros_like(conv_full)
            w[conv_full.argmax()] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ---- gradient of the smoothed max ----
        # full convolution w * a has length 3n-2; we need the centre n entries
        grad_full_len = 3 * n - 2
        grad_full = convolve_fft(w, a, grad_full_len)
        # central part corresponds to indices [n-1, n-1 + n)
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # ---- Adam update ----
        lr = lr0 / (1.0 + lr_decay * iter_cnt)   # simple decay
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        bias_correction1 = 1.0 - beta1 ** (iter_cnt + 1)
        bias_correction2 = 1.0 - beta2 ** (iter_cnt + 1)
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # ---- projection onto simplex and symmetry enforcement ----
        a = project_simplex(a)
        a = (a + a[::-1]) / 2.0   # keep the distribution symmetric

        # ---- occasional random jitter to escape plateaus ----
        if iter_cnt % 1200 == 0 and iter_cnt > 0:
            jitter = np.random.randn(n) * (1e-4 / np.sqrt(iter_cnt + 1))
            a = project_simplex(a + jitter)
            a = (a + a[::-1]) / 2.0

        # ---- evaluate current candidate (exact) ----
        cur_score = float(evaluate_sequence(a.tolist()))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{iter_cnt:5d}] improvement: {best_score:.8f}")
        else:
            no_improve += 1

        # ---- schedule update for the soft‑max temperature ----
        beta = min(beta_end, beta + beta_inc)

        # ---- restart if stagnating for a long time ----
        if no_improve >= max_no_improve:
            a = sqrt_series(n)
            a = (a + a[::-1]) / 2.0
            a = project_simplex(a + np.random.randn(n) * 1e-3)
            m.fill(0.0)
            v.fill(0.0)
            beta = beta_start
            no_improve = 0
            print(f"[{iter_cnt:5d}] restart triggered")

        iter_cnt += 1

    # --------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # --------------------------------------------------------------
    print(f"[final] best evaluation = {best_score:.8f} after {iter_cnt} iterations")
    return best_seq.tolist()
