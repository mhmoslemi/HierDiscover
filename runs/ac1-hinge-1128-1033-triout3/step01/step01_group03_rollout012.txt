"""
Hybrid search for a step‑function with a low 2·n·max(conv) value.

Approach
--------
* Build a collection of analytically motivated candidates:
    - the classic (1‑x)^(-1/2) (square‑root) series,
    - shifted square‑root series  a_i ∝ 1/√(i+c),
    - pure power‑law a_i ∝ (i+1)^(-p),
    - symmetric beta family a_i ∝ (i+½)^(α‑1)(n‑i‑½)^(α‑1),
    - a few random Dirichlet vectors.
  All are normalised to sum to 1 and projected onto the probability simplex.
* Keep the best of those as a starting point.
* Refine the best point with an **annealed smooth‑max gradient descent**.
  The true objective max(conv) is approximated by the smooth
  log‑sum‑exp:  M_λ(a)= (1/λ)·log Σ_k exp(λ·(a∗a)_k).
  Its gradient is a simple convolution, which we compute with FFT‑based
  `np.convolve`.  Adam‑style adaptive steps and a schedule that slowly
  increases λ from a gentle value (λ≈1) to a large one (λ≈200) are used.
* The iterate is projected onto the simplex after every update.
  Occasionally a tiny random jitter is added and a simple restart
  (re‑initialise from the current best + jitter) is performed if no
  improvement is seen for many iterations.
* The best sequence found before the time budget expires is returned.

The routine respects a 1000 s budget, uses only NumPy (no file or network I/O)
and works with the supplied `evaluate_sequence` function.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto {x ≥ 0, sum x = 1} (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Analytic families of candidate sequences
# --------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Coefficients of (1‑x)^(-1/2) (C_k = binom(2k,k)/4^k), normalised."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


def _shifted_sqrt_series(n: int, c: float) -> np.ndarray:
    """a_i ∝ 1/√(i + c), c>0, normalised."""
    idx = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt(idx + c)
    w /= w.sum()
    return w


def _power_series(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** (-p)
    w /= w.sum()
    return w


def _beta_series(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric beta‑like shape:
        a_i ∝ (i+½)^{α‑1} (n‑i‑½)^{α‑1},
    normalised.  α>0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i ** (alpha - 1.0)) * ((n - i) ** (alpha - 1.0))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.ones(n, dtype=np.float64) / n
    else:
        w /= s
    return w


def _random_simplex(n: int) -> np.ndarray:
    """A random point on the simplex (Dirichlet with α=1)."""
    x = np.random.rand(n)
    return _project_simplex(x)


# --------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp) of the convolution objective
# --------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, lam: float):
    """
    Returns (smooth_max, gradient) for the objective
        M_λ(a) = (1/λ) log Σ_k exp(λ·(a∗a)_k)

    The gradient ∂M_λ/∂a = 2·(a_rev * w)[n‑1:n‑1+n],
    where w_k = exp(λ·c_k)/Σ_j exp(λ·c_j) and c_k = (a∗a)_k.
    """
    conv = np.convolve(a, a)               # full convolution, length 2n‑1
    max_c = conv.max()
    # softmax weights, numerically stable
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp                 # length 2n‑1, sums to 1

    # gradient via a single convolution (a_rev * w)
    grad_full = np.convolve(a[::-1], w)    # length 3n‑2
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1:n - 1 + n]   # slice of length n
    smooth_max = max_c + np.log(sum_exp) / lam
    return smooth_max, grad


# --------------------------------------------------------------
# Main candidate generator
# --------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation value.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    n = kwargs.get("length", 5808)               # default length (as in the
                                                # previous best construction)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – generate a portfolio of analytic / random candidates
    # ------------------------------------------------------------------
    candidates = []

    # classic square‑root series
    candidates.append(_sqrt_series(n))

    # shifted square‑root series (different singularity shifts)
    for c in [0.5, 1.0, 2.0, 5.0, 10.0]:
        candidates.append(_shifted_sqrt_series(n, c))

    # pure power‑law families
    for p in np.linspace(0.30, 0.80, 6):
        candidates.append(_power_series(n, p))

    # symmetric beta‑like families
    for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
        candidates.append(_beta_series(n, alpha))

    # a few random points on the simplex
    for _ in range(5):
        candidates.append(_random_simplex(n))

    best_seq = None
    best_score = np.inf

    print("[phase 1] evaluating analytic / random starters")
    for idx, seq in enumerate(candidates):
        seq = _project_simplex(seq)                 # safety – ensure sum=1
        conv = np.convolve(seq, seq)
        score = 2.0 * n * conv.max()                # exact objective (no clipping)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        print(f"  starter {idx:02d}: eval = {score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – annealed smooth‑max Adam optimisation from the best starter
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (always on simplex)
    m = np.zeros_like(a)                    # Adam first moment
    v = np.zeros_like(a)                    # Adam second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.6                           # base learning rate for Adam
    lam_start = 1.0
    lam_end = 200.0
    max_iters = 200000                      # generous upper bound
    iter_cnt = 0
    no_improve = 0
    restart_interval = 8000                 # trigger restart after many stalls

    print("[phase 2] start smooth‑max Adam optimisation")
    while time.time() < deadline and iter_cnt < max_iters:
        # ----- evaluate current point (cheap; we already have the convolution)
        conv = np.convolve(a, a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{iter_cnt:06d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ----- annealing schedule for λ (smoothness)
        lam = lam_start + (lam_end - lam_start) * (iter_cnt / max_iters)
        lam = min(lam, lam_end)

        # ----- smooth‑max value and its gradient
        _, grad = _smooth_max_and_grad(a, lam)

        # ----- Adam update (step = base_lr / √(iteration) essentially)
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        step = base_lr / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a = _project_simplex(a_new)          # stay on the simplex

        # ----- occasional jitter to help escape shallow plateaus
        if iter_cnt % 500 == 0:
            jitter = np.random.randn(n) * (1e-4 / np.sqrt(iter_cnt + 1))
            a = _project_simplex(a + jitter)

        # ----- simple restart if we seem stuck for a while
        if no_improve >= restart_interval:
            # restart from the current best plus a modest random perturbation
            a = best_seq.copy()
            a = _project_simplex(a + np.random.randn(n) * 1e-3)
            m[:] = 0.0
            v[:] = 0.0
            no_improve = 0
            print(f"[{iter_cnt:06d}] restart triggered")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best evaluation (approx.) = {best_score:.6f} "
          f"after {iter_cnt} iterations, length={n}")
    return best_seq.tolist()
