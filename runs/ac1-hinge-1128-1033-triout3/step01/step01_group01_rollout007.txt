"""
Propose a candidate step‑function (list of non‑negative heights).

The algorithm is a multi‑restart projected Adam optimiser that
minimises the quantity

    M = 2·n·max(conv(p,p))

under the simplex constraint Σ p_i = 1 , p_i ≥ 0.
The objective is homogeneous, therefore we optimise a probability
vector p and return it unchanged (the evaluator rescales it
automatically).

Strategy
---------
* Several random restarts are performed; each restart starts from a
  random Dirichlet point or from a noisy perturbation of the current
  best solution.
* Inside a restart a smoothed version of the max‑operator is used:
  w_k ∝ exp(α·(p∗p)_k).  For α → ∞ the weights become a one‑hot
  vector on the maximal entry, i.e. the true sub‑gradient of the
  max‑norm.  α is increased slowly during optimisation.
* The gradient of the smoothed objective can be written as a single
  correlation (implemented with `np.convolve`).  Adam’s adaptive
  moment estimates are applied to the gradient, followed by Euclidean
  projection onto the probability simplex.
* Every few hundred iterations a small Gaussian jitter is added to
  escape shallow plateaus.
* After the time budget is almost exhausted a cheap coordinate‑wise
  hill‑climbing phase fine‑tunes the best vector found.
* The best vector (clipped to the evaluator’s admissible range) is
  returned as a plain Python list.

The method is deliberately lightweight (only NumPy and the standard
library) and respects the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0 , Σ x_i = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    restart = 0
    # ------------------------------------------------------------------
    # Outer loop: independent restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        restart += 1

        # --------------------------------------------------------------
        # Initialise p
        # --------------------------------------------------------------
        if best_p is not None and rng.random() < 0.5:

            p = best_p + rng.normal(scale=0.01, size=n)
            p = np.clip(p, 0.0, None)
            if p.sum() == 0.0:
                p = np.full(n, 1.0 / n)
            else:
                p = p / p.sum()
        else:
            # completely random start (Dirichlet‑like)
            p = rng.random(n)
            p = p / p.sum()

        # Adam state
        m = np.zeros(n)
        v = np.zeros(n)
        t_iter = 0

        # Hyper‑parameters (tuned empirically)
        lr = 0.2                # base learning rate
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # smoothing parameter for the soft‑max approximation
        alpha = 0.5
        alpha_mul = 1.003

        # ----------------------------------------------------------------
        # Inner optimisation loop for this restart
        # ----------------------------------------------------------------
        while time.time() < deadline:
            t_iter += 1

            # ----- convolution of p with itself (the quantity we minimise)
            conv = np.convolve(p, p)                # length 2n‑1

            # ----- smooth approximation of the max‑operator (soft‑max)
            z = alpha * conv
            max_z = np.max(z)

            #   ∇ = 2 * correlation(p, w) (restricted to the n relevant entries)

            m = beta1 * m + (1.0 - beta1) * grad

    # Final cheap local‑search (coordinate hill‑climbing)
    # ----------------------------------------------------------------------
    while time.time() < deadline:

        if i == j:
            continue

        # propose moving a small amount of probability mass from i to j
        delta = rng.normal(scale=0.01) * best_p[i]
        if delta > best_p[i] or -delta > best_p[j]:

        new_p[i] -= delta
        new_p[j] += delta

        # because the move preserves the total sum we do not need a projection
        # (the sum stays equal to 1).  Still guard against tiny negative values.
        if new_p[i] < -1e-12 or new_p[j] < -1e-12:
            continue

        # evaluate the move
        new_score = 2.0 * n * np.max(np.convolve(new_p, new_p))
        if new_score < best_score:
            best_score = new_score
            best_p = new_p
            print(f"[local] improved M = {best_score:.6f}")

        # stop if we have already beaten the target
        if best_score <= 1.503:
            break

    # ----------------------------------------------------------------------
    # Return the best vector (clipped to the evaluator's admissible range)
    # ----------------------------------------------------------------------
    if best_p is None:                     # should never happen, but be safe
        best_p = np.full(n, 1.0 / n)

    result = np.clip(best_p, 0.0, 1000.0).tolist()
    return result
