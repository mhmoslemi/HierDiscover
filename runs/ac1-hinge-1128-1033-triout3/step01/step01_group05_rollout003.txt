def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, scipy.sparse, scipy.optimize

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # Parameters
    N = 2500  # length of initial sequence; adjust.
    # generate central binomial coefficients
    coeffs = generate_binomial_prefix(N)
    # choose best length based on analytic score (maybe 3000 original had best)
    # but for simplicity just use N
    f = coeffs.tolist()
    # Evaluate initial
    best_seq = f[:]
    best_score = evaluate_sequence(best_seq)
    print(f"[init] n={len(f)} score={best_score:.6f}")

    # iteration parameters
    max_iter = 30
    t = 0.2
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        # compute current max convolution
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())
        # Build constraint matrix A_ub
        n = len(f)
        # using diags
        diags = [np.full(n - i, f[i], dtype=np.float64) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        A = scipy.sparse.diags(diags, offsets, shape=(2*n - 1, n), format='csr')
        b = np.full(2*n - 1, M, dtype=np.float64)

        # Linear program: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        # Solve
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"[lp] iteration {iteration} LP failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = float(g0.sum())
        if sum_g0 <= 1e-12:
            break
        # scale g to keep L1 norm equal to current sum_f
        sum_f = float(np.sum(f))
        scale = sum_f / sum_g0
        g = g0 * scale

        # mixing
        f_new = (1 - t) * np.array(f) + t * g

        # ensure nonnegative
        f_new = np.maximum(f_new, 0.0)

        # evaluate
        try:
            score_new = evaluate_sequence(f_new.tolist())
        except Exception:
            # fallback, stop
            break

        print(f"[iter {iteration}] M={M:.6f} sum={sum_f:.6f} t={t:.3f} score={score_new:.6f}")

        if score_new < best_score:
            best_score = score_new
            best_seq = f_new.tolist()
            print(f"[improve] new best score {best_score:.6f} at iter {iteration}")

        # Update for next iteration
        f = f_new.tolist()
        # optionally reduce t for stability
        t = max(0.05, t * 0.95)

    # final clamp and return
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return best_seq

"""
Propose a step‑function (list of non‑negative heights) that improves the
baseline central‑binomial construction.

Approach
--------
1.  Use the analytically optimal “central‑binomial’’ coefficients
    b_k = C(2k,k)/4**k and pick the prefix length that gives the smallest
    analytic score 2·n/(∑b_k)² (this yields the ~π/2 ≈ 1.5709 bound).

2.  Starting from that prefix, repeatedly solve a linear programme:
        maximise   ∑ g_j
        subject to  (f * g)[k] ≤ M   for all k,
                    g_j ≥ 0,
    where f is the current sequence and M = max(f*f) (a linear constraint
    because f is fixed).  The LP finds a “partner’’ sequence g that can
    carry much more L1‑mass without raising the convolution bound.

3.  Rescale g so that it has the same L1‑norm as f, then mix:
        f ← (1‑t)·f + t·g    (t≈0.2, slowly reduced).
    The mixture keeps the total sum unchanged but, thanks to the
    (typically) smaller self‑convolution of g, reduces the max‑convolution
    M.  The objective 2·n·M/(∑f)² therefore drops.

4.  Iterate the LP‑mix step while time permits (≈10‑20 iterations).
    The algorithm is deterministic (seeded) and returns the best sequence
    found.

The LP is built with a sparse Toeplitz matrix representing the convolution
operator, so each iteration costs O(n²) memory (~25 MiB for n≈3000) and is
solved by SciPy’s HiGHS interior‑point method, well within the 1000 s budget.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def generate_binomial_prefix(max_n: int) -> np.ndarray:
    """Return b_k = C(2k,k)/4**k for k=0..max_n-1 using the recurrence."""
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, max_n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


def build_conv_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the matrix A (size (2n‑1)×n) such that A @ g = f * g (zero‑padded convolution).
    The k‑th row of A contains f_i on the diagonal offset i (i≥0).
    """
    n = f.shape[0]
    # each diagonal i (offset = i) has length n-i and value f[i]
    diags = [np.full(n - i, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = scipy.sparse.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def conv_max(seq: np.ndarray) -> float:
    """Return max(conv(seq, seq)) using NumPy's convolution."""
    return float(np.convolve(seq, seq).max())


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves on the
    baseline 1.5709 score.  The routine runs for up to ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # safety margin

    # ----- Phase 1: analytic scan for the best central‑binomial prefix -----
    MAX_LEN = 4000               # enough for the analytic optimum
    coeffs = generate_binomial_prefix(MAX_LEN)

    cumulative = 0.0
    best_len = 1
    best_analytic = float("inf")
    for n in range(1, MAX_LEN + 1):
        cumulative += coeffs[n - 1]
        # max of self‑convolution of the *full* infinite series is 1,
        # and the truncated prefix inherits this for k < n.
        score = 2.0 * n / (cumulative * cumulative)
        if score < best_analytic:
            best_analytic = score
            best_len = n

    # ----- Phase 2: initialise from that prefix -----
    f = coeffs[:best_len].astype(np.float64)   # current working sequence
    best_seq = f.tolist()
    try:
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_score = float("inf")
    print(f"[init] length={best_len}, analytic={best_analytic:.6f}, real={best_score:.6f}")

    # ----- Phase 3: iterative LP‑mix optimisation -----
    max_iters = 30                # generous but stops early if time runs out
    t = 0.2                       # initial mixing factor
    iteration = 0

    while time.time() < deadline and iteration < max_iters:
        iteration += 1
        n = f.shape[0]

        # current max‑convolution
        M = conv_max(f)

        # build the Toeplitz convolution matrix for the constraints
        A = build_conv_matrix(f)                # (2n‑1) × n sparse
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # linear programme: maximise sum(g)  <=> minimise –sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        # solve
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[lp] iter {iteration} failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = float(g0.sum())
        if sum_g0 <= 1e-12:
            print("[lp] zero‑mass solution, stopping.")
            break

        # scale g0 so that it has the same L1‑norm as f (keeps denominator unchanged)
        sum_f = float(f.sum())
        scale = sum_f / sum_g0
        g = g0 * scale

        # convex combination
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)          # guard against tiny negatives

        # evaluate the true objective
        try:
            score_new = evaluate_sequence(f_new.tolist())
        except Exception:
            print("[eval] evaluation failed, aborting.")
            break

        print(
            f"[iter {iteration:2d}] M={M:.6f}  sum={sum_f:.6f}  t={t:.3f}  score={score_new:.6f}"
        )

        # keep the best ever seen
        if score_new < best_score:
            best_score = score_new
            best_seq = f_new.tolist()
            print(f"[improve] new best score {best_score:.6f} (iter {iteration})")

        # prepare for next iteration
        f = f_new
        t = max(0.05, t * 0.95)   # slowly decay the step size

        # early exit if we already beat the target
        if best_score < 1.503:
            print("[target] achieved target bound, stopping early.")
            break

    # final sanity: clamp to the limits enforced by evaluate_sequence
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]
    return best_seq
