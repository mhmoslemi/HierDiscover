def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    from scipy.optimize import linprog

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    def binomial_prefix(n):
        coeff = np.empty(n, dtype=np.float64)
        cur = 1.0
        coeff[0] = cur
        for k in range(1, n):
            cur = cur * (2*k - 1) / (2*k)
            coeff[k] = cur
        return coeff

    # candidate lengths
    lengths = [200, 400, 600, 800]   # adjust according to time

    best_seq = None
    best_score = float('inf')
    # iterate over lengths
    for n in lengths:
        if time.time() >= deadline:
            break
        f = binomial_prefix(n)    # initial sequence
        # No need to scale
        # optionally compute initial objective
        s = evaluate_sequence(f.tolist())
        if s < best_score:
            best_score = s
            best_seq = f.tolist()
            print(f"[init len={n}] score={s:.6f}")
        # optimization loop
        t = 0.1
        iter_no = 0
        while time.time() < deadline and iter_no < 200:
            conv_f = np.convolve(f, f)
            M = conv_f.max()
            # Build constraint matrix A
            rows = 2 * n - 1
            A = np.zeros((rows, n), dtype=np.float64)
            # fill columns
            for j in range(n):
                A[j:j+n, j] = f
            # Solve LP
            c = -np.ones(n)
            ub = np.full(rows, M)
            res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)] * n, method='highs', options={'presolve': True, 'quiet': True})
            if not res.success:
                print(f"[lp fail] n={n} iter={iter_no}")
                break
            b = res.x
            # mixing
            f = (1 - t) * f + t * b
            # compute new score
            new_score = evaluate_sequence(f.tolist())
            if new_score < best_score - 1e-9:
                best_score = new_score
                best_seq = f.tolist()
                print(f"[improve n={n} iter={iter_no} t={t:.3f}] score={new_score:.6f}")
                # optionally decrease t slowly for stability
                t = max(t * 0.9, 0.001)
            else:
                # reduce step
                t = max(t * 0.9, 0.001)
            iter_no += 1
            # early stop if no big change
            if iter_no % 20 == 0:
                # recompute M and bounds maybe if there is stalling
                pass

    # final returned best sequence trimmed zeros and ensured positivity
    final_seq = [float(max(0.0, x)) for x in best_seq]
    print(f"[final] length={len(final_seq)} best_score={best_score:.6f}")
    return final_seq

assistantfinal**Solution Overview**

We improve the previous “constant‑convolution” construction by an **iterative linear‑programming
scheme** that is very close to the method described in the statement.

*   Start from a binomial prefix `f`.  
    For this sequence the self‑convolution is bounded by `M = max(conv(f,f)) = 1`
    and the objective equals  

    \[
    \text{score}= \frac{2\,n\,M}{(\sum f)^2}= \frac{2n}{(\sum f)^2}\; .
    \]

*   Keep `f` fixed, compute `M`, and solve the **LP**

    \[
    \begin{aligned}
    \max\;&\mathbf 1^{\!\top} b \\
    \text{s.t.}&\; (f * b)_k \le M \quad\forall k=0,\dots,2n-2\\
                &\; b_j \ge 0 .
    \end{aligned}
    \]

    The constraints are linear because the convolution `f * b` is linear in `b`.
    The optimal vector `b` has a **larger total sum** than `f` while still respecting
    the same bound `M`.

*   Update the current sequence by a small convex combination  

    \[
    f \leftarrow (1-t)\,f + t\,b ,
    \qquad t\in(0,1) .
    \]

    The mixing raises the total sum but only modestly increases the peak of the
    self‑convolution (the factor `t` is kept small).  
    After the update we recompute the real score and keep the best sequence seen.

*   The whole process is repeated until the time budget (≈ 1000 s) runs out.
    The algorithm is executed for a few different lengths (200 … 800).  
    The best sequence among all runs is returned.

The implementation uses `numpy` for the convolutions and `scipy.optimize.linprog`
(high‑s performance of the HiGHS solver) for the linear programs.  
All arithmetic stays far below the 0 – 1000 clamps required by `evaluate_sequence`.

```python
"""
propose_candidate
-----------------

Iterative LP‑based optimisation starting from the optimal “constant‑convolution”
(binomial) prefix.  In each iteration we

1. compute M = max(conv(f,f));
2. solve   max sum(b)  s.t. conv(f,b) ≤ M,  b ≥ 0;
3. replace f ← (1‑t)·f + t·b with a small mixing coefficient t;
4. keep the best sequence found.

The procedure is run for several lengths (200, 400, 600, 800) and stops
as soon as the 1000 s budget is exhausted.  The final sequence is returned
as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------
# Helper: binomial (constant‑convolution) coefficients
# --------------------------------------------------------------
def generate_binomial_prefix(max_n: int) -> np.ndarray:
    """
    Return the first ``max_n`` coefficients of
        b_k = C(2k, k) / 4**k
    using the recurrence
        b_{k+1} = b_k * (2k+1) / (2(k+1)).
    """
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, max_n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


# --------------------------------------------------------------
# Core optimisation loop for a single length
# --------------------------------------------------------------
def optimise_length(initial_f: np.ndarray,
                    max_iter: int,
                    time_deadline: float,
                    length_id: int) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑mixing optimisation for a fixed length.
    Returns the best sequence found and its score.
    """
    f = initial_f.copy()                     # current sequence (non‑negative)
    n = len(f)

    # evaluate the initial sequence once (using the same formula as evaluate_sequence)
    conv = np.convolve(f, f)
    M = conv.max()
    score = 2.0 * n * M / (np.sum(f) ** 2)
    best_f = f.copy()
    best_score = score

    # mixing parameter – start relatively large and shrink slowly
    t = 0.12

    # ----------------------------------------------------------
    # Main iteration
    # ----------------------------------------------------------
    it = 0
    no_improve = 0
    while it < max_iter and time.time() < time_deadline:
        # 1) build the constraint matrix A where A @ b = conv(f,b)
        #    (row k, column j) = f[k‑j] if 0 ≤ k‑j < n else 0.
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f

        # 2) linear programme – maximise 1ᵀb  (i.e. minimise –1ᵀb)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(b)
        ub = np.full(rows, M, dtype=np.float64)    # conv(f,b) ≤ M
        bounds = [(0.0, None)] * n

        res = linprog(c,
                      A_ub=A,
                      b_ub=ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'quiet': True})

        if not res.success:
            # if the LP fails we abort this length – it should be extremely rare
            print(f"[{length_id}] LP failed at iter {it}, status: {res.message}")
            break

        b = res.x

        # 3) convex combination (small step)
        f = (1.0 - t) * f + t * b

        # 4) evaluate the new objective
        conv = np.convolve(f, f)
        M = conv.max()
        cur_score = 2.0 * n * M / (np.sum(f) ** 2)

        # keep the global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{length_id}] iter {it:04d}  t={t:.4f}  score={best_score:.6f}")
            no_improve = 0
            # gradually shrink the step size after a successful move
            t = max(t * 0.97, 0.001)
        else:
            no_improve += 1
            # if we see many non‑improving steps, reduce the step size
            if no_improve >= 30:
                t = max(t * 0.7, 0.001)
                no_improve = 0

        it += 1

    return best_f, best_score


# --------------------------------------------------------------
# Public entry point required by the grader
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises
        2 * n * max(conv(a,a)) / (sum(a)**2)

    The routine runs several independent optimisation threads (different
    lengths) within the allotted time budget and returns the best sequence
    discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # small safety margin

    # lengths we will try – short enough for fast LP solves,
    # long enough to give a good n·M trade‑off.
    lengths = [200, 400, 600, 800]

    best_overall = None
    best_overall_score = float('inf')

    for idx, n in enumerate(lengths):
        if time.time() > deadline:
            break

        # ----- initialise with the binomial prefix -----
        start_seq = generate_binomial_prefix(n)

        # ----- run the iterative LP optimiser -----
        max_iters = 500                     # plenty for the given budget
        seq, score = optimise_length(start_seq,
                                     max_iter=max_iters,
                                     time_deadline=deadline,
                                     length_id=idx)

        # keep the best of all lengths
        if score < best_overall_score:
            best_overall_score = score
            best_overall = seq.copy()
            print(f"[global] new best length={n} score={score:.6f}")

    # final sanitisation – ensure non‑negative numbers and respect the
    # 0‑1000 clamp imposed by evaluate_sequence.
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_overall]

    # final diagnostic
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] length={len(final_seq)} score={final_score:.6f}")
    except Exception:
        # should never happen – we keep the internally computed score instead
        print(f"[final] length={len(final_seq)} (internal score={best_overall_score:.6f})")
    return final_seq
```
