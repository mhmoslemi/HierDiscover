"""
Propose a candidate step‑function.

The algorithm starts from the optimal “arcsine” (inverse‑square‑root) series,
which yields the classical bound π/2 ≈ 1.5708.  To improve on this bound we
perform a simple simulated‑annealing style stochastic optimisation that moves
mass from the high‑impact early coefficients to the low‑impact tail.  Reducing
the early terms lowers many convolution entries (the maximum is attained for
all k < n), while the added mass at large indices only influences convolution
entries that still have slack (they are far from the saturated part).  The
search repeatedly transfers a random amount of weight between a randomly chosen
early index and a randomly chosen late index, accepts moves that improve the
evaluation score (or occasionally accepts worsening moves according to a
temperature schedule), and keeps the best sequence seen.

The whole search runs within the allotted time budget (default 950 s, leaving a
small safety margin).  The returned list is the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – they are defined at module level on purpose
# ----------------------------------------------------------------------


def _compute_score(seq: list[float]) -> float:
    """
    Fast evaluation of the objective using an FFT based convolution.
    Returns the same value as ``evaluate_sequence`` (the function is provided
    by the evaluation environment).  The implementation mirrors the
    reference implementation for speed.
    """
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    # FFT length – next power of two ≥ 2*n-1
    L = 1 << ((2 * n - 2).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _arcsine_coeffs(n: int) -> np.ndarray:
    """
    First ``n`` coefficients of (1‑z)^(-1/2) = Σ binom(2i,i)/4^i z^i.
    The recurrence a_{i+1}=a_i*(2i+1)/(2(i+1)) is numerically stable.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def _random_transfer(src: int, dst: int, seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Propose a new sequence by moving a random amount of mass from ``src`` to
    ``dst`` (both indices are distinct).  The move respects the hard bounds
    0 ≤ a_i ≤ 1 (the current max of the convolution never exceeds 1, therefore
    any coefficient larger than 1 would certainly increase the max).  The
    function returns the new sequence and the amount transferred.
    """
    if src == dst:
        return seq, 0.0
    # maximal amount we can take from src and add to dst without violating bounds
    max_take = seq[src]
    max_give = 1.0 - seq[dst]
    max_delta = min(max_take, max_give)
    if max_delta <= 0.0:
        return seq, 0.0

    # sample a fraction – we favor moderate moves but allow very small ones
    delta = max_delta * np.random.rand()
    if delta <= 0.0:
        return seq, 0.0

    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    return new_seq, delta


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, length: int = 5000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds.  The routine stops a fraction of a second
        before this limit to avoid being terminated.
    length : int, optional
        Length of the initial arcsine series.  Values around 5000 give a good
        trade‑off between quality and speed.

    Returns
    -------
    list[float]
        The best sequence found before the time budget expires.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.   Deterministic start with the arcsine (inverse‑square‑root) series
    # ------------------------------------------------------------------
    seq = _arcsine_coeffs(length)
    best_seq = seq.copy()
    best_score = _compute_score(best_seq.tolist())
    print(f"[init] length={length}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.   Stochastic refinement – simulated annealing style
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin
    n = length

    # Index ranges for the source (early, high impact) and destination
    # (late, low impact) moves.
    early_end = max(1, n // 10)                # first 10% of the array
    late_start = n // 2                       # second half of the array

    # Simple linear temperature schedule
    T_start = 1e-2
    T_end = 1e-6

    iteration = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Temperature – linearly decreasing
        # --------------------------------------------------------------
        elapsed = time.time() - start
        frac = 1.0 - elapsed / (deadline - start)
        temperature = T_start * frac + T_end * (1.0 - frac)

        # --------------------------------------------------------------
        # Choose a source index (early) with probability proportional to its
        # current magnitude – this focuses the search on coefficients that
        # have the biggest impact on the max convolution.
        # --------------------------------------------------------------
        src = np.random.randint(0, early_end)
        if seq[src] <= 1e-12:
            continue

        # --------------------------------------------------------------
        # Choose a destination index (late) – uniform in the second half.
        # --------------------------------------------------------------
        dst = np.random.randint(late_start, n)

        # --------------------------------------------------------------
        # Propose the move
        # --------------------------------------------------------------
        new_seq, delta = _random_transfer(src, dst, seq)
        if delta == 0.0:
            continue

        # --------------------------------------------------------------
        # Compute the new score
        # --------------------------------------------------------------
        cur_score = _compute_score(new_seq.tolist())

        # --------------------------------------------------------------
        # Acceptance test – Metropolis criterion
        # --------------------------------------------------------------
        accept = False
        if cur_score < best_score:
            accept = True
        else:
            # probabilistic acceptance of worsening moves
            prob = np.exp(-(cur_score - best_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            seq = new_seq
            if cur_score < best_score:
                best_score = cur_score
                best_seq = new_seq.copy()
                if iteration % 100 == 0:
                    print(f"[iter {iteration:06d}] improved score={best_score:.6f}")

        iteration += 1

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # The evaluate_sequence routine will clamp values to [0,1000] anyway,
    # but we keep the numbers as‑is.
    return best_seq.tolist()
