    for it in range(max_iters):
        # time‑budget check
        if time.time() - start_time > budget_s - 5.0:
            print("[timeout] stopping early")
            break

        # current convolution maximum
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # Build the actual sparse matrix A with the current f.
        # data = repeat each f[i] exactly n times (one per row in its column)
        data_vals = np.repeat(f, n)
        A = sparse.csr_matrix((data_vals, (row_idx, col_idx)),
                              shape=(2 * n - 1, n))

        # Linear programme: maximise Σ g   →   minimise -Σ g
        c = -np.ones(n, dtype=np.float64)                # objective
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)   # right‑hand side

        # Solve with HiGHS (fast interior‑point for large sparse LPs)
        res = optimize.linprog(c,
                               A_ub=A,
                               b_ub=b_ub,
                               bounds=[(0, None)] * n,
                               method='highs',
                               options={'presolve': True, 'time_limit': budget_s - (time.time() - start_time)})

        if not res.success:
            print(f"[iter {it}] LP failed – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            print(f"[iter {it}] degenerate solution (sum_g={sum_g})")
            break

        # Rescale to keep the required normalisation Σg = √(2 n)
        scale = target_sum / sum_g
        g = g * scale

        # Mix the new candidate with the current sequence
        new_f = (1.0 - t) * f + t * g

        # Evaluate the mixed sequence
        new_score = evaluate_sequence(new_f.tolist())
        print(f"[iter {it}] score={new_score:.6f}  M={M:.6f}  t={t:.3f}")

        if new_score < best_score - 1e-9:
            best_score = new_score
            best_seq = new_f.copy()
            f = new_f
            no_improve_cnt = 0
            print(f"  [improved] new best score={best_score:.6f}")
            # slowly increase mixing strength to move faster while we improve
            t = min(0.5, t * 1.05)
        else:
            no_improve_cnt += 1
            # shrink the mixing factor if we are not improving
            t = max(0.05, t * 0.7)
            if no_improve_cnt >= 4:
                print("[no improvement] terminating early")
                break

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {it+1} iterations")
    return best_seq.tolist()
