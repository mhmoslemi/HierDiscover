"""
Iterative linear‑programming refinement of the arcsine step function.

The algorithm starts from the classic arcsine (inverse‑square‑root) sequence,
which yields a constant convolution value of 1.  For the current sequence
`f`, the maximum convolution entry `M = max(f*f)` is computed.  A linear
program is then solved:

    maximise    Σ g_j
    subject to  (f * g)_k ≤ M   for all k
                g_j ≥ 0

Because `f` is fixed, each convolution constraint is linear in the
variables `g_j`.  The solution `g₀` uses the available “budget’’ `M`
most efficiently and typically shifts mass towards indices where `f`
is small, thereby lowering the future self‑convolution of the mixed
sequence.

`g₀` is rescaled to have the same total mass as `f` (the evaluation
function is scale‑invariant) and then replaces `f`.  Repeating this step
monotonically reduces the maximal convolution value while preserving the
ℓ₁‑norm, which reduces the evaluation score
`2·n·max(b)/(Σa)²`.  The loop stops when no further improvement is seen
or the time budget expires.

If `cvxpy` is not available, the routine falls back to the original
stochastic mass‑transfer refinement.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine coefficients (inverse‑square‑root power series)
def _arcsine_coeffs(n: int):
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs.tolist()


# ----------------------------------------------------------------------
# Fast evaluation matching `evaluate_sequence`
def _compute_score(seq):
    seq = np.asarray(seq, dtype=np.float64)
    n = len(seq)
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = float(np.max(conv))
    sum_a = float(np.sum(seq))
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Linear‑programming step (requires cvxpy)
def _lp_optimize(seq, M):
    """
    Solve   maximise Σ g_j
            subject to conv(seq, g) ≤ M  (element‑wise)
                     g_j ≥ 0
    Returns the optimal vector g (as a NumPy array).
    """
    try:
        import cvxpy as cp
    except Exception as e:
        raise RuntimeError("cvxpy is required for the LP step") from e

    n = len(seq)
    g = cp.Variable(n, nonneg=True)          # variables g_j ≥ 0
    f = cp.Parameter(n, value=np.array(seq, dtype=np.float64))
    # convolution with one argument a Parameter is a linear expression
    conv_fg = cp.conv(f, g)                  # length 2n‑1
    constraints = [conv_fg <= M]             # element‑wise inequality

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # ECOS works well for dense linear programs; warm_start speeds up repeats
    prob.solve(solver="ECOS", warm_start=True,
               max_iters=5000, abstol=1e-8,
               reltol=1e-8, feastol=1e-8, verbose=False)

    if g.value is None:
        raise RuntimeError(f"LP did not solve to optimality (status {prob.status})")
    return np.asarray(g.value, dtype=np.float64)


# ----------------------------------------------------------------------
# Fallback stochastic refinement (the original simple method)
def _stochastic_refine(seq, deadline):
    """Very cheap mass‑transfer hill‑climbing; used when cvxpy is absent."""
    n = len(seq)
    best_seq = seq.copy()
    best_score = _compute_score(best_seq)

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j or seq[i] <= 0.0:
            continue
        # take a tiny exponential fraction of the source entry
        delta = np.random.exponential(scale=1e-3) * seq[i]
        if delta > seq[i]:
            continue

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        cur_score = _compute_score(new_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            seq = new_seq
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, length: int = 1500):
    """
    Produce a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The method uses an LP‑based refinement of the
    arcsine sequence; if the LP machinery is unavailable it falls back
    to a simple stochastic hill‑climber.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # 1. deterministic start with the arcsine series
    seq = np.array(_arcsine_coeffs(length), dtype=np.float64)
    best_seq = seq.copy()
    best_score = _compute_score(best_seq)
    print(f"[init] length={length}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Try the LP‑based refinement – it is the core improvement technique.
    try:
        iteration = 0
        while time.time() < deadline and iteration < 30:   # cap iterations
            # current convolution bound M = max(f * f)
            conv_ff = np.convolve(seq, seq)
            M = float(np.max(conv_ff))

            # solve the linear program for a better companion sequence
            g0 = _lp_optimize(seq.tolist(), M)

            # rescale to keep the total ℓ₁‑norm unchanged
            sum_seq = float(np.sum(seq))
            sum_g0 = float(np.sum(g0))
            if sum_g0 == 0:
                print("[lp] zero solution – aborting LP phase")
                break
            scale = sum_seq / sum_g0
            g = g0 * scale

            # evaluate the candidate
            new_score = _compute_score(g)

            if new_score < best_score:
                best_score = new_score
                best_seq = g.copy()
                seq = g.copy()
                print(f"[iter {iteration}] improved score={best_score:.6f}")
            else:
                print(f"[iter {iteration}] no improvement (score={new_score:.6f}); stopping LP")
                break

            iteration += 1

        print(f"[final] best score after LP stage={best_score:.6f}")

    except Exception as e:
        # If any part of the LP approach fails, fall back to stochastic search
        print("[fallback] LP phase failed:", e)
        seq = best_seq.copy()
        best_seq, best_score = _stochastic_refine(seq, deadline)

    # ------------------------------------------------------------------
    # As a safety net, perform a brief stochastic refinement on whatever we have
    if time.time() < deadline:
        seq = best_seq.copy()
        best_seq, best_score = _stochastic_refine(seq, deadline)

    print(f"[return] final best score={best_score:.6f}")
    # convert back to plain Python list for the caller
    return best_seq.tolist()
