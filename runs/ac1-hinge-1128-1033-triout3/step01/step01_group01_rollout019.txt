def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    default_n = 2048
    try:
        n0 = len(height_sequence_1)  # if defined
        n = max(n0, default_n)
    except Exception:
        n = default_n

    # Prepare FFT length
    conv_len = 2 * n - 1
    N_fft = 1 << (conv_len - 1).bit_length()  # next power of two

    # Initialize p:
    if rng.random() < 0.3 and 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)
        if len(seq) < n:
            seq = np.concatenate([seq, np.zeros(n - len(seq))])
        else:
            seq = seq[:n]
        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total > 0:
            p = seq / total
        else:
            p = rng.random(n)
            p = p / p.sum()
    else:
        p = rng.random(n)
        p = p / p.sum()

    # ensure non-neg
    p = np.maximum(p, 0.0)
    p = _simplex_projection(p)

    # function to compute M
    def compute_M(vec):
        fft_vec = np.fft.rfft(vec, n=N_fft)
        conv = np.fft.irfft(fft_vec * fft_vec, n=N_fft)[:conv_len]
        max_conv = conv.max()
        return float(2.0 * n * max_conv), conv, max_conv

    # initial evaluation
    cur_M, cur_conv, cur_max = compute_M(p)
    best_M = cur_M
    best_p = p.copy()
    print(f"Initial M: {best_M:.6f}")

    # hyperparameters
    deadline = time.time() + budget_s - 5.0
    iter_count = 0
    eta = 0.5   # initial step size
    eta_decay = 0.9995
    temp = 0.01  # temperature for simulated annealing
    temp_decay = 0.9995

    while time.time() < deadline:
        iter_count += 1

        # locate index where conv is maximal
        k_star = int(np.argmax(cur_conv))

        # compute subgradient
        indices = k_star - np.arange(n)
        mask = (indices >= 0) & (indices < n)
        g = np.zeros_like(p)
        valid = np.where(mask)[0]
        if valid.size > 0:
            g[valid] = 2.0 * p[indices[valid]]

        # gradient descent step
        p_new = p - eta * g
        p_new = _simplex_projection(p_new)

        # occasional random jitter to escape local minima
        if iter_count % 200 == 0:
            jitter = rng.normal(scale=0.002, size=n)
            p_new = _simplex_projection(p_new + jitter)

        M_new, conv_new, max_new = compute_M(p_new)

        # accept if improvement or with probability exp(-(Δ)/temp)
        if M_new < cur_M or rng.random() < np.exp(-(M_new - cur_M) / max(temp, 1e-12)):
            p = p_new
            cur_M = M_new
            cur_conv = conv_new
            cur_max = max_new

        # track best
        if cur_M < best_M:
            best_M = cur_M
            best_p = p.copy()
            print(f"Iter {iter_count:6d} – new best M: {best_M:.6f}")

        # update schedules
        eta *= eta_decay
        temp *= temp_decay

        # early stop if target reached
        if best_M <= 1.503:
            break

    # final clip to bounds [0,1000]
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Finished after {iter_count} iterations – best M: {best_M:.6f}")
    return result

"""
Propose a candidate step‑function.

The search proceeds in two phases:

1. **Parametric sweep** – simple analytic families (exponential,
   power‑law, symmetric versions) are generated for a moderately large
   length (default 2048).  For each parameter the objective
   `M = 2·n·max(conv(p,p))` is evaluated with an FFT‑based convolution.
   The best candidate of this sweep is taken as the starting point.

2. **Projected (sub‑)gradient descent with simulated annealing** – the
   exact max‑convolution is used, its index `k*` gives a sub‑gradient
   `g_i = 2·p_{k*-i}` (zero outside the support).  The iterate is moved
   opposite the gradient, projected onto the probability simplex, and
   occasionally perturbed by small Gaussian noise.  An annealing
   temperature permits occasional uphill moves, helping escape local
   minima.  The step size and temperature decay exponentially.

The algorithm stops after the allotted time (default 1000 s) or as soon
as a value `M ≤ 1.503` is found.  The returned list is already normalised
to sum = 1, non‑negative and within the clipping bounds required by
`evaluate_sequence`.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x >= 0, sum x = 1}."""
    if np.abs(v.sum() - 1) < 1e-12 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate distributions
# ----------------------------------------------------------------------
def _exp_decay(n, lam):
    """Monotone exponential p_i ∝ lam**i."""
    arr = np.power(lam, np.arange(n, dtype=np.float64))
    return arr / arr.sum()


def _sym_exp_decay(n, lam):
    """Symmetric exponential centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    arr = np.power(lam, np.abs(idx - centre))
    return arr / arr.sum()


def _power_decay(n, s):
    """Monotone power‑law p_i ∝ (i+1)^{-s}."""
    arr = np.power(np.arange(1, n + 1, dtype=np.float64), -s)
    return arr / arr.sum()


def _sym_power_decay(n, s):
    """Symmetric U‑shaped power‑law."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    centre = (n + 1) / 2.0
    arr = np.power(np.abs(idx - centre) + 1.0, -s)
    return arr / arr.sum()


def _edge_u_shape(n, s):
    """Edge‑concentrated shape: (i+1)^{-s} + (n-i)^{-s}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    arr = np.power(i, -s) + np.power(n - i + 1, -s)
    return arr / arr.sum()


# ----------------------------------------------------------------------
# Core evaluation (FFT‑based max convolution)
# ----------------------------------------------------------------------
def _max_conv_and_obj(p: np.ndarray, n: int, fft_len: int):
    """Return (M, conv_array, max_conv) for a probability vector p."""
    fft_p = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(fft_p * fft_p, n=fft_len)[: 2 * n - 1]
    max_c = float(conv.max())
    M = float(2.0 * n * max_c)
    return M, conv, max_c

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    DEFAULT_N = 2048                     # power‑of‑two, convenient for FFT
    try:

    except Exception:
        n = DEFAULT_N

    fft_len = 1 << ((conv_len - 1).bit_length())

        else:

        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total > 0.0:
            p = seq / total
        else:
            p = rng.random(n)

    else:

    # make sure we are on the simplex

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    p = best_p.copy()

        # --- sub‑gradient from the current maximal index ---
        k_star = int(np.argmax(cur_conv))

        accept = False

        else:

            prob = math.exp(-(M_new - cur_M) / max(temp, 1e-12))

        if best_M <= 1.503:

            break

    # 5. Return the best sequence found.
    # --------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Finished after {it} iterations – best M: {best_M:.6f}")
    return result
