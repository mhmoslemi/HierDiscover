"""
Propose a candidate step‑function sequence by alternating linear‑programming
updates and a small stochastic refinement.

The method follows the “alternating LP” idea:
* Keep a current non‑negative vector a (∑a = 1).
* Let M = max(a * a).  (a * a denotes the ordinary convolution.)
* Solving the LP
        maximise   1⋅g
        subject to conv_a_g ≤ M   (component‑wise)
                   g ≥ 0
  yields a vector g that can be added to a without increasing the current
  maximum self‑convolution.
* Mix a and the normalised g (a ← (1‑t) a + t·g/∑g).  This strictly reduces
  the maximal convolution entry unless a is already optimal.
* Repeat until the time budget is exhausted.

After the LP phase a short simulated‑annealing style refinement is performed
to squeeze out any remaining improvements.

The algorithm is completely self‑contained, uses only NumPy, CVXPY (ECOS) and
the provided `evaluate_sequence` function, and respects the 1000 s budget.
"""

import time
import math
import numpy as np

# CVXPY is optional – the submission environment provides it.
import cvxpy as cp


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A @ g == np.convolve(a, g)
    for any vector g of length n.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # each column i contributes a shifted copy of a
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _solve_lp_max_sum(a: np.ndarray, M: float):
    """
    Solve   maximise   sum(g)   subject to   conv(a,g) ≤ M   and   g ≥ 0.
    Returns the optimal g (as a 1‑D NumPy array) or None on failure.
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-6, reltol=1e-6, feastol=1e-6, verbose=False)
    except Exception as e:                # pragma: no cover – safety net
        print("[LP] solver exception:", e)
        return None

    if g.value is None:
        print("[LP] no solution (status:", prob.status, ")")
        return None

    return np.array(g.value).reshape(-1)


def _mix_vectors(a: np.ndarray, g: np.ndarray, step: float) -> np.ndarray:
    """
    Return (1‑step)·a + step·(g/∑g).  Both a and g are non‑negative and sum(a)=1.
    """
    sum_g = g.sum()
    if sum_g <= 0:
        return a.copy()
    g_norm = g / sum_g
    return (1.0 - step) * a + step * g_norm

    # Helper for a fast convolution update when moving a mass δ from i→j.
    def apply_move(vec, conv_vec, i, j, delta):
        n_loc = vec.shape[0]
        # update vector
        vec_new = vec.copy()
        vec_new[i] -= delta
        vec_new[j] += delta

        # update convolution (O(n))
        conv_new = conv_vec.copy()
        idx = np.arange(n_loc)
        mask = (idx != i) & (idx != j)
        q = idx[mask]
        a_q = vec[q]

        conv_new[i + q] += 2.0 * (-delta) * a_q
        conv_new[j + q] += 2.0 * (+delta) * a_q

        # self‑pair terms
        conv_new[2 * i] += 2.0 * (-delta) * vec[i] + (-delta) * (-delta)
        conv_new[2 * j] += 2.0 * (+delta) * vec[j] + (+delta) * (+delta)

        # cross term (i,j)
        cross = i + j
        conv_new[cross] += 2.0 * (vec[i] * (+delta) + vec[j] * (-delta) + (-delta) * (+delta))
        return vec_new, conv_new

    # targeting the current maximal convolution entry
    def targeted_move(vec, conv_vec):
        n_loc = vec.shape[0]
        k = int(np.argmax(conv_vec))
        i_min = max(0, k - (n_loc - 1))
        i_max = min(n_loc - 1, k)
        i_candidates = np.arange(i_min, i_max + 1, dtype=int)
        j_candidates = k - i_candidates
        contrib = vec[i_candidates] * vec[j_candidates]
        src = int(i_candidates[np.argmax(contrib)])
        dst = np.random.randint(n_loc - 1)
        if dst >= src:
            dst += 1
        return src, dst

    # run a brief annealing loop (a few seconds)
    end_sa = time.time() + min(10.0, deadline - time.time())
    cur_vec = best_a.copy()
    cur_conv = np.convolve(cur_vec, cur_vec)
    cur_val = 2.0 * n * np.max(cur_conv)

    T_start = 0.2
    T_end = 1e-4
    it_sa = 0
    while time.time() < end_sa:
        progress = (time.time() - start_time) / (budget_s - 5.0)
        T = T_start * (1.0 - progress) + T_end * progress

        if np.random.rand() < 0.8:
            src, dst = targeted_move(cur_vec, cur_conv)
        else:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

        max_delta = cur_vec[src] * 0.5
        if max_delta < 1e-12:
            continue
        delta = np.random.rand() * max_delta

        vec_new, conv_new = apply_move(cur_vec, cur_conv, src, dst, delta)
        new_val = 2.0 * n * np.max(conv_new)

        improve = new_val < cur_val
        if improve or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_vec, cur_conv, cur_val = vec_new, conv_new, new_val
            if new_val < best_val:
                try:
                    official = evaluate_sequence(vec_new.tolist())
                    if official < best_val:
                        best_val = official
                        best_a = vec_new.copy()
                        print(f"[SA {it_sa:06d}] new best (official) = {best_val:.6f}")
                except Exception:
                    pass
        it_sa += 1

    # ------------------------------------------------------------------
    # 4.  Final safety check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] best evaluation reported by evaluate_sequence =", final_score)
    except Exception as exc:                       # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
