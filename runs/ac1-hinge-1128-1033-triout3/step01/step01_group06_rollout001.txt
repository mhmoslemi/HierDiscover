def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent flattening of the self‑convolution.

We minimise  2·n·max(a∗a)  (the sum of a is kept equal to 1) by a
projected‑gradient method.  The non‑smooth max is approximated by a
soft‑max with a temperature λ that is increased during the run.
The gradient of  Σ_k w_k·(a∗a)[k]  w.r.t. a is
    ∇ = 2 * corr(w, a)                     (valid‑mode correlation)
and the update a←a−η∇ is projected onto the probability simplex.
Fast FFT based convolution / correlation (via scipy.signal) keeps each
iteration O(n log n).  The routine runs for the allotted time and
returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.signal import fftconvolve, correlate

# ----------------------------------------------------------------------
# Simplex projection  (Euclidean projection onto {x≥0, sum x = 1})
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive -> uniform distribution
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        # ----- periodic logging ----------------------------------------
        if now - last_log > 30.0:
            elapsed = now - start_time
            print(f"[{iteration:7d}] t={elapsed:.1f}s best={best_val:.6f} step={step_eta:.4e} λ={lam:.3f}")
            last_log = now

        iteration += 1

    # ------------------------------------------------------------------
    # Final verification with the provided evaluate_sequence
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
