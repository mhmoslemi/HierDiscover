"""
Propose a sequence minimizing `evaluate_sequence`.

Algorithm
---------
1. Initialise a probability vector on the simplex using a small grid of
   generalized series a_i ∝ (γ)_i / i!  (the coefficients of (1‑x)^‑γ).  The
   classic γ=0.5 gives the “square‑root” shape; other values are tried and
   the best is kept as a starting point.
2. Run a projected Adam optimiser on the non‑negative simplex.
   * The loss is 2·n·max_k (a∗a)_k (sum(a)=1) – we minimise the maximum
     convolution entry.
   * Instead of the sub‑gradient at the active index only, we use a
     soft‑max approximation:
           p_k ∝ exp(β·((a∗a)_k – max(a∗a)))
     and the gradient becomes a weighted sum of the shifted vectors.
     The temperature β is increased slowly, turning the smooth gradient
     into the true sub‑gradient.
   * After each Adam step the vector is projected onto the probability
     simplex (Duchi et al. 2008).
3. Occasionally inject a tiny random jitter and, after a long period
   without improvement, restart from a perturbed best point.
4. Keep the best sequence observed until the time budget expires.

The code uses FFT‑based convolutions (O(n log n)) so that many thousands
of iterations fit comfortably into the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x>=0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex using the algorithm of Duchi et al."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: generate the (1‑x)^‑γ series truncated to length n
# ----------------------------------------------------------------------
def _gen_gamma_series(n: int, gamma: float) -> np.ndarray:
    """Return the normalised coefficients of (1-x)^-gamma, length n."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (gamma + k - 1) / k
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# FFT‑based convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper to obtain the next power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the linear convolution of a and b using a real FFT of length `size`.
    Returns a 1‑D array of length len(a)+len(b)-1.
    """
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine respects the given
    time budget and returns the best sequence found.
    """
    np.random.seed(seed)

    # length of the sequence (default from the previous best solution)
    n = kwargs.get("length", 5808)

    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1) initialise with a small grid over the gamma parameter
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    gammas = np.linspace(0.45, 0.55, 11)  # includes the classic 0.5

    for gamma in gammas:
        a = _gen_gamma_series(n, gamma)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
    # also try a uniform start and a few random Dirichlet vectors
    uniform = np.full(n, 1.0 / n)
    score_uni = evaluate_sequence(uniform.tolist())
    if score_uni < best_score:
        best_score = score_uni
        best_a = uniform.copy()

    for _ in range(3):
        rand = np.random.dirichlet(np.ones(n))
        s = evaluate_sequence(rand.tolist())
        if s < best_score:
            best_score = s
            best_a = rand.copy()

    print(f"[init] length={n}, best initial evaluation={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) set up Adam optimiser and soft‑max temperature schedule
    # ------------------------------------------------------------------
    lr = 0.15
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(best_a)
    v = np.zeros_like(best_a)

    # FFT size large enough for both a*a (2n‑1) and a*rev(p) (3n‑2)
    fft_sz = _next_pow2(3 * n - 2)

    iter_cnt = 0
    no_improve = 0

    while time.time() < deadline:
        # ---- convolution a*a -------------------------------------------------
        conv = _conv_fft(best_a, best_a, fft_sz)          # length 2n‑1
        max_val = conv.max()

        # ---- soft‑max weights over the convolution entries --------------------
        # temperature β grows slowly (from ~0.5 to about 10)
        beta_temp = 0.5 + min(10.0, iter_cnt * 5e-4)       # linear schedule
        diff = conv - max_val                              # all ≤ 0
        # exponentiation is safe because diff≤0
        exp_term = np.exp(beta_temp * diff)
        p = exp_term / exp_term.sum()                     # shape (2n‑1,)

        # ---- gradient of the smooth max --------------------------------------
        # grad = 2 * Σ_k p_k * shift_k(a)   ↔   2 * (a * rev(p)) slice
        p_rev = p[::-1]
        conv2 = _conv_fft(best_a, p_rev, fft_sz)          # length 3n‑2
        # slice indices n‑1 … n‑1+n‑1 (inclusive start, exclusive end)
        grad_slice = conv2[n - 1 : n - 1 + n][::-1]
        grad = 2.0 * grad_slice

        # ---- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (iter_cnt + 1))
        v_hat = v / (1.0 - beta2 ** (iter_cnt + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = best_a - step
        a_new = _project_simplex(a_new)

        # occasional tiny jitter to help escape flat regions
        if iter_cnt % 500 == 0 and iter_cnt > 0:
            jitter = np.random.randn(n) * (1e-4 / math.sqrt(iter_cnt + 1))
            a_new = _project_simplex(a_new + jitter)

        cur_score = evaluate_sequence(a_new.tolist())

        # ---- accept/reject ----------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            no_improve = 0
            print(f"[{iter_cnt:5d}] improvement: {best_score:.6f}")
        else:
            # simulated‑annealing style occasional acceptance
            # temperature decays roughly linearly with remaining time
            T = max(1e-4, (deadline - time.time()) / budget_s)
            if np.random.rand() < math.exp(-(cur_score - best_score) / (T + 1e-12)):
                best_a = a_new.copy()   # move without updating best_score
            no_improve += 1

        # ---- restart if stuck -------------------------------------------------
        if no_improve >= 2000:
            # perturb the current best and restart Adam moments
            a_restart = best_a + np.random.randn(n) * 0.01
            best_a = _project_simplex(a_restart)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{iter_cnt:5d}] restart after stagnation")

        iter_cnt += 1

    print(f"[final] best evaluation = {best_score:.6f} after {iter_cnt} iterations")
    return best_a.tolist()
