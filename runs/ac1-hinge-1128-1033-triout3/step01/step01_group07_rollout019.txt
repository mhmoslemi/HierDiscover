import time
import numpy as np
from math import sqrt, log10, ceil

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    """
    Compute the maximum entry of the linear convolution of *seq* with itself.
    Uses an FFT of the next power‑of‑two size for speed.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


def _score(seq: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a given sequence.
    Mirrors the logic used inside the evaluation harness but is much
    faster because it works on ``np.ndarray`` directly.
    """
    total = float(seq.sum())
    if total < 0.01:
        return np.inf
    max_conv = _conv_max(seq)
    n = seq.shape[0]
    return 2.0 * n * max_conv / (total * total)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative list of heights) with a
    low ``evaluate_sequence`` value.

    Strategy
    --------
    *   Use a short “core’’ that follows the optimal 1/√i decay.
    *   Append a long constant tail.  The tail adds a lot of ℓ¹‑mass
        while contributing only a modest amount to the maximal
        self‑convolution, thereby decreasing the objective.
    *   Exhaustively sweep a modest grid of core lengths, total
        lengths and tail heights (log‑spaced).  The search space is
        kept small enough to finish well under the 1000‑second budget.
    *   Finally perform a quick hill‑climbing refinement on the best
        tail height found.

    The routine returns a plain Python list ready for the evaluation
    harness.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1) Parameter grid
    # ------------------------------------------------------------------
    # allowed total length (larger than the previous 8k – up to 30k)
    MAX_TOTAL = 30000

    # core lengths – keep them relatively small; they dominate the
    # convolution peak but we want a decent ℓ¹‑mass from them.
    core_lengths = np.arange(10, 501, 10)            # 10,20,…,500

    # total lengths are multiples of the core length; we explore a few
    # scaling factors to get various tail sizes.
    scale_factors = np.array([5, 10, 15, 20, 30, 40, 50], dtype=int)

    best_val = np.inf
    best_seq = None
    best_params = None   # (core_len, total_len, delta)

    # iterate over the grid – the total number of candidates stays well
    # below a few‑thousand, easily fitting into the time budget.
    for core_len in core_lengths:
        # core pattern (unscaled)
        core = 1.0 / np.sqrt(np.arange(1, core_len + 1, dtype=np.float64))

        # pre‑compute the max‑convolution of the pure core – this is the
        # dominant part of the final max value.
        core_max = _conv_max(core)

        for factor in scale_factors:
            total_len = core_len * factor
            if total_len > MAX_TOTAL:
                continue
            tail_len = total_len - core_len
            if tail_len <= 0:
                continue

            # Upper bound for the tail height: we keep the extra tail‑square
            # contribution below the core peak to avoid blowing up the max.
            # safety_factor < 1 leaves a margin.
            safety_factor = 0.8
            delta_upper = sqrt(safety_factor * core_max / max(1, tail_len))
            # make sure the upper bound is not absurdly tiny (otherwise the
            # log‑grid would be degenerate)
            delta_upper = max(delta_upper, 1e-6)

            # log‑spaced tail heights
            log_h = np.linspace(-6, log10(delta_upper), 30)
            deltas = 10 ** log_h

            for delta in deltas:
                # build the candidate
                seq = np.concatenate([core, np.full(tail_len, delta, dtype=np.float64)])
                val = _score(seq)

                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    best_params = (core_len, total_len, delta)
                    print(
                        f"[grid] new best: {best_val:.6f} "
                        f"(core={core_len}, tot={total_len}, δ={delta:.2e})"
                    )

                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) Simple hill‑climbing refinement on the tail height
    # ------------------------------------------------------------------
    if best_seq is not None:
        core_len, total_len, delta = best_params
        tail_len = total_len - core_len
        core = 1.0 / np.sqrt(np.arange(1, core_len + 1, dtype=np.float64))

        cur_delta = delta
        cur_seq = best_seq.copy()
        cur_val = best_val

        start = time.time()
        iteration = 0
        while time.time() < deadline:
            iteration += 1
            # propose a new tail height – multiplicative Gaussian perturbation
            proposal = cur_delta * np.exp(np.random.randn() * 0.1)
            if proposal <= 0:
                continue
            # respect a slightly tighter upper bound to stay safe
            delta_limit = sqrt(0.9 * _conv_max(core) / max(1, tail_len))
            proposal = min(proposal, delta_limit)

            new_seq = np.concatenate([core, np.full(tail_len, proposal, dtype=np.float64)])
            new_val = _score(new_seq)

            # accept if improvement (simple greedy accept)
            if new_val < cur_val:
                cur_delta, cur_seq, cur_val = proposal, new_seq, new_val
                if new_val < best_val:
                    best_val, best_seq = new_val, new_seq.copy()
                    print(f"[refine] improved to {best_val:.6f} (δ={cur_delta:.2e})")
        print(f"final best value: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback – return something valid if the search failed
    # ------------------------------------------------------------------
    if best_seq is None:
        # uniform tiny sequence – safe, but will give a poor score
        n = 1000
        best_seq = np.full(n, 1e-3, dtype=np.float64)

    # Return a plain Python list for the harness
    return best_seq.tolist()
