"""
Propose a low‑score step‑function using a deterministic LP “flatten‑the‑convolution”
phase followed by a fast stochastic polishing phase.

The LP phase implements the iteration described in the reference:
    • keep the current vector f (non‑negative, sum = 1)
    • let M = max_k (f*f)[k]
    • solve the linear program
          maximise   Σ b_j
          subject to 0 ≤ b,
                     (f * b)_k ≤ M   for every convolution index k
      The feasible b is a direction that can be mixed with f without
      increasing the peak of the self‑convolution.
    • normalise b to a probability vector g, mix f ← (1‑t)·f + t·g
      (t is reduced if the mix does not keep the peak ≤ M).

The LP is built with a sparse Toeplitz‑type matrix so that the size
remains manageable (n≈800‑1600).  After a handful of LP iterations we
obtain a “flat” distribution for a modest length.  The distribution is
then interpolated to a larger length (≈5000) and refined by a cheap
simulated‑annealing/gradient local search (the same technique used in
the previous baseline).  The whole routine respects the 1000 s budget
and prints progress; the best sequence found is returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (used by both the LP and the stochastic phases)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex:
        { w ≥ 0, Σ w = 1 }.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape” with a power‑law decay from the boundaries.
        a[i] = (min(i, n‑1‑i) + 1)^{‑p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-12           # avoid exact zeros
    a /= a.sum()
    return a


def internal_score(a: np.ndarray) -> float:
    """
    Fast internal equivalent of `evaluate_sequence` for a vector that
    already satisfies Σ a_i = 1.
    Returns 2·n·max(conv(a, a)).
    """
    n = len(a)
    max_c = np.convolve(a, a).max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# LP utilities
# ----------------------------------------------------------------------


def build_convolution_matrix(f: np.ndarray) -> "scipy.sparse.csr_matrix":
    """
    Build the (2n‑1) × n matrix A such that (A @ b)[k] = (f * b)[k].
    Each column j contains the vector f shifted by j positions.
    """
    import scipy.sparse as sp

    n = f.shape[0]
    rows = []
    cols = []
    data = []
    f_list = f.tolist()
    for j in range(n):
        rows.extend(range(j, j + n))        # rows k = j + i , i = 0..n‑1
        cols.extend([j] * n)                # column index j repeated
        data.extend(f_list)                 # f[i] for i = 0..n‑1
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_step(current: np.ndarray, M: float) -> np.ndarray:
    """
    Perform a single linear‑programming step.
    Returns a new probability vector g (normalised) or None if the LP fails.
    """
    from scipy.optimize import linprog

    n = current.shape[0]

    # Build constraints (f * b) ≤ M
    A = build_convolution_matrix(current)
    rhs = np.full(2 * n - 1, M, dtype=float)

    # Bounds: b_j ≥ 0
    bounds = [(0, None)] * n

    # Objective: maximise Σ b_j  ⇒ minimise −Σ b_j
    c = -np.ones(n)

    res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    b = res.x
    s = b.sum()
    if s <= 1e-12:
        return None
    g = b / s                      # normalise to sum = 1
    return g

                    mask = (idx >= 0) & (idx < n)
                    grad = np.zeros(n)
                    grad[mask] = 2.0 * cur[idx[mask]]
                    cand = cur - grad_step * grad
                    cand = np.maximum(cand, 0.0)
                    cand = simplex_projection(cand)   # re‑normalise to sum = 1

                cand_score = internal_score(cand)

                # Simulated‑annealing acceptance
                accept = False
                if cand_score < cur_score:
                    accept = True
                else:
                    prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
                    if np.random.rand() < prob:
                        accept = True

                if accept:
                    cur = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        print(f"[Refine] N={N:5d} it={it:5d} new best {best_score:.6f}")

                # cooling
                temperature *= cooling_rate
                grad_step *= cooling_rate
                if temperature < 1e-6:
                    temperature = 1e-6

    # ------------------------------------------------------------------
    # 4.  Fallback (if everything failed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_score = internal_score(best_seq)
        print(f"Fallback to uniform of length {n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final sanity checks and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)   # stay inside evaluate_sequence limits
    print(f"Search finished – best internal score = {best_score:.6f} "
          f"(length {len(best_seq)})")
    # The external evaluator will recompute the true score.
    return best_seq.tolist()
