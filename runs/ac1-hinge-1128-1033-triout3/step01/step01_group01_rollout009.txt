"""
Propose a candidate step‑function (list of non‑negative heights).

The algorithm is a high‑precision projected Adam optimiser that minimises
the maximum entry of the self‑convolution of a probability vector.
The optimisation is performed on the log‑sum‑exp smooth approximation
of the max‑operator; the temperature (α) is slowly increased so that the
approximation becomes tighter over time.  Convolutions and the required
correlations are computed with FFTs (O(n log n)) which allows us to run
many thousands of iterations within the 1000 s budget.  The current
best vector is kept and returned after the time limit (or when the
target bound 1.503 is reached).

All values are kept in the simplex {p_i ≥ 0, Σ p_i = 1}, which makes the
objective scale‑invariant.  The final sequence is clipped to the range
[0, 1000] as required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_power_of_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    M = 2·n·max(conv(a,a)) / (sum(a)**2).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The routine stops a few seconds early
        to avoid forced termination.
    kwargs : dict
        Ignored – present for API compatibility.

    Returns
    -------
    list[float]
        The best sequence found (values are already non‑negative and
        lie in the interval [0, 1000]).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine problem size -------------------------------------------------
    # Use the length of a previously discovered construction if available,
    # otherwise fall back to the default length 1153.
    # ------------------------------------------------------------------
    default_len = 1153
    try:
        n = len(height_sequence_1)  # type: ignore[name‑defined]
    except Exception:
        n = default_len

    # ------------------------------------------------------------------
    # 2. FFT planning (length must accommodate a correlation of length 3n‑2)
    # ------------------------------------------------------------------
    fft_len = _next_power_of_two(3 * n - 2)

    # ------------------------------------------------------------------
    # 3. Initial probability vector (sum = 1)
    # ------------------------------------------------------------------
    if rng.random() < 0.3 and "height_sequence_1" in globals():
        # start from the known construction (interpolated if lengths differ)
        old = np.array(height_sequence_1, dtype=float)  # type: ignore[name‑defined]
        old = np.clip(old, 0.0, 1000.0)
        if old.sum() > 0:
            old = old / old.sum()
        if len(old) != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, len(old))
            x_new = np.linspace(0.0, 1.0, n)
            p = np.interp(x_new, x_old, old)
            p = p / p.sum()
        else:
            p = old.copy()
    else:
        # random Dirichlet‑like start
        p = rng.random(n)
        p = p / p.sum()

    best_p = p.copy()
    # initial exact objective (sum(p)=1 → denominator = 1)
    F = np.fft.rfft(p, n=fft_len)
    conv_full = np.fft.irfft(F * F, n=fft_len)
    conv = conv_full[: 2 * n - 1]
    best_score = float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # 4. Adam optimiser hyper‑parameters
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(p)
    v = np.zeros_like(p)

    lr = 0.05          # initial learning rate for Adam
    lr_decay = 0.9995  # global decay of the learning‑rate

    alpha = 5.0        # temperature for the smooth‑max
    alpha_mul = 1.003  # increase factor per iteration
    alpha_max = 5000.0

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0  # safety margin
    iteration = 0
    target_score = 1.503

    while time.time() < deadline:
        iteration += 1

        # ----- FFT of the current vector ---------------------------------
        F = np.fft.rfft(p, n=fft_len)

        # ----- self‑convolution -----------------------------------------
        conv_full = np.fft.irfft(F * F, n=fft_len)
        conv = conv_full[: 2 * n - 1]          # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- smooth approximation of the max (log‑sum‑exp) ----------
        z = alpha * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)            # stabilised exponentials
        w = exp_z / exp_z.sum()               # soft‑max weights (sum = 1)

        # ----- gradient of the smoothed objective -----------------------
        w_rev = w[::-1]
        G = np.fft.rfft(w_rev, n=fft_len)
        corr_full = np.fft.irfft(F * G, n=fft_len)
        corr = np.real(corr_full[: 3 * n - 2])   # length 3n‑2

        idx = 2 * n - 2 - np.arange(n)          # indices needed for gradient
        grad = 2.0 * corr[idx]

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        p_new = p - step
        p_new = _simplex_projection(p_new)

        # ----- occasional random jitter to escape shallow minima -------
        if iteration % 500 == 0:
            jitter = rng.normal(scale=0.001, size=n)
            p_new = _simplex_projection(p_new + jitter)

        p = p_new

        # ----- hyper‑parameter schedules --------------------------------
        lr *= lr_decay
        alpha = min(alpha * alpha_mul, alpha_max)

        # ----- keep the best solution -----------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"Iter {iteration:6d} – new best M = {best_score:.6f}")

        # ----- early exit ------------------------------------------------
        if best_score <= target_score:
            print(f"Target reached (M ≤ {target_score}) after {iteration} iterations.")
            break

    # ------------------------------------------------------------------
    # 6. Return the best sequence (clip to the allowed range)
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    return result
