"""
Propose a sequence of non‑negative heights that (hopefully) improves the
known bound  ≈ π/2 ≈ 1.5708.

Algorithm
---------
* Initialise the coefficients from the analytically optimal “square‑root”
  series (the coefficients of (1‑x)^{‑1/2}) or, if available, from the global
  variable ``height_sequence_1`` (the best sequence found so far by the
  harness).
* Optimise a smoothed version of the objective
      F(a) = 2·n·max_k (a∗a)_k
  by replacing the max with a soft‑max (log‑sum‑exp) with a temperature
  parameter β.  For a given β the gradient of the smoothed objective can be
  written as a cross‑correlation of ``a`` with the soft‑max weights and is
  computed efficiently with one FFT.
* Perform Adam‑style updates on the probability simplex (projected onto the
  simplex after each step).  β is increased slowly so that the algorithm
  first sees a smooth landscape and eventually focuses on the true max.
* Occasionally inject a tiny random jitter and, if no improvement has been
  seen for a while, restart from the analytical start (with a small random
  perturbation).  The best sequence seen during the whole run is returned.

# Helper: projection onto the probability simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Duchi et al. (2008) Euclidean projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic “square‑root” series (coefficients of (1‑x)^{-1/2})
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """Return the first n coefficients of (1‑x)^{-1/2}, normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2 * (k - 1) + 1) / (2 * k)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Fast convolution of a with itself (real FFT, zero‑padded)
# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a∗a (full convolution) using an FFT of length L."""
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Cross‑correlation a ⋆ w (full) using the same FFT length L.
# The result is indexed so that result[i] = Σ_j a_j w_{i+j}.
# ----------------------------------------------------------------------
def _cross_correlation_fft(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """Return the cross‑correlation a⋆w (full)."""
    A = np.fft.rfft(a, L)
    W = np.fft.rfft(w, L)
    corr_full = np.fft.irfft(A * np.conj(W), L)
    # slice the part that corresponds to i = 0 … n‑1
    offset = a.shape[0] - 1
    return corr_full[offset : offset + a.shape[0]]


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) beats the
    π/2 bound.  The routine runs for at most ``budget_s`` seconds and
    prints progress information.

    Parameters
    ----------
    seed : int
        Random seed for stochastic parts.
    budget_s : int
        Time budget in seconds.
    **kwargs : dict
        Optional keyword arguments.  The key ``length`` can be used to
        specify the length of the sequence (default 8192).

    Returns
    -------
    list[float]
        The best sequence discovered (elements sum to ≈ 1).
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose the length and the initial vector
    # ------------------------------------------------------------------
    if "length" in kwargs:
        n = int(kwargs["length"])
    else:
        # power‑of‑two length is convenient for FFT speed
        n = 8192

    # if the harness provides a previously found sequence, reuse its length
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.size > 0:
            a = seq0.copy()
            a = a / a.sum()
            n = a.shape[0]          # respect the given length
        else:
            a = _sqrt_series(n)
    else:
        a = _sqrt_series(n)

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # Adam parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    t_iter = 0

    # Soft‑max temperature (β in the analysis).  We start gentle and grow.
    beta = 5.0               # low β → smooth approximation of the max
    beta_max = 200.0
    beta_inc = 0.01

    # Step‑size (learning‑rate) for Adam
    lr = 0.05

    # Restart / jitter control
    no_improve = 0
    restart_after = 4000    # iterations without improvement → restart

    # Pre‑compute an FFT length that safely contains all required products
    L = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_seq = a.copy()
    # Direct evaluation (sum(a) ≈ 1) is 2·n·max(conv)
    best_conv = _self_convolution_fft(a, L).max()
    best_score = 2.0 * n * best_conv
    print(f"[init] n={n}, initial evaluation={best_score:.6f}")

    deadline = time.time() + budget_s - 0.5   # tiny safety margin
    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- 1. current convolution and max value -----
        conv = _self_convolution_fft(a, L)
        max_conv = conv.max()

        # ----- 2. evaluate -----
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{iteration:6d}] new best {best_score:.6f}")
        else:
            no_improve += 1

        # ----- 3. soft‑max weights for the active constraints -----
        # w_i ∝ exp(beta * (conv_i - max_conv)); the maximal entry gets weight 1
        w_raw = np.exp(beta * (conv - max_conv))
        w_sum = w_raw.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            # fall back to a hard max (should rarely happen)
            w = np.zeros_like(conv)
            w[conv == max_conv] = 1.0 / np.count_nonzero(conv == max_conv)
        else:
            w = w_raw / w_sum

        # ----- 4. gradient of the smoothed objective (cross‑correlation) -----
        grad = 2.0 * _cross_correlation_fft(a, w, L)

        # ----- 5. Adam update -----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a = _project_simplex(a_new)

        # ----- 6. occasional jitter to escape plateaus -----
        if iteration % 500 == 0 and iteration > 0:
            jitter = np.random.randn(n) * (5e-5 / math.sqrt(iteration + 1))
            a = _project_simplex(a + jitter)

        # ----- 7. gradually sharpen the soft‑max -----
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ----- 8. restart if stuck -----
        if no_improve >= restart_after:
            a = _sqrt_series(n)
            a = _project_simplex(a + np.random.randn(n) * 1e-3)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            no_improve = 0
            print(f"[{iteration:6d}] restart triggered")

        iteration += 1

    # ------------------------------------------------------------------
    # Final report (use the supplied evaluate_sequence for exact consistency)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best evaluation = {final_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
