"""
Propose a candidate step‑function.

The algorithm searches over simple power‑law shapes
    a_i = (i+1)^p   for i = 0 … n‑1
subject to the constraint a_i ≤ 1000 (the evaluation routine clips larger
values, so we keep them below the bound).  For a fixed exponent p the length
n is chosen as the largest integer with n^p ≤ 1000, i.e.
    n = floor(1000^{1/p}).

For a pure power law the evaluation score 2·n·max(conv)/(sum^2) has a
closed‑form limit that decreases with p, but the bound n ≤ 1000^{1/p}
makes the discrete case deviate from the asymptotic value.  We therefore
enumerate a fine grid of exponents (0.5 … 5.0) and compute the exact score
with the supplied ``evaluate_sequence`` function, keeping the best
sequence found.  The search is deterministic, very cheap (the longest
convolution is ≈ 31·log 31), and finishes in a fraction of a second,
well within the 1000 s budget.

The best sequence discovered is returned as a plain list of floats.
"""
import time
import math
import numpy as np

# tiny helper – pure‑Python version of the evaluation (faster for tiny n)
def _fast_score(seq):
    """Return the same value as evaluate_sequence for non‑negative seq."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.size
    # full convolution (length 2*n‑1)
    conv = np.convolve(a, a)
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """
    Search over power‑law step functions a_i = (i+1)^p (with a_i ≤ 1000)
    and return the sequence that minimizes ``evaluate_sequence``.
    """
    # the evaluator is already imported in the execution environment
    # (we also keep a fast pure‑Python version for speed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # search grid – denser near the region that gives the smallest scores
    # (exponents 1.5 … 3.5 are the most interesting)
    exponents = np.concatenate((
        np.linspace(0.5, 1.5, 21, endpoint=False),   # coarse low‑p part
        np.linspace(1.5, 3.5, 41, endpoint=False),   # fine middle part
        np.linspace(3.5, 5.0, 16)                    # coarse high‑p tail
    ))
    # ensure deterministic order
    exponents = np.unique(np.round(exponents, 6))

    for p in exponents:
        if time.time() > deadline:
            break
        # maximal length respecting the 1000‑cap
        n = int(math.floor(1000.0 ** (1.0 / p)))
        if n < 2:
            continue
        # generate the sequence a_i = (i+1)^p  (floats)
        seq = [(float(i + 1) ** p) for i in range(n)]
        # safety – clip just in case of rounding errors
        seq = [min(1000.0, max(0.0, x)) for x in seq]

        # evaluate (use the provided evaluator; fallback to fast version)
        try:
            cur_score = evaluate_sequence(seq)   # type: ignore[name-defined]
        except Exception:                       # fall back if unavailable
            cur_score = _fast_score(seq)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq
            print(f"[improve] p={p:.3f}, n={n}, score={best_score:.6f}")

    # -----------------------------------------------------------------
    # (Optional) a tiny local refinement: move a tiny amount of mass
    # between two positions while keeping the total sum constant.
    # This sometimes squeezes a few extra digits out of the best shape.
    # -----------------------------------------------------------------
    if best_seq is not None:
        seq = np.array(best_seq, dtype=np.float64)
        n = len(seq)
        # simple stochastic hill climbing
        while time.time() < deadline:
            i, j = np.random.randint(n, size=2)
            if i == j or seq[i] <= 0.0:
                continue
            # transfer a small random fraction of a[i] to a[j]
            delta = np.random.exponential(scale=1e-3) * seq[i]
            delta = min(delta, seq[i])
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            cur_score = _fast_score(new_seq.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = new_seq.tolist()
                seq = new_seq
                print(f"[refine] score={best_score:.6f}")

    if best_seq is None:
        # fallback – a minimal non‑zero uniform vector (should never happen)
        best_seq = [1.0, 1.0]

    print(f"[final] best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq
