```
def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, cvxpy as cp, math

    deadline = time.time() + budget_s - 5

    # Choose n maybe 1500 based on time
    n = 1500
    # generate binomial prefix
    coeffs = generate_binomial_prefix(n)
    f = coeffs.copy()

    # Compute initial objective
    best_seq = f.tolist()
    best_score = evaluate_sequence(best_seq)
    print("[init] n=", n, "score=", best_score)

    # Precompute convolution matrix builder
    def build_A(vec):
        n = len(vec)
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # j = k-i
            # For j in [i_min,i_max], f_{k-j}=vec_{k-j}
            # This sets A[k,j] = vec[k-j]
            # Equivalent:
            for j in range(i_min, i_max+1):
                A[k, j] = vec[k - j]
        return A

    # Loop
    iteration = 0
    while time.time() < deadline:
        # compute M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Build constraint matrix
        A = build_A(f)  # shape (2n-1, n)
        # Solve LP
        g = cp.Variable(n)
        constraints = [A @ g <= M * np.ones(2*n-1), g >= 0]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception as e:
            print("LP solve error:", e)
            break
        if g.value is None:
            print("LP failed to find solution")
            break
        g_vec = np.maximum(0.0, np.array(g.value, dtype=np.float64))
        sum_g = g_vec.sum()
        sum_f = f.sum()
        if sum_g <= sum_f * (1 + 1e-6):
            # no improvement, stop
            print("[stop] No sum improvement iteration", iteration)
            break

        # try blending
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            f_new = (1 - t) * f + t * g_vec
            # evaluate
            seq_new = f_new.tolist()
            try:
                score_new = evaluate_sequence(seq_new)
            except Exception:
                continue
            if score_new < best_score - 1e-6:
                print(f"[improve] it{iteration} t={t:.3f} score={score_new:.6f}")
                f = f_new
                best_seq = seq_new
                best_score = score_new
                improved = True
                break
        if not improved:
            print("[no improve] iteration", iteration)
            break

        iteration += 1
        # possibly early stop if objective near target
        if best_score < 1.55:  # optional early break
            pass

    # final clean
    best_seq = [float(max(0.0, x)) for x in best_seq]
    return best_seq
```

The evaluation function is  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

If we keep the maximal auto‑convolution value `M = max(a*a)` small while increasing the
`ℓ₁`‑norm `S = Σ a_i`, the score drops.
The well‑known “constant‑convolution’’ (central binomial) sequence satisfies  
`max(a*a)=1` and `S ≈ 1.128·√n`, giving the limit `π/2 ≈ 1.5708`.

A better sequence can be obtained by the classic **linear‑programming iteration**:

1. start from the binomial prefix `f`.
2. let `M = max(f*f)`.
3. solve a linear program  

   \[
   \max\; Σ g_i \qquad
   \text{s.t.}\; (f * g)_k \le M\;(k=0..2n-2),\; g_i\ge 0 .
   \]

   Because the convolution with a *fixed* `f` is linear in `g`, this is a genuine LP.
4. replace `f ← (1‑t)·f + t·g` with a small step size `t`.  
   The new sequence keeps `M` (or makes it smaller) while its sum grows.
5. repeat until the time budget expires.

Every LP iteration pushes mass to indices where `f` is already tiny,
therefore the self‑convolution peak `M` can drop, and the objective improves.
A tiny random bias is added to the LP objective (`ε·w·g`) to avoid
getting stuck at the same extreme point.

The implementation below follows this scheme, uses **CVXPY** for the LP,
and falls back to a lightweight stochastic hill‑climb if CVXPY is unavailable.
It respects the 1000 s time budget, prints progress, and finally returns the
best sequence found.

```python
"""Propose a step‑function with a low evaluation score.

The algorithm:
1. Initialise with the optimal “constant‑convolution’’ (central‑binomial)
   sequence of length n (default 1500).
2. Repeatedly
   • compute M = max(conv(f,f));
   • solve a linear programme  max Σ g_i  s.t.  conv(f,g) ≤ M , g ≥ 0;
     a tiny random linear term (ε·w·g) breaks ties so the solution g differs
     from f;
   • blend f ← (1‑t)·f + t·g with a line‑search over t∈{0.5,0.3,…,0.01};
     keep the blend only if the true score improves.
3. Stop when time runs out or no further improvement is possible.

If CVXPY cannot be imported, a simple random‑walk hill‑climber (the previous
method) is used as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the constant‑convolution (central‑binomial) coefficients
# ----------------------------------------------------------------------
def generate_binomial_prefix(max_n: int):
    """Return the first max_n coefficients  b_k = C(2k,k)/4**k."""
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0  # b_0
    coeffs[0] = cur
    for k in range(1, max_n):
        # recurrence: b_{k} = b_{k-1} * (2k-1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) minimises
    evaluate_sequence(...).  The function runs for at most *budget_s* seconds.
    """
    import math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise with the binomial prefix
    # ------------------------------------------------------------------
    N = 1500                     # length of the working vector (tunable)
    f = generate_binomial_prefix(N).astype(np.float64)   # numpy array
    conv_f = np.convolve(f, f)
    M = float(conv_f.max())      # current maximal auto‑convolution
    best_seq = f.tolist()
    best_score = evaluate_sequence(best_seq)
    print(f"[init] n={N}, M={M:.6f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Try to import CVXPY – the LP engine
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
        have_cvx = True
    except Exception:
        have_cvx = False
        print("[info] CVXPY unavailable – falling back to simple hill‑climb.")

    # ------------------------------------------------------------------
    # 3.  Core optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    max_iter = 30                     # safety guard
    epsilon = 1e-7                    # tiny bias in the LP objective
    while time.time() < deadline and iteration < max_iter:
        # ---- recompute M (may have changed after previous blend) ----
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        if have_cvx:
            # ---------- Linear programme: maximise Σ g_i ----------
            n = len(f)
            g = cp.Variable(n)

            # convolution f * g (full linear convolution)
            conv_fg = cp.conv(f, g)           # yields a vector of length 2n‑1
            constraints = [
                conv_fg <= M * np.ones(2 * n - 1),   # element‑wise bound
                g >= 0
            ]

            # add a tiny random linear term to break degeneracy
            w = np.random.uniform(-1.0, 1.0, size=n)
            objective = cp.Maximize(cp.sum(g) + epsilon * w @ g)

            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
            except Exception as exc:
                print("[lp] solver failed:", exc)
                break

            if g.value is None:
                print("[lp] no solution – aborting LP stage.")
                break
            g_vec = np.maximum(0.0, np.asarray(g.value, dtype=np.float64))

            # if the LP does not increase the sum, we have reached a fixed point
            if g_vec.sum() <= f.sum() * (1.0 + 1e-9):
                print("[lp] sum(g) ≤ sum(f); stop iteration.")
                break
        else:
            # --------- Fallback: tiny random perturbation ----------
            # generate a candidate by perturbing a random index
            g_vec = f.copy()
            idx = np.random.randint(N)
            delta = np.random.randn() * 0.05 * max(g_vec[idx], 1e-12)
            g_vec[idx] = max(0.0, g_vec[idx] + delta)

        # ---- 4.  Line‑search blending of f and g ----
        improved = False
        # try larger steps first – they give quick gains when possible
        for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            candidate = (1.0 - t) * f + t * g_vec
            # evaluate with the provided (clamped) function
            try:
                score = evaluate_sequence(candidate.tolist())
            except Exception:
                continue

            if score < best_score - 1e-9:
                best_score = score
                best_seq = candidate.tolist()
                f = candidate               # accept the new sequence
                improved = True
                print(f"[iter{iteration}] t={t:.3f}  score={score:.6f}")
                break

        if not improved:
            # No blending step improved the score – stop the loop.
            print("[stop] No improvement from blending; terminating.")
            break

        iteration += 1

    # ------------------------------------------------------------------
    # 5.  Final sanitisation – enforce non‑negativity and limits 0‑1000
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    print(f"[final] iterations={iteration}, final score={best_score:.6f}, "
          f"length={len(final_seq)}")
    return final_seq
```
