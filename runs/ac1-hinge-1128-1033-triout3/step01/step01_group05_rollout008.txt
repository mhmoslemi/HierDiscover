def central_binomial_prefix(n):
    coeffs = np.empty(n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, n):
        cur = cur * (2*k-1)/(2*k)
        coeffs[k] = cur

        i_max = min(k, n - 1)

        i_vals = np.arange(i_min, i_max+1)
        j_vals = k - i_vals  # corresponds to column indices
        A[k, j_vals] = f[i_vals]
    return A

    # ensure f sum = target
    f = f * (target_sum / np.sum(f))

"""
propose_candidate
-----------------

Searches a step‑function (list of non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

by a Frank‑Wolfe‑style iteration:

1.  Initialise a vector `f` (central‑binomial, uniform or random) of length *n*
    and scale it so that   sum(f) = √(2 n).  With this normalisation the
    objective equals the maximal self‑convolution `M = max(conv(f,f))`.

2.  While time permits, solve a linear program  

        maximise   Σ g_j
        subject to g_j ≥ 0
                   (f * g)_k ≤ M   for all k = 0 … 2n‑2

    (the convolution is linear in `g`).  The LP gives a direction `g₀`
    with a larger ℓ₁‑norm than `f`.  `g₀` is re‑scaled to the same ℓ₁‑norm
    as `f` (still √(2 n)) producing `g`.

3.  Mix the current point towards `g`:

        f ← (1‑t)·f + t·g      with several trial steps t∈{0.1,0.25,0.5,0.75,0.9}

    The mixture keeps the ℓ₁‑norm fixed.  If the new maximal self‑convolution
    `M'` is smaller, the step is accepted; otherwise the iteration stops.

4.  The procedure is repeated for a few different lengths (200‑500) and a few
    initialisations; the best sequence (lowest real score) is returned.

The algorithm stays within the 1000 s budget, uses only `numpy` and `scipy`
(and the supplied `evaluate_sequence`) and obeys all problem constraints.
"""

import time
import numpy as np
from scipy.optimize import linprog


def central_binomial_prefix(n: int) -> np.ndarray:
    """
    Returns the first `n` terms of the sequence
        b_k = C(2k,k) / 4**k
    using the recurrence  b_{k+1} = b_k * (2k+1) / (2(k+1)).
    """
    coeffs = np.empty(n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, n):
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that
        (A @ g)[k] = (f * g)[k]      (linear convolution)
    for every k = 0 … 2n‑2.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # each row k contains a shifted copy of f
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1, dtype=np.int64)
        j_vals = k - i_vals                # column indices that receive f[i]
        A[k, j_vals] = f[i_vals]
    return A


def _optimize_fixed_length(f: np.ndarray,
                           max_iter: int = 30,
                           t_vals: tuple = (0.1, 0.25, 0.5, 0.75, 0.9),
                           deadline: float = 1e9) -> tuple:
    """
    Perform the Frank‑Wolfe relaxation for a fixed length.
    Returns the best vector found and the associated maximal self‑convolution.
    """
    n = f.size
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / np.sum(f))          # enforce the normalisation

    M = np.max(np.convolve(f, f))
    best_f, best_M = f.copy(), M

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # linear program: maximise Σ g_j  (⇒ minimise -Σ g_j)
        A = build_conv_matrix(f)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)      # -Σ g_j
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            # LP failed – exit the inner loop
            print(f"[opt] LP failure at iter {it}: {res.message}")
            break

        g0 = res.x
        sum_g0 = np.sum(g0)
        if sum_g0 <= 0:
            break

        # scale g0 back to the required ℓ₁‑norm
        g = g0 * (target_sum / sum_g0)

        improved = False
        for t in t_vals:
            f_new = (1.0 - t) * f + t * g
            M_new = np.max(np.convolve(f_new, f_new))
            if M_new < best_M - 1e-9:          # genuine improvement
                best_f = f_new.copy()
                best_M = M_new
                f = f_new
                M = M_new
                improved = True
                print(f"[opt] iter {it:02d} t={t:.2f} M={M:.6f}")
                break
        if not improved:
            # no step size gave improvement → stop iterating
            print("[opt] no improvement, terminating")
            break
        if best_M <= 1.01:                     # cannot beat the theoretical bound 1
            break

    return best_f, best_M


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """
    Main entry point used by the evaluator.  Tries a few problem sizes and
    initialisations, runs the optimisation routine, and returns the best
    sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float('inf')

    # candidate lengths – larger gives a better theoretical bound but costs more LP solves
    lengths = [200, 300, 400, 500]
    init_kinds = ('central', 'uniform', 'random')

    for n in lengths:
        if time.time() >= deadline:
            break
        for kind in init_kinds:
            if time.time() >= deadline:
                break

            # ----- initialise the start vector -----
            if kind == 'central':
                f0 = central_binomial_prefix(n)
            elif kind == 'uniform':
                f0 = np.ones(n, dtype=np.float64)
            else:                               # random
                f0 = np.random.rand(n)

            # ----- run the optimisation -----
            f_opt, _ = _optimize_fixed_length(f0,
                                              max_iter=40,
                                              t_vals=(0.1, 0.25, 0.5, 0.75, 0.9),
                                              deadline=deadline)

            # ----- evaluate the real objective -----
            cand = f_opt.tolist()
            try:
                score = evaluate_sequence(cand)
            except Exception:
                continue

            if score < best_score:
                best_score = score
                best_seq = cand
                print(f"[global] new best {best_score:.6f} (n={n}, init={kind})")

    # fallback – should never happen, but keep a safe answer
    if best_seq is None:
        best_seq = central_binomial_prefix(200).tolist()

    # final clamp to the allowed range (the evaluator does this again, but we keep it tidy)
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return best_seq
