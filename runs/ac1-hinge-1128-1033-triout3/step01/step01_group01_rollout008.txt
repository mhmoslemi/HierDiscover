def solve_lp(f, M, sum_target):
    n = len(f)
    m = 2*n-1
    # Build A matrix shape (m,n)
    row_idxs = []
    col_idxs = []
    data = []
    for i in range(n):
        # column i contributes to rows i..i+n-1
        start = i
        end = i + n  # exclusive
        # rows = np.arange(start, end)
        row_idxs.extend(range(start, end))
        col_idxs.extend([i]*n)
        data.extend(f.tolist())
    A = scipy.sparse.csr_matrix((data, (row_idxs, col_idxs)), shape=(m,n))
    # constraints A @ g <= M * np.ones(m)
    b = np.full(m, M)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    g = res.x
    return g, res.success, res.fun

    alpha = 5.0           # starting smoothing factor
    alpha_mul = 1.001

    # tracking
    best_p = p.copy()
    conv = np.convolve(p, p)
    best_score = 2 * n * np.max(conv)
    print(f"Initial score: {best_score:.6f}")
    # time limit
    deadline = time.time() + budget_s - 5.0
    iter = 0
    last_improve_iter = 0
    while time.time() < deadline:
        iter += 1
        # compute conv and smoothed max
        conv = np.convolve(p, p)
        max_conv = np.max(conv)
        # compute softmax weights
        # shift for numerical stability
        shift = np.max(alpha * conv)

        w = exp_weights / exp_weights.sum()

        corr = np.convolve(p, w[::-1])
        idx = 2 * n - 2 - np.arange(n)

        # projection onto simplex
        p_new = _simplex_projection(p_new)

        if iter % 500 == 0:

        p = p_new
        # update schedules
        lr *= lr_decay
        alpha *= alpha_mul
        # evaluate objective
        cur_score = 2 * n * np.max(np.convolve(p, p))
        if cur_score < best_score:

            best_p = p.copy()

            print(f"Iter {iter:6d} – restart after stagnation")

    result = np.clip(best_p, 0.0, 1000.0).tolist()

"""
Propose a candidate step‑function (list of non‑negative heights).

The algorithm starts from the best known construction (if available) or
from a uniform distribution, then performs a long run of projected Adam
optimisation on a smooth approximation of the L∞‑norm of the
self‑convolution.  The smoothing (soft‑max) parameter is slowly increased
so that the optimisation first sees a smooth landscape and then focuses
on the true max.  After each gradient step the vector is projected onto
the probability simplex to keep feasibility.  Periodic random jitter
helps to escape shallow plateaus.  The best sequence seen during the
time budget is returned.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the simplex {x | x_i >= 0, Σ x_i = 1}.
    Implementation follows Duchi et al. (ICML 2008).
    """
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Main routine
# ----------------------------------------------------------------------

    seed : int

        Desired time budget (seconds).  The function stops a few seconds

        Ignored – kept for compatibility with the surrounding framework.

        evaluation routine will apply the same clipping internally).

    p = _simplex_projection(p + rng.normal(scale=0.001, size=n))

    # --------------------------------------------------------------
    # 3. Adam optimiser state.
    # --------------------------------------------------------------
    m = np.zeros_like(p)          # first moment
    v = np.zeros_like(p)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # --------------------------------------------------------------
    # 4. Scheduling parameters.
    # --------------------------------------------------------------
    lr = 0.05                     # base learning rate for Adam
    lr_decay = 0.9999             # slow exponential decay
    alpha = 5.0                   # smoothing factor for soft‑max (low ⇒ smooth)
    alpha_mul = 1.001             # increase smoothness gradually

    # --------------------------------------------------------------
    # 5. Book‑keeping.
    # --------------------------------------------------------------
    conv = np.convolve(p, p)
    best_score = 2.0 * n * np.max(conv)        # exact objective (sum(p)=1)
    best_p = p.copy()
    print(f"initial score: {best_score:.6f}")

    last_improve = 0
    iter_cnt = 0
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 6. Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline:
        iter_cnt += 1

        # ----- self‑convolution -----
        conv = np.convolve(p, p)               # length 2n‑1
        max_conv = np.max(conv)

        # ----- smooth approximation of max (soft‑max) -----
        # Using the usual log‑sum‑exp trick for numerical stability.
        z = alpha * conv
        z_max = np.max(z)
        w = np.exp(z - z_max)                 # shifted exponentials
        w /= w.sum()                          # soft‑max weights (sum to 1)

        # ----- gradient of the smoothed objective -----
        # ∂/∂p_i Σ_k w_k (p * p)_k = 2 Σ_k w_k p_{k-i}
        # This is a correlation of p with the reversed weight vector.
        corr = np.convolve(p, w[::-1])        # length 3n‑2
        idx = 2 * n - 2 - np.arange(n)        # mapping described above
        grad = 2.0 * corr[idx]

        # ----- Adam update + projection onto simplex -----
        m = beta1 * m + (1.0 - beta1) * grad

        alpha *= alpha_mul

        if best_score <= 1.45:               # theoretical optimum is 1.0
