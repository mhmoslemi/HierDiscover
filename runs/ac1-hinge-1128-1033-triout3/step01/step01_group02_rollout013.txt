        conv_f = np.convolve(best_f, best_f)
        M = conv_f.max()
        # Build convolution matrix B

            break

iteration = 0
while time.time() < deadline:
    iteration += 1
    # Compute self-convolution and max
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    # Build convolution matrix B: use toeplitz
    # B shape (2*n-1, n)
    ccol = np.concatenate([f, np.zeros(n-1)])
    crow = np.concatenate([f[:1], np.zeros(n-1)])
    B = toeplitz(ccol, crow)  # dense ndarray
    # Prepare LP:
    c = -np.ones(n)  # maximize sum -> minimize -sum
    # Add small epsilon to ensure feasibility
    b = np.full(2*n-1, M * (1.0 + 1e-12))
    # Solve
    res = linprog(c, A_ub=B, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        print(f"[lp] iteration {iteration} failed: {res.message}")
        break
    g = res.x
    sum_g = g.sum()
    sum_f = f.sum()
    # Check improvement
    if sum_g <= sum_f * (1.0 + 1e-9):
        print(f"[lp] iteration {iteration} no sum increase")
        break
    # Scale g to match sum_f
    g_scaled = g * (sum_f / sum_g)
    # Mix f and g_scaled
    t = 0.5
    new_f = (1 - t) * f + t * g_scaled
    # Compute new score
    new_score = evaluate_sequence(new_f.tolist())
    if new_score < best_score:
        best_score = new_score
        f = new_f
        print(f"[improve] iter {iteration} new best score={best_score:.6f}")
    else:
        # try reducing t
        for t2 in (0.3, 0.1):
            new_f2 = (1 - t2) * f + t2 * g_scaled
            new_score2 = evaluate_sequence(new_f2.tolist())
            if new_score2 < best_score:
                best_score = new_score2
                f = new_f2
                print(f"[improve] iter {iteration} t={t2} new best score={best_score:.6f}")
                break
        else:
            print(f"[no improve] iter {iteration}")
            break

"""
propose_candidate
-----------------

Implements an iterative improvement of the Catalan (central‑binomial) step‑function.
The algorithm works as follows:

1. Build the classic Catalan sequence (c₀=1,  cₖ₊₁ = cₖ·(2k+1)/(2(k+1))) of length *n*
   and scale it so that its ℓ₁‑norm equals √(2 n).  At this scale the evaluation
   score equals the maximal self‑convolution value (≈π/2 ≈ 1.5708).

2. Repeatedly solve a linear programme

       maximise   Σ gⱼ
       subject to  (f * g)ₖ ≤ M   for all k
                    gⱼ ≥ 0

   where *f* is the current sequence,  M = max(f*f) and (f * g) denotes the linear
   convolution (a Toeplitz matrix applied to *g*).  The LP finds a feasible
   “direction” *g* that can carry more total mass without breaking the current
   convolution bound.

3. Rescale *g* to the fixed ℓ₁‑norm ( √(2 n) ) and move partway towards it:

       f ← (1−t)·f + t·g_scaled   (t≈0.5, reduced if no improvement)

   Because (f·g)≤M, this convex combination never raises the bound and usually
   reduces it.  The process is repeated until the time budget expires or no
   progress is possible.

4. A tiny stochastic refinement (random coordinate perturbations) is performed
   for any remaining time.

The LP is solved with SciPy’s *highs* interior‑point solver; the convolution
matrix is built as a dense Toeplitz matrix (size ≈2n×n) and passed to the
solver as a CSR sparse matrix.  The whole routine respects the 1000 s
budget and always returns a non‑negative sequence.
