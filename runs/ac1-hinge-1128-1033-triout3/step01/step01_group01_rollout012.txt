"""Propose a candidate step‑function.

The optimizer combines several ideas:
* multiple random restarts within the overall time budget,
* a projected Adam optimisation on a soft‑max approximation of
  the max‑convolution,
* periodic random jitter and optional sparsification,
* a short stochastic local‑search phase on the best solution found.

All iterates stay on the probability simplex (non‑negative,
sum‑to‑one) so the evaluation function reduces to 2·n·max(conv).
The routine stops a few seconds before the budget expires and
returns the best list of heights discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008)."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------

    """Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s``
    seconds and returns the best sequence (as a plain Python list) it
    discovered.  The implementation uses a projected Adam optimiser
    with a soft‑max smoothing of the max‑convolution, many random
    restarts and a final stochastic local‑search refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size (use the length of a known construction if present)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 1153  # default length used by the benchmark

    # ------------------------------------------------------------------
    # 2. Global bookkeeping
    # ------------------------------------------------------------------
    best_score = np.inf
    best_seq = None

    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 3. Main loop – keep launching independent optimisation runs
    # ------------------------------------------------------------------
    run_id = 0
    while time.time() < deadline:
        run_id += 1

        # ----- 3.1 Initial distribution ---------------------------------
        # With some probability start from the previously known construction
        if rng.random() < 0.15 and 'height_sequence_1' in globals():

            if total > 0.0:
                p = seq / total
            else:

        else:

            power = 1.0 + rng.random() * 2.0   # between 1 and 3
            p = rng.random(n) ** power
            p /= p.sum()

        m = np.zeros_like(p)
        v = np.zeros_like(p)

        lr = 0.2                     # base learning rate
        lr_decay = 0.9997

        # smoothing of the max‑operator
        alpha = 5.0
        alpha_mul = 1.005

        it = 0
        while time.time() < deadline:
            # compute self‑convolution
            conv = np.convolve(p, p)               # length 2n‑1

            # smooth approximation of the maximum (log‑sum‑exp)
            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)              # stability shift
            w = exp_z / exp_z.sum()                # soft‑max weights, sum to 1

            # gradient of the smoothed objective:
            #   ∂/∂p_i Σ_k w_k (p * p)_k = 2 Σ_k w_k p_{k-i}
            w_rev = w[::-1]
            corr = np.convolve(p, w_rev)           # length 3n‑2
            idx = 2 * n - 2 - np.arange(n)

            # ----- Adam update -------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional sparsification – keep only the most significant entries
            if it % 500 == 0 and it > 0:
                # keep ~20 % largest components
                thresh = np.percentile(p_next, 80)
                mask = p_next >= thresh
                if mask.sum() >= 2:
                    p_next = p_next * mask
                # renormalise (fallback to uniform if everything vanished)
                s = p_next.sum()
                if s == 0.0:
                    p_next = rng.random(n)
                p_next /= p_next.sum()

            # enforce feasibility (simplex)
            p = _simplex_projection(p_next)

            alpha *= alpha_mul
            lr *= lr_decay

"""
Propose a candidate step‑function.

The optimiser works inside the allotted time budget with several
independent restarts.  Each run performs a projected Adam optimisation
on a smoothed (log‑sum‑exp) approximation of the max‑convolution.
The iterate is always projected onto the probability simplex, i.e.
non‑negative and summing to 1, so the evaluation function reduces to
``2·n·max(conv(p,p))``.  The algorithm incorporates

* a gradually tightening soft‑max (parameter α),
* occasional random jitter,
* optional sparsification of tiny components,
* a short stochastic hill‑climbing phase on the best solution found.

The best sequence discovered before the time limit is returned as a
plain Python list (values are already clipped to ``[0,1000]``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0 , Σ x_i = 1}.
    Implements the algorithm of Duchi et al., ICML 2008.
    """
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
