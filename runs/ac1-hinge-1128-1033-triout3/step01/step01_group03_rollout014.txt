import time
import numpy as np

def _project_simplex(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        # fallback to uniform distribution
        n = v.shape[0]
        return np.full(n, 1.0 / n)
    return v / s

def _sqrt_series(n):
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k-1] * (2.0 * (k-1) + 1.0) / (2.0 * k)
    w /= w.sum()
    return w

def _next_pow_two(x):
    return 1 << ((x - 1).bit_length())

def _objective_from_conv(a, conv):
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * conv.max() / (sum_a * sum_a))

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimize the evaluation function using a smooth approximation of the max‑convolution.
    The algorithm starts from the analytical (1‑x)^(-1/2) sequence, then performs
    Adam‑style updates on the probability simplex.  The max is approximated by a
    soft‑max (log‑sum‑exp) with a temperature that is decreased during the run,
    allowing the optimiser to escape plateaux and to converge to a tighter bound.
    Several random perturbations and simple restarts are added to avoid stagnation.
    """
    np.random.seed(seed)
    n = kwargs.get("length", 5808)
    deadline = time.time() + budget_s - 0.5

    # Initialise with the analytically optimal “square‑root” shape
    a = _sqrt_series(n)
    best_a = a.copy()
    best_score = _objective_from_conv(best_a, np.convolve(best_a, best_a))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.05          # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature schedule
    temp_factor = 0.5    # start with 0.5·max(conv)
    temp_decay = 0.9995  # multiplicative decay per iteration
    min_temp = 1e-10

    # Pre‑compute FFT length (power of two)
    conv_len = 2 * n - 1
    fft_len = _next_pow_two(conv_len)

    it = 0
    no_improve = 0
    while time.time() < deadline:
        # ---- full convolution via FFT ----
        fa = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(fa * fa, fft_len)[:conv_len]
        max_conv = conv_full.max()
        cur_score = _objective_from_conv(a, conv_full)

        # track best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{it:6d}] new best: {best_score:.6f}")
        else:
            no_improve += 1

        # ---- soft‑max weights for the active part of the max ----
        temp = max(max_conv * temp_factor, min_temp)
        # shift by max for numerical stability
        shifted = conv_full - max_conv
        exp_w = np.exp(shifted / temp)
        w = exp_w / exp_w.sum()

        # ---- gradient of the smoothed objective (Adam) ----
        a_rev = a[::-1]
        fw = np.fft.rfft(w, fft_len)
        frev = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(frev * fw, fft_len)[: n + conv_len - 1]  # length 3n-2
        grad = 2.0 * grad_full[n-1 : n-1 + n]   # slice of length n

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a = _project_simplex(a_new)

        # occasional random jitter to diversify search
        if it % 2000 == 0 and it > 0:
            jitter = np.random.randn(n) * (1e-4 / np.sqrt(it + 1))
            a = _project_simplex(a + jitter)

        # simple restart if we have been stagnant for a while
        if no_improve >= 5000:
            a = _sqrt_series(n)
            a = _project_simplex(a + np.random.randn(n) * 1e-3)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{it:6d}] restart triggered")

        # decay temperature
        temp_factor *= temp_decay
        it += 1

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    # Return plain Python list
    return best_a.tolist()

"""
Optimise the “max‑convolution” objective with a smooth approximation.

The optimiser starts from the analytically optimal (1‑x)^(-1/2) sequence
and then performs Adam‑style updates on the probability simplex.
A soft‑max (log‑sum‑exp) is used to obtain a differentiable surrogate for the
maximum of the self‑convolution; its temperature is slowly reduced so the
method first explores a smooth landscape and finally concentrates on the
true max.  Simple random jitter and occasional restarts avoid stagnation.
The implementation uses FFTs for fast full convolutions (O(n log n) per
iteration) and a cheap projection onto the simplex (clip + renormalise).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clip negatives to zero and renormalise to sum‑one."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:                       # fall back to the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s

# ----------------------------------------------------------------------
# Analytic ‑√‑series (coefficients of (1‑x)^(-1/2)), normalised to a sum of 1
# ----------------------------------------------------------------------
def _sqrt_series(n: int) -> np.ndarray:
    """First n coefficients of (1‑x)^(-1/2) normalised to sum = 1."""
    w = np.empty(n, dtype=np.float64)
    w[0] = 1.0
    for k in range(1, n):
        w[k] = w[k - 1] * (2.0 * (k - 1) + 1.0) / (2.0 * k)
    w /= w.sum()
    return w

# ----------------------------------------------------------------------
# Helper: next power of two (used for zero‑padding in FFT)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Fast full convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolve_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full linear convolution a*b using a pre‑allocated FFT length."""
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len)
    return conv[: a.shape[0] + b.shape[0] - 1]

# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The algorithm uses a smooth approximation of the
    max‑convolution and Adam updates on the simplex.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = kwargs.get("length", 5808)           # keep the same scale as the reference
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # initialise with the known “square‑root” shape
    # ------------------------------------------------------------------
    a = _sqrt_series(n)                      # already sums to 1
    best_a = a.copy()
    # exact evaluation of the current candidate
    conv0 = np.convolve(best_a, best_a)
    best_score = float(2.0 * n * conv0.max() / (best_a.sum() ** 2))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam hyper‑parameters (tuned for stability)
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.02                               # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # ------------------------------------------------------------------
    # soft‑max temperature schedule (starts relatively high, decays)
    # ------------------------------------------------------------------
    temp_factor = 0.5                        # start with 0.5·max(conv)
    temp_decay = 0.9995
    min_temp = 1e-10

    # ------------------------------------------------------------------
    # pre‑compute an FFT length that is a power of two (fast transforms)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = _next_pow_two(conv_len)

    it = 0
    no_improve = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) full self‑convolution (FFT based, O(n log n))
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(fa * fa, fft_len)[:conv_len]
        max_conv = conv_full.max()
        cur_score = float(2.0 * n * max_conv / (a.sum() ** 2))

        # --------------------------------------------------------------
        # keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{it:6d}] new best: {best_score:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 2) soft‑max weights (smooth approximation of the max)
        # --------------------------------------------------------------
        temp = max(max_conv * temp_factor, min_temp)
        # shift by the maximum for numerical stability
        shifted = conv_full - max_conv
        w = np.exp(shifted / temp)
        w /= w.sum()

        # --------------------------------------------------------------
        # 3) gradient of the smoothed objective (via FFT)
        # --------------------------------------------------------------
        a_rev = a[::-1]                       # reversed copy of a
        fw = np.fft.rfft(w, fft_len)
        frev = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(frev * fw, fft_len)[: n + conv_len - 1]   # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]                         # slice of length n

        # --------------------------------------------------------------
        # 4) Adam update on the simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step

            v.fill(0.0)
            no_improve = 0
            print(f"[{it:6d}] restart triggered")

        # --------------------------------------------------------------
        # decay the soft‑max temperature
        # --------------------------------------------------------------
        temp_factor *= temp_decay

        it += 1

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    # --------------------------------------------------------------
    # return the best sequence found (plain Python list)
    # --------------------------------------------------------------
    return best_a.tolist()
