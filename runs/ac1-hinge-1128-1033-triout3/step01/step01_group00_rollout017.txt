"""
Propose a low‑score step‑function.

The approach has three stages:

1. **Deterministic seeds** – generate many candidates (uniform random,
   classic windows, exponential decays, U‑shapes) for a few lengths and
   keep the best few according to the exact objective
   2·n·max(conv(a,a))/sum(a)^2 (the sum is 1 after normalisation).

2. **Soft‑max gradient descent on the simplex** – for each elite seed we
   minimise the soft‑max of the self‑convolution:
        f(a)=log∑_k exp(α·conv_k(a))
   using exponentiated‑gradient updates (mirror descent) that stay on the
   probability simplex.  The gradient of f is a weighted sum of the
   sub‑gradients of the individual convolution entries and can be
   computed with a single FFT‑based convolution.

3. **Random perturbations** – occasionally move a small amount of mass
   between two positions to escape shallow minima.

All heavy convolutions are performed with FFTs (O(n log n)) so the
procedure can run many thousands of iterations within the 1000 s budget.
The best sequence found (as a plain Python list) is returned.
"""

import time
import numpy as np

# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    if x <= 0:
        return 1
    return 1 << (int(np.ceil(np.log2(x))))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_max_fft(a: np.ndarray):
    """
    Compute the linear (zero‑padded) self‑convolution and its maximum
    using FFTs.  Returns (max_value, full_convolution, FFT_of_a).
    """
    n = a.shape[0]
    L = next_pow2(3 * n)               # long enough for both convolutions
    Fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    max_c = float(conv.max())
    return max_c, conv, Fa, L


def gradient_softmax(a: np.ndarray, Fa: np.ndarray, L: int, w: np.ndarray):
    """
    Gradient of the soft‑max objective:
        g_i = 2 * Σ_k w_k * a_{k-i}
    Computed via a single FFT convolution.
    """
    n = a.shape[0]
    # reverse the weight vector
    w_rev = w[::-1]
    Fw = np.fft.rfft(w_rev, n=L)
    # convolution a * w_rev (linear, zero‑padded)
    conv_full = np.fft.irfft(Fa * Fw, n=L)[: n + (2 * n - 1) - 1]  # length 3n‑2
    # indices 2n-2‑i give the required sums
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad


def score_from_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a vector with sum=1."""
    n = a.shape[0]
    max_c = float(np.convolve(a, a).max())
    return 2.0 * n * max_c
