def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math, random, sys
    rng = np.random.default_rng(seed)
    start = time.time()
    end_time = start + budget_s - 5.0

    # set n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 1153

    best_p = None
    best_score = np.inf

    # Helper to evaluate objective
    def objective(p):
        return 2.0 * n * np.max(np.convolve(p, p))

    # Define gradient of smoothed max (softmax)
    def softmax_grad(p, alpha):
        conv = np.convolve(p, p)
        # compute softmax weights
        z = alpha * conv
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        w = exp_z / exp_z.sum()
        # correlation
        w_rev = w[::-1]
        corr = np.convolve(p, w_rev)
        idx = 2 * n - 2 - np.arange(n)
        grad = 2.0 * corr[idx]
        return grad

    # Parameter schedule
    # We'll start with relatively high smoothing alpha0=1.0, then gradually increase.
    # We'll use EG with learning rate eta0 = 0.5 decaying.

    # number of restarts
    restarts = 5
    # allocate some time for restarts
    for r in range(restarts):
        if time.time() > end_time:
            break
        # init p
        if r == 0 and 'height_sequence_1' in globals():
            seq = np.array(height_sequence_1, dtype=float)
            seq = np.clip(seq, 0.0, 1000.0)
            total = seq.sum()
            if total > 0:
                p = seq / total
            else:
                p = rng.random(n)
                p = p / p.sum()
        else:
            p = rng.random(n)
            p = p / p.sum()
        # create local copy
        cur_p = p.copy()
        cur_alpha = 1.0
        cur_lr = 0.5

        # Perform gradient descent iterations for this restart
        # We'll loop until some iteration count or time limited
        # Let's allow up to 3000 iterations or time.
        for it in range(5000):
            if time.time() > end_time:
                break
            # compute gradient
            grad = softmax_grad(cur_p, cur_alpha)
            # Exponentiated gradient update
            cur_p = cur_p * np.exp(-cur_lr * grad)
            # renormalize
            cur_p_sum = cur_p.sum()
            if cur_p_sum == 0:
                # fallback to uniform
                cur_p = np.full(n, 1.0/n)
            else:
                cur_p = cur_p / cur_p_sum
            # update schedules
            cur_lr *= 0.9995
            cur_alpha *= 1.001  # slowly increase
            # evaluate objective
            cur_score = objective(cur_p)
            if cur_score < best_score:
                best_score = cur_score
                best_p = cur_p.copy()
                print(f"Restart {r}, iter {it}, new best {best_score:.6f}")
            # occasional jitter
            if it % 250 == 0 and it > 0:
                noise = rng.normal(scale=0.005, size=n)
                cur_p = cur_p + noise
                cur_p = np.clip(cur_p, 0, None)
                cur_p /= cur_p.sum()
        # next restart
    # After restarts, perform simulated annealing starting from best_p if we have time left
    if best_p is None:
        # fallback
        best_p = rng.random(n)
        best_p = best_p / best_p.sum()
        best_score = objective(best_p)
    # Simulated annealing
    cur_p = best_p.copy()
    cur_score = best_score
    # temperature schedule: start high (maybe 0.1) then exponential decay
    T0 = 0.1
    T_end = 1e-5
    # determine total SA steps based on remaining time
    remaining = end_time - time.time()
    # if still time left
    if remaining > 0:
        max_steps = int(remaining * 200)  # assume ~200 steps per second; adjust
        # approximate step time we can compute runtime
        # We'll also adapt step count by while loop
        step = 0
        while time.time() < end_time and step < max_steps:
            step += 1
            # temperature decreasing linearly on log scale
            t_frac = step / max_steps
            T = T0 * ((T_end / T0) ** t_frac)  # exponential decay
            # propose move: choose i,j
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # choose delta fraction of p_i
            delta = rng.random() * 0.1 * cur_p[i]  # up to 10% of p_i
            if delta <= 0:
                continue
            new_p = cur_p.copy()
            new_p[i] -= delta
            new_p[j] += delta
            # ensure nonnegatives
            if new_p[i] < 0:
                continue
            # compute objective
            new_score = objective(new_p)
            diff = new_score - cur_score
            if diff < 0 or rng.random() < math.exp(-diff / T):
                cur_p = new_p
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_p = cur_p.copy()
                    print(f"SA step {step}, new best {best_score:.6f}")
            # occasionally re-normalize (should preserve sum)
            # also small jitter occasionally
            if step % 1000 == 0:
                # random perturbation
                noise = rng.normal(scale=0.001, size=n)
                cur_p = np.clip(cur_p + noise, 0, None)
                cur_p /= cur_p.sum()
        # end while
    # final result: clip and convert to list
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Final best score: {best_score:.6f}")
    return result

"""
Propose a candidate step‑function (list of non‑negative heights).

The routine combines three ideas:

1. **Projected exponentiated‑gradient descent** on a smooth
   approximation of the objective  `max(conv(p,p))`.  The smooth
   approximation is the log‑sum‑exp with a temperature `alpha`
   that is gradually increased, turning the soft‑max into the
   true max.  Each EG step keeps the vector non‑negative and
   normalised to a probability simplex.

2. **Random jitter** (Gaussian noise) every few hundred iterations
   to escape shallow basins.

3. **Simulated‑annealing fine‑tuning** starting from the best
   distribution found by the gradient phase.  Small mass transfers
   between two coordinates are accepted if they improve the
   objective or with a probability that decreases with a temperature
   schedule.

The algorithm runs until the supplied time budget expires (a small
safety margin is left) and returns the best sequence discovered.
All numbers are non‑negative, sum to one and are safely clipped to
the interval required by ``evaluate_sequence``.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex {x | x_i >= 0, sum(x) = 1}
    (Duchi et al. 2008).  Returned array has the same shape as ``v``.
    """
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed gradient of max(conv(p,p))
# ----------------------------------------------------------------------
def _softmax_gradient(p: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the gradient of the smoothed objective
        f(p) = (1/alpha) * log( sum_k exp(alpha * (p*p)_k) )
    w.r.t. the probability vector ``p``.
    """
    n = p.shape[0]
    conv = np.convolve(p, p)                     # length 2n‑1
    z = alpha * conv
    # stable soft‑max
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()                      # soft‑max weights

    # correlation with reversed weights gives the gradient before the factor 2
    w_rev = w[::-1]
    corr = np.convolve(p, w_rev)                # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)              # pick the right slice
    grad = 2.0 * corr[idx]                       # gradient vector
    return grad


# ----------------------------------------------------------------------
# Objective (independent of scaling of p)
# ----------------------------------------------------------------------
def _objective(p: np.ndarray, n: int) -> float:
    """Exact evaluation of 2·n·max(conv(p,p)) (sum(p)=1)."""
    return 2.0 * n * np.max(np.convolve(p, p))


# ----------------------------------------------------------------------
# Main entry point expected by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the routine stops a few seconds early).
    kwargs : dict
        Ignored – kept for API compatibility.

    Returns
    -------
    list[float]
        A sequence of non‑negative heights (already normalised to sum‑1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin so the system does not kill us
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Choose length n
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # type: ignore[name‑defined]
    except Exception:
        n = 1153                                   # default length used previously

    # ------------------------------------------------------------------
    # 2. Helper for evaluating a candidate (fast, no clipping)
    # ------------------------------------------------------------------
    def eval_obj(p):
        return _objective(p, n)

    # ------------------------------------------------------------------
    # 3. Gradient‑descent phase (exponentiated gradient + projection)
    # ------------------------------------------------------------------
    best_p = None
    best_score = np.inf

    # Hyper‑parameter families for different starts – a tiny diversity boost
    hyper_sets = [
        (5.0, 1.003, 0.5, 0.9995),   # (alpha0, alpha_mul, lr0, lr_decay)
        (2.0, 1.005, 0.8, 0.999),   # a cooler schedule
        (8.0, 1.002, 0.3, 0.9998),  # more aggressive smoothing
    ]

    # Number of random restarts – we keep iterating until time runs out
    restart = 0
    while time.time() < deadline:
        # pick hyper‑parameters cyclically
        alpha, alpha_mul, lr, lr_decay = hyper_sets[restart % len(hyper_sets)]

        # --------------------------------------------------------------
        # Initialise p
        # --------------------------------------------------------------
        if restart == 0 and 'height_sequence_1' in globals():
            seq = np.array(height_sequence_1, dtype=float)   # type: ignore[name‑defined]
            seq = np.clip(seq, 0.0, 1000.0)
            total = seq.sum()
            if total > 0.0:
                p = seq / total
            else:
                p = rng.random(n)
                p = p / p.sum()
        else:
            p = rng.random(n)
            p = p / p.sum()

        # --------------------------------------------------------------
        # Run the EG optimiser
        # --------------------------------------------------------------
        iteration = 0
        while time.time() < deadline:
            iteration += 1

            # gradient of the smooth approximation
            grad = _softmax_gradient(p, alpha)

            # exponentiated‑gradient update (all entries stay non‑negative)
            p = p * np.exp(-lr * grad)

            # re‑normalise to the simplex
            sum_p = p.sum()
            if sum_p == 0.0:
                p = np.full(n, 1.0 / n)
            else:
                p = p / sum_p

            # schedules
            lr *= lr_decay
            alpha *= alpha_mul

            # occasional random jitter to shake the iterate
            if iteration % 250 == 0:
                noise = rng.normal(scale=0.005, size=n)
                p = np.clip(p + noise, 0.0, None)
                p = p / p.sum()

            # evaluate exact objective (no smoothing)
            cur_score = eval_obj(p)
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                print(f"[GD] restart {restart} iter {iteration:5d} – new best {best_score:.6f}")

            # early exit if we already beat the target
            if best_score <= 1.503:
                break

        # finished one restart – move to the next
        restart += 1
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing fine‑tuning (only if time permits)
    # ------------------------------------------------------------------
    if best_p is None:
        best_p = rng.random(n)
        best_p = best_p / best_p.sum()
        best_score = eval_obj(best_p)

    cur_p = best_p.copy()
    cur_score = best_score

    # temperature schedule (exponential decay)
    T_start = 0.1
    T_end = 1e-6
    # estimate how many SA steps we can afford (~2000 steps per second)
    sa_steps_budget = int(max(0, (deadline - time.time()) * 2000))

    for step in range(sa_steps_budget):
        # linear index for temperature on a log‑scale
        t_frac = step / max(1, sa_steps_budget - 1)
        T = T_start * ((T_end / T_start) ** t_frac)

        # propose moving a small amount of mass from i to j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 10 % of the source mass
        delta = rng.random() * 0.1 * cur_p[i]
        if delta <= 0.0:
            continue
        new_p = cur_p.copy()
        new_p[i] -= delta
        new_p[j] += delta
        # safety – numerical errors
        if new_p[i] < 0.0:
            continue

        new_score = eval_obj(new_p)
        diff = new_score - cur_score
        if diff < 0.0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            cur_p = new_p
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_p = cur_p.copy()
                print(f"[SA] step {step:07d} – new best {best_score:.6f}")
        # occasional global jitter to keep diversity
        if step % 10000 == 0 and step > 0:
            noise = rng.normal(scale=0.001, size=n)
            cur_p = np.clip(cur_p + noise, 0.0, None)
            cur_p = cur_p / cur_p.sum()

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ------------------------------------------------------------------
    result = np.clip(best_p, 0.0, 1000.0).tolist()
    print(f"Final best score: {best_score:.6f} (n={n})")
    return result
