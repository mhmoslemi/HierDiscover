def simplex_projection(v):
    # v is np.ndarray
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Sub‑gradient descent on the max‑self‑convolution objective.

The evaluation of a step sequence a is  2·n·max(conv(a,a))  (the sum of a is
normalised to 1, therefore the denominator in ``evaluate_sequence`` is 1).
The function   f(a) = max_k (a*a)[k]   is convex‑like but non‑smooth.
We use a simple sub‑gradient method:

*   Find the index k* where the convolution attains its current maximum.
*   The sub‑gradient of max(conv) at a is the gradient of conv[k*],
    which is   g_i = 2·a[k*−i]   whenever the partner index lies inside the
    vector and 0 otherwise.
*   Take a step a ← a – η·g and project back onto the probability simplex
    (a_i ≥ 0, Σ a_i = 1).
*   The step‑size η is decreased linearly from a few ×10⁻¹ to 10⁻⁴.
*   Every few hundred iterations a tiny random mass‑transfer is applied
    (similar to a simulated‑annealing jump) to escape possible plateaus.

The convolution is computed with a real FFT (O(n log n)) and the projection
onto the simplex uses the O(n log n) algorithm of Duchi et al.  The method is
fast enough to run many thousands of iterations on a sequence of a few
thousand entries within the 1000 s budget.

The best sequence encountered during the run is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: probability‑simplex projection (Duchi et al.)
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safeguard – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # final renormalisation (to counter rounding errors)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ------------------------------------------------------------
# Helper: full convolution via real FFT (fast for many iterations)
# ------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Returns the full (length 2·n‑1) convolution a*a using an FFT.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    # minor negative rounding errors are clipped to zero
    np.maximum(conv, 0, out=conv)
    return conv
