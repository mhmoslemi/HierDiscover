"""
Propose a step‑function using a smooth “soft‑max” approximation of the
maximum of the self‑convolution and projected (or exponentiated) gradient
descent on the probability simplex.  The algorithm starts from the best
known sequence (the global variable ``height_sequence_1`` if present,
otherwise a random U‑shape), then repeatedly:

* evaluates the exact objective 2·n·max(conv(a,a))
* builds a soft‑max weight vector p ∝ exp(α·conv(a,a))
* computes the gradient of the smooth objective:
        ∇F_i = 2·∑_k p_k·a_{i+k}   (zero‑padding outside the array)
* performs a projected gradient step on the simplex (with a simple
  momentum term) or, with some probability, a random mass‑transfer move
* slowly increases the soft‑max temperature α so that the optimisation
  focuses more on the true peak, and restarts with a small perturbation
  if no improvement has been seen for a long while.

The routine respects a time budget (default 1000 s), prints progress,
and finally returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex { w >= 0, sum(w) = 1 }.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """
    Exact value returned by ``evaluate_sequence`` for a vector a that
    is already normalised to sum(a) == 1.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Main proposal routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence found (as a plain ``list[float]``).
    """
    np.random.seed(seed)
    # small safety margin so we finish before the external timeout
    deadline = time.time() + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Initialise a feasible point (probability simplex, sum == 1)
    # --------------------------------------------------------------
    try:
        # use the best sequence from the previous run if it exists
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        s = init.sum()
        if s > 0:
            a = init / s
        else:
            raise ValueError
        n = a.shape[0]
        print(f"Initialised from existing height_sequence_1, length {n}")
    except Exception:
        # fall back to a simple “U‑shape” – generally a good seed
        n = 5808                       # length of the previous best construction
        d = np.arange(n)
        d = np.minimum(d, n - 1 - d)   # distance to nearer edge
        p0 = 0.6                        # exponent that gave a decent seed earlier
        a = (d + 1) ** (-p0)
        a = a / a.sum()
        print(f"Initialised from synthetic U‑shape, length {n}")

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"Start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Hyper‑parameters
    # --------------------------------------------------------------
    alpha = 1.0                # soft‑max temperature (low → smooth)
    alpha_mul = 1.001          # increase it slowly each iteration
    max_alpha = 500.0          # we never need it larger than this
    momentum = np.zeros_like(a)
    beta = 0.9                 # momentum factor
    last_improve = 0           # iteration number of the last improvement
    it = 0

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- evaluate current objective (exact) -----
        conv = np.convolve(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- keep the global best -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = it
            print(f"Iter {it:6d}: new best score = {best_score:.6f}")

        # ----- occasionally try a cheap random mass‑transfer move -----
        if np.random.rand() < 0.15:
            i, j = np.random.choice(n, 2, replace=False)
            if a[i] > 0.0:
                # move up to 30 % of the source mass
                delta = a[i] * np.random.rand() * 0.3
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = cand / cand.sum()          # stays on the simplex
                a = cand
                continue

        # ----- gradient step based on a smooth approximation of max -----
        # increase the soft‑max temperature gradually
        alpha = min(alpha * alpha_mul, max_alpha)

        # soft‑max weights on the convolution entries
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_w = np.exp(scaled - max_scaled)      # numerically stable
        p = exp_w / exp_w.sum()                  # shape (2n‑1,)

        # gradient of  (1/α)·log∑exp(α·conv)  w.r.t. a
        # ∇_i ≈ 2·∑_k p_k·a_{i+k}   (zero padding outside the array)
        # this can be obtained via a convolution with a reversed copy of a
        temp = np.convolve(p, a[::-1])           # length 3n‑2
        # extract the needed n entries (the “reversed” slice)
        grad = 2.0 * temp[(n - 1)::-1][:n]

        # ----- momentum & adaptive step size -----
        momentum = beta * momentum + (1 - beta) * grad
        gnorm = np.linalg.norm(momentum)
        if gnorm > 0:
            # step size chosen so that a typical coordinate moves by ~O(0.1)
            step = 0.5 / gnorm
        else:
            step = 0.0

        # ----- projected gradient update on the simplex -----
        cand = a - step * momentum
        cand = simplex_projection(cand)

        # safeguard: numerical issues can occasionally produce NaNs
        if np.any(np.isnan(cand)):
            cand = np.random.rand(n)
            cand = cand / cand.sum()

        a = cand

        # ----- restart if we are stuck for a long time -----
        if it - last_improve > 25000:
            # perturb the current best a little to escape a flat region
            noise = np.random.rand(n) * 1e-3
            a = best_seq + noise
            a = a.clip(min=0.0)                  # keep non‑negative
            a = a / a.sum()
            momentum = np.zeros_like(a)
            last_improve = it
            print(f"Iter {it:6d}: stagnation detected – restart with small noise")

        # occasional progress report
        if it % 2000 == 0:
            elapsed = time.time() + budget_s - deadline - (budget_s - 5.0)
            print(f"Iter {it:6d}: current score = {cur_score:.6f}, best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Return the best sequence found
    # --------------------------------------------------------------
    print(f"\nSearch finished after {it} iterations. Best score = {best_score:.6f}")
    # The external evaluator will clip and normalise again; we simply return
    # the raw (already normalised) probabilities.
    return best_seq.tolist()
