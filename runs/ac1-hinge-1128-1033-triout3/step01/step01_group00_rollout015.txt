"""
Propose a low‑score step‑function using a parametric “edge‑decay” family
(a_i ∝ (c + distance_to_nearest_edge)^‑p) and a projected sub‑gradient
descent that respects symmetry and the simplex constraint.  The algorithm
first performs a fast random sweep over the two parameters (p, c) to find
a promising seed, then refines it with a deterministic gradient step
based on the sub‑gradient of the maximal convolution entry.  Simulated
annealing style acceptance and occasional random mass‑transfer moves
let the search escape shallow minima.  All operations are implemented with
FFT‑based convolutions for O(n log n) speed, enabling many iterations
within the 1000 s budget.
"""

import time
import numpy as np

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def symmetric(v: np.ndarray) -> np.ndarray:
    """Force a vector to be symmetric: (v + rev(v))/2."""
    return (v + v[::-1]) * 0.5

def edge_decay_vector(n: int, p: float, c: float) -> np.ndarray:
    """
    Construct a symmetric vector a_i ∝ (c + d_i)^(-p),
    where d_i = min(i, n-1-i) is distance to nearest edge.
    The vector is normalised to sum == 1 and symmetrised.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (c + d) ** (-p) + 1e-12   # avoid exact zero
    a /= a.sum()
    return symmetric(a)

def conv_max_fft(a: np.ndarray) -> float:
    """
    Compute max_{k} (a * a)[k] using FFT (real‑valued input).
    """
    n = a.shape[0]
    size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())

def score_from_normed(a: np.ndarray) -> float:
    """Score for a vector already normalised to sum == 1."""
    n = a.shape[0]
    return 2.0 * n * conv_max_fft(a)

    The routine respects the time budget, prints progress and returns the best
    sequence (as a plain Python list) it found.
    """

    # ------------------------------------------------------------------

    lengths = [5000, 6000, 8000]

        if best_seq is not None:

Propose a low‑score step‑function using a parametric “edge‑decay” family
(a_i ∝ (c + distance_to_nearest_edge)^‑p) together with a projected
sub‑gradient descent that respects symmetry and the simplex constraint.
The algorithm first performs a fast random sweep over the two parameters
(p, c) to find a good seed, then refines it with a deterministic gradient
step based on the sub‑gradient of the maximal self‑convolution entry.
Occasional random mass‑transfer moves and a simulated‑annealing style
acceptance rule let the search escape shallow minima.  FFT‑based
convolutions give O(n log n) speed, allowing many iterations within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetric(v: np.ndarray) -> np.ndarray:
    """Force a vector to be symmetric: (v + v[::-1]) / 2."""
    return (v + v[::-1]) * 0.5


def edge_decay_vector(n: int, p: float, c: float) -> np.ndarray:
    """
    Construct a symmetric vector a_i ∝ (c + d_i)^(-p),
    where d_i = min(i, n‑1‑i) is the distance to the nearest edge.
    The vector is normalised to sum == 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)                 # distance to nearest edge
    a = (c + d) ** (-p) + 1e-12                  # avoid exact zero
    a /= a.sum()
    return symmetric(a)


def conv_max_fft(a: np.ndarray) -> float:
    """
    Compute max_k (a * a)[k] using a real‑valued FFT.
    """
    n = a.shape[0]
    size = 1 << int(np.ceil(np.log2(2 * n - 1)))   # power of two for speed
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())


def score_from_normed(a: np.ndarray) -> float:
    """
    Score for a vector already normalised to sum == 1.
    The evaluate_sequence function would return exactly this value.
    """
    n = a.shape[0]
    return 2.0 * n * conv_max_fft(a)
