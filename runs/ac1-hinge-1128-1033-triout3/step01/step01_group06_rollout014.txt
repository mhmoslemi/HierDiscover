```
def solve_lp_max_sum(a, M):
    n = len(a)
    m = 2 * n - 1
    # Build A_ub matrix
    A = np.zeros((m, n))
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        A[k, i_min:i_max+1] = a[i_min:i_max+1]
    # objective maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # RHS vector
    b_ub = M * np.ones(m)
    # Solve
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # maybe debug
        return None
    b = res.x
    return b
```

"""
Propose a step‐function (non‑negative height sequence) that minimises
2·n·max(conv(a,a)) / (sum(a)**2).  The algorithm combines three ideas:

1.  Fast O(n) simulated‑annealing moves that shift a small amount of mass
    from a source index to a destination index.  The source is chosen as
    the index that contributes most to the current maximal convolution
    entry, and the destination is the index that contributes least.

2.  Periodic global updates obtained by solving a linear program:
        maximise   Σ b_i
        subject to 0 ≤ b_i
                    (a * b)_k ≤ M   for all k,
    where M is the current maximal self‑convolution value.
    The LP yields a new distribution b that can carry more total mass
    without raising the convolution peak.  After scaling b to have the
    same ℓ¹‑norm as a we either replace a with b or take a convex
    combination (a←(1‑t)a + t b).  This step provides a strong global
    descent direction.

3.  Regular full recomputation of the convolution to avoid accumulated
    floating‑point drift.

The routine respects the 1000 s time budget, prints intermediate progress
and returns the best sequence found as a plain list of floats.
"""

import numpy as np
import time
import math
import scipy.optimize


def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Move `delta` mass from a[i] to a[j] and update the self‑convolution
    `conv` (length 2n‑1) in O(n) using vectorised operations.
    Returns the new (a, conv) pair.
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta
    dj = +delta

    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    conv_new = conv.copy()

    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]
    a_q = a[q]

    # contributions involving index i (with all other positions)
    conv_new[i + q] += 2.0 * di * a_q
    # contributions involving index j (with all other positions)
    conv_new[j + q] += 2.0 * dj * a_q

    # self‑pair contributions
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


def targeted_move2(a: np.ndarray, conv: np.ndarray):
    """
    Choose a source index that contributes most to the current peak
    of the convolution and a destination index that contributes least.
    Both choices are based on the product a[p]·a[k‑p] where k is the
    location of the maximal convolution entry.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))

    # indices that actually take part in the k‑th convolution entry
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_vals = np.arange(i_min, i_max + 1)
    j_vals = k - i_vals

    contrib = a[i_vals] * a[j_vals]
    src = int(i_vals[np.argmax(contrib)])

    # destination – minimise the same product (or set it to zero if out of range)
    dst_product = np.full(n, np.inf)
    valid = (k - np.arange(n) >= 0) & (k - np.arange(n) < n)
    idx_valid = np.where(valid)[0]
    dst_product[idx_valid] = a[idx_valid] * a[k - idx_valid]

    # slight bias towards already small entries
    score = dst_product + a * 0.1
    score[src] = np.inf          # never move mass back to the source
    dst = int(np.argmin(score))
    return src, dst


def solve_lp_max_sum(a: np.ndarray, M: float):
    """
    Linear programme:
        maximise   Σ b_i
        subject to   b_i ≥ 0
                     (a * b)_k ≤ M   for all k = 0 … 2n‑2
    Returns the optimal b (numpy array) or None if the solver fails.
    """
    n = a.shape[0]
    m = 2 * n - 1

    # Build the constraint matrix A_ub where (A_ub @ b)[k] = Σ_i a_i b_{k-i}
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        A[k, i_min:i_max + 1] = a[i_min:i_max + 1]

    c = -np.ones(n, dtype=np.float64)                # maximise Σb → minimise -Σb
    bounds = [(0, None)] * n
    b_ub = M * np.ones(m, dtype=np.float64)

    # Use the high‑performance HiGHS solver that ships with SciPy
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=bounds, method='highs',
                                options={'presolve': True})
    if res.success:
        return res.x
    return None

                            best_a = g.copy()
                            print(f"[LP {iteration:6d}] new best = {best_val:.6f}")

                        # Accept if it improves the current state, otherwise try a convex mix
                        if val_g < cur_val:
                            a, conv, cur_val = g, conv_g, val_g
                        else:
                            t = 0.5
                            a_mix = (1.0 - t) * a + t * g
                            a_mix = np.maximum(a_mix, 0.0)
                            a_mix = a_mix / a_mix.sum()
                            conv_mix = np.convolve(a_mix, a_mix)
                            val_mix = 2.0 * n * np.max(conv_mix)
                            if val_mix < cur_val:
                                a, conv, cur_val = a_mix, conv_mix, val_mix
                                if val_mix < best_val:
                                    best_val, best_a = val_mix, a_mix.copy()
                                    print(f"[LPmix {iteration:6d}] new best = {best_val:.6f}")
        # ------------------------------------------------------------------
        # Local simulated‑annealing move
        # ------------------------------------------------------------------
        # Occasionally fall back to a completely random move to maintain exploration
        if np.random.rand() < 0.07:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
        else:
            src, dst = targeted_move2(a, conv)

        max_delta = a[src] * 0.5
        if max_delta < 1e-14:
            iteration += 1
            continue

        delta = np.random.rand() * max_delta
        a_new, conv_new = apply_move(a, conv, src, dst, delta)
        new_val = 2.0 * n * np.max(conv_new)

        improve = new_val < cur_val
        if improve or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a, conv, cur_val = a_new, conv_new, new_val
            accept_cnt += 1
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{iteration:6d}] new best = {best_val:.6f}")

        iteration += 1

        # Re‑compute the full convolution periodically to avoid accumulated rounding errors
        if iteration % 8000 == 0:
            conv = np.convolve(a, a)

    # ------------------------------------------------------------------
    # Final safety check with the provided evaluation routine
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
