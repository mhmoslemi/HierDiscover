"""
Gradient‑exchange optimizer for the self‑convolution bound.

The evaluation function is homogeneous: scaling the whole sequence does not
change the score.  Hence we normalise the ℓ¹‑norm to 1 and minimise the
maximum of the linear convolution a*a.  For a given maximum M the active
constraints are the indices k where (a*a)[k] ≈ M.  The sub‑gradient of the
max‑functional at a is the (non‑negative) sum of the reversed copies of a
that participate in those active constraints:

    g_i = 2 * Σ_{k∈A} a_{k-i}   (indices out of range give 0)

Moving a small amount of mass from indices with large g_i to indices with
small g_i strictly reduces the maximal convolution value (or at worst
leaves it unchanged).  The optimiser repeatedly performs such “exchange”
steps, picking i_max from the top‑few gradient entries and i_min from the
bottom‑few entries, with a decreasing step size.  Randomisation in the
choice of i_max/i_min and a small simulated‑annealing acceptance rule help
escape shallow local minima.

The algorithm starts from the best known construction (global variable
``height_sequence_1`` if present) and falls back to an inverse‑square‑root
core with a tiny constant tail.  It runs for the allotted time (≤ 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT‑based convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: n + m - 1]


def _conv_self_full(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution a*a."""
    return _conv_full_fft(a, a)


def _conv_self_max(a: np.ndarray) -> float:
    """Maximum value of the self‑convolution a*a."""
    return float(_conv_self_full(a).max())


# ----------------------------------------------------------------------
# Gradient of the max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of  max_k (a*a)[k]  at the point a.

    The active set A consists of all indices k where (a*a)[k] ≥ M−eps,
    where M is the current maximum.  The returned vector g has length
    len(a) and satisfies g_i = 2 * Σ_{k∈A} a_{k-i} (out‑of‑range terms are 0).
    """
    n = a.shape[0]
    conv = _conv_self_full(a)
    M = conv.max()
    # pick a small absolute tolerance – ensures at least one active index
    tol = max(eps * M, eps)
    act = np.zeros(2 * n - 1, dtype=np.float64)
    act[np.where(conv >= M - tol)[0]] = 1.0

    # gradient = 2 * (a * act)   (only first n entries are needed)
    grad_full = _conv_full_fft(a, act)   # length = 3n-2
    grad = 2.0 * grad_full[:n]
    return grad, M

    return evaluate_sequence(seq.tolist())

    n = a.shape[0]
    best_seq = a.copy()
    best_val = _score(best_seq)
    print(f"[info] initial score = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2.  Parameters for the exchange loop
    # --------------------------------------------------------------
    # step size (fraction of the donor entry); decays over time
    step_frac = 0.15
    min_step = 5e-5
    # How many of the largest / smallest gradient entries we consider
    top_k = max(5, n // 1000)   # at least 5, more for very long sequences

    # Simple simulated‑annealing style temperature for occasional uphill moves
    temperature = best_val * 0.02
    temp_decay = temperature / (budget_s * 0.8)   # linear decay

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # Compute sub‑gradient and current maximum convolution
        # ----------------------------------------------------------
        grad, M = _max_subgradient(a, eps=1e-12)
        cur_val = 2.0 * n * M / (1.0 ** 2)   # ℓ¹‑norm = 1 ⇒ denominator = 1
        # early exit if already at the known lower bound (π/2 ≈ 1.5708) and
        # cannot improve significantly – keep searching anyway.
        # ----------------------------------------------------------

        # ----------------------------------------------------------
        # Choose donor (i_max) and receiver (i_min) indices
        # ----------------------------------------------------------
        # indices with largest gradient entries (most responsible for the peak)
        top_idxs = np.argpartition(-grad, top_k - 1)[:top_k]
        # indices with smallest gradient entries (least responsible)
        bot_idxs = np.argpartition(grad, top_k - 1)[:top_k]

        i_max = top_idxs[np.argmax(grad[top_idxs])]
        i_min = bot_idxs[np.argmin(grad[bot_idxs])]

        if i_max == i_min:
            # degenerate case – pick a random pair
            i_max, i_min = np.random.choice(n, size=2, replace=False)

        # ----------------------------------------------------------
        # Propose a mass exchange
        # ----------------------------------------------------------
        donor_amount = a[i_max]
        if donor_amount == 0:
            # cannot donate – pick another donor
            continue
        delta = step_frac * donor_amount
        # ensure we do not move a vanishing amount
        if delta < min_step:
            delta = min_step
        delta = min(delta, donor_amount)  # cannot give more than we have

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta
        # no need to renormalise – total ℓ¹‑norm stays 1

        # ----------------------------------------------------------
        # Evaluate the proposal
        # ----------------------------------------------------------
        new_val = _score(new_a)

        # ----------------------------------------------------------
        # Acceptance rule (deterministic improvement + occasional uphill)
        # ----------------------------------------------------------
        accept = False
        if new_val < best_val:
            accept = True
            best_val, best_seq = new_val, new_a.copy()
            print(f"[iter {iteration}] NEW BEST = {best_val:.6f} "
                  f"(i_max={i_max}, i_min={i_min}, Δ={delta:.2e})")
        elif new_val < cur_val:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            if temperature > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / temperature):
                accept = True

        if accept:
            a = new_a
        # ----------------------------------------------------------
        # Cooling
        # ----------------------------------------------------------
        temperature = max(0.0, temperature - temp_decay)

        # ----------------------------------------------------------
        # Adaptive step size – slowly reduce if we see no improvement
        # ----------------------------------------------------------
        if iteration % 200 == 0:
            # If the best value hasn't improved in the last 200 steps, shrink step_frac
            if best_val >= cur_val * 0.9999:
                step_frac = max(min_step, step_frac * 0.9)

    # --------------------------------------------------------------
    # 3.  Final reporting
    # --------------------------------------------------------------
    print(f"[final] best value = {best_val:.6f} after {iteration} iterations")
    return best_seq.tolist()
