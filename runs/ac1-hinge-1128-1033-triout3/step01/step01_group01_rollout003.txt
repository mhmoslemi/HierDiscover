import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the simplex {x | x_i ≥ 0 , Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where the condition holds
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    p_fft = np.fft.rfft(p, n=m_fft)

    cur_score = 2.0 * n * np.max(conv_self)

    swap_attempts = 2000
    for _ in range(swap_attempts):
        if time.time() >= deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_move = 1e-4 * rng.random()
        if p[i] <= eps_move:
            continue
        new_p = p.copy()
        new_p[i] -= eps_move
        new_p[j] += eps_move

        p_fft = np.fft.rfft(new_p, n=m_fft)
        conv_self = np.fft.irfft(p_fft * p_fft, n=m_fft)[:2 * n - 1]
        new_score = 2.0 * n * np.max(conv_self)

        if new_score < cur_score:
            p = new_p
            cur_score = new_score
            best_p = p.copy()
            best_score = cur_score
            print(f"Swap improvement → {best_score:.6f}")

    print(f"Final best score: {best_score:.6f}")
    # Clip to the range enforced by evaluate_sequence and return as plain list
    return np.clip(best_p, 0.0, 1000.0).tolist()
