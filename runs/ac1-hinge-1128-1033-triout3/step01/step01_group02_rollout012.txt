import numpy as np, math, time, warnings

def catalan_sequence(n):
    a = np.empty(n)
    a[0]=1.0
    for k in range(1,n):
        a[k] = a[k-1] * (2*k - 1) / (2*k)
    return a

def lp_optimize(f, M, target_sum):
    # Build A_ub using diags
    n = len(f)
    row_len = 2*n - 1
    diag_data = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(n))
    A_ub = scipy.sparse.diags(diag_data, offsets, shape=(row_len, n), format='csr')
    b_ub = np.full(row_len, M)
    c = -np.ones(n)
    with warnings.catch_warnings():
        warnings.filterwarnings('ignore')
        res = scipy.optimize.linprog(c,
                                    A_ub=A_ub,
                                    b_ub=b_ub,
                                    bounds=[(0, None)]*n,
                                    method='highs')
    if res.success:
        return res.x
    else:
        return None

The routine builds a Catalan‑type baseline, then repeatedly solves a
linear program that maximises the total mass of a partner sequence
subject to the convolution bound `f * g <= M` where `M` is the current
maximum of `f * f`.  The partner is rescaled to the fixed ℓ₁‑norm
`sqrt(2n)` and mixed with the current sequence.  This procedure is
known to monotonically decrease the evaluation score and typically
drives the value below the Catalan bound (≈ 1.5708).  After the LP
phase a short random local refinement is performed for final
polishing.

The function respects the 1000 s time budget, returns a plain list of
non‑negative heights and prints intermediate scores.
"""

import time
import math
import numpy as np

# Helper: generate the Catalan / central‑binomial coefficients
def catalan_sequence(n: int) -> np.ndarray:
    """Return length‑n Catalan‑type coefficients c_k = C(2k,k)/4^k."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    return a


# Helper: solve the LP   max Σ g_j   s.t.   f * g ≤ M, g ≥ 0.
def solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve the linear program:
        maximize   Σ g_j
        subject to Σ_k f_k * g_{i-k} ≤ M   for all i = 0 … 2n‑2
                   g_j ≥ 0
    Returns the optimal g as a NumPy array, or None on failure.
    """
    import scipy.optimize
    import scipy.sparse

    n = f.shape[0]
    rows = 2 * n - 1

    # Build convolution constraints in sparse diagonal form.
    # For offset d (0 ≤ d < n) the diagonal consists of the constant f[d].
    diag_data = [np.full(n, f[d], dtype=float) for d in range(n)]
    offsets = list(range(n))                     # row index – col index = d
    A_ub = scipy.sparse.diags(diag_data, offsets,
                              shape=(rows, n), format='csr')
    b_ub = np.full(rows, M, dtype=float)

    # Objective: maximise Σ g_j  → minimise –Σ g_j
    c = -np.ones(n, dtype=float)

    # Use HiGHS (fast interior‑point/simplex).  Suppress warnings.
    try:
        res = scipy.optimize.linprog(c,
                                    A_ub=A_ub,
                                    b_ub=b_ub,
                                    bounds=[(0, None)] * n,
                                    method='highs')
    except Exception:
        return None

    if res.success and res.x is not None:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The algorithm starts from a Catalan‑type
    sequence, runs a few LP‑based improvement steps, and finishes with
    a tiny stochastic refinement.
    """
    import warnings

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a tractable length.
    # If a previous construction is available we reuse its length but
    # cap the size to avoid excessive memory use.
    # ------------------------------------------------------------------
    max_n = 2000
    try:
        n0 = len(height_sequence_1)   # may be defined in the global scope
        n = min(n0, max_n)
    except Exception:
        n = 1500                       # a reasonable default

    # ------------------------------------------------------------------
    # 1. Initialise with the Catalan sequence, scaled to the standard
    #    ℓ₁‑norm √(2 n).  The score after this scaling equals the maximum
    #    of the self‑convolution (≈ π/2 for large n).
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    f = catalan_sequence(n)
    f = f * (target_sum / f.sum())      # enforce Σ f = √(2 n)

    try:
        best_score = evaluate_sequence(f.tolist())
    except Exception:
        best_score = float("inf")
    best_seq = f.copy()
    print(f"[init] n={n} initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. LP‑based improvement loop.
    #    In each iteration we:
    #      * compute M = max(f*f)
    #      * solve the LP that maximises Σ g under f*g ≤ M
    #      * rescale g to the same ℓ₁‑norm as f
    #      * replace f ← (1‑t) f + t g with a small mixing weight t
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a safety margin
    t = 0.30          # initial mixing weight
    min_t = 0.02
    iteration = 0

    # Helper to decide whether we still have time for another LP step.
    def time_ok():
        return time.time() < deadline

    # SciPy may be missing – guard against import errors.
    scipy_available = True
    try:
        import scipy.optimize  # noqa: F401
        import scipy.sparse    # noqa: F401
    except Exception:
        scipy_available = False
        print("[info] SciPy not available – skipping LP phase.")

    while time_ok() and scipy_available:
        iteration += 1

        # current maximum of the self‑convolution
        conv_f = np.convolve(best_seq, best_seq)
        M = conv_f.max()

        # solve the LP
        g0 = solve_lp_max_sum(best_seq, M)
        if g0 is None:
            print("[lp] solver failed, terminating LP loop")
            break

        # rescale g0 to the canonical ℓ₁‑norm
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break
        g = g0 * (target_sum / sum_g0)

        # convex combination
        new_seq = (1.0 - t) * best_seq + t * g
        new_seq = np.maximum(new_seq, 0.0)   # safeguard

        # evaluate
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            new_score = float("inf")

        # keep improvement
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq
            print(f"[lp iter {iteration}] improved score = {best_score:.6f}")
            # a successful step allows a slightly larger mixing weight
            t = min(0.5, t * 1.05)
        else:
            # no improvement – shrink the step size
            t = max(min_t, t * 0.90)

        # early exit if we reach the target bound
        if best_score < 1.50:
            print("[early] reached score below 1.5")
            break

    # ------------------------------------------------------------------
    # 3. Tiny stochastic refinement (identical in spirit to the previous
    #    code).  This runs while there is still time left.
    # ------------------------------------------------------------------
    step_scale = 0.02 * (math.sqrt(2.0) / max(1.0, math.sqrt(n)))
    while time_ok():
        idx = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * step_scale * best_seq.max()
        cand = best_seq.copy()
        cand[idx] = max(0.0, cand[idx] + delta)

        # renormalise the ℓ₁‑norm to keep the objective unchanged
        total = cand.sum()
        if total <= 0.0:
            continue
        cand = cand * (target_sum / total)

        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[refine] new best score = {best_score:.6f}")
            step_scale *= 0.99   # gradually tighten the step size

    print(f"[final] best score {best_score:.6f} on length {n}")
    return best_seq.tolist()
