"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Choose a moderate length (≈2000‑2500) – larger lengths give a bigger
   factor ``2·n`` in the score but also a larger search space.
   All vectors are normalised to the *theoretical* optimum sum
   ``S = sqrt(2·n)``.  For such a normalisation the evaluation function
   reduces to the peak of the self‑convolution, i.e.  ``score = max(f*f)``.

2. Initialise the search with a simple symmetric “U‑shape’’
   ``a[i] ∝ (dist_to_edge+1)^{-p}``.  Several exponents ``p`` are tried
   and the best seed is kept.

3. Refine the seed by the iterative scheme described in the statement:
   given the current vector ``f`` with peak ``M = max(f*f)``,
   solve the linear programme

        maximise   sum(g_j)
        subject to  (f * g)[k] ≤ M   for all k = 0 … 2n‑2
                    g_j ≥ 0

   (the convolution is linear in ``g`` for fixed ``f``).  The LP is
   solved with ``scipy.optimize.linprog`` (HiGHS backend).  The optimum
   ``g`` is scaled back to the target sum ``S`` and mixed with the old
   vector:

        f ← (1‑t)·f + t·g_scaled      (t starts at 0.5 and decays)

   If ``g_scaled`` alone already yields a smaller peak it replaces ``f``.
   This step is repeated until the time budget is exhausted or no
   improvement is observed.

4. A short simulated‑annealing / random‑mass‑transfer refinement is
   applied to the best vector found so far.  It is cheap and helps to
   escape tiny local minima.

The whole routine respects the 1000 s time budget, prints occasional
progress, and finally returns the best coefficient list (as plain
Python floats).  All values are non‑negative and well below the
``evaluate_sequence`` clipping threshold of 1000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric “U‑shape’’:
        a[i] = (min(i, n‑1‑i) + 1)^{‑p}
    Normalised to sum = 1.
    """
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14           # avoid exact zero
    a /= a.sum()
    return a


def convolution_peak(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of ``a``."""
    return float(np.convolve(a, a).max())


def maxconv_score(a: np.ndarray) -> float:
    """
    Score used throughout the optimisation: the peak of the
    self‑convolution (the exact value returned by ``evaluate_sequence``
    when ``a`` is normalised to sum = sqrt(2·len(a))).
    """
    return convolution_peak(a)


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix ``A`` such that ``A @ g = np.convolve(f, g)``.
    Shape of ``A`` is (2·n‑1, n).
    """
    from scipy.linalg import toeplitz

    n = f.shape[0]
    # First column = f padded with zeros, first row = [f0, 0, …, 0]
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    A_full = toeplitz(c, r)
    return A_full[: 2 * n - 1, :n]


def lp_step(f: np.ndarray, S: float, t: float) -> tuple[np.ndarray, float, bool]:
    """
    Perform a single LP‑based refinement step.

    Parameters
    ----------
    f : np.ndarray
        Current vector (already normalised to sum = S).
    S : float
        Desired total sum (sqrt(2·n)).
    t : float
        Mixing weight (0 < t ≤ 1).

    Returns
    -------
    f_new : np.ndarray
        Updated vector (still normalised to sum = S).
    new_score : float
        Peak of the self‑convolution of ``f_new``.
    improved : bool
        ``True`` if the peak decreased.
    """
    from scipy.optimize import linprog

    n = f.shape[0]
    # 1) current peak
    M = maxconv_score(f)

    # 2) build linear constraints  (f * g)[k] ≤ M
    A = build_conv_matrix(f)                       # shape (2n‑1, n)
    b = np.full(A.shape[0], M, dtype=float)

    # 3) solve maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n, dtype=float)                    # minimise -∑g
    bounds = [(0.0, None)] * n

    # Use HiGHS – fast for dense problems
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # LP failed – return previous vector unchanged
        return f, M, False

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return f, M, False

    # 4) rescale g to the target sum S
    g = g * (S / sum_g)

    # 5) decide whether to accept g directly or mix
    M_g = maxconv_score(g)
    if M_g < M:
        f_new = g
        new_score = M_g
        improved = True
    else:
        # Convex combination
        f_new = (1.0 - t) * f + t * g
        f_new /= f_new.sum()        # re‑normalise (numerical drift)
        f_new *= S                   # enforce exact sum S
        new_score = maxconv_score(f_new)
        improved = new_score < M - 1e-12

    return f_new, new_score, improved


def stochastic_refinement(
    seq: np.ndarray, best_score: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Cheap simulated‑annealing style local search that moves a tiny
    amount of mass between two randomly chosen positions.
    """
    n = seq.shape[0]
    cur = seq.copy()
    cur_score = best_score
    temperature = 1.0
    cooling = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1
        # move a random fraction of mass from i -> j
        i, j = np.random.randint(0, n, size=2, replace=False)
        if cur[i] == 0.0:
            continue
        delta = cur[i] * np.random.rand() * 0.5
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # re‑normalise to the exact sum required by the evaluator
        S = np.sqrt(2.0 * n)
        cand /= cand.sum()
        cand *= S

        cand_score = maxconv_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                # print useful progress information
                print(f"Iter {it:6d}: new best score = {best_score:.6f}")

        temperature = max(temperature * cooling, 1e-6)

    return cur, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (heuristically) minimises
    ``evaluate_sequence``.  The routine uses a parametric U‑shape seed,
    an LP‑based alternating optimisation (the method from the
    statement), and a short stochastic local search.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a small safety margin

    # --------------------------------------------------------------
    # 1. Choose a manageable length (≈2500) – large enough to beat
    #    the reference 5808‑item solution, but small enough for LP.
    # --------------------------------------------------------------
    n = 2500
    target_sum = np.sqrt(2.0 * n)  # normalisation used by the evaluator

    # --------------------------------------------------------------
    # 2. Scan a few exponents for the deterministic U‑shape seed.
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    candidate_ps = [0.3, 0.45, 0.6, 0.75, 0.9, 1.05]

    print("=== Scanning U‑shape exponents ===")
    for p in candidate_ps:
        if time.time() > deadline:
            break
        a = u_shaped_distribution(n, p) * target_sum
        sc = maxconv_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  seed p={p:.2f} → score={sc:.6f}")

    # If by any chance a global variable ``height_sequence_1`` exists and
    # is shorter than our limit, try it as an extra seed.
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size and external.size <= 3000:
            external = np.maximum(external, 0.0)
            external = external / external.sum() * target_sum
            sc_ext = maxconv_score(external)
            if sc_ext < best_score:
                best_score = sc_ext
                best_seq = external.copy()
                print(f"  external seed → score={sc_ext:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3. LP‑based alternating optimisation (main engine)
    # --------------------------------------------------------------
    # Import linprog once – if it fails we fall back to the simple
    # stochastic method used in the previous submission.
    try:
        from scipy.optimize import linprog  # noqa: F401
        have_scipy = True
    except Exception:
        have_scipy = False
        print("SciPy not available – skipping LP refinement.")

    if have_scipy and best_seq is not None:
        f = best_seq.copy()
        t = 0.5
        iteration = 0
        print("\n=== LP alternating optimisation ===")
        while time.time() < deadline:
            iteration += 1
            f, new_score, improved = lp_step(f, target_sum, t)
            if new_score < best_score:
                best_score = new_score
                best_seq = f.copy()
                print(f"LP iter {iteration:4d}: new best score = {best_score:.6f}")

            # decay mixing weight; keep it above a tiny floor
            t = max(0.05, t * 0.95)

            # stop if no improvement in a few consecutive steps
            if not improved and t <= 0.05:
                break
            if iteration >= 60:       # generous upper bound for safety
                break

    # --------------------------------------------------------------
    # 4. Final cheap stochastic refinement (optional)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n=== Final stochastic refinement ===")
        final_seq, final_score = stochastic_refinement(
            best_seq, best_score, deadline
        )
        if final_score < best_score:
            best_seq = final_seq
            best_score = final_score

    # --------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------
    if best_seq is None:
        # very unlikely – fall back to a uniform vector
        best_seq = np.full(n, target_sum / n, dtype=float)

    print("\n=== Search finished ===")
    print(f"Best score (max convolution) = {best_score:.6f}")
    print(f"Length of vector = {len(best_seq)}")
    return best_seq.tolist()
