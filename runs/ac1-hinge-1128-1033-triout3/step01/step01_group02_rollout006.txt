"""
Propose a step‑function sequence using an iterative linear‑programming refinement.

1.  Start from the classic Catalan (central–binomial) coefficients.
    This shape gives a very low evaluation score (≈π/2 ≈ 1.5708).

2.  For a fixed length n we treat the problem:
        max (f * f) ≤ M      with  M = max(f*f)
        maximise  ∑ g_j   subject to  f * g ≤ M ,  g ≥ 0
    The convolution f*g is linear in g, therefore the sub‑problem is a
    linear program (LP).  Its solution g has a much larger ℓ₁‑norm than f
    while still respecting the same convolution bound M.

3.  Scale g back to the original ℓ₁‑norm of f and mix the two
    sequences:   f ← (1‑t)·f + t·g   (t≈0.3).  The mixing keeps the total
    sum unchanged and never increases the worst‑case convolution value.
    Re‑evaluate the score; keep the new f if it improves.

4.  Repeat the LP→mix step until the time budget is exhausted (or
    improvements cease).  Finally perform a tiny random local search
    around the best sequence found.

The algorithm uses `scipy.optimize.linprog` with a sparse constraint
matrix (≈ n² non‑zeros) and therefore works comfortably for n≈2000
within the 1000 s budget.  If SciPy is unavailable we fall back to the
previous random‑perturbation method.
"""

import time
import math
import numpy as np

# Try to import the heavy libraries once, fall back gracefully.
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_OK = True
except Exception:  # pragma: no cover – SciPy may be missing in some environments
    _SCIPY_OK = False


def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients (unnormalised)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def _convolution_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the matrix A (shape (2n‑1, n)) such that A @ g = np.convolve(f, g).
    Each column contains the whole vector f shifted downwards.
    """
    n = f.shape[0]
    # rows[i, j] = j + i, cols[i, j] = j, data = f[i] (repeated for each column)
    i = np.arange(n, dtype=np.int64)          # 0 … n‑1  (offset inside a column)
    j = np.arange(n, dtype=np.int64)          # column index
    rows = (j[:, None] + i[None, :]).ravel()  # shape (n*n,)
    cols = np.repeat(j, n)                    # each column repeated n times
    data = np.tile(f, n)                      # f repeated for every column
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_maximise_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise 1ᵀg   s.t.   f * g ≤ M  (componentwise), g ≥ 0.
    Returns the optimal g (may contain very large entries).
    """
    n = f.shape[0]
    A = _convolution_matrix(f)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)    # same bound for every row
    c = -np.ones(n, dtype=np.float64)              # linprog minimises, so –1 gives maximise

    # HiGHS (default) can handle sparse A efficiently.
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  The routine starts from a Catalan sequence,
    repeatedly expands its ℓ₁‑norm via a linear program, mixes the
    improvement back into the current sequence and finishes with a tiny
    stochastic fine‑tune.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a few seconds spare for cleanup

    # ------------------------------------------------------------------
    # 1.  Choose the length n.
    # ------------------------------------------------------------------
    MAX_N = 2000      # safe size for the LP; larger values explode memory/time
    try:
        # If a previous solution exists we reuse its length but cap it.
        n = min(len(height_sequence_1), MAX_N)
    except NameError:
        n = MAX_N

    # ------------------------------------------------------------------
    # 2.  Initialise with Catalan coefficients.
    # ------------------------------------------------------------------
    f = _catalan_sequence(n)
    best_seq = f.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float("inf")
    print(f"[init] n={n}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Iterative LP‑based refinement (only if SciPy is present).
    # ------------------------------------------------------------------
    if _SCIPY_OK:
        t_mix = 0.35           # initial mixing weight
        last_improvement = time.time()
        iteration = 0

        while time.time() < deadline:
            iteration += 1
            # 3a. current bound M = max(f * f)
            conv_ff = np.convolve(f, f)
            M = float(conv_ff.max())
            if M <= 0:
                break  # degenerate case – should not happen

            # 3b. solve the LP to boost the ℓ₁ norm
            try:
                g0 = _lp_maximise_sum(f, M)
            except Exception as exc:
                print(f"[lp‑fail] iteration {iteration}: {exc}")
                break

            sum_f = float(f.sum())
            sum_g0 = float(g0.sum())
            if sum_g0 <= sum_f * (1.0 + 1e-12):
                # No further gain possible – stop early.
                print(f"[stop] iteration {iteration}: LP cannot increase ℓ₁ norm.")
                break

            # 3c. rescale g0 to have the same ℓ₁ norm as f
            scale = sum_f / sum_g0
            g = g0 * scale

            # 3d. try to mix the new direction – reduce t if needed
            improved = False
            t_try = t_mix
            while t_try > 1e-4:
                f_new = (1.0 - t_try) * f + t_try * g
                # evaluate – the objective is scale‑invariant, so we can use
                # evaluate_sequence directly.
                try:
                    new_score = evaluate_sequence(f_new.tolist())
                except Exception:
                    new_score = float("inf")

                if new_score < best_score - 1e-12:
                    # Accept the new sequence
                    best_seq = f_new.copy()
                    best_score = new_score
                    f = f_new
                    improved = True
                    print(f"[iter {iteration}] t={t_try:.4f} → score={best_score:.6f}")
                    # slowly increase mixing weight for the next round (up to 0.5)
                    t_mix = min(0.5, t_mix * 1.05)
                    break
                else:
                    # try a smaller step
                    t_try *= 0.5

            if not improved:
                # No improvement with any step size → assume convergence.
                print(f"[converged] iteration {iteration}: no mixing improvement.")
                break

            # occasional aggressive shrink of the mixing weight to avoid stalls
            if iteration % 7 == 0:
                t_mix = max(0.05, t_mix * 0.9)

            # keep an eye on time
            if time.time() - last_improvement > 120:  # 2 min without progress → exit
                print("[timeout] no improvement for 2 min, leaving loop.")
                break
            last_improvement = time.time()
    else:
        print("[fallback] SciPy unavailable – using stochastic refinement only.")

    # ------------------------------------------------------------------
    # 4.  Final tiny stochastic refinement (works with or without LP).
    # ------------------------------------------------------------------
    # The objective is homogeneous, so we keep the total sum fixed while
    # exploring the neighbourhood.
    step_scale = 0.02 * (math.sqrt(2.0) / max(1.0, math.sqrt(n)))
    while time.time() < deadline:
        idx = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * step_scale
        cand = best_seq.copy()
        cand[idx] = max(0.0, cand[idx] + delta)

        # Renormalise to preserve the ℓ₁ norm (does not change the score).
        s = np.sum(cand)
        if s > 0:
            cand *= (np.sum(best_seq) / s)
        else:
            continue

        try:
            cand_score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[local] improved to {best_score:.6f}")

        # slowly shrink the perturbation magnitude
        if np.random.rand() < 0.01:
            step_scale *= 0.95

    # ------------------------------------------------------------------
    # 5.  Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
