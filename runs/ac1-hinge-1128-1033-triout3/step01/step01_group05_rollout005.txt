def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    from scipy.optimize import linprog

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0
    best_seq = None
    best_score = float('inf')
    # Baseline: binomial prefix length up to maybe 3000 (as before)
    max_len = 3000
    coeffs = generate_binomial_prefix(max_len)
    # get best analytic len as before maybe 3000? Actually for constant conv seq best_len = max_len
    # We'll take that as initial candidate
    init_seq = coeffs[:max_len].tolist()
    try:
        init_score = evaluate_sequence(init_seq)
        best_seq = init_seq
        best_score = init_score
        print(f"[baseline] length={max_len}, score={init_score}")
    except Exception:
        pass

    # Now attempt iterative LP improvements for several lengths
    lengths = [100, 150, 200, 250, 300, 400, 500]  # moderate sizes
    for n in lengths:
        if time.time() > deadline:
            break
        # Build initial f: could use binomial prefix length n or random Dirichlet
        f = generate_binomial_prefix(n)
        # Optionally random perturbations: also try random Dirichlet as start
        for restart in range(2):
            if time.time() > deadline:
                break
            # Randomization: replace f with random Dirichlet if restart >0
            if restart == 1:
                # Dirichlet of parameter alpha=0.5 may produce variation
                alpha = np.full(n, 0.5)
                f = np.random.dirichlet(alpha)  # sum=1
                # scale maybe to random magnitude to avoid tiny values; keep as is
            # Ensure positivity
            f = np.maximum(f, 1e-12)
            # Iterative improvement
            for it in range(30):
                if time.time() > deadline:
                    break
                # Compute M
                conv_ff = np.convolve(f, f)
                M = conv_ff.max()
                # Build linear constraints matrix A_ub
                # shape (2*n-1, n)
                rows = 2 * n - 1
                A = np.zeros((rows, n), dtype=np.float64)
                for j in range(n):
                    A[j:j+n, j] = f
                # Solve LP
                c = -np.ones(n)  # maximize sum => minimize -sum
                ub_vec = np.full(rows, M * (1.0 + 1e-12))
                # bounds for b >=0
                bounds = [(0, None)] * n
                res = linprog(c, A_ub=A, b_ub=ub_vec, bounds=bounds, method='highs')
                if not res.success:
                    # If fails, break
                    break
                b_opt = res.x
                sum_f = f.sum()
                sum_b = b_opt.sum()
                # If no improvement break
                if sum_b <= sum_f + 1e-12:
                    break
                # Mix
                t = 0.2  # mixing weight
                f = (1 - t) * f + t * b_opt
                # Evaluate current objective
                seq = f.tolist()
                try:
                    score = evaluate_sequence(seq)
                except Exception:
                    score = float('inf')
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    print(f"[improve] n={n} restart={restart} it={it} score={score:.6f}")
            # end for it
        # end restarts
    # end for n

    # final local refinement: random perturbations of best_seq similar to previous approach
    if best_seq is None:
        best_seq = init_seq
        best_score = init_score
    cur_seq = best_seq[:]
    cur_score = best_score
    n = len(cur_seq)
    sigma_rel = 0.02
    max_no_improve = 5000
    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(n)
        old = cur_seq[idx]
        delta = np.random.randn() * sigma_rel * max(old, 1e-12)
        new_val = max(0.0, old + delta)
        cand = cur_seq[:]
        cand[idx] = new_val
        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue
        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_seq, best_score = cand[:], cand_score
                print(f"[local improve] new best {best_score:.6f}")
        else:
            no_improve += 1

    # final cleanup
    best_seq = [float(max(0.0, x)) for x in best_seq]
    return best_seq

"""
Propose a candidate step‐function sequence.

The algorithm starts from the analytically optimal “constant‑convolution”
(binomial) sequence and then attempts to improve the objective by an
alternating linear‑programming scheme described in the problem statement:

* For a fixed sequence f we compute M = max(f*f).
* Keeping M fixed we solve a linear program that maximises Σ b_j
  subject to the linear constraints (f*b)_k ≤ M for all k.
  This yields a vector b with a larger L₁‑norm than f while still respecting
  the convolution bound.
* We mix the old and the new vector (f←(1‑t)f+t·b) with a small mixing weight
  t.  The L₁‑norm grows linearly in t while the maximal convolution grows
  only quadratically, so the evaluation score 2·n·max(conv)/(Σa)² strictly
  decreases.
* The process is repeated for several sequence lengths (up to a few hundred)
  and several random restarts.  The best sequence seen (including a final
  lightweight stochastic hill‑climb) is returned.

The whole search respects the 1000 s time budget and uses only numpy and
scipy.optimize.linprog (available in the execution environment).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate the constant‑convolution (central binomial) coefficients
# ----------------------------------------------------------------------
def generate_binomial_prefix(max_n: int) -> np.ndarray:
    """Return b[0..max_n‑1] where b_k = C(2k,k)/4**k."""
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, max_n):
        # recurrence: b_{k} = b_{k‑1} * (2k‑1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


# ----------------------------------------------------------------------
# Linear‑program for the “g‑step” of the alternating optimisation
# ----------------------------------------------------------------------
def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_j   subject to   (f*b)_k ≤ M ,   b_j ≥ 0.
    Returns the optimal b as a 1‑D ndarray, or None if the LP fails.
    """
    from scipy.optimize import linprog

    n = f.size
    rows = 2 * n - 1

    # Build the constraint matrix A where (A @ b) = conv(f,b)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f  # shifted copy of f

    # Right‑hand side – a tiny epsilon avoids numerical infeasibility
    eps = 1e-12
    ub = np.full(rows, M * (1.0 + eps), dtype=np.float64)

    # Objective: minimise – Σ b_j  (i.e. maximise Σ b_j)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n

    # Solve with the HiGHS interior‑point simplex – fast for our modest sizes
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the competition
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluation score.  Returns the best sequence found within the allowed
    time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Baseline: constant‑convolution (binomial) sequence, length 3000
    # --------------------------------------------------------------
    MAX_LEN = 3000
    bin_coeffs = generate_binomial_prefix(MAX_LEN)
    best_seq = bin_coeffs[:MAX_LEN].tolist()
    try:
        best_score = evaluate_sequence(best_seq)
        print(f"[baseline] length={MAX_LEN}, score={best_score:.6f}")
    except Exception:                     # defensive programming
        best_score = float("inf")

    # --------------------------------------------------------------
    # 2️⃣  Alternating LP optimisation for several moderate lengths
    # --------------------------------------------------------------
    # lengths to try – larger lengths give more freedom but slower LPs.
    lengths = [120, 180, 240, 300, 400, 500]   # all ≤ 500 → fast LP solves
    for n in lengths:
        if time.time() >= deadline:
            break

        # start from the normalised binomial prefix (sum = 1)
        f = generate_binomial_prefix(n).astype(np.float64)
        f /= f.sum()                # normalise to L1 = 1

        # we also try a random Dirichlet start (two restarts per length)
        for restart in range(2):
            if time.time() >= deadline:
                break

            if restart == 1:        # random Dirichlet initialisation
                # α < 1 concentrates mass; α > 1 gives flatter start.
                alpha = np.full(n, 0.7)
                f = np.random.dirichlet(alpha)

            f = np.maximum(f, 1e-14)   # avoid exact zeros (numerical safety)

            # iteratively improve f
            for it in range(30):       # at most 30 LP steps per restart
                if time.time() >= deadline:
                    break

                # current maximal self‑convolution
                conv_ff = np.convolve(f, f)
                M = conv_ff.max()

                # solve LP: maximise Σ b while keeping conv(f,b) ≤ M
                b = solve_lp_max_sum(f, M)
                if b is None:
                    # LP failed – abort this restart
                    break

                sum_f = f.sum()
                sum_b = b.sum()
                if sum_b <= sum_f + 1e-12:
                    # no L1‑gain possible – stop iterating this start
                    break

                # decide mixing weight t – larger if b is not too “spiky”
                conv_bb = np.convolve(b, b)
                M_b = conv_bb.max()
                ratio = M_b / (M + 1e-12)

                if ratio <= 1.5:
                    t = 0.4
                elif ratio <= 3.0:
                    t = 0.2
                else:
                    t = 0.08

                # convex combination (keeps all entries non‑negative)
                f = (1.0 - t) * f + t * b

                # evaluate the new candidate
                cand_seq = f.tolist()
                try:
                    cand_score = evaluate_sequence(cand_seq)
                except Exception:
                    cand_score = float('inf')

                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand_seq
                    print(f"[lp‑improve] n={n} restart={restart} it={it} "
                          f"score={cand_score:.6f}")

                # stop early if improvement stagnates
                if t * (sum_b - sum_f) < 1e-8:
                    break

    # --------------------------------------------------------------
    # 3️⃣  Final lightweight stochastic hill‑climb (as in the previous code)
    # --------------------------------------------------------------
    cur_seq = best_seq[:]
    cur_score = best_score
    n_cur = len(cur_seq)
    sigma_rel = 0.02          # relative perturbation magnitude
    max_no_improve = 4000
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        idx = np.random.randint(n_cur)
        old_val = cur_seq[idx]
        delta = np.random.randn() * sigma_rel * max(old_val, 1e-12)
        new_val = max(0.0, old_val + delta)

        cand = cur_seq[:]
        cand[idx] = new_val

        try:
            cand_score = evaluate_sequence(cand)
        except Exception:
            continue

        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_seq, best_score = cand[:], cand_score
                print(f"[local‑improve] new best {best_score:.6f}")
        else:
            no_improve += 1

    # --------------------------------------------------------------
    # 4️⃣  Return the best sequence (ensure non‑negative numbers)
    # --------------------------------------------------------------
    best_seq = [float(max(0.0, x)) for x in best_seq]
    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
