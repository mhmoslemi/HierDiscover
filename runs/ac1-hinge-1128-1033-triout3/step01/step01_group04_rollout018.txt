def propose_candidate(seed=0, budget_s=1000, length=4000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    
    seq = _arcsine_coeffs(length)
    # normalize sum to 1
    seq = np.array(seq, dtype=np.float64)
    seq /= seq.sum()
    
    best_seq = seq.copy()
    best_score = _compute_score(seq.tolist())
    cur_seq = seq.copy()
    cur_score = best_score
    
    T = 1e-3
    T_min = 1e-7
    decay = 0.99995
    
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # choose mutation type
        if np.random.rand() < 0.7:
            # transfer
            i, j = np.random.randint(length, size=2)
            if i == j or cur_seq[i] < 1e-12:
                continue
            # delta fraction of source
            frac = np.random.rand() * 0.5  # up to 50% of source
            delta = cur_seq[i] * frac
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        else:
            # perturb multiplicative factor at a random index
            i = np.random.randint(length)
            factor = np.exp(np.random.normal(scale=0.1))
            new_seq = cur_seq.copy()
            new_seq[i] *= factor
        
        # ensure non-negative
        new_seq = np.maximum(new_seq, 0.0)
        # renormalize sum
        s = new_seq.sum()
        if s < 1e-12:
            continue
        new_seq /= s
        
        new_score = _compute_score(new_seq.tolist())
        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[iter {iteration}] new best {best_score:.6f}")
        # cool
        T = max(T * decay, T_min)
    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()

"""
Propose a step‑function with a lower evaluation than the classic arcsine
construction (≈ π/2 ≈ 1.5708).  

The algorithm follows the “linear‑programming update” described in the
statement:

1. Start from the arcsine coefficients a_i = C(2i,i)/4^i and scale them so
   that Σ a_i = √(2 n).  (The scale does not change the evaluation value.)

2. Compute the current maximal convolution value M = max(a * a).

3. Solve a linear program that **maximises the L₁‑norm** of a new vector
   g under the *same* convolution bound M:

        maximise   Σ_j g_j
        subject to g_j ≥ 0
                   (a ∗ g)_k ≤ M   for every k = 0…2n‑2

   For fixed a the constraints are linear because (a ∗ g)_k = Σ_i a_i g_{k‑i}.
   The LP gives a feasible g₀ whose L₁‑norm is larger than the current one.

4. Rescale g₀ back to the original normalisation (same Σ a_i) and mix
   the two vectors:

        a ← (1‑t)·a + t·g      with a small step size t (≈ 0.05).

   The mixture has the same total mass as before but a strictly smaller
   maximal convolution value.  Repeating steps 2‑4 drives the evaluation
   down.

5. After a few LP iterations a short stochastic refinement (mass transfer)
   is applied to polish the result.

All heavy work is done with NumPy / SciPy (FFT for fast evaluation,
SciPy’s high‑performance LP solver “highs”).  The whole search runs for
the allotted time (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _compute_score(seq):
    """Fast evaluation of the objective (identical to evaluate_sequence)."""
    a = np.asarray(seq, dtype=np.float64)
    n = len(a)
    # FFT length – next power of two ≥ 2·n‑1
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _arcsine_coeffs(n):
    """First n coefficients of (1‑z)^{‑½} = Σ binom(2i,i)/4^i·z^i."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def _build_conv_matrix(f):
    """
    Build the (2n‑1)×n constraint matrix A such that
        (A @ g)[k] = (f ∗ g)[k]
    for all k.
    Uses a sparse diagonal construction (each diagonal of offset i
    contains the value f[i]).
    """
    n = len(f)
    diags = []
    offsets = []
    # diagonal i corresponds to the contribution of f[i] to (f∗g)[k]
    for i in range(n):
        length = min(n, 2 * n - 1 - i)          # number of entries on this diagonal
        diags.append(np.full(length, f[i]))
        offsets.append(i)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _lp_improve(a, M, target_sum, mix_factor):
    """
    One LP‑based improvement step.
    *a*   – current sequence (numpy array, length n)
    M     – current maximal convolution value
    target_sum – normalisation Σ a_i = √(2 n)
    mix_factor – step size t for the convex combination
    Returns the new sequence and the new objective value.
    """
    n = len(a)

    # constraint matrix for (a ∗ g) ≤ M
    A = _build_conv_matrix(a)

    # maximise Σ g_j  <=>  minimise –Σ g_j
    c = -np.ones(n, dtype=np.float64)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # solve the LP (highs is fast and handles sparse A)
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if res.status != 0 or res.x is None:
        # LP failed – return the original vector unchanged
        return a, _compute_score(a.tolist())

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= target_sum * (1.0 + 1e-12):
        # no improvement possible
        return a, _compute_score(a.tolist())

    # rescale to keep the L1‑norm fixed (normalisation)
    g = g0 * (target_sum / sum_g0)

    # convex combination (mixing)
    new_a = (1.0 - mix_factor) * a + mix_factor * g

    # small numerical clean‑up
    new_a[new_a < 0.0] = 0.0

    new_score = _compute_score(new_a.tolist())
    return new_a, new_score


def _stochastic_refinement(seq, deadline):
    """
    Lightweight stochastic hill‑climbing that runs until the time
    budget expires.  It moves a tiny amount of mass from one position
    to another while keeping the total sum unchanged.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_score = _compute_score(best_seq.tolist())
    cur_seq = seq.copy()
    cur_score = best_score

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j or cur_seq[i] <= 0.0:
            continue

        # move a small fraction (up to 5 %) of the source entry
        delta = cur_seq[i] * np.random.rand() * 0.05
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_score = _compute_score(new_seq.tolist())
        if new_score < cur_score:
            cur_seq, cur_score = new_seq, new_score
            if new_score < best_score:
                best_seq, best_score = new_seq, new_score
                print(f"[refine] new best {best_score:.6f}")
        else:
            # occasional uphill move (probability decays with the worsening)
            if np.random.rand() < math.exp(-(new_score - cur_score) * 20):
                cur_seq, cur_score = new_seq, new_score

    return best_seq, best_score


def propose_candidate(seed=0, budget_s=1000, length=500):
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The method performs a few LP‑based updates that
    strictly decrease the maximal convolution bound, followed by a short
    stochastic refinement.  It respects the 1000‑second time budget.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Total wall‑clock time budget (seconds).
    length : int, optional
        Length of the initial arcsine sequence.  Values around 400–800
        give a good trade‑off between optimisation power and LP solve time.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    n = length

    # ---- 1. start from the arcsine construction --------------------
    a = _arcsine_coeffs(n).astype(np.float64)

    # Normalise to the theoretical bound Σ a_i = √(2 n)
    target_sum = math.sqrt(2.0 * n)
    a *= target_sum / a.sum()

    best_seq = a.copy()
    best_score = _compute_score(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ---- 2. LP based improvement loop -----------------------------
    mix_factor = 0.07           # convex‑mixing weight
    max_lp_iters = 12           # limited iterations – each LP is the costliest part
    lp_iter = 0

    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1
        # current maximal convolution value
        conv = np.convolve(a, a)
        M = conv.max()

        a, cur_score = _lp_improve(a, M, target_sum, mix_factor)

        if cur_score < best_score:
            best_seq = a.copy()
            best_score = cur_score
            print(f"[lp {lp_iter}] improved score={best_score:.6f}")

        # simple early‑stop if improvement stalls
        if lp_iter > 2 and best_score / cur_score > 0.9999:
            break

    # ---- 3. Optional stochastic polishing -------------------------
    if time.time() < deadline:
        best_seq, best_score = _stochastic_refinement(best_seq, deadline)

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
