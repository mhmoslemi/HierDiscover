def directed_move(current, max_conv_idx):
    n = len(current)
    # compute gradient array as earlier: 
    # For i where j = max_conv_idx - i in [0,n-1]:
    # partner = j
    # grad[i] = 2 * current[partner]
    k = max_conv_idx
    grad = np.zeros(n)
    idx = k - np.arange(n)   # k - i
    mask = (idx >= 0) & (idx < n)
    grad[mask] = 2.0 * current[idx[mask]]
    # find source and target indices
    i_max = np.argmax(grad)
    i_min = np.argmin(grad)
    # Ensure i_max != i_min and source has mass
    if current[i_max] <= 1e-12:
        # fallback to random transfer
        i, j = np.random.choice(n, 2, replace=False)
        delta = current[i] * np.random.rand() * 0.3
        cand = current.copy()
        cand[i] -= delta
        cand[j] += delta
        return cand
    delta = current[i_max] * np.random.rand() * 0.5  # move up to 50% of source mass
    cand = current.copy()
    cand[i_max] -= delta
    cand[i_min] += delta
    # ensure non-negative
    cand = np.maximum(cand, 0.0)
    # renormalize
    cand_sum = cand.sum()
    if cand_sum > 0:
        cand /= cand_sum
    else:
        cand = np.ones_like(cand) / n
    return cand

"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Broad seeding** – generate many deterministic “shapes’’
   (U‑shaped power‑law, product power‑law, exponential decay,
   symmetric exponential sum) for several lengths and a grid of
   parameters.  The best of those seeds is kept.

2. **Random exploration** – while time permits random Dirichlet
   vectors and random convex combinations of the deterministic
   bases are tried.  This gives a cheap way to discover better
   shapes that are not covered by the fixed grids.

3. **Stochastic refinement** – a simulated‑annealing loop that
   applies four different mutations:
      * random mass transfer between two positions,
      * a directed transfer that moves mass from the index that
        contributes most to the current maximal convolution entry
        to the index that contributes least,
      * a small gradient step on the active convolution index,
      * a mixing step that blends the current vector with a fresh
        random seed.
   Moves are accepted by a Metropolis rule, the temperature is
   slowly cooled and the global best is updated whenever an
   improvement is found.

The whole routine respects a user‑supplied time budget (default
1000 s) and finally returns the best sequence as a plain list of
floats – the external evaluator will clip/normalise it again.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def u_shaped_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric “U” – distance to the nearest edge raised to –p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)          # distance to nearest edge
    a = (d + 1) ** (-p) + 1e-14          # avoid exact zeros
    a /= a.sum()
    return a


def product_power_law_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n-i))^{‑p} – very steep edges."""
    i = np.arange(1, n + 1)               # 1‑based index for formula
    a = (i * (n + 1 - i)) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Monotone decreasing exponential a[i] ∝ exp(-alpha*i/n)."""
    i = np.arange(n)
    a = np.exp(-alpha * i / n) + 1e-14
    a /= a.sum()
    return a


def symmetric_exponential_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric sum of two exponentials a[i] ∝ e^{-α i/n}+e^{-α (n-1‑i)/n}."""
    i = np.arange(n)
    a = (np.exp(-alpha * i / n) + np.exp(-alpha * (n - 1 - i) / n)) + 1e-14
    a /= a.sum()
    return a


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a vector already
    normalised to sum == 1.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    return 2.0 * n * max_c

# Main proposal routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function runs for at most ``budget_s``
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1.  Deterministic grid search over a few lengths and shape families
    # ------------------------------------------------------------------
    base_lengths = [2000, 4000, 8000]                # cheap enough for many seeds
    shape_names = ["u", "product", "exp", "symexp"]

    # parameter grids (coarse)
    p_grid_u = np.linspace(0.1, 3.0, 25)            # for U‑shape and product
    p_grid_prod = np.linspace(0.2, 2.5, 25)
    alpha_grid_exp = np.linspace(0.3, 5.0, 25)      # exponential parameter α

    print("=== Deterministic seeding phase ===")
    for n in base_lengths:
        print(f"--- length = {n} ---")
        # uniform baseline (historical bound = 2.0)
        a_uni = np.full(n, 1.0 / n)
        sc_uni = score_from_vector(a_uni)
        if sc_uni < best_score:
            best_score = sc_uni
            best_seq = a_uni.copy()
            print(f"  uniform    score={sc_uni:.6f}")

        # loop over shape families
        for shape in shape_names:
            if shape == "u":
                for p in p_grid_u:
                    a = u_shaped_distribution(n, p)
                    sc = score_from_vector(a)
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(f"  u‑shape   n={n:5d} p={p:4.2f}  score={sc:.6f}")
            elif shape == "product":
                for p in p_grid_prod:
                    a = product_power_law_distribution(n, p)
                    sc = score_from_vector(a)
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(f"  product   n={n:5d} p={p:4.2f}  score={sc:.6f}")
            elif shape == "exp":
                for a_param in alpha_grid_exp:
                    a = exponential_distribution(n, a_param)
                    sc = score_from_vector(a)
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(f"  exp‑left  n={n:5d} α={a_param:4.2f}  score={sc:.6f}")
            else:   # symexp
                for a_param in alpha_grid_exp:
                    a = symmetric_exponential_distribution(n, a_param)
                    sc = score_from_vector(a)
                    if sc < best_score:
                        best_score, best_seq = sc, a.copy()
                        print(f"  sym‑exp   n={n:5d} α={a_param:4.2f}  score={sc:.6f}")

        # small early‑exit if time is out
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2.  Try the external construction if it exists (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size > 0:
            external = np.maximum(external, 0.0)
            if external.sum() > 0:
                external /= external.sum()
            else:
                external = np.full_like(external, 1.0 / external.size)
            sc_ext = score_from_vector(external)
            if sc_ext < best_score:
                best_score, best_seq = sc_ext, external.copy()
                print(f"  external  length={len(external)}  score={sc_ext:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3.  Randomised exploration while time permits
    # ------------------------------------------------------------------
    print("\n=== Random exploration phase ===")
    while time.time() < deadline:
        # pick a random length (between 2000 and 16000, even numbers)
        n = np.random.randint(2000, 16001)
        n = (n // 2) * 2                     # make it even (convolution symmetry nice)
        # random Dirichlet vector
        a = np.random.rand(n)
        a /= a.sum()
        sc = score_from_vector(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"  random    n={n:5d}  score={sc:.6f}")

        # Random convex combination of two deterministic bases
        shape1, shape2 = np.random.choice(shape_names, 2, replace=False)
        # pick random parameters for each shape
        if shape1 == "u":
            p1 = np.random.uniform(0.05, 4.0)
            a1 = u_shaped_distribution(n, p1)
        elif shape1 == "product":
            p1 = np.random.uniform(0.1, 3.0)
            a1 = product_power_law_distribution(n, p1)
        elif shape1 == "exp":
            a1 = exponential_distribution(n, np.random.uniform(0.2, 6.0))
        else:  # symexp
            a1 = symmetric_exponential_distribution(n, np.random.uniform(0.2, 6.0))

        if shape2 == "u":
            p2 = np.random.uniform(0.05, 4.0)
            a2 = u_shaped_distribution(n, p2)
        elif shape2 == "product":
            p2 = np.random.uniform(0.1, 3.0)
            a2 = product_power_law_distribution(n, p2)
        elif shape2 == "exp":
            a2 = exponential_distribution(n, np.random.uniform(0.2, 6.0))
        else:
            a2 = symmetric_exponential_distribution(n, np.random.uniform(0.2, 6.0))

        w = np.random.rand()
        a = (w * a1 + (1 - w) * a2)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        sc = score_from_vector(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"  mix       n={n:5d}  score={sc:.6f}")

        # break if we are close to the time limit
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 4.  Stochastic local refinement (simulated‑annealing + directed moves)
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fallback – a simple uniform vector
        n = 3000
        best_seq = np.full(n, 1.0 / n)
        best_score = score_from_vector(best_seq)

    current = best_seq.copy()
    current_score = best_score
    n = current.shape[0]

    temperature = 1.0
    cooling_rate = 0.9993
    it = 0
    last_improvement_it = 0

    print("\n=== Stochastic refinement phase ===")
    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # Choose a mutation type
        # ------------------------------------------------------------------
        r = np.random.rand()
        if r < 0.35:
            # ----- random mass transfer between two positions -----
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] <= 0.0:
                continue
            delta = current[i] * np.random.rand() * 0.5      # up to 50 % of source mass
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta
        elif r < 0.65:
            # ----- directed transfer based on the active convolution index -----
            conv = np.convolve(current, current)
            k_max = int(np.argmax(conv))

            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros(n)
            grad[mask] = 2.0 * current[idx[mask]]

            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))

            if current[i_max] <= 0.0:
                # fallback to random transfer
                i, j = np.random.choice(n, 2, replace=False)
                delta = current[i] * np.random.rand() * 0.5
                cand = current.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                delta = current[i_max] * np.random.rand() * 0.5
                cand = current.copy()
                cand[i_max] -= delta
                cand[i_min] += delta
        elif r < 0.85:
            # ----- small gradient step on the active index -----
            conv = np.convolve(current, current)
            k_max = int(np.argmax(conv))

            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros(n)
            grad[mask] = 2.0 * current[idx[mask]]

            step = 0.02 * temperature          # adapt with temperature
            cand = current - step * grad
            cand = np.maximum(cand, 0.0)
            # simple re‑normalisation (project onto the simplex)
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand /= cand_sum
            else:
                cand = np.full_like(cand, 1.0 / n)
        else:
            # ----- blend with a fresh random seed (global perturbation) -----
            shape = np.random.choice(shape_names)
            if shape == "u":
                p = np.random.uniform(0.05, 4.0)
                seed_vec = u_shaped_distribution(n, p)
            elif shape == "product":
                p = np.random.uniform(0.1, 3.0)
                seed_vec = product_power_law_distribution(n, p)
            elif shape == "exp":
                seed_vec = exponential_distribution(n, np.random.uniform(0.2, 6.0))
            else:  # symexp
                seed_vec = symmetric_exponential_distribution(n, np.random.uniform(0.2, 6.0))

            alpha = np.random.rand() * 0.15       # mixing weight (≤ 15 %)
            cand = (1.0 - alpha) * current + alpha * seed_vec
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand /= s
            else:
                cand = np.full_like(cand, 1.0 / n)

        # ------------------------------------------------------------------
        # Evaluate candidate (fast internal version – sum already 1)
        # ------------------------------------------------------------------
        cand_score = score_from_vector(cand)

        # Simulated‑annealing acceptance
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = math.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improvement_it = it
                print(f"Iter {it:6d}: NEW BEST score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Temperature schedule & occasional restart
        # ------------------------------------------------------------------
        temperature *= cooling_rate
        if temperature < 1e-6:
            temperature = 1e-6

        # If we have stalled for a long while, perturb the current state
        if it - last_improvement_it > 50000:
            # small Gaussian jitter followed by re‑normalisation
            jitter = np.random.randn(n) * 0.001
            cand = current + jitter
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand /= s
            else:
                cand = np.full_like(cand, 1.0 / n)
            current = cand
            current_score = score_from_vector(current)
            last_improvement_it = it
            print(f"Iter {it:6d}: restart jitter applied")

    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    # Return as plain Python list (the external evaluator will re‑clip & normalise)
    return best_seq.tolist()
