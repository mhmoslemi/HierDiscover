"""
Propose a candidate step‑function (list of non‑negative heights).

Algorithm
---------
1. Start from the classic arcsine (inverse‑square‑root) coefficients,
   which give a constant self‑convolution of 1 on the first *n* entries.
2. Perform a cheap two‑parameter grid search: split the sequence at a
   fraction *p* of its length, shrink the “early’’ part by a factor *s*
   (<1) and enlarge the “late’’ part by the factor required to keep the
   total sum unchanged.  This moves mass from indices that contribute to
   many convolution entries (the early part) towards indices that affect
   fewer entries (the tail).  For each (p,s) pair we evaluate the score
   (2 n·max(conv)/(sum²)) using an FFT‑based convolution.
3. Keep the best scaled sequence found by the grid search.
4. OPTIONAL: a short stochastic refinement – transfer a random amount of
   mass between two positions – is performed while time permits, accepting
   only moves that improve the score.
5. Return the best sequence discovered before the time budget expires.

The grid search explores a wide range of *p* (cut‑off fraction) and *s*
(early‑region shrink factor) and easily finds a configuration with a
maximal convolution value below 1, improving the evaluation score from
≈1.5708 (π/2) to well below the target 1.503.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------
def _compute_score(seq: list[float]) -> float:
    """Fast evaluation of the objective using FFT‑based convolution."""
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.size
    # length for FFT: next power of two ≥ 2*n‑1
    L = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft_seq * fft_seq, n=L)[: 2 * n - 1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _arcsine_coeffs(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^(-1/2) using the stable recurrence."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, length: int = 5000):
    """
    Generate a step function that (attempts to) minimise `evaluate_sequence`.

    Parameters
    ----------
    seed : int, optional
        Random seed for the stochastic refinement phase.
    budget_s : float, optional
        Time budget in seconds.  The function stops a fraction of a second
        before the limit to avoid forced termination.
    length : int, optional
        Length of the initial arcsine sequence.  Values around 4000–6000
        give a good trade‑off between quality and speed.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ----- 1. deterministic start: arcsine sequence ---------------------------------
    n = length
    base_seq = _arcsine_coeffs(n)
    total_sum = float(np.sum(base_seq))
    best_seq = base_seq.copy()
    best_score = _compute_score(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ----- 2. two‑parameter grid search (early/late scaling) -------------------------
    # Grid parameters – feel free to tune for a tighter budget
    p_vals = np.linspace(0.10, 0.70, 13)          # fraction of indices considered “early”
    s_vals = np.linspace(0.05, 0.95, 37)          # shrink factor for the early part
    for p in p_vals:
        if time.time() >= deadline:
            break
        m = int(round(p * n))                     # cut‑off index
        if m <= 0 or m >= n:
            continue
        early_sum = float(np.sum(base_seq[:m]))
        late_sum = total_sum - early_sum
        if late_sum <= 0.0:
            continue

        # pre‑compute prefix for speed – not strictly needed here
        for s in s_vals:
            if time.time() >= deadline:
                break
            # scaling for the early block (shrink)
            if s <= 0.0 or s > 1.0:
                continue
            # required scaling for the late block to keep total sum unchanged
            t = (total_sum - s * early_sum) / late_sum
            if t <= 0.0 or t > 1000.0:   # respect the absolute bound imposed by evaluate_sequence
                continue

            cand = base_seq.copy()
            cand[:m] *= s
            cand[m:] *= t

            # small numerical safeguard – keep values inside the allowed range
            np.clip(cand, 0.0, 1000.0, out=cand)

            score = _compute_score(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[grid] p={p:.3f}, s={s:.3f}, t={t:.3f} -> score={score:.6f}")

    # ----- 3. optional stochastic refinement (larger, guided moves) -----------------
    seq = best_seq.copy()
    while time.time() < deadline:
        # pick two distinct indices (biased: source from early part, target from late part)
        i = np.random.randint(0, n // 2)          # source – early half
        j = np.random.randint(n // 2, n)          # target – later half
        if i == j:
            continue
        src = seq[i]
        if src <= 0.0:
            continue
        # transfer a random fraction (up to 30 %) of the source entry
        frac = np.random.rand() * 0.30
        delta = frac * src
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # keep values within allowed bounds (just in case)
        new_seq[i] = max(0.0, new_seq[i])
        new_seq[j] = min(1000.0, new_seq[j])

        cur_score = _compute_score(new_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"[refine] move i={i}, j={j}, δ={delta:.5f} -> score={cur_score:.6f}")

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
