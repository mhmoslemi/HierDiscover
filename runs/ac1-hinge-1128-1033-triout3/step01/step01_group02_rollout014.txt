Search for a step‑function with a lower evaluation score than the

# Helper: Catalan / central‑binomial coefficients (the classic start)
# ----------------------------------------------------------------------
def catalan_sequence(n: int) -> np.ndarray:
    """Return the first ``n`` Catalan‑type coefficients:

    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2 * k - 1) / (2 * k)
    return a

#          constraints (f * g)_k = Σ_i f_i g_{k-i}.
# ----------------------------------------------------------------------
