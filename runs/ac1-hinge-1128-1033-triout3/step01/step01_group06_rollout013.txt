        conv = np.convolve(a, a)

        # Build matrix A

def simulated_annealing(a_initial, conv_initial, n, deadline):
    a = a_initial.copy()
    conv = conv_initial.copy()
    cur_val = 2.0 * n * np.max(conv)
    best_a = a.copy()
    best_val = cur_val
    iteration = 0
    T_start = 0.1
    T_end = 1e-4
    total_budget = deadline - time.time()
    while time.time() < deadline:
        progress = (time.time() - start_time) / (deadline - start_time)
        T = T_start * (1.0 - progress) + T_end * progress
        if np.random.rand() < 0.7:
            src, dst = targeted_move(a, conv)
        else:
            src = np.random.randint(n)
            dst = np.random.randint(n-1)
            if dst >= src:
                dst += 1
        max_delta = a[src] * 0.5
        if max_delta < 1e-12:
            continue
        delta = np.random.rand() * max_delta
        a_new, conv_new = apply_move(a, conv, src, dst, delta)
        new_val = 2.0 * n * np.max(conv_new)
        improve = new_val < cur_val
        if improve or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[SA {iteration}] new best={best_val:.6f}")
        iteration += 1
        if iteration % 5000 == 0:
            conv = np.convolve(a, a)
    return best_a, best_val

"""
Propose a step‑function sequence that minimises the evaluation metric
2·n·max(conv(a,a)) / (sum(a)²).  The algorithm consists of two phases:

1. **LP improvement phase** –  Following the method described in the
   statement, we repeatedly solve a linear program
        max   Σ b_j
        s.t.  (a * b)ₖ ≤ M   ∀k,
              b_j ≥ 0,
   where M = max(a*a) and a is currently normalised so that
   Σ a = √(2n).  If the LP yields Σb > Σa we can rescale b to the same
   total mass and mix it with a, which provably does not increase M.
   A few iterations of this step quickly drive the objective down.

2. **Fine‑tuning phase** –  A fast simulated‑annealing search with O(n)
   convolution updates (same machinery as the previous baseline) is run
   on the resulting vector for the remaining time budget.

The implementation uses only NumPy and, if available, CVXPY (ECOS
solver) for the linear program.  All numbers are kept non‑negative and
the final sequence is returned as a plain Python list.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast O(n) convolution update when moving a mass δ from index i to j
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Return new (a, conv) after moving `delta` from a[i] to a[j].
    a      – 1‑D array of length n, sum(a) = √(2n)
    conv   – convolution a*a (length 2n‑1)
    i, j   – distinct indices, 0 ≤ i,j < n
    delta  – 0 < delta ≤ a[i]
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta            # change at i
    dj = +delta            # change at j

    # new vector (exact, sum stays unchanged)
    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    # fast convolution update
    conv_new = conv.copy()

    # other indices (vectorised)
    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]                # all other positions
    a_q = a[q]

    # contributions of the changed entries to all sums
    conv_new[i + q] += 2.0 * di * a_q    # (i, q) and (q, i)
    conv_new[j + q] += 2.0 * dj * a_q    # (j, q) and (q, j)

    # self‑pair contributions (i,i) and (j,j)
    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    # cross term (i,j) and (j,i)
    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


# ----------------------------------------------------------------------
# Produce a targeted move focusing on the current maximum of conv
# ----------------------------------------------------------------------
def targeted_move(a: np.ndarray, conv: np.ndarray):
    """
    Choose (src, dst) that are most responsible for the current peak
    of conv and a random destination different from src.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))                     # position of the current peak

    # feasible i so that both i and k-i lie inside the vector
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_candidates = np.arange(i_min, i_max + 1, dtype=int)
    j_candidates = k - i_candidates

    # contribution of each unordered pair to the peak
    contrib = a[i_candidates] * a[j_candidates]
    src = int(i_candidates[np.argmax(contrib)])

    # random destination (different from src)
    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1
    return src, dst


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (works on vectors with Σa = √(2n))
# ----------------------------------------------------------------------
def simulated_annealing(a0: np.ndarray, conv0: np.ndarray,
                        deadline: float, start_time: float):
    n = a0.shape[0]
    a = a0.copy()
    conv = conv0.copy()
    cur_val = float(np.max(conv))
    best_a = a.copy()
    best_val = cur_val
    iteration = 0

    T_start = 0.10
    T_end = 1e-4

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear temperature schedule
        progress = (now - start_time) / (deadline - start_time)
        T = T_start * (1.0 - progress) + T_end * progress

        # move type
        if np.random.rand() < 0.7:          # 70 % targeted
            src, dst = targeted_move(a, conv)
        else:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1

        max_delta = a[src] * 0.5
        if max_delta < 1e-12:
            continue
        delta = np.random.rand() * max_delta

        a_new, conv_new = apply_move(a, conv, src, dst, delta)
        new_val = float(np.max(conv_new))

        improve = new_val < cur_val
        if improve or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a = a_new
            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[SA {iteration:8d}] new best = {best_val:.6f}")

        iteration += 1

        # every few thousand steps recompute convolution to keep numerics clean
        if iteration % 5000 == 0:
            conv = np.convolve(a, a)

    return best_a, best_val
