def solve_lp(f, M):
    n = len(f)
    K = 2*n - 1
    # Build A matrix as shape (K, n)
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(n):
        # fill rows i..i+n-1
        row_start = i
        row_end = min(i + n, K)
        length = row_end - row_start
        A[row_start:row_end, i] = f[:length]
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b_ub = np.full(K, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None

Hybrid linear‑programming / simulated‑annealing optimizer.

1.  Start from a known or simple non‑negative step vector (sum = 1).
2.  Repeatedly solve a linear program:
        max  Σ g_i
        s.t. g_i ≥ 0 and (f * g)_k ≤ M   for all k,
    where f is the current vector and M = max(f*f).
    The LP finds a distribution g that spreads mass to positions where f
    contributes little to the current maximal self‑convolution.
3.  If the self‑convolution of the normalized LP solution improves the
    objective, replace f by this solution; otherwise blend the current
    vector with the LP solution using a decreasing step‑size.
4.  When time permits, run a fast targeted simulated‑annealing
    refinement (the same move logic as the previous best code) to
    escape any remaining local minima.

The evaluation metric is scale‑invariant, so all vectors are kept
normalised to sum = 1.  The algorithm stops after the allotted time and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑programming helper (SciPy if available, otherwise cvxpy)
# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_i  s.t.  g ≥ 0  and  f * g ≤ M.
    Returns the optimal g as a NumPy array, or None on failure.
    """
    n = f.shape[0]
    K = 2 * n - 1

    # Build matrix A where (A @ g)_k = (f * g)_k
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(n):
        row_start = i
        row_end = i + n
        if row_end > K:
            row_end = K
        length = row_end - row_start
        A[row_start:row_end, i] = f[:length]

    # Objective: maximise Σ g_i  → minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(K, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # Try SciPy first (fast for dense LPs)
    try:
        from scipy.optimize import linprog

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass

    # Fallback to cvxpy
    try:
        import cvxpy as cp

        g = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
        prob.solve(solver=cp.ECOS, warm_start=False, verbose=False)
        if prob.status == cp.OPTIMAL and g.value is not None:
            return np.maximum(np.asarray(g.value).flatten(), 0.0)
    except Exception:
        pass

    return None


# ----------------------------------------------------------------------
# Fast convolution update for simulated‑annealing moves
# ----------------------------------------------------------------------
def apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Move `delta` from index i to index j, updating both the vector
    and its self‑convolution in O(n) time.
    """
    n = a.shape[0]
    a_i = a[i]
    a_j = a[j]

    di = -delta
    dj = +delta

    a_new = a.copy()
    a_new[i] = a_i - delta
    a_new[j] = a_j + delta

    conv_new = conv.copy()

    idx = np.arange(n)
    mask = (idx != i) & (idx != j)
    q = idx[mask]
    a_q = a[q]

    conv_new[i + q] += 2.0 * di * a_q
    conv_new[j + q] += 2.0 * dj * a_q

    conv_new[2 * i] += 2.0 * di * a_i + di * di
    conv_new[2 * j] += 2.0 * dj * a_j + dj * dj

    cross = i + j
    conv_new[cross] += 2.0 * (a_i * dj + a_j * di + di * dj)

    return a_new, conv_new


def targeted_move(a: np.ndarray, conv: np.ndarray):
    """
    Choose a source index that participates most in the current maximal
    convolution entry and a random destination.
    """
    n = a.shape[0]
    k = int(np.argmax(conv))

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i_cand = np.arange(i_min, i_max + 1, dtype=int)
    j_cand = k - i_cand

    contrib = a[i_cand] * a[j_cand]
    src = int(i_cand[np.argmax(contrib)])

    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1
    return src, dst
