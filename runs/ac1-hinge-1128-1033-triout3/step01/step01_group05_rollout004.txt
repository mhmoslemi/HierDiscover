"""
Propose a step‑function sequence that minimizes the evaluation metric.
The algorithm starts from the analytically optimal binomial coefficients,
normalises the sum to √(2 n) (so the metric equals the maximum of the
self‑convolution).  It then performs a few iterations of the
classical LP‑based refinement described in the problem statement:
    maximise 𝟙⋅g  subject to g ≥ 0 and  f∗g ≤ M,
where M = max(f∗f).  After each LP we scale g back to the normalisation,
mix it with the current f using a small step size and recompute M.
A short stochastic hill‑climb finishes the search.
All operations respect the 1000 s time budget.
"""

import time
import numpy as np

# SciPy is allowed – we use the high‑performance “highs” LP solver.
from scipy.optimize import linprog
from scipy.sparse import diags, csr_matrix


def generate_binomial_prefix(max_n: int) -> np.ndarray:
    """Return the first ``max_n`` central‑binomial coefficients
    b_k = C(2k,k)/4**k.
    """
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, max_n):
        # recurrence b_{k} = b_{k-1} * (2k-1)/(2k)
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """Return the sparse matrix A such that A @ g = f ∗ g (full convolution).
    The matrix has shape (2·len(f)-1, len(f)).
    """
    n = f.shape[0]
    # each diagonal with offset i (0 ≤ i < n) contains the constant f[i]
    diagonals = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)  # 0,1,…,n‑1
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The procedure consists of:
      1. initialise with a binomial prefix,
      2. a few LP‑based refinement steps (the “alternating LP” method),
      3. a thin stochastic hill‑climb to polish the result.
    The function respects a 1000 s time budget and returns the best
    sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a safety margin

    best_seq = None
    best_score = float('inf')
    best_len = None

    # Lengths to try – short enough for a fast LP but long enough to
    # approach the asymptotic bound.
    length_candidates = [600, 1000, 1500, 2000, 2500]

    # Global step size for the convex combination (t in the description)
    t_step = 0.4

    for n in length_candidates:
        if time.time() >= deadline:
            break

        # ---- Phase 0 : initialise ------------------------------------------------
        coeffs = generate_binomial_prefix(n)
        target_sum = np.sqrt(2.0 * n)               # we keep ∑a = √(2n)
        f = coeffs * (target_sum / np.sum(coeffs))  # normalised start vector

        # current objective = max convolution (because sum is fixed)
        cur_M = np.max(np.convolve(f, f))
        cur_score = cur_M                           # identical to evaluate_sequence
        # record global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.tolist()
            best_len = n
            print(f"[init] n={n}, score={cur_score:.6f}")

        # ---- Phase 1 : alternating LP (few iterations) -------------------------
        max_lp_iters = 12
        t = t_step

        for it in range(max_lp_iters):
            if time.time() >= deadline:
                break

            # Build the convolution matrix for the current f
            A = build_conv_matrix(f)
            b = np.full(2 * n - 1, cur_M)

            # Linear program: maximise 1·g   ⇔   minimise –1·g
            c = -np.ones(n)                # objective vector
            bounds = [(0, None)] * n

            # solve
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:
                # If the LP fails we abort the inner loop – it is very unlikely.
                print(f"[LP fail] n={n}, it={it}, msg={res.message}")
                break

            g = res.x
            sum_g = np.sum(g)
            if sum_g <= 0:
                break

            # Rescale g to keep the normalisation ∑g = √(2n)
            g = g * (target_sum / sum_g)

            # Mix the new candidate with the current f
            new_f = (1.0 - t) * f + t * g

            # Compute the new maximum of the self‑convolution
            new_M = np.max(np.convolve(new_f, new_f))

            # Because the sum stays constant, the evaluation score equals new_M.
            new_score = new_M

            # Accept the update only if we made progress.
            if new_score < cur_score - 1e-12:
                f = new_f
                cur_M = new_M
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = f.tolist()
                    best_len = n
                    print(f"[improve] n={n}, it={it}, score={new_score:.6f}")
                # keep the same step size
            else:
                # Reduce the mixing factor – a smaller step is more likely to
                # maintain the “max‑convolution ≤ M” property.
                t *= 0.5
                if t < 1e-4:
                    break   # nothing more can be gained for this length

        # ---- Phase 2 : light stochastic hill‑climb ------------------------------
        # Use the time that is still left for a quick random walk.
        cur_seq = f
        cur_score = cur_M
        sigma_rel = 0.07          # modest relative scale for Gaussian moves
        no_improve = 0
        max_no_improve = 3000

        while time.time() < deadline and no_improve < max_no_improve:
            idx = np.random.randint(n)
            old_val = cur_seq[idx]
            delta = np.random.randn() * sigma_rel * max(old_val, 1e-12)
            new_val = max(0.0, old_val + delta)

            cand = cur_seq.copy()
            cand[idx] = new_val

            # evaluate – the metric is just the max convolution because the sum
            # is still normalised.
            cand_M = np.max(np.convolve(cand, cand))
            cand_score = cand_M

            if cand_score < cur_score:
                cur_seq = cand
                cur_score = cand_score
                no_improve = 0
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.tolist()
                    best_len = n
                    print(f"[hill‑climb] n={n}, idx={idx}, score={cand_score:.6f}")
            else:
                no_improve += 1

        # End of this length – we keep the best sequence globally.
        # If we have already beaten the target we can stop early.
        if best_score < 1.503:
            break

    # Final sanitisation – clamp to the range required by evaluate_sequence.
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Final debug print
    print(f"[final] length={best_len}, score={best_score:.6f}")

    return best_seq

"""
Propose a step‑function sequence that minimises the metric used by
`evaluate_sequence`.  The method follows the LP‑based refinement described
in the statement:

    * start from the analytically optimal binomial prefix,
    * normalise so that Σa = √(2 n) (this makes the objective equal to
      max(a∗a)),
    * repeatedly solve the linear programme
          maximise Σ g_j  subject to  g≥0  and  f∗g ≤ M,
      where M = max(f∗f) and f is the current sequence,
    * rescale g to the normalisation, mix it with f using a small
      convex‑combination step, and recompute M,
    * finish with a lightweight stochastic hill‑climb.

All operations respect the 1000 s budget; the best sequence found is
returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags, csr_matrix


def generate_binomial_prefix(max_n: int) -> np.ndarray:
    """Return the first ``max_n`` coefficients
    b_k = C(2k, k) / 4**k (central binomial / 4^k)."""
    coeffs = np.empty(max_n, dtype=np.float64)
    cur = 1.0
    coeffs[0] = cur
    for k in range(1, max_n):
        # recurrence  b_k = b_{k-1} * (2k-1) / (2k)
        cur = cur * (2 * k - 1) / (2 * k)
        coeffs[k] = cur
    return coeffs


def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Construct the sparse matrix A such that (A @ g) == f ∗ g,
    where ∗ is the discrete full convolution.
    Shape of A is (2·len(f)-1, len(f)).
    """
    n = f.shape[0]
    # each diagonal with offset i (0 ≤ i < n) consists of the constant f[i]
    diagonals = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)               # 0, 1, …, n‑1
    A = diags(diagonals, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The algorithm consists of:
      1. initial binomial prefix,
      2. a few LP‑based refinement steps (the “alternating LP” method),
      3. a short stochastic hill‑climb for polishing.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a safety margin

    best_seq = None
    best_score = float('inf')
    best_len = None

    # Lengths to try – moderate sizes keep the LP cheap while still
    # approaching the asymptotic bound.
    length_candidates = [600, 1000, 1500, 2000, 2500]

    # Convex‑combination step size (t in the description)
    base_t = 0.4

    for n in length_candidates:
        if time.time() >= deadline:
            break

        # -----------------------------------------------------------------
        # Phase 0 – initialise from the binomial prefix (normalised)
        # -----------------------------------------------------------------
        coeffs = generate_binomial_prefix(n)
        target_sum = np.sqrt(2.0 * n)                # we enforce Σa = √(2n)
        f = coeffs * (target_sum / np.sum(coeffs))   # normalised start vector

        cur_M = np.max(np.convolve(f, f))            # max self‑convolution
        cur_score = cur_M                            # because Σa² = 2n
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.tolist()
            best_len = n
            print(f"[init] n={n}, score={cur_score:.6f}")

        # -----------------------------------------------------------------
        # Phase 1 – alternating LP refinement (few iterations)
        # -----------------------------------------------------------------
        max_lp_iters = 12
        t = base_t

        for it in range(max_lp_iters):
            if time.time() >= deadline:
                break

            # Build the convolution matrix for the current f
            A = build_conv_matrix(f)
            # Slightly relax the RHS to avoid numerical infeasibility
            b = np.full(2 * n - 1, cur_M + 1e-12)

            # Linear programme: maximise Σ g_j  ⇔  minimise –Σ g_j
            c = -np.ones(n)               # objective vector
            bounds = [(0, None)] * n       # g_j ≥ 0

            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})

            if not res.success:
                print(f"[LP fail] n={n}, it={it}, msg={res.message}")
                break

            g = res.x
            sum_g = np.sum(g)
            if sum_g <= 0:
                break

            # Rescale g to keep the normalisation Σg = √(2n)
            g = g * (target_sum / sum_g)

            # Convex combination of current f and the new g
            new_f = (1.0 - t) * f + t * g

            # New max of the self‑convolution (still the objective value)
            new_M = np.max(np.convolve(new_f, new_f))
            new_score = new_M

            # Accept only if we make progress
            if new_score < cur_score - 1e-12:
                f = new_f
                cur_M = new_M
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = f.tolist()
                    best_len = n
                    print(f"[improve] n={n}, it={it}, score={new_score:.6f}")
                # keep the same step size
            else:
                # Reduce the mixing factor – a smaller step is safer
                t *= 0.5
                if t < 1e-4:
                    break

        # -----------------------------------------------------------------
        # Phase 2 – light stochastic hill‑climb (uses the same normalisation)
        # -----------------------------------------------------------------
        cur_seq = f
        cur_score = cur_M
        sigma_rel = 0.07      # relative std‑dev for Gaussian proposals
        no_improve = 0
        max_no_improve = 3000

        while time.time() < deadline and no_improve < max_no_improve:

            old_val = cur_seq[idx]
            delta = np.random.randn() * sigma_rel * max(old_val, 1e-12)
            new_val = max(0.0, old_val + delta)

            cand = cur_seq.copy()
            cand[idx] = new_val
            cand_M = np.max(np.convolve(cand, cand))
            cand_score = cand_M

            if cand_score < cur_score:
                cur_seq = cand
                cur_score = cand_score
                no_improve = 0
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.tolist()
                    best_len = n
                    print(f"[hill‑climb] n={n}, idx={idx}, score={cand_score:.6f}")
            else:
                no_improve += 1

        # If we have already beaten the target we can stop early.
        if best_score < 1.503:
            break

    # -----------------------------------------------------------------
    # Final sanitisation (clamping to the range required by evaluate_sequence)
    # -----------------------------------------------------------------
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Compute the true score with the provided evaluator for reporting.
    final_score = evaluate_sequence(best_seq)
    print(f"[final] length={best_len}, score={final_score:.6f}")

    return best_seq
