"""
This proposal builds a starting step‑function from a short inverse‑sqrt core
plus a constant tail, then refines it through a sequence of linear‑program
updates.  In each iteration we fix the current shape `f`, compute the current
maximum self‑convolution `M = max(f*f)`, and solve the LP

    max   Σ b_i
    s.t.  (f * b)_k ≤ M   for all k,
          b_i ≥ 0.

The optimal `b` gives the largest possible ℓ¹‑norm that does not increase the
convolution bound.  We scale `b` to the normalisation √(2 n) and move the
current shape a small step towards it.  This monotone update keeps `M`
non‑increasing while the ℓ¹‑norm grows, which monotonically reduces the
objective `2·n·M/Σa²`.  After a few LP steps a short simulated‑annealing
finetuning is applied.  The whole procedure runs within the allotted time
budget and returns the best sequence found.
"""

import time
import numpy as np

# optional imports
try:
    from scipy.linalg import toeplitz
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:
    _SCIPY_AVAILABLE = False

def _conv_max(seq: np.ndarray) -> float:
    """
    Compute the maximum of the linear convolution seq * seq using FFT.
    """
    n = seq.shape[0]
    # FFT size is next power of two for speed
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _score(seq: np.ndarray) -> float:
    """
    Exact evaluation of the benchmark objective.
    """
    n = seq.shape[0]
    total = float(seq.sum())
    if total < 0.01:
        return np.inf
    max_conv = _conv_max(seq)
    return 2.0 * n * max_conv / (total * total)

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n-1) × n) representing the linear operator
    b ↦ f * b.  Row k contains f_{k-j} for column j whenever the index is
    inside the support of f, otherwise 0.
    """
    n = f.shape[0]
    # first column = f padded with zeros below
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    # first row = f_0 followed by zeros (so that A[0, j>0]=0)
    first_row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = toeplitz(first_col, first_row)[: 2 * n - 1, :]
    return A

def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear program:
        maximize   sum(b)
        subject to  (f * b)_k ≤ M   for every convolution offset k,
                    b_i ≥ 0.
    Returns the optimal vector b (may be all zeros if infeasible).
    """
    if not _SCIPY_AVAILABLE:
        raise RuntimeError("SciPy is required for the LP step")
    n = f.shape[0]
    A = _build_conv_matrix(f)          # shape (2n-1, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    # linprog solves a minimization problem, so we minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    # Use the HiGHS interior‑point solver (fast for dense problems)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a step‑function sequence with low evaluation score.

    The routine proceeds in three phases:
      1. a cheap grid search over a simple core‑+‑tail family (inverse‑sqrt
         core, constant tail);
      2. a few iterations of the LP “mass‑redistribution” step described
         above;
      3. an optional short simulated‑annealing polish.

    The best sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a working length.  2000 gives a good balance between quality
    # and the size of the LP (≈8 MiB dense matrix).
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1", [])
    n = min(len(global_seq), 2000) if global_seq else 2000
    if n < 500:          # guard against pathological very short inputs
        n = 800

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1) Fast grid search over core‑fraction, core exponent and tail height.
    # ------------------------------------------------------------------
    core_fracs = np.linspace(0.05, 0.5, 9)     # 5 % … 50 %
    exponents = [0.5, 0.55, 0.45]              # around the theoretical ½
    log_heights = np.linspace(-6, -0.5, 20)    # 10⁻⁶ … 10⁻⁰·⁵

    for p in exponents:
        for frac in core_fracs:
            core_len = max(2, int(frac * n))
            core = 1.0 / (np.arange(1, core_len + 1, dtype=np.float64) ** p)

            tail_len = n - core_len
            if tail_len <= 0:
                continue

            for logh in log_heights:
                delta = 10.0 ** logh
                tail = np.full(tail_len, delta, dtype=np.float64)
                seq = np.concatenate([core, tail])
                val = _score(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    print(f"[grid] new best: {best_val:.6f} (p={p:.2f}, core={core_len}, δ={delta:.2e})")
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform sequence if grid search produced nothing.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _score(best_seq)

    # ------------------------------------------------------------------
    # 2) LP based refinement (only if SciPy is present).
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE:
        f = best_seq.copy()
        best_f = f.copy()
        best_score = best_val
        M = _conv_max(f)

        # limit the number of LP iterations – each is relatively cheap
        lp_iter = 0
        while time.time() < deadline and lp_iter < 30:
            try:
                b = _solve_lp_max_sum(f, M)
            except Exception as e:
                print(f"[LP] iteration {lp_iter}: solve failed – {e}")

"""
A two‑phase optimiser for the step‑function benchmark.

Phase 1 builds a quick “core+tail” candidate (inverse‑sqrt core, constant
tail) and tunes three simple hyper‑parameters on a fine grid.

Phase 2 refines the candidate with the classic linear‑program update
described in the statement: for the current shape `f` we compute
`M = max(f*f)`, then solve

    maximise   Σ b_i
    subject to  (f * b)_k ≤ M   for every convolution offset k,
                b_i ≥ 0.

The optimal `b` gives the largest possible ℓ¹‑norm that does not increase
the convolution bound.  After scaling `b` to the normalisation
√(2 n) we move a small convex step towards it; this never raises `M`
and therefore monotonically decreases the evaluation function

        L = 2·n·M / (∑a)²  =  M          (because we keep ∑a = √(2 n))

The LP phase is followed by a brief simulated‑annealing polish that
preserves the ℓ¹‑norm.  The whole routine respects the 1000 s budget and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# optional SciPy imports – required for the LP step
# ----------------------------------------------------------------------
try:
    from scipy.linalg import toeplitz
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:                     # pragma: no cover
    _SCIPY_OK = False

# ----------------------------------------------------------------------
# convolution utilities
# ----------------------------------------------------------------------
def _conv_max(seq: np.ndarray) -> float:
    """Maximum of the linear convolution of `seq` with itself (FFT)."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())

def _score(seq: np.ndarray) -> float:
    """Exact evaluation of the benchmark objective."""
    n = seq.shape[0]
    total = float(seq.sum())
    if total < 0.01:
        return np.inf
    max_conv = _conv_max(seq)
    return 2.0 * n * max_conv / (total * total)

# ----------------------------------------------------------------------
# LP helpers (only usable when SciPy is available)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2n‑1)×n) that maps a vector `b` to the
    linear convolution `f * b`.  Row `k` contains `f_{k‑j}` for each column
    `j` (zero outside the support of `f`).
    """
    n = f.shape[0]
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    first_row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = toeplitz(first_col, first_row)[: 2 * n - 1, :]
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Linear program:
        maximise   Σ b_i
        subject to  (f * b)_k ≤ M   for all offsets k,
                    b_i ≥ 0.
    Returns the optimal vector `b`.
    """
    if not _SCIPY_OK:                       # pragma: no cover
        raise RuntimeError("SciPy is required for the LP step")
    n = f.shape[0]
    A = _build_conv_matrix(f)               # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises, so we minimise -Σb.
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS interior‑point solver (fast for dense problems)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:                     # pragma: no cover
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights with a low evaluation score.

    The function runs a fast grid search, an LP‑based improvement loop,
    and a short simulated‑annealing polishing phase, always respecting the
    time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a safety margin

    # --------------------------------------------------------------
    # Choose a working length.  2000 gives a good trade‑off between
    # solution quality and LP size (~8 MiB dense matrix).
    # --------------------------------------------------------------
    global_seq = globals().get("height_sequence_1", [])
    n = min(len(global_seq), 2000) if global_seq else 2000
    if n < 500:                              # defensive minimum
        n = 800

    best_raw = None
    best_val = np.inf

    # --------------------------------------------------------------
    # 1) QUICK GRID SEARCH (core + constant tail)
    # --------------------------------------------------------------
    core_fracs = np.linspace(0.05, 0.5, 9)          # 5 % … 50 %
    exponents = [0.5, 0.55, 0.45]                  # explore around ½
    log_heights = np.linspace(-6, -0.5, 20)        # tail heights 10⁻⁶ … 10⁻⁰·⁵

    for p in exponents:
        for frac in core_fracs:
            core_len = max(2, int(frac * n))
            core = 1.0 / (np.arange(1, core_len + 1, dtype=np.float64) ** p)

            tail_len = n - core_len
            if tail_len <= 0:
                continue

            for logh in log_heights:
                delta = 10.0 ** logh
                tail = np.full(tail_len, delta, dtype=np.float64)
                seq = np.concatenate([core, tail])
                val = _score(seq)
                if val < best_val:
                    best_val = val
                    best_raw = seq.copy()
                    print(f"[grid] new best: {best_val:.6f} (p={p:.2f}, "
                          f"core={core_len}, δ={delta:.2e})")
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # Fallback to a simple uniform sequence if nothing was explored.
    if best_raw is None:
        best_raw = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _score(best_raw)

    # --------------------------------------------------------------
    # 2) NORMALISE to √(2 n)  – after this the score equals the
    #    maximal self‑convolution M.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    f = best_raw * (target_sum / best_raw.sum())
    M = _conv_max(f)          # here M == current score
    best_f = f.copy()
    best_M = M
    print(f"[init] normalised score (max conv) = {M:.6f}")

    # --------------------------------------------------------------
    # 3) LP‑based refinement – only if SciPy is present.
    # --------------------------------------------------------------
    if _SCIPY_OK:
        lp_iters = 0
        while time.time() < deadline and lp_iters < 30:
            try:
                b = _solve_lp_max_sum(f, M)
            except Exception as exc:               # pragma: no cover
                print(f"[LP] solve failed ({exc}), aborting LP phase")
                break

            sum_b = b.sum()
            if sum_b < 1e-12:
                print("[LP] zero‑mass solution – stop")
                break

            # Scale to the normalisation √(2 n)
            g = b * (target_sum / sum_b)

            # Convex step towards g; start with t=0.2 and back‑track.
            t = 0.2
            while t > 1e-4:
                f_new = (1.0 - t) * f + t * g
                M_new = _conv_max(f_new)
                if M_new <= M * (1.0 + 1e-12):
                    f, M = f_new, M_new
                    if M_new < best_M:
                        best_M = M_new
                        best_f = f_new.copy()
                        print(f"[LP] iter {lp_iters:02d} improved M to {best_M:.6f} (t={t:.4f})")
                    break
                t *= 0.5
            else:
                # No acceptable step size – stop the LP loop.
                print("[LP] cannot find a non‑increasing step, terminating LP phase")
                break

            lp_iters += 1

    # --------------------------------------------------------------
    # 4) SHORT simulated‑annealing polish (keeps ℓ¹‑norm fixed).
    # --------------------------------------------------------------
    cur_seq = best_f.copy()
    cur_M = best_M
    start = time.time()
    T0 = max(0.01, 0.1 * cur_M)          # initial temperature
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j or cur_seq[i] <= 0:
            continue
        # Transfer up to 2 % of the donor entry.
        amount = cur_seq[i] * np.random.rand() * 0.02
        new_seq = cur_seq.copy()
        new_seq[i] -= amount
        new_seq[j] += amount
        # Sum stays unchanged, so denominator of the score is constant.
        new_M = _conv_max(new_seq)

        if new_M < cur_M or np.random.rand() < np.exp(-(new_M - cur_M) / T0):
            cur_seq, cur_M = new_seq, new_M
            if new_M < best_M:
                best_M = new_M
                best_f = new_seq.copy()
                print(f"[SA] improved M to {best_M:.6f}")

        # Linear cooling.
        elapsed = time.time() - start
        T0 = max(1e-6, T0 * (1.0 - elapsed / budget_s))

    print(f"final best value (max convolution) = {best_M:.6f}")
    # The evaluation harness expects the raw list; the score will be computed
    # by `evaluate_sequence` later.
    return best_f.tolist()
