def propose_candidate(seed=0, budget_s=1000, length=500):
     np.random.seed(seed)
     deadline = time.time() + budget_s - 0.5

     n = length
     # generate arcsine coefficients
     a = arcsine_coeffs(n)
     a = np.array(a, dtype=np.float64)
     # normalize sum to sqrt(2n)
     target_sum = np.sqrt(2.0 * n)
     a = a * (target_sum / a.sum())
     # compute initial score: objective = M = max(conv)
     conv = np.convolve(a, a)
     M = conv.max()
     best_score = M
     best_seq = a.tolist()
     print(f"[init] n={n} initial M={M:.6f}")

     # iteration parameters
     t = 0.2  # step size
     iteration = 0
     while time.time() < deadline:
         iteration += 1
         # Build matrix A
         # A_ub in shape (2n-1, n)
         # might be built using SciPy's sparse matrix for speed; but we can use dense.
         rows = 2*n - 1
         A = np.zeros((rows, n), dtype=np.float64)
         for k in range(rows):
             i_start = max(0, k - (n - 1))
             i_end = min(k, n - 1)
             # vectorized fill:
             # j indices = range(i_start to i_end) maybe not directly
             # We'll loop i perhaps (small).
             for i in range(i_start, i_end+1):
                 j = k - i
                 A[k, j] = a[i]
         # Solve LP maximize sum g subject to A @ g <= M (with M as current max)
         c = -np.ones(n)  # minimize -sum
         bounds = [(0, None)] * n
         res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
         if not res.success:
             # If LP fails, break
             print("[lp] failed")
             break
         g = res.x
         # If sum is zero or very small, break
         sum_g = g.sum()
         if sum_g < 1e-12:
             break
         # Scale g to have target sum
         g = g * (target_sum / sum_g)

         # Update f via convex combination
         a_new = (1 - t) * a + t * g
         conv_new = np.convolve(a_new, a_new)
         M_new = conv_new.max()
         # Accept if improves (i.e., lower M)
         if M_new < M - 1e-12:
             a = a_new
             M = M_new
             if M_new < best_score:
                 best_score = M_new
                 best_seq = a.tolist()
                 print(f"[iter {iteration}] improved M={M_new:.6f}")
         else:
             # maybe reduce step size gradually
             t *= 0.9
         # Optionally break if M near theoretical bound > 1.0 but can't go below ???.
     
     print(f"[final] best M={best_score:.6f}")
     return best_seq

"""
Propose a candidate step‑function for the “evaluate_sequence” benchmark.

Algorithm
---------
1. Initialise a sequence with the first *n* coefficients of the
   power‑series (1‑z)^(-1/2) (the “arcsine” sequence).  After a simple
   scaling the sum of the coefficients equals √(2 n).  For this
   normalisation the evaluation function reduces to the maximal
   self‑convolution value *M* = max(a*a).

2. Repeatedly improve the sequence by a convex‑combination step.
   For the current sequence *f* (still normalised) we solve a linear
   programme

        maximise   Σ g_i
        subject to g_i ≥ 0
                   (f * g)_k ≤ M   for all k = 0 … 2n‑2

   The constraints are linear because (f*g)_k = Σ_i f_i·g_{k‑i}.
   The LP is solved with ``scipy.optimize.linprog`` (HiGHS).  The
   optimiser returns a vector *g₀* whose sum is typically **much**
   larger than √(2 n) while still respecting the convolution bound.

3. Scale *g₀* back to the required normalisation
   (multiply by √(2 n) / Σg₀).  The scaled vector *g* has the same sum as
   *f* but a strictly smaller cross‑convolution (f*g) ≤ M.

4. Update the current sequence by a small convex step
        f ← (1‑t)·f + t·g
   (t is reduced adaptively when a step does not improve the score).
   Because the sum is preserved, the new evaluation score is simply
   the new maximal self‑convolution *M*.

5. Keep the best sequence seen throughout the allotted time budget and
   finally return it.

The method is an implementation of the “alternating linear‑program”
scheme described in the statement, but it builds the convolution matrix
explicitly with a Toeplitz structure and works with a fixed step size
that is automatically damped.  With a moderate length (≈ 800‑1000)
the linear programmes are tiny (≤ 2 000 constraints, ≤ 1000 variables)
and many iterations comfortably fit into the 1000 s time limit.

If SciPy is unavailable the function falls back to the plain arcsine
construction.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def arcsine_coeffs(n: int) -> np.ndarray:
    """Return the first *n* coefficients of (1‑z)^(-1/2)."""
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return coeffs


def max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the full convolution a*a."""
    return np.max(np.convolve(a, a))


def build_toeplitz_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that A·g = f * g (full convolution).
    Row k contains the shifted copy of f that multiplies g in the k‑th
    entry of the convolution.
    """
    n = f.shape[0]
    # First column: f padded with zeros to length 2n‑1
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    # First row: only the first element of f, rest zeros
    row = np.concatenate([np.array([f[0]], dtype=np.float64),
                          np.zeros(n - 1, dtype=np.float64)])
    # Use SciPy's toeplitz implementation (fallback to manual if unavailable)
    try:
        from scipy.linalg import toeplitz
        A = toeplitz(col, row)
    except Exception:  # pragma: no cover
        # Manual construction – slower but functional
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for k in range(2 * n - 1):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                A[k, j] = f[i]
    return A


# ------------------------------------------------------------
# Main entry point required by the evaluation harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 0,
                      budget_s: float = 1000,
                      length: int = 800) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best score (≈ 1.5708).  The routine follows the LP‑based
    refinement described above and respects the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise with the arcsine (inverse‑square‑root) sequence.
    # ------------------------------------------------------------------
    n = max(2, int(length))                     # length must be at least 2
    target_sum = np.sqrt(2.0 * n)               # required normalisation

    f = arcsine_coeffs(n)
    f = f * (target_sum / f.sum())              # normalise sum to √(2n)

    # Current score is just the maximal self‑convolution (see comment in docstring)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  If SciPy's optimiser is unavailable we cannot do anything else.
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:                         # pragma: no cover
        print("[info] SciPy linprog not available – returning arcsine sequence.")
        return best_seq.tolist()

    # ------------------------------------------------------------------
    # 3.  Iterative LP‑based improvement.
    # ------------------------------------------------------------------
    step_size = 0.3               # initial convex‑combination coefficient
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ----- recompute current maximal convolution M -----------------
        M = max_self_convolution(f)

        # ----- build the linear‑program matrix A -----------------------
        A = build_toeplitz_convolution_matrix(f)          # shape (2n‑1, n)

        # ----- linear programme: maximise Σg_i  subject to A·g ≤ M -----
        c = -np.ones(n, dtype=np.float64)                # linprog minimises
        bounds = [(0.0, None)] * n
        rhs = np.full(2 * n - 1, M, dtype=np.float64)

        # HiGHS is the default high‑performance solver in recent SciPy.
        res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # In the unlikely event that the LP fails we abort the loop.
            print(f"[iter {iteration}] LP failed (status: {res.status}); terminating.")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            print(f"[iter {iteration}] LP returned zero sum; terminating.")
            break

        # ----- rescale g back to the required sum √(2n) -----------------
        g = g * (target_sum / sum_g)

        # ----- convex combination step ---------------------------------
        f_candidate = (1.0 - step_size) * f + step_size * g

        # Ensure numerical non‑negativity (tiny negatives can appear)
        f_candidate = np.maximum(f_candidate, 0.0)

        # ----- evaluate the candidate ---------------------------------
        cand_score = evaluate_sequence(f_candidate.tolist())

        if cand_score < best_score - 1e-12:
            # improvement – keep the new sequence and possibly increase step
            best_score = cand_score
            best_seq = f_candidate.copy()
            f = f_candidate
            print(f"[iter {iteration:4d}] improved score={best_score:.6f}, step={step_size:.3f}")
            # gently increase step size to accelerate progress
            step_size = min(0.9, step_size * 1.05)
        else:
            # no improvement – shrink the step size to become more conservative
            step_size = max(1e-4, step_size * 0.7)

        # occasional early termination if we are already extremely close
        if best_score <= 1.01:  # far below the current known bound
            print("[info] Score already close to theoretical lower bound – stopping.")
            break

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
