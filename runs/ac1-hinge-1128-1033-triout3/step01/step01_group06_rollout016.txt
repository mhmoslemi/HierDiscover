"""
Iterative Frank‑Wolfe style optimisation.

We maintain a non‑negative vector `a` (∑a = 1).  For the current
distribution we compute

    M = max_k (a * a)[k]                (linear convolution)

and solve the linear programme

    maximise    ∑ b_j
    subject to  a * b ≤ M    (componentwise)
                b_j ≥ 0

where the convolution `a*b` is linear in `b`.  The LP gives a vector `b`
with a larger total mass than `a`.  After normalising it to a unit sum we
obtain `g`.  Mixing `g` with `a` strictly reduces the maximal
self‑convolution, and we repeat the process until the time budget expires.

The LP is solved with ``scipy.optimize.linprog`` (HiGHS).  The mixing
parameter is chosen by a cheap line‑search over a few candidates.
The best sequence seen so far is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the matrix A (shape (2n‑1, n)) such that (A @ b)[k] ==
    (a * b)[k] where ``*`` denotes linear convolution.
    Each column j of A contains the coefficients a[i] for rows i+j.
    """
    n = a.shape[0]
    # rows: for each column j, rows are j, j+1, …, j+n‑1
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)               # a[i] appears in each column
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _line_search_mix(a, g, conv_aa, conv_ag, conv_gg, n):
    """
    Find a mixing coefficient t ∈ [0,1] that minimises
        2 * n * max( ( (1‑t)² conv_aa  + 2 t (1‑t) conv_ag + t² conv_gg ) )
    Returns the new vector a_new and the corresponding value.
    """
    best_val = np.inf
    best_a = a
    best_t = 0.0

    # coarse grid, then a refined local search around the best point
    grid = np.linspace(0.0, 1.0, 21)   # 0,0.05,…,1
    for t in grid:
        if t == 0.0:
            continue
        coeff0 = (1 - t) ** 2
        coeff1 = 2 * t * (1 - t)
        coeff2 = t ** 2
        conv_mix = coeff0 * conv_aa + coeff1 * conv_ag + coeff2 * conv_gg
        val = 2.0 * n * conv_mix.max()
        if val < best_val:
            best_val = val
            best_a = (1 - t) * a + t * g
            best_t = t

    # a few refinement steps around the best grid point
    if best_t not in (0.0, 1.0):
        left = max(0.0, best_t - 0.05)
        right = min(1.0, best_t + 0.05)
        refined = np.linspace(left, right, 21)
        for t in refined:
            coeff0 = (1 - t) ** 2
            coeff1 = 2 * t * (1 - t)
            coeff2 = t ** 2
            conv_mix = coeff0 * conv_aa + coeff1 * conv_ag + coeff2 * conv_gg
            val = 2.0 * n * conv_mix.max()
            if val < best_val:
                best_val = val
                best_a = (1 - t) * a + t * g
                best_t = t
    return best_a, best_val, best_t


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Frank‑Wolfe style optimiser for the step‑function evaluation.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Initialise a – use a previously found construction if available,
    # otherwise start from an “edge‑heavy’’ sine profile.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.array(height_sequence_1, dtype=float)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)
    else:
        x = np.arange(n, dtype=float)
        a = 1.0 - np.sin(np.pi * x / (n - 1))   # edge‑heavy shape
        a = np.maximum(a, 0.0)

    a = a / a.sum()                    # enforce ∑a = 1
    conv_aa = np.convolve(a, a)
    cur_val = 2.0 * n * conv_aa.max()
    best_a = a.copy()
    best_val = cur_val

    print(f"[init] n={n}, initial value={cur_val:.6f}")

    iteration = 0
    no_improve_iters = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # recompute convolution and bound M
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()

        # ------------------------------------------------------------------
        # Build the linear‑programming matrix for the constraint a * b ≤ M
        # ------------------------------------------------------------------
        A = _build_convolution_matrix(a)

        # ------------------------------------------------------------------
        # Solve the LP: maximise ∑b  subject to A b ≤ M, b ≥ 0
        # ------------------------------------------------------------------
        c = -np.ones(n)                      # minimise -∑b  → maximise ∑b
        bounds = [(0.0, None)] * n
        lp_res = linprog(
            c,
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not lp_res.success:
            print(f"[{iteration:6d}] LP failed: {lp_res.message}")
            break

        b_opt = lp_res.x
        sum_b = b_opt.sum()

        # If the LP cannot increase the total mass, we have (almost) converged
        if sum_b <= 1.0 + 1e-12:
            print(f"[{iteration:6d}] No improvement from LP (sum_b={sum_b:.6f})")
            no_improve_iters += 1
            if no_improve_iters >= 5:
                break
            # inject a tiny random perturbation to try escaping a flat region
            a += 1e-5 * np.random.rand(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            continue

        # Normalise the LP solution to obtain a unit‑sum candidate g
        g = b_opt / sum_b

        # ------------------------------------------------------------------
        # Precompute the needed convolutions for line‑search
        # ------------------------------------------------------------------
        conv_ag = np.convolve(a, g)
        conv_gg = np.convolve(g, g)

        # ------------------------------------------------------------------
        # Find a mixing factor t that yields the largest decrease
        # ------------------------------------------------------------------
        a_new, val_new, t_star = _line_search_mix(
            a, g, conv_aa, conv_ag, conv_gg, n
        )

        # Accept the improvement if any
        if val_new < cur_val - 1e-12:
            a = a_new
            cur_val = val_new
            no_improve_iters = 0
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{iteration:6d}] new best = {best_val:.6f}  (t={t_star:.3f})")
        else:
            # No improvement – try a small random move (similar to SA)
            no_improve_iters += 1
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            max_delta = a[src] * 0.2
            if max_delta > 1e-12:
                delta = np.random.rand() * max_delta
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, 0.0)
                a /= a.sum()
            else:
                # if source is essentially zero, just re‑normalise
                a = np.maximum(a, 0.0)
                a /= a.sum()

        # give a chance to break if we have not improved for a while
        if no_improve_iters >= 30:
            print(f"[{iteration:6d}] stagnation – terminating early")
            break

    # ------------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence()
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
