"""
propose_candidate: heuristic search for a non‑negative step‑function that
minimises  2·n·max_k (a*a)[k]  (the evaluation function used in the
challenge).

The algorithm consists of three stages:

1. Deterministic sweep over several parametric, symmetric families
   (U‑shapes, shifted U‑shapes, product‑distance, exponential, cosine)
   for a range of lengths.  The best vector from this sweep seeds the
   stochastic phase.

2. Stochastic refinement on the probability simplex using mirror‑descent
   (exponentiated‑gradient) driven by a *smoothed* sub‑gradient of the
   max‑convolution term.  The smoothing is a soft‑max (log‑sum‑exp)
   with a temperature β that is slowly increased, turning the smooth
   gradient into the true sub‑gradient as the optimisation proceeds.
   An annealing schedule allows occasional uphill moves, and random
   mass‑transfer steps help escape shallow basins.  Stagnation triggers a
   small “re‑heat”.

3. The best vector found (as a plain Python list) is returned.  All
   numbers are non‑negative and sum to one, so the external evaluator
   rescales them automatically.

The routine respects a user‑specified time budget (default 1000 s) and
uses only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (not strictly needed because exponential update
# preserves the simplex, but kept for completeness)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Fast convolution helpers (real FFT)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)-1) using rFFT."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def full_convolution_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via rFFT."""
    n = len(x) + len(y) - 1
    size = 1 << int(np.ceil(np.log2(n)))  # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n]

# ----------------------------------------------------------------------
# Scoring utilities (exact score for a vector with sum = 1)
# ----------------------------------------------------------------------
def score_from_vector(a: np.ndarray) -> float:
    """Return the evaluation score (2·n·max(conv)) for a simplex vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (all symmetric)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))‑p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Soft‑max weighted gradient of the max‑convolution term
# ----------------------------------------------------------------------
def softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute grad = Σ_k w_k * ∂(a*a)[k]/∂a with
    w_k = softmax(beta * conv_k).
    The result is a vector of length len(a).
    """
    # soft‑max weights (sum to 1, numerically stable)
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient contribution for each k: 2 * a_{k-i}
    # Using convolution of reversed a with w
    a_rev = a[::-1]
    conv_rev = full_convolution_fft_general(a_rev, w)          # length 3n‑2
    start = len(w) // 2                                      # = n‑1
    grad = 2.0 * conv_rev[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) update on the simplex
# ----------------------------------------------------------------------
def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform a mirror‑descent update:
        a' ∝ a * exp(-η·grad)
    The result is renormalised to sum = 1.
    """
    # Clip exponent to avoid overflow/underflow
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0 or not np.isfinite(s):
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new

            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (U‑min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (U‑prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

            for c in shift_cs:
                a = u_min_shift_distribution(n, p, c)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"new best (shift)  n={n:5d} p={p:4.2f} c={c:.1f}  score={sc:.6f}")

        # exponential family
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (exp)    n={n:5d} lam={lam:.5f}  score={sc:.6f}")

        # cosine family
        for p in cos_exps:
            a = cos_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (cos)    n={n:5d} p={p:3.1f}  score={sc:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 2. Incorporate an external construction if available
    # --------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        if external.sum() > 0.0:
            external /= external.sum()
            sc = score_from_vector(external)
            if sc < best_score:
                best_score, best_seq = sc, external.copy()
                print(f"new best (external)  score={sc:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3. Fallback random initialise (unlikely to be hit)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 4096
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f"fallback random init n={n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Stochastic refinement (soft‑max mirror descent + annealing)
    # --------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    current_conv = full_convolution_fft(current)
    current_score = 2.0 * n * current_conv.max()

    # hyper‑parameters
    eta = 0.3                 # base step size for EG
    eta_decay = 0.9999
    temperature = 1.0
    temp_decay = 0.9998
    beta = 0.5                # start with a smooth approximation
    beta_inc = 0.0006
    beta_max = 30.0

    iteration = 0
    last_improve = 0

    print("\n=== Stochastic refinement start ===")
    while time.time() < deadline:
        iteration += 1

        # 4a – soft‑max weighted gradient of the max‑convolution term
        grad = softmax_gradient(current, current_conv, beta)

        # 4b – exponentiated‑gradient proposal
        cand = exponential_update(current, grad, eta)

        # 4c – occasional random mass‑transfer move (helps escape)
        if iteration % 128 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] > 0.0:
                delta = current[i] * np.random.rand() * 0.5
                cand2 = current.copy()
                cand2[i] -= delta
                cand2[j] += delta
                cand2 = np.maximum(cand2, 0.0)
                cand2 /= cand2.sum()
                # evaluate both candidates, keep the better
                cand_conv = full_convolution_fft(cand)
                cand_score = 2.0 * n * cand_conv.max()
                cand2_conv = full_convolution_fft(cand2)
                cand2_score = 2.0 * n * cand2_conv.max()
                if cand2_score < cand_score:
                    cand, cand_conv, cand_score = cand2, cand2_conv, cand2_score
                else:
                    # cand_conv already computed earlier
                    pass
            else:
                cand_conv = full_convolution_fft(cand)
                cand_score = 2.0 * n * cand_conv.max()
        else:
            cand_conv = full_convolution_fft(cand)
            cand_score = 2.0 * n * cand_conv.max()

        # 4d – simulated‑annealing acceptance
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            current_conv = cand_conv
            current_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # 4e – schedule updates
        temperature *= temp_decay
        eta *= eta_decay
        beta = min(beta_max, beta + beta_inc)

        # 4f – stagnation handling (re‑heat)
        if iteration - last_improve > 200000:
            bump = np.random.rand(n) * 1e-5
            current = np.maximum(current + bump, 0.0)
            current /= current.sum()
            current_conv = full_convolution_fft(current)
            current_score = 2.0 * n * current_conv.max()
            temperature = 1.0
            eta = 0.3
            beta = 0.5
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    # --------------------------------------------------------------
    # 5. Return the best sequence found
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
