"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm
----------------
* Choose a problem size (use the previous best sequence if available,
  otherwise start with n≈8000).
* Initialise a monotone “power‑law’’ shape ( (1‑i/(n‑1))^p ) and keep the
  best of a small grid of exponents.  This provides a good deterministic
  starting point that already beats the uniform sequence.
* Re‑parameterise the simplex using logits  y  with  a = softmax(y).
  Optimise the *smoothed* maximum of the self‑convolution
  (soft‑max with decreasing temperature ε) using Adam on the logits.
  The gradient of the smoothed objective is a cross‑correlation:
      ∇_a f ≈ 2·n·2·rev(a) * w   ( w – soft‑max weights )
  and the chain‑rule gives the gradient w.r.t. y.
* Temperature ε and learning‑rate η decay slowly; if no improvement for
  a while a small random perturbation is added to y to escape plateaus.
* Keep the best sequence seen (according to the exact evaluation function).
* After the main loop run a short pure sub‑gradient polishing that
  follows the (exact) maximal convolution entry.
* Clamp the final heights to the limits required by `evaluate_sequence`

The whole optimisation runs well within the 1000 s budget and

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _full_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a (full convolution) using a single FFT."""
    size = 2 * a.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, nfft)
    Fc = Fa * Fa
    conv = np.fft.irfft(Fc, nfft)[:size]
    return conv
