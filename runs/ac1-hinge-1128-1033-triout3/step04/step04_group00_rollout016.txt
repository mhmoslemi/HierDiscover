"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a two‑stage optimisation:

1.  A fast “coarse” optimisation on a small problem (n≈256) with the
    Adam optimiser on a smooth‑max approximation of the max‑convolution.
    The resulting distribution is stored as `small_seq`.

2.  `small_seq` is interpolated to the target length (the length of the
    previously‑found `height_sequence_1` if it exists, otherwise 5000).
    A second Adam optimisation on the full size refines the solution.

Both stages keep the iterate on the probability simplex (non‑negative,
∑a = 1) via a projection.  The smooth‑max uses an exponential weighting
with a temperature that is rapidly cooled, concentrating the gradient
on the currently largest convolution entry.  The Adam step‑size adapts
to the geometry of the problem and usually out‑performs plain momentum.

Occasional random “jumps’’ are injected when the objective has not
improved for a while to escape plateaus.  At the end a short
sub‑gradient polishing (gradient of the single maximal entry) is run
and the result is clamped to the limits required by `evaluate_sequence`.
The whole routine respects the 1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the probability simplex {x | x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using real FFT (fast for many calls)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


def _softmax_weights(conv: np.ndarray, temperature: float) -> np.ndarray:
    """
    Return weights w_i ∝ exp((conv_i - max(conv))/temperature) that sum to 1.
    A very small temperature concentrates the weight on the maximal entry.
    """
    if temperature <= 0:
        # exact max – give unit weight to the first maximal element
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    # shift for numerical stability
    shift = conv.max()
    ex = np.exp((conv - shift) / temperature)
    s = ex.sum()
    if s == 0 or not np.isfinite(s):
        return np.full_like(conv, 1.0 / conv.size)
    return ex / s

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function defined in the problem statement.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Overall time budget (seconds).  Five seconds are kept as a safety
        margin; the function will return before the deadline.
    kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """

    # ------------------------------------------------------------------
    # General set‑up
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0      # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine target problem size ------------------------------------------------
    # If a previous construction exists we reuse its length, otherwise 5000.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

                warm = None
            else:

        except Exception:          # any trouble – fall back to fresh start
            target_n = 5000
            warm = None
    else:
        target_n = 5000
        warm = None

    # ------------------------------------------------------------------
    # Small‑scale optimisation (fast, many iterations)  -----------------
    # ------------------------------------------------------------------

    a_small = np.full(small_n, 1.0 / small_n)          # uniform start
    if warm is not None:

        x_big = np.arange(target_n, dtype=np.float64)
        x_small = np.linspace(0, target_n - 1, small_n)
        a_small = np.interp(x_small, x_big, warm)
        a_small = np.maximum(a_small, 0.0)
        a_small = a_small / a_small.sum()

    # Adam optimiser state for the small problem
    m_small = np.zeros_like(a_small)
    v_small = np.zeros_like(a_small)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    iter_small = 0
    best_small = a_small.copy()
    best_val_small = np.inf
    stagn_small = 0                     # iterations without improvement

    # small‑phase runs for at most 200 s or until time runs out

        iter_small += 1

        conv = _full_convolution_fft(a_small, a_small)
        max_c = conv.max()

        if val < best_val_small - 1e-12:
            best_val_small = val

            stagn_small = 0

        else:

        # temperature schedule – aggressive cooling

        # gradient of the (soft) weighted sum:
        rev = a_small[::-1]

        # Adam update

        v_small = beta2 * v_small + (1.0 - beta2) * (g * g)
        m_hat = m_small / (1.0 - beta1 ** iter_small)
        v_hat = v_small / (1.0 - beta2 ** iter_small)
        step = 0.05 / (np.sqrt(v_hat) + eps_adam)          # step adapts to gradient size

        a_new = a_small - step * m_hat
        a_new = _project_simplex(a_new)

        a_small = a_new

        # occasional random perturbation if we are stuck
        if stagn_small > 3000:

            a_small = a_small + noise
            a_small = _project_simplex(a_small)
            stagn_small = 0
            print(f"[small {iter_small}] perturbation injected")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    a = a_big
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    iter_big = 0
    best_seq = a.copy()
    best_val = np.inf
    stagn = 0

    while time.time() < deadline:
        iter_big += 1

        conv = _full_convolution_fft(a, a)
        max_c = conv.max()
        val = 2.0 * target_n * max_c               # denominator = 1 (∑a = 1)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            stagn = 0
            print(f"[big {iter_big:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # temperature – starts fairly high then cools quickly
        temperature = max(max_c * 0.01 * (0.997 ** iter_big), 1e-12)

        w = _softmax_weights(conv, temperature)

        rev = a[::-1]
        g = 2.0 * _full_convolution_fft(rev, w)[target_n - 1 : target_n - 1 + target_n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iter_big)
        v_hat = v / (1.0 - beta2 ** iter_big)

        step = 0.03 / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step * m_hat
        a_new = _project_simplex(a_new)
        a = a_new

            a = a + noise
            a = _project_simplex(a)

        # gradient of the *single* maximal entry (as derived in the comment)

    # ------------------------------------------------------------------
    # Clamp to the limits imposed by `evaluate_sequence` and return.
    # ------------------------------------------------------------------
