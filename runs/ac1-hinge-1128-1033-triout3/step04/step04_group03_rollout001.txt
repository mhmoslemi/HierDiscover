def compute_grad_from_weights(a, w):
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((2*n - 1 + n - 1).bit_length())
    # Could just use np.fft.rfft with size = next pow2 of (len(w)+len(a)-1)
    size = 1 << ((len(w) + len(a) - 1).bit_length())
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa_rev, size)
    # conv length = size, but we need slice [n-1:n-1+n]
    start = n-1
    cross = conv[start:start+n]  # length n
    grad = 2.0 * cross
    return grad
```

"""
Propose a candidate step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm:

1. Initialise from the previously‑found best construction (global
   height_sequence_1) or from a random simplex point.
2. Perform a mirror‑descent / exponentiated‑gradient optimisation on the
   probability simplex.  Instead of using the sub‑gradient of the *hard*
   max, we use a *soft* max based on a soft‑max (log‑sum‑exp) with a
   temperature parameter α that is slowly increased (continuation).  The
   gradient of the smoothed objective is
        ∇ = 2 * ( w ⨂ a )
   where w_i ∝ exp(α·conv_i) and ⨂ denotes the cross‑correlation.
   This yields a dense, informative gradient that helps escape plateaus.
3. An exponential‑gradient update a' ∝ a·exp(−η·∇) is applied,
   followed by a projection onto the simplex (normalisation).  A
   simulated‑annealing acceptance rule and occasional random mass‑transfer
   moves provide additional exploration.
4. If the search stagnates for many iterations a small random “re‑heat”
   perturbation is added.
5. The best sequence seen during the run is returned.

The routine respects a time budget (default 1000 s) and prints progress.

import time

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) of the convolution.
    grad_i = 2 * sum_j w[i+j] * a[j]   (cross‑correlation of w and a)
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = len(w) + n - 1               # length needed for full cross‑correlation
    size = 1 << (L.bit_length())     # next power of two >= L
    fw = np.fft.rfft(w, n=size)
    fa = np.fft.rfft(a_rev, n=size)
    full = np.fft.irfft(fw * fa, n=size)
    start = n - 1                     # location of the “valid” part
    cross = full[start:start + n]
    grad = 2.0 * cross
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent / exponentiated‑gradient step on the simplex:
        a' ∝ a * exp(-η * grad)
    A tiny floor guarantees positivity.
    """
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    a_new = np.maximum(a_new, 1e-12)   # keep all entries strictly positive
    a_new /= a_new.sum()
    return a_new

            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score

            # update best solution seen so far
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # -------- schedule updates ----------
        temperature *= temp_decay
        eta *= eta_decay

        # slowly increase the soft‑max sharpness
        if iteration % 20000 == 0 and alpha_idx < len(alphas) - 1:
            alpha_idx += 1

        # re‑heat if no improvement for a long time
        if iteration - last_improve > 250000:
            noise = np.random.rand(n) * 1e-4
            a = np.maximum(a + noise, 1e-12)
            a /= a.sum()
            cur_score = 2.0 * n * full_convolution_fft(a).max()
            temperature = 1.0
            eta = 1.5
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

        # time check (extra safety)
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 4. Return the best sequence found
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
