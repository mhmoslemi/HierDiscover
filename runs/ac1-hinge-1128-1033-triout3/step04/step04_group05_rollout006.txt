"""
Propose a candidate step‑function using large‑scale Adam optimisation.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) by
optimising a *softened* max‑convolution objective

    L(a) = max_k (a * a)[k]  (soft‑max approximation).

Variables are stored in log‑space (x, a = softmax(x)) so that the simplex
constraint holds automatically.  Each iteration performs:

* FFT‑based auto‑convolution a*a (single forward FFT of a).
* A temperature‑controlled soft‑max over the convolution entries to obtain
  smooth weights w.
* Gradient of the smoothed max with respect to a, computed via a second
  convolution (a with the reversed weight vector) – again using the
  previously computed FFT of a.
* Adam update on the logits x, with a linear learning‑rate schedule.
* Occasional random “jitter’’ moves (mass transfer) to escape plateaux;
  the Adam state is reset after such a move.

The search runs for the whole time budget (default 1000 s) on a modestly
large sequence (n≈5000).  The best sequence encountered is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _auto_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full linear convolution of a with itself (length 2·len(a)-1) using FFT.
    The result is returned as a real array of length exactly 2·n‑1.
    """
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, size: int, fa: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smoothed max‑convolution w.r.t. a.
    Implements g_i = 2 * Σ_k w_k * a_{k-i} using a single convolution.
    """
    # reverse the weight vector and convolve with a
    fw = np.fft.rfft(w[::-1], size)
    conv_full = np.fft.irfft(fa * fw, size)
    n = a.shape[0]
    # the indices that correspond to the original a‑length
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv_full[idx]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a non‑negative step‑function (list of heights) to minimise
    the evaluation function.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so the runner does not cut us off early
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a problem size
    # ------------------------------------------------------------------
    n = 5000                       # larger than before – still fast with FFT

    # ------------------------------------------------------------------
    # Initialise the coefficients on the simplex.
    # We start from a smooth “sine’’ shape (strictly positive) to give a
    # better start than the uniform distribution.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n)      # strictly positive on (0,π)
    a = np.maximum(a, 1e-12)              # avoid exact zeros
    a = a / a.sum()                       # normalise to sum = 1

    # Log‑space representation (softmax guarantees the constraints)
    eps = 1e-14
    x = np.log(np.maximum(a, eps))

    # Adam optimiser state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.90, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps_adam = 1e-8
    base_lr = 0.35                     # initial learning‑rate

    # Pre‑compute the FFT size (must accommodate a * w_rev convolution)
    size = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Book‑keeping of the best sequence seen so far
    # ------------------------------------------------------------------
    conv = _auto_convolution_fft(a, size)
    M = conv.max()
    cur_val = 2.0 * n * M
    best_val = cur_val
    best_a = a.copy()
    print(f"[init] n={n}, initial objective={best_val:.6f}")

    iteration = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1

        # ----- forward FFT of the current a (reused below) -----
        fa = np.fft.rfft(a, size)

        # ----- auto‑convolution a*a -----
        conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
        M = conv.max()
        cur_val = 2.0 * n * M

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ----- temperature schedule for the soft‑max (annealing) -----
        progress = (now - start_time) / (deadline - start_time)  # in [0,1)
        # start with a fairly soft approximation, finish almost exact
        tau_start = M * 0.40
        tau_end   = M * 1e-7
        tau = tau_start * (1.0 - progress) + tau_end * progress

        # ----- soft‑max weights over the convolution entries -----
        if tau > 0.0:
            exp_args = (conv - M) / tau
            max_arg = exp_args.max()
            exp_vals = np.exp(exp_args - max_arg)          # shift for stability
            sum_exp = exp_vals.sum()
            if sum_exp > 0.0 and np.isfinite(sum_exp):
                w = exp_vals / sum_exp
            else:
                # fallback to a hard‑max if something goes wrong
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ----- gradient of the smoothed max w.r.t. a -----
        grad_a = _gradient_from_weights(a, w, size, fa)

        # ----- chain rule: gradient w.r.t. logits x (softmax) -----
        dot = np.dot(grad_a, a)                # ⟨grad_a , a⟩
        grad_x = a * (grad_a - dot)            # ∇_x softmax(x)

        # ----- Adam update for x -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # linear decay of the learning‑rate
        lr = base_lr * (1.0 - progress) + 1e-5 * progress
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- obtain a from the updated logits (softmax) -----
        max_x = x.max()
        exp_x = np.exp(x - max_x)        # shift for numerical stability
        a = exp_x / exp_x.sum()          # stays on the simplex

        # ----- occasional random mass‑transfer (helps escape plateaux) -----
        if np.random.rand() < 0.04:      # ~4 % chance per iteration
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            # move a random fraction of the mass at i to j
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta
            a = a / a.sum()
            # reset the optimiser – the sudden change needs a fresh start
            x = np.log(np.maximum(a, eps))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0

        # ----- a cheap hill‑climbing move targeting the current maximal conv -----
        if iteration % 2000 == 0 and np.random.rand() < 0.30:
            # locate the index where the convolution attains its maximum
            k_max = np.argmax(conv)
            # the contributing pairs (i, k_max‑i) lie in a short interval
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            # product contributions for each i in the interval
            idx_range = np.arange(lo, hi + 1)
            contrib = a[idx_range] * a[k_max - idx_range]
            if contrib.sum() > 0:
                probs = contrib / contrib.sum()
                i_sel = np.random.choice(idx_range, p=probs)
                # choose a destination index different from i_sel
                j_sel = np.random.randint(n - 1)
                if j_sel >= i_sel:
                    j_sel += 1
                delta = np.random.rand() * a[i_sel] * 0.3
                a[i_sel] -= delta
                a[j_sel] += delta
                a = a / a.sum()            # renormalise
                # recompute logits and reset Adam state
                x = np.log(np.maximum(a, eps))
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow, beta2_pow = 1.0, 1.0

    # ------------------------------------------------------------------
    # Final reporting (the framework will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
