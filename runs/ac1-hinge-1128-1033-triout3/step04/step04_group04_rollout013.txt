def deterministic_sweep(deadline):

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimises

    2 * n * max_k (a * a)[k]  / (sum(a)**2)

The routine consists of three stages:

1. **Deterministic sweep** – try a variety of parametric families
   (inverse‑distance, inverse‑product‑distance, symmetric & left‑hand
   exponentials) for several lengths, also mixing each candidate with the
   uniform distribution.  The best vector found is kept as a seed.

2. **Fast pairwise hill‑climbing** – repeatedly move a small amount of mass
   from one position to another.  The convolution is updated in O(n) by
   adding the exact linear and quadratic changes, which makes millions of
   trials feasible.  The step size is slowly reduced and occasional random
   re‑heats help to escape plateaus.

3. **Projected sub‑gradient polishing** – using the sub‑gradient of the
   maximal convolution entry we perform a classic projected gradient descent
   (with momentum–type step‑size adaptation) to obtain a final fine‑tuned
   vector.

All steps respect the 1000 s time budget; the best sequence found is
returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2*n-1)."""
    n = a.shape[0]
    size = 2 * n                     # zero‑padding length (power of two not required)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]

def score_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a vector (any scale)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    max_b = conv.max()
    sum_a = a.sum()
    # avoid division by zero (evaluate_sequence would return inf)
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a (holding sum(a) fixed).
    Uses all indices where the convolution attains (or almost attains) its maximum.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each ordered pair (i,j) contributes a[j] to ∂/∂a[i]
        grad[i_vals] += a[j_vals]
    # because max(k) appears twice in the full convolution (ordered pairs),
    # we multiply by 2 to obtain the true gradient of the objective factor.
    return 2.0 * grad

# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+1)^{‑p}   (symmetric, edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1.0) ** (-p) + 1e-14
    a /= a.sum()
    return a

def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{‑p}   (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1.0) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a

def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))   (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a

def left_exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i)   (asymmetric, decreasing from the left)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a
