"""
Propose a step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Work with a soft‑max parametrisation:
        a_i = exp(w_i) / Σ_j exp(w_j)        (a_i ≥ 0, Σ a_i = 1)
  The objective 2·n·max(conv) is scale‑invariant, so fixing Σa_i=1 is
  harmless.

* Optimise a smooth surrogate of the max‑convolution:
        f_ε(a) = 2·n·ε·log Σ_k exp(conv_k / ε) ,
  where conv = a * a (full convolution).  As ε → 0 the surrogate
  converges to the true objective.

* The gradient of f_ε w.r.t. the soft‑max parameters w can be
  expressed in closed‑form using FFT‑based convolutions (O(n log n)).
  A continuation schedule decreases ε from a relatively large value
  (≈5·10⁻²) down to 10⁻⁹, running a limited‑memory BFGS optimisation
  at each level.

* Several diverse initialisations are tried (uniform, a previous
  solution if available, random Dirichlet draws and simple power‑law
  shapes).  The best exact value seen (via ``evaluate_sequence``) is
  retained.

* After the continuation phase a cheap hill‑climbing refinement performs
  random mass transfers while keeping Σa constant.  The refinement runs
  until the time budget expires.

* The function respects the 1000 s wall‑time limit and prints progress.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)      # eliminate tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective and its gradient (w.r.t. the soft‑max parameters)
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Evaluate the smooth surrogate  f_ε(a) = 2·n·ε·log Σ exp(conv/ε)
    and its gradient w.r.t. the soft‑max parameters w.

    Parameters
    ----------
    w : np.ndarray
        Log‑weights (any real numbers).
    eps : float
        Smoothing parameter (positive).
    n : int
        Length of the underlying sequence.

    Returns
    -------
    L : float
        Smooth surrogate value.
    grad_w : np.ndarray
        Gradient ∂L/∂w (same shape as ``w``).
    """
    # ----- soft‑max (feasible a, Σa = 1) -----
    w_shift = w - np.max(w)          # for numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum                    # a_i ≥ 0, Σ a_i = 1

    # ----- full auto‑convolution a*a -----
    conv = _full_convolution(a)      # length 2·n‑1

    # ----- smooth max via log‑sum‑exp -----
    # L = 2·n·ε·log Σ exp(conv/ε)
    # we shift by the (global) maximum to avoid overflow.
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)     # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * (max_conv + eps * log_sum)

    # ----- gradient w.r.t. a (closed form) -----
    # p_k = exp(conv_k/ε) / Σ_j exp(conv_j/ε)
    # Using the shifted version we have:
    weights = np.exp(shifted)        # proportional to exp(conv/ε)
    p = weights / weights.sum()      # shape (2·n‑1,)

    # g_i = 4·n· Σ_k p_k·a_{k‑i}   (a outside [0,n‑1] ⇒ 0)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)    # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]  # middle n entries correspond to Σ_k p_k a_{k‑i}
    grad_a = 4.0 * n * seg[::-1]    # reverse to get correct ordering

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)      # ∂L/∂w

    return float(L), grad_w.astype(float)


# ----------------------------------------------------------------------
# Hill‑climbing refinement (random mass transfers)
# ----------------------------------------------------------------------


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Simple stochastic refinement: repeatedly move a small fraction of
    mass between two randomly chosen positions while keeping Σa constant.
    Accept only improving moves.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Transfer at most 5 % of the source mass.
        delta = cur_a[i] * 0.05 * rng.random()
        if delta <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # Evaluate exactly (the official, expensive function)
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (step‑function) with a low
    ``evaluate_sequence`` score.  The routine respects a 1000 s time
    budget, tries several initial guesses, runs a continuation L‑BFGS‑B
    optimisation on a smooth surrogate and finishes with a cheap
    hill‑climbing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------
    # Problem size – larger n→more flexibility, still cheap to FFT.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Build a list of diverse starting points.
    # --------------------------------------------------------------
    init_candidates = []

    # 1) Use the previously stored construction if it exists.
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # Linear interpolation to required length.
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        init_candidates.append(a0)
        print("[init] loaded height_sequence_1")
    except Exception:
        pass

    # 2) Uniform start.
    init_candidates.append(np.full(n, 1.0 / n))

    # 3) Random Dirichlet draws (more spiky than uniform).
    rng = np.random.default_rng(seed + 12345)
    for concentration in (0.5, 0.3):
        a = rng.dirichlet(np.full(n, concentration))
        init_candidates.append(a)

    # 4) Simple power‑law shapes a_i ∝ (i+1)^{-p} + (n-i)^{-p}
    for p in (0.2, 0.5, 0.8):
        idx = np.arange(n, dtype=float)
        a = (1.0 / (idx + 1) ** p) + (1.0 / (n - idx) ** p)
        a = a / a.sum()
        init_candidates.append(a)

    best_a = None
    best_val = np.inf

    # --------------------------------------------------------------
    # Continuation schedule for ε (logarithmic spacing).
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-2), -9, num=22)   # 5·10⁻² → 10⁻⁹

    # --------------------------------------------------------------
    # Optimisation loop for each start.
    # --------------------------------------------------------------
    for start_id, a0 in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        # Ensure feasibility.
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        w = np.log(a0 + 1e-12)          # soft‑max parameters

        # Keep track of the best exact value seen in this run.
        try:
            cur_val = evaluate_sequence(a0.tolist())
        except Exception:
            cur_val = np.inf
        if cur_val < best_val:
            best_val = cur_val
            best_a = a0.copy()
            print(f"[start {start_id}] initial exact value = {cur_val:.6f}")

        # Iterate over the ε‑schedule.
        for eps in eps_schedule:
            if time.time() >= deadline:
                break

            # L‑BFGS‑B on the smooth surrogate.
            # Allow more iterations than the previous attempt.
            res = minimize(
                _smooth_obj_and_grad,
                w,
                args=(eps, n),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 800,
                         "ftol": 1e-14,
                         "gtol": 1e-14,
                         "disp": False}
            )
            w = res.x

            # Exact evaluation of the current iterate.
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a = a / a.sum()
            try:
                cur_val = evaluate_sequence(a.tolist())
            except Exception:
                cur_val = np.inf

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[start {start_id} | eps {eps:.1e}] new best = {best_val:.6f}")

            # Occasionally perform a tiny random perturbation to avoid
            # stagnation (only if time permits).
            if time.time() < deadline and rng.random() < 0.05:
                # Random mass transfer (very small) followed by a single L‑BFGS step.
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    delta = a[i] * 0.01 * rng.random()
                    a2 = a.copy()
                    a2[i] -= delta
                    a2[j] += delta
                    w2 = np.log(a2 + 1e-12)
                    # one optimisation step with a moderate ε
                    res = minimize(
                        _smooth_obj_and_grad,
                        w2,
                        args=(max(eps, 1e-4), n),
                        method="L-BFGS-B",
                        jac=True,
                        options={"maxiter": 50,
                                 "ftol": 1e-12,
                                 "gtol": 1e-12,
                                 "disp": False}
                    )
                    w = res.x

        # End of ε‑schedule for this start
        # (no explicit early‑stop; we simply continue with next start.)

    # --------------------------------------------------------------
    # Final hill‑climbing refinement (random mass transfers).
    # --------------------------------------------------------------
    if best_a is None:
        # Fallback – should never happen, but guard against it.
        best_a = np.full(n, 1.0 / n)
        best_val = evaluate_sequence(best_a.tolist())

    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # --------------------------------------------------------------
    # Final report.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
