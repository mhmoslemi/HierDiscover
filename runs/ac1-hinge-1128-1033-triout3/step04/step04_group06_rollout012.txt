"""
Propose a step‑function sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of two phases:

1) **Projected exponentiated‑gradient descent** on a smooth
   soft‑max approximation of the infinity‑norm.  The update is
   multiplicative (mirror descent on the simplex) with momentum,
   a slowly decaying learning‑rate and occasional random jitter.
   This phase runs for roughly 60 % of the time budget and quickly
   drives the peak of the auto‑convolution down.

2) **LP‑based refinement**.  For the current sequence `f` we solve a
   linear program that maximises `∑ g_i` under the constraints
   `f * g ≤ max(f * f)`.  The solution `g` is a very “flat” partner
   to `f`.  Mixing `f ← (1‑t)·f + t·g` (with `t≈0.5`) provably reduces
   the maximum convolution value.  Several LP steps are performed
   until the time budget expires.

Both phases keep the best sequence seen so far.  The final sequence
is returned as a plain list of non‑negative floats (automatically
normalised to sum‑to‑one, which leaves the evaluation invariant).
"""
import time
import numpy as np
from scipy.optimize import linprog

# -------------------------------------------------------------
# Helper: full (linear) convolution using a real FFT
# -------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative artefacts caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv

# -------------------------------------------------------------
# Gradient of the soft‑max approximation of max(a*a)
# -------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i}
    via a single convolution: reverse w, convolve with a,
    then extract the appropriate slice.
    """
    n = a.shape[0]
    # reverse weight vector (length 2n‑1)
    u = w[::-1]
    # full convolution u * a → length 3n‑2
    conv = _full_convolution_fft(u, a)
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g

    """
    Returns a list of non‑negative heights (a step function) that
    minimises the evaluation function within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin – we finish well before the hard deadline
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 1) initialise the sequence
    # ------------------------------------------------------------------
    try:
        # Use the previously discovered construction if it exists
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = a.shape[0]
    except Exception:
        # Otherwise start from a modestly sparse Dirichlet draw
        n = 2000                     # feel free to enlarge (up to ~2500 is safe)
        a = np.random.dirichlet(alpha=np.full(n, 0.5))
    # normalise to the probability simplex (sum = 1)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    best_a = a.copy()
    conv_a = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv_a.max())
    print(f"[init] n={n}, initial evaluation={best_val:.6f}")

    momentum = np.zeros_like(a)

    # hyper‑parameters for the first phase (mirror descent)
    eta_start = 0.30          # initial multiplicative step size
    eta_end   = 0.01          # final step size (slow decay)
    beta = 0.90               # momentum decay
    jitter_prob = 0.018       # how often we perform a random mass transfer
    eps = 1e-15               # small epsilon to avoid division by zero

    # temperature parameters (soft‑max smoothing)
    tau_factor_start = 0.25
    tau_factor_end   = 5e-6

    # decide how long the first phase should last (≈60 % of the budget)
    phase1_deadline = start_time + 0.60 * (deadline - start_time)

    iteration = 0

    # ------------------------------------------------------------------
    # Phase 1: exponentiated‑gradient descent with soft‑max
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= phase1_deadline:
            break

        # current convolution and objective
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()
        cur_val = float(2.0 * n * M)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[phase1 {iteration:7d}] new best = {best_val:.6f}")

        # progress (0→1) for schedules within phase 1
        prog = (now - start_time) / (phase1_deadline - start_time)

        # ---- temperature for the soft‑max (annealed) ----
        tau = (tau_factor_start * (1.0 - prog) + tau_factor_end * prog) * M
        if tau <= 0:
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0
        else:
            # stable soft‑max: shift by the maximum to avoid overflow
            exp_args = (conv_a - M) / tau
            np.clip(exp_args, -80, 80, out=exp_args)   # safety
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()

            else:

        # ---- gradient of the smoothed objective ----
        grad = _gradient_from_weights(a, w)

        momentum = beta * momentum + (1.0 - beta) * grad

        # ---- step‑size schedule (multiplicative) ----
        eta = eta_start * (1.0 - prog) + eta_end * prog

        # ---- multiplicative (exponentiated) update ----
        step = -eta * momentum
        np.clip(step, -50, 50, out=step)   # keep exponent bounded

                src = np.random.choice(src_candidates)
                dst = np.random.randint(n - 1)

                    dst += 1

                a[src] -= delta
                a[dst] += delta

        a = np.maximum(a, eps)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2: Linear‑programming refinement
    # ------------------------------------------------------------------
    # Pre‑allocate the constraint matrix (will be overwritten each LP)
    C = np.empty((2 * n - 1, n), dtype=float)

    # LP mixing parameters
    mix_t_start = 0.6      # initial mixing weight
    mix_t_end   = 0.1      # final mixing weight (decays)
    lp_iteration = 0

    while time.time() < deadline:
        # current convolution and objective
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()
        cur_val = float(2.0 * n * M)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[phase2 {lp_iteration:7d}] new best = {best_val:.6f}")

        # Build the constraint matrix C where
        #   (C @ g)_k = (f * g)_k   for all k = 0 … 2n‑2
        #   C[k, j] = f_{k-j}  if 0 ≤ k‑j < n else 0
        # Fill column‑wise – this is fast in NumPy.
        C.fill(0.0)
        for j in range(n):
            C[j:j + n, j] = a

        # Linear program:
        #   maximise 1·g          (=> minimise -1·g)
        #   subject to C·g ≤ M·1   and g ≥ 0
        c = -np.ones(n, dtype=float)               # minimise -sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=float)  # RHS = M
        bounds = [(0.0, None)] * n                  # g_i ≥ 0

        # solve with HiGHS (the default in recent SciPy)
        lp_res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})

        if lp_res.success:
            g = lp_res.x
            sum_g = g.sum()
            if sum_g > 0:
                g_norm = g / sum_g
                # mixing weight decays slowly during phase 2
                prog2 = (time.time() - phase1_deadline) / (deadline - phase1_deadline)
                mix_t = mix_t_start * (1.0 - prog2) + mix_t_end * prog2
                a = (1.0 - mix_t) * a + mix_t * g_norm
                # renormalise (should already sum to 1, but numerical drift possible)
                a = np.maximum(a, eps)
                a = a / a.sum()
        else:
            # If LP fails we fall back to a tiny random perturbation
            if np.random.rand() < 0.2:
                src = np.random.randint(n)
                dst = np.random.randint(n - 1)
                if dst >= src:
                    dst += 1
                delta = np.random.rand() * a[src] * 0.3
                a[src] -= delta
                a[dst] += delta
                a = np.maximum(a, eps)
                a = a / a.sum()

        lp_iteration += 1

    # ------------------------------------------------------------------
    # Final reporting (the outer framework also calls evaluate_sequence)
    # ------------------------------------------------------------------
    try:

        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
