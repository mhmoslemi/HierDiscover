simplex automatically via the soft‑max representation a = softmax(u).  A
few hundred thousand FFT‑based gradient steps are performed within the

The best sequence seen is returned as a plain Python list of floats.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast full linear convolution using real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x) = 1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: numerically stable soft‑max.
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Return soft‑max of `x` as a probability vector."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)

    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = base.shape[0]
            a0 = np.maximum(0.0, base)
            if a0.sum() == 0.0:
                a0 = rng.random(n)
            a0 = a0 / a0.sum()
        except Exception:
            n = 5000
            a0 = None
    else:
        n = 5000
        a0 = None

    if a0 is None:
        # Build a mixed “nice’’ shape (similar to the previous code)
        x = np.arange(n, dtype=np.float64)

        # Catalan‑type decreasing sequence
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)
            catalan[k] = cur
        catalan = np.maximum(catalan, 0.0)

        # Sine shape (positive everywhere)
        sine = np.sin(np.pi * (x + 0.5) / n)

        # Semi‑circle
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

        # Gaussian centred in the middle
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

        # Simple triangular tent
        triangle = np.minimum(x, n - 1 - x) + 1.0

        bases = [catalan, sine, semicircle, gauss, triangle]
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a0 = sum(w * b for w, b in zip(weights, bases))
        a0 = np.maximum(a0, 0.0)

    # Normalise to a probability vector (sum = 1) and guarantee positivity
    a0 = np.clip(a0, 1e-12, None)
    a0 = a0 / a0.sum()

    # Initialise logits for exponentiated‑gradient (soft‑max) representation
    u = np.log(a0)

    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    # ------------------------------------------------------------------
    lr = 0.5                     # initial learning rate for logits
    lr_min = 5e-5
    lr_decay = 0.9999

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    t_iter = 0

    # ------------------------------------------------------------------
    # Temperature for the soft‑max smoothing of the max‐convolution
    # ------------------------------------------------------------------
    # initialise a temporary sequence to get a reasonable start temperature
    conv0 = _full_convolution_fft(a0, a0)
    max_conv0 = conv0.max()
    temp = max(max_conv0 * 0.5, 1e-9)            # start temperature
    temp_min = max(1e-12, max_conv0 * 1e-6)     # lower bound

    # ------------------------------------------------------------------
    # Tracking the best solution found so far
    # ------------------------------------------------------------------
    best_val = float('inf')
    best_seq = None
    stagn = 0                               # iterations without improvement

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- obtain current probability vector -----
        a = _softmax(u)                     # always sums to 1, non‑negative

        # ----- compute convolution and current score -----
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv           # sum(a)==1 ⇒ denominator = 1

        # ----- keep the best seen -----
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter] new best = {best_val:.12f}  lr={lr:.2e}  eps={temp:.2e}")
        else:
            stagn += 1

        # ----- temperature schedule (annealing) -----
        temp = max(temp * 0.9995, temp_min)

        # ----- soft‑max weights for the smoothed max -----
        # w_i ∝ exp((conv_i - max_conv) / temp)   (all ≤ 1)
        w = np.exp((conv - max_conv) / temp)
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient of the weighted sum (cross‑correlation) -----
        rev_a = a[::-1]
        conv_rev = _full_convolution_fft(rev_a, w)       # length 3n‑2
        g_a = 2.0 * conv_rev[n - 1 : n - 1 + n]          # size n

        # ----- gradient w.r.t. logits (soft‑max Jacobian) -----
        avg = np.dot(a, g_a)
        grad_u = a * (g_a - avg)

        # ----- Adam update of logits -----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- learning‑rate decay -----
        lr = max(lr_min, lr * lr_decay)

        # ----- occasional random perturbation to escape plateaus -----
        if stagn > 3000:
            # add a modest Gaussian jitter to the logits
            u += rng.normal(scale=0.1, size=n)
            stagn = 0
            lr = max(lr, 0.001)
            print("[perturb] added jitter to logits")

    # ------------------------------------------------------------------
    # If for some reason no improvement was recorded, fall back to the
    # last soft‑max vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _softmax(u)

    # ------------------------------------------------------------------
    # Final polishing: pure max‑entry sub‑gradient steps.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    step = 5e-5                     # initial step size for polishing
    for it in range(5000):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        idx = int(conv.argmax())

        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]

        a_new = a - step * g
        a_new = _project_simplex(a_new)

        # evaluate whether we made progress
        new_mx = _full_convolution_fft(a_new, a_new).max()
        if new_mx < mx - 1e-14:
            a = a_new
            step *= 1.03                # gently increase step when improving
        else:
            step *= 0.7                 # otherwise shrink step

        if (it + 1) % 500 == 0:
            cur_score = 2.0 * n * new_mx
            print(f"[polish {it + 1:4d}] score = {cur_score:.12f}")

    # ------------------------------------------------------------------
    # Clamp to the limits enforced by `evaluate_sequence` and return.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score = {final_score:.12f}, length = {len(final_seq)}")
    return final_seq
