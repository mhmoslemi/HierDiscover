from scipy.optimize import linprog

def solve_lp_for_g(f, M):
    n = len(f)
    # Build A_ub: shape (2n-1, n)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    # Target: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    # ub: A @ g <= M
    b_ub = np.full(rows, M)
    # Solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': maybe?})
    if not res.success:
        # fallback to uniform
        return None
    g = res.x
    # could be all zeros if M too small; but then sum small.

"""
Propose a candidate step‑function (non‑negative vector) that minimises

        2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm combines

1. A brief deterministic sweep over a few simple symmetric families
   (U‑shapes, shifted U‑shapes, exponential) to obtain a decent seed.

2. Mirror‑descent on the probability simplex using the exponentiated‑gradient
   update with a simulated‑annealing acceptance rule (the same idea as the
   previous “stochastic refinement” but with a more aggressive schedule).

3. A periodic linear‑programming refinement: for the current iterate `f`
   we solve the LP

        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f * g)_k ≤ M   for all k,
   where M = max_k (f * f)_k.
   The optimal `g` has the largest possible ℓ₁‑norm while respecting the
   convolution bound.  Mixing `f` and `g` with a small step size can strictly
   reduce the maximal convolution value.  The LP is solved with
   ``scipy.optimize.linprog`` (high‑s) on a dense matrix; the cost is modest
   for n≈2500, so we perform this refinement only occasionally.

The whole search respects a configurable time budget (default 1000 s) and
returns the best vector found (as a plain Python list with sum≈1).  The code
uses only NumPy and SciPy and contains no closures or lambdas, as required.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: n + m - 1]


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a."""
    return conv_fft(a, a)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution entry
# ----------------------------------------------------------------------
def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] with respect to a.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        i_range = np.arange(i0, i1 + 1, dtype=np.int64)
        j_range = k - i_range
        grad[i_range] += 2.0 * a[j_range]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) step on the simplex
# ----------------------------------------------------------------------
def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Perform the update a' ∝ a * exp(-η * grad) and renormalise to sum = 1.
    """
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new


# ----------------------------------------------------------------------
# Small parametric families (all symmetric, already normalised)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming helper (maximise ℓ₁‑norm of g under convolution bound)
# ----------------------------------------------------------------------
def build_A_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ g) = f * g (full convolution).
    Row k contains f_i for i+j = k, i.e. the j‑th column is f shifted by j.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f * g)_k ≤ M  (for all k).
    Returns a normalised vector g (Σ g = 1) or None on failure.
    """
    n = f.shape[0]
    A = build_A_matrix(f)                # shape (2n‑1, n)
    c = -np.ones(n)                      # maximise Σg_i  → minimise -Σg_i
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    # use the HiGHS solver (fast, reliable)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s < 1e-12:
        return None
    g = g / s
    return g

# ----------------------------------------------------------------------

                    sc = float(2.0 * n * conv_h.max())
                    if sc < best_local_score:
                        best_local_score = sc
                        best_h = h
                        best_h_conv = conv_h

                if best_h is not None and best_local_score < current_score:
                    # accept the mixed vector
                    current = best_h
                    current_conv = best_h_conv
                    current_score = best_local_score
                    if best_local_score < best_score:
                        best_score = best_local_score
                        best_seq = best_h.copy()
                        last_improve = iteration
                        print(f"Iter {iteration:7d} (LP): new best score = {best_score:.6f}")

        # ---- 4f.  Schedule updates
        temperature *= temp_decay
        eta *= eta_decay

        # ---- 4g.  Stagnation handling (re‑heat)
        if iteration - last_improve > 120000:
            bump = np.random.rand(n) * 1e-5
            current = np.maximum(current + bump, 0.0)
            current /= current.sum()
            current_conv = full_convolution_fft(current)
            current_score = float(2.0 * n * current_conv.max())
            temperature = 1.0
            eta = 0.9
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation (LP calls={lp_calls})")

    # ------------------------------------------------------------------
    # 5️⃣  Return the best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
