We can implement projection onto simplex using known algorithm O(n log n) (sort).

Pseudo:

def simplex_projection(v):
    # Project onto simplex {x >=0, sum x =1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def projected_subgradient_descent(a_init, max_iter, deadline, init_step=0.1, decay=0.99):
    a = a_init.copy()
    best_a = a.copy()
    best_val = approx_obj(best_a)
    step = init_step
    n = len(a)
    while time.time() < deadline and max_iter>0:
        # compute conv
        conv = _full_convolution(a)
        M = conv.max()
        k_star = conv.argmax()
        # compute gradient: g_i = a_{k_star - i}
        # to avoid out-of-bounds, use zero for missing indices.
        idx = np.arange(n)
        # compute reversed mapping:
        # need a_{k*-i} = shift a by k*
        # Create grad = zeros(n)
        grad = np.zeros(n)
        # for i where k*-i in [0,n-1] => i in [k*- (n-1), k*]
        low = max(0, k_star - (n-1))
        high = min(n-1, k_star)
        # For i in low..high:
        # a_{k*-i} is a[k_star - i]
        grad[low:high+1] = a[k_star - np.arange(low, high+1)]
        # gradient of O = 4 * n * grad
        grad = 4.0 * n * grad
        # step
        a = a - step * grad
        # project onto simplex
        a = _simplex_projection(a)
        # compute new obj
        cur_val = approx_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgrad] iter {max_iter}, val {best_val:.6f}, step {step:.6f}")
        # step size schedule
        step *= decay
        max_iter -= 1
    return best_a, best_val

def _simplex_projection(v):
    # ensure sum of v = 1 after projection onto nonnegative orthant
    # algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

assistantfinal**Solution Overview**

The evaluation metric is  

\[
E(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where \((a*a)_k\) is the full self‑convolution of the step heights.  
The objective is *scale‑invariant*, therefore we can (and do) keep the
vector on the probability simplex \(\sum_i a_i=1\).  
On the simplex  

\[
E(a)=2n\;\max_k (a*a)_k .
\]

Consequences

* The only term that matters is the maximal convolution entry `M`.
* The gradient of `M` (when the active index `k*` is unique) is
  \(\partial M/\partial a_j = 2 a_{k*-j}\) (outside the support the
  term is zero).
* Hence a **projected sub‑gradient step** is  

  \[
  a \leftarrow \Pi_{\Delta}\Bigl(a -
            \eta\;4n\;g\Bigr),\qquad
  g_j = a_{k*-j},
  \]

  where \(\Pi_{\Delta}\) projects onto the simplex.

The algorithm consists of three phases  

1. **Fast parameter sweep** – try simple analytic families
   (Beta‑distributions) on the final size `n` and keep the best
   candidate.
2. **Projected sub‑gradient descent** – starting from the best
   candidate, repeatedly reduce the active convolution peak.
3. **Lightweight hill‑climbing** – random mass transfers for a final
   polish.

All heavy work (convolution, gradient) uses an FFT based
implementation, which is *O(n log n)* and easily fits into the 1000 s
budget for `n≈4000`.

The code below follows this plan and returns the best sequence found
before the time budget expires.  It prints intermediate scores so you
can see the progress.

"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence.

Algorithm
---------
1. Scan a small analytic family (Beta(p,p) densities) on the final
   problem size n, keep the best sequence (approximate objective).
2. Starting from the best sequence, run a projected sub‑gradient
   descent on the exact objective 2·n·max(conv) (sum is kept equal to 1
   by a simplex projection).  The gradient of the max‑convolution is
   g_j = a_{k*‑j} (k* is the index where the convolution attains its
   maximum).  A diminishing step size is used.
3. Finish with a cheap random hill‑climbing (mass transfers) that
   respects the sum‑constraint; acceptance is based on the exact
   evaluate_sequence value.
4. The best exact value seen is returned.

All convolutions are performed with FFT (O(n log n)) for speed.
The whole procedure respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negative values caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (onto {a_i ≥ 0, Σ a_i = 1})
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Approximate objective (fast, uses the same formula as evaluate_sequence)
# ----------------------------------------------------------------------
def _approx_obj(a: np.ndarray) -> float:
    """Return the value of the evaluation function (no clamping)."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a)
    M = conv.max()
    return 2.0 * n * M / (s * s)


# ----------------------------------------------------------------------
# Sub‑gradient step (projected onto simplex)
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Perform one projected sub‑gradient step for the objective
    2·n·max(conv) under the constraint Σa=1, a≥0.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k_star = int(conv.argmax())               # index of maximal entry
    # Build g_j = a_{k_star - j} (zero if out of range)
    low = max(0, k_star - (n - 1))
    high = min(n - 1, k_star)
    grad = np.zeros_like(a)
    # i runs from low … high, then k_star‑i is in [0,n‑1]
    idx = np.arange(low, high + 1)
    grad[idx] = a[k_star - idx]
    # Gradient of the objective: 4·n·g
    grad *= 4.0 * n
    # Gradient descent step
    a_new = a - step * grad
    # Project back onto the simplex
    a_proj = _simplex_projection(a_new)
    return a_proj


# ----------------------------------------------------------------------
# Hill‑climbing refinement (random mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """Small random perturbations that keep Σa constant."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Transfer a random fraction of the mass at i to j
        delta = cur_a[i] * 0.05 * rng.random()
        if delta <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # Quick approximate check first
        approx_val = _approx_obj(new_a)
        if approx_val < cur_val - 1e-8:
            # Verify with the official function (more expensive)
            try:
                exact_val = evaluate_sequence(new_a.tolist())
            except Exception:
                continue
            if exact_val < cur_val:
                cur_a = new_a
                cur_val = exact_val
                print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------

    uniform = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Mid‑point grid (avoids the exact 0 and 1 which would be infinite for p<1)
