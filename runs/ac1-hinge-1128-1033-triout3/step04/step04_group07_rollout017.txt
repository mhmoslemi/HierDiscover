   optionally mix them with the uniform distribution.
   Each candidate is projected onto the probability simplex
   (sum = 1) and its exact objective (using FFT convolution) is
   evaluated.  The best shape from this sweep is kept.

2. **Sub‑gradient descent** – starting from the best candidate we
   iteratively minimise the exact maximal self‑convolution.
   The active set of indices attaining the maximum is used to
   build a sub‑gradient; after each step the vector is projected
   back onto the simplex.  A simple geometric decay of the step size
   is employed and the best point ever seen is stored.

3. **Tiny random‑swap hill‑climber** – a lightweight stochastic
   refinement that moves a small amount of mass between two
   coordinates whenever the objective improves.

All operations respect the non‑negativity and normalisation
constraints, and the resulting list is clipped to the
[0, 1000] range required by `evaluate_sequence`.  The whole
search is bounded by the supplied time budget (default 1000 s).
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, exact for the purpose of the objective)
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of p using an FFT of size m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _max_convolution(p: np.ndarray, m_fft: int) -> float:
    """Maximum entry of the self‑convolution of p."""
    return float(_conv_self(p, m_fft).max())


def _objective(p: np.ndarray, n: int, m_fft: int) -> float:
    """Exact value of the evaluation objective for a probability vector."""
    return 2.0 * n * _max_convolution(p, m_fft)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a sequence of non‑negative heights with a low evaluation score."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a problem size.  Using a larger n reduces discretisation
    # error and gives a better asymptotic bound.
    # --------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 0

    # target length – a few thousand points are still cheap for the final
    # convolution (the evaluate_sequence function uses a simple O(n²) loop,
    # but for n ≲ 4000 the cost is well below the 1000 s budget).
    target_n = 3000
    n = max(base_n, target_n)

    # FFT size – next power‑of‑two that holds a full convolution.
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Build a pool of promising initial vectors.
    # --------------------------------------------------------------
    candidates = []

    # Uniform distribution (baseline)
    uniform = np.full(n, 1.0 / n)
    candidates.append(("uniform", uniform))

    # Arcsine shape (α = 0.5) – the classic construction.
    i = np.arange(n, dtype=float)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-15)
    arcsine /= arcsine.sum()
    candidates.append(("arcsine", arcsine))

    # Beta‑shaped families for many α values, plus a few mixes with uniform.
    alphas = np.concatenate(
        [
            np.linspace(0.2, 0.9, 15),      # below, at and a bit above 0.5
            np.linspace(0.95, 1.5, 6),     # more uniform‑like shapes
        ]
    )
    mix_fractions = (0.0, 0.25, 0.5, 0.75)  # λ in convex combo with uniform

    t = (np.arange(n, dtype=float) + 0.5) / n            # points in (0,1)

    for a in alphas:
        # beta density (symmetrised) : (t·(1‑t))^{α‑1}
        shape = np.power(t * (1.0 - t), a - 1.0)
        # Numerical safeguard – the extreme ends can be huge for α<0.5.
        shape = np.where(np.isfinite(shape), shape, 0.0)
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        candidates.append((f"beta_{a:.3f}", shape))
        for lam in mix_fractions:
            mixed = lam * shape + (1.0 - lam) * uniform
            # The mixture already sums to one.
            candidates.append((f"beta_{a:.3f}_lam{lam:.2f}", mixed))

    # A few random Dirichlet draws for extra diversity.
    for k in range(5):
        rnd = rng.random(n)
        rnd /= rnd.sum()
        candidates.append((f"rand_{k}", rnd))

    # Historic construction (if supplied) – simply normalised.
    if base_seq is not None:
        seq = np.clip(base_seq, 0.0, 1000.0)
        s = seq.sum()
        if s > 0:
            hist = seq / s
            # pad / truncate to the chosen length
            if len(hist) < n:
                hist = np.concatenate([hist, np.zeros(n - len(hist))])
            elif len(hist) > n:
                hist = hist[:n]
            candidates.append(("historic", hist))

    # --------------------------------------------------------------
    # Quick evaluation of all candidates.
    # --------------------------------------------------------------
    best_score = float("inf")
    best_vec = None
    best_name = None

    deadline = time.time() + budget_s - 0.5   # safety margin

    for name, vec in candidates:
        if time.time() >= deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0:
            continue
        vec = vec / vec.sum()
        score = _objective(vec, n, m_fft)
        if score < best_score:
            best_score = score
            best_vec = vec.copy()
            best_name = name
            print(f"Initial candidate {name:20s} → score {best_score:.6f}")
        if best_score <= 1.503:
            break   # we already reached the target

    # --------------------------------------------------------------
    # Sub‑gradient descent on the *exact* max entry (active‑set).
    # --------------------------------------------------------------
    if best_vec is not None and best_score > 1.503:
        p = best_vec.copy()
        step = 0.5
        decay = 0.9995
        min_step = 1e-8
        max_iter = 120_000

        for it in range(max_iter):
            if time.time() >= deadline:
                break
            # exact convolution and maximum
            conv = _conv_self(p, m_fft)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_vec = p.copy()
                print(f"Sub‑grad iter {it:05d} – new best {best_score:.6f}")
                if best_score <= 1.503:
                    break

            # active set (indices attaining the max up to a tiny tolerance)
            tol = 1e-12
            active = np.where(conv >= max_val - tol)[0]

            # sub‑gradient of max(conv) w.r.t. p
            g = np.zeros_like(p)
            for idx in active:
                shift = idx - (n - 1)          # actual lag: negative = left shift
                if shift >= 0:
                    # overlap region: p[k] with p[k+shift]
                    g[: n - shift] += 2.0 * p[shift:]
                else:
                    sft = -shift
                    g[sft:] += 2.0 * p[: n - sft]
            g /= float(len(active))

            # projected update
            p = p - step * g
            p = _simplex_projection(p)

            step = max(step * decay, min_step)

        # --------------------------------------------------------------
        # Tiny random‑swap hill‑climber (fine polishing)
        # --------------------------------------------------------------
        if best_vec is not None:
            p_swap = best_vec.copy()
            cur_score = best_score
            for _ in range(200_000):
                if time.time() >= deadline:
                    break
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                eps_move = 1e-4 * rng.random()      # tiny transfer
                if p_swap[i] <= eps_move:
                    continue
                new_p = p_swap.copy()
                new_p[i] -= eps_move
                new_p[j] += eps_move
                new_score = _objective(new_p, n, m_fft)
                if new_score < cur_score - 1e-12:
                    p_swap = new_p
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_vec = p_swap.copy()
                        print(f"Swap improvement → {best_score:.6f}")
                    if best_score <= 1.503:
                        break

    # ------------------------------------------------------------------
    # Fallback – should never happen.
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = uniform
        best_score = _objective(best_vec, n, m_fft)

    print(f"\nFinal best ({best_name}) score: {best_score:.6f} (length {n})")

    # Clip to the bounds enforced by evaluate_sequence (0 ≤ x ≤ 1000)
    # The shape is already normalised (sum = 1) so the scaling does not
    # affect the objective; clipping is only a safety measure.
    return np.clip(best_vec, 0.0, 1000.0).tolist()
```
