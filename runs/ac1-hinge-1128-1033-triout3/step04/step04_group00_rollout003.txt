"""
Propose a step‑function (non‑negative sequence) that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using an Adam‑based projected gradient descent with a soft‑max approximation of
the max‑operator, occasional random restarts and a final sub‑gradient polishing.
The algorithm works directly on the probability simplex (sum = 1, a_i ≥ 0)
so the evaluation function reduces to 2·n·max(conv).  A previous best
solution (available as ``height_sequence_1``) is used as a warm start;
if it is missing a power‑law shape is generated.  The optimisation runs for
the full time budget (≤ 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x | x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using a real FFT (returns length len(a)+len(b)-1)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Generate a simple power‑law decreasing shape a_i ∝ (i+1)^(-p)
# ----------------------------------------------------------------------
def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a

    progress messages and finally returns the best sequence as a list of

    """

    # ------------------------------------------------------------------

        try:

            n = base.shape[0]
            a0 = np.maximum(0.0, base)
            if a0.sum() == 0.0:
                a0 = rng.random(n)
            a0 = a0 / a0.sum()
        except Exception:
            n = 5000
            a0 = None
    else:
        n = 5000
        a0 = None

    if a0 is None:
        # random power‑law shape – exponent drawn uniformly in a reasonable range
        p = rng.uniform(0.35, 0.85)
        a0 = _powerlaw_shape(n, p)
        # add a hint of noise and project back onto the simplex
        a0 = a0 + 0.02 * rng.random(n)
        a0 = _project_simplex(a0)

    # ------------------------------------------------------------------
    # Tracking the best solution seen so far
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a0, a0)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv          # because sum(a0) == 1
    best_seq = a0.copy()
    print(f"[init] n={n}, initial_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    a = a0.copy()                         # current candidate
    m = np.zeros_like(a)                  # 1st moment (Adam)
    v = np.zeros_like(a)                  # 2nd moment (Adam)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    lr = 0.30                              # base learning rate for Adam
    eps_adam = 1e-8

    # temperature for the soft‑max approximation of max(conv)
    eps = max(max_conv * 0.5, 1e-12)
    eps_min = max(max_conv * 1e-5, 1e-12)
    eps_decay = 0.9995

    stagn = 0          # iterations without improvement
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (projected Adam + occasional restart)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ---- convolution and current objective value ----
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # ---- keep the best sequence seen so far ----
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it:07d}] new_best={best_val:.12f} eps={eps:.2e}")
        else:
            stagn += 1

        # ---- temperature schedule (soft‑max) ----
        eps = max(eps * eps_decay, eps_min)

        # ---- soft‑max weights for the current convolution ----
        w = np.exp((conv - max_conv) / eps)      # all entries ≤ 1
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- gradient of the weighted sum (cross‑correlation) ----
        rev_a = a[::-1]
        g_full = _full_convolution_fft(rev_a, w)    # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]          # slice back to size n

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = _project_simplex(a_new)

        # ---- occasional random restart if we stagnate for a while ----
        if stagn > 5000:
            # blend the current best with a fresh random power‑law shape
            p = rng.uniform(0.3, 1.0)
            fresh = _powerlaw_shape(n, p)
            a = 0.6 * best_seq + 0.4 * fresh
            a = _project_simplex(a + 0.01 * rng.random(n))

            # reset optimizer state
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            stagn = 0
            # reset temperature to a relatively large value
            eps = max(max_conv * 0.5, 1e-12)
            print(f"[restart] iteration {it}")

        # ---- tiny safety re‑normalisation (should be unnecessary) ----
        if it % 10000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Final polishing: pure sub‑gradient steps on the *max* convolution entry
    # ------------------------------------------------------------------
    a = best_seq.copy()
    step_sz = 5e-4
    polishing_start = time.time()
    for sub_it in range(4000):
        # stop early if we are out of time
        if time.time() > deadline - 1.0:
            break
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        # find the first index attaining the maximum (any will do)
        idx = int(np.flatnonzero(conv >= mx - 1e-15)[0])

        # sub‑gradient of the *single* maximal entry:
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]

        a = _project_simplex(a - step_sz * g)

        # gradually shrink the step size
        step_sz *= 0.9999

    # ------------------------------------------------------------------
    # Convert to plain Python list, clip to the limits required by evaluate_sequence
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
