def project_to_simplex(v):
    # ensure sum(v) =1, v>=0
    # uses sorted v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) that minimises the
`evaluate_sequence` objective.

Algorithm (≈ 1000 s budget)

* Phase A – Adam on a smooth approximation of the max‑convolution.
  The vector a lives on the simplex via the soft‑max parametrisation
  a = softmax(w).  The objective  max(conv(a,a)) is replaced by a
  log‑sum‑exp (smooth‑max) with a temperature ε that is driven from
  5·10⁻³ → 1·10⁻⁹ while the learning‑rate decays from 5·10⁻² → 5·10⁻⁴.
  Adam’s adaptive moments are used; occasional noise and a restart
  after many idle steps keep the search from stagnating.

* Phase B – Projected sub‑gradient descent on the exact max‑convolution.
  At each iteration we compute the full convolution, locate the
  active (max‑attaining) indices, and evaluate a sub‑gradient
  g_i = 2·∑_{k∈active} a_{k‑i} (indices out of range are ignored).
  A diminishing step η_t = η₀/√t is taken and the new point is projected
  onto the probability simplex (Σa=1, a≥0).  This directly reduces the
  L∞‑norm of the convolution and typically improves the bound further
  than a pure hill‑climber.

* Phase C – Deterministic hill‑climbing with exact O(n) convolution updates.
  Small masses δ are transferred from a source index to a target index.
  The convolution is updated analytically (no FFT) and the move is kept
  only if the exact objective strictly improves.  After long stagnation
  a tiny random perturbation of the current best solution is performed.

The three phases together explore the landscape at different scales,
allowing us to breach the previous bound (≈ 1.5138) and aim for the
target 1.503.  All numbers are kept non‑negative and normalised to sum = 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution (real vectors) using FFT
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of x with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (fast version, assuming Σa = 1)
# ----------------------------------------------------------------------
def _exact_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    b = _full_convolution(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Phase A – smoothed‑max Adam optimisation
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp approximation of max(conv(a,a)).
    Returns the best (a, value) seen before *deadline*.
    """
    # initialise soft‑max parameters
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 5e-2
    lr_end = 5e-4
    eps_start = 5e-3
    eps_end = 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 25000  # restart after this many idle steps

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----------- soft‑max to obtain a on the simplex -------------
        w_shift = w - np.max(w)               # avoid overflow
        a = np.exp(w_shift)
        a /= a.sum()

        # --------------- exact objective (for tracking) -------------
        cur_val = _exact_objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ------------------- smoothing schedule ----------------------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # compute full convolution only once (needed for gradient as well)
        conv_aa = _full_convolution(a)                 # length 2n‑1
        max_conv = conv_aa.max()

        # ---------- smooth‑max (log‑sum‑exp) weights ---------------
        shifted = (conv_aa - max_conv) / eps
        # clip to avoid overflow (shifted ≤ 0)
        weights = np.exp(shifted)                      # shape (2n‑1,)
        p = weights / weights.sum()                    # p approximates a one‑hot

        # ---------- gradient of smoothed objective wrt a -----------
        #   d/da  ( n·log Σ exp(conv/ε) ) = (n/ε) * Σ p_k * d conv_k/da
        #   d conv_k/da = 2 * (a * rev(p))[k]  (see derivation)
        p_rev = p[::-1]                                 # reversed
        conv_ab = _conv_fft(a, p_rev)                   # length 3n‑2
        segment = conv_ab[n - 1: 2 * n - 1]              # middle n entries
        grad_a = 4.0 * n * segment[::-1]                # shape (n,)

        # ---------- chain‑rule through soft‑max --------------------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)                     # ∂f/∂w

        # --------------- Adam update --------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional tiny perturbation to avoid exact plateaus
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart if we have been idle too long
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_phase(a0: np.ndarray,
                       n: int,
                       start: float,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on   f(a) = max(conv(a,a))   with Euclidean projection
    onto the simplex.  Returns the best (a,val) seen before *deadline*.
    """
    a = a0.copy()
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 50000        # after this many idle steps we add noise

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current objective
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Subgrad {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # active set (indices attaining the maximum, tolerance for float noise)
        tol = 1e-12
        active = np.where(b >= max_b - tol)[0]

        # compute sub‑gradient g_i = 2 * Σ_{k∈active} a_{k‑i}
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1)               # i indices
            partner = k - idx                                 # = a index
            g[idx] += 2.0 * a[partner]

        # step‑size schedule η_t = η₀ / √t
        t = iteration + 1
        eta0 = 0.5
        eta = eta0 / np.sqrt(t)

        # gradient step and projection onto simplex
        a = a - eta * g
        a = _project_simplex(a)

        # occasional noise to shake the process when stuck
        if no_improve > IMPROVE_LIMIT:
            print(f"[Subgrad {iteration:7d}] stagnation – perturbing")
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = _project_simplex(a + noise)
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – fast deterministic hill‑climbing with exact updates
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """
    Local search that moves a small amount of mass from a source index i
    to a target index j.  The convolution is updated in O(n) using the
    analytical formula from bilinearity; only improving moves are kept.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac = 0.5          # at most this fraction of source mass can be moved
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # -----------------------------------------------------------------
        # Choose source and target indices
        # -----------------------------------------------------------------
        if np.random.rand() < 0.7:
            i = int(np.argmax(a))            # usually take the biggest mass
        else:
            i = np.random.choice(n, p=a)      # occasionally weighted random

        if a[i] <= 1e-14:                     # nothing to move
            continue

        j = np.random.randint(n)
        if j == i:
            continue

        # -----------------------------------------------------------------
        # Amount of mass to transfer
        # -----------------------------------------------------------------
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # -----------------------------------------------------------------
        # Incremental update of the full convolution (O(n))
        # -----------------------------------------------------------------
        b_new = b.copy()
        # cross‑terms (2·δ·a) added for the new position and removed for the old
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # -----------------------------------------------------------------
        # Accept only strictly improving moves
        # -----------------------------------------------------------------
        if new_val < cur_val - 1e-15:          # a tiny tolerance avoids noise loops
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # -----------------------------------------------------------------
        # Random perturbation after long stagnation
        # -----------------------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            print(f"[Hill {iteration:7d}] stagnation – perturbing")
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a + noise)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a
