def softmax_weights(conv, temperature):
    # conv: array length 2n-1.
    max_val = conv.max()
    # compute exp(...), numeric stable:
    delta = conv - max_val
    exps = np.exp(delta / temperature)
    sum_exps = exps.sum()
    if sum_exps == 0 or not np.isfinite(sum_exps):
        return np.full_like(conv, 1.0/conv.size)
    return exps / sum_exps

The function below searches a non‑negative step‑function (a vector `a`) that minimizes  

\[
E(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum a\bigr)^2}\;.
\]

Because `E` is scale‑invariant we keep the vector on the probability simplex
(`∑a = 1 , a_i ≥ 0`).  
The algorithm consists of three phases

1. **Initialisation** – take a warm‑start from the previously found best
   sequence (if it exists) and up‑sample it, otherwise start from a mixture of
   smooth shapes (sine, semi‑circle, Gaussian, triangle).  
2. **Main optimisation** – minimise a smooth approximation of the `max`
   using the **Adam** optimiser.  
   * At each iteration the self‑convolution `c = a*a` is computed by FFT.  
   * A soft‑max over `c` (temperature `τ`) yields weights `w` that concentrate
     the gradient on the currently large entries.  
   * The gradient of `∑ w_i c_i` equals `2·(rev(a) * w)` and is also obtained
     by FFT.  
   * Adam’s adaptive step is taken and the iterate is projected back onto the
     simplex.  
   * Learning‑rate and temperature decay slowly; after a long period without
     improvement a small random perturbation is added to escape plateaus.
3. **Polishing** – a short sub‑gradient descent on the *exact* maximum entry
   (no soft‑max) refines the best candidate found.

All operations are vectorised; the dominant work is a few FFTs per
iteration, which easily fits the 1000 s budget for `n ≈ 10 000`.  
The final sequence is clamped to `[0,1000]` as required by
`evaluate_sequence`.

"""
Propose a non‑negative step function that (approximately) minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using an Adam optimiser on a soft‑max approximation of the max‑convolution.
The routine starts from a warm‑start (if available), otherwise builds a smooth
mixed shape, runs many FFT‑based iterations, adds occasional random
perturbations to escape plateaus and finishes with a pure max‑entry
sub‑gradient polishing.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x | x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Duchi et al. (2008) projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b (real FFT)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

        a = a / a.sum()

    # --------------------------------------------------------------
    # Optimiser state (Adam)
    # --------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    lr = 0.05                     # initial learning rate
    lr_min = 1e-5
    lr_decay = 0.9998

    # --------------------------------------------------------------
    # Temperature for the soft‑max (smooth max)
    # --------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    temp = max(max_conv * 0.5, 1e-12)          # start relatively high
    temp_min = max(max_conv * 1e-6, 1e-12)
    temp_decay = 0.9999

    # --------------------------------------------------------------
    # Tracking the best solution
    # --------------------------------------------------------------
    best_score = 2.0 * n * max_conv   # because sum(a) == 1
    best_seq = a.copy()
    best_iter = 0
    stagn = 0
    it = 0

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- current convolution & score -----
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # ----- keep the best -----
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            best_iter = it
            stagn = 0
            if it % 2000 == 0:
                print(f"[iter {it:6d}] new_best={best_score:.12f}  lr={lr:.2e}  τ={temp:.2e}")
        else:
            stagn += 1

        # ----- soft‑max weights (smooth max) -----
        #   w_i ∝ exp((conv_i - max_conv) / τ)
        delta = conv - max_conv
        w = np.exp(delta / temp)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of weighted sum: g = 2 * (rev(a) * w) -----
        rev_a = a[::-1]
        conv_rev = _full_convolution_fft(rev_a, w)      # length = 3n‑2
        g = 2.0 * conv_rev[n - 1 : n - 1 + n]            # slice to size n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a = _project_simplex(a_new)      # keep on the simplex

        # ----- schedule -----
        lr = max(lr_min, lr * lr_decay)
        temp = max(temp_min, temp * temp_decay)

        # ----- occasional random perturbation to escape plateaus -----
        if stagn > 5000:
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            # add mild Gaussian noise and re‑project
            noise = rng.normal(scale=0.01, size=n)
            a = _project_simplex(a + noise)
            stagn = 0
            lr = max(lr_min, lr * 2.0)
            print(f"[perturb] iter {it:6d}  lr boosted to {lr:.2e}")

        # ----- occasional renormalisation (numerical safety) -----
        if it % 5000 == 0:
            a = a / a.sum()

    # --------------------------------------------------------------
    # Final polishing: pure max‑entry sub‑gradient steps
    # --------------------------------------------------------------
    a = best_seq.copy()
    for _ in range(3000):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        idxs = np.flatnonzero(conv >= mx - 1e-15)
        idx = int(idxs[0])                     # take the first maximal index

        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        # gradient of the *single* maximal entry (2*a[j] except the diagonal)
        g[i_range] = 2.0 * a[j]
        # correct the diagonal where i==j (appears only once)
        if idx % 2 == 0:
            centre = idx // 2
            if 0 <= centre < n:
                g[centre] = a[centre]          # derivative = a[centre] (not doubled)

        a = _project_simplex(a - 5e-6 * g)

    # --------------------------------------------------------------
    # Convert to the format expected by `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]
    final_score = evaluate_sequence(final_seq)

    print(f"[result] final_score={final_score:.12f}  best_score={best_score:.12f} "
          f"length={len(final_seq)}  iters={it}  best_iter={best_iter}")

    return final_seq
