"""
Mirror‑descent on a soft‑max (and final hard) approximation of the
∞‑norm of the auto‑convolution.  Several diverse initialisations
(uniform, arcsine‑like, Dirichlet‑draw, previous best) are tried in
sequence; each run uses an Adam‑style exponentiated‑gradient update
on the probability simplex.  After the main phase a short hard‑max
refinement is performed.  The best sequence encountered before the
time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution using a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative round‑off values
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑norm objective (via a single convolution)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Given probabilities a (length n) and soft‑max weights w (length 2n‑1),
    return the gradient of  Σ_k w_k·(a*a)_k with respect to a.
    The formula is  g_i = 2·∑_k w_k·a_{k-i}.
    """
    n = a.shape[0]
    # reverse the weight vector (full length 2n‑1)
    u = w[::-1]
    # full convolution u * a  → length 3n‑2
    conv = _full_convolution_fft(u, a)
    # indices that correspond to the gradient entries
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Soft‑max (stable) of the auto‑convolution entries
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Return a probability vector w proportional to exp(conv/τ).
    If τ is non‑positive or numerically unstable, use a hard one‑hot at
    the maximum entry.
    """
    if tau <= 0.0 or not np.isfinite(tau):
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w

    M = conv.max()
    # shift for numerical stability
    exp_args = (conv - M) / tau
    np.clip(exp_args, -80, 80, out=exp_args)
    exp_vals = np.exp(exp_args)
    sum_exp = exp_vals.sum()
    if sum_exp == 0.0 or not np.isfinite(sum_exp):
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exp_vals / sum_exp


# ----------------------------------------------------------------------
# Core optimisation for a single initial vector
# ----------------------------------------------------------------------
def _optimise_one_start(a_init: np.ndarray,
                        deadline: float,
                        seed: int = 0) -> tuple[np.ndarray, float]:
    """
    Run an Adam‑style exponentiated‑gradient descent on the soft‑max
    surrogate of the max‑convolution.  After the main loop a short
    hard‑max refinement is performed.  Returns the best vector and
    its objective value (2·n·max(b)).
    """
    np.random.seed(seed)
    a = a_init.copy()
    n = a.shape[0]

    # bookkeeping of the best seen
    conv = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv.max())
    best_a = a.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # hyper‑parameters (linear schedules)
    eta_start = 0.6      # early multiplicative step size
    eta_end = 5e-5       # final step size
    tau_factor_start = 0.25
    tau_factor_end = 1e-6
    jitter_prob = 0.02

    iteration = 0
    start_time = time.time()

    while True:
        now = time.time()
        if now >= deadline:
            break

        # --- objective value (full conv) ---------------------------------
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[seed {seed:02d} it {iteration:06d}] new best = {best_val:.6f}")

        # progress ∈ [0,1] for schedules
        progress = (now - start_time) / (deadline - start_time)

        # --- temperature -------------------------------------------------
        tau = ((1.0 - progress) * tau_factor_start +
               progress * tau_factor_end) * M

        # --- soft‑max weights -------------------------------------------
        w = _softmax_weights(conv, tau)

        # --- gradient ----------------------------------------------------
        grad = _gradient_from_weights(a, w)

        # --- Adam update (in log‑space) ---------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        # learning‑rate schedule
        eta = (1.0 - progress) * eta_start + progress * eta_end

        step = -eta * m_hat / (np.sqrt(v_hat) + eps)
        np.clip(step, -50, 50, out=step)
        a = a * np.exp(step)

        # --- occasional random jitter (helps escape plateaus) ----------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move a random fraction (up to 50 %) of src's mass to dst
            delta = np.random.rand() * a[src] * 0.5
            a[src] -= delta
            a[dst] += delta

        # --- projection onto the probability simplex --------------------
        a = np.maximum(a, eps)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        iteration += 1

        # occasional sanity check (cheap, no output)
        if iteration % 3000 == 0:
            _ = _full_convolution_fft(a, a)

    # ------------------------------------------------------------------
    # Hard‑max refinement (τ → 0) – few Newton‑style steps
    # ------------------------------------------------------------------
    # reuse Adam state (re‑initialise for a clean fine‑tune)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    hard_steps = 200
    for it in range(hard_steps):
        now = time.time()
        if now >= deadline:
            break
        conv = _full_convolution_fft(a, a)
        k = int(np.argmax(conv))
        w_hard = np.zeros_like(conv)
        w_hard[k] = 1.0
        grad = _gradient_from_weights(a, w_hard)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        eta_refine = 5e-5
        step = -eta_refine * m_hat / (np.sqrt(v_hat) + eps)
        np.clip(step, -50, 50, out=step)
        a = a * np.exp(step)

        a = np.maximum(a, eps)
        a = a / a.sum()

    # Final evaluation of the refined candidate
    conv = _full_convolution_fft(a, a)
    final_val = float(2.0 * n * conv.max())
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
        print(f"[seed {seed:02d} hard‑refine] final best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Performs a multi‑start Adam‑mirrored optimisation of the
    max‑auto‑convolution objective.  The best sequence found before
    the time budget expires is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin before the hard deadline
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Length of the sequence – a balance between expressiveness and speed
    # --------------------------------------------------------------
    n = 1800                     # can be tuned; 1800 is a good compromise

    # --------------------------------------------------------------
    # Build a collection of diverse initial vectors
    # --------------------------------------------------------------
    inits = []

    # 1. Uniform distribution
    inits.append(np.full(n, 1.0 / n))

    # 2. Arcsine‑like distribution (heavy at the ends)
    idx = np.arange(n, dtype=float)
    x = (idx + 0.5) / n                     # centre of each bin
    # avoid division by zero at the boundaries
    eps = 1e-12
    arcs = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    arcs = arcs / arcs.sum()
    inits.append(arcs)

    # 3. Sparse Dirichlet draw (α < 1 gives many near‑zeros)
    inits.append(np.random.dirichlet(alpha=np.full(n, 0.3)))

    # 4. Previous best if it is available in the global scope
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if prev.shape[0] < n:
            prev = np.pad(prev, (0, n - prev.shape[0]), constant_values=0.0)
        elif prev.shape[0] > n:
            prev = prev[:n]
        prev = np.maximum(prev, 0.0)
        prev = prev / prev.sum()
        inits.append(prev)
    except Exception:
        pass

    # 5. A random perturbation of the best uniform (helps break symmetry)
    pert = np.copy(inits[0])
    # add a small Gaussian noise and renormalise
    pert += np.random.normal(scale=0.01, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum()
    inits.append(pert)

    # --------------------------------------------------------------
    # Run optimisation on each start, sharing the overall time budget
    # --------------------------------------------------------------
    best_overall_a = None
    best_overall_val = float("inf")
    start_counter = 0

    while time.time() < deadline and start_counter < len(inits):
        # give each start a proportional slice of the remaining time
        remaining = deadline - time.time()
        # leave a little room for the final call to evaluate_sequence
        sub_deadline = time.time() + remaining * 0.8
        a0 = inits[start_counter]
        a_opt, val_opt = _optimise_one_start(a0, sub_deadline, seed + start_counter)
        if val_opt < best_overall_val:
            best_overall_val = val_opt
            best_overall_a = a_opt.copy()
            print(f"[global] new overall best = {best_overall_val:.6f} (seed {seed + start_counter})")
        start_counter += 1

    # In the unlikely case nothing was found (should not happen), fall back to uniform
    if best_overall_a is None:
        best_overall_a = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Final reporting (optional – the outer framework will also call it)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_overall_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_overall_a.tolist()
