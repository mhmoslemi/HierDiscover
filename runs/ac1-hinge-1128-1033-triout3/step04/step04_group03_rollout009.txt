        n = n0

        else:

    # convert init_a to logits

    logits = np.log(init_a + eps)  # add epsilon to avoid -inf

    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1_pow = beta1
    beta2_pow = beta2

    beta_mult = 1.02
    max_beta = 50.0

    best_score = compute_score(best_a)
    best_iter = 0
    iter_no_improve = 0

    # Precompute size for FFT
    size = 1 << ((3*n - 2).bit_length())
    # Main loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute a via softmax of logits
        max_logit = np.max(logits)
        a = np.exp(logits - max_logit)
        a_sum = a.sum()
        a = a / a_sum

        # FFT of a
        a_fft = np.fft.rfft(a, n=size)

        # Full convolution a*a
        conv_fft = a_fft * a_fft
        conv = np.fft.irfft(conv_fft, n=size)[:2*n-1]

        # compute current score for tracking
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # update best if needed
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            best_iter = iteration
            iter_no_improve = 0
            print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")
        else:
            iter_no_improve += 1

        # compute smoothed max using softmax weights
        # Scale conv by smooth_beta
        scaled = conv * smooth_beta
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()

        # compute correlation w_rev * a via FFT
        w_rev = w[::-1]

        start = 2*n - 2
        grad_a = 2.0 * corr[start:start+n]

        # compute gradient wrt logits
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_logits
        v = beta2 * v + (1 - beta2) * (grad_logits**2)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        beta1_pow *= beta1
        beta2_pow *= beta2

def run_optimizer(n, init_a, time_budget):
    # ensure init_a shape = n
    if init_a.shape[0] != n:
        # upscale/downscale to n (maybe fallback uniform)
        init_a = upscale_distribution(init_a, n)
    # add epsilon to avoid zero log
    eps = 1e-12
    logits = np.log(init_a + eps)

    # FFT size
    size = 1 << ((3*n-2).bit_length())
    # Adam parameters
    lr = 0.05  # may tune
    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1_pow = beta1
    beta2_pow = beta2

    # schedule for smoothing beta
    smooth_beta = 1.0
    beta_mult = 1.02
    max_beta = 30.0

    # best records
    best_seq = None
    best_score = np.inf
    best_iter = 0
    iter_no_improve = 0

    deadline = time.time() + time_budget
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # compute a
        max_logit = np.max(logits)
        a = np.exp(logits - max_logit)
        a_sum = a.sum()
        a = a / a_sum

        # FFT of a
        a_fft = np.fft.rfft(a, n=size)

        # convolution a*a
        conv_fft = a_fft * a_fft
        conv = np.fft.irfft(conv_fft, n=size)[:2*n-1]

        # score
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            best_iter = iteration
            iter_no_improve = 0
            print(f"Iter {iteration:6d} (n={n}) new best = {best_score:.6f}")
        else:
            iter_no_improve += 1

        # smoothed max weights w
        scaled = conv * smooth_beta
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()

        # gradient wrt a
        w_rev = w[::-1]
        w_rev_fft = np.fft.rfft(w_rev, n=size)
        grad_fft = a_fft * w_rev_fft
        conv_full = np.fft.irfft(grad_fft, n=size)[:3*n-2]
        start = 2*n - 2
        idx = start - np.arange(n)
        grad_a = 2.0 * conv_full[idx]

        # gradient wrt logits
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_logits
        v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # update power terms
        beta1_pow *= beta1
        beta2_pow *= beta2

        # schedule smoothing factor
        if smooth_beta < max_beta:
            smooth_beta *= beta_mult
            if smooth_beta > max_beta:
                smooth_beta = max_beta

        # reheat if stagnated
        if iter_no_improve > 50000:
            # add small noise
            noise = (np.random.rand(n) - 0.5) * 1e-4
            logits += noise
            iter_no_improve = 0
            print(f"Iter {iteration:6d}: reheat (stagnation)")

    # end loop
    return best_seq, best_score

"""
Propose a candidate step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm is a three‑phase
smooth‑max exponential‑gradient descent on the probability simplex
(parameterised by a soft‑max of unconstrained logits) with an
Adam‑style step‑size schedule and occasional re‑heating.  It first
optimises a modest length, upsamples the result and refines it at larger
scales, re‑using any external construction (`height_sequence_1`) if
available.  All helper functions are top‑level and the routine respects
the given time budget (default 1000 s).  The returned list is a plain
Python list; the external evaluator rescales it as required.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Basic convolution helpers (FFT based, O(n log n))
# ------------------------------------------------------------


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a) - 1.
    """
    n = a.shape[0]
    size = 1 << ((2 * n).bit_length())            # power‑of‑two ≥ 2n
    af = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(af * af, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact score 2 * n * max(conv) for a probability vector (sum(a)=1).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Utility: up‑sample / down‑sample a probability distribution
# ------------------------------------------------------------


def upscale_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resample a discrete probability distribution to a new length
    using linear interpolation of the cumulative distribution.
    """
    if a.ndim != 1:
        raise ValueError("a must be 1‑D")
    if a.size == 0:
        raise ValueError("empty distribution")
    # cumulative (including 0 and 1 at the ends)
    cum = np.concatenate([[0.0], np.cumsum(a)])
    src = np.linspace(0.0, 1.0, cum.size)          # len = old_n + 1
    dst = np.linspace(0.0, 1.0, new_n + 1)
    cum_new = np.interp(dst, src, cum)
    return np.diff(cum_new)


# ------------------------------------------------------------
# Core optimiser for a fixed length
# ------------------------------------------------------------


def run_optimizer(
    n: int,
    init_a: np.ndarray,
    time_budget: float,
    *,
    seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponential‑gradient descent on the simplex for a fixed size.
    Returns (best_sequence, best_score) found within `time_budget` seconds.
    """
    np.random.seed(seed)
    eps = 1e-12

    # initialise logits from the starting probability vector
    a0 = init_a.astype(np.float64)
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / s0
    logits = np.log(a0 + eps)                     # ensures >‑inf

    # FFT size chosen once for the whole run
    size = 1 << ((3 * n - 2).bit_length())        # ≥ 3n‑2 for the gradient step

    # Adam hyper‑parameters
    lr = 0.07
    beta1 = 0.90
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1_pow = beta1
    beta2_pow = beta2

    # smoothing for the max (soft‑max) – start gentle, increase to focus on the true max
    smooth_beta = 1.0
    beta_mult = 1.02
    max_beta = 30.0

    # record the best solution seen
    best_seq = None
    best_score = np.inf
    iter_no_improve = 0
    iteration = 0

    deadline = time.time() + time_budget
    while time.time() < deadline:
        iteration += 1

        # ----------- probability vector from logits (soft‑max) -----------
        max_logit = np.max(logits)
        a = np.exp(logits - max_logit)
        a /= a.sum()

        # ----------- forward convolution a * a (FFT) --------------------
        a_fft = np.fft.rfft(a, n=size)
        conv_fft = a_fft * a_fft
        conv = np.fft.irfft(conv_fft, n=size)[: 2 * n - 1]

        # current (exact) score – used for bookkeeping only
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            iter_no_improve = 0
            print(f"Iter {iteration:6d} (n={n}) new best = {best_score:.6f}")
        else:
            iter_no_improve += 1

        # ----------- smooth‑max weights w for sub‑gradient -------------
        scaled = conv * smooth_beta
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()               # shape (2n‑1,)

        # ----------- gradient w.r.t. a (using correlation) ------------
        # grad_a_i = 2 * sum_j a_j * w_{i+j}
        w_rev = w[::-1]                                 # reverse for correlation
        w_rev_fft = np.fft.rfft(w_rev, n=size)
        corr_fft = a_fft * w_rev_fft
        corr_full = np.fft.irfft(corr_fft, n=size)[: 3 * n - 2]

        start = 2 * n - 2                               # index where i = 0
        idx = start - np.arange(n)                      # descending slice
        grad_a = 2.0 * corr_full[idx]                   # shape (n,)

        # ----------- gradient w.r.t. logits (mirror descent) ---------
        avg = np.dot(a, grad_a)                         # ⟨a,∇_a L⟩
        grad_logits = a * (grad_a - avg)                # ∂L/∂z_i

        # ----------- Adam update ------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # update bias‑correction factors
        beta1_pow *= beta1
        beta2_pow *= beta2

        # ----------- schedule smoothing factor -----------------------
        if smooth_beta < max_beta:
            smooth_beta = min(max_beta, smooth_beta * beta_mult)

        # ----------- re‑heat on long stagnation ----------------------
        if iter_no_improve > 50000:
            noise = (np.random.rand(n) - 0.5) * 1e-4
            logits += noise
            iter_no_improve = 0
            print(f"Iter {iteration:6d}: re‑heat (stagnation)")

    return best_seq, best_score

        seq, sc = run_optimizer(
            n=n,
            init_a=init_dist,
            time_budget=phase_time,
            seed=seed + n,       # deterministic but different per phase
        )

        # Keep the global best across phases
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"Phase n={n} yielded new global best score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4) Final tidy‑up and return
    # -----------------------------------------------------------------
    if best_seq is None:                           # extremely unlikely
        best_seq = np.full(init_n, 1.0 / init_n)
        best_score = score_from_vector(best_seq)

    best_seq = np.clip(best_seq, 0.0, 1000.0)       # respect evaluator limits
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
