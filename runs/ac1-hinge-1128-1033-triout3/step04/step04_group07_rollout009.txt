            A = np.zeros((L, n))
            # fill
            # using broadcasting: each column i gets f[i] at rows i .. i+n-1
            for i_idx in range(n):
                A[i_idx:i_idx+n, i_idx] = f[i_idx]
            # Solve LP
            c = -np.ones(n)  # maximize sum(g) -> minimize -sum(g)
            b_ub = np.full(L, cur_M)
            bounds = [(0, None)] * n
            try:
                from scipy.optimize import linprog
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
                if res.success:
                    g = res.x
                    g_sum = g.sum()
                    if g_sum > 0:
                        g = g / g_sum
                        # mixing
                        t = 0.2
                        p_swap = (1 - t) * p_swap + t * g
                        p_swap = _simplex_projection(p_swap)  # ensures sum 1
                        cur_score_swap = objective(p_swap)
                        if cur_score_swap < best_score:
                            best_score = cur_score_swap
                            best_p = p_swap.copy()
                            print(f" LP refinement {lp_iter} new best {best_score:.6f}")
                            if best_score <= 1.503:
                                break
                else:
                    # LP failed, break
                    break
            except Exception as e:
                print(f"LP exception: {e}")
                break
        
        # if reached target early, break out
        if best_score <= 1.503:
            break
    
    # Final fallback
    if best_p is None:
        best_p = np.full(n, 1.0 / n)
        best_score = objective(best_p)
    
    print(f"\nFinal best score: {best_score:.6f}")
    # Clip for safety (though values should be within [0,1000])
    best_p = np.clip(best_p, 0.0, 1000.0)
    return best_p.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimises

    2 · n · max(conv_self) / (sum(a)²)

The algorithm proceeds in three stages:

1. **Exploration** – generate a diverse pool of candidate shapes
   (arcsine‑like with a tunable exponent, geometric decays,
   uniform, a few random Dirichlet draws and the historic sequence if
   available) and keep the few best according to the exact objective.

2. **Local optimisation** – for each promising candidate run a
   *projected* optimisation consisting of:
   * a short sub‑gradient descent on the true max,
   * a long Adam optimisation on a log‑sum‑exp smoothing of the max,
   * a fine‑grained sub‑gradient polishing,
   * a random‑swap hill‑climber.
   All updates are projected onto the probability simplex
   (non‑negative, sum = 1).

3. **LP refinement** – one (or a few) steps of the classic
   linear‑programming “alternating‑LP’’ update:
   for the current vector `f` (sum = 1) solve
   `max Σ g_j   s.t.   g≥0,  f * g ≤ M`  (where `M` is the current
   maximal self‑convolution).  The optimum `g` is normalised and mixed
   with `f`.  The LP is solved with `scipy.optimize.linprog`.

The routine respects the 1000 s time budget, continuously checks the
deadline and returns the best sequence found before the timeout.
All numbers are clipped to the `[0, 1000]` range required by
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – ensures non‑negative entries and sum=1.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution utilities (FFT based) – full linear self‑convolution.
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Return full self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the log‑sum‑exp smoothed max (used by Adam phase).
# ----------------------------------------------------------------------
def _grad_logsumexp(p: np.ndarray, p_fft: np.ndarray,
                    w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Compute ∇_p  Σ_i w_i (p*p)[i]  where w are soft‑max weights.
    """
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    # correlation of p with w_rev, length up to 3·n‑2
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]
    idx = 2 * p.size - 2 - np.arange(p.size)      # selection for gradient
    grad = 2.0 * corr[idx]
    return grad


# ----------------------------------------------------------------------
# Exact objective (scale invariant – sum is assumed to be 1).
# ----------------------------------------------------------------------
def _objective(p: np.ndarray, m_fft: int) -> float:
    conv = _conv_self(p, m_fft)
    return float(2.0 * p.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size (length of the step function)
    # --------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = len(base_seq)
        have_base = True
    except Exception:
        base_seq = None
        n = 1153
        have_base = False

    # FFT size – next power‑of‑two that can hold any convolution we need
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Build a pool of diverse initial vectors (no optimisation yet)
    # --------------------------------------------------------------
    i = np.arange(n, dtype=float)

    init_vectors = []

    # 1) Arcsine‑like with tunable exponent p ∈ [0.30, 0.80]
    for exp in np.linspace(0.30, 0.80, 9):
        denom = ((i + 0.5) * (n - 0.5 - i) + 1e-12) ** exp
        vec = 1.0 / denom
        vec /= vec.sum()
        init_vectors.append(vec)

    # 2) Geometric decays (r close to 1)
    for r in (0.999, 0.997, 0.995, 0.99, 0.985):
        geom = r ** i
        geom /= geom.sum()
        init_vectors.append(geom)

    # 3) Uniform
    init_vectors.append(np.full(n, 1.0 / n))

    # 4) A few random Dirichlet draws (actually simple normalised randoms)
    for _ in range(5):
        rnd = rng.random(n)
        rnd /= rnd.sum()
        init_vectors.append(rnd)

    # 5) Historic construction, if available
    if have_base:
        seq = np.clip(base_seq, 0.0, 1000.0)
        if seq.sum() > 0.0:
            init_vectors.append(seq / seq.sum())

    # --------------------------------------------------------------
    # Quick pre‑evaluation – keep the best few candidates only.
    # --------------------------------------------------------------
    scored = []
    for vec in init_vectors:
        p = np.maximum(vec, 0.0)
        p = p / p.sum()
        sc = _objective(p, m_fft)
        scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    # keep the top 4 (adjustable)
    top_candidates = [p for _, p in scored[:4]]
    print(f"Pre‑evaluation kept {len(top_candidates)} candidates, best score {scored[0][0]:.6f}")

    # --------------------------------------------------------------
    # Optimisation helpers (sub‑gradient, Adam, hill‑climber)
    # --------------------------------------------------------------
    def _subgradient_phase(p0: np.ndarray, max_iter: int,
                           step0: float, decay: float,
                           name: str) -> tuple[np.ndarray, float]:
        p = p0.copy()
        step = step0
        best_local = _objective(p, m_fft)
        best_vec = p.copy()
        for it in range(max_iter):
            if time.time() >= deadline:
                break
            conv = _conv_self(p, m_fft)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_local - 1e-12:
                best_local = cur_score
                best_vec = p.copy()
                # global update will happen outside
            # active set (indices attaining max within a tiny tolerance)
            tol = 1e-13
            active = np.where(conv >= max_val - tol)[0]
            g = np.zeros_like(p)
            for idx in active:
                if idx < n:
                    g[: idx + 1] += 2.0 * p[idx::-1]
                else:
                    start = idx - n + 1
                    g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
            g /= float(len(active))
            p = p - step * g
            p = _simplex_projection(p)
            step = max(step * decay, 1e-9)
        return best_vec, best_local

    def _adam_phase(p0: np.ndarray,
                    max_iter: int,
                    base_lr: float,
                    alpha_start: float,
                    alpha_mul: float,
                    alpha_max: float,
                    name: str) -> tuple[np.ndarray, float]:
        p = p0.copy()
        m1 = np.zeros_like(p)
        m2 = np.zeros_like(p)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        alpha = alpha_start
        best_local = _objective(p, m_fft)
        best_vec = p.copy()
        for it in range(max_iter):
            if time.time() >= deadline:
                break
            p_fft = np.fft.rfft(p, n=m_fft)
            conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

            # log‑sum‑exp smoothing
            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            w = exp_z / np.sum(exp_z)

            grad = _grad_logsumexp(p, p_fft, w, m_fft)

            # Adam update
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            m_hat = m1 / (1.0 - beta1 ** (it + 1))
            v_hat = m2 / (1.0 - beta2 ** (it + 1))

            lr = base_lr / np.sqrt(alpha)          # shrink LR as α grows
            step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

            p = p - step_vec
            p = _simplex_projection(p)

            alpha = min(alpha * alpha_mul, alpha_max)

            # occasional exact evaluation
            if (it & 1023) == 0:
                cur_score = _objective(p, m_fft)
                if cur_score < best_local - 1e-12:
                    best_local = cur_score
                    best_vec = p.copy()
        return best_vec, best_local

    def _hillclimb_phase(p0: np.ndarray,
                         max_iter: int,
                         name: str) -> tuple[np.ndarray, float]:
        p = p0.copy()
        best_score = _objective(p, m_fft)
        best_vec = p.copy()
        for _ in range(max_iter):
            if time.time() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps_move = 0.001 * rng.random()
            if p[i] <= eps_move:
                continue
            new_p = p.copy()
            new_p[i] -= eps_move
            new_p[j] += eps_move
            new_score = _objective(new_p, m_fft)
            if new_score < best_score - 1e-15:
                p = new_p
                best_score = new_score
                best_vec = p.copy()
        return best_vec, best_score

    # --------------------------------------------------------------
    # LP refinement (alternating LP as described in the statement).
    # --------------------------------------------------------------
    def _lp_refine(p: np.ndarray,
                   max_iter: int,
                   mix_weight: float,
                   name: str) -> tuple[np.ndarray, float]:
        cur = p.copy()
        best_score = _objective(cur, m_fft)
        best_vec = cur.copy()
        for it in range(max_iter):
            if time.time() >= deadline:
                break
            conv = _conv_self(cur, m_fft)
            M = conv.max()
            L = 2 * n - 1
            # Build constraint matrix A (L × n)  :  (f * g)[k] = Σ_i f_i g_{k-i}
            A = np.zeros((L, n), dtype=float)
            for idx in range(n):
                A[idx:idx + n, idx] = cur[idx]
            # Linear program: maximise Σ g_j   → minimise -Σ g_j
            c = -np.ones(n)
            b_ub = np.full(L, M)
            bounds = [(0, None)] * n
            try:
                res = linprog(c, A_ub=A, b_ub=b_ub,
                              bounds=bounds, method='highs')
            except Exception as exc:          # fallback – abort LP phase
                print(f" LP exception ({name} iter {it}): {exc}")
                break
            if not res.success:
                # infeasible / numerical trouble – stop LP refinement
                break
            g = res.x
            if g.sum() == 0.0:
                break
            g_norm = g / g.sum()
            # convex combination
            cur = (1.0 - mix_weight) * cur + mix_weight * g_norm
            cur = _simplex_projection(cur)
            score = _objective(cur, m_fft)
            if score < best_score - 1e-12:
                best_score = score
                best_vec = cur.copy()
            # early stop if already excellent
            if best_score <= 1.503:
                break
        return best_vec, best_score

    # --------------------------------------------------------------
    # Core optimisation loops over the selected candidates.
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5    # tiny safety margin
    global_best_score = float("inf")
    global_best = None

    cand_id = 0
    for cand in top_candidates:
        if time.time() >= deadline:
            break
        cand_id += 1
        print(f"\n=== Optimising candidate {cand_id}/{len(top_candidates)} ===")
        p = np.maximum(cand, 0.0)
        p = p / p.sum()

        # (A) Sub‑gradient descent on the exact max
        p, sc = _subgradient_phase(p,
                                   max_iter=15000,
                                   step0=0.4,
                                   decay=0.9997,
                                   name="subgrad")
        if sc < global_best_score:
            global_best_score, global_best = sc, p.copy()
            print(f"  After sub‑grad: new global best {global_best_score:.6f}")

        # (B) Adam on smooth log‑sum‑exp max
        p, sc = _adam_phase(p,
                            max_iter=250000,
                            base_lr=0.08,
                            alpha_start=1.0,
                            alpha_mul=1.001,
                            alpha_max=400.0,
                            name="adam")
        if sc < global_best_score:
            global_best_score, global_best = sc, p.copy()
            print(f"  After Adam: new global best {global_best_score:.6f}")

        # (C) Fine‑grained sub‑gradient polishing
        p, sc = _subgradient_phase(p,
                                   max_iter=60000,
                                   step0=0.2,
                                   decay=0.9999,
                                   name="fine_subgrad")
        if sc < global_best_score:
            global_best_score, global_best = sc, p.copy()
            print(f"  After fine‑subgrad: new global best {global_best_score:.6f}")

        # (D) Random‑swap hill‑climber
        p, sc = _hillclimb_phase(p,
                                 max_iter=50000,
                                 name="swap")
        if sc < global_best_score:
            global_best_score, global_best = sc, p.copy()
            print(f"  After hill‑climb: new global best {global_best_score:.6f}")

        # (E) LP refinement – a few aggressive steps
        p, sc = _lp_refine(p,
                           max_iter=5,
                           mix_weight=0.2,
                           name="lp")
        if sc < global_best_score:
            global_best_score, global_best = sc, p.copy()
            print(f"  After LP refine: new global best {global_best_score:.6f}")

        # early exit if target already reached
        if global_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if global_best is None:
        global_best = np.full(n, 1.0 / n)
        global_best_score = _objective(global_best, m_fft)

    print(f"\nFinal best score: {global_best_score:.6f}")
    # Clip to the bounds enforced by evaluate_sequence (0 ≤ x ≤ 1000)
    return np.clip(global_best, 0.0, 1000.0).tolist()
