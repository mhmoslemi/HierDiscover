"""
Propose a step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. Initialise a candidate from the global ``height_sequence_1`` if present,
   otherwise use the uniform distribution (sum = 1).
2. Perform several *restarts* of a smooth surrogate optimisation:
   * Represent a as a soft‑max of parameters w.
   * Minimise a log‑sum‑exp approximation of the max‑convolution
     using L‑BFGS‑B with a continuation schedule on the smoothing
     parameter ε.
   * After each ε‑stage evaluate the exact objective and keep the best
     iterate.
3. With the best solution found, run a *simulated‑annealing* (SA) stage:
   * Randomly move a small fraction of mass from one index to another.
   * Accept moves that improve the objective, otherwise accept with a
     probability `exp(-(Δ)/T)` where T is a temperature that decays
     exponentially.
   * Fast evaluation of the max‑convolution is performed with an FFT‑based
     full convolution (O(n log n)).
4. Return the best sequence seen before the time budget expires.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself (length 2·len(a)-1)
    using a real FFT. The result is clipped to be non‑negative to avoid
    tiny negative rounding errors.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    ft = np.fft.rfft(a, size)
    conv = np.fft.irfft(ft * ft, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors (length len(x)+len(y)-1) via FFT.
    Result is clipped to be non‑negative.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient w.r.t. soft‑max params
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) for a given parameter vector w and smoothing ε.
    L is the smooth approximation of 2·n·max(conv) and grad_w is the
    gradient for the L‑BFGS‑B optimizer.
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)            # numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # full auto‑convolution a*a
    conv = _full_convolution(a)         # length 2·n‑1
    max_conv = np.max(conv)

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps   # ≤ 0
    log_sum = logsumexp(shifted)        # = log ∑_k exp((conv_k‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)           # exp((conv‑max)/ε)
    p = weights / np.sum(weights)       # soft‑max over convolution entries

    # g_i = 4·n·∑_k p_k·a_{k‑i}  (indices outside [0,n‑1] give 0)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)       # length 3·n‑2
    seg = conv_ap[n - 1 : 2 * n - 1]    # middle n entries
    grad_a = 4.0 * n * seg[::-1]        # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation on the smooth surrogate (single restart)
# ----------------------------------------------------------------------
def _lbfgs_optimize(w0: np.ndarray, n: int, deadline: float):
    """
    Runs the continuation L‑BFGS‑B scheme starting from soft‑max parameters
    ``w0``. Returns the best feasible sequence (as a NumPy array) and its
    exact evaluation obtained via ``evaluate_sequence``.
    """
    w = w0.copy()
    # continuation schedule (geometric progression in log‑space)
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-7), num=14)

    best_local_a = None
    best_local_val = np.inf

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        # L‑BFGS‑B with a modest iteration budget (fast per ε‑stage)
        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 150,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # exact evaluation of the current iterate
        a = np.exp(w - np.max(w))
        a = a / np.sum(a)
        try:
            val = evaluate_sequence(a.tolist())
        except Exception:
            continue
        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            print(f"[L‑BFGS ε={eps:.1e}] new best = {best_local_val:.6f}")

    # If the optimizer never produced a valid point, fall back to the final w
    if best_local_a is None:
        a = np.exp(w - np.max(w))
        a = a / np.sum(a)
        best_local_a = a
        best_local_val = evaluate_sequence(a.tolist())
    return best_local_a, best_local_val

# Simulated‑annealing refinement (fast FFT‑based evaluation)

def _simulated_annealing(a_start: np.ndarray, n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Starting from ``a_start`` (already normalised to sum=1) performs a
    simulated‑annealing search using fast FFT convolution. Returns the
    best sequence found and its exact evaluation.
    """
    rng = np.random.default_rng()
    a = a_start.copy()
    # current objective (exact, because sum=1)
    cur_conv = _full_convolution(a)
    cur_M = np.max(cur_conv)
    cur_val = 2.0 * n * cur_M

    best_a = a.copy()
    best_val = cur_val

    # temperature schedule (exponential decay)
    T0 = max(0.1 * cur_val, 1e-3)      # start relatively warm
    Tend = 1e-8
    # approximate number of iterations we can afford (empirical)
    max_iters = 200_000
    i = 0
    while time.time() < deadline and i < max_iters:

        T = T0 * (Tend / T0) ** progress
