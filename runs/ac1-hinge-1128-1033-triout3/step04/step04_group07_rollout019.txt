def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_value(seq: np.ndarray) -> float:
    """Return the exact value of evaluate_sequence for a numpy array."""
    # evaluate_sequence does its own clipping, but we mimic the behaviour.
    seq = np.maximum(0.0, np.minimum(1000.0, seq.astype(float)))
    n = len(seq)

    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))

    """
    Solve the linear programme described in the problem statement:
        maximise   Σ g_i
        subject to f * g  ≤ max_conv   (pointwise)
                   g_i ≥ 0
    Return a convex combination (1‑t)·f + t·ĝ where ĝ is the normalised
    solution of the LP.

        # If SciPy is missing we simply return f unchanged.
        return f

    c = -np.ones(n)

                  A_ub=A,
                  b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})

        return f
