"""
Propose a step‑function (non‑negative vector) that minimises

    2 · n · max(conv_self) / (sum(a)²)

The algorithm proceeds in three stages:

1. **Diverse initialisations** –  arcsine shape, geometric decay,
   uniform, several β‑distribution shapes (β⁽α,α⁾ with α∈(0.1,0.5))
   and random Dirichlet draws.

2. **Annealed exponential‑smooth optimisation** –  we optimise the
   log‑sum‑exp approximation of the maximal convolution entry.  The
   probability vector is represented by soft‑max logits and updated
   with Adam.  The smoothing parameter α starts at 1.0 and is multiplied
   by a factor ≈1.15 each “temperature” until a large value (≈5 000)
   is reached, thus turning the smooth objective into the true max.
   The Adam step size is scaled by 1/√α to keep updates stable.

3. **Exact‑max polishing** –  a short projected sub‑gradient phase on the
   genuine max‑convolution followed by a tiny random‑swap hill‑climber.
   All updates are projected onto the probability simplex (via the
   Euclidean projection) guaranteeing feasibility for `evaluate_sequence`.

The whole routine respects the 1000 s budget, keeps the best vector seen,
and returns it (clipped to the [0, 1000] range required by the evaluator).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full linear self‑convolution of p using a pre‑computed FFT size."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _gradient_from_weights(p: np.ndarray, p_fft: np.ndarray,
                          w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed max w.r.t. p.
    w – soft‑max weights (length 2·n‑1) that sum to 1.
    """
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad


# ----------------------------------------------------------------------
# Exact objective (used for logging / final polishing)
# ----------------------------------------------------------------------
def _exact_score(p: np.ndarray, n: int, m_fft: int) -> float:
    """Return 2·n·max(p∗p) for a probability vector p (sum(p)=1)."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * n * conv.max())

    arcsine = arcsine / arcsine.sum()
    init_vectors.append(arcsine)

    # 2) geometric decay
    r = 0.99
    geom = r ** i
    init_vectors.append(geom / geom.sum())

    # 3) uniform
    init_vectors.append(np.full(n, 1.0 / n))

    # 4) β‑shapes with various α ∈ (0.2,0.5)
    for alpha in (0.20, 0.25, 0.30, 0.35, 0.40, 0.45):
        w = (i + 0.5) ** (alpha - 1.0) * (n - 0.5 - i) ** (alpha - 1.0)
        w = np.maximum(w, 0.0)

        init_vectors.append(rnd / rnd.sum())

        if s > 0:

            for it in range(steps_this_alpha):

                max_log = np.max(logits)
                exp_shift = np.exp(logits - max_log)
                p = exp_shift / exp_shift.sum()

                # ---------- FFT of p ----------
                p_fft = np.fft.rfft(p, n=m_fft)

                # ---------- self‑convolution ----------
                conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

                # ---------- log‑sum‑exp weights ----------
                z = alpha * conv
                max_z = np.max(z)
                exp_z = np.exp(z - max_z)           # stable softmax
                w = exp_z / exp_z.sum()             # w sums to 1

                # ---------- gradient w.r.t. p ----------
                grad_p = _gradient_from_weights(p, p_fft, w, m_fft)

                # ---------- gradient w.r.t. logits (soft‑max chain rule) ----------
                dot = np.dot(p, grad_p)
                grad_logits = p * (grad_p - dot)

                # ---------- Adam update ----------
                m = beta1 * m + (1.0 - beta1) * grad_logits
                v = beta2 * v + (1.0 - beta2) * (grad_logits * grad_logits)
                t = it + 1
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)

                # Learning rate scaled down as α grows (more peaked objective)
                lr = base_lr / np.sqrt(alpha)
                step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                logits -= step

                # ---------- occasional exact evaluation ----------
                if (it & 511) == 0:          # every 512 iterations

                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best_p = p.copy()

                        if best_score <= 1.503:
                            break

            # increase smoothing
            alpha = min(alpha * alpha_factor, alpha_max)

        # --------------------------------------------------------------
        # (B) Exact‑max sub‑gradient polishing (few thousand iterations)
        # --------------------------------------------------------------
        # start from the best vector found for this start (or current p)
        p = best_p.copy() if best_p is not None else p
        step = 0.12
        decay = 0.9998
        min_step = 5e-9

        for it in range(9000):
            if time.time() >= deadline:
                break
            conv = _conv_self(p, m_fft)
            max_val = conv.max()
            cur_score = float(2.0 * n * max_val)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                print(f"  Subgrad it={it:5d} → new best {best_score:.6f}")
                if best_score <= 1.503:
                    break

            # identify the (approximately) active set
            tol = 1e-13
            active = np.where(conv >= max_val - tol)[0]

            # gradient: average over active indices
            g = np.zeros_like(p)
            for idx in active:
                if idx < n:
                    g[: idx + 1] += 2.0 * p[idx::-1]
                else:
                    start = idx - n + 1
                    # p[n-1 : start-1 : -1] is the reversed slice we need
                    g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
            g /= float(len(active))

            p = p - step * g
            p = _simplex_projection(p)

        # --------------------------------------------------------------
        if best_p is None:

        else:

        cur_score = _exact_score(candidate, n, m_fft)

    return np.clip(best_p, 0.0, 1000.0).tolist()
