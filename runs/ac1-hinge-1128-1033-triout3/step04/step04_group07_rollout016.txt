"""
propose_candidate
-----------------

Search for a non‑negative sequence a (treated as a probability vector,
Σ a_i = 1) that minimizes

    2·n·max(conv(a,a)) / (Σ a_i)²   =   2·n·max(conv(a,a))

where ``conv`` is the full discrete self‑convolution.
The algorithm is a multi‑start projected optimisation:

1. **Initial pool** – a few hand‑crafted shapes (uniform,
   geometric, arcsine, several exponent‑shaped vectors) and a handful of
   random Dirichlet draws.
2. **Phase A – sub‑gradient descent** on the exact max‑entry using the
   active set of binding lags.
3. **Phase B – Adam on a smooth log‑sum‑exp approximation** of the max,
   with a temperature that grows to force the smooth objective to the
   true max.
4. **Phase C – short fine‑grained sub‑gradient polishing.**
5. **Phase D – tiny random‑swap hill climbing** (with a decreasing
   step‑size) to escape possible local minima.

All updates are projected onto the probability simplex, guaranteeing
non‑negative coefficients that sum to one (exactly the domain expected
by ``evaluate_sequence``).  The routine runs until the supplied time
budget expires (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of p using FFT of size m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _objective(p: np.ndarray, n: int, m_fft: int) -> float:
    """Exact objective 2·n·max(conv(p,p)).  p is assumed to sum to 1."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * n * np.max(conv))


def _max_gradient(p: np.ndarray,
                  conv: np.ndarray,
                  n: int,
                  tol: float = 1e-13) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at p.
    Uses the active set of lags whose convolution value is within ``tol``
    of the maximum.
    """
    max_val = conv.max()
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(p)
    L = len(active)
    if L == 0:
        return g
    for idx in active:
        if idx < n:
            # overlap length = idx+1
            g[: idx + 1] += 2.0 * p[idx::-1]
        else:
            start = idx - n + 1
            # overlapping slice from start … n‑1, reversed
            g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
    g /= float(L)
    return g


def _smooth_gradient(p: np.ndarray,
                     p_fft: np.ndarray,
                     w: np.ndarray,
                     n: int,
                     m_fft: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp (soft‑max) approximation of the max.
    ``w`` are the soft‑max weights, summing to 1.
    """
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    # correlation of p with w_rev (full length 3n‑2)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * corr[idx]


def _exponent_shape(n: int, exp: float) -> np.ndarray:
    """
    Produce a monotone shape a_i ∝ ((i+½)(n‑½‑i))^{-exp}.
    This family includes the classic arcsine (exp=0.5).
    """
    i = np.arange(n, dtype=float)
    denom = (i + 0.5) * (n - 0.5 - i) + 1e-12
    v = denom ** (-exp)
    v = np.maximum(v, 0.0)
    v /= v.sum()
    return v

        eps_adam = 1e-8
        base_lr = 0.05
        alpha = 1.0
        alpha_mul = 1.001
        alpha_max = 300.0

        m1 = np.zeros_like(p)
        m2 = np.zeros_like(p)
        max_iter_b = 120000

        for it in range(max_iter_b):
            if time.time() >= deadline:
                break

            # FFT of current p (used both for evaluation and gradient)
            p_fft = np.fft.rfft(p, n=m_fft)
            conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

            # soft‑max weights (stable log‑sum‑exp)
            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            w = exp_z / exp_z.sum()

            # gradient of the smooth max
            grad = _smooth_gradient(p, p_fft, w, n, m_fft)

            # Adam update
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            m_hat = m1 / (1.0 - beta1 ** (it + 1))
            v_hat = m2 / (1.0 - beta2 ** (it + 1))

            lr = base_lr / np.sqrt(alpha)          # shrink learning‑rate as α grows
            step_vec = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            p = p - step_vec
            p = _simplex_projection(p)

            # raise temperature (σ) gradually
            alpha = min(alpha * alpha_mul, alpha_max)

            # occasional exact evaluation (every 1024 steps)
            if (it & 1023) == 0:
                cur_score = _objective(p, n, m_fft)
                if cur_score + 1e-12 < best_score:
                    best_score = cur_score
                    best_p = p.copy()
                    print(f"  Adam iter {it:6d} – new best {best_score:.6f}")

        # --------------------------------------------------------------
        # Phase C – fine‑grained sub‑gradient polishing
        # --------------------------------------------------------------
        step_c = 0.15
        decay_c = 0.9999
        min_step_c = 1e-9
        max_iter_c = 40000

        for it in range(max_iter_c):
            if time.time() >= deadline:
                break

            conv = _conv_self(p, m_fft)
            cur_score = float(2.0 * n * conv.max())
            if cur_score + 1e-12 < best_score:
                best_score = cur_score
                best_p = p.copy()
                print(f"  Fine‑subgrad {it:5d} – new best {best_score:.6f}")

            g = _max_gradient(p, conv, n, tol=1e-13)
            p = p - step_c * g
            p = _simplex_projection(p)

            step_c = max(step_c * decay_c, min_step_c)

        # --------------------------------------------------------------
        # Phase D – tiny random‑swap hill‑climber (simulated‑annealing style)
        # --------------------------------------------------------------
        p_swap = best_p.copy() if best_p is not None else p.copy()
        cur_score = _objective(p_swap, n, m_fft)

        max_swaps = 50000
        # start with a relatively large “temperature” for the swaps
        max_eps = 0.02

        for swap_id in range(max_swaps):
            if time.time() >= deadline:
                break

            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            # move a small fraction of mass from i to j
            eps_move = max_eps * rng.random()
            if p_swap[i] <= eps_move:
                continue

            new_p = p_swap.copy()
            new_p[i] -= eps_move
            new_p[j] += eps_move
            new_score = _objective(new_p, n, m_fft)

            # Accept only improvements (simple hill‑climb)
            if new_score + 1e-15 < cur_score:
                p_swap = new_p
                cur_score = new_score
                if cur_score + 1e-12 < best_score:
                    best_score = cur_score
                    best_p = p_swap.copy()
                    print(f"  Swap improvement → {best_score:.6f}")

            # gradually reduce the allowed move size – akin to cooling
            max_eps *= 0.99995

        # early exit if we already reach the historic target
        if best_score <= 1.503:
            print("Target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_p is None:
        best_p = np.full(n, 1.0 / n)
        best_score = _objective(best_p, n, m_fft)

    print(f"\nFinal best score (before clipping): {best_score:.6f}")

    # Clip to the bounds enforced by evaluate_sequence (0 ≤ x ≤ 1000)
    return np.clip(best_p, 0.0, 1000.0).tolist()
