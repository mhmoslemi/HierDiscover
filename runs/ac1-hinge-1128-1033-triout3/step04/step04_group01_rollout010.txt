"""
Propose a step‑function (non‑negative heights) that minimises
`evaluate_sequence`.

Algorithm
---------
* Phase A – smooth‑max Adam optimisation.
  The vector a is represented by soft‑max weights w so that Σa = 1
  and a_i ≥ 0.  We minimise a smooth log‑sum‑exp surrogate of
  max(conv(a,a)).  The temperature ε is annealed from 1e‑2 to 1e‑9
  while the learning‑rate decays from 1e‑1 to 1e‑3.
  Adam (β1=0.9, β2=0.999) handles the steps.  The best candidate
  seen is kept.

* Phase B – targeted hill‑climbing on the *exact* objective.
  From the best point of Phase A we keep the full convolution b
  (length 2·n‑1).  At each iteration we locate the index k where
  b is maximal.  The contribution of a_i to this worst entry is
  a_i·a_{k‑i}.  We therefore choose a source index i with large
  a_{k‑i} and transfer a small amount of mass to a target index j
  where a_{k‑j} is small.  The convolution can be updated in
  O(n) time using the bilinearity of the convolution, so millions
  of moves fit easily into the remaining budget.  After long
  stagnation a tiny random perturbation of the current best
  solution is performed to escape local minima.

Both phases together usually push the bound well below the
previous 1.5138 within the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolution (real data)
# ----------------------------------------------------------------------
def full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of `x` with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the best (a, value) found before *deadline*.
    """
    # soft‑max parametrisation, w lives in ℝⁿ
    w = np.log(a0 + 1e-12)

    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.10          # fairly aggressive start
    lr_end = 0.001
    eps_start = 1e-2
    eps_end = 1e-9

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 30_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # --------- map w → a (feasible point) ----------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a_sum = a.sum()
        if a_sum == 0.0:          # numeric safeguard
            a = np.full_like(a, 1.0 / n)
            a_sum = 1.0
        a /= a_sum                # Σa = 1, a_i ≥ 0

        # --------- exact objective (for monitoring) ----------
        b = full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # --------- schedule the smoothing parameter ε ----------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # --------- smooth‑max weights (log‑sum‑exp) ----------
        shifted = (b - b.max()) / eps          # stable shift
        wgt = np.exp(shifted)
        p = wgt / wgt.sum()                    # soft‑max over convolution entries

        # --------- gradient of smooth‑max w.r.t. a ----------
        # derivative of b_k w.r.t. a is 2 * (a reversed at offset k)
        p_rev = p[::-1]
        grad_ab = conv_fft(a, p_rev)           # length 3·n‑2
        # interior slice gives ∂/∂a_i of ∑_k p_k b_k
        segment = grad_ab[n - 1: 2 * n - 1]     # length n
        grad_a = 4.0 * n * segment[::-1]       # factor 4 comes from the chain rule below

        # --------- chain rule through soft‑max (w) ----------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # --------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # --------- occasional tiny perturbation ----------
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.005, size=n)

        # --------- restart if we are stuck ----------
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _targeted_hillclimb(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> np.ndarray:
    """
    Fast local search that reduces the *max* entry of the convolution
    by moving mass from indices that contribute strongly to the worst
    entry to indices that contribute weakly.
    Returns the best sequence seen before *deadline*.
    """
    a = a_start.copy()
    b = full_convolution(a)                 # length 2·n‑1
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max

    # parameters controlling the amount of transferred mass
    max_frac = 0.5
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- locate the worst convolution entry ----
        k = int(np.argmax(b))                # 0 … 2·n‑2

        # partner[i] = a_{k-i} if the index exists, else 0
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        partner = np.zeros_like(a)
        partner[mask] = a[idx[mask]]

        # ---- choose a source index i ----
        # we prefer an index with a_i > 0 and large partner[i]
        candidates = np.where(a > 1e-14)[0]
        if candidates.size == 0:
            break
        src_vals = partner[candidates]
        src = candidates[np.argmax(src_vals)]

        # ---- choose a target index j ----
        # prefer a location where partner[j] is small
        low_thr = np.quantile(partner, 0.10)
        low_cands = np.where(partner <= low_thr)[0]
        low_cands = low_cands[low_cands != src]          # avoid self‑move
        if low_cands.size == 0:
            # fallback to a random index different from src
            j = np.random.randint(n)
            while j == src:
                j = np.random.randint(n)
        else:
            j = np.random.choice(low_cands)

        # ---- amount of mass to transfer ----
        delta = a[src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) ----
        b_new = b.copy()
        # cross‑terms (2·δ·a) entering when we add/remove mass
        b_new[j:j + n] += 2.0 * delta * a
        b_new[src:src + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- accept only improving moves ----
        if new_val < cur_val:
            a[src] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---- stagnation handling ----
        if no_improve > IMPROVE_LIMIT:
            # perturb the current best ever found and resume from there
            a = best_a.copy()
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a

        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)                # linear interpolation
        a0 = np.maximum(a0, 0.0)
    except Exception:
        # fall back to a simple Dirichlet (α = 0.5) to obtain a non‑uniform start
        a0 = np.random.gamma(shape=0.5, scale=1.0, size=n)
    a0 /= a0.sum()                                         # Σa = 1

    best_a = a0.copy()
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Phase A – smooth‑max Adam optimisation (≈ 35 % of the budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * budget_s
    best_a, best_val = _adam_phase(a0, n, start_time, phase_a_deadline,
                                   best_a, best_val)

    # ------------------------------------------------------------------
    # Phase B – targeted hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    best_a = _targeted_hillclimb(best_a, n, phase_a_deadline,
                                 deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
