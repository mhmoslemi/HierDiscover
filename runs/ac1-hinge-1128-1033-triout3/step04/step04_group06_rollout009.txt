def propose_candidate(...):

    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0

    Determine n.

    Initialize a (probability vector) as above.

    x = np.log(a)

    Initialize m, v = zeros.

    best_x = x.copy()
    best_a = a.copy()
    best_val = 2 * n * np.max(_full_convolution_fft(a, a))
    print(f"[init] n={n}, eval={best_val}")

    iteration = 0
    while time.time() < deadline:
        # conv of a with itself
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()
        cur_val = 2.0 * n * M

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            best_x = x.copy()
            print(f"[{iteration}] new best = {best_val:.6f}")

        # compute softmax weight w
        progress = (time.time() - start_time) / (deadline - start_time)
        # Temperature schedule
        tau_factor = (0.5 * (1 - progress) + 1e-6 * progress)  # factor
        tau = max(tau_factor * M, 1e-12)

        # softmax
        exp_args = (conv_a - M) / tau
        # clip for stability
        np.clip(exp_args, -80, 80, out=exp_args)
        exp_vals = np.exp(exp_args)
        sum_exp = exp_vals.sum()
        if sum_exp <= 0 or not np.isfinite(sum_exp):
            # fallback: use indicator of max index
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0
        else:
            w = exp_vals / sum_exp

        # gradient w.r.t a
        grad_a = _gradient_from_weights(a, w)

        # scalar dot for softmaxed gradient averaging
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update
        iteration += 1
        beta1 = 0.9
        beta2 = 0.999
        eps_Adam = 1e-8

        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)

        m_hat = m / (1 - beta1 ** iteration)

"""
Propose a candidate step‑function using a simplex‑parameterised Adam optimiser.

The algorithm works on the probability simplex:
    a_i = exp(x_i) / Σ_j exp(x_j)   (a_i ≥ 0, Σ a_i = 1).

At each iteration we
  * compute the full auto‑convolution a*a via a real FFT,
  * approximate the ∞‑norm with a soft‑max (temperature τ is annealed),
  * obtain the smoothed gradient w.r.t. a (g_a = 2·conv(w_rev, a)),
  * map this gradient to the x‑space (g_x = a·(g_a – ⟨g_a,a⟩)),
  * perform an Adam update on x,
  * rebuild a from the updated x,
  * occasionally apply a tiny random jitter to help escape plateaus,
  * keep track of the best sequence seen so far.

The routine runs until the time budget is exhausted (≈ 1000 s) and
returns the best sequence (as a plain list of floats).  It can start
from the previously discovered `height_sequence_1` if that global
variable is present, otherwise it starts from a random Dirichlet draw.
Larger n (≈ 4000) gives the optimiser more freedom, while all
operations stay fast thanks to FFT‑based convolutions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of vectors x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient from soft‑max weights (same as the baseline)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented as a single convolution of the reversed weight vector with a.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    # reverse the full weight vector (length 2n-1)
    w_rev = w[::-1]
    # full convolution length = (2n-1) + n - 1 = 3n-2
    conv = _full_convolution_fft(w_rev, a)
    # indices that give the gradient entries:
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:

    Optimise a non‑negative step‑function via Adam on the simplex.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin before the hard deadline
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # Determine the length of the sequence (a few thousand works well)
    # --------------------------------------------------------------
    try:
        # If a previous construction exists, use its length as a lower bound
        prev = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        base_len = prev.shape[0]
    except Exception:
        prev = None
        base_len = 0

    # Target length – at least twice the previous one, but not too large.
    n = max(4000, base_len * 2)
    # ------------------------------------------------------------------
    # Initialise a on the probability simplex
    # ------------------------------------------------------------------
    eps_init = 1e-12
    a = np.full(n, eps_init, dtype=float)

    if prev is not None and base_len > 0:
        # Normalise the imported sequence and embed it centrally
        prev = np.maximum(prev, 0.0)
        prev_sum = prev.sum()
        if prev_sum > 0:
            prev = prev / prev_sum
        else:
            prev = np.full(base_len, 1.0 / base_len)
        start = (n - base_len) // 2
        a[start:start + base_len] = np.maximum(prev, eps_init)

    # If the previous sequence is tiny or missing, fall back to a Dirichlet draw
    if a.sum() <= 0:
        a = np.random.dirichlet(alpha=np.full(n, 0.5))

    # Normalise to the simplex
    a = a / a.sum()
    # Parameterisation on the log‑probability simplex
    x = np.log(a + 1e-300)                     # avoid log(0)

    # ------------------------------------------------------------------
    # Adam bookkeeping
    # ------------------------------------------------------------------
    m = np.zeros_like(x)                      # first moment
    v = np.zeros_like(x)                      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Learning‑rate schedule (multiplicative in the x‑space)
    eta_start = 0.30
    eta_end   = 0.001

    # Temperature (soft‑max) schedule factors
    tau_factor_start = 0.25
    tau_factor_end   = 1e-6

    # Jitter probability (helps escaping flat plateaux)
    jitter_prob = 0.012

    conv_a = _full_convolution_fft(a, a)

    iteration = 0

    while True:

        if now >= deadline:

        # --------------------------------------------------------------

            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):

            else:
                w = exp_vals / sum_exp

        # ---- gradient w.r.t. a (smooth approximation of sub‑gradient) ----
        grad_a = _gradient_from_weights(a, w)

        # ---- map gradient to x‑space (softmax Jacobian) ----
        dot = np.dot(grad_a, a)               # ⟨grad_a, a⟩
        grad_x = a * (grad_a - dot)

        # ---- Adam moment updates ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # ---- learning‑rate schedule ----
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # ---- Adam step on the log‑probabilities ----
        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        x -= step

        # ---- reconstruct a from x (stable softmax) ----
        max_x = np.max(x)
        ex = np.exp(x - max_x)                # prevent overflow
        a = ex / ex.sum()

        # ---- occasional random jitter (mass transfer) ----
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move a random fraction (up to half) of the mass at src to dst
            delta = np.random.rand() * a[src] * 0.5
            a[src] -= delta
            a[dst] += delta
            # re‑normalise (sum stays 1, but we guard against tiny negatives)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            # keep x consistent with the modified a
            x = np.log(a + 1e-300)

        # ---- safety re‑normalisation (numeric drift) ----
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # ---- optional re‑initialisation if stuck for long ----
        if now - last_improve > 45.0 and iteration % 1000 == 0:
            # perturb the current best slightly to escape a possible plateau
            perturb = np.random.dirichlet(alpha=np.full(n, 0.3))
            a = 0.95 * best_a + 0.05 * perturb
            a = np.maximum(a, 0.0)
            a /= a.sum()
            x = np.log(a + 1e-300)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{iteration:7d}] re‑initialising after stagnation")

    # ----------------------------------------------------------------------
    # Final reporting
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
