def simplex_projection(v):
    # Project v onto the probability simplex {x: x>=0, sum(x)=1}
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

"""
Search for a step‐function (non‑negative heights) that minimises
`evaluate_sequence`.  The algorithm works in two stages.

Stage A – a smoothed‑max Adam optimiser on the simplex (soft‑max
parameterisation).  It minimises a log‑sum‑exp approximation of the
convolution maximum while keeping the coefficients non‑negative and
∑a=1.

Stage B – a deterministic sub‑gradient descent on the *exact* objective.
At each iteration the index k with the largest convolution entry is
identified, the gradient of that entry (2·a shifted by k) is taken,
a step is performed and the point is projected back onto the simplex.
Learning‑rate is decreased when progress stalls and occasional random
perturbations prevent getting stuck in local minima.

Both stages run within the allotted time budget; the best sequence found
is returned as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Σx=1, x≥0)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.05
    lr_end = 0.001
    eps_start = 5e-3
    eps_end = 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 30000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- soft‑max to retrieve a (∑a=1, a≥0) -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ----- exact objective for tracking -----
        conv_aa = _full_convolution(a)
        cur_val = 2.0 * n * conv_aa.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve_cnt += 1

        # ----- smoothing parameter ε (linear decay) -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # ----- soft‑max probabilities over convolution entries -----
        maxi = conv_aa.max()
        shifted = (conv_aa - maxi) / eps
        weights = np.exp(shifted)               # safe in float64
        p = weights / weights.sum()             # p_k ≈ one‑hot when eps tiny

        # ----- gradient of smoothed objective w.r.t. a -----
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)           # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]     # middle n entries
        grad_a = 4.0 * n * segment[::-1]        # shape (n,)

        # ----- chain rule through the soft‑max (gradient w.r.t. w) -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update on w -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- occasional random perturbation -----
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # ----- restart if we are stuck -----
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restart after {no_improve_cnt} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – deterministic sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_phase(a_start: np.ndarray,
                       n: int,
                       start: float,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> np.ndarray:
    """
    Sub‑gradient descent on the exact evaluation function
    2·n·max(conv(a,a)).  At each step we move opposite to the
    gradient of the currently maximal convolution entry and project back
    onto the simplex.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    lr = 0.2                       # initial learning‑rate
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # index of the current maximum entry
        k = int(np.argmax(b))

        # gradient of b_k w.r.t. a  ( = 2 * shifted a )
        grad = np.zeros_like(a)
        if k < n:
            grad[:k + 1] = a[k::-1]
        else:
            start_i = k - n + 1
            grad[start_i:] = a[n - 1:start_i - 1:-1]
        grad *= 2.0

        # trial step with current learning‑rate
        a_trial = _simplex_projection(a - lr * grad)
        b_trial = _full_convolution(a_trial)
        new_val = 2.0 * n * b_trial.max()

        # line‑search: keep halving the step‑size until we see improvement
        trials = 0
        while new_val >= cur_val and trials < 5:
            lr *= 0.5
            a_trial = _simplex_projection(a - lr * grad)
            b_trial = _full_convolution(a_trial)
            new_val = 2.0 * n * b_trial.max()
            trials += 1

        if new_val < cur_val:
            # accept the move
            a = a_trial
            b = b_trial
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Subgrad {iteration:7d}] new best = {best_val:.6f}")
        else:
            # no improvement – shrink learning‑rate more aggressively
            lr *= 0.7
            no_improve += 1

        # if we have been idle for a long time, jitter the current best
        if no_improve > 20000:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = _simplex_projection(a + noise)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Subgrad {iteration:7d}] Stagnation – jittering")

        iteration += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a step‑function under the given time budget and returns the
    list of heights that (hopefully) yields the smallest evaluation score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 4000 is a good compromise between expressiveness and speed
    n = 4000

    # initialise with the uniform distribution (Σa = 1)
    a0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Phase A – smoothed‑max Adam optimisation (≈ 70 % of the budget)
    # --------------------------------------------------------------
    phase_a_deadline = start_time + 0.70 * budget_s
    best_a, best_val = _adam_phase(
        a0, n, start_time, phase_a_deadline, a0.copy(), float('inf')
    )

    # --------------------------------------------------------------
    # Phase B – exact sub‑gradient descent (remaining time)
    # --------------------------------------------------------------
    best_a = _subgradient_phase(
        best_a, n, phase_a_deadline, deadline, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
