Propose a non‑negative step‑function (probability vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a multi‑stage optimisation:

1.  Several diverse initial vectors (uniform, arcsine, geometric decay,
    random Dirichlet draws and sparse “binary” patterns).
2.  Exact sub‑gradient descent on the true maximal convolution entry.
3.  Adam optimisation on a log‑sum‑exp (smooth‑max) surrogate with a
    continuation on the smoothing parameter α.
4.  Exponentiated‑gradient (multiplicative) refinement at high α.
5.  A final precise sub‑gradient polishing phase.
6.  A tiny random‑swap hill‑climber.

All updates are projected onto the probability simplex
{ x_i ≥ 0 , Σ x_i = 1 } so the objective is invariant under scaling.
The routine stops as soon as the target value 1.503 is reached or the
time budget (default 1000 s) expires.
"""

import time

# --------------------------------------------------------------
# Simplex projection (Euclidean)
# --------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Convolution utilities (FFT based)
# --------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of p using FFT size m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _gradient_from_weights(p: np.ndarray, p_fft: np.ndarray,
                          w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed max.
    w – soft‑max weights (length 2·n‑1) that sum to 1.
    """
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]
    # mapping indices back to gradient as derived in the original code
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad

    conv = _conv_self(p, m_fft)
    return float(2.0 * n * conv.max())

        # (B) Adam on a log‑sum‑exp smooth approximation of max
        # ------------------------------------------------------
        m1 = np.zeros_like(p)
        m2 = np.zeros_like(p)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        base_lr = 0.12
        alpha = 1.0
        alpha_mul = 1.002
        alpha_max = 20000.0

        for it in range(250_000):
            if time.time() >= deadline:
                break

            p_fft = np.fft.rfft(p, n=m_fft)
            conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

            # soft‑max weights (stable)
            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            w = exp_z / exp_z.sum()

            # gradient of the smooth max
            grad = _gradient_from_weights(p, p_fft, w, m_fft)

            # Adam update
            m1 = beta1 * m1 + (1.0 - beta1) * grad
            m2 = beta2 * m2 + (1.0 - beta2) * (grad ** 2)

            m_hat = m1 / (1.0 - beta1 ** (it + 1))
            v_hat = m2 / (1.0 - beta2 ** (it + 1))

            lr = base_lr / np.sqrt(alpha)          # shrink as α grows
            step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)

            p = p - step_vec
            p = _simplex_projection(p)

            # raise smoothing parameter gradually
            alpha = min(alpha * alpha_mul, alpha_max)

            # occasional exact evaluation
            if (it & 1023) == 0:
                cur_score = _exact_score(p, n, m_fft)
                if cur_score < best_score - 1e-12:
                    best_score, best_seq = cur_score, p.copy()
                    print(f"[Adam] iter {it:6d} – new best {best_score:.9f}")
                    if best_score <= 1.503:
                        print("Target reached during Adam phase.")
                        return np.clip(best_seq, 0.0, 1000.0).tolist()

        # ------------------------------------------------------
        # (C) Exponentiated‑gradient refinement at very high α
        # ------------------------------------------------------
        # Ensure α is huge so the smooth max is essentially the true max.
        alpha = max(alpha, 5000.0)
        lr_mult = 0.03  # multiplicative step size
        for it in range(120_000):
            if time.time() >= deadline:
                break
            p_fft = np.fft.rfft(p, n=m_fft)
            conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            w = exp_z / exp_z.sum()

            grad = _gradient_from_weights(p, p_fft, w, m_fft)

            # multiplicative (mirror descent) update
            p = p * np.exp(-lr_mult * grad)
            p = p / p.sum()   # automatically stays on simplex

            if (it & 1023) == 0:
                cur_score = _exact_score(p, n, m_fft)
                if cur_score < best_score - 1e-12:
                    best_score, best_seq = cur_score, p.copy()
                    print(f"[EG] iter {it:6d} – new best {best_score:.9f}")
                    if best_score <= 1.503:
                        print("Target reached during EG phase.")
                        return np.clip(best_seq, 0.0, 1000.0).tolist()

        # ------------------------------------------------------
        # (D) Final exact sub‑gradient polishing (small step)
        # ------------------------------------------------------
        step = 0.04
        decay = 0.9999
        min_step = 1e-9
        p_polish = best_seq.copy() if best_seq is not None else p.copy()
        for it in range(60_000):
            if time.time() >= deadline:
                break
            conv = _conv_self(p_polish, m_fft)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-12:
                best_score, best_seq = cur_score, p_polish.copy()
                print(f"[Polish] iter {it:5d} – new best {best_score:.9f}")
                if best_score <= 1.503:
                    print("Target reached during polishing phase.")
                    return np.clip(best_seq, 0.0, 1000.0).tolist()

            tol = 1e-13
            active = np.where(conv >= max_val - tol)[0]
            g = np.zeros_like(p_polish)
            for idx in active:
                if idx < n:
                    g[: idx + 1] += 2.0 * p_polish[idx::-1]
                else:
                    start = idx - n + 1
                    g[start:] += 2.0 * p_polish[n - 1 : start - 1 : -1]
            g /= float(len(active))

            p_polish = p_polish - step * g
            p_polish = _simplex_projection(p_polish)
            step = max(step * decay, min_step)

        # ------------------------------------------------------
        # (E) Tiny random‑swap hill‑climber around the best found
        # ------------------------------------------------------
        if best_seq is None:
            p_swap = p.copy()
            cur_score = _exact_score(p_swap, n, m_fft)
        else:
            p_swap = best_seq.copy()
            cur_score = best_score

        for _ in range(40_000):
            if time.time() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps_move = 0.001 * rng.random()
            if p_swap[i] <= eps_move:
                continue
            new = p_swap.copy()
            new[i] -= eps_move
            new[j] += eps_move
            new_score = _exact_score(new, n, m_fft)
            if new_score < cur_score - 1e-15:
                p_swap, cur_score = new, new_score
                if cur_score < best_score:
                    best_score, best_seq = cur_score, p_swap.copy()
                    print(f"[Swap] improvement → {best_score:.9f}")
                    if best_score <= 1.503:
                        print("Target reached during swap phase.")
                        return np.clip(best_seq, 0.0, 1000.0).tolist()

        # early exit if target already met
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _exact_score(best_seq, n, m_fft)

    print(f"\nFinal best score: {best_score:.9f}")
    # Clip to the bounds enforced by evaluate_sequence (0 ≤ x ≤ 1000)
    return np.clip(best_seq, 0.0, 1000.0).tolist()
