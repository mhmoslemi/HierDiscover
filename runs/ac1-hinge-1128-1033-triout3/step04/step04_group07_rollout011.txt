def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    try: n = len(height_sequence_1)
    except NameError: n = 1153
    deadline = time.time() + budget_s - 0.1

    # Precompute i array
    i = np.arange(n, dtype=float)

    best_obj = float('inf')
    best_p = None

    # function to evaluate given p (assume p normalized)
    def obj(p):
        conv = np.convolve(p, p)
        # max conv
        return 2.0 * n * np.max(conv)

    # Candidate generators:
    def arcsine(alpha):
        eps = 1e-12
        w = ((i+0.5)*(n-0.5-i) + eps)**(-alpha)
        return w / w.sum()

    def beta(beta):
        w = ((i+1)*(n-i) + 1e-12)**beta
        return w / w.sum()

    def cosine(pow):
        # shape: cos(pi * (i+0.5)/n) >=0 for i <= n/2, negative after; take abs.
        w = np.abs(np.cos(np.pi * (i+0.5) / n))**pow
        return w / w.sum()

    def lorentz(sigma):
        center = (n-1)/2.0
        w = 1.0 / (1.0 + ((i - center) / sigma)**2)
        return w / w.sum()

    def mix(alpha, lam):
        # lam in [0,1] weighting uniform + arcsine shape
        w_ar = arcsine(alpha)
        w = lam * np.full_like(w_ar, 1.0/n) + (1.0 - lam) * w_ar
        return w / w.sum()

    # Search loops
    # 1. param sweep for arcsine/exponent
    for alpha in np.linspace(0.3, 0.8, 101):
        if time.time() >= deadline: break
        p = arcsine(alpha)
        val = obj(p)
        if val < best_obj:
            best_obj = val; best_p = p.copy()
            print(f"Arcsine alpha={alpha:.3f} -> {best_obj:.6f}")

    # 2. param sweep for beta exponent (beta negative)
    for beta_val in np.linspace(-0.8, -0.2, 61):
        if time.time() >= deadline: break
        p = beta(beta_val)
        val = obj(p)
        if val < best_obj:
            best_obj = val; best_p = p.copy()
            print(f"Beta exponent={beta_val:.3f} -> {best_obj:.6f}")

    # 3. cosine pow
    for pow_val in np.linspace(0.5, 6.0, 56):
        if time.time() >= deadline: break
        p = cosine(pow_val)
        val = obj(p)
        if val < best_obj:
            best_obj = val; best_p = p.copy()
            print(f"Cosine pow={pow_val:.3f} -> {best_obj:.6f}")

    # 4. lorentz sigma
    for sigma in np.linspace(50, 600, 111):
        if time.time() >= deadline: break
        p = lorentz(sigma)
        val = obj(p)
        if val < best_obj:

"""
propose_candidate
-----------------

Search for a non‑negative step function (a probability vector) that
minimises

    2·n·max(conv_self(p)) / (sum(p)²)

where *conv_self* is the full self‑convolution of the vector.
The strategy is:

1. **Exploratory parametric sweep** – generate many “U‑shaped’’ candidates
   of the form p[i] ∝ ((i+δ)*(n‑1‑i+δ))⁻ᵅ   (δ ≈ 0.5, α≈0.5 is the classic
   arcsine shape).  A fine grid over (δ,α) plus a few simple alternatives
   (uniform, cosine‑shaped, mixtures) is evaluated with a fast FFT‑based
   convolution; the best candidate is kept.

2. **Local refinement** – starting from the best candidate, perform a
   projected sub‑gradient descent on the *exact* max‑convolution entry.
   The sub‑gradient for a given active index k is
   `g[j] = 2·p[k‑j]` (when the index is valid).  Steps are decreased
   geometrically and the iterate is projected onto the probability simplex.

3. **Fine‑grained stochastic hill‑climbing** – a tiny random‑swap
   optimiser (move a small amount of mass from one coordinate to another)
   is run until the time budget expires.  Any improvement updates the
   current best solution.

All updates respect the constraints required by ``evaluate_sequence``:
non‑negative entries that sum to at least 0.01 (the algorithm keeps a
probability vector, i.e. sum = 1).  The final vector is clipped to the
allowed range [0, 1000] before being returned.
"""

import numpy as np
import time

# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# --------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x | x_i ≥ 0, Σ x_i = 1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------
# Convolution utilities (FFT based)
# --------------------------------------------------------------
def conv_self_fft(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full (2·n‑1) self‑convolution of p using an FFT of size m_fft."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv_full = np.fft.irfft(p_fft * p_fft, n=m_fft)
    n = p.shape[0]
    return conv_full[: 2 * n - 1]

def subgradient_max(p: np.ndarray, n: int, m_fft: int):
    """
    Sub‑gradient of the *exact* maximal convolution entry.
    Returns (g, max_val) where g has the same shape as p.
    """
    conv = conv_self_fft(p, m_fft)
    max_val = conv.max()
    # tolerance for the active set – a tiny absolute tolerance works well
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(p)
    for idx in active:
        if idx < n:
            # contribution from the left part of the convolution
            g[: idx + 1] += 2.0 * p[idx::-1]
        else:
            # contribution from the right part (wrap‑around of the index)
            start = idx - n + 1
            g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
    g /= float(len(active))
    return g, max_val

# --------------------------------------------------------------
# Objective (exact, assumes sum(p)=1)
# --------------------------------------------------------------
def objective_exact(p: np.ndarray, n: int, m_fft: int) -> float:
    conv = conv_self_fft(p, m_fft)
    return float(2.0 * n * conv.max())

# Main entry point required by the evaluation framework.
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (length of the step function)
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 may be provided by the surrounding framework
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = len(base_seq)
    except Exception:
        # fallback length used in the reference code
        n = 1153

    # FFT size – next power of two that can hold the full convolution
    m_fft = 1 << ((2 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Search state
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.1   # safety margin
    best_score = float("inf")
    best_p = None

    i = np.arange(n, dtype=float)

    # ------------------------------------------------------------------
    # Helper to update the incumbent solution
    # ------------------------------------------------------------------
    def maybe_update(p_candidate: np.ndarray):
        nonlocal best_score, best_p
        cur = objective_exact(p_candidate, n, m_fft)
        if cur < best_score - 1e-12:
            best_score = cur
            best_p = p_candidate.copy()
            print(f"  New incumbent: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) PARAMETRIC SWEEP – families that are known to give low max‑conv
    # ------------------------------------------------------------------
    eps = 1e-12

    # a) Generalised arcsine family  p[i] ∝ ((i+δ)*(n‑1‑i+δ))⁻ᵅ
    deltas = np.linspace(0.3, 3.0, 28)            # offset (δ)
    alphas = np.linspace(0.30, 0.85, 56)          # exponent (α)
    for delta in deltas:
        if time.time() >= deadline:
            break
        # pre‑compute the distance factor (same for all alphas)
        base = (i + delta) * (n - 1 - i + delta) + eps
        for alpha in alphas:
            if time.time() >= deadline:
                break
            w = base ** (-alpha)
            p = w / w.sum()                     # normalise to sum = 1
            maybe_update(p)

    # b) Cosine‑power family  p[i] ∝ |cos(π·(i+0.5)/n)|ᵞ
    gammas = np.linspace(0.5, 6.0, 56)
    cos_vals = np.abs(np.cos(np.pi * (i + 0.5) / n))
    for gamma in gammas:
        if time.time() >= deadline:
            break
        w = cos_vals ** gamma + eps
        p = w / w.sum()
        maybe_update(p)

    # c) Uniform candidate (baseline)
    p_uniform = np.full(n, 1.0 / n)
    maybe_update(p_uniform)

    # d) Simple mixture of the best arcsine shape with uniform mass
    # (we reuse the best shape found so far once it exists)
    if best_p is not None:
        for lam in np.linspace(0.0, 0.4, 41):   # λ = weight of uniform part
            if time.time() >= deadline:
                break
            mix = lam * p_uniform + (1.0 - lam) * best_p
            mix = mix / mix.sum()
            maybe_update(mix)

    # e) Random Dirichlet draws for extra diversity
    while time.time() < deadline:
        # concentration parameter controls how “spiky’’ the draw is
        conc = rng.uniform(0.1, 2.5)
        gamma = rng.gamma(shape=conc, scale=1.0, size=n)
        p = gamma / gamma.sum()
        maybe_update(p)

    # ------------------------------------------------------------------
    # 2) LOCAL REFINEMENT – sub‑gradient descent on the true max entry
    # ------------------------------------------------------------------
    if best_p is not None:
        p = best_p.copy()
        step = 0.4
        decay = 0.9998
        min_step = 1e-8
        it = 0

            g, max_val = subgradient_max(p, n, m_fft)
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                print(f"  Subgrad refinement iter {it:5d} → {best_score:.6f}")
                if best_score <= 1.503:
                    break
            # gradient step & projection
            p = p - step * g
            p = simplex_projection(p)
            step = max(step * decay, min_step)
            it += 1

    # ------------------------------------------------------------------
    # 3) FINAL TINY RANDOM‑SWAP HILL‑CLIMBER
    # ------------------------------------------------------------------
    if best_p is not None:
        p_swap = best_p.copy()
        cur_score = objective_exact(p_swap, n, m_fft)
        attempts = 0
        while time.time() < deadline and attempts < 200_000:
            i_idx, j_idx = rng.integers(0, n, size=2)
            if i_idx == j_idx:

            if p_swap[i_idx] <= eps_move:
                continue

            cand[i_idx] -= eps_move
            cand[j_idx] += eps_move
            cand_score = objective_exact(cand, n, m_fft)
            if cand_score < cur_score - 1e-15:
                p_swap = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_p = p_swap.copy()
                    print(f"  Swap improvement → {best_score:.6f}")
                # optional: reset attempts counter to allow deeper search
                attempts = 0
            else:
                attempts += 1

    # ------------------------------------------------------------------
    # FALLBACK (should never happen)
    # ------------------------------------------------------------------
    if best_p is None:
        best_p = np.full(n, 1.0 / n)
        best_score = objective_exact(best_p, n, m_fft)

    print(f"\nFinal best score achieved: {best_score:.6f}")

    # Clip to the bounds enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    return np.clip(best_p, 0.0, 1000.0).tolist()
