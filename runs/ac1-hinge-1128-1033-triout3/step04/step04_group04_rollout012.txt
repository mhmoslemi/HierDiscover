```python
def conv_full_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx*fy, n=size)
    return conv[:n]
```

"""
Propose candidate step function using smooth‑max gradient descent + sub‑gradient polishing.

The optimizer works on the probability simplex (non‑negative entries summing to 1).
It starts from the best external seed if available (or from a uniform vector),
then minimizes a smoothed approximation of the objective

    f(a) = 2 * n * max_k (a*a)[k]

by a projected Adam update.  The smoothing uses a soft‑max with an increasing
temperature parameter α, which gradually turns the smooth objective into the
true max‑norm.  After the smooth phase a short sub‑gradient polishing (as in the
baseline) is performed.  The routine respects a 1000 s budget and returns the
best vector found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def full_convolution_fft_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT. Length = 2*len(a)-1."""
    n = a.shape[0]
    size = 2 * n                     # power-of‑two is not required, FFT works anyway
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y using real FFT (any lengths)."""
    n = len(x) + len(y) - 1
    # next power of two for speed
    size = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n]


def score_vector(a: np.ndarray) -> float:
    """Exact value returned by evaluate_sequence for a vector with sum ≈ 1."""
    n = a.shape[0]
    conv = full_convolution_fft_self(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (used for final polishing)
# ----------------------------------------------------------------------
def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each (i,j) pair contributes 2*a[j] to ∂/∂a[i]
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a good non‑negative height sequence within the time budget.
    Returns a plain Python list (∑a ≈ 1).  The algorithm uses a smooth‑max
    gradient descent followed by a short sub‑gradient polishing.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a little buffer

    # ---- 1. initialise -------------------------------------------------
    # Use external seed if available, otherwise a uniform vector.
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        if external.sum() == 0.0:
            raise ValueError
        a = external / external.sum()
        n = len(a)
        print(f"Using external seed, length {n}")
    except Exception:
        # default length – a balance between expressiveness and speed
        n = 6000
        a = np.ones(n, dtype=float) / n
        print(f"No external seed – start with uniform vector, length {n}")

    # ensure exact simplex compliance
    a = simplex_projection(a)
    a /= a.sum()

    best_vec = a.copy()
    best_score = score_vector(best_vec)
    print(f"Initial score = {best_score:.6f}")

    # ---- 2. smooth‑max gradient descent (Adam) ------------------------
    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule of smoothness parameters (α) – start smooth, finish sharp
    alphas = [1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
    # base learning rate (tuned empirically)
    base_lr = 0.30

    for alpha in alphas:
        # learning rate adapted to α – larger α -> smaller steps
        lr = base_lr / (np.sqrt(alpha) + 1.0)
        max_iter_alpha = 2500          # generous but will stop early if out of time
        print(f"\n--- smooth‑max phase α = {alpha:.1f}, lr = {lr:.4f} ---")
        for _ in range(max_iter_alpha):
            if time.time() > deadline:
                break

            t += 1
            # self‑convolution
            conv = full_convolution_fft_self(a)
            max_c = conv.max()

            # soft‑max weights (stable by subtracting the max)
            w = np.exp(alpha * (conv - max_c))
            p = w / w.sum()

            # gradient = 2 * (p * a_rev) convolution slice
            a_rev = a[::-1]
            conv2 = full_convolution_fft(p, a_rev)          # length = 3n‑2
            grad = 2.0 * conv2[n - 1 : n - 1 + n]           # keep n entries

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # project back onto the simplex
            a = simplex_projection(a)
            a /= a.sum()

            # evaluate (fast exact score because sum≈1)
            cur_score = score_vector(a)
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.copy()
                print(f"Iter {t:6d} α={alpha:5.1f}: new best = {best_score:.6f}")

        if time.time() > deadline:
            break

    # ---- 3. sub‑gradient polishing (as in the baseline) --------------
    print("\n--- sub‑gradient polishing ---")
    current = best_vec.copy()
    current_score = best_score
    step = 0.15
    step_decay = 0.9995
    it = 0
    last_improve = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft_self(current)
        grad = compute_subgradient(current, conv)

        cand = current - step * grad
        cand = simplex_projection(np.maximum(cand, 0.0))
        cand /= cand.sum()
        cand_score = score_vector(cand)

        if cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                last_improve = it
                print(f"Polish it {it:6d}: new best = {best_score:.6f}")

        step *= step_decay
        if step < 1e-7:
            break
        if it - last_improve > 3000:
            # small random re‑heat
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current /= current.sum()
            current_score = score_vector(current)
            step = 0.15
            last_improve = it
            print(f"Polish it {it:6d}: re‑heat")

    # ---- 4. final output ------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_vec)})")
    return best_vec.tolist()
