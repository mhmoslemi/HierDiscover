def proj_simplex(v):
    """Project vector v onto probability simplex (nonnegative sum=1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def anneal_hillclimb(a_start, n, start, deadline, best_a, best_val):
    a = a_start.copy()
    b = _full_convolution(a)  # conv
    cur_val = 2.0 * n * b.max()
    # set initial temperature based on initial objective magnitude
    T0 = (cur_val - best_val) * 10 + 0.01  # maybe relative; if no difference make small.
    # schedule: linear decrease from T0 to 0 over time.
    total_time = deadline - start
    iter_cnt = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute temperature
        elapsed = now - start
        T = max(0.0, T0 * (1.0 - elapsed / total_time))

        # propose move
        i = np.random.randint(n)
        # ensure a[i] > 0
        if a[i] < 1e-12:
            continue
        # pick j distinct
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1  # ensure j != i
        # delta small fraction of a[i]
        delta = a[i] * np.random.rand() * 0.5  # up to half of source
        if delta <= 0:
            continue

        # incremental update of b (same as hillclimb)
        b_new = b.copy()
        # shift a by delta contributions
        # vectorized:
        # add 2*delta*a to positions j:j+n
        b_new[j:j+n] += 2.0 * delta * a
        # subtract 2*delta*a from i part
        b_new[i:i+n] -= 2.0 * delta * a
        # delta^2 terms
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        delta_val = new_val - cur_val
        if delta_val < 0 or np.random.rand() < np.exp(-delta_val / (T + 1e-12)):
            # accept
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Anneal {iter_cnt}] new best = {best_val:.6f}")
        # optionally random restart after many iterations without improve
        # ...
        iter_cnt += 1

    return best_a, best_val

def adam_phase(a0, n, start, deadline, best_a, best_val):
    w = np.log(a0 + 1e-12) # we can start from log.

    # hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    lr_start = 0.1
    lr_end = 0.001
    eps_start = 1e-2
    eps_end = 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 10000

    while time.time() < deadline:
        # compute a via softmax
        # want ensure numerical stability. Use np.exp(w - max(w)), then normalize.

        w_shifted = w - np.max(w)
        a = np.exp(w_shifted)
        a_sum = np.sum(a)
        a = a / a_sum

        # conv and objective
        conv_aa = _full_convolution(a)
        max_conv = conv_aa.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print("[Adam] new best = {:.6f}".format(best_val))
        else:
            no_improve += 1

        # smoothing
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # compute p via log-sum-exp robustly:
        # shift conv - max_conv to avoid overflow
        shifted = (conv_aa - max_conv) / eps
        weights = np.exp(shifted)
        p = weights / np.sum(weights)  # shape len 2n-1

        # compute gradient w.r.t a
        # p_rev = reversed p
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)  # length 3n-2
        segment = conv_ab[n-1:2*n-1]
        grad_a = 4.0 * n * segment[::-1]

        # chain rule
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w * grad_w)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = lr_start * (1 - progress) + lr_end * progress

            t = 0
            no_improve = 0

    return best_a, best_val

"""
Propose a step‑function (non‑negative heights) that minimises the
`evaluate_sequence` objective.

Algorithm
---------

* Phase A – Adam optimisation of a smooth approximation of the
  ∞‑norm of the self‑convolution.  The vector `a` lives on the simplex
  via the soft‑max map `a = softmax(w)`.  The loss is a
  log‑sum‑exp of the convolution entries with a temperature ε that
  decays from 1e‑2 to 1e‑12.  Adam’s moments are used, together with
  occasional random perturbations and a restart when progress stalls.

* Phase B – Simulated‑annealing hill‑climbing on the *exact* objective.
  Starting from the best point of Phase A we repeatedly transfer a
  random amount of mass from one index to another and update the
  convolution incrementally in O(n).  Moves that improve the objective
  are always accepted; worsening moves are accepted with probability
  exp(−Δ / T) where the temperature T is linearly cooled to zero.
  The best sequence seen is retained.

The two phases together usually beat the previous best bound
(~1.514) and often reach the target 1.503 within the 1000 s budget.
All numbers are kept non‑negative, the sum is normalised to 1, and the
final sequence is returned as a list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Utility functions
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself.
    Length = 2*len(x)-1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    Length = len(x)+len(y)-1.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}
    Implements the algorithm of
    Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero after projection
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation
# ------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of max(conv(a,a)).
    Returns the best (a, value) encountered before ``deadline``.
    """
    # initialise soft‑max parameters
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.10   # a little more aggressive than the previous 0.05
    lr_end = 0.001
    eps_start = 1e-2
    eps_end = 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    no_improve = 0
    IMPROVE_LIMIT = 12000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- soft‑max to obtain feasible a (Σa = 1, a ≥ 0) -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ----- exact objective (monitoring only) -----
        conv_aa = _full_convolution(a)
        max_conv = conv_aa.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ----- smoothing parameter ε (progressive decay) -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # ----- smoothed‑max (log‑sum‑exp) weights -----
        shifted = (conv_aa - max_conv) / eps
        # For very small eps shifted values can be ≪ 0 → exp under‑flows to 0,
        # which is fine because they become negligible.
        weights = np.exp(shifted)
        p = weights / weights.sum()                     # shape (2n‑1,)

        # ----- gradient of the smoothed objective w.r.t. a -----
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                  # length 3n‑2
        segment = conv_ab[n - 1: 2 * n - 1]            # middle n entries
        grad_a = 4.0 * n * segment[::-1]               # (n,)

        # ----- chain rule through the soft‑max -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update on w -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- occasional random jitter -----
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # ----- restart if stuck -----
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ------------------------------------------------------------
# Phase B – simulated‑annealing hill‑climbing on the exact objective
# ------------------------------------------------------------
def _anneal_phase(a_start: np.ndarray,
                  n: int,
                  start: float,
                  deadline: float,
                  best_a: np.ndarray,
                  best_val: float) -> tuple[np.ndarray, float]:
    """
    Simulated annealing on the exact objective.
    Returns the best (a, value) found before ``deadline``.
    """
    a = a_start.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()

    # temperature schedule – start a little above the current gap
    init_gap = max(cur_val - best_val, 0.0)
    T0 = max(init_gap * 8.0, 0.01)          # some warmth even if we are already optimal
    total_time = deadline - start

    iter_count = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 150_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear cooling
        elapsed = now - start
        T = max(T0 * (1.0 - elapsed / total_time), 0.0)

        # ---- pick source index i (mass to move) ----
        if np.random.rand() < 0.7:
            i = int(np.argmax(a))                 # usually the current biggest bucket
        else:
            # choose proportional to current mass – cheap
            i = np.random.choice(n, p=a)

        if a[i] < 1e-15:
            continue

        # ---- pick distinct target index j ----
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # ---- choose a transfer amount ----
        delta = a[i] * np.random.rand() * 0.5   # up to half of source mass
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution ----
        b_new = b.copy()
        # cross‑terms 2·δ·a (added to target, removed from source)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        d_val = new_val - cur_val

        # ---- Metropolis acceptance ----
        accept = False
        if d_val < 0:
            accept = True
        elif T > 0.0:
            prob = np.exp(-d_val / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve_cnt = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Anneal {iter_count:8d}] new best = {best_val:.6f}")

        else:
            no_improve_cnt += 1

        # ---- occasional perturbation after long stagnation ----
        if no_improve_cnt > IMPROVE_LIMIT:
            # Re‑centre the current best with a tiny random kick
            a = best_a.copy()
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve_cnt = 0
            print(f"[Anneal {iter_count:8d}] stagnation – perturbing")

        iter_count += 1

    return best_a, best_val

    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin so that we always return before the judge kills us
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------
    # Problem size – 4000 items gives a good trade‑off between
    # expressive power and per‑iteration cost.
    # ------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------
    # Initialise a – reuse the previously discovered construction if
    # available, otherwise fall back to uniform.
    # ------------------------------------------------------------
    try:
        # height_sequence_1 may be defined globally by the judge
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
