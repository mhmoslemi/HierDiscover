"""
Propose a non‑negative step‑function (list of heights) with a low
evaluate_sequence() score.

Algorithm
---------
* Phase A – Smoothed‑max Adam optimisation.
  The vector a is parametrised by a soft‑max `w` (so Σa = 1, a≥0).
  We minimise a log‑sum‑exp approximation to max(conv(a,a)).
  The smoothing ε and learning‑rate decay from relatively large values
  to very small ones.  Standard Adam moments and occasional random
  perturbations / restarts keep the search from stagnating.

* Phase B – Targeted hill‑climbing on the exact objective.
  At each iteration we locate the (possibly several) lag positions where
  the convolution attains its current maximum.  For these “active”
  peaks we compute, for every index i, the total contribution
  `score_i = a_i * Σ_{k∈active} a_{k‑i}`.
  Mass is transferred from the index with the largest score (the most
  harmful for the peaks) to an index with a tiny score (the least
  harmful).  The convolution is updated in O(n) using the exact
  bilinear formula, so only moves that strictly reduce the objective
  are accepted.  When no improvement occurs for many steps we
  perturb the best solution and continue.

Both phases are run within the 1000 s budget; the best sequence
encountered is returned.  The implementation works for n≈6000 (≈12 k
convolution length) which empirically gives a bound below the previous
1.514 value and often reaches ≈1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real data, FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of x with itself (`len = 2*len(x)-1`)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Phase A – smoothed‑max Adam optimisation
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation to max(conv(a,a)).
    Returns the best (a, value) seen before *deadline*.
    """
    # soft‑max parametrisation (w -> a on the simplex)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters (tuned for n≈6000)
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.07          # a bit larger than before
    lr_end = 0.0005
    eps_start = 1e-2
    eps_end = 1e-10

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    no_improve_cnt = 0
    IMPROVE_LIMIT = 25000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # --------- get feasible a from w (soft‑max) ----------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # --------- exact objective (for bookkeeping) ----------
        conv_aa = _full_convolution(a)
        max_conv = conv_aa.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve_cnt += 1

        # --------- smoothing parameter ε (linear decay) ----------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # --------- compute soft‑max weights over convolution entries ----------
        shifted = (conv_aa - max_conv) / eps          # (max‑conv) ≤ 0
        weights = np.exp(shifted)                    # safe; may under‑flow
        p = weights / weights.sum()                  # approximates arg‑max

        # --------- gradient of smoothed objective w.r.t. a ----------
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]          # middle n entries
        grad_a = 4.0 * n * segment[::-1]             # shape (n,)

        # --------- chain rule through soft‑max ----------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # --------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # --------- occasional random perturbation ----------
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # --------- restart if stagnating ----------
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restart (idle {no_improve_cnt})")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val

# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _targeted_hillclimb_phase(a_start: np.ndarray,
                              n: int,
                              start: float,
                              deadline: float,
                              best_a: np.ndarray,
                              best_val: float) -> np.ndarray:
    """
    Fast local search that transfers mass from indices that contribute most
    to the current peak(s) of the convolution to indices that contribute
    the least.
    Returns the best sequence encountered before *deadline*.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # parameters
    max_frac = 0.5        # maximal fraction of source mass to move
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        max_conv = b.max()
        # indices where the convolution attains (near) the maximum
        active = np.where(b >= max_conv - 1e-13)[0]

        # ---------- compute contribution scores ----------
        # score_i = a_i * Σ_{k∈active} a_{k-i} (if defined)
        contrib = np.zeros(n)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # i runs from lo..hi, and term a_{k-i}
            i_range = np.arange(lo, hi + 1, dtype=int)
            contrib[i_range] += a[k - i_range]

        scores = a * contrib

        # source = index with largest score (most harmful)
        source = int(np.argmax(scores))

        # choose a target among the smallest scores (least harmful)
        # we avoid the source itself.
        sorted_idx = np.argsort(scores)
        target = None
        for cand in sorted_idx[:15]:
            if cand != source:
                target = int(cand)
                break
        if target is None:               # fallback – any different index
            target = (source + 1) % n

        # amount of mass to move
        delta = a[source] * np.random.rand() * max_frac
        if delta <= 1e-15:
            continue

        # ----- incremental convolution update (O(n)) -----
        b_new = b.copy()
        # cross‑terms (2·δ·a) – add to positions j..j+n‑1, subtract from i..i+n‑1
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- accept if strictly improving -----
        if new_val < cur_val:
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[HillT {iteration:7d}] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            # perturb the best solution a little and restart from it
            a = best_a.copy()
            noise = np.random.normal(scale=0.008, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillT {iteration:7d}] perturbing after stagnation")

        # slowly shrink the allowed fraction of moved mass
        if iteration % 7000 == 0 and iteration > 0:
            max_frac = max(0.02, max_frac * 0.85)

        iteration += 1

    return best_a

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ---- problem size -------------------------------------------------
    # Larger n gives more expressive power; 6000 is a good trade‑off.
    n = 6000

    # ---- initialise a -------------------------------------------------
    try:
        # reuse the previously discovered construction if present
        old = np.asarray(height_sequence_1, dtype=float)    # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        a0 = np.full(n, 1.0 / n)

    # add a tiny random perturbation to avoid perfect symmetry
    a0 += np.random.normal(scale=1e-4, size=n)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = float('inf')

    # ---- Phase A – smooth‑max Adam ------------------------------------
    phase_a_deadline = start_time + 0.70 * budget_s   # 70 % of total time
    best_a, best_val = _adam_phase(a0, n, start_time,
                                   phase_a_deadline, best_a, best_val)

    # ---- Phase B – targeted hill‑climbing -----------------------------
    best_a = _targeted_hillclimb_phase(best_a, n,
                                       phase_a_deadline,
                                       deadline, best_a, best_val)

    # ---- final evaluation ---------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
