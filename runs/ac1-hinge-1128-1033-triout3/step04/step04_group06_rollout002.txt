"""
Subgradient Mirror Descent for the max‑auto‑convolution objective.

The evaluation function is

    f(a) = 2 * n * max_k (a * a)[k] / (sum_i a_i)^2 .

Because the objective is convex (maximum of convex quadratics) we can
optimise it directly on the probability simplex (a_i ≥ 0, Σ a_i = 1)
using an exponentiated‑gradient (mirror‑descent) step with the *exact*
sub‑gradient of the active convolution entry (or the average of all
active entries).  The algorithm

* starts from the best known construction if it exists,
* repeatedly evaluates the full auto‑convolution using a real FFT,
* selects the indices where the convolution attains its maximum,
* builds a weight vector supported on those indices,
* obtains the sub‑gradient via a single convolution (the same formula
  used in the previous baseline),
* performs an exponentiated update `a_i ← a_i·exp(-η·g_i)`,
* renormalises to the simplex,
* occasionally adds a small random “jitter’’ to escape plateaus,
* keeps the best sequence seen so far.

The step size η is linearly annealed from a relatively large value to a
tiny one, while the whole loop runs until the time‑budget expires.
The returned sequence is the best one encountered during the run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Clip tiny negative values caused by numerical noise.
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient from a weight vector w (same formula as the baseline code)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i}
    via a single convolution: reverse w and convolve with a,
    then take the appropriate slice.
    """
    n = a.shape[0]
    # reverse weight vector (full length = 2n‑1)
    u = w[::-1]
    # full convolution u * a → length = 3n‑2
    conv = _full_convolution_fft(u, a)
    # Indices that correspond to the gradient entries:
    # start = 2n‑2, then step backwards
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g
