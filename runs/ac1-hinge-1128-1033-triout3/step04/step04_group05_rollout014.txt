import cvxpy as cp

def solve_lp(a, M):
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    # using constant a
    conv_expr = cp.conv(cp.Constant(a), b)  # shape 2n-1
    constraints = [conv_expr <= M]
    # maybe also ensure sum(b) >= 0? Not needed.
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None
    return b.value
```

"""
Propose a step‑function sequence using an alternating linear‑program / mixing scheme.

The objective of ``evaluate_sequence`` (for a sum‑normalised sequence) is
    2·n·max_k (a * a)[k] .
We start from a random Dirichlet vector (ensuring ∑a = 1, a_i ≥ 0) and repeatedly
perform the following update:

1.  Compute M = max_k (a * a)[k].
2.  Solve the linear program
        maximise   1ᵀ b
        subject to b ≥ 0,
                   (a * b)[k] ≤ M   for all k = 0 … 2·n‑2 .
    The convolution (a * b) is linear in the optimisation variable b, so this
    is a genuine LP.  The solution b tends to concentrate mass in positions
    that do **not** increase the current maximum convolution.
3.  Normalise b to obtain a candidate distribution g (∑g = 1).
4.  Mix the old and the new distribution: a ← (1‑t)·a + t·g.
    Several candidate mixing factors t∈{0.1,0.25,0.5,0.75} are tested and the
    one giving the smallest objective is kept.
5.  Occasionally add a small random jitter to escape plateaux.

The LP is built once using CVXPY parameters (a and M) and a variable `b`.
Only the parameter values are updated each iteration, allowing warm‑starts
and keeping the per‑iteration cost low.  Convolutions are evaluated with an
FFT‑based routine for speed.  The algorithm runs until the provided time
budget expires and returns the best sequence observed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via FFT (real‑valued, length 2n‑1)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray) -> np.ndarray:
    """
    Compute the full linear convolution x*x using a real FFT.
    Returns a non‑negative array of length 2·len(x)‑1.
    """
    n = x.shape[0]
    total = 2 * n - 1
    # next power of two for FFT speed
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:total]
    # Numerical noise may produce tiny negatives – clip them
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Linear‑program solver (uses CVXPY with a constant‑time‑varying LP)
# ----------------------------------------------------------------------
def _solve_lp(a_vec: np.ndarray, bound: float, b_var, a_param, m_param, prob):
    """
    Solve:
        maximise   sum(b)
        subject to b ≥ 0,
                   (a_vec * b)[k] ≤ bound   ∀k.
    The CVXPY problem ``prob`` is built once; only the parameter values are
    updated on each call.  Returns the optimal b as a NumPy array or ``None``
    if the solver failed.
    """
    a_param.value = a_vec
    m_param.value = bound
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=5000, abstol=1e-7, reltol=1e-7, feastol=1e-7)
    except Exception:                      # pragma: no cover
        return None

    if b_var.value is None:
        return None
    # Clip tiny negatives due to numerical error
    b_opt = np.maximum(b_var.value, 0.0)
    return b_opt
