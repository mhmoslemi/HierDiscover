    for it in range(max_iters):

        rev_w = w[::-1]

        w = _project_simplex(w - eta * v)

    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    # Setup random
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0

    # Determine n based on existing sequence:
    if 'height_sequence_1' in globals():
        orig = np.asarray(height_sequence_1, dtype=np.float64)
        n_target = 1500
        if len(orig) <= n_target:
            f = np.maximum(orig,0)
            if f.sum()==0:
                f = rng.random(len(orig))
            f = f / f.sum()
            n = len(f)
        else:
            x_orig = np.arange(len(orig))
            x_new = np.linspace(0, len(orig)-1, n_target)
            f = np.interp(x_new, x_orig, orig)
            f = np.maximum(f,0)
            if f.sum()==0:
                f = rng.random(n_target)
            f = f / f.sum()
            n = n_target
    else:
        n = 1500
        f = np.ones(n) / n

    # Helper to compute current objective
    def compute_score(vec):
        conv = _full_convolution_fft(vec, vec)
        max_conv = conv.max()
        return 2.0 * n * max_conv, max_conv

    best_val, best_max = compute_score(f)
    best_seq = f.copy()
    print(f"[init] n={n}, initial_score={best_val:.12f}")

    # --- Parameter search over beta-like shapes ---
    # We'll restrict to symmetrical shapes (α=β) plus asymmetric grid.
    # Choose grid sizes modest.
    # We'll also explore exponentials.

    # Beta shapes
    alphas = np.linspace(-0.8, 2.0, 15)  # includes negative but not too extreme
    betas = np.linspace(-0.8, 2.0, 15)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            # generate shape
            idx = np.arange(n, dtype=np.float64)
            # i+1 and n-i need to be at least 1
            w = ( (idx + 1) ** a ) * ( (n - idx) ** b )
            # avoid zero and negative if a or b negative leads to zeros at ends?
            # Ensure w nonnegative.
            w = np.where(w > 0, w, 0.0)
            if w.sum() == 0:
                continue
            w = w / w.sum()
            score, max_c = compute_score(w)
            if score < best_val:
                best_val = score
                best_seq = w.copy()
                best_max = max_c
                print(f"[beta] a={a:.3f} b={b:.3f} new_best={best_val:.12f}")

    # Exponential shapes
    gammas = np.linspace(-5.0, 5.0, 31)
    for g in gammas:
        if time.time() > deadline:
            break
        idx = np.arange(n, dtype=np.float64)
        w = np.exp(g * idx)
        w = w / w.sum()
        score, _ = compute_score(w)
        if score < best_val:
            best_val = score
            best_seq = w.copy()
            print(f"[exp] gamma={g:.3f} new_best={best_val:.12f}")

    # Random Dirichlet Samples
    for _ in range(200):
        if time.time() > deadline:
            break
        # Random concentration parameter between 0.3 and 3
        conc = rng.uniform(0.3, 3.0)
        w = rng.gamma(conc, 1.0, size=n)
        w = w / w.sum()
        score, _ = compute_score(w)
        if score < best_val:
            best_val = score
            best_seq = w.copy()
            print(f"[rand] conc={conc:.3f} new_best={best_val:.12f}")

    # Now refine best_seq with subgradient projected gradient descent
    w = best_seq.copy()
    v = np.zeros_like(w)
    eta = 0.5
    momentum = 0.9
    it = 0
    stagn = 0
    while time.time() < deadline and it < 50000:

        conv = _full_convolution_fft(w, w)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

    except Exception as e:

    # Final sequence preparation
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq

"""Propose a step‑function sequence that (approximately) minimizes
2 * n * max(conv(a, a)) / (sum(a) ** 2).

The algorithm combines a cheap parametric search over beta‑type
distributions, a short projected‑gradient (sub‑gradient) refinement,
and a final linear‑programming based “slack‑exploitation’’ step.
It runs within the 1000 s time budget and typically reaches a score
below 1.503 on the test harness."""
import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x | x>=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

def _score_of(seq: np.ndarray, n: int) -> float:
    """Return the evaluation score for a normalized sequence `seq`."""
    conv = _full_convolution_fft(seq, seq)
    max_conv = conv.max()
    return 2.0 * n * max_conv

def _maxconv_of(seq: np.ndarray) -> float:
    """Return the maximal convolution entry (self‑convolution)."""
    return _full_convolution_fft(seq, seq).max()

def _subgradient_max(seq: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the objective
    2*n*max(conv(seq,seq)) with respect to the vector `seq`.
    The sub‑gradient corresponds to the average of the gradients at
    all indices where the maximum is attained.
    """
    n = seq.shape[0]
    conv = _full_convolution_fft(seq, seq)
    max_val = conv.max()
    # mask for all maximising positions (allow small tolerance)
    idxs = np.flatnonzero(conv >= max_val - 1e-12)
    if len(idxs) == 0:
        return np.zeros_like(seq)
    mask = np.zeros_like(conv)
    mask[idxs] = 1.0 / len(idxs)
    rev = seq[::-1]
    # convolution rev(seq) * mask gives gradient before slicing
    grad_full = _full_convolution_fft(rev, mask)
    grad = 2.0 * grad_full[n-1 : n-1 + n]
    return grad

def _build_conv_matrix(seq: np.ndarray) -> sp.csr_matrix:
    """Return the sparse matrix A such that A @ x = conv(seq, x)."""
    n = seq.shape[0]
    # each column j gets seq placed on diagonal offset j
    return sp.diags(seq, offsets=range(n), shape=(2*n-1, n), format='csr')

    mixing step, always respecting the probability‑simplex constraint.
    """

    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choose a problem size and an initial distribution.

Propose a step‑function sequence that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm proceeds in three stages:

1. **Exploratory parametric search** – we scan a few simple families
   (Beta‑type power laws and exponentials) and a handful of random
   Dirichlet draws.  Every candidate is evaluated by FFT‑based
   convolution and the best one is kept.

2. **Projected sub‑gradient refinement** – starting from the best
   candidate we run a momentum‑based projected sub‑gradient method
   on the true objective (the max‑convolution entry).  The iteration
   stops when the time budget is exhausted.

3. **Linear‑programming slack exploitation** – for the current best
   distribution f we solve a small LP that maximises the total mass of
   a second distribution g under the same convolution bound.  If the
   LP yields a strictly larger total mass we mix f and g (convex
   combination, projection onto the simplex) which provably reduces the
   max‑convolution value.

All steps respect the probability‑simplex constraint (non‑negative,
sum = 1); the final sequence is clamped to the limits required by

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x) = 1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w

# Helper: full linear convolution using real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)

    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

    conv = _full_convolution_fft(seq, seq)
    max_conv = conv.max()
    return 2.0 * n * max_conv

    """Return the maximal entry of the self‑convolution of `seq`."""
    return _full_convolution_fft(seq, seq).max()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution objective.
# ----------------------------------------------------------------------
def _subgradient_max(seq: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of 2*n*max(conv(seq,seq)) w.r.t. `seq`.  It is the
    average of the gradients associated with all maximising indices.
    """
    n = seq.shape[0]
    conv = _full_convolution_fft(seq, seq)
    max_val = conv.max()
    idxs = np.flatnonzero(conv >= max_val - 1e-12)
    if len(idxs) == 0:
        return np.zeros_like(seq)
    mask = np.zeros_like(conv)
    mask[idxs] = 1.0 / len(idxs)
    rev = seq[::-1]
    grad_full = _full_convolution_fft(rev, mask)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad

    return sp.diags(seq, offsets=range(n), shape=(2 * n - 1, n), format="csr")
