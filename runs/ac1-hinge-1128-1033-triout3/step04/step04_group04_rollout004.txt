def exponential_gradient_polish(current, deadline):
    step_size = eta_start = 0.5
    decay = 0.999
    it = 0
    best_local = current.copy()
    best_local_score = score_vector(current)
    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft(current)
        grad = compute_subgradient(current, conv)
        eta = eta_start * decay**it
        exp_factor = np.exp(-eta * grad)
        # avoid overflow/underflow:
        exp_factor = np.clip(exp_factor, 1e-20, 1e20)
        cand = current * exp_factor
        cand_sum = cand.sum()
        if cand_sum == 0:
            cand = np.full_like(current, 1.0/current.size)
            cand_sum = 1.0
        cand = cand / cand_sum
        cand_score = score_vector(cand)
        if cand_score < best_local_score:
            best_local_score = cand_score
            best_local = cand.copy()
            # also update global best
        # accept if improved else keep old? We'll keep best overall but may also continue from cand (maybe accepted if improvement else keep)
        if cand_score < best_local_score:
            current = cand
        else:
            # optionally occasionally accept with probability
            # but for now keep current unchanged
            pass
        # if step becomes too small, break
        if eta < 1e-5:
            break

"""
propose_candidate
-----------------
Find a non‑negative step‑function (vector of heights) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The algorithm consists of

1.  A deterministic sweep over several parametric families (U‑shaped,
    exponential, shifted power, and mixtures with the uniform distribution)
    for a handful of lengths.  This gives a strong initial candidate.

2.  A Differential‑Evolution search on the probability simplex (population
    size ≈ 40) to explore the space globally.

3.  Two polishing stages:
      * exponentiated‑gradient (multiplicative) updates using the sub‑gradient
        of the max‑convolution,
      * a simple mass‑redistribution hill‑climber that moves a small amount of
        probability from indices with large gradient to indices with small
        gradient.

All phases respect the 1000 s time budget; intermediate progress and scores
are printed.  The returned list sums to 1 (the scale is irrelevant for the
objective because it is homogeneous).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w>=0, sum w = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n                     # zero‑padding length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a vector with sum == 1."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices where the convolution attains (or almost attains) its
    maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each (i,j) pair contributes 2*a[j] to ∂/∂a[i]
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

# Parametric families (deterministic sweep)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+1)^{-p}   (symmetric, edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{-p}   (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))   (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def shifted_power_distribution(n: int, p: float, shift: float) -> np.ndarray:
    """a[i] ∝ (i+1+shift)^{-p} + (n-i+shift)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0 + shift) ** (-p) + (n - i + shift) ** (-p)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def mix_uniform_min(n: int, p: float, w: float) -> np.ndarray:
    """Mixture (1‑w)*uniform + w*u_min(p).  w∈[0,1]."""
    uniform = np.full(n, 1.0 / n)
    umin = u_min_distribution(n, p)
    a = (1.0 - w) * uniform + w * umin
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Resampling helper (used for a cheap length‑scaling try after polishing)
# ----------------------------------------------------------------------
def resample_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex vector onto a new length."""
    old_n = v.shape[0]
    if new_n == old_n:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    v_new = np.interp(x_new, x_old, v)
    v_new = np.maximum(v_new, 0.0)
    v_new /= v_new.sum()
    return v_new


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a high‑quality step‑function under the given time budget.
    Returns a plain Python list (∑a ≈ 1, all entries ≥ 0).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin so we can always return something
    deadline = start_time + max(budget_s - 5.0, 10.0)

    best_vec = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 1. Deterministic sweep over several families
    # --------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 10000]
    p_vals = np.linspace(0.3, 4.0, 15)          # exponents
    lam_vals = np.logspace(-4, -0.3, 12)       # exponential rates
    shift_vals = [0.0, 0.5, 1.0, 2.0, 5.0]     # shift for power family
    w_vals = np.linspace(0.0, 1.0, 6)          # mixture weight

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break
        # uniform baseline
        uniform = np.full(n, 1.0 / n)
        sc = score_vector(uniform)
        if sc < best_score:
            best_score, best_vec = sc, uniform.copy()
            print(f" new best (uniform)      n={n:5d}  score={sc:.6f}")

        # U‑shaped families
        for p in p_vals:
            if time.time() > deadline:
                break
            a = u_min_distribution(n, p)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f" new best (u_min)        n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f" new best (u_prod)       n={n:5d} p={p:4.2f}  score={sc:.6f}")

            # mixtures with uniform
            for w in w_vals:
                a = mix_uniform_min(n, p, w)
                sc = score_vector(a)
                if sc < best_score:
                    best_score, best_vec = sc, a.copy()
                    print(f" new best (mix)          n={n:5d} p={p:4.2f} w={w:3.1f}  score={sc:.6f}")

        # exponential family
        for lam in lam_vals:
            if time.time() > deadline:
                break
            a = exp_distribution(n, lam)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec = sc, a.copy()
                print(f" new best (exp)          n={n:5d} λ={lam:.5e}  score={sc:.6f}")

        # shifted power family
        for p in p_vals:
            if time.time() > deadline:
                break
            for shift in shift_vals:
                a = shifted_power_distribution(n, p, shift)
                sc = score_vector(a)
                if sc < best_score:
                    best_score, best_vec = sc, a.copy()
                    print(f" new best (shift)        n={n:5d} p={p:4.2f} s={shift:4.1f}  score={sc:.6f}")

    # --------------------------------------------------------------
    # 2. Use external seed if present
    # --------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        external = np.maximum(external, 0.0)
        s = external.sum()
        if s > 0.0:
            external /= s
            sc = score_vector(external)
            if sc < best_score:
                best_score, best_vec = sc, external.copy()
                print(f" external seed used, length {len(external)} score={sc:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # 3. Fallback (should never happen)
    # --------------------------------------------------------------
    if best_vec is None:
        n = 6000
        best_vec = np.random.rand(n)
        best_vec /= best_vec.sum()
        best_score = score_vector(best_vec)
        print(f" fallback random seed, n={n} score={best_score:.6f}")

    n = best_vec.shape[0]
    print(f"\n=== Selected problem size: n = {n} ===")
    print(f" Seed score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Differential Evolution on the simplex
    # --------------------------------------------------------------
    pop_size = 40
    F = 0.8
    CR = 0.9

    population = np.empty((pop_size, n), dtype=np.float64)
    pop_scores = np.empty(pop_size, dtype=np.float64)

    # initialise with a few curated seeds
    seeds = [best_vec]
    seeds.append(u_min_distribution(n, 1.2))
    seeds.append(u_prod_distribution(n, 1.2))
    seeds.append(exp_distribution(n, 5e-3))
    seeds.append(mix_uniform_min(n, 1.0, 0.5))
    seeds.append(shifted_power_distribution(n, 1.0, 1.0))

    i = 0
    for vec in seeds[:pop_size]:
        population[i] = vec
        pop_scores[i] = score_vector(vec)
        i += 1
    # fill the rest with Dirichlet draws
    while i < pop_size:
        rnd = np.random.rand(n)
        population[i] = rnd / rnd.sum()
        pop_scores[i] = score_vector(population[i])
        i += 1

    best_idx = int(pop_scores.argmin())
    best_vec = population[best_idx].copy()
    best_score = float(pop_scores[best_idx])

    generation = 0
    print("\n=== DE optimisation start ===")
    while time.time() < deadline:
        generation += 1
        for idx in range(pop_size):
            # select three distinct donors
            idxs = np.arange(pop_size)
            idxs = idxs[idxs != idx]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            donor = population[a] + F * (population[b] - population[c])
            donor = np.maximum(donor, 0.0)

            # binomial crossover
            cross_mask = np.random.rand(n) < CR
            if not cross_mask.any():
                cross_mask[np.random.randint(n)] = True
            trial = np.where(cross_mask, donor, population[idx])

            # project to simplex
            trial = simplex_projection(trial)

            trial_score = score_vector(trial)

            if trial_score < pop_scores[idx]:
                population[idx] = trial
                pop_scores[idx] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    print(f"Gen {generation:4d} ind {idx:2d}: new best = {best_score:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 5. Exponentiated‑gradient polishing (multiplicative updates)
    # --------------------------------------------------------------
    print("\n=== Exponentiated‑gradient polishing start ===")
    cur = best_vec.copy()
    cur_score = best_score
    eta0 = 0.6               # initial step size in exponent
    decay = 0.9998           # geometric decay
    it = 0
    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft(cur)
        grad = compute_subgradient(cur, conv)

        eta = eta0 * (decay ** it)
        # shift gradient to avoid overflow/underflow
        shift = grad.min()
        exp_factor = np.exp(-eta * (grad - shift))
        # protect from numerical overflow
        exp_factor = np.clip(exp_factor, 1e-20, 1e20)

        cand = cur * exp_factor
        cand = simplex_projection(cand)
        cand_score = score_vector(cand)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                print(f"EG it {it:6d}: new best = {best_score:.6f}")

        # stop when step gets too tiny
        if eta < 1e-7:
            break

    # --------------------------------------------------------------
    # 6. Simple hill‑climber (move tiny mass from high‑grad to low‑grad)
    # --------------------------------------------------------------
    print("\n=== Gradient‑mass‑shift polishing start ===")
    cur = best_vec.copy()
    cur_score = best_score
    eps = 1e-4               # amount of mass to relocate each step
    it = 0
    while time.time() < deadline and it < 5000:
        it += 1
        conv = full_convolution_fft(cur)
        grad = compute_subgradient(cur, conv)
        i_max = np.argmax(grad)   # index that hurts the objective most
        i_min = np.argmin(grad)   # index that helps the most

        if cur[i_max] <= eps:
            # nothing to move from the worst index, break early
            break

        new = cur.copy()
        new[i_max] -= eps
        new[i_min] += eps
        # projection is not needed because sum is unchanged and both stay non‑negative
        new_score = score_vector(new)

        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = new.copy()
                print(f"Shift it {it:5d}: new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Optional cheap length‑scaling trial (interpolate to a nearby n)
    # --------------------------------------------------------------
    try:
        for factor in (1.2, 0.8):
            if time.time() > deadline:
                break
            new_n = int(round(n * factor))
            if new_n < 2000 or new_n > 20000 or new_n == n:
                continue
            cand = resample_vector(best_vec, new_n)
            cand_score = score_vector(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand
                n = new_n
                print(f"Resample to n={new_n}: new best = {best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 8. Return the best sequence found
    # --------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_vec)})")
    return best_vec.tolist()
