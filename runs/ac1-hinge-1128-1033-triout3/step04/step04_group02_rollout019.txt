def propose_candidate(seed=0, budget_s=1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5
    best_val = float('inf')
    best_seq = None

    # Define candidate n values - maybe up to 20000
    n_candidates = [2000, 4000, 8000, 16000] # adjust within 1000s
    # p values up to maybe 10 (or up to 12)
    p_candidates = list(range(1, 12))

    for n in n_candidates:
        # Precompute array of x = i/(n-1)
        i_arr = np.arange(n, dtype=float)
        x = i_arr / (n - 1) if n>1 else np.zeros(1)
        for p in p_candidates:
            weights = (1.0 - x) ** p
            if weights.sum() == 0:
                continue
            a = weights / weights.sum()
            # optional: enforce non-negativity (already)
            # evaluate exact function
            try:
                val = evaluate_sequence(a.tolist())
            except Exception:
                continue
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[candidate] n={n}, p={p}, val={val:.6f}")
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # optionally do hill climbing minor refinement if time left
    # but not needed.

    if best_seq is None:
        raise RuntimeError("No valid sequence found")

    # final evaluation again to confirm
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"[final] best evaluation {final_val:.6f}")

    return best_seq.tolist()

"""
Optimise the step‑function by exploring simple decreasing power‑law shapes.
For a given length n and exponent p we set a_i ∝ (1−i/(n−1))^p,
normalise to Σa_i = 1, evaluate with evaluate_sequence and keep the best.
The search scans a handful of n and p values within the time budget,
returning the sequence achieving the lowest exact evaluation.
"""
import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5

    best_val = float('inf')
    best_seq = None

    # Candidate lengths – we stay below a few hundred‑thousand
    n_candidates = [2000, 4000, 8000, 16000, 32000, 64000]
    # Exponents to try (power‑law decreasing shapes)
    p_candidates = list(range(0, 11))  # 0 = uniform, 1 = linear, ...

    # iterate with increasing n, p
    for n in n_candidates:
        if time.time() >= deadline:
            break
        # pre‑compute the normalised positions once
        idx = np.arange(n, dtype=float)
        denom = n - 1 if n > 1 else 1.0
        x = idx / denom
        for p in p_candidates:
            if time.time() >= deadline:
                break
            # Power‑law shape, non‑negative by construction
            weights = (1.0 - x) ** p
            # Guard against degenerate case where all weights are zero
            total = weights.sum()
            if total == 0.0:
                continue
            a = weights / total   # Σa = 1

            # Quick sanity – sum must be >0.01 after scaling (it is 1)
            try:
                val = evaluate_sequence(a.tolist())
            except Exception:
                # If evaluate_sequence raises, ignore this candidate
                continue

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[improve] n={n}, p={p}, val={val:.6f}")

        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_val = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    # Final report
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"[final] best evaluation = {final_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
