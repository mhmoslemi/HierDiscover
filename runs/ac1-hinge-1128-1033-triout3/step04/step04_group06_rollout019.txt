def project_simplex(v):
   # returns projection onto the probability simplex (sum=1, v>=0)
   n = len(v)
   u = np.sort(v)[::-1]   # descending
   cssv = np.cumsum(u) - 1
   ind = np.arange(n) + 1
   cond = u - cssv / ind > 0
   rho = ind[cond][-1]
   theta = cssv[cond][-1] / rho
   w = np.maximum(v - theta, 0)
   return w

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0

    # Set length
    n = 4000 # Could adapt based on budget: if time limited, maybe smaller.
    # Starting distribution: Dirichlet with concentration 0.3
    a = np.random.dirichlet(alpha=np.full(n, 0.3))
    # Ensure positivity
    a = np.maximum(a, 0)
    a = a / a.sum()

    best_a = a.copy()
    # compute initial objective
    conv = _full_convolution_fft(a, a)
    M = conv.max()
    best_val = float(2.0 * n * M / (a.sum()**2))

    # other variables
    momentum = np.zeros_like(a)

    # schedule parameters
    eta_start = 0.6
    eta_end = 1e-5
    beta = 0.9
    jitter_prob = 0.02
    jitter_frac = 0.4

    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # Compute conv and objective
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M / (a.sum()**2))

        # Update best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:6d}] new best = {best_val:.6f}")

        # progress
        progress = (now - start_time) / (deadline - start_time)

        # compute gradient w.r.t max
        # get index of max
        k = np.argmax(conv)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # momentum
        momentum = beta * momentum + (1.0 - beta) * g

        # step size
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # exponentiated update
        step = -eta * momentum
        # clip step to avoid overflow
        np.clip(step, -50, 50, out=step)
        a = a * np.exp(step)

        # jitter random transfer
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n-1)
            if dst >= src:
                dst += 1
            # choose fraction to move
            frac = jitter_frac * np.random.rand()
            delta = frac * a[src]
            a[src] -= delta
            a[dst] += delta

        # projection: renormalize to sum 1, clip negative
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        iteration += 1

        # occasionally recompute M to avoid drift
        if iteration % 2000 == 0:
            conv = _full_convolution_fft(a, a)

    # compute final evaluation
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence error:", exc)

    return best_a.tolist()

Mirror‑descent (KL‑mirror) on the soft‑max of the auto‑convolution,
augmented with an Adam‑style adaptive step‑size and occasional random
jitter.  The algorithm works on the probability simplex (a_i ≥ 0,
∑ a_i = 1) and directly minimizes the true objective
    2·n·max(a*a) / (∑ a)^2
by progressively “hardening” the soft‑max (τ → 0).  The adaptive
Adam‑step generally reaches lower values faster than the plain
momentum used in the reference implementation.
"""

import time
import numpy as np

def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for repeated calls)."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()      # next power of 2
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    np.maximum(conv, 0.0, out=conv)           # discard tiny negatives
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max objective.
    Computes   g = 2·(w_rev * a)   where * denotes full linear convolution.
    """
    n = a.shape[0]
    # reverse the weight vector (full length = 2n‑1)
    u = w[::-1]
    # full convolution u * a  → length 3n‑2
    conv = _full_convolution_fft(u, a)
    # indices that correspond to the gradient entries:
    # start = 2n‑2, then step backwards
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Stable soft‑max of the convolution vector with temperature τ.
    Returns a probability vector that sums to one.
    """
    if tau <= 0 or not np.isfinite(tau):
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    M = conv.max()
    # shift by the max for numerical stability
    exp_args = (conv - M) / tau
    np.clip(exp_args, -80, 80, out=exp_args)
    exp_vals = np.exp(exp_args)
    s = exp_vals.sum()
    if s == 0 or not np.isfinite(s):
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_vals / s
    return w

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin before the hard timeout
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose the length of the vector (reuse a known construction if any)
    # --------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = a0.shape[0]
        a = np.maximum(a0, 0.0)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()
        print(f"[init] using supplied sequence, n={n}")
    except Exception:
        # otherwise start from a random Dirichlet draw (sparse start)
        n = 3000                     # size large enough to tighten the lower bound
        conc = 0.4                   # concentration < 1 → sparse initialization
        a = np.random.dirichlet(alpha=np.full(n, conc))
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        print(f"[init] random start, n={n}")

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best_a = a.copy()
    conv = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv.max() / (a.sum() ** 2))
    print(f"[init] initial evaluation = {best_val:.6f}")

    # Adam‑style moment estimates (first and second)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.995
    eps = 1e-12

    # Hyper‑parameters (linearly annealed)
    lr_start = 0.45                # initial learning rate for the exponentiated update
    lr_end   = 5e-5                # final learning rate
    tau_start_factor = 0.35        # τ ≈ factor·max(conv) initially
    tau_end_factor   = 1e-6        # final factor (hard max)

    jitter_prob = 0.018            # frequency of random mass‑transfer
    jitter_max_frac = 0.45         # up‑to this fraction of the source mass is moved

    iteration = 0
    last_improve = 0               # iteration index of the last improvement

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- current objective -------------------------------------------------
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M / (a.sum() ** 2))

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = iteration
            print(f"[{iteration:6d}] new best = {best_val:.6f}")

        # ---- schedules (progress in [0,1]) ------------------------------------
        progress = (now - start_time) / (deadline - start_time)
        lr = lr_start * (1.0 - progress) + lr_end * progress
        tau = (tau_start_factor * (1.0 - progress) +
               tau_end_factor   * progress) * M

        # ---- soft‑max weights and gradient ------------------------------------
        w = _softmax_weights(conv, tau)
        g = _gradient_from_weights(a, w)                     # shape (n,)

        # ---- Adam accumulation -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # ---- multiplicative (exponentiated) update -----------------------------
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        np.clip(step, -50, 50, out=step)          # keep the exponent in a safe range
        a = a * np.exp(step)

        # ---- occasional random jitter (helps escape plateaus) -----------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            frac = jitter_max_frac * np.random.rand()
            delta = frac * a[src]
            a[src] -= delta
            a[dst] += delta

        # ---- projection back onto the simplex (simple renormalisation) ----------
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0 or not np.isfinite(total):
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

        iteration += 1

        # ---- restart if we have been stuck for a long time --------------------
        if iteration - last_improve > 30000:
            print(f"[{iteration:6d}] restart – no improvement for many iterations")
            a = np.random.dirichlet(alpha=np.full(n, conc))
            a = a / a.sum()
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration

        # occasional recomputation just to keep numerical drift tiny
        if iteration % 5000 == 0:
            conv = _full_convolution_fft(a, a)

    # ----------------------------------------------------------------------
    # Final reporting (the outer framework will also invoke evaluate_sequence)
    # ----------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
