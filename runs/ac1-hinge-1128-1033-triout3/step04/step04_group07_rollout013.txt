"""
Propose a step‑function (non‑negative probability vector) that minimises

    2·n·max(conv_self) / (sum(a)²)

The algorithm is a two‑phase optimiser:

1. **Exponentiated‑gradient Adam** on a log‑sum‑exp smooth approximation of the
   maximal autocorrelation.  The probabilities are represented as soft‑max
   of log‑weights `v`; this automatically satisfies `p_i ≥ 0` and Σp_i = 1,
   so no explicit simplex projection is needed.  The smoothing parameter `α`
   is increased gradually, turning the smooth objective into the true `max`
   and allowing a stable optimisation path.

2. **Exact sub‑gradient polishing** on the true maximal convolution entry,
   followed by a tiny random‑swap hill‑climber.  These last stages remove any
   remaining bias introduced by the smoothing.

Multiple diverse initialisations (arcsine shape, geometric decay,
uniform, several Dirichlet draws and the historic sequence if available) are
tried; the best vector found before the time budget expires is returned.

The whole search respects a user supplied time budget (default 1000 s) and
uses only NumPy (no filesystem or network access).  All numbers are clipped
to `[0, 1000]` as required by `evaluate_sequence`.

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the probability simplex {x | x_i ≥ 0, Σx_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    vmax = np.max(v)
    e = np.exp(v - vmax)
    return e / e.sum()


def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution using an FFT of size `m_fft`."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _gradient_from_weights(p: np.ndarray, p_fft: np.ndarray,
                          w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_i w_i·conv_i.
    `w` must have length 2·n‑1 and Σ w_i = 1.
    """
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad


def _exact_score(p: np.ndarray, m_fft: int, n: int) -> float:
    """Exact evaluation 2·n·max(conv_self(p))."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * n * np.max(conv))

        min_step = 1e-8
        # limit to avoid overrunning the budget
        for it in range(40_000):
            if time.time() >= deadline:
                break
            conv = _conv_self(p, m_fft)
            max_val = conv.max()
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                print(f"   Sub‑grad it {it:5d} – new best {best_score:.6f}")
                if best_score <= 1.503:
                    deadline = time.time()
                    break

            # active set (indices within a tiny tolerance of the max)
            tol = 1e-13
            active = np.where(conv >= max_val - tol)[0]

            g = np.zeros_like(p)
            for idx in active:
                if idx < n:
                    g[: idx + 1] += 2.0 * p[idx::-1]
                else:
                    start = idx - n + 1
                    g[start:] += 2.0 * p[n - 1 : start - 1 : -1]
            g /= float(len(active))

            p = p - step * g
            p = _simplex_projection(p)

            step = max(step * decay, min_step)

        # ----------------------------------------------------------
        # (C) Tiny random‑swap hill‑climber
        # ----------------------------------------------------------
        if best_p is None:
            continue
        p_swap = best_p.copy()
        cur_score = 2.0 * n * np.max(_conv_self(p_swap, m_fft))
        step_scale = 0.05
        for _ in range(80_000):
            if time.time() >= deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if p_swap[i] <= 1e-12:
                continue
            eps_move = step_scale * rng.random() * p_swap[i]
            new_p = p_swap.copy()
            new_p[i] -= eps_move
            new_p[j] += eps_move
            new_score = 2.0 * n * np.max(_conv_self(new_p, m_fft))
            if new_score < cur_score - 1e-15:
                p_swap = new_p
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_p = p_swap.copy()
                    print(f"Swap improvement → {best_score:.6f}")
                    if best_score <= 1.503:
                        deadline = time.time()
                        break
            # slowly shrink the move magnitude
            step_scale = max(step_scale * 0.9999, 1e-5)

        # early stop if we have already hit the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen but keeps the contract safe
    # ------------------------------------------------------------------
    if best_p is None:
        best_p = np.full(n, 1.0 / n)
        best_score = _exact_score(best_p, m_fft, n)

    print(f"\nFinal best score: {best_score:.6f}")

    # Clip to the bounds enforced by `evaluate_sequence` (0 ≤ x ≤ 1000)
    return np.clip(best_p, 0.0, 1000.0).tolist()
