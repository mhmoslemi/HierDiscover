"""
Propose a step‑function sequence that minimizes
    2 * n * max(a * a) / (sum(a)**2)

The algorithm is a two‑phase mirror‑descent / Adam optimiser on the
probability simplex combined with occasional random mass transfers.
* Phase 1 – warm‑up: exponentiated‑gradient updates with a soft‑max
  approximation of the ℓ∞‑norm (the same idea as the baseline but with
  Adam‑style per‑coordinate learning‑rate adaptation and a slightly
  more aggressive schedule).
* Phase 2 – refinement: the temperature is driven to (practically) zero,
  i.e. the weight vector becomes a hard‑max, and we keep performing
  Adam‑updates.  In parallel a small “jitter” step randomly moves a
  fraction of mass between two positions – this helps escape plateaus.

The optimiser works entirely on the simplex (a_i ≥ 0, Σ a_i = 1);
the objective is invariant under scaling, so normalising to a
probability vector does not affect the score returned by
`evaluate_sequence`.  The implementation is fully vectorised and uses
FFT‑based full convolutions for speed.  The routine runs until the time
budget expires and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    np.maximum(conv, 0, out=conv)    # eliminate tiny negatives caused by round‑off
    return conv


# ----------------------------------------------------------------------
# Gradient of the (soft‑max) smoothed max‑convolution
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute   g_i = 2 * Σ_k w_k * a_{k-i}
    via a single convolution: reverse w, convolve with a,
    then extract the appropriate slice.
    """
    n = a.shape[0]
    # reverse the weight vector (full length = 2n‑1)
    u = w[::-1]
    # full convolution u * a  → length 3n‑2
    conv = _full_convolution_fft(u, a)
    # indices that correspond to the gradient entries:
    start = 2 * n - 2
    idx = start - np.arange(n)   # descending indices
    g = 2.0 * conv[idx]
    return g

# Main entry point required by the competition framework
# ----------------------------------------------------------------------

                     **kwargs) -> list[float]:

    Optimise a non‑negative step function under the evaluation metric.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # finish a little early to avoid hard termination
    deadline = start_time + budget_s - 1.0

    # --------------------------------------------------------------
    # Choose length of the sequence.
    # --------------------------------------------------------------
    # If a previous construction exists, reuse it (and optionally pad).
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        # Pad with zeros to give a little more freedom (up to ~2500 entries)
        target_n = max(2500, a0.shape[0])
        if a0.shape[0] < target_n:
            pad = np.zeros(target_n - a0.shape[0], dtype=float)
            a = np.concatenate([a0, pad])
        else:
            a = a0.copy()
    except Exception:
        # No previous construction – start from a sparse Dirichlet draw.
        target_n = 2500
        concentration = 0.4           # <1 gives a fairly sparse start
        a = np.random.dirichlet(alpha=np.full(target_n, concentration))
    n = a.shape[0]

    # ------------------------------------------------------------------
    # Normalise to the probability simplex (Σ a_i = 1) and keep a non‑zero floor
    # ------------------------------------------------------------------
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    eps = 1e-15
    a = np.maximum(a, eps)
    a /= a.sum()

    # ------------------------------------------------------------------
    # Optimisation bookkeeping
    # ------------------------------------------------------------------
    best_a = a.copy()
    conv_a = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv_a.max())
    print(f"[init] n={n}, initial evaluation={best_val:.6f}")

    # Adam‑style moments (all zeros initially)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    epsilon = 1e-8
    t_iter = 0

    # Hyper‑parameters (linear schedules)
    eta_start = 0.80        # larger initial step than the baseline
    eta_end   = 1e-4
    tau_factor_start = 0.25
    tau_factor_end   = 1e-6

    beta_mom = 0.92         # momentum used only for the “jitter” variant
    jitter_prob = 0.03      # a bit more aggressive random mass‑transfer
    max_jitter_frac = 0.5   # up to 50 % of the source mass can be moved

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while True:
        now = time.time()

        conv_a = _full_convolution_fft(a, a)   # length 2n‑1

        # progress ∈ [0,1] for schedule interpolation

        # ---- temperature for the soft‑max (annealed) ----
        # after 80 % of the time use a hard max (τ ~ 0)

            w[np.argmax(conv_a)] = 1.0

            else:
                w = exp_vals / sum_exp

        # ---- gradient of the smoothed (or hard) objective ----
        grad = _gradient_from_weights(a, w)   # shape (n,)

        # ---- Adam moment updates ----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # ---- step‑size schedule (multiplicative) ----
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # ---- exponentiated Adam update (mirror descent on simplex) ----
        step = eta * m_hat / (np.sqrt(v_hat) + epsilon)   # direction to move *down*
        # we move opposite to the gradient, hence a ← a * exp(-step)
        np.clip(step, -50, 50, out=step)   # avoid overflow/under‑flow
        a = a * np.exp(-step)

        # ---- occasional random jitter (mass transfer) ----
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move a random fraction (up to max_jitter_frac) of the source mass
            delta = np.random.rand() * a[src] * max_jitter_frac
            a[src] -= delta
            a[dst] += delta

        # ---- projection back onto the simplex (simple renormalisation) ----
        a = np.maximum(a, eps)      # keep strictly positive
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # ---- occasional full recomputation (stability) ----
        if t_iter % 3000 == 0:
            # sanity check – recompute convolution, ignored otherwise
            _ = _full_convolution_fft(a, a)

    # ------------------------------------------------------------------
    # Final reporting (the outer framework will also call it)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
