"""
Propose a step‑function (non‑negative heights) that minimises the
`evaluate_sequence` objective.

The algorithm consists of three stages:

1.  **Fast shape exploration** – a handful of analytic families
    (uniform, symmetric power‑laws, sine‑shapes and a discretised
    arcsine density) are built, normalised and scored with the
    provided `evaluate_sequence`.  The best of those shapes is used as
    a warm‑start for the subsequent optimisation.

2.  **Smoothed‑max Adam optimisation** – the vector `a` is parametrised
    with a soft‑max (`a = softmax(w)`) so that the constraints
    `a_i ≥ 0, Σ a_i = 1` are always satisfied.  A log‑sum‑exp
    approximation to `max(conv(a,a))` yields a smooth surrogate that
    can be differentiated.  Adam updates are performed with a
    gradually shrinking smoothing parameter `ε` and learning‑rate.
    Random restarts keep the search from stagnating.

3.  **Targeted hill‑climbing on the exact objective** – after each
    Adam run we switch to a deterministic local search that moves a
    small amount of mass from a *source* index (chosen as the pair
    contributing most to the current maximal convolution entry) to a
    *target* index (the current smallest coefficient).  The convolution
    is updated in O(n) using the bilinearity of the operation, and a
    move is accepted only if the exact `evaluate_sequence` score
    strictly improves.  After long periods without improvement a
    light random perturbation is applied.

The whole routine respects the 1000 s time budget, uses a modest
problem size (`n ≈ 5000`), and leverages only NumPy and SciPy which are
available in the execution environment.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT‑based full convolution (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*x using an FFT (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """Run Adam on a log‑sum‑exp surrogate of max(conv(a,a))."""
    # soft‑max parametrisation
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.05
    lr_end = 0.001
    eps_start = 5e-3
    eps_end = 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    no_improve = 0
    IMPROVE_LIMIT = 50000   # restart after many idle steps

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- obtain feasible a (non‑negative, Σa = 1) -----
        w_shift = w - np.max(w)               # numerical stability
        a = np.exp(w_shift)
        a /= a.sum()

        # ----- exact current objective (tracking only) -----
        conv_aa = _full_convolution(a)
        cur_val = 2.0 * n * conv_aa.max()

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ----- progressive smoothing parameter ε -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # ----- smoothed max via log‑sum‑exp -----
        max_conv = conv_aa.max()
        shifted = (conv_aa - max_conv) / eps      # values ≤ 0
        weights = np.exp(shifted)                # softmax weights
        p = weights / weights.sum()              # approx. one‑hot for tiny ε

        # ----- gradient of the surrogate w.r.t. a -----
        #   ∂/∂a  Σ_k p_k (a*a)[k]  =  2·(p * a)  (convolution form)
        p_rev = p[::-1]                          # reversed for convolution
        conv_ab = _conv_fft(a, p_rev)            # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]      # middle n entries
        grad_a = 4.0 * n * segment[::-1]         # shape (n,)

        # ----- chain rule through the soft‑max -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update on w -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional small random jitter to escape flat plateaus
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart if we see no progress for a long time
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {iteration:7d}] Restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _targeted_hillclimb(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> np.ndarray:
    """Hill‑climb that always moves mass away from the pair that creates
    the current maximal convolution entry."""
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac = 0.5                           # maximal fraction of source mass to move
    IMPROVE_LIMIT = 200_000                  # after this many idle steps we perturb
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- locate the peak of the convolution ----
        k_star = int(np.argmax(b))

        # ---- find the source index which contributes most to that peak ----
        idx = np.arange(n)
        j_partner = k_star - idx
        mask = (j_partner >= 0) & (j_partner < n)
        products = np.zeros_like(a)
        products[mask] = a[idx[mask]] * a[j_partner[mask]]
        i_source = int(np.argmax(products))

        if a[i_source] <= 1e-14 or products[i_source] == 0.0:
            # fallback to a random source if the above fails
            i_source = int(np.argmax(a))

        # ---- pick a target index with smallest mass (different from source) ----
        #    the goal is to move probability away from the heavy pair.
        j_target = int(np.argmin(a + (np.arange(n) == i_source) * 1e9))

        # ---- amount of mass to transfer ----
        delta = a[i_source] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) ----
        b_new = b.copy()
        # cross‑terms 2·δ·a shifted
        b_new[j_target:j_target + n] += 2.0 * delta * a
        b_new[i_source:i_source + n] -= 2.0 * delta * a
        # pure δ² terms
        b_new[2 * j_target] += delta * delta
        b_new[2 * i_source] += delta * delta
        b_new[i_source + j_target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ---- accept only strict improvements ----
        if new_val + 1e-12 < cur_val:
            a[i_source] -= delta
            a[j_target] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---- occasional perturbation after long stagnation ----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] Stagnation – perturbing")

        iteration += 1

    return best_a

# Helper: generate a variety of analytically motivated shapes
# ----------------------------------------------------------------------
def _generate_initial_shapes(n: int):
    """Yield a list of candidate arrays (already normalised)."""
    shapes = []

    # uniform
    shapes.append(np.full(n, 1.0 / n))

    # symmetric power‑law (min distance to an endpoint) for several exponents
    for alpha in (0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60):
        d = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1  # distance ≥1
        a = 1.0 / (d.astype(float) ** alpha)
        a /= a.sum()
        shapes.append(a)

    # one‑sided power‑law (decreasing from the left)
    for alpha in (0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60):
        i = np.arange(1, n + 1)
        a = 1.0 / (i.astype(float) ** alpha)
        a /= a.sum()
        shapes.append(a)

    # half‑sine (smooth bump)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a /= a.sum()
    shapes.append(a)

    # sine‑square (more mass near the centre)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))) ** 2
    a /= a.sum()
    shapes.append(a)

    # discrete arcsine density: 1 / sqrt(x (1‑x))
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a /= a.sum()
    shapes.append(a)

    return shapes


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights under a 1000 s budget.
    Returns a list of non‑negative floats (Σ = 1) that minimises
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # -------- problem size ------------------------------------------------
    # 5000 gives a good trade‑off between expressive power and speed.
    n = 5000

    # -------- initial shape selection ------------------------------------
    best_val = float('inf')
    best_a = None

    # generate and score a handful of analytically motivated proposals
    for cand in _generate_initial_shapes(n):
        # evaluate via the provided wrapper (the wrapper enforces all guards)
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            print(f"[Init] new best = {best_val:.6f}")

    # -------- Phase A – smooth‑max Adam optimisation ----------------------
    phase_a_deadline = start_time + 0.55 * budget_s
    best_a, best_val = _adam_phase(best_a, n, start_time,
                                   phase_a_deadline, best_a, best_val)

    # -------- Phase B – targeted hill‑climbing ---------------------------
    best_a = _targeted_hillclimb(best_a, n,
                                 phase_a_deadline, deadline,
                                 best_a, best_val)

    # -------- final report ------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
