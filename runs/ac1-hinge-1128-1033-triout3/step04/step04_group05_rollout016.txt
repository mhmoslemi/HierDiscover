def _auto_convolution_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    total = 2*n - 1
    size = 1 << (total - 1).bit_length()
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)[:total]
    np.maximum(conv, 0, out=conv)
    return conv

def optimize_sequence(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    x = np.log(np.maximum(a, eps))
    # Adam parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0
    eps_adam = 1e-8
    base_lr = 0.2
    best_a = a.copy()
    best_val = 2.0 * n * np.max(_auto_convolution_fft(a))
    it = 0
    while time.time() < deadline:
        # compute conv and max
        conv = _auto_convolution_fft(a)
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{it}] new best {best_val:.6f}")
        # compute softmax weights
        progress = (time.time() - start_time) / (deadline - start_time)
        # Temperature schedule
        tau_start_factor = 0.7   # larger initial smoothing
        tau_end_factor = 1e-6    # final factor
        tau = M * (tau_start_factor * (1 - progress) + tau_end_factor * progress)
        # compute w
        if tau > 0:
            exp_args = (conv - M) / tau
            max_arg = exp_args.max()
            exp_vals = np.exp(exp_args - max_arg)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        # gradient
        grad_a = _gradient_from_weights(a, w)
        # compute grad_x
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        # adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        lr = base_lr * (1 - progress) + 1e-5 * progress
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # reconstruct a
        max_x = x.max()
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        # random jitter
        if np.random.rand() < 0.05:
            # random mass transfer
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            delta = np.random.rand() * a[i] * 0.7
            a[i] -= delta
            a[j] += delta
            a = np.clip(a, 0, None)   # ensure non-neg
            a = a / a.sum()
            # recompute logits
            max_x = np.log(np.maximum(a, eps)).max()
            x = np.log(np.maximum(a, eps))
            m.fill(0.0); v.fill(0.0)
            beta1_pow = 1.0; beta2_pow = 1.0
        it += 1
    return best_a, best_val

Given a (size n). Compute conv_current via FFT. To compute next conv after moving delta, we can compute:

```

# new a' = a - eff*e_i + eff*e_j

# conv' = conv + 2*eff*(conv_of_ej - conv_of_ei) + eff^2*(delta_conv_self)

Where conv_of_ei = convolution of e_i with a (i.e., a shifted). conv_of_ei is vector length 2n-1: entry at k is a_{k-i} if within bounds else 0. This is just a shifted version: conv_of_ei = np.concatenate([np.zeros(i), a, np.zeros(n-1-i)]) ??? Actually conv of e_i (single mass at i) with a yields a smeared copy of a where indices ranging from i to i+n-1 correspond to a[0..n-1] i.e., conv_of_ei[k] = a_{k-i} if 0 <= k-i < n else 0. So conv_of_ei = a shifted by i positions and zero-padded to length 2n-1.

Similarly conv_of_ej is shift by j.

"""
Step‑function optimizer.

The evaluation metric for a non‑negative vector a (length n) is

    E(a) = 2·n·max_k (a * a)[k] / (sum a)^2 .

Because the numerator and denominator scale quadratically, the objective is
scale‑invariant; we may therefore restrict ourselves to probability vectors
(sum a = 1).  For such vectors the objective reduces to 2·n·max(conv), where
conv = a * a is the full linear convolution.

The algorithm consists of two phases that together respect the 1000 s budget:

1. **Smoothed max‑gradient descent (Adam).**
   a is represented in log‑space (soft‑max) so that the simplex constraint is
   satisfied automatically.  The “max’’ is approximated by a temperature‑scaled
   soft‑max (log‑sum‑exp) which provides a smooth sub‑gradient.  The learning
   rate and the temperature are annealed with the overall progress.

2. **Local stochastic search.**
   Starting from the best point found by the gradient phase we perform many
   small random mass‑transfer moves (i → j).  The convolution is updated
   incrementally in O(n) using the analytic formula for the effect of a single
   transfer, and the new objective is accepted if it improves or with a
   diminishing Metropolis probability.

Both phases keep the overall best vector seen so far; the final vector is
returned as a plain ``list`` of floats.  The implementation works for any
reasonable length – we use n = 5000 which provides many degrees of freedom while
still fitting comfortably into the time budget.

No external files or network access are required; the only external call is the
provided ``evaluate_sequence`` which is invoked once at the end for reporting.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) auto‑convolution via real FFT
# ----------------------------------------------------------------------
def _auto_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    total = 2 * n - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fa * Fa, size)[:total]
    np.maximum(conv, 0, out=conv)                # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max objective w.r.t. a
#   grad_a = 2 * Σ_k w_k * a_{k-i}
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ_k w_k·(a*a)[k] w.r.t. a.
    Implemented as a single convolution of a with the reversed weight vector.
    """
    n = a.shape[0]
    u = w[::-1]                                   # reversed weights
    total = n + len(w) - 1                         # = 3n‑2
    size = 1 << (total - 1).bit_length()
    Fu = np.fft.rfft(u, size)
    Fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(Fu * Fa, size)[:total]    # length 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)                    # indices that give the gradient
    return 2.0 * conv[idx]


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (works on the soft‑max logits)
# ----------------------------------------------------------------------
def _adam_optimize(
    a_start: np.ndarray,
    start_time: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the smoothed max objective until *deadline*.
    Returns the best vector found and its objective value (2·n·max(conv)).
    """
    a = a_start.copy()
    n = a.shape[0]
    eps = 1e-12
    x = np.log(np.maximum(a, eps))          # logits (soft‑max representation)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.90, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps_adam = 1e-8
    base_lr = 0.25                           # initial learning rate

    # bookkeeping of the best solution
    conv = _auto_convolution_fft(a)
    best_val = float(2.0 * n * conv.max())
    best_a = a.copy()

    iteration = 0
    print(f"[init] n={n}, start objective={best_val:.6f}")

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- current convolution & objective -----
        conv = _auto_convolution_fft(a)
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ----- progress‑dependent schedules -----
        progress = (now - start_time) / (deadline - start_time)

        # temperature for the soft‑max (smooth max)
        tau_start = M * 0.70               # fairly smooth at the beginning
        tau_end = M * 1e-6
        tau = tau_start * (1.0 - progress) + tau_end * progress

        # ----- soft‑max weights over convolution entries -----
        if tau > 0.0:
            exp_args = (conv - M) / tau
            max_arg = exp_args.max()
            exp_vals = np.exp(exp_args - max_arg)   # shift for stability
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ----- gradient w.r.t. a (smoothed) -----
        grad_a = _gradient_from_weights(a, w)

        # ----- chain‑rule to obtain gradient w.r.t. logits x -----
        dot = np.dot(grad_a, a)                     # <grad_a , a>
        grad_x = a * (grad_a - dot)                 # ∇_x soft‑max = diag(a)*(grad_a-<grad_a,a>)

        # ----- Adam update for x -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule (linear decay)
        lr = base_lr * (1.0 - progress) + 1e-5 * progress
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- obtain a from the updated logits (soft‑max) -----
        max_x = x.max()
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()                     # stays on the simplex

        # ----- occasional random “jitter’’ to escape plateaux -----
        if np.random.rand() < 0.05:                 # ~5 % per iteration
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            # move a random fraction of mass from i to j
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            # reset Adam state after a sudden change
            x = np.log(np.maximum(a, eps))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Local stochastic search (random mass transfers with incremental update)
# ----------------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    start_time: float,
    deadline: float,
    best_a_global: np.ndarray,
    best_val_global: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a simulated‑annealing style random‑transfer search.
    The convolution is updated incrementally, which makes each trial O(n).
    """
    a = a_start.copy()
    n = a.shape[0]
    conv = _auto_convolution_fft(a)                # current convolution
    best_a = best_a_global.copy()
    best_val = best_val_global

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose moving a fraction of mass from i to j
        max_delta = a[i] * 0.5
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * max_delta

        # candidate vector (only needed for the incremental update)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # ----- incremental update of the convolution -----
        conv_new = conv.copy()

        # linear term: 2·δ·(shift(j) – shift(i))
        # shift(i) = a placed at positions i … i+n-1
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a

        # quadratic self‑term for the moved mass
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        # new objective
        new_M = conv_new.max()
        new_val = 2.0 * n * new_M

        # annealing temperature (linear decay)
        progress = (now - start_time) / (deadline - start_time)
        T = 1e-3 * (1.0 - progress) + 1e-6

        # acceptance rule
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / T):
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[search {iteration:7d}] new best = {best_val:.6f}")

        iteration += 1

    return best_a, best_val

    # ------------------------------------------------------------------
    phase1_deadline = start_time + (budget_s * 0.68) - 2.0
    best_a, best_val = _adam_optimize(a0, start_time, phase1_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – local stochastic search (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _local_search(
            best_a,
            start_time,
            deadline,
            best_a,
            best_val,
        )

    # ------------------------------------------------------------------
    # Final reporting (the framework will also call evaluate_sequence)
    # ------------------------------------------------------------------
    result = best_a.tolist()
    try:
        final_score = evaluate_sequence(result)
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return result
