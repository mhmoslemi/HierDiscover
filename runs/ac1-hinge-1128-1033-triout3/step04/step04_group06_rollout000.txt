Implementation details for simplex projection:

def project_simplex(v):
    # ensure sum(v) = 1 and v>=0.
    # Using algorithm:
    # Sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function for the competition.

The algorithm is a projected Adam optimizer on a smooth approximation
of the objective  f(a)=max_k (a*a)_k .  The max is replaced by a
soft‑max with an annealed temperature τ, which gradually turns the
smooth objective into the true ∞‑norm.  The variables a_i live on the
probability simplex (a_i ≥ 0, Σ a_i = 1); after each Adam step we project
back onto the simplex with the Euclidean projection of Duchi et al.
A small “jitter’’ (random mass transfer) is performed occasionally to
escape plateaus and we restart from a fresh random Dirichlet draw if we
see no improvement for a while.  The code runs for the whole time
budget (default 1000 s) and returns the best sequence observed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full linear convolution using a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the full linear convolution of x and y using a real FFT.
    The result has length len(x)+len(y)-1.
    """
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate very small negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the (soft‑max) smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a probability vector a (length n) and a weight vector w of
    length 2n‑1 (soft‑max over the convolution a*a),
    compute the gradient of Σ_k w_k (a*a)_k with respect to a.
    Result has length n.
    """
    n = a.shape[0]
    # reverse the weight vector (full length 2n‑1)
    u = w[::-1]
    # convolution u * a → length 3n‑2
    conv = _full_convolution_fft(u, a)
    # extract the n entries that form the gradient
    start = 2 * n - 2
    idx = start - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# (Duchi, Shalev‑Shwartz, Singer & Chandra, 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector v onto the simplex {x | x_i ≥ 0, Σ x_i = 1}.
    Returns a new array of the same shape.
    """
    # shift by the maximum to improve numerical stability
    # (not strictly necessary but helps with large negative entries)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative; return the uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # due to numerical issues, renormalise exactly
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  Uses a projected Adam optimiser with an
    annealed soft‑max approximation of the ∞‑norm.  The function runs
    for the allotted time (default 1000 s) and returns the best
    sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Leave a small safety margin before the hard deadline
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose the length of the sequence.
    # Larger n gives a tighter theoretical lower bound (≈ 2n/(2n‑1)),
    # but also costs more per iteration.  4000 works well in practice.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initial point: a Dirichlet draw with concentration < 1
    # (leads to a reasonably sparse start that can be refined).
    # --------------------------------------------------------------
    a = np.random.dirichlet(alpha=np.full(n, 0.5))
    a = _project_simplex(a)               # ensure exact simplex membership

    # --------------------------------------------------------------
    # Adam bookkeeping
    # --------------------------------------------------------------
    m = np.zeros(n)                       # first moment estimate
    v = np.zeros(n)                       # second moment estimate
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8

    # --------------------------------------------------------------
    # Hyper‑parameter schedules
    # --------------------------------------------------------------
    lr_start = 0.5        # initial learning rate (Adam's α)
    lr_end   = 0.001      # final learning rate
    tau_factor_start = 0.5   # τ ≈ factor * max_conv at the beginning
    tau_factor_end   = 1e-5  # final factor (practically zero)
    jitter_prob = 0.012        # probability of a random mass‑transfer
    restart_interval = 150.0   # seconds without improvement → restart

    # --------------------------------------------------------------
    # Tracking of the best solution seen so far
    # --------------------------------------------------------------
    cur_conv = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * cur_conv.max())
    best_a = a.copy()
    best_iter = 0
    time_last_improve = start_time
    iteration = 0

    print(f"[init] n={n}, initial evaluation={best_val:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----------------------------------------------------------
        # 1) Evaluate current objective
        # ----------------------------------------------------------
        conv_a = _full_convolution_fft(a, a)   # length 2n‑1
        M = conv_a.max()
        cur_val = float(2.0 * n * M)

        # ----------------------------------------------------------
        # 2) Keep the best solution so far
        # ----------------------------------------------------------
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            best_iter = iteration
            time_last_improve = now
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ----------------------------------------------------------
        # 3) Determine progress (for annealing schedules)
        # ----------------------------------------------------------
        progress = (now - start_time) / (deadline - start_time)

        # ----------------------------------------------------------
        # 4) Soft‑max temperature τ (annealed)
        # ----------------------------------------------------------
        tau_factor = (tau_factor_start * (1.0 - progress) +
                      tau_factor_end   * progress)
        tau = tau_factor * M

        if tau <= 0 or not np.isfinite(tau):
            # fall back to a hard max (sub‑gradient of the true ∞‑norm)
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0
        else:
            # stable soft‑max: shift by the maximum to avoid overflow
            exp_args = (conv_a - M) / tau
            np.clip(exp_args, -80, 80, out=exp_args)   # guard against overflow
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv_a)
                w[np.argmax(conv_a)] = 1.0
            else:
                w = exp_vals / sum_exp

        # ----------------------------------------------------------
        # 5) Gradient of the smoothed objective
        # ----------------------------------------------------------
        grad = _gradient_from_weights(a, w)   # shape (n,)

        # ----------------------------------------------------------
        # 6) Adam update (with step‑size schedule)
        # ----------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # learning‑rate annealing (linear)
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # Adam's parameter update (note the minus sign: we minimise)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----------------------------------------------------------
        # 7) Projection onto the simplex
        # ----------------------------------------------------------
        a = _project_simplex(a)

        # ----------------------------------------------------------
        # 8) Occasional random jitter to escape plateaus
        # ----------------------------------------------------------
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # transfer a random fraction (up to 30 %) of the mass at src
            delta = a[src] * np.random.rand() * 0.30
            a[src] -= delta
            a[dst] += delta
            # re‑project to maintain feasibility
            a = _project_simplex(a)

        # ----------------------------------------------------------
        # 9) Restart if we have been stuck for a while
        # ----------------------------------------------------------
        if now - time_last_improve > restart_interval:
            # new random start, keep the best seen so far
            a = np.random.dirichlet(alpha=np.full(n, 0.5))
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            iteration = 0
            time_last_improve = now
            print(f"[restart] re‑initialised after {restart_interval}s without improvement.")

        # ----------------------------------------------------------
        # 10) Periodic sanity check (optional)
        # ----------------------------------------------------------
        if iteration % 5000 == 0:
            # recompute convolution just to keep numeric drift in check
            _ = _full_convolution_fft(a, a)   # result ignored

    # --------------------------------------------------------------
    # Final evaluation using the provided evaluate_sequence()
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return the best sequence found (as plain Python list)
    return best_a.tolist()
