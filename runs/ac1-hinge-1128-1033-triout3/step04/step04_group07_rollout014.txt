Propose a step‑function (non‑negative vector) that minimises

        2 · n · max(conv_self) / (sum(a)²)

The algorithm proceeds in three stages:

1. **Generate a diverse pool of initial candidates**
   – arcsine shape, power‑law shapes, Beta‑distribution shapes,
     convex mixtures of the above and a few random Dirichlet draws.
   All candidates are normalised to sum = 1.

2. **Select the best few candidates (by the exact objective) and
   refine each of them with a long projected sub‑gradient descent
   on the *exact* max‑convolution entry.**  
   The sub‑gradient is obtained from the active set of indices whose
   convolution value is within a tiny tolerance of the current maximum.
   The step‑size is slowly decayed; every 50 000 iterations a tiny
   random perturbation is added to escape shallow local minima.
   The iterate is projected onto the probability simplex after each
   update.

3. **A final cheap random‑swap hill‑climber** explores the neighbourhood
   of the best vector found so far.

Both the generation and the refinement respect the security
checks performed by `evaluate_sequence`.  The whole routine stops
as soon as the 1000 s time budget is exhausted and returns the best
vector discovered.
"""

import numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# --------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1} using the O(n) algorithm."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Convolution utilities (FFT based)
# --------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full self‑convolution of a probability vector `p` using FFT."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _objective(p: np.ndarray, n: int, m_fft: int) -> float:
    """Exact objective 2·n·max(conv_self(p)).  `p` must already sum to 1."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Candidate generators
# --------------------------------------------------------------
def _gen_powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-0.5-i))^{-p}, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i) + 1e-12
    a = denom ** (-p)
    a /= a.sum()
    return a


def _gen_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(alpha,beta) density sampled at regular points, then normalised."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n  # mid‑point of each bin
    # avoid exact 0/1 when exponents < 1
    eps = 1e-12
    xa = np.maximum(x, eps) ** (alpha - 1.0)
    xb = np.maximum(1.0 - x, eps) ** (beta - 1.0)
    a = xa * xb
    a /= a.sum()
    return a


def _gen_mixture(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam*a + (1-lam)*b (both already normalised)."""
    return lam * a + (1.0 - lam) * b

                # stop early if already at the target
                if best_score <= 1.503:
                    break

        # stop altogether if the target has been reached
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – just in case no candidate survived the time limit.
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)
        best_score = _objective(best_vec, n, m_fft)

    print(f"\nFinal best score: {best_score:.6f}")

    # Clip to the bounds enforced by `evaluate_sequence` (0 ≤ x ≤ 1000)
    return np.clip(best_vec, 0.0, 1000.0).tolist()
