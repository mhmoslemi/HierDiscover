# ----------------------------------------------------------------------
# Helper: fast full convolution using a real FFT (zero‑padded to next pow‑2)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of 1‑D arrays `x` and `y`."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power‑of‑two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Clip tiny negatives from round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max approximation of max(a*a)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g = 2 * (reverse(w) * a), where * denotes full convolution.
    The result has the same length as `a`.
    """
    n = a.shape[0]
    # reverse the (full) weight vector (length = 2n‑1)
    u = w[::-1]

    start = 2 * n - 2          # last index that overlaps a completely
    idx = start - np.arange(n)  # descending indices of length n
    g = 2.0 * conv[idx]
    return g

        a = a / s

    # --------------------------------------------------------------
    # Optimisation bookkeeping
    # --------------------------------------------------------------
    conv_a = _full_convolution_fft(a, a)
    best_val = float(2.0 * n * conv_a.max())
    best_a = a.copy()
    print(f"[init] n={n}, initial evaluation={best_val:.6f}")

    # momentum (Exponential moving average of the gradient)
    momentum = np.zeros_like(a)

    # ------------------- hyper‑parameters (linear schedules) ----------
    eta_start = 0.45          # initial multiplicative step size
    eta_end   = 1e-5          # final step size
    tau_factor_start = 0.40   # τ ≈ factor * max_conv at the beginning
    tau_factor_end   = 1e-5   # final factor (practically zero)

    beta = 0.95               # momentum decay
    jitter_prob = 0.04        # probability of a tiny random mass move
    large_perturb_prob = 0.003  # probability of a larger Dirichlet mix
    eps = 1e-15               # to keep entries strictly positive

    iteration = 0
    # --------------------------------------------------------------
    # Main optimisation loop (soft‑max EG)
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- convolution and current objective value -----
        conv_a = _full_convolution_fft(a, a)      # length 2n‑1
        M = conv_a.max()
        cur_val = float(2.0 * n * M)               # sum(a)=1, so formula simplifies

        # keep the best ever seen
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{iteration:7d}] new best = {best_val:.6f}")

        # ----- progress for schedule -----
        progress = (now - start_time) / (deadline - start_time)

        # ----- temperature τ (soft‑max) -----
        tau = (tau_factor_start * (1.0 - progress) +
               tau_factor_end   * progress) * M
        if tau <= 0.0:
            # fall back to a hard sub‑gradient (single active index)
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0
        else:
            # stable soft‑max: shift by the maximum to avoid overflow
            exp_args = (conv_a - M) / tau
            np.clip(exp_args, -80, 80, out=exp_args)   # safety
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv_a)
                w[np.argmax(conv_a)] = 1.0
            else:
                w = exp_vals / sum_exp

        # ----- gradient of the smoothed objective -----
        grad = _gradient_from_weights(a, w)

        # ----- momentum update (EMA) -----
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- step‑size schedule (multiplicative) -----
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # ----- multiplicative EG update -----
        step = -eta * momentum
        np.clip(step, -50, 50, out=step)          # avoid overflow/underflow
        a = a * np.exp(step)

        # ----- occasional tiny jitter (random mass transfer) -----
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move a random fraction (up to 50 %) of the source mass
            delta = np.random.rand() * a[src] * 0.5
            a[src] -= delta
            a[dst] += delta

        # ----- occasional larger perturbation (mix with fresh Dirichlet) -----
        if np.random.rand() < large_perturb_prob:
            t = np.random.rand() * 0.3   # blend factor up to 0.3
            fresh = np.random.dirichlet(alpha=np.full(n, 0.5))
            a = (1.0 - t) * a + t * fresh

        # ----- project back onto the simplex -----
        a = np.maximum(a, eps)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        iteration += 1

        # occasional status report (does not affect speed much)
        if iteration % 5000 == 0:
            print(f"[{iteration:7d}] best so far = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final local‑search refinement (uses the exact evaluation)
    # --------------------------------------------------------------
    # Compute remaining time; keep about 0.5 s margin for safety.
    time_left = deadline - time.time()
    if time_left > 0.5:
        # number of tiny perturbation attempts we can afford
        max_trials = int(min(5000, time_left / 0.0015))
        for _ in range(max_trials):
            # propose a tiny random mass transfer
            cand = best_a.copy()
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move a small fraction (≤10 %) of source mass
            delta = np.random.rand() * cand[src] * 0.1
            if delta <= 0.0:
                continue
            cand[src] -= delta
            cand[dst] += delta
            # keep on the simplex (sum unchanged)
            # evaluate exactly (this is the real objective)
            try:
                val = evaluate_sequence(cand.tolist())
            except Exception:
                continue
            if val < best_val:
                best_val = val
                best_a = cand
                print(f"[refine] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final report (optional – the outer framework may also call it)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
