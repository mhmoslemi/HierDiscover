"""
Greedy local search for minimizing `evaluate_sequence`.

The algorithm works in three stages:

1. **Initialisation** – start from the best known construction if the
   global `height_sequence_1` is available, otherwise fall back to a
   uniform distribution on `n` points (n≈4000).  The vector is
   normalised to sum to 1.

2. **Optional quick Adam phase** – a very short Adam optimisation of a
   smooth log‑sum‑exp surrogate of the max‑convolution.  This gives a
   decent starting point but uses only a few percent of the budget.

3. **Greedy hill‑climbing** – the exact objective is `2·n·max(a*a)`.
   At each iteration we locate the index `k` where the convolution is
   maximal, compute the “partner” values `p_i = a_{k‑i}` (zero outside
   the support) and move mass from the source index with the largest
   partner to the target index with the smallest partner.  The move
   size is chosen analytically from the current gap to the second
   largest convolution entry; the convolution is updated incrementally
   in `O(n)` time.  When no improving move is found we perform a small
   random perturbation of the current best solution to escape local
   minima.  The loop runs until the time budget expires.

The whole procedure respects the non‑negativity and sum constraints
and returns the best sequence discovered before the deadline.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# FFT based full convolution (used only for the initial exact evaluation)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Very short Adam phase (optional – improves the start a little)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """A lightweight Adam optimiser on the log‑sum‑exp smooth max."""
    w = np.log(a0 + 1e-12)

    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.05
    lr_end = 0.001
    eps_start = 5e-3
    eps_end = 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 3000

    while time.time() < deadline:
        # soft‑max to stay on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # exact objective (for tracking only)
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # linear schedule for ε and learning‑rate
        progress = (time.time() - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # smooth‑max weights
        max_conv = conv.max()
        shifted = (conv - max_conv) / eps
        weights = np.exp(shifted - np.max(shifted))      # avoid overflow
        p = weights / weights.sum()

        # gradient of smoothed objective w.r.t. a
        p_rev = p[::-1]
        grad_a = 4.0 * n * np.fft.irfft(np.fft.rfft(a, 2 * n) *
                                        np.fft.rfft(p_rev, 2 * n))[:n]
        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter
        if np.random.rand() < 0.01:
            w += np.random.normal(scale=0.01, size=n)

        if no_improve > IMPROVE_LIMIT:
            # simple restart
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0
            print("[Adam] restart due to stagnation")

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray,
                      n: int,
                      start: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """
    Deterministic local search:
    - move mass from the index that contributes most to the current peak
      to the index that contributes least;
    - step size is chosen from the gap to the second‑largest convolution entry;
    - after many idle iterations a small random perturbation is applied.
    Returns the best sequence seen before *deadline*.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()
    if cur_val < best_val:
        best_val = cur_val
        best_a = a.copy()

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0

    while time.time() < deadline:
        # index of the current maximum of the convolution
        k_max = int(np.argmax(b))
        max_b = b[k_max]

        # second largest value (used to size the step)
        b_tmp = b.copy()
        b_tmp[k_max] = -np.inf
        second_max = b_tmp.max()
        gap = max_b - second_max

        # partner values p_i = a_{k_max‑i} (0 outside the support)
        idx = np.arange(n)
        src = k_max - idx
        mask = (src >= 0) & (src < n)
        partner = np.zeros(n, dtype=float)
        partner[mask] = a[src[mask]]

        # --- choose source index ------------------------------------------------
        source_mask = a > 1e-12
        if not np.any(source_mask):
            break
        # ignore indices with zero mass
        partner_src = partner.copy()
        partner_src[~source_mask] = -np.inf
        i_src = int(np.argmax(partner_src))

        # --- choose target index ------------------------------------------------
        j_tgt = int(np.argmin(partner))
        if j_tgt == i_src:
            # pick the second smallest partner value
            sorted_idx = np.argsort(partner)
            for cand in sorted_idx:
                if cand != i_src:
                    j_tgt = int(cand)
                    break

        diff = partner[i_src] - partner[j_tgt]
        if diff <= 0:
            # nothing can improve the current peak
            break

        # step size: try to bring the peak down to the second‑largest value
        if gap > 0:
            delta_needed = gap / (2.0 * diff)
            delta = min(a[i_src] * 0.5, delta_needed * 0.9)
        else:
            # no gap (plateau) – take a modest step
            delta = a[i_src] * 0.01

        if delta <= 0:
            delta = a[i_src] * 0.01

        # incremental update of the convolution (exact)
        # slice updates for the cross‑terms
        b_new = b.copy()
        b_new[j_tgt:j_tgt + n] += 2.0 * delta * a
        b_new[i_src:i_src + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j_tgt] += delta * delta
        b_new[2 * i_src] += delta * delta
        b_new[i_src + j_tgt] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept move
            a[i_src] -= delta
            a[j_tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Greedy {iteration:7d}] new best = {best_val:.6f}")

        else:
            # try smaller fractions of the suggested delta
            accepted = False
            for factor in (0.5, 0.25, 0.125):
                d = delta * factor
                if d < 1e-12:
                    continue
                b_tmp = b.copy()
                b_tmp[j_tgt:j_tgt + n] += 2.0 * d * a
                b_tmp[i_src:i_src + n] -= 2.0 * d * a
                b_tmp[2 * j_tgt] += d * d
                b_tmp[2 * i_src] += d * d
                b_tmp[i_src + j_tgt] -= 2.0 * d * d
                val_tmp = 2.0 * n * b_tmp.max()
                if val_tmp < cur_val - 1e-12:
                    a[i_src] -= d
                    a[j_tgt] += d
                    b = b_tmp
                    cur_val = val_tmp
                    no_improve = 0
                    accepted = True
                    if val_tmp < best_val:
                        best_val = val_tmp
                        best_a = a.copy()
                        print(f"[Greedy {iteration:7d}] new best (small step) = {best_val:.6f}")
                    break
            if not accepted:
                no_improve += 1

        # ---- stagnation handling ------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            # perturb the best solution a little and restart from there
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Greedy {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation framework

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list of floats (Σ = 1) that minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Choose a problem size.  n≈4000 gives a good balance between
    # expressive power and speed (the bound does not depend on n).
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise a – reuse the best known construction if available.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # Interpolate onto a new grid of length n
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        # fallback: uniform distribution (guaranteed feasible)
        a0 = np.full(n, 1.0 / n)

    a0 /= a0.sum()
    best_a = a0.copy()
    # initial exact objective (for reference)
    cur_val = 2.0 * n * _full_convolution(a0).max()
    best_val = cur_val
    print(f"[Init] objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # (Optional) very short Adam phase – uses ~5 % of the budget.
    # --------------------------------------------------------------
    adam_deadline = start_time + 0.05 * budget_s
    if adam_deadline < deadline:
        best_a, best_val = _adam_phase(a0, n, start_time, adam_deadline,
                                       best_a, best_val)

    # --------------------------------------------------------------
    # Main greedy hill‑climbing phase.
    # --------------------------------------------------------------
    best_a, best_val = _greedy_hillclimb(best_a, n, start_time, deadline,
                                         best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
