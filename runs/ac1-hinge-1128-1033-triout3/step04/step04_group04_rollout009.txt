        cand_score = float(2.0 * n * full_convolution_fft(cand).max())

        if cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_score_sofar:
                best_score_sofar = cand_score
                best_vec = cand.copy()
                last_improve = it
                print(f"Subgrad it {it:6d}: new best = {cand_score:.6f}")

        step *= step_decay
        if step < 1e-7:
            break
        if it - last_improve > 3000:
            # tiny random kick to get unstuck
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            current_score = float(2.0 * n * full_convolution_fft(current).max())
            step = 0.15
            last_improve = it
            print(f"Subgrad it {it:6d}: random kick applied")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score discovered: {best_score_sofar:.6f} (length {len(best_vec)})")
    return best_vec.tolist()
