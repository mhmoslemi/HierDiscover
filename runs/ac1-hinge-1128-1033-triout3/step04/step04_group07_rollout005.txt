```
            j = rng.integers(0, n-1)

            a_new[j] += delta

            if dE < 0 or rng.random() < np.exp(-dE / T):

                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"Anneal {idx}-{it}: new best {best_score:.6f}")
```

assistantfinal**Solution Overview**

The objective  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

is invariant to a global scaling, therefore we can enforce the *probability simplex*  

\[
\sum_i a_i = 1,\;a_i\ge 0 .
\]

The problem is to make the self‑convolution `a*a` as flat as possible
(i.e. minimise its $\ell_\infty$‑norm).  
The programme below performs a multi‑stage global search:

1. **Broad initial pool** – arcsine‑shaped families, geometric decays,
   uniform, a few random Dirichlet draws and, if available, the historic
   construction.
2. **Exact sub‑gradient descent** on the true maximum of the convolution
   (projected onto the simplex).  
   The sub‑gradient is computed analytically and the step size is
   exponentially decayed.
3. **Adam optimisation** on a *log‑sum‑exp* smooth approximation of the max.
   The smoothing parameter `α` is increased gradually, turning the smooth
   objective into the true one.
4. **Simulated‑annealing / random‑swap hill‑climber** – a cheap
   metropolis move that walks the simplex and can escape shallow local minima.
5. **Best‑ever tracking** – every improvement updates the global best.

All operations use FFT‑based convolutions (complexity $O(n\log n)$) and
the Euclidean projection onto the simplex to keep the iterates feasible.
The algorithm stops as soon as the target value `1.503` is reached or the
time budget expires.

```python
"""
propose_candidate

A multi‑stage optimiser for the self‑convolution L∞‑norm problem.
Starts from a diverse pool (arcsine‑family, geometric, uniform,
random Dirichlet, historic sequence), performs exact sub‑gradient descent,
Adam on a log‑sum‑exp smoothing, and finishes with a simulated‑annealing
hill‑climber.  All iterates stay on the probability simplex.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x_i ≥ 0, Σ x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based convolution utilities.
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, m_fft: int) -> np.ndarray:
    """Full linear self‑convolution of a using a pre‑computed FFT size."""
    a_fft = np.fft.rfft(a, n=m_fft)
    conv = np.fft.irfft(a_fft * a_fft, n=m_fft)[: 2 * a.size - 1]
    return conv


def _gradient_logsumexp(a: np.ndarray, a_fft: np.ndarray,
                        w: np.ndarray, m_fft: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed max.
    w – soft‑max weights (length 2·n‑1) that sum to 1.
    Returns ∇_a  (∑ w_i (a*a)_i ).
    """
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    # correlation of a with reversed weights (length 3·n‑2)
    corr = np.fft.irfft(a_fft * w_rev_fft, n=m_fft)[: 3 * a.size - 2]
    idx = 2 * a.size - 2 - np.arange(a.size)
    grad = 2.0 * corr[idx]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (length of the step function)
    # --------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = len(base_seq)
    except Exception:
        n = 1153  # default size used in the reference run

    # FFT size – next power‑of‑two that can hold any convolution we need
    m_fft = 1 << ((2 * n - 2).bit_length())

    # --------------------------------------------------------------
    # Objective (exact, no extra normalisation needed)
    # --------------------------------------------------------------
    def _objective(p: np.ndarray) -> float:
        """Exact objective 2·n·max(conv_self) / (sum(p)^2)."""
        s = p.sum()
        if s < 1e-12:
            return np.inf
        conv = _conv_self(p, m_fft)
        return float(2.0 * n * np.max(conv) / (s * s))

    # --------------------------------------------------------------
    # Build a diverse pool of promising initial vectors.
    # --------------------------------------------------------------
    init_vectors = []

    i = np.arange(n, dtype=float)

    # 1) Arcsine family with a few exponents (s = 0.4 … 0.6)
    for s_exp in (0.4, 0.45, 0.5, 0.55, 0.6):
        # a_i ∝ ((i+0.5)*(n-0.5-i))^{-s_exp}
        denom = ((i + 0.5) * (n - 0.5 - i) + 1e-12) ** s_exp
        p = 1.0 / denom
        p = p / p.sum()
        init_vectors.append(p)

    # 2) Geometric decays (several r values)
    for r in (0.99, 0.97, 0.95):
        geom = r ** i
        p = geom / geom.sum()
        init_vectors.append(p)

    # 3) Uniform distribution
    init_vectors.append(np.full(n, 1.0 / n))

    # 4) Random Dirichlet draws
    for _ in range(5):
        rnd = rng.random(n)
        rnd = rnd / rnd.sum()
        init_vectors.append(rnd)

    # 5) Historic construction, if it exists
    if 'height_sequence_1' in globals():
        seq = np.clip(np.array(height_sequence_1, dtype=float), 0.0, 1000.0)
        if seq.sum() > 0:
            init_vectors.append(seq / seq.sum())

    # --------------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # safety margin
    best_score = np.inf
    best_vec = None

    # --------------------------------------------------------------
    # Core optimisation loops (one per start vector)
    # --------------------------------------------------------------
    for start_id, p0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break

        print(f"\n=== Start {start_id + 1}/{len(init_vectors)} =====================")
        a = np.maximum(p0, 0.0)
        a = a / a.sum()                      # genuine probability vector

        # ------------------------------------------------------------------
        # (A) Exact sub‑gradient descent on the true max entry
        # ------------------------------------------------------------------
        step = 0.35
        decay = 0.9996
        min_step = 1e-7
        for it in range(25000):
            if time.time() >= deadline:
                break

            conv = _conv_self(a, m_fft)
            max_val = conv.max()
            cur = 2.0 * n * max_val / (a.sum() ** 2)

            if cur + 1e-12 < best_score:
                best_score = cur
                best_vec = a.copy()
                print(f"  Subgrad {it:5d} – new best {best_score:.6f}")
                if best_score <= 1.503:
                    break

            # active set (indices that are (almost) maximal)
            tol = 1e-12
            active = np.where(conv >= max_val - tol)[0]

            # sub‑gradient of the max entry (average over the active set)
            g = np.zeros_like(a)
            for idx in active:
                if idx < n:                               # left half of the convolution
                    g[: idx + 1] += 2.0 * a[idx::-1]
                else:                                     # right half
                    start = idx - n + 1
                    g[start:] += 2.0 * a[n - 1 : start - 1 : -1]
            g /= float(len(active))

            a = a - step * g
            a = _simplex_projection(a)
            step = max(step * decay, min_step)

        # ------------------------------------------------------------------
        # (B) Adam optimisation on a log‑sum‑exp smoothing of the max
        # ------------------------------------------------------------------
        m1 = np.zeros_like(a)
        m2 = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        base_lr = 0.07
        alpha = 1.0                 # smoothing factor
        alpha_mul = 1.001
        alpha_max = 250.0

        for it in range(300000):
            if time.time() >= deadline:
                break

            a_fft = np.fft.rfft(a, n=m_fft)
            conv = np.fft.irfft(a_fft * a_fft, n=m_fft)[: 2 * n - 1]

            # soft‑max (log‑sum‑exp) weights
            z = alpha * conv
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            w = exp_z / exp_z.sum()

            # gradient of the smoothed objective w.r.t. a
            g = _gradient_logsumexp(a, a_fft, w, m_fft)

            # Adam update (project after each step)
            m1 = beta1 * m1 + (1.0 - beta1) * g
            m2 = beta2 * m2 + (1.0 - beta2) * (g ** 2)

            m1_hat = m1 / (1.0 - beta1 ** (it + 1))
            m2_hat = m2 / (1.0 - beta2 ** (it + 1))

            lr = base_lr / np.sqrt(alpha)            # slower for large α
            step_vec = lr * m1_hat / (np.sqrt(m2_hat) + eps_adam)

            a = a - step_vec
            a = _simplex_projection(a)

            # increase smoothing steadily
            alpha = min(alpha * alpha_mul, alpha_max)

            # occasional exact evaluation
            if (it & 1023) == 0:
                cur = _objective(a)
                if cur + 1e-12 < best_score:
                    best_score = cur
                    best_vec = a.copy()
                    print(f"  Adam {it:6d} – new best {best_score:.6f}")
                    if best_score <= 1.503:
                        break

        # ------------------------------------------------------------------
        # (C) Simulated‑annealing / tiny random‑swap hill‑climber
        # ------------------------------------------------------------------
        # start the annealer from the best vector found so far for this run
        if best_vec is not None:
            a = best_vec.copy()
        cur_score = _objective(a)

        T = 0.02 * cur_score          # initial temperature
        T_decay = 0.99995
        move_scale = 0.02             # relative amount of mass moved per step

        for it in range(80000):
            if time.time() >= deadline:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if i == j or a[i] < 1e-12:
                continue

            delta = min(a[i], move_scale * rng.random())
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_score = _objective(a_new)
            diff = new_score - cur_score
            if diff < 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
                a = a_new
                cur_score = new_score
                if cur_score + 1e-12 < best_score:
                    best_score = cur_score
                    best_vec = a.copy()
                    print(f"  Anneal {it:5d} – new best {best_score:.6f}")
                    if best_score <= 1.503:
                        break
            T = max(T * T_decay, 1e-9)

        if best_score <= 1.503:
            break    # target reached – stop all starts

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)
        best_score = _objective(best_vec)

    print(f"\nFinal best score: {best_score:.6f}")

    # Clip to the bounds enforced by evaluate_sequence (0 ≤ x ≤ 1000)
    return np.clip(best_vec, 0.0, 1000.0).tolist()
```
