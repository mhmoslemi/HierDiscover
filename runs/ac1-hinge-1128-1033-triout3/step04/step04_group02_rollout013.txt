"""
Propose a step‑function with a low max‑convolution value.

Algorithm
---------
1. **Initialisation** – start from the previous best sequence (if it exists)
   or from a uniform distribution and normalise the sum to 1.  (The
   evaluation is scale‑invariant, so fixing Σa = 1 is convenient.)

2. **Smooth optimisation** – minimise a smooth log‑sum‑exp surrogate of
   the max‑convolution with several restarts of L‑BFGS‑B (continuation on
   ε).  This quickly drives the objective into the neighbourhood of a
   good local minimum.

3. **Projected sub‑gradient / mirror descent** – repeatedly locate the
   index k* where the convolution a∗a attains its maximum, compute the
   sub‑gradient (g_i = 2·a_{k*‑i}) and perform an exponentiated‑gradient
   (mirror) update a_i ← a_i·exp(−η·g_i).  The step‑size η decays as
   1/√t.  After each iteration we re‑normalise a to keep Σa = 1 and keep
   the best exact value seen so far.

4. **Local hill‑climbing** – small random mass‑transfer moves are tried
   until the time budget is exhausted; any improvement is kept.

The whole routine respects the 1000 s budget, prints progress and finally
returns the best list of heights found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negative values caused by rounding.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient wrt soft‑max params.
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Return (L_ε, ∂L_ε/∂w) for parameters w (soft‑max representation) and
    smoothing ε.
    """
    # soft‑max (Σa = 1)
    w_shift = w - np.max(w)               # for numeric stability
    a = np.exp(w_shift)
    a /= a.sum()

    # full self‑convolution
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smoothed max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # = log Σ exp((conv‑max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient wrt a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)         # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]       # middle n entries
    grad_a = 4.0 * n * seg[::-1]          # shape (n,)

    # chain rule through soft‑max
    dot = grad_a @ a
    grad_w = a * (grad_a - dot)           # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Mirror‑descent step for the exact max‑convolution objective.
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, eta: float):
    """
    Perform one exponentiated‑gradient (mirror‑descent) step that aims to
    decrease max_k (a∗a)[k].

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector (Σa = 1, a_i ≥ 0).
    eta : float
        Positive step‑size.

    Returns
    -------
    a_new : np.ndarray, shape (n,)
        Updated vector, still on the simplex.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    k_star = int(conv.argmax())          # index where the max occurs

    # sub‑gradient g_i = 2·a_{k_star‑i}  (zero outside the range)
    i = np.arange(n)
    j = k_star - i
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]

    # mirror‑descent (exponentiated update) and projection onto simplex
    a_new = a * np.exp(-eta * g)
    a_new_sum = a_new.sum()
    if a_new_sum == 0.0:                 # pathological, fall back to uniform
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= a_new_sum
    return a_new


# ----------------------------------------------------------------------
# Hill‑climbing refinement (random mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """Simple random mass‑transfer hill climbing, keeping the best value."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small random fraction of the mass at i towards j
        delta = cur_a[i] * 0.05 * rng.random()
        if delta == 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # exact evaluation (fast version, sum=1)
        max_conv = _full_convolution(new_a).max()
        val = 2.0 * n * max_conv

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a non‑negative step‑function heights sequence under the
    given evaluation function.  The routine respects a 1000 s budget,
    prints intermediate progress and finally returns the best list found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives a tighter theoretical bound.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise a from a previously saved construction, if any.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = np.full(n, 1.0 / n)    # uniform start

    a0 = np.maximum(a0, 0.0)
    a0_sum = a0.sum()
    if a0_sum == 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 /= a0_sum                            # Σa = 1

    # best seen so far
    best_a = a0.copy()
    best_val = 2.0 * n * _full_convolution(best_a).max()
    print(f"[init] n={n}, initial (exact) evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Smooth optimisation via L‑BFGS‑B (multiple restarts)
    # --------------------------------------------------------------
    n_restarts = 4
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-8), num=15)

    for restart in range(n_restarts):
        if time.time() >= deadline:
            break

        # initialise soft‑max parameters w
        if restart == 0:
            w = np.log(best_a + 1e-12)
        else:
            # random start – Gaussian around 0
            w = rng.normal(loc=0.0, scale=1.0, size=n)

        for eps in eps_schedule:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                L, g = _smooth_obj_and_grad(w_vec, eps, n)
                return L, g

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 250,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # exact evaluation of the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = 2.0 * n * _full_convolution(a).max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e} | restart {restart}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Mirror‑descent (sub‑gradient) phase – many cheap iterations
    # --------------------------------------------------------------
    a = best_a.copy()
    iter_cnt = 0
    eta0 = 1.0                # initial step size
    while time.time() < deadline - 5.0:   # keep a few seconds for refinement
        iter_cnt += 1
        eta = eta0 / np.sqrt(iter_cnt)

        # one mirror‑descent step
        a = _subgradient_step(a, eta)

        # evaluate exact objective (fast version)
        cur_val = 2.0 * n * _full_convolution(a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgrad iter {iter_cnt}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final hill‑climbing refinement (random mass transfers)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # Final report using the official evaluate_sequence for safety
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
