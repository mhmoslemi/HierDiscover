"""
Multi‑phase optimizer for the step‑function evaluation problem.

Phase 1 – Exponentiated‑gradient descent on a soft‑max approximation
          of the ∞‑norm of the (linear) self‑convolution.  This is
          essentially the “mirror‑descent” baseline but with a slightly
          different schedule and a stronger jitter term.

Phase 2 – Hard‑max hill‑climbing / simulated annealing.  After the
          temperature has been annealed to (almost) zero we focus on the
          actual maximal convolution coefficient.  One iteration picks a
          maximal index `k`, removes a small amount of mass from the
          position that contributes most to `c_k` and moves it to a
          random other position.  Improvements are always kept; worse
          moves are accepted with a slowly decreasing temperature.

Both phases keep the iterate on the probability simplex (non‑negative,
unit sum) and continuously track the best sequence seen.  The routine
runs until the supplied time budget (default 1000 s) expires and returns
the best list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using an FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative values caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient for the smooth (soft‑max) objective
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i}
    via a single convolution: reverse w and convolve with a.
    """
    n = a.shape[0]
    u = w[::-1]                       # reverse the weight vector
    conv = _full_convolution_fft(u, a)  # length 3n‑2
    start = 2 * n - 2                 # index of the first needed entry
    idx = start - np.arange(n)        # decreasing indices
    g = 2.0 * conv[idx]
    return g

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum

    # --------------------------------------------------------------
    # Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best_a = a.copy()
    # fast objective (exact because sum = 1)
    best_val = float(2.0 * n * _full_convolution_fft(a, a).max())
    print(f"[init] n={n}, initial objective={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent on a soft‑max bound
    # --------------------------------------------------------------
    # Hyper‑parameters (linear schedules)
    eta_start = 0.50
    eta_end   = 5e-5
    tau_factor_start = 0.35     # τ ≈ factor·max_conv initially
    tau_factor_end   = 5e-7     # final factor (practically hard max)
    beta = 0.94                 # momentum decay
    jitter_prob = 0.018

    momentum = np.zeros_like(a)
    eps = 1e-15                 # for projection

    # allocate ~60 % of the budget to phase 1
    phase1_deadline = start_time + 0.60 * (deadline - start_time)
    it = 0
    while time.time() < phase1_deadline:
        # current convolution and objective
        conv_a = _full_convolution_fft(a, a)
        M = conv_a.max()
        cur_val = float(2.0 * n * M)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[EG {it:7d}] new best = {best_val:.6f}")

        # progress fraction (0 .. 1) for schedule interpolation
        progress = (time.time() - start_time) / (deadline - start_time)

        # ---- temperature for the soft‑max (annealed) ----
        tau = (tau_factor_start * (1.0 - progress) +
               tau_factor_end   * progress) * M
        if tau <= 0.0:
            # fall back to a hard max‑weight
            w = np.zeros_like(conv_a)
            w[np.argmax(conv_a)] = 1.0
        else:
            # stable soft‑max: shift by the maximum to avoid overflow
            exp_args = (conv_a - M) / tau
            np.clip(exp_args, -80, 80, out=exp_args)
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv_a)
                w[np.argmax(conv_a)] = 1.0
            else:
                w = exp_vals / sum_exp

        # gradient of the smoothed objective (before scaling)
        grad = _gradient_from_weights(a, w)

        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # step‑size schedule (multiplicative)
        eta = eta_start * (1.0 - progress) + eta_end * progress

        # exponentiated‑gradient multiplicative update
        step = -eta * momentum
        np.clip(step, -50, 50, out=step)          # avoid overflow/underflow
        a = a * np.exp(step)

        # occasional random jitter to escape plateaux
        if np.random.rand() < jitter_prob:
            src = np.random.randint(n)
            dst = np.random.randint(n - 1)
            if dst >= src:
                dst += 1
            # move a random fraction (up to 30 %) of the mass at src to dst
            delta = np.random.rand() * a[src] * 0.30
            a[src] -= delta
            a[dst] += delta

        # projection back onto the simplex (renormalise)
        a = np.maximum(a, eps)
        a_sum = a.sum()
        a = a / a_sum

        it += 1
        # keep the convolution numerically clean from time to time
        if it % 2500 == 0:
            _ = _full_convolution_fft(a, a)

    # --------------------------------------------------------------
    # Phase 2 – hard‑max hill climbing with simulated annealing
    # --------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 0.0:
        # temperature schedule for the annealing moves
        temp_start = 1e-2
        temp_end   = 1e-5
        iter_hc = 0
        while time.time() < deadline:
            # recompute convolution for the current iterate
            conv_a = _full_convolution_fft(a, a)
            M = conv_a.max()
            cur_val = float(2.0 * n * M)

            # pick one of the maximal indices uniformly
            max_idxs = np.where(conv_a == M)[0]
            k = np.random.choice(max_idxs)

            # contribution of each position to c_k = a_i * a_{k-i}
            contrib = np.zeros(n)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            if i_min <= i_max:
                idxs = np.arange(i_min, i_max + 1, dtype=int)
                contrib[idxs] = a[idxs] * a[k - idxs]

            # choose a source index with the largest contribution
            if contrib.sum() > 0.0:
                i_src = int(np.argmax(contrib))
            else:
                i_src = np.random.randint(n)

            # choose a destination index distinct from the source
            i_dst = np.random.randint(n - 1)
            if i_dst >= i_src:
                i_dst += 1

            # amount to move – up to 10 % of the source mass
            eps_move = np.random.rand() * a[i_src] * 0.10
            eps_move = min(eps_move, a[i_src])   # stay non‑negative

            if eps_move <= 0.0:
                # nothing to move; skip this iteration
                continue

            a_new = a.copy()
            a_new[i_src] -= eps_move
            a_new[i_dst] += eps_move

            # safety – clip tiny negatives that could appear from rounding
            a_new = np.maximum(a_new, 0.0)
            # renormalise (the total mass is still ~1, but rounding may drift)
            a_new_sum = a_new.sum()
            if a_new_sum == 0.0:
                continue
            a_new /= a_new_sum

            # evaluate the candidate
            conv_new = _full_convolution_fft(a_new, a_new)
            M_new = conv_new.max()
            new_val = float(2.0 * n * M_new)

            # always accept improvements
            if new_val < cur_val:
                a = a_new
                if new_val < best_val:
                    best_val = new_val
                    best_a = a_new.copy()
                    print(f"[HC {iter_hc:7d}] new best = {best_val:.6f}")
            else:
                # occasional uphill moves (simulated annealing)
                prog = (time.time() - (deadline - remaining_time)) / remaining_time
                temp = temp_start * (1.0 - prog) + temp_end * prog
                accept_prob = math.exp(-(new_val - cur_val) / (temp + 1e-12))
                if np.random.rand() < accept_prob:
                    a = a_new

            iter_hc += 1
            # occasional full recomputation for numerical stability
            if iter_hc % 3000 == 0:
                _ = _full_convolution_fft(a, a)

    # --------------------------------------------------------------
    # Final reporting (the competition harness will also call it)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                         # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # return a plain list of floats
    return best_a.tolist()
