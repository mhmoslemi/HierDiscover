"""
propose_candidate
-----------------

A two‑phase search for a non‑negative step function (probability vector)
minimising 2·n·max_k (a*a)[k] :

1. Deterministic parametric sweep on several families (U‑shape based,
   product‑shape, exponential, and a new “arcsine‑like’’ family).  The
   sweep covers many lengths (up to 20000) and a fine grid of the shape
   parameters, keeping the best vector seen.

2. Local polishing:
   * a simple self‑adaptive Differential Evolution (jDE) on the simplex
     (population ≈ 60, per‑individual F/CR), run only while time permits;
   * a projected sub‑gradient descent (hard max) with a decaying step
     size, starting from the best vector found by the sweep.
   * occasional “re‑heat’’ moves to escape stagnation.

Both phases respect the 1000 s budget and always return a vector whose
entries are non‑negative and sum to 1, which can be passed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex:
        { w >= 0 , sum(w) = 1 }
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution & scoring utilities
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    size = 2 * n                     # zero‑padding length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """
    Exact value returned by evaluate_sequence for a vector that already
    satisfies sum(a) == 1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Contributions come from all indices where the convolution attains
    (or almost attains) its maximum.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        # each pair (i,j) contributes 2*a[j] to ∂/∂a[i]
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric families (used for the deterministic sweep)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+1)^{-p}   (symmetric, edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{-p}   (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))   ( symmetric exponential )."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def arcsine_distribution(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """
    a[i] ∝ ((i+offset+1)*(n-i+offset))^{-p}
    This family contains the classic arcsine‑like shape when p≈0.5
    and offset≈0.0.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1.0 + offset) * (n - i + offset)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a
