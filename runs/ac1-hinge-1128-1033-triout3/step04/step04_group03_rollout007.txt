def correlation(a, w):
    n = a.shape[0]
    m = w.shape[0]   # 2n-1
    L = 1 << ( (n + m - 1).bit_length() )
    a_rev = a[::-1]
    # compute FFT
    Fa = np.fft.rfft(a_rev, L)
    Fw = np.fft.rfft(w, L)
    conv = np.fft.irfft(Fa * Fw, L)
    # conv length L, need slice n-1:n-1+n
    grad = conv[n-1:n-1+n]   # shape n array
    return grad
```

    deadline = time.time() + budget_s - 5.0

    # define functions: (as top-level defined outside propose_candidate)
    # We'll place them earlier in code.

    # deterministic sweep
    best_seq, best_score = deterministic_sweep(...)

    # gradient descent
    cur = best_seq.copy()
    n = cur.shape[0]
    cur = np.maximum(cur, 1e-14)
    cur /= cur.sum()

    cur_conv = full_convolution_fft(cur)
    cur_score = float(2.0*n*cur_conv.max())

    # hyperparameters
    eta = 0.2
    eta_decay = 0.99995
    alpha = 20.0
    alpha_mult = 1.001
    alpha_max = 2000.0
    grad_ema = np.zeros_like(cur)
    beta = 0.9
    temperature = 0.1
    temp_decay = 0.99995
    iter_count = 0
    last_best_iter = 0

    while time.time() < deadline:
        iter_count += 1

        # compute conv (already have cur_conv)
        conv = cur_conv
        max_c = conv.max()
        # compute weight vector w (softmax)
        # using shift to avoid overflow
        shift = conv - max_c  # <=0

        # ensure minimal positivity

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # small probability to accept uphill
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = cand
            cur_conv = cand_conv
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()

The evaluator computes  

\[
\text{score}= \frac{2\,n\;\max_k(a*a)[k]}{(\sum_i a_i)^2}\,,
\]

so the problem is *scale–invariant*: we may normalise the step‑function to a probability vector
(`sum(a)=1`).  
The goal is therefore to minimise  

\[
\Phi(a)=2\,n\;\max_k (a*a)[k]\qquad\text{with}\qquad a_i\ge 0,\;\sum_i a_i=1 .
\]

The code below follows a two‑phase strategy

1. **Deterministic sweep** – a fast scan over several parametric, symmetric and
   asymmetric families (power‑law “U‑shapes”, shifted U‑shapes, product‑distance,
   exponential, cosine‑based and monotone power‑law).  
   The best vector of the sweep seeds the stochastic phase.

2. **Smooth exponential‑gradient descent** – we minimise a smooth surrogate of the
   max‑operator  

   \[
   f_\alpha(a)=\frac{1}{\alpha}\log\!\Big(\sum_k e^{\alpha (a*a)[k]}\Big),
   \]

   whose gradient can be written as a *cross‑correlation* that is computed
   with a single FFT.  
   Mirror‑descent on the simplex (exponentiated‑gradient update) together with
   a slowly decaying step‑size, a growing smoothing parameter `α`,
   a tiny simulated‑annealing temperature and occasional random re‑heating
   drives the search to better solutions.

All heavy work is performed with `numpy`’s FFT (real‑valued) – a full
convolution of length `2·n‑1` costs only a few milliseconds even for
`n≈20000`.  The algorithm respects the 1000 s time budget and returns the
best sequence it has found.

"""
propose_candidate – optimise a non‑negative step function to minimise
2 * n * max_k (a * a)[k].

Algorithm
---------
1. Deterministic sweep over several parametric families (U‑shapes,
   shifted U‑shapes, product‑distance, exponential, cosine and a monotone
   power‑law).  The best vector of the sweep seeds the stochastic phase.

2. Smooth optimisation:
   * minimise the log‑sum‑exp surrogate of the max‑convolution,
   * use the gradient (cross‑correlation) computed via a real FFT,
   * perform an exponentiated‑gradient (mirror descent) update on the
     probability simplex,
   * gradually increase the smoothness parameter α, slowly decay the
     step‑size η and a tiny simulated‑annealing temperature,
   * occasional random re‑heating prevents stagnation.

The routine stops before the given deadline and returns the best
sequence (as a plain Python list) it discovered.

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex { w ≥ 0 , sum(w)=1 }.
    Implementation follows Wang & Carreira‑Perpiñán (O(n log n)).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full length 2*len(a)-1) using a real FFT.
    """
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def cross_correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the cross‑correlation (a ⋆ w)[k] = Σ_i a_i w_{i+k}
    for k = 0 … len(a)-1 using a real FFT.
    Both a and w are assumed real and non‑negative.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2*n-1
    L = 1 << ((n + m - 1).bit_length())   # next power of two
    a_rev = a[::-1]
    Fa = np.fft.rfft(a_rev, L)
    Fw = np.fft.rfft(w, L)
    conv = np.fft.irfft(Fa * Fw, L)
    start = n - 1                 # slice that gives (a ⋆ w)[0 … n-1]
    return conv[start:start + n]


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact score returned by evaluate_sequence when Σ a = 1.
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())

# Parametric families (all return a normalised probability vector)

def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))⁻p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


def monotone_powerlaw(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)⁻p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p) + 1e-14
    a /= a.sum()
    return a


def linear_ramp(n: int) -> np.ndarray:
    """Simple linear drop from the left edge to the right."""
    a = np.linspace(1.0, 0.0, n) + 1e-14
    a /= a.sum()
    return a
