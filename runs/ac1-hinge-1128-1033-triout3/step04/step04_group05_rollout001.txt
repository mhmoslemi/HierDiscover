"""
Propose a step‑function height sequence that minimises
evaluate_sequence(sequence).

The algorithm is a two‑phase optimiser:

1. **Mirror‑descent on the simplex**  
   * Variables are kept in the log‑softmax (x = log a) so the
     constraints a_i ≥ 0 and Σ a_i = 1 are always satisfied.
   * At each iteration we compute the exact sub‑gradient of the
     objective   f(a)=2·n·max_k (a*a)[k] .
   * The sub‑gradient is obtained by averaging the gradients of the
     convolution entries that attain (or are very close to) the maximum.
   * An Adam optimiser with a simple time‑based learning‑rate schedule
     drives the updates.

2. **Stochastic local search (simulated‑annealing style)**  
   * When 80 % of the time budget has elapsed we switch to a fast
     random‑pair mass‑transfer heuristic.
   * A small amount of probability mass is moved from one index to another.
   * The move is always accepted if it improves the objective, otherwise
     it may be accepted with a temperature‑dependent probability.
   * The current best solution is kept throughout.

Both phases keep a copy of the best sequence seen so far and return it
when the time budget expires.  The implementation relies only on NumPy,
uses FFT‑based convolutions for speed and respects the required
non‑negative, finite input format.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full convolution (real input → real output)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max (or exact sub‑gradient) w.r.t. a
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via one convolution: reverse w, convolve with a.
    a : shape (n,)
    w : shape (2n-1,)
    Returns an array of length n.
    """
    n = a.shape[0]
    u = w[::-1]                              # reversed weight vector
    conv = _full_convolution_fft(u, a)       # length = 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (used only for sanity)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("vector expected")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Initialise a “arcsine” shaped distribution (heavy at the ends)
# ----------------------------------------------------------------------
def _init_arcsine(n: int) -> np.ndarray:
    """Return a normalised arcsine‑like vector of length n."""
    i = np.arange(n, dtype=np.float64)
    # shift by 0.5 to avoid division by zero at the boundaries
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Objective value for a probability vector a (sum(a)=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max_k (a*a)[k]  (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()

                    best_val = val_new
                    best_a = a.copy()
                    print(f"[SA {iteration:7d}] new best = {best_val:.6f}")

            iteration += 1

        # ------------------------------------------------------------------
        # Safety: break if we have run an absurd number of iterations
        # ------------------------------------------------------------------
        if iteration > 10_000_000:
            break

    # ------------------------------------------------------------------
    # Final reporting (framework may also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list of floats
    return best_a.tolist()
```
