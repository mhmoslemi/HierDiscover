def refine_lp(f, deadline, max_iter=10, t_init=0.2):
    n = len(f)
    rows, cols = build_convolution_matrix_structure(n)
    best_f = f.copy()
    best_score = score_vector(f)
    t = t_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        conv = full_convolution_fft(f)
        M = conv.max()
        data = np.repeat(f, n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
        # maximize sum(b) => minimize -sum(b)
        c = -np.ones(n)
        b_ub = np.full(2*n - 1, M, dtype=np.float64)
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                          method='highs', options={'presolve': True})
        if not res.success:
            print(f"LP iteration {it} failed: {res.message}")
            break
        b_vec = res.x
        s = b_vec.sum()
        if s <= 0:
            break
        g = b_vec / s
        f_new = (1 - t) * f + t * g
        # renormalize just in case
        f_new = np.maximum(f_new, 0)
        f_new /= f_new.sum()
        new_score = score_vector(f_new)
        if new_score + 1e-12 < best_score:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            print(f"LP iter {it} improved score to {new_score:.6f}")
            # maybe increase t slightly?
            t = min(t * 1.05, 0.5)
        else:
            # No improvement, reduce step
            t *= 0.5
            if t < 1e-3:
                break
    return best_f, best_score

def subgradient_polish(f, deadline, max_iter=20000):
    n = len(f)
    best_f = f.copy()
    best_score = score_vector(f)
    step = 0.2
    step_decay = 0.9996
    it = 0
    last_improve_it = 0
    while time.time() < deadline and step > 1e-8 and it < max_iter:
        it += 1
        conv = full_convolution_fft(f)
        grad = compute_subgradient(f, conv)
        cand = f - step * grad
        cand = np.maximum(cand, 0)
        cand = simplex_projection(cand)
        cand /= cand.sum()
        cand_score = score_vector(cand)

            best_f = cand.copy()
            f = cand

def refine_lp(f, deadline, max_iter=10, t_init=0.2):
    n = len(f)
    rows = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    best_f = f.copy()
    best_score = score_vector(f)
    t = t_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        conv = full_convolution_fft(f)
        M = conv.max()
        data = np.repeat(f, n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"LP iter {it} failed: {res.message}")
            break
        b = res.x
        s = b.sum()
        if s <= 0:
            break
        g = b / s
        # mix
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        f_new /= f_new.sum()
        new_score = score_vector(f_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            print(f"LP iteration {it}, improved to {new_score:.6f}")
            # maybe increase step slightly for next iteration
            t = min(t * 1.05, 0.5)
        else:
            # reduce step and maybe break
            t *= 0.5
            if t < 1e-4:
                break
    return best_f, best_score

def subgradient_polish(f, deadline, max_iters=50000):
    n = len(f)
    best_f = f.copy()
    best_score = score_vector(f)
    step = 0.2
    decay = 0.9996
    it = 0
    last_improve = 0
    while time.time() < deadline and step > 1e-8 and it < max_iters:
        it += 1
        conv = full_convolution_fft(f)
        grad = compute_subgradient(f, conv)
        cand = f - step * grad
        cand = np.maximum(cand, 0)
        cand = simplex_projection(cand)
        cand /= cand.sum()
        cand_score = score_vector(cand)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_f = cand.copy()
            f = cand
            last_improve = it
            print(f"Subgrad it {it}, new best {best_score:.6f}")
        # step decay
        step *= decay
        if it - last_improve > 3000:
            bump = np.random.rand(n) * 1e-5
            f = simplex_projection(f + bump)
            f /= f.sum()
            step = max(step, 0.15)
            last_improve = it
            print(f"Subgrad it {it}: reheated")
    return best_f, best_score

#!/usr/bin/env python
"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimises

    2 * n * max_k (a * a)[k]

where * denotes the full linear convolution.  The algorithm works in three
stages:

1. **Deterministic sweep** over a variety of symmetric parametric families
   (inverse distance, inverse product distance, exponential decay and a
   shifted version) for several lengths.  This quickly finds a good
   starting shape.

2. **LP refinement** (inspired by the Frank‑Wolfe / linear‑programming
   construction from the paper).  For a fixed vector f we
   maximise Σ b_j under the linear constraints (f * b)_k ≤ M
   (M = max (f * f)).  The optimiser returns b with a larger ℓ₁–norm,
   which after normalisation gives a new direction g that strictly
   reduces the objective when mixed with f.  Only moderate dimensions
   (n ≤ 3000) are treated in this step to keep the constraint matrix
   manageable.

3. **Projected sub‑gradient polishing** – a simple but effective
   first‑order method that directly minimises the max‑convolution
   objective while staying on the probability simplex.

All three phases respect the global 1000 s budget and print progress.
The final vector is returned as a plain Python list (∑a ≈ 1, a_i ≥ 0).
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w≥0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = 2 * n                      # zero‑padding length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """
    Exact objective value returned by evaluate_sequence for a vector with
    sum(a) == 1 (i.e. 2 * n * max_k (a*a)[k]).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices where the convolution attains (or almost attains) its
    maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric “U‑shaped’’ families (used for the deterministic seed)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+1)^{-p}   (symmetric, edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{-p}   (edge‑heavy, product of distances)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))   (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def shifted_min_distribution(n: int, p: float, c: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+c)^{-p}  (extra shift to tune flatness)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Deterministic sweep – cheap but gives a very good seed
# ----------------------------------------------------------------------
def deterministic_sweep(deadline: float):
    best_score = np.inf
    best_vec = None
    best_n = None

    # length candidates (small enough for quick FFTs, large enough to
    # explore the scaling factor 2·n)
    lengths = [1000, 2000, 3000, 4000, 5000, 6000, 8000, 10000, 12000]

    for n in lengths:
        if time.time() > deadline:
            break
        #--- inverse‑distance families ---------------------------------
        for p in np.linspace(0.4, 2.5, 12):
            a = u_min_distribution(n, p)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec, best_n = sc, a.copy(), n
                print(f"[sweep] new best (u_min)  n={n:5d} p={p:5.3f}  score={sc:.6f}")

        for p in np.linspace(0.4, 2.5, 12):
            a = u_prod_distribution(n, p)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec, best_n = sc, a.copy(), n
                print(f"[sweep] new best (u_prod) n={n:5d} p={p:5.3f}  score={sc:.6f}")

        #--- exponential families ---------------------------------------
        for lam in np.logspace(-4, -1, 10):
            a = exp_distribution(n, lam)
            sc = score_vector(a)
            if sc < best_score:
                best_score, best_vec, best_n = sc, a.copy(), n
                print(f"[sweep] new best (exp)    n={n:5d} λ={lam:.5e}  score={sc:.6f}")

        #--- shifted min families ---------------------------------------
        for c in (0.01, 0.1, 0.5):
            for p in (0.8, 1.2, 1.6):
                a = shifted_min_distribution(n, p, c)
                sc = score_vector(a)
                if sc < best_score:
                    best_score, best_vec, best_n = sc, a.copy(), n
                    print(f"[sweep] new best (shift) n={n:5d} p={p:.1f} c={c:.2f} score={sc:.6f}")

    return best_vec, best_score, best_n


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step’’ from the paper)
# ----------------------------------------------------------------------
def refine_lp(f: np.ndarray, deadline: float, max_iter: int = 12, t_init: float = 0.25):
    """
    Perform a few LP refinement iterations.
    Each iteration solves:
        maximise Σ b_j   subject to   f * b ≤ M   and   b ≥ 0,
    where M = max (f*f).  The solution b is normalised,
    mixed with the current vector f on the simplex, and accepted
    if it reduces the objective.
    """
    n = len(f)
    # Build the *fixed* sparsity pattern of the convolution matrix once.
    # rows[k,i] = i + k  (where i = column index)
    rows = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    cols = np.repeat(np.arange(n, dtype=np.int32), n)

    best_f = f.copy()
    best_score = score_vector(f)
    t = t_init

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # current bound M
        conv = full_convolution_fft(f)
        M = conv.max()

        # matrix A such that (A @ b) = f * b
        data = np.repeat(f, n)
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # solve LP (maximize Σ b_j  ⇔  minimize -Σ b_j)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})

        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        s = b.sum()
        if s <= 0.0:
            break

        g = b / s                     # normalise to sum = 1
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()

        new_score = score_vector(f_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            print(f"[LP] iter {it:2d} improved → score {new_score:.6f}")
            # slightly enlarge the mixing factor for the next round
            t = min(t * 1.05, 0.5)
        else:
            # no progress – shrink step size
            t *= 0.5
            if t < 1e-4:
                break

    return best_f, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing
# ----------------------------------------------------------------------
def subgradient_polish(f: np.ndarray, deadline: float, max_iters: int = 50000):
    n = len(f)
    best_f = f.copy()
    best_score = score_vector(f)

    step = 0.20                # initial step size
    decay = 0.9996
    it = 0
    last_improve = 0

    while time.time() < deadline and step > 1e-8 and it < max_iters:
        it += 1
        conv = full_convolution_fft(f)
        grad = compute_subgradient(f, conv)

        cand = f - step * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()

        cand_score = score_vector(cand)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_f = cand.copy()
            f = cand
            last_improve = it
            print(f"[subgrad] it {it:5d} → new best {best_score:.6f}")

        # decay step size
        step *= decay

        # if we have been stuck for a while, add a tiny random nudge
        if it - last_improve > 3000:
            bump = np.random.rand(n) * 1e-5
            f = simplex_projection(f + bump)
            f /= f.sum()
            step = max(step, 0.15)
            last_improve = it
            print(f"[subgrad] it {it:5d} – reheated")

    return best_f, best_score


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the external
# evaluator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise the
    evaluation function.  The routine respects a 1000 s budget, prints
    progress, and finally returns the best vector found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 0.  Optional external seed (height_sequence_1)
    # ------------------------------------------------------------------
    external_vec = None
    try:
        external_vec = np.array(height_sequence_1, dtype=float)
        external_vec = np.maximum(external_vec, 0.0)
        if external_vec.sum() > 0:
            external_vec /= external_vec.sum()
            ext_score = score_vector(external_vec)
            print(f"[seed] external vector score = {ext_score:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1. Deterministic sweep for a strong initial guess
    # ------------------------------------------------------------------
    best_vec, best_score, best_n = deterministic_sweep(deadline)

    # incorporate external seed if it beats the sweep result
    if external_vec is not None:
        ext_score = score_vector(external_vec)
        if ext_score < best_score:
            best_vec, best_score, best_n = external_vec.copy(), ext_score, len(external_vec)
            print(f"[seed] using external vector (score={ext_score:.6f})")

    print(f"\n=== Selected seed: n={best_n}, score={best_score:.6f} ===")

    # ------------------------------------------------------------------
    # 2. LP refinement (only for modest dimensions to keep memory low)
    # ------------------------------------------------------------------
    LP_MAX_N = 3000
    if best_vec.shape[0] <= LP_MAX_N:
        print("[refine] starting LP refinement")
        best_vec, best_score = refine_lp(best_vec, deadline,
                                         max_iter=12, t_init=0.25)
        print(f"[refine] after LP: score={best_score:.6f}, n={len(best_vec)}")
    else:
        print(f"[refine] skipping LP (n={len(best_vec)} > {LP_MAX_N})")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient polishing for final fine‑tuning
    # ------------------------------------------------------------------
    print("[polish] starting sub‑gradient polishing")
    best_vec, best_score = subgradient_polish(best_vec, deadline,
                                              max_iters=40000)
    print(f"\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f} (length {len(best_vec)})")

    # Return as plain Python list – evaluate_sequence will handle the final
    # validation and scaling.
    return best_vec.tolist()
