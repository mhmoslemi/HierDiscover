"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a two‑phase heuristic:

1.  Adam‑style projected gradient descent with a soft‑max weighting of the
    convolution entries.  This quickly reduces the dominant peaks while
    keeping the vector on the probability simplex.

2.  A fast random “swap” local search that moves a tiny amount of mass
    between two coordinates and keeps the change only if it improves the
    score.  The remaining time budget is spent here, allowing the algorithm
    to escape the small plateau where the smooth gradient becomes too
    flat.

Both phases finish with a short max‑entry sub‑gradient polishing step.
The whole routine respects the 1000 s time limit and returns the best
sequence discovered before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project ``v`` onto the simplex {x | x >= 0, sum(x) = 1}
    (Duchi et al. 2008).  Works for 1‑D arrays.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution via FFT (full linear convolution).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of 1‑D real arrays ``a`` and ``b`` using real FFT.
    """
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

    # ------------------------------------------------------------------
    # Phase 2 – random pairwise mass swaps (remaining time)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    best_val_local = best_val
    max_step = 0.02                     # maximal fraction of a[i] to move

        if i == j:

        if delta <= 0.0:
            continue

        a_candidate[i] -= delta
        a_candidate[j] += delta

        # compute new score
        conv_cand = _full_convolution_fft(a_candidate, a_candidate)
        cand_score = 2.0 * n * conv_cand.max()

        # accept only if it improves the best known value
        if cand_score < best_val_local - 1e-12:
            best_val_local = cand_score
            best_seq = a_candidate.copy()

            max_step *= 0.99                     # gradually fine‑tune
            print(f"[local] new_best={best_val_local:.12f}, max_step={max_step:.5f}")
        else:

            if rng.random() < 0.001:

    # Final polishing: deterministic sub‑gradient steps on the max entry.
    # ------------------------------------------------------------------

        mx = conv.max()
