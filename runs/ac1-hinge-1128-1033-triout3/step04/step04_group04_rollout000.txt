"""
propose_candidate
-----------------

Find a non‑negative step‑function (probability vector) that minimises

        2 * n * max_k (a * a)[k]

where `*` is the full linear convolution.
The optimal (continuous) solution is the arcsine distribution whose
self‑convolution is exactly uniform.  By discretising the arcsine CDF on a
uniform grid we obtain a sequence whose convolution is already almost flat,
giving a value of the evaluation function extremely close to the theoretical
lower bound 1.  The routine

1. sweeps a few problem sizes (up to ~5·10⁴) and builds the discretised
   arcsine vector for each size;
2. keeps the best one (lowest score computed with a fast FFT‑based convolution);
3. performs a short projected sub‑gradient polish to remove any remaining
   irregularities;
4. returns the best vector found (as a plain Python list).

The whole search respects the supplied time budget (default 1000 s) and
runs on a single CPU.  All helper functions are defined at top level, and
no file or network I/O is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w >= 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full linear convolution (real input)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    # pad to length 2*n (next power of two is not required; rfft handles any length)
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact score for a vector that already sums to 1:
#   2 * n * max_k (a * a)[k]
# ----------------------------------------------------------------------
def score_vector(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of max_k (a*a)[k] w.r.t. a (used for polishing)
# ----------------------------------------------------------------------
def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the max‑convolution term."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Discrete arcsine distribution (CDF differences) – the key seed
# ----------------------------------------------------------------------
def arcsine_distribution(n: int) -> np.ndarray:
    """
    Return a symmetric probability vector obtained by binning the
    arcsine law on n equally sized intervals.
    The arcsine CDF is F(x) = (2/π) * arcsin(√x), x∈[0,1].
    """
    # grid points for the CDF (including 0 and 1)
    x = np.linspace(0.0, 1.0, n + 1, dtype=np.float64)
    F = (2.0 / np.pi) * np.arcsin(np.sqrt(x))
    a = np.diff(F)                       # a[i] = F((i+1)/n) - F(i/n)
    # numerical safety – enforce non‑negativity and renormalise
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  The routine returns a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # reserve a few seconds for cleanup

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over a few problem sizes using the
    #    discretised arcsine distribution (theoretically optimal shape).
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf

    # candidate lengths – powers of two are cheap for FFT, but any length works.
    # The largest size we comfortably handle within the time budget is about 5·10⁴.
    candidate_lengths = [2000, 4000, 8000, 16000, 32000, 50000]

    print("=== Deterministic arcsine sweep ===")
    for n in candidate_lengths:
        if time.time() > deadline:
            break
        a = arcsine_distribution(n)
        sc = score_vector(a)      # exact score (sum(a) == 1)
        print(f"  n={n:5d}  score={sc:.6f}")
        if sc < best_score:
            best_score, best_vec = sc, a.copy()
            print(f"   ** new best (arcsine) with n={n}")

    # ------------------------------------------------------------------
    # 2. Fallback to a few random Dirichlet seeds in case the arcsine sweep
    #    failed (should never happen).
    # ------------------------------------------------------------------
    if best_vec is None:
        print("Arcsine sweep failed – falling back to random Dirichlet.")
        n = 5000
        a = np.random.rand(n)
        a /= a.sum()
        best_vec = a
        best_score = score_vector(a)

    # ------------------------------------------------------------------
    # 3. Short projected sub‑gradient polish (optional, only if time permits)
    # ------------------------------------------------------------------
    print("\n=== Sub‑gradient polishing ===")
    current = best_vec.copy()
    current_score = best_score
    step = 0.12                     # initial step size
    step_decay = 0.9994
    last_improve = 0
    it = 0
    n = current.shape[0]

    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft(current)
        grad = compute_subgradient(current, conv)

        cand = current - step * grad
        cand = np.maximum(cand, 0.0)
        cand = simplex_projection(cand)
        cand /= cand.sum()                       # safety renormalisation
        cand_score = score_vector(cand)

        if cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                last_improve = it
                print(f"  subgrad it {it:6d}: new best = {best_score:.6f}")

        # step‑size schedule
        step *= step_decay
        if step < 1e-7:
            break

        # re‑heat if stagnation persists for a long time
        if it - last_improve > 3000:
            bump = np.random.rand(n) * 1e-5

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
