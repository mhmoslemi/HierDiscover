"""Step‑function optimiser.

The algorithm explores a diverse set of initial probability vectors
(arcsine, uniform, geometric, several beta‑shapes and random Dirichlet
samples).  For each start it first performs a short Adam optimisation on
a log‑sum‑exp smooth approximation of the maximal autocorrelation.
After that a long projected sub‑gradient descent on the *exact* max
convolution entry is executed – the step size decays slowly and
stays at a tiny constant for many iterations, allowing fine‑grained
improvements beyond the previous best (≈1.517).  Finally a cheap
pair‑swap hill‑climber polishes the result.  The best vector found
within the time budget is returned as a plain Python list.
"""

import numpy as np
import time

# --------------------------------------------------------------
# Simplex projection (Euclidean) – keeps a vector non‑negative and sum‑1.
# --------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Full self‑convolution using a pre‑computed FFT size.
# --------------------------------------------------------------
def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Return the linear (full) self‑convolution of p."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


# --------------------------------------------------------------
# Gradient of the log‑sum‑exp smoothed max.
# w – soft‑max weights (length 2·n‑1) that sum to 1.
# --------------------------------------------------------------
def _gradient_from_weights(p: np.ndarray, p_fft: np.ndarray,
                          w: np.ndarray, m_fft: int) -> np.ndarray:
    """Gradient of the smoothed max with respect to p."""
    w_rev = w[::-1]                       # reversed weight vector
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * p.size - 2]
    # indices that map the correlation result back to p‑gradient
    idx = 2 * p.size - 2 - np.arange(p.size)
    grad = 2.0 * corr[idx]
    return grad
