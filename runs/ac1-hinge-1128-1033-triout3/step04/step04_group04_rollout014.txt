def combined_distribution(n, w1, w2, w3, p1, p2):
    a1 = (np.minimum(np.arange(n), n-1-np.arange(n)) + 1) ** (-p1)

def combined_distribution(n,w1,w2,w3,p1,p2):
    d = np.arange(n)

def full_convolution_fft(a):
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)

Propose a step‑function (non‑negative list) that minimises the score
2 * n * max_k (a * a)[k] (the evaluate_sequence objective).

Algorithm
---------

1. **Broad deterministic sweep** – evaluate a large family of simple
   parametric shapes (inverse‑distance, inverse‑product‑distance,
   symmetric exponential and a 5‑parameter mixture of those) over a
   range of lengths.  Keep the best vector found.

2. **Differential Evolution on the 5‑parameter mixture** – treat the
   mixture coefficients and the two exponents as a low‑dimensional
   optimisation problem.  The population lives in ℝ⁵, each individual is
   turned into a full probability vector of the chosen length and
   scored with a fast FFT‑based convolution.

3. **Projected sub‑gradient polishing** – using the exact
   sub‑gradient of the max‑convolution, perform a projected gradient
   descent on the simplex to squeeze out the last bits of improvement.

4. **Optional hill‑climbing (simulated‑annealing‑style) fallback** –
   if time permits, a simple random‑perturbation search is run after
   polishing.

The routine respects the 1000 s budget, prints progress and finally
returns the best list of floats it discovered.

# ----------------------------------------------------------------------
# 1. Simplex projection (Wang & Carreira‑Perpiñán, O(n log n))
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto {w >= 0, sum(w) = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0
    if rho.size:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2. Convolution utilities (full linear convolution using real FFT)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 2 * n                     # zero‑padding length
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_vector(a: np.ndarray) -> float:
    """Exact score for a probability vector (sum == 1)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only indices where the convolution attains (or almost attains) its
    maximum contribute.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

# 3. Parametric families of step functions
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ (min(i, n‑1‑i)+1)^{-p}  (edge‑heavy)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d) + 1.0
    a = d ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """a[i] ∝ ((i+1)*(n‑i))^{-p}  (edge‑heavy, product distance)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1.0) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """a[i] ∝ exp(-lam*i) + exp(-lam*(n‑1‑i))  (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def combined_distribution(n: int,
                         w1: float, w2: float, w3: float,
                         p1: float, p2: float) -> np.ndarray:
    """
    5‑parameter mixture:
        w1 * (min(i,n-1-i)+1)^{-p1} +
        w2 * ((i+1)*(n-i))^{-p2}   +
        w3 * 1
    Normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)

    d = np.minimum(i, n - 1 - i) + 1.0
    prod = (i + 1.0) * (n - i)
    a = w1 * (d ** (-p1)) + w2 * (prod ** (-p2)) + w3
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a
