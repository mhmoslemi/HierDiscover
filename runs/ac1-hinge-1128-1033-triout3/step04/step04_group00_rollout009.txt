"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a two‑phase optimiser:

* Phase 1 – Adam‑based projected gradient descent on a smooth
  approximation of the max‑convolution (soft‑max with a slowly
  decreasing temperature).  The iterate lives on the probability
  simplex, enforced by a Euclidean projection.
* Phase 2 – Simulated‑annealing fine‑tuning that repeatedly moves a
  small amount of mass from one index to another.  It accepts
  uphill moves with a temperature schedule, which helps to escape
  the remaining plateaus.
* A short sub‑gradient polishing step is performed at the end.

The whole routine respects the 1000 s budget; it prints progress and
returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using a real FFT (fast).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    # next power of two for efficient FFT
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Compute objective (2 * n * max(conv)) – sum is always 1 in our code.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    conv = _full_convolution_fft(a, a)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Gradient of the smoothed max (soft‑max with temperature eps).
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """Return gradient of eps * log‑sum‑exp(conv/eps) w.r.t `a`."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)               # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (stable)
    w = np.exp((conv - max_c) / eps)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w / w_sum

    # gradient = 2 * (rev(a) * w)   (slice to size n)
    rev_a = a[::-1]
    conv_rev = _full_convolution_fft(rev_a, w)       # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]          # keep central n entries
    return grad


# ----------------------------------------------------------------------
# Simulated annealing neighbour proposal (move a small amount of mass).
# ----------------------------------------------------------------------
def _sa_neighbour(a: np.ndarray, rng: np.random.Generator, step_frac: float):
    """Return a new vector obtained by moving a tiny amount of mass."""
    n = a.shape[0]
    # draw source proportional to current mass (more mass → more chances to move)
    src = rng.choice(n, p=a)
    # ensure destination differs
    dst = rng.integers(0, n - 1)
    if dst >= src:
        dst += 1
    # amount to move (at most a[src])
    max_move = a[src]
    delta = step_frac * max_move * rng.random()
    if delta > max_move:
        delta = max_move
    new_a = a.copy()
    new_a[src] -= delta
    new_a[dst] += delta
    # numerical safety – projections not needed because total mass unchanged
    return new_a


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that (approximately)
    minimises ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence discovered.
    """
    # ---------- setup ----------
    deadline = time.time() + budget_s - 5.0          # safety margin
    rng = np.random.default_rng(seed)

    # ---------- problem size ----------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(base.shape[0])
            a = np.maximum(0.0, base)
            if a.sum() == 0.0:
                a = rng.random(n)
            a = a / a.sum()
        except Exception:   # fall back to a fresh start
            n = 5000
            a = None
    else:
        n = 5000
        a = None

    # ---------- initial shape (mixture of classic windows) ----------
    if a is None:
        x = np.arange(n, dtype=np.float64)

        # Catalan decreasing sequence (classical start)
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)
            catalan[k] = cur

        # Sine (positive on whole interval)
        sine = np.sin(np.pi * (x + 0.5) / n)

        # Semicircle (zero at the ends)
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

        # Gaussian centred in the middle
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

        # Simple triangle
        triangle = np.minimum(x, n - 1 - x) + 1.0

        bases = [catalan, sine, semicircle, gauss, triangle]
        w = rng.random(len(bases))
        w /= w.sum()
        a = sum(weight * base for weight, base in zip(w, bases))
        a = np.clip(a, 0.0, None)
        a /= a.sum()

    # ---------- bookkeeping ----------
    best_a = a.copy()
    best_val = _objective(best_a)
    print(f"[init] n={n}, initial_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam projected gradient descent on soft‑max of the conv.
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta = 0.05                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_step = 0

    # soft‑max temperature schedule
    conv0 = _full_convolution_fft(a, a)
    max_conv = conv0.max()
    eps = max(max_conv * 0.5, 1e-12)    # start relatively high
    eps_min = max(max_conv * 1e-5, 1e-12)
    eps_decay = 0.9998

    # stagnation counter – used for occasional random perturbations
    stagn = 0
    phase1_deadline = time.time() + (budget_s - 5) * 0.55

    while time.time() < phase1_deadline:
        # gradient of the smooth max
        grad = _softmax_gradient(a, eps)

        # Adam update on the *unconstrained* variable
        t_step += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)
        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = _project_simplex(a_new)

        a = a_new

        # evaluate objective
        cur_val = _objective(a)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            stagn = 0
            print(f"[gd] iter={t_step:6d} score={best_val:.12f} eps={eps:.2e} eta={eta:.2e}")
        else:
            stagn += 1

        # temperature decay
        eps = max(eps * eps_decay, eps_min)

        # optional tiny random kicks to escape plateaus
        if stagn > 2500:
            noise = rng.normal(scale=0.005, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            eta = max(1e-5, eta * 0.8)
            print("[gd] perturbation applied")

        # slowly decay the base learning rate (helps fine tuning)
        eta = max(1e-5, eta * 0.99995)

    # ------------------------------------------------------------------
    # Phase 2 – Simulated annealing fine‑tuning.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val

    # annealing schedule
    T = cur_val * 0.05                 # initial temperature (a small fraction of score)
    T_min = 1e-6
    T_decay = 0.9996

    # step size for moving mass
    step_frac = 0.08
    step_decay = 0.9999

    iter_sa = 0
    while time.time() < deadline:
        iter_sa += 1

        cand_a = _sa_neighbour(cur_a, rng, step_frac)
        cand_val = _objective(cand_a)

        if cand_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            diff = cand_val - cur_val
            accept = rng.random() < np.exp(-diff / max(T, 1e-12))

        if accept:
            cur_a = cand_a
            cur_val = cand_val

            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[sa] iter={iter_sa:6d} new_best={best_val:.12f}")

        # temperature and step decay
        T = max(T_min, T * T_decay)
        step_frac = max(0.001, step_frac * step_decay)

        # occasional re‑normalisation (numerical safety)
        if iter_sa % 5000 == 0:
            cur_a = cur_a / cur_a.sum()

    # ------------------------------------------------------------------
    # Final polishing – max‑entry sub‑gradient steps (tiny corrections).
    # ------------------------------------------------------------------
    a = best_a.copy()
    for _ in range(1500):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        # locate the first maximal index
        idx = int(np.flatnonzero(conv >= mx - 1e-15)[0])

        # gradient of the *single* maximal entry
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]

        a = _project_simplex(a - 5e-6 * g)

    # ------------------------------------------------------------------
    # Final evaluation and return.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
