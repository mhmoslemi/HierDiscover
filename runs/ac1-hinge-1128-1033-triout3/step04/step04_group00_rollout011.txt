"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a two‑phase projected optimisation:

1.  A fast “soft‑max” gradient descent with momentum (very similar to the
    reference implementation) to quickly drive the objective down.
2.  A high‑precision refinement that uses the exact sub‑gradient of the
    max‑convolution entry together with an Adam‐style adaptive step size.
    Random perturbations are injected if progress stalls.

Both phases keep the vector on the probability simplex (non‑negative,
unit sum) via a Duchi‑et‑al. projection.  The best sequence seen during the
whole run is returned.  The routine respects the 1000 s budget and prints
progress information.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution via FFT (full linear convolution).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative height sequence that (approximately) minimizes
    the evaluation function.  The algorithm runs for at most `budget_s`
    seconds and prints intermediate scores.
    """
    # ------------------------------------------------------------------
    # Timing and RNG setup
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – try a modestly larger size than the reference.
    # ------------------------------------------------------------------
    n = 6000  # a size that still fits comfortably in memory

    # ------------------------------------------------------------------
    # Initialise a candidate.
    # If a previous good sequence exists we reuse it; otherwise we form a
    # random mixture of a few classic shapes.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] != n:
                # simply truncate / pad to the chosen size
                if base.shape[0] > n:
                    base = base[:n]
                else:
                    pad = np.zeros(n - base.shape[0], dtype=np.float64)
                    base = np.concatenate([base, pad])
            a = np.maximum(0.0, base)
            if a.sum() == 0.0:
                a = rng.random(n)
            a = a / a.sum()
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        x = np.arange(n, dtype=np.float64)

        # Catalan‑type decreasing shape
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)
            catalan[k] = cur

        # Sine shape (always positive)
        sine = np.sin(np.pi * (x + 0.5) / n)

        # Semi‑circle
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))

        # Gaussian centred in the middle
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

        # Simple triangle
        triangle = np.minimum(x, n - 1 - x) + 1.0

        bases = [catalan, sine, semicircle, gauss, triangle]
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a = sum(w * b for w, b in zip(weights, bases))
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Tracking the best solution seen so far.
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv   # sum(a) == 1
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent (momentum + occasional noise)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s   # ~45 % of total time
    eta = 0.8                     # initial step size
    eta_min = 1e-5
    eta_decay = 0.9997

    eps = max(max_conv * 0.5, 1e-12)   # temperature for soft‑max
    eps_min = max(max_conv * 1e-5, 1e-12)
    eps_decay = 0.9995

    momentum = 0.9
    vel = np.zeros_like(a)

    it = 0
    stagn = 0

    while time.time() < phase1_end:
        it += 1

        # convolution and current score
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # keep the best
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phase1 iter {it:6d}] new_best={best_val:.12f}  eta={eta:.2e}  eps={eps:.2e}")
        else:
            stagn += 1

        # soft‑max weights
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient (cross‑correlation)
        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # momentum step
        vel = momentum * vel + (1.0 - momentum) * g

        # gradient step + projection onto the simplex
        a_new = a - eta * vel
        a = _project_simplex(a_new)

        # decay schedules
        eta = max(eta_min, eta * eta_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional perturbation if we are stuck
        if stagn > 2500:
            noise = rng.normal(scale=0.005, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            eta = max(eta_min, eta * 1.5)
            print(f"[phase1 perturb] iteration {it}")

        # safeguard – renormalise every few thousand steps
        if it % 4000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Phase 2 – exact max sub‑gradient descent with Adam‑style adaption.
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eta0 = 0.12                     # base learning rate for Adam

    m = np.zeros_like(a)           # first moment
    v = np.zeros_like(a)           # second moment
    beta1_pow = 1.0
    beta2_pow = 1.0
    t_adam = 0
    stagn = 0

    while time.time() < deadline:
        t_adam += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        # current convolution and score
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # keep the best overall
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phase2 iter {t_adam:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # --- exact max sub‑gradient (average over all active entries) ---
        # tolerance for deciding which entries are “active”
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        # simple acceptance rule: keep the candidate (it is already projected)
        a = a_candidate

        # occasional random perturbation to escape flat regions
        if stagn > 4000:
            noise = rng.normal(scale=0.003, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            # reset Adam moments to avoid stale direction
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            print(f"[phase2 perturb] iter {t_adam}")

        # keep the vector well‑scaled
        if t_adam % 5000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Convert best sequence to the form expected by evaluate_sequence
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    # (Optional) check the final score for logging purposes
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        # In the unlikely event evaluate_sequence is not available at import
        print("[result] final_score could not be computed (evaluate_sequence missing)")
    return final_seq
