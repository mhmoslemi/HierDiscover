def _full_convolution_fft(a, b):

"""
A fast projected‑Adam optimiser for the step‑function problem.

The algorithm works in two stages:

1️⃣  Soft‑max Adam – minimise a smooth approximation of
    `max(conv(a,a))` by repeatedly
    * computing the self‑convolution,
    * turning the convolution entries into a probability distribution
      via a temperature‑controlled soft‑max,
    * obtaining the gradient as a cross‑correlation,
    * taking an Adam step and projecting back onto the probability simplex.

2️⃣  Hard‑max polishing – when the temperature becomes very low the
    weights concentrate on the almost‑maximal entries.
    We then continue the Adam updates with the binary mask of the
    truly maximal indices.  A tiny constant learning‑rate decay together
    with occasional random perturbations lets the procedure escape
    plateaus and shave a few more digits off the score.

The routine respects the 1000 s time budget, keeps track of the best
sequence found, and finally returns it (clamped to the limits required
by `evaluate_sequence`).  It can start from a warm‑start stored in the
global `height_sequence_1`, otherwise builds a mixed “nice’’ shape.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative height sequence that (approximately) minimises
    `2 * n * max(conv(a, a)) / (sum(a) ** 2)`.
    The function respects the supplied time budget (in seconds).
    """
    # ------------------------------------------------------------------
    # General set‑up.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0          # small safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a starting size and initialise a.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = base.shape[0]
            a = np.maximum(0.0, base)
            if a.sum() == 0.0:
                a = rng.random(n)
            a = a / a.sum()
        except Exception:
            n = 5000
            a = None
    else:
        n = 5000
        a = None

    if a is None:
        # Build a mixture of several “nice’’ shapes.
        x = np.arange(n, dtype=np.float64)

        # (approximately) Catalan decreasing sequence
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)
            catalan[k] = cur

        # Sine, semi‑circle, Gaussian and triangular shapes.
        sine = np.sin(np.pi * (x + 0.5) / n)
        t = 2.0 * x / (n - 1) - 1.0
        semicircle = np.sqrt(np.clip(1.0 - t * t, 0.0, None))
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
        triangle = np.minimum(x, n - 1 - x) + 1.0

        bases = [catalan, sine, semicircle, gauss, triangle]
        weights = rng.random(len(bases))
        weights /= weights.sum()
        a = sum(w * b for w, b in zip(weights, bases))
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Keep track of the best solution seen so far.
    # ------------------------------------------------------------------
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv           # sum(a)==1 ⇒ denominator = 1
    best_seq = a.copy()
    print(f"[init] n={n}, initial_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for Adam + soft‑max.
    # ------------------------------------------------------------------
    # Adam
    lr = 0.12               # initial learning‑rate
    lr_min = 1e-6
    lr_decay = 0.99996      # very slow decay – keep enough momentum
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature
    eps = max(max_conv * 0.5, 1e-12)
    eps_min = max(max_conv * 1e-6, 1e-12)
    eps_decay = 0.99992

    stagn = 0               # iterations without improvement
    it = 0

    # ------------------------------------------------------------------
    # Phase 1 – smooth optimisation with Adam.
    # ------------------------------------------------------------------
    while time.time() < deadline * 0.68:   # reserve ~32 % of the budget for polishing
        it += 1

        # ----- convolution and current score ---------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        # ----- keep the best --------------------------------------------------
        if score < best_val - 1e-13:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it:6d}] new_best={best_val:.12f}  lr={lr:.2e}  eps={eps:.2e}")
        else:
            stagn += 1

        # ----- soft‑max weighting --------------------------------------------
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum > 0.0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # ----- gradient of the weighted sum (cross‑correlation) -------------
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]            # slice to size n

        # ----- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates (avoid overflow for very large it)
        beta1_t = beta1 ** it
        beta2_t = beta2 ** it
        m_hat = m / (1.0 - beta1_t)
        v_hat = v / (1.0 - beta2_t)

        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a + step
        a_new = _project_simplex(a_new)

        a = a_new

        # ----- schedules -------------------------------------------------------
        lr = max(lr_min, lr * lr_decay)
        eps = max(eps_min, eps * eps_decay)

        # ----- occasional random perturbation ---------------------------------
        if stagn > 3000:
            noise = rng.normal(scale=0.008, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            lr = max(lr_min, lr * 1.3)
            print(f"[perturb] iteration {it}")

        # ----- periodic re‑normalisation (numerical safety) --------------------
        if it % 5000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Phase 2 – hard‑max polishing (temperature essentially zero).
    # ------------------------------------------------------------------
    # The temperature is already tiny; we just keep using the same Adam
    # machinery but with a binary mask of the true maximal entries.
    while time.time() < deadline:
        it += 1

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-13:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[polish {it:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # binary mask of (approximately) maximal entries
        tol = max_conv * 1e-9 + 1e-14
        mask = (conv >= max_conv - tol).astype(np.float64)
        w_sum = mask.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = mask / w_sum

        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # A tiny constant step size works well in this regime.
        step = - 2e-5 * grad
        a = a + step
        a = _project_simplex(a)

        # occasional random jitter to avoid being stuck
        if stagn > 5000:
            a = a + rng.normal(scale=0.004, size=n)
            a = _project_simplex(a)
            stagn = 0
            print(f"[jitter] iteration {it}")

    # ------------------------------------------------------------------
    # Final polishing using a few thousand pure max‑entry sub‑gradient steps.
    # ------------------------------------------------------------------
    a = best_seq.copy()
    for _ in range(5000):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        # mask of entries that are (almost) maximal
        active = (conv >= mx - 1e-15).astype(np.float64)
        w_sum = active.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = active / w_sum
        rev_a = a[::-1]
        grad_full = _full_convolution_fft(rev_a, w)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]
        # decaying sub‑gradient step
        step = 4e-6 * grad
        a = _project_simplex(a - step)

    # ------------------------------------------------------------------
    # Convert to the format required by `evaluate_sequence` and report.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    return final_seq
