def beta_shape(n, alpha, beta):
    indices = np.arange(n) + 0.5
    x = indices / n
    # avoid zero/one endpoints
    eps=1e-12
    x = np.clip(x, eps, 1-eps)
    w = np.power(x, alpha-1) * np.power(1 - x, beta-1)
    w_sum = w.sum()
    return w / w_sum

"""
Propose a step‑function sequence that minimises the objective in
``evaluate_sequence``.

The algorithm has three stages:

1. **Fast shape search** – on a modest length (≈600) a random search
   over Beta‑shaped distributions (parameterised by two positive
   exponents) is performed.  The best shape (α,β) is kept.

2. **Upscaling** – the selected shape is transferred to a larger
   length (≈2000) and a small random jitter is added.  This provides a
   high‑quality initial point that is already better than the uniform
   start.

   optimised on the probability simplex using a log‑space (softmax)
   parametrisation.  The max‑convolution is smoothed with a temperature
   that linearly decays to zero; the resulting gradient is fed to Adam.
   Random “jitter” moves and occasional restarts (when progress stalls)
   help escape plateaux.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence discovered before the deadline.  All numbers
are non‑negative and sum to one, satisfying the constraints of
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full (linear) convolution using real FFT – fast for many calls
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y (length len(x)+len(y)-1) via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny numerical negatives
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution (soft‑max weights)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = 2 * Σ_k w_k * a_{k-i} for all i.
    Implemented via a single convolution: reverse w and convolve with a.
    """
    n = a.shape[0]
    # reverse the weight vector
    u = w[::-1]
    conv = _full_convolution_fft(u, a)          # length = 3n‑2
    # indices that correspond to the original length n:
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Beta‑shape generator (continuous Beta density discretised on a grid)
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Return a probability vector of length n with Beta(α,β) shape."""
    eps = 1e-12
    # mid‑point of each bin
    idx = np.arange(n, dtype=np.float64) + 0.5
    x = idx / n
    # avoid exact 0/1 which would cause infinities when α<1 or β<1
    x = np.clip(x, eps, 1.0 - eps)
    w = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    s = w.sum()
    # fallback to uniform if something went wrong
    if s == 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a non‑negative step‑function sequence under the given budget.
    Returns a Python list of floats (the best sequence found).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – the framework kills the process a few
    # seconds after the deadline.
    deadline = start_time + budget_s - 4.0

    # --------------------------------------------------------------
    # Phase 1 – cheap random search over Beta-shaped families (small n)
    # --------------------------------------------------------------
    n_small = 600
    best_val_small = float("inf")
    best_alpha = None
    best_beta = None

    # initialise with the uniform distribution (baseline)
    uniform_a = np.full(n_small, 1.0 / n_small, dtype=np.float64)
    conv = np.convolve(uniform_a, uniform_a)
    best_val_small = 2.0 * n_small * conv.max()
    best_alpha = 1.0
    best_beta = 1.0

    # search while we still have at least ~30 s for later phases
    while time.time() < deadline - 30.0:
        # draw α,β from a log‑uniform range [0.2, 5]
        log_a = np.random.uniform(np.log(0.2), np.log(5.0))
        log_b = np.random.uniform(np.log(0.2), np.log(5.0))
        alpha = np.exp(log_a)
        beta = np.exp(log_b)
        a = _beta_shape(n_small, alpha, beta)
        conv = np.convolve(a, a)
        cur_val = 2.0 * n_small * conv.max()
        if cur_val < best_val_small:
            best_val_small = cur_val
            best_alpha = alpha
            best_beta = beta
            print(f"[shape] new best value {best_val_small:.6f} "
                  f"α={alpha:.4f}, β={beta:.4f}")

    # --------------------------------------------------------------
    # Phase 2 – up‑scale the best shape to a larger dimension
    # --------------------------------------------------------------
    n = 2000                      # final length (still fast enough)
    a = _beta_shape(n, best_alpha, best_beta)

    # tiny random jitter to break perfect symmetry
    jitter = 0.001 * np.random.rand(n) * a
    a = np.clip(a + jitter, 0.0, None)
    a = a / a.sum()               # keep the simplex constraint

    # --------------------------------------------------------------
    # Phase 3 – Adam optimisation with annealed soft‑max smoothing
    # --------------------------------------------------------------
    eps = 1e-12
    # log‑space representation (softmax guarantees non‑negativity & sum=1)
    x = np.log(np.maximum(a, eps))

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.90, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps_adam = 1e-8
    base_lr = 0.20                # a slightly smaller start learning rate

    # bookkeeping of the overall best solution
    a_curr = a.copy()
    conv = _full_convolution_fft(a_curr, a_curr)
    best_val = float("inf")
    best_a = a_curr.copy()
    iteration = 0
    it_no_improve = 0
    RESTART_THRESHOLD = 15000     # restart if no improvement for many iters

    print(f"[init] n={n}, start objective={2.0 * n * conv.max():.6f}")

    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration += 1

        # --- current objective ------------------------------------------------
        conv = _full_convolution_fft(a_curr, a_curr)
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_curr.copy()
            it_no_improve = 0
            print(f"[{iteration:7d}] new best = {best_val:.6f}")
        else:
            it_no_improve += 1

        # --- progress & temperature (annealing of the soft‑max) ---------------
        progress = (now - start_time) / (deadline - start_time)

        # temperature for the soft‑max (relative to the current max)
        tau_start = M * 0.5
        tau_end   = M * 1e-6
        tau = tau_start * (1.0 - progress) + tau_end * progress

        # --- soft‑max weights over all convolution entries --------------------
        if tau > 0.0:
            exp_args = (conv - M) / tau
            # numerical stabilisation
            max_arg = exp_args.max()
            exp_vals = np.exp(exp_args - max_arg)
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # --- gradient w.r.t. a (smoothed sub‑gradient) -----------------------
        grad_a = _gradient_from_weights(a_curr, w)

        # --- gradient w.r.t. logits x (softmax chain rule) --------------------
        dot = np.dot(grad_a, a_curr)                 # <grad_a , a>
        grad_x = a_curr * (grad_a - dot)

        # --- optional gradient clipping (helps stability) --------------------
        grad_norm = np.linalg.norm(grad_x)
        max_norm = 5.0
        if grad_norm > max_norm:
            grad_x = grad_x * (max_norm / grad_norm)

        # --- Adam update for the log‑parameters x ----------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule (linear decay to a tiny value)
        lr = base_lr * (1.0 - progress) + 1e-5 * progress
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- obtain a from the updated logits (softmax) ----------------------
        max_x = x.max()
        exp_x = np.exp(x - max_x)
        a_curr = exp_x / exp_x.sum()

        # --- occasional random “jitter” moves (helps escape plateaux) -------
        if np.random.rand() < max(0.05 * (1.0 - progress), 0.005):
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = np.random.rand() * a_curr[i] * 0.4
            a_curr[i] -= delta
            a_curr[j] += delta
            a_curr = np.clip(a_curr, 0.0, None)
            a_curr /= a_curr.sum()
            # reset the optimiser state after a sudden change
            x = np.log(np.maximum(a_curr, eps))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            it_no_improve = 0

        # --- restart if we have been stuck for a while ----------------------
        if it_no_improve >= RESTART_THRESHOLD:
            # re‑initialise from the best shape (with fresh random perturbation)
            a_restart = _beta_shape(n, best_alpha, best_beta)
            a_restart += 0.001 * np.random.rand(n) * a_restart
            a_restart = np.clip(a_restart, 0.0, None)
            a_restart /= a_restart.sum()
            x = np.log(np.maximum(a_restart, eps))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            a_curr = a_restart.copy()
            it_no_improve = 0
            print(f"[{iteration:7d}] restart triggered (no improvement)")

    # ------------------------------------------------------------------
    # Final reporting (the external framework will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = "
              f"{final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the best sequence discovered (as a plain Python list)
    return best_a.tolist()
