    max_val = conv.max()
    # active mask
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]

    # For each active idx compute gradient contributions similar to previous code:

    # else:

"""
propose_candidate
-----------------

Search for a non‑negative step function (a probability distribution on
{0,…,n‑1}) that minimizes

    2·n·max(conv_self(a)) / (sum(a))**2

The evaluation routine `evaluate_sequence` removes scaling, therefore we
fix the L¹‑norm to 1 (simplex projection) and minimise the maximal entry
of the full self‑convolution.

Algorithm
~~~~~~~~~
1.  Build a diverse pool of initial vectors (uniform, geometric decays,
    power‑law, random monotone, Dirichlet draws and a spectral‑flattening
    heuristic).
2.  For each start run a two‑phase optimiser
    * **Phase A – Adam on a smooth max**  
      The true max is approximated by a log‑sum‑exp with a temperature
      α.  The smoothed objective is  
        fα(p)= (1/α)·log Σ exp(α·conv(p)) .  
      Its gradient can be obtained by a single FFT convolution with the
      soft‑max weights.  Adam with a step size  base_lr/√α  is applied,
      and α is slowly increased up to a large value (≈400).  Every
      1024 iterations the exact objective is evaluated and the best
      vector is recorded.
    * **Phase B – Exact sub‑gradient descent**  
      Using the current vector we compute the exact maximal convolution
      entry, build a uniform weight over all active (≈maximal) indices
      and compute the sub‑gradient again via FFT.  A decreasing step‑size
      (geometric decay) is used.  Periodically the exact objective is
      checked.
3.  The best vector found over all starts is returned (clipped to the
    limits enforced by `evaluate_sequence`).

All linear constraints (non‑negativity and unit L¹‑norm) are enforced
by Euclidean projection onto the probability simplex after each update.
The implementation is pure NumPy, runs on a single CPU and respects the
1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(p: np.ndarray, m_fft: int) -> np.ndarray:
    """Full linear self‑convolution of ``p`` using an FFT of size ``m_fft``."""
    p_fft = np.fft.rfft(p, n=m_fft)
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * p.size - 1]
    return conv


def _objective_exact(p: np.ndarray, n: int, m_fft: int) -> float:
    """Exact objective value 2·n·max(conv_self(p))."""
    conv = _conv_self(p, m_fft)
    return float(2.0 * n * np.max(conv))


def _smooth_max_and_grad(p: np.ndarray, alpha: float,
                         n: int, m_fft: int):
    """
    Smooth approximation of the max (log‑sum‑exp) together with its gradient.

    Returns
    -------
    smooth_max : float
        (1/α)·log Σ exp(α·conv[p])
    grad : np.ndarray
        Gradient of the smooth max w.r.t. ``p`` (size n).
    """
    # FFT of the current vector
    p_fft = np.fft.rfft(p, n=m_fft)

    # full convolution of p with itself
    conv = np.fft.irfft(p_fft * p_fft, n=m_fft)[: 2 * n - 1]

    # log‑sum‑exp (smooth max)
    z = alpha * conv
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)          # stability trick
    sum_exp = np.sum(exp_z)
    smooth_max = (max_z + np.log(sum_exp)) / alpha

    # soft‑max weights (they sum to 1)
    w = exp_z / sum_exp

    # gradient = 2 * correlation(p, reversed(w))
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * corr[idx]                     # ∂/∂p (w·conv)
    return smooth_max, grad


def _subgradient_step(p: np.ndarray, step: float,
                      n: int, m_fft: int):
    """
    One sub‑gradient step on the *exact* max of the convolution.
    The sub‑gradient is obtained by averaging the gradients of all
    active (maximal) convolution entries.
    """
    conv = _conv_self(p, m_fft)
    max_val = conv.max()
    active = np.where(conv >= max_val - 1e-12)[0]

    # uniform weight over active indices
    w = np.zeros_like(conv)
    w[active] = 1.0 / len(active)

    # gradient via FFT (same formula as in the smooth case)
    p_fft = np.fft.rfft(p, n=m_fft)
    w_rev_fft = np.fft.rfft(w[::-1], n=m_fft)
    corr = np.fft.irfft(p_fft * w_rev_fft, n=m_fft)[: 3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * corr[idx]

    return _simplex_projection(p - step * grad)


def _spectral_flattening_initial(p: np.ndarray, n: int, m_fft: int,
                                iters: int = 200):
    """
    Heuristic that repeatedly forces the DFT magnitude of ``p`` to be
    constant while keeping the vector non‑negative and on the simplex.
    """
    q = p.copy()
    for _ in range(iters):
        q_fft = np.fft.rfft(q, n=m_fft)
        mags = np.abs(q_fft)
        avg_mag = np.mean(mags)
        # replace magnitude by the average, keep original phase
        new_fft = avg_mag * np.exp(1j * np.angle(q_fft))
        q = np.fft.irfft(new_fft, n=m_fft)[:n]
        q = np.maximum(q, 0.0)
        q = _simplex_projection(q)
    return q

# ----------------------------------------------------------------------

        eps_adam = 1e-8
        m1 = np.zeros_like(p)
        m2 = np.zeros_like(p)

        alpha = 0.5
        alpha_mul = 1.001
        alpha_max = 400.0

        for it in range(200_000):

            smooth_max, grad = _smooth_max_and_grad(p, alpha, n, m_fft)

            # Adam update (gradient is for the smooth max; scale by 2·n later via lr)
