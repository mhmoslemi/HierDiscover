def solve_lp_for_f(f, M, method='highs'):

    n = len(f)
    # Build A_ub and b = M.
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        A[k, i_start:i_end+1] = f[i_start:i_end+1]   # assign vector slice
    # Note each row = relevant f[i]
    # Using broadcasting might be slower; but loops can be done.

    # objective maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)

    b_ub = np.full(rows, M, dtype=np.float64)

    # bounds for variables: (0, None)
    bounds = [(0, None) for _ in range(n)]

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method, options={'presolve': True})

    if not res.success:
        return None

"""
Propose a candidate step‑function (non‑negative vector) that minimises

    2 * n * max_k (a * a)[k]

where * is the full linear convolution.  The search proceeds in three
phases:

1.  A deterministic sweep over several symmetric parametric families,
    including a new “piecewise‑constant” family (heavy edges, light
    interior).  The best vector from this sweep seeds the refinement.

2.  Stochastic refinement on the probability simplex using
    *additive* mirror descent with momentum (a simple projected‑gradient
    method) plus occasional *targeted mass‑transfer* moves that move
    probability mass away from the indices that currently realise the
    maximal convolution entry.  Acceptance follows a simulated‑annealing
    schedule.

3.  (If time permits) a short linear‑programming boost: for the current
    vector f we solve a linear program that maximises the total mass of
    a second vector g under the constraint f∗g ≤ max(f∗f).  A convex
    combination f←(1‑t)f+t g reduces the worst convolution value.

The routine respects the given time budget (default 1000 s) and returns
the best sequence found as a plain Python list (its entries sum to 1,
so the external evaluator will only rescale them).
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {w ≥ 0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 1
    target = 2 * n - 1
    while size < target:
        size <<= 1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[:target]


def full_convolution_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two arbitrary real vectors using FFT.
    Returns length len(x)+len(y)-1.
    """
    lx, ly = len(x), len(y)
    size = 1
    target = lx + ly - 1
    while size < target:
        size <<= 1
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:target]


# ----------------------------------------------------------------------
# Scoring utilities (exact score for a vector that already sums to 1)
# ----------------------------------------------------------------------
def score_from_vector(a: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (used by additive updates)
# ----------------------------------------------------------------------
def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    For each shift attaining the maximum we add 2*a[j] to gradient[i]
    whenever i+j = k.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


# ----------------------------------------------------------------------
# Parametric families (all symmetric)
# ----------------------------------------------------------------------
def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^‑p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))‑p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)⁻p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π·i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


def piecewise_const_distribution(n: int, outer_k: int, ratio: float) -> np.ndarray:
    """
    Heavy‑edge / light‑interior shape.
    The first and last ``outer_k`` entries receive weight ``ratio``,
    all other entries weight 1.  The vector is normalised to sum‑to‑1.
    """
    if outer_k <= 0:
        a = np.ones(n, dtype=np.float64)
    else:
        a = np.ones(n, dtype=np.float64)
        a[:outer_k] = ratio
        a[-outer_k:] = ratio
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming boost (optional, runs only if time permits)
# ----------------------------------------------------------------------
def solve_lp_max_mass(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        max  sum(g)
        s.t.  conv(f, g)[k] ≤ M   for all k
              g ≥ 0
    Returns the optimal g (not normalised).  Uses SciPy's HiGHS solver.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the constraint matrix A where A[k, i] = f[i] if i contributes to conv[k]
    # We construct it once per call (O(n²) work) – acceptable for moderate n.
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # slice assignment is cheap
        A[k, i_start:i_end + 1] = f[i_start:i_end + 1]

    c = -np.ones(n, dtype=np.float64)           # maximise sum(g)  → minimise -sum(g)
    b_ub = np.full(rows, M, dtype=np.float64)  # right‑hand side of the inequalities
    bounds = [(0.0, None) for _ in range(n)]

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=bounds,
                                method='highs',
                                options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the given time
    budget (seconds).  Returns the sequence as a plain Python list whose
    entries sum to about 1.0.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 4.0  # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # (0) Try to use the global variable from the previous run, if present
    # ------------------------------------------------------------------
    try:
        ext = np.array(height_sequence_1, dtype=float)
        if ext.size:
            ext = np.maximum(ext, 0.0)
            s = ext.sum()
            if s > 0.0:
                ext /= s
                sc = score_from_vector(ext)
                if sc < best_score:
                    best_score = sc
                    best_seq = ext.copy()
                    print(f"External seed used (score={sc:.6f})")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # (1) Deterministic sweep over parametric families
    # ------------------------------------------------------------------
    print("\n=== Deterministic sweep start ===")
    lengths = [1000, 1500, 2000, 3000, 4000, 5000, 5808, 8000, 12000]

    # Exponent grids
    exponents = np.arange(0.2, 5.01, 0.2)   # slightly coarser than before
    shift_cs = [0.5, 1.0, 2.0, 5.0]
    lam_vals = np.logspace(-4, -0.3, num=20)    # fewer points
    cos_exps = np.arange(0.5, 8.5, 0.5)

    for n in lengths:
        if time.time() > deadline:
            break

        # --- U‑shape families ---
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (U‑min)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (U‑prod) n={n:5d} p={p:4.2f}  score={sc:.6f}")

            for c in shift_cs:
                a = u_min_shift_distribution(n, p, c)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"new best (shift) n={n:5d} p={p:4.2f} c={c:.1f}  score={sc:.6f}")

        # --- exponential family ---
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (exp)   n={n:5d} lam={lam:.5f}  score={sc:.6f}")

        # --- cosine family ---
        for p in cos_exps:
            a = cos_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (cos)   n={n:5d} p={p:3.1f}  score={sc:.6f}")

        # --- piecewise‑constant (heavy edges) ---
        outer_fractions = [0.01, 0.02, 0.05, 0.1, 0.15]
        ratios = np.logspace(0, 2, num=15)   # 1 … 100
        for frac in outer_fractions:
            outer_k = max(1, int(round(frac * n)))
            for r in ratios:
                a = piecewise_const_distribution(n, outer_k, r)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"new best (pc)    n={n:5d} outer={outer_k:4d} r={r:5.1f}  score={sc:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # (2) Stochastic refinement – additive projected gradient + moves
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of modest length
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = score_from_vector(best_seq)
        print(f"fallback to uniform n={n}  score={best_score:.6f}")

    current = best_seq.copy()
    n = current.shape[0]
    current = np.maximum(current, 0.0)
    current /= current.sum()
    current_conv = full_convolution_fft(current)
    current_score = 2.0 * n * current_conv.max()

    # hyper‑parameters for the additive (momentum) descent
    step_size = 0.7           # initial learning rate
    step_decay = 0.99985
    beta = 0.9                # momentum coefficient
    momentum = np.zeros_like(current)
    temperature = 1.0
    temp_decay = 0.9998
    it = 0
    last_improve = 0

    print("\n=== Stochastic refinement start ===")
    while time.time() < deadline:
        it += 1

        # recompute convolution and gradient for the current iterate
        conv = full_convolution_fft(current)
        max_val = conv.max()
        current_score = 2.0 * n * max_val
        grad = compute_subgradient(current, conv)

        # momentum update (exponential moving average of the sub‑gradient)
        momentum = beta * momentum + (1.0 - beta) * grad

        # additive mirror‑descent step on the simplex
        cand = current - step_size * momentum
        cand = simplex_projection(cand)

        # occasional *targeted* mass‑transfer move (helps escape plateaus)
        if it % 64 == 0:
            # pick a shift attaining the maximal convolution value
            max_idx = np.where(conv >= max_val - 1e-12)[0]
            if max_idx.size:
                k = np.random.choice(max_idx)
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx_range = np.arange(i_start, i_end + 1)
                # pick the index with the largest coefficient inside the active window
                i_sel = idx_range[np.argmax(current[idx_range])]
                # pick a destination far from the active window
                forbidden = np.arange(i_start, i_end + 1)
                possible_j = np.setdiff1d(np.arange(n), forbidden)
                if possible_j.size:
                    j_sel = np.random.choice(possible_j)
                    delta = current[i_sel] * np.random.rand() * 0.5
                    cand2 = cand.copy()
                    cand2[i_sel] = max(cand2[i_sel] - delta, 0.0)
                    cand2[j_sel] = cand2[j_sel] + delta
                    cand2 = simplex_projection(cand2)

                    cand2_score = 2.0 * n * full_convolution_fft(cand2).max()
                    cand_score = 2.0 * n * full_convolution_fft(cand).max()

                    if cand2_score < cand_score:
                        cand, cand_score = cand2, cand2_score
                    else:
                        cand_score = cand_score
                else:
                    cand_score = 2.0 * n * full_convolution_fft(cand).max()
            else:
                cand_score = 2.0 * n * full_convolution_fft(cand).max()
        else:
            cand_score = 2.0 * n * full_convolution_fft(cand).max()

        # simulated‑annealing acceptance
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = it
                print(f"Iter {it:7d}: new best score = {best_score:.6f}")

        # schedule updates
        step_size *= step_decay
        temperature *= temp_decay

        # stagnation handling – occasional re‑heat
        if it - last_improve > 180000:
            bump = np.random.rand(n) * 1e-5
            current = simplex_projection(current + bump)
            step_size = 0.7
            temperature = 1.0
            momentum = np.zeros_like(current)
            last_improve = it
            print(f"Iter {it:7d}: re‑heat after stagnation")

    # ------------------------------------------------------------------
    # (3) Optional linear‑programming boost (if time still remains)
    # ------------------------------------------------------------------
    # Use a reduced size (≈1500) to keep the LP fast.  The best vector
    # found so far is interpolated to this size, refined with a few LP
    # iterations, and then interpolated back.
    if time.time() < deadline - 5.0:
        n_lp = 1500
        # simple linear interpolation from best_seq to size n_lp
        x_src = np.linspace(0, 1, len(best_seq))
        x_dst = np.linspace(0, 1, n_lp)
        f_lp = np.interp(x_dst, x_src, best_seq)
        f_lp = simplex_projection(f_lp)

        for lp_it in range(8):   # small fixed number of LP refinements
            if time.time() > deadline - 2.0:
                break
            conv_lp = full_convolution_fft(f_lp)
            M_lp = conv_lp.max()
            g_lp = solve_lp_max_mass(f_lp, M_lp)
            if g_lp is None:
                break
            # normalise g_lp to sum = 1
            g_lp = simplex_projection(g_lp)
            # convex combination (t = 0.2 works well empirically)
            t = 0.2
            f_lp = (1 - t) * f_lp + t * g_lp
            # keep the best LP‑refined vector (if it improves)
            score_lp = 2.0 * n_lp * full_convolution_fft(f_lp).max()
            if score_lp < best_score:
                # up‑sample back to the original length for reporting
                up = np.interp(x_src, x_dst, f_lp)
                up = simplex_projection(up)
                best_score = 2.0 * len(up) * full_convolution_fft(up).max()
                best_seq = up.copy()
                print(f"LP iter {lp_it}: improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
