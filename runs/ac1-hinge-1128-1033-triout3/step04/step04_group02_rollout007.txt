"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
* Work with a soft‑max parametrisation: a = softmax(w) → a_i ≥ 0, ∑a_i = 1.
* Optimise a smooth surrogate of the max‑convolution:
        L_ε(a) = 2·n·( max(conv) + ε·log∑_k exp((conv_k‑max)/ε) )
  where conv = a * a (full convolution).  Its gradient w.r.t. w is
  available in closed form and can be evaluated with two FFT‑based
  convolutions.
* Perform a continuation in ε (log‑spaced from 1e‑2 down to 1e‑8) using
  L‑BFGS‑B.  After each ε‑stage evaluate the exact objective and keep the
  best iterate.
* Run several random restarts (the budget is 1000 s, so we can afford a
  few dozen restarts).  Each restart starts from the base sequence
  perturbed with Gaussian noise.
* Finally apply a cheap hill‑climbing refinement that repeatedly moves a
  small random fraction of mass from one index to another while keeping
  the ℓ₁‑norm constant.  Accept only improvements.
* The best sequence seen (exactly evaluated) is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT helpers (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[:out_len]
    # Numerical noise may cause small negatives – clip.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Soft‑max utilities
# ----------------------------------------------------------------------
def _softmax(w: np.ndarray) -> np.ndarray:
    """Return soft‑max of w (guaranteed non‑negative and summed to 1)."""
    w_shift = w - np.max(w)          # numeric stability
    e = np.exp(w_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed surrogate and its gradient (w.r.t. the soft‑max parameters)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute (L_ε, ∂L_ε/∂w) for a given soft‑max parameter vector w.
    L_ε = 2·n·(max(conv) + ε·log∑_k exp((conv_k‑max)/ε)).
    """
    a = _softmax(w)                     # shape (n,)

    # full auto‑convolution a*a
    conv = _full_convolution(a)         # length 2·n‑1
    max_c = conv.max()

    # ---- smoothed max (log‑sum‑exp) ----
    shifted = (conv - max_c) / eps       # ≤0
    log_sum = logsumexp(shifted)        # log ∑ exp(...)
    L = 2.0 * n * (max_c + eps * log_sum)

    # ---- gradient w.r.t. a (closed form) ----
    # p_k = exp((conv_k‑max)/eps) / Σ_j exp((conv_j‑max)/eps)
    p = np.exp(shifted - log_sum)       # already normalised, shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    # compute convolution a * rev(p)
    p_rev = p[::-1]                     # reversed p
    conv_ap = _conv_fft(a, p_rev)       # length 3·n‑2
    seg = conv_ap[n - 1 : 2 * n - 1]    # middle n entries, shape (n,)
    grad_a = 4.0 * n * seg[::-1]        # reverse back to align with a

    # ---- chain rule (soft‑max) ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)          # ∂L/∂w
    return L, grad_w


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, cur_val: float,
                deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Perform cheap random mass transfers while we still have time."""
    n = a.shape[0]
    best_a = a.copy()
    best = cur_val
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a fraction of mass from i to j
        delta = best_a[i] * 0.03 * rng.random()   # up to 3 % of mass at i
        if delta <= 0.0:
            continue
        candidate = best_a.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # evaluate exactly
        try:
            val = evaluate_sequence(candidate.tolist())
        except Exception:
            continue
        if val < best:
            best = val
            best_a = candidate
            print(f"[hill] new best = {best:.6f}")
    return best_a, best
