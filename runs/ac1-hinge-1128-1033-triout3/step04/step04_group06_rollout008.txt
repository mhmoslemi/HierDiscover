def propose_candidate(...):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # 0.5 sec margin

    n = 3000
    a = np.random.dirichlet(alpha=np.full(n, 0.5))
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print('initial val', best_val)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    iter = 0
    while time.time() < deadline:
        # compute conv
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print('[%d] new best = %.6f' % (iter, best_val))

        # schedule for temperature and learning rate
        progress = (time.time() - start) / (deadline - start)
        # Temperature factor
        tau_factor = (0.3 * (1-prog) + 1e-6 * prog) * M
        # Compute w
        if tau_factor <= 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            # stable log-sum-exp
            exp_args = (conv - M) / tau_factor
            np.clip(exp_args, -80, 80, out=exp_args)
            exp_vals = np.exp(exp_args)
            sum_exp = exp_vals.sum()
            w = exp_vals / sum_exp

        # gradient
        grad = _gradient_from_weights(a, w)  # shape n
        # ADAM update on log a (since we want multiplicative update)
        # we can perform update: log_a = log_a - lr * grad / (norm...)
        # but to stay on simplex, we can use exponentiated gradient:
        # use step size lr * grad, a = a * exp(-lr * grad) and renorm
        # Let's incorporate ADAM on gradient: m = b1*m + (1-b1)*grad ; v = b2*v + (1-b2)*grad**2
        # compute bias-corrected
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**(iter+1))
        v_hat = v / (1 - beta2**(iter+1))
        # step
        eta0 = 0.5
        eta = eta0 * (1 - progress) + 0.001 * progress
        # Compute update factor
        step = - eta * m_hat / (np.sqrt(v_hat) + eps)
        np.clip(step, -50, 50, out=step)
        a = a * np.exp(step)

        # optional jitter
        if np.random.rand() < 0.02:
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            # choose random fraction
            delta_frac = np.random.rand() * 0.5
            delta = min(delta_frac * a[i], a[i])
            a[i] -= delta
            a[j] += delta

        # projection onto simplex: ensure positivity and renorm
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        iter += 1
        # occasionally recompute conv for safety
        if iter % 2000 == 0:
            conv = _full_convolution_fft(a,a)  # no op
    # final report
    try:
        final_val = evaluate_sequence(best_a.tolist())
        print('[final] best evaluation =', final_val)
    except Exception as exc:
        print('final evaluation error', exc)
    return best_a.tolist()

"""
Propose a non‑negative step‑function (probability vector) that minimizes
the evaluation function

    evaluate_sequence(a) = 2 * n * max(conv(a, a))

where n = len(a) and conv is linear convolution.
The algorithm uses exponentiated‑gradient (mirror‑descent) on a
soft‑max approximation of the infinity norm, an annealed temperature
schedule, momentum and occasional random “jitter”.  In the final
stage it applies a greedy “max‑conv reduction” move that explicitly
shifts a small amount of probability mass away from the indices that
contribute most to the current maximal convolution value.  The whole
procedure runs until the time budget expires and returns the best
sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution using a real FFT (fast for many calls)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x and y using a real FFT."""
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    # size is the next power of two (FFT length)
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max (log‑sum‑exp) approximation of the max‑conv.
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of 2 * Σ_k w_k * (a * a)[k] with respect to a.
    The result is 2 * (conv_rev(w) * a) evaluated at the original indices.
    """
    n = a.shape[0]
    # reverse the weight vector (full length = 2*n-1)
    rev_w = w[::-1]
    # full convolution rev_w * a  → length 3*n-2
    conv = _full_convolution_fft(rev_w, a)
    # the gradient entries correspond to indices start, start-1, ..., start-(n-1)
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Greedy refinement step: move a small fraction of probability away from the
# indices that contribute most to the current maximal convolution value.
# ----------------------------------------------------------------------
def _greedy_reduce_max(a: np.ndarray, conv: np.ndarray, n: int) -> np.ndarray:
    """Shift a tiny mass from a high‑contributing index to a low one."""
    # location of the current maximum of the auto‑convolution
    k_max = int(np.argmax(conv))

    # indices i for which a_i * a_{k_max-i} is defined
    i_low = max(0, k_max - (n - 1))
    i_high = min(n - 1, k_max)
    i_range = np.arange(i_low, i_high + 1, dtype=int)

    # contributions of each i to the peak
    contrib = a[i_range] * a[k_max - i_range]

    # source index = the one with largest contribution
    src = i_range[int(np.argmax(contrib))]

    # destination index = a random index with relatively small contribution
    # (or uniform random if all contributions are similar)
    perc = np.percentile(contrib, 20)  # 20‑th percentile
    low_candidates = i_range[contrib <= perc]
    if low_candidates.size > 0:
        dst = int(np.random.choice(low_candidates))
    else:
        dst = int(np.random.randint(n))

    if src == dst:
        return a

    # move at most 20 % of the source mass (capped to keep positivity)
    delta = min(a[src] * 0.2, a[src] * np.random.rand() * 0.5)
    a[src] -= delta
    a[dst] += delta
    return a
