def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    # global variable access
    global height_sequence_1

    # Determine target length
    target_n = 4000  # choose moderate large
    if 'height_sequence_1' in globals():
        base_seq = np.array(height_sequence_1, dtype=float)
        # Up-sample to target_n
        x_old = np.arange(len(base_seq))
        x_new = np.linspace(0, len(base_seq)-1, target_n)
        a = np.interp(x_new, x_old, base_seq)
    else:
        a = np.full(target_n, 1.0/target_n)
    # Normalize
    a = a / a.sum()
    n = a.shape[0]
    # Convert to logit x
    eps = 1e-12
    x = np.log(np.maximum(a, eps))

    # Adam state variables
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps_adam = 1e-8
    base_lr = 0.5

    # Compute initial objective
    conv = _full_convolution_fft(a, a)
    M = conv.max()
    cur_val = float(2.0 * n * M)
    best_val = cur_val
    best_a = a.copy()
    print(f"[init] n={n}, init val={cur_val:.6f}")

    # Stage durations
    total_budget = deadline - start_time
    stage1_budget = total_budget * 0.6  # 60% for gradient descent
    stage1_end = start_time + stage1_budget

    # Main loop for stage1
    iteration = 0
    while True:
        now = time.time()
        if now >= stage1_end:
            break
        # Compute conv, M
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        cur_val = float(2.0 * n * M)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[gd {iteration}] new best = {best_val:.6f}")

        # Temperature schedule
        progress = (now - start_time) / total_budget
        tau_start = M * 0.5
        tau_end = M * 1e-7
        tau = tau_start * (1.0 - progress) + tau_end * progress

        # compute soft-max weights
        if tau > 0:
            # shift to avoid overflow
            exp_args = (conv - M) / tau
            max_arg = exp_args.max()
            exp_vals = np.exp(exp_args - max_arg)
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # gradient wrt a
        grad_a = _gradient_from_weights(a, w)

        # gradient wrt x (softmax derivative)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        lr = base_lr * (1.0 - progress) + 1e-5 * progress
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # compute a from x
        max_x = x.max()
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # occasional random jitter
        if np.random.rand() < 0.02:
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta
            # re-normalize and reset Adam
            a = a / a.sum()
            x = np.log(np.maximum(a, eps))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0

        iteration += 1

    # Stage2: Simulated annealing
    # Use best_a as starting point
    a = best_a.copy()
    conv = _full_convolution_fft(a, a)
    M = conv.max()
    cur_val = float(2.0 * n * M)
    best_val = cur_val
    best_a = a.copy()
    # Temperature schedule for SA
    # Start temperature as fraction of current objective
    T0 = cur_val * 0.1
    # Move factor start
    move_factor_start = 0.5
    move_factor_end = 0.01

    # precompute FFT size for later use (function uses internal compute)
    while True:
        now = time.time()
        if now >= deadline:
            break
        # progress for SA
        prog = (now - stage1_end) / (deadline - stage1_end)
        T = T0 * (1.0 - prog)
        move_factor = move_factor_start * (1.0 - prog) + move_factor_end * prog

        # choose random i,j, ensure i!=j
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # sample delta
        max_delta = a[i] * move_factor
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        # propose new a
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute objective for new a
        conv_new = _full_convolution_fft(a_new, a_new)
        M_new = conv_new.max()
        new_val = float(2.0 * n * M_new)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # accept with probability exp(-(new_val-cur_val)/T)
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True
        if accept:
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sa {iteration}] new best = {best_val:.6f}")
        iteration += 1

    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence exception:", exc)

    return best_a.tolist()

"""
Propose a good step‑function.

The algorithm works in two phases:
1.  Gradient descent on a smooth approximation of the max‑convolution.
    The coefficients are kept on the probability simplex via a softmax
    (log‑space) parametrisation; Adam optimisation with a linearly‑decaying
    learning‑rate is used.  A temperature τ gradually drives the soft‑max
    weights from a smooth average to the true max, giving a sub‑gradient
    of the objective.
2.  Simulated‑annealing fine‑tuning.  Starting from the best vector found
    by phase‑1 we repeatedly move a small random amount of mass from one
    index to another.  Moves that improve the objective are always kept;
    worsening moves are accepted with a probability exp(−Δ/T) where T
    cools to zero.  This helps escape plateaus that the gradient method
    can get stuck in.

A previously discovered solution (``height_sequence_1``) is up‑sampled to a
larger length (4000) before optimisation – more degrees of freedom allow
the max‑convolution peak to be reduced further.  The entire search runs
until the supplied time budget (≈ 1000 s) is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _auto_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a*a using a single FFT (real‑valued).
    Returns an array of length 2*len(a)-1.
    Tiny negative values due to floating‑point error are clamped to 0.
    """
    n = a.shape[0]
    total = 2 * n - 1
    size = 1 << (total - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:total]
    np.maximum(conv, 0, out=conv)
    return conv


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution x*y via FFT.
    Used for the gradient w.r.t. a (convolution of reversed weights with a).
    """
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    return conv


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute   g_i = 2 * Σ_k w_k * a_{k-i}   for all i.
    This is implemented as a single convolution of reversed w with a.
    """
    n = a.shape[0]
    u = w[::-1]                          # reverse the weight vector
    conv = _full_convolution_fft(u, a)   # length = 3n‑2
    start = 2 * n - 2
    idx = start - np.arange(n)           # pick the n entries that correspond to a
    g = 2.0 * conv[idx]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence minimising
        2·n·max_k (a*a)[k]   (sum(a)=1)

    Returns the best sequence (list of floats) found before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Choose the problem size and an initial point
    # --------------------------------------------------------------
    TARGET_N = 4000                     # larger than before, still fast enough
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        # Linear interpolation to the larger size
        old_x = np.arange(base.size)
        new_x = np.linspace(0, base.size - 1, TARGET_N)
        a = np.interp(new_x, old_x, base)
    else:
        a = np.full(TARGET_N, 1.0 / TARGET_N)

    # Normalise to the simplex
    a = a / a.sum()
    n = a.shape[0]

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on a softened max‑convolution
    # --------------------------------------------------------------
    eps = 1e-12
    x = np.log(np.maximum(a, eps))          # log‑softmax parameters

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.90, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps_adam = 1e-8
    base_lr = 0.45                           # start learning rate

    # initial objective
    conv = _auto_convolution_fft(a)
    cur_val = float(2.0 * n * conv.max())
    best_val = cur_val
    best_a = a.copy()
    print(f"[init] n={n}, initial objective={best_val:.6f}")

    # allocate fractions of the total budget
    total_budget = deadline - start_time
    phase1_budget = total_budget * 0.60          # 60 % of the time
    phase1_end = start_time + phase1_budget

    iteration = 0
    while True:
        now = time.time()
        if now >= phase1_end:
            break

        iteration += 1

        # ---- current convolution and objective ----
        conv = _auto_convolution_fft(a)
        M = conv.max()
        cur_val = float(2.0 * n * M)

        if cur_val < best_val:

            best_a = a.copy()
            print(f"[gd {iteration:7d}] new best = {best_val:.6f}")

        # ---- temperature for the soft‑max (linear schedule) ----
        progress = (now - start_time) / total_budget
        tau_start = M * 0.5
        tau_end   = M * 1e-7
        tau = tau_start * (1.0 - progress) + tau_end * progress

        # ---- soft‑max weights over the convolution ----
        if tau > 0.0:

            max_arg = exp_args.max()
            exp_vals = np.exp(exp_args - max_arg)
            sum_exp = exp_vals.sum()
            if sum_exp == 0.0 or not np.isfinite(sum_exp):
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_vals / sum_exp
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ---- gradient w.r.t. a (smoothed sub‑gradient) ----
        grad_a = _gradient_from_weights(a, w)

        # ---- chain rule for the soft‑max parametrisation ----
        dot = np.dot(grad_a, a)            # <grad_a , a>
        grad_x = a * (grad_a - dot)        # ∇_x (softmax)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = base_lr * (1.0 - progress) + 1e-5 * progress
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- back‑to‑probability‑simplex (softmax) ----
        max_x = x.max()
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ---- occasional random mass–transfer (helps escape plateaux) ----
        if np.random.rand() < 0.02:        # 2 % per iteration
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            delta = np.random.rand() * a[i] * 0.5
            a[i] -= delta
            a[j] += delta
            a = a / a.sum()
            x = np.log(np.maximum(a, eps))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = 1.0, 1.0

    # --------------------------------------------------------------
    # Phase 2 – Simulated annealing fine‑tuning
    # --------------------------------------------------------------
    a = best_a.copy()
    conv = _auto_convolution_fft(a)
    cur_val = float(2.0 * n * conv.max())
    best_val = cur_val
    best_a = a.copy()

    # Temperature starts as a fraction of the current objective
    T0 = cur_val * 0.10
    move_start = 0.5
    move_end = 0.02

    iteration_sa = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        iteration_sa += 1

        # schedule progress for this stage
        prog = (now - phase1_end) / (deadline - phase1_end)
        T = T0 * (1.0 - prog)                # linear cooling
        move_factor = move_start * (1.0 - prog) + move_end * prog

        # pick a random ordered pair (i -> j) and a random transfer amount
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        max_delta = a[i] * move_factor
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * max_delta

        # propose the new vector
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate the proposal
        conv_new = _auto_convolution_fft(a_new)
        new_val = float(2.0 * n * conv_new.max())

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(new_val - cur_val) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[sa {iteration_sa:7d}] new best = {best_val:.6f}")
