def phase_projection(a, iters=10):
    n = len(a)
    L = 1 << int(np.ceil(np.log2(2*n)))  # next pow2 >= 2n
    for _ in range(iters):

        total_energy = np.sum(a*a)
        # set magnitude to sqrt(total_energy)
        X = np.sqrt(total_energy) * np.exp(1j * np.angle(X))
        a = np.real(np.fft.ifft(X))[:n]
        # clip to nonnegative and renormalize sum to 1
        a = np.maximum(a, 0)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
    return a

"""
Propose a non‑negative step‑function (probability vector) that minimises

        2 * n * max_k (a * a)[k]

where * denotes the full linear convolution.  The algorithm consists of

1.  A fast deterministic sweep over several symmetric parametric families
    (U‑shapes, shifted U‑shapes, product‑U, exponential, cosine).  The best
    vector from this sweep seeds the stochastic phase.

2.  Stochastic refinement that mixes several complementary update rules:
    •  Exponential‑gradient (mirror descent) on the simplex.
    •  Projected Euclidean‑gradient descent.
    •  Gradient‑guided mass‑transfer (move mass from the index that most
       increases the current maximal convolution to the index that least
       increases it).
    •  Pure random mass‑transfer (as a source of exploration).
    •  Every few thousand iterations a “phase‑retrieval” style projection
       that forces the magnitude of the FFT to be constant – this tends to
       flatten the autocorrelation.

    Candidates are evaluated with the exact score (full convolution via FFT);
    a simple Metropolis‑type rule accepts improvements or, with a temperature
    schedule, occasional deteriorations.  Stagnation is detected and a small
    random re‑heat is performed.

The routine respects the given time budget (default 1000 s) and returns the
best sequence found as a plain Python list whose entries sum to ≈ 1.
"""

import time

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Projects a vector onto the probability simplex { w >= 0 , sum(w) = 1 }."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 2 * n          # power‑of‑2 zero‑padding is not required here
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score for a probability vector (sum ≈ 1)."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    Only the positions attaining (or very close to) the maximum are used.
    """
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        i_vals = np.arange(start_i, end_i + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Mirror‑descent / exponentiated‑gradient step on the simplex."""
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new


def additive_update(a: np.ndarray, grad: np.ndarray, lr: float) -> np.ndarray:
    """Projected Euclidean‑gradient step on the simplex."""
    a_new = a - lr * grad
    a_new = simplex_projection(a_new)
    return a_new


def guided_mass_transfer(a: np.ndarray, grad: np.ndarray) -> np.ndarray:
    """
    Transfer a random fraction of mass from the index with the largest
    gradient (most harmful) to the index with the smallest gradient.
    """
    n = a.shape[0]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst or a[src] == 0.0:
        return a.copy()
    # move at most 50 % of the source mass
    delta = a[src] * np.random.rand() * 0.5
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


def random_mass_transfer(a: np.ndarray) -> np.ndarray:
    """A pure random redistribution of a small amount of mass."""
    n = a.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    if a[i] == 0.0:
        return a.copy()
    delta = a[i] * np.random.rand() * 0.5
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


def phase_projection(a: np.ndarray, iters: int = 12) -> np.ndarray:
    """
    Alternating‑projection inspired step that enforces a constant magnitude
    of the FFT (the “flat spectrum” heuristic).  The iteration keeps the
    vector non‑negative and on the probability simplex.
    """
    n = a.shape[0]
    # length for zero‑padding – next power of two ≥ 2·n
    L = 1 << int(np.ceil(np.log2(2 * n)))
    a_cur = a.copy()
    for _ in range(iters):
        # FFT of the zero‑padded vector
        X = np.fft.fft(a_cur, n=L)
        total_energy = np.sum(a_cur * a_cur)
        # impose constant magnitude
        X = np.sqrt(total_energy) * np.exp(1j * np.angle(X))
        # back‑transform, keep the first n entries (the rest are padding)
        a_cur = np.real(np.fft.ifft(X))[:n]
        # enforce non‑negativity and normalise
        a_cur = np.maximum(a_cur, 0.0)
        s = a_cur.sum()
        if s == 0.0:
            a_cur = np.full(n, 1.0 / n)
        else:
            a_cur /= s
    return a_cur

def u_min_distribution(n: int, p: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for the best non‑negative height sequence under the time budget."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 3.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # A modest collection of lengths – more lengths give more flexibility
    lengths = [4000, 6000, 8000, 12000, 16000, 20000, 26000, 32000]

    # Power‑law exponents
    exponents = np.arange(0.1, 6.01, 0.1)

    # Shift constants for the shifted U‑shape
    shift_cs = [0.5, 1.0, 2.0, 5.0]

    # Exponential decay (log‑spaced)
    lam_vals = np.logspace(-4, -0.2, num=35)

    # Cosine exponents
    cos_exps = np.arange(0.5, 9.5, 0.5)

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break
        # --- power‑law families ---
        for p in exponents:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"new best (min)   n={n:5d} p={p:4.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"new best (prod)  n={n:5d} p={p:4.2f}  score={sc:.6f}")

            for c in shift_cs:
                a = u_min_shift_distribution(n, p, c)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"new best (shift) n={n:5d} p={p:4.2f} c={c:3.1f}  score={sc:.6f}")

        # --- exponential family ---
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"new best (exp)   n={n:5d} lam={lam:.5f}  score={sc:.6f}")

        # --- cosine family ---
        for p in cos_exps:
            a = cos_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"new best (cos)   n={n:5d} p={p:3.1f}  score={sc:.6f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. External construction (if provided)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score = sc
                    best_seq = external.copy()
                    print(f"using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f"fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]

    # Ensure a clean probability vector
    current = np.maximum(current, 0.0)
    current /= current.sum()

    current_conv = full_convolution_fft(current)
    current_score = score_from_vector(current)

    # Hyper‑parameters
    temperature = 1.0
    temp_decay = 0.9998
    eta = 0.7            # exponential‑gradient step size
    eta_decay = 0.9997
    lr_add = 0.12        # additive‑gradient step size
    lr_decay = 0.9999

    iteration = 0
    last_improve = 0
    phase_interval = 8500   # how often we run the phase‑projection heuristic

    print("\n=== Stochastic refinement start ===")
    while time.time() < deadline:
        iteration += 1

        # ---- gradient information (sub‑gradient of the current max) ----
        grad = compute_subgradient(current, current_conv)

        # ---- Candidate 1: exponential‑gradient (mirror descent) ----
        cand_eg = exponential_update(current, grad, eta)

        # ---- Candidate 2: additive projected gradient descent ----
        cand_add = additive_update(current, grad, lr_add)

        # ---- Candidate 3: guided mass‑transfer (source/high‑grad → target/low‑grad) ----
        cand_guided = guided_mass_transfer(current, grad)

        # ---- Candidate 4: pure random mass‑transfer (exploration) ----
        if iteration % 256 == 0:
            cand_rand = random_mass_transfer(current)
        else:
            cand_rand = None

        # ---- Evaluate all candidates (skip None) ----
        candidates = [
            (cand_eg, "eg"),
            (cand_add, "add"),
            (cand_guided, "guided"),
        ]
        if cand_rand is not None:
            candidates.append((cand_rand, "rand"))

        best_cand = None
        best_cand_score = np.inf
        best_cand_conv = None
        best_cand_tag = None

        for cand, tag in candidates:
            conv_c = full_convolution_fft(cand)
            sc = float(2.0 * n * conv_c.max())
            if sc < best_cand_score:
                best_cand_score = sc
                best_cand = cand
                best_cand_conv = conv_c
                best_cand_tag = tag

        # ---- Candidate 5: periodic phase‑projection (every phase_interval steps) ----
        if iteration % phase_interval == 0:
            cand_phase = phase_projection(current, iters=8)
            conv_phase = full_convolution_fft(cand_phase)
            sc_phase = float(2.0 * n * conv_phase.max())
            if sc_phase < best_cand_score:
                best_cand = cand_phase
                best_cand_score = sc_phase
                best_cand_conv = conv_phase
                best_cand_tag = "phase"
                # note that phase_projection tends to be expensive; we keep it sparse

        # ---- Simulated‑annealing acceptance ----
        accept = False
        if best_cand_score < current_score:
            accept = True
        else:
            prob = np.exp((current_score - best_cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = best_cand
            current_conv = best_cand_conv
            current_score = best_cand_score
            if best_cand_score < best_score:
                best_score = best_cand_score
                best_seq = best_cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best {best_cand_tag} score = {best_score:.6f}")

        # ---- schedule updates ----
        temperature *= temp_decay
        eta *= eta_decay
        lr_add *= lr_decay

        # ---- stagnation handling (re‑heat) ----
        if iteration - last_improve > 250000:
            bump = np.random.rand(n) * 1e-5
            current = np.maximum(current + bump, 0.0)
            current /= current.sum()
            current_conv = full_convolution_fft(current)
            current_score = score_from_vector(current)
            temperature = 1.0
            eta = 0.7
            lr_add = 0.12
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    # ------------------------------------------------------------------
    # 5. Return best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
