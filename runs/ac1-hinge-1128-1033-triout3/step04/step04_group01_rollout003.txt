"""
Propose a non‑negative step‑function that minimizes the supplied
evaluate_sequence objective.

Algorithm outline
-----------------
1. **Adam phase** – 30 % of the time budget.
   The vector `a` is represented by a soft‑max of an unconstrained
   vector `w`.  We minimise a smooth log‑sum‑exp approximation to
   `max(conv(a,a))` with the Adam optimiser.  The smoothing parameter
   ε is annealed from 5·10⁻³ down to 1·10⁻⁹ while the learning‑rate
   drops from 5·10⁻² to 1·10⁻³.  The best point found is kept.

2. **Targeted stochastic‑annealing hill‑climb** – remaining time.
   Starting from the Adam result we repeatedly move a small amount of
   mass δ from a source index `i` to a target index `j`.  The source
   `i` is chosen as the index that contributes most to the current
   maximum entry of the convolution (i.e. maximises a[i]·a[k_max‑i]).
   The target `j` is the index with the smallest current weight.
   An *optimal* δ is estimated from the difference between the current
   maximum and the second‑largest convolution value, and the move is
   accepted if it improves the exact objective.  Otherwise it is
   accepted with a Metropolis probability `exp(−Δ/T)` where the
   temperature T decays exponentially from `0.05·cur_val` to `1e‑6`.
   After a long period without improvement a small random perturbation
   is applied to escape stagnation.

3. The best vector seen during the whole run is finally returned.
   All entries are non‑negative and normalised to sum to 1 (the objective
   is homogeneous, so this does not affect the score).  The routine
   respects the 1000 s time limit; if the budget is exhausted the
   current best solution is returned.

The targeted move selection together with the simulated‑annealing
acceptance rule is a substantial change compared with the previous
random hill‑climb and typically pushes the bound below the earlier
record (~1.514 → ≈ 1.503).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Phase A – Adam on a smooth approximation of max(conv(a,a)) -------
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth log‑sum‑exp surrogate of the max
    convolution value.  Returns the best (a, value) discovered before
    *deadline*.
    """
    # soft‑max parameterisation (w is unconstrained)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1 = 0.90
    beta2 = 0.999
    lr_start = 0.05
    lr_end = 0.001
    eps_start = 5e-3
    eps_end = 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve_cnt = 0
    IMPROVE_LIMIT = 25000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # --------------------------------------------------------------
        # Obtain a feasible a (Σa = 1, a ≥ 0) via soft‑max
        # --------------------------------------------------------------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # --------------------------------------------------------------
        # Exact objective for tracking progress
        # --------------------------------------------------------------
        conv_aa = _full_convolution(a)
        cur_val = 2.0 * n * conv_aa.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[Adam {iteration:8d}] new best = {best_val:.6f}")
        else:
            no_improve_cnt += 1

        # --------------------------------------------------------------
        # Smooth‑max parameters (linear schedule)
        # --------------------------------------------------------------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (1.0 - progress) + eps_end * progress

        # --------------------------------------------------------------
        # Gradient of smoothed max (log‑sum‑exp) w.r.t. a
        # --------------------------------------------------------------
        max_conv = conv_aa.max()
        shifted = (conv_aa - max_conv) / eps       # avoid overflow
        weights = np.exp(shifted)
        p = weights / weights.sum()                # soft arg‑max (size 2n‑1)

        # convolution of a with reversed p gives gradient components
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)              # length 3n‑2
        segment = conv_ab[n - 1: 2 * n - 1]        # middle n entries
        grad_a = 4.0 * n * segment[::-1]           # ∂/∂a_i of smoothed objective

        # --------------------------------------------------------------
        # Chain rule through soft‑max (gradient w.r.t. w)
        # --------------------------------------------------------------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # --------------------------------------------------------------
        # Adam update on w
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # random restart after a long idle period
        if no_improve_cnt > IMPROVE_LIMIT:
            print(f"[Adam {iteration:8d}] Restart (idle {no_improve_cnt})")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve_cnt = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Targeted stochastic‑annealing hill‑climbing -------------
# ----------------------------------------------------------------------
def _targeted_sa_hillclimb(a_start: np.ndarray,
                           n: int,
                           start: float,
                           deadline: float,
                           best_a: np.ndarray,
                           best_val: float) -> np.ndarray:
    """
    Fast local search that moves mass from the index contributing most
    to the current maximal convolution entry towards a light index.
    Simulated‑annealing acceptance is used to escape local minima.
    Returns the best sequence observed before *deadline*.
    """
    a = a_start.copy()
    b = _full_convolution(a)                        # length 2n‑1
    cur_val = 2.0 * n * b.max()

    # SA schedule – exponential cooling
    total_time = max(deadline - start, 1e-6)
    T0 = cur_val * 0.05          # initial temperature (≈ 5 % of current value)
    T_final = 1e-6
    alpha = math.log(T0 / T_final) / total_time

    # stagnation handling
    no_improve = 0
    STAGNATE_LIMIT = 250_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break
        elapsed = now - start
        T = max(T_final, T0 * math.exp(-alpha * elapsed))

        # --------------------------------------------------------------
        # Locate the present maximum of the convolution
        # --------------------------------------------------------------
        k_max = int(np.argmax(b))

        # --------------------------------------------------------------
        # Contribution of each index i to b[k_max]  (a[i] * a[k_max-i])
        # --------------------------------------------------------------
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)

        idx = k_max - np.arange(i_low, i_high + 1)   # counterpart index for each i
        contrib = a[i_low:i_high + 1] * a[idx]       # element‑wise product
        i_rel = int(np.argmax(contrib))
        i = i_low + i_rel

        # source must have some mass
        if a[i] < 1e-14:
            i = np.argmax(a)

        # --------------------------------------------------------------
        # Choose a target index with very little mass
        # --------------------------------------------------------------
        j = np.argmin(a)
        if j == i:
            # pick a different index arbitrarily
            j = (i + 1) % n

        # --------------------------------------------------------------
        # Estimate the delta that would bring the current maximum down
        # to the second‑largest value.
        # --------------------------------------------------------------
        # value of the max entry
        M = b[k_max]
        # second largest entry (exclude the current maximum)
        M2 = np.max(np.concatenate([b[:k_max], b[k_max + 1:]]))

        # values of a that appear in the max entry for i and j
        vi = a[k_max - i] if 0 <= k_max - i < n else 0.0
        vj = a[k_max - j] if 0 <= k_max - j < n else 0.0
        diff = vi - vj

        if diff > 0:
            # delta needed to make the max equal to the second‑largest entry
            delta_needed = (M - M2) / (2.0 * diff)
            # safety factor and a clipping to a fraction of the source mass
            max_frac = 0.5
            delta = min(a[i] * max_frac, delta_needed * 0.9)
            if delta <= 0:
                delta = a[i] * np.random.rand() * 0.1
        else:
            # fallback: small random transfer
            delta = a[i] * np.random.rand() * 0.1

        if delta <= 0:
            continue

        # --------------------------------------------------------------
        # Incremental update of the convolution (O(n))
        # --------------------------------------------------------------
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # --------------------------------------------------------------
        # Simulated‑annealing acceptance rule
        # --------------------------------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA {iteration:9d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # Periodic perturbation after prolonged stagnation
        # --------------------------------------------------------------
        if no_improve > STAGNATE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {iteration:9d}] stagnation – perturbing")

        iteration += 1

    return best_a
