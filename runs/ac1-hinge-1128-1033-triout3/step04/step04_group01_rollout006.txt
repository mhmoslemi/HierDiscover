def _project_simplex(v):
    # from https://arxiv.org/pdf/1101.6081.pdf
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step‑function that minimises `evaluate_sequence`.

Algorithm
---------

1. **Initialisation** – start from the known construction `height_sequence_1`
   (if present) or from a random Dirichlet vector on the simplex.

2. **Phase C – Gradient descent on the L²‑error of the convolution**  
   We want the autoconvolution `b = a∗a` to be as flat as possible.
   Let `b̄ = 1/(2·n‑1)` be the target constant (the average of `b`
   when `∑a = 1`).  The loss `L = Σ(b_k‑b̄)²` is smooth; its gradient
   can be written as a pair of FFT‑based convolutions:

       grad = 2·( conv(b‑b̄ , a)   +   conv(b‑b̄ , a_rev) )[n‑1:2·n‑1]

   After each gradient step we **project** the trial vector back onto the
   probability simplex (non‑negative entries summing to 1).  The step size
   decays linearly during the allotted time.

   The loss directly penalises any peak of `b`, so minimising it drives
   `max(b)` – and consequently the evaluation score – down.

3. **Phase D – Deterministic hill‑climbing**  
   Starting from the best point found in Phase C we perform cheap
   mass‑transfer moves: a small amount `δ` is moved from a source index
   `i` to a target `j`.  The effect on the full convolution can be
   updated in `O(n)` time using the bilinearity of the convolution.
   Only moves that strictly improve the true objective
   `2·n·max(b)` are accepted; after long stagnation we apply a small
   random perturbation to escape local minima.

Both phases fit easily into the 1000 s budget (≈55 % for the gradient
phase, the remainder for hill‑climbing).  The code is self‑contained,
uses only NumPy and the provided `evaluate_sequence`, and returns a list
of floats ready for scoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolutions (real data)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x≥0, Σx=1})
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # find the rho value
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)

    w = v - theta
    w[w < 0] = 0.0
    # numerical drift may cause sum != 1; renormalise
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase D – deterministic hill‑climbing (same as before, slightly tweaked)
# ----------------------------------------------------------------------
def _hillclimb_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """
    Fast local search that moves infinitesimal mass between coordinates
    while keeping Σa = 1.  Returns the best sequence seen before *deadline*.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac = 0.3                           # max fraction of source mass to move
    IMPROVE_LIMIT = 150_000                  # stagnation trigger
    no_improve = 0
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # Choose source – usually the current largest entry
        if np.random.rand() < 0.85:
            i = int(np.argmax(a))
        else:
            i = np.random.choice(n, p=a)

        if a[i] <= 1e-14:
            continue

        # Choose target ≠ source
        j = np.random.randint(n)
        if j == i:
            continue

        # Amount to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of convolution (O(n))
        b_new = b.copy()
        # cross‑terms (2·δ·a) added/subtracted by shifting a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # Accept only improving moves
        if new_val < cur_val:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # Random perturbation after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturbing")

        iteration += 1

    return best_a
