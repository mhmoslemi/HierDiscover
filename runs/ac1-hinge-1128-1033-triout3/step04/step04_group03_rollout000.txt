"""
Propose a candidate step‑function (non‑negative vector) that minimizes

    2 * n * max_k (a * a)[k] / (sum a)^2

The algorithm is a two‑phase search:

1. **Conditional‑gradient (Frank–Wolfe) using a linear program**  
   For a fixed length `n` we start from a simple parametric shape
   (U‑shaped power law).  Given the current vector `f` we compute
   `M = max (f*f)`.  We then solve the linear program

        maximise   sum(g)
        subject to  f * g ≤ M          (elementwise)
                    g ≥ 0

   where `f * g` is a linear convolution, therefore the constraints are
   linear in `g`.  The LP yields a vector `g` with a possibly larger
   total mass while respecting the same peak bound `M`.  By scaling `g`
   back to the original ℓ₁‑norm of `f` and convex‑combining
   `f ← (1‑t) f + t g` we strictly reduce the peak `M`.  Repeating this
   step a few times drives the objective down.

2. **Exponential‑gradient refinement**  
   After the LP phase we fine‑tune the result on the probability simplex
   using the same mirror‑descent scheme that the previous baseline used.

The whole routine respects the given time budget (default 1000 s) and
returns the best sequence found as a plain Python list (ℓ₁‑norm = 1).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex { w >= 0, sum(w)=1 }."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a via real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def score_from_vector(a: np.ndarray) -> float:
    """Exact score returned by evaluate_sequence for a sum‑1 vector."""
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


def compute_subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a."""
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-15
    max_idx = np.where(conv >= max_val - eps)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        grad[i_vals] += 2.0 * a[j_vals]
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient (mirror descent) update on the simplex."""
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new


# ----------------------------------------------------------------------
# Parametric families (all symmetric)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on distance to the nearer edge: (min+1)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """U‑shape based on product of distances to both edges: ((i+1)*(n‑i))^-p."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted U‑shape: (min(i,n‑1‑i)+c)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based U‑shape: (1 + cos(π i/(n‑1)))^p."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (Frank–Wolfe step)
# ----------------------------------------------------------------------


def lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   sum(g)
        subject to  f * g ≤ M   (full convolution)
                    g ≥ 0
    Returns the optimal vector g (may be scaled later).
    """
    n = f.shape[0]
    # Build the constraint matrix A where (A @ g)[k] = (f * g)[k]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Each column i of A is f shifted by i positions
    for i in range(n):
        A[i:i + n, i] = f
    b = np.full(rows, M, dtype=np.float64)

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # Solve with HiGHS (fast interior‑point/simplex)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        # In the unlikely event of failure, fall back to the current f
        return f.copy()
    return res.x


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the allotted time.
    Returns a plain Python list (ℓ₁‑norm ≈ 1).  The implementation combines a
    Frank–Wolfe LP phase with an exponential‑gradient refinement phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0. Optional external seed
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score = sc

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400]   # moderate sizes for LP tractability

        f = init_f.copy()

    # 5. Return result
    # ------------------------------------------------------------------
    print("\n=== Search completed ===")
    print(f"Best score = {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
