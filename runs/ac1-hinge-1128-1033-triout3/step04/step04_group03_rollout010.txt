"""
Propose a candidate non‑negative step‑function (probability vector) that
minimises

        2 * n * max_k (a * a)[k]

where * is the full linear convolution.
The algorithm works in two phases:

1️⃣  Deterministic sweep over many *parametric families* – both symmetric
    (the original U‑, exponential, cosine families) and *asymmetric*
    (left‑hand power‑law, left‑hand exponential, linear‑decline).  For each
    length and each set of parameters the exact score (2·n·max(conv))
    is evaluated; the best vector found seeds the stochastic phase.

2️⃣  Stochastic refinement on the probability simplex using a *smooth
    surrogate* for the max‑operator.
    We replace max(conv) by a soft‑max with a temperature α
    (log‑sum‑exp).  The gradient of this smooth objective is obtained in
    closed form using a convolution with the soft‑max weights.
    An exponentiated‑gradient (mirror‑descent) step is taken,
    the iterate is projected back onto the simplex and a simulated‑annealing
    acceptance rule (with a slowly cooling temperature) decides whether to
    keep the move.  α is increased progressively (the surrogate becomes
    sharper) while the step‑size η and temperature decay.
    Random “mass‑transfer’’ moves and periodic re‑heating help escape
    stagnation.

The whole search respects the given time budget (default 1000 s) and
returns the best vector found as a plain Python list (its sum is ≈ 1, the
external evaluator will rescale it as required).
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {w>=0, sum(w)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u - (cssv - 1) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of real vectors a and b using an FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[:n]


def full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (real FFT)."""
    return convolution_fft(a, a)


def score_from_vector(a: np.ndarray) -> float:
    """
    Exact score used by evaluate_sequence for a vector with sum(a)=1.
    Returns 2 * n * max(conv).
    """
    n = a.shape[0]
    conv = full_convolution_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families (both symmetric and asymmetric)
# ----------------------------------------------------------------------


def u_min_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape based on distance to the nearer edge: (min+1)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_prod_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric U‑shape based on product of distances to both edges."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)
    a = prod ** (-p) + 1e-14
    a /= a.sum()
    return a


def u_min_shift_distribution(n: int, p: float, c: float) -> np.ndarray:
    """Shifted symmetric U‑shape: (min+ c)^-p."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + c) ** (-p) + 1e-14
    a /= a.sum()
    return a


def exp_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def cos_distribution(n: int, p: float) -> np.ndarray:
    """Cosine‑based symmetric U‑shape."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** p + 1e-14
    a /= a.sum()
    return a


def left_power_distribution(n: int, p: float) -> np.ndarray:
    """Monotonically decreasing power‑law (left‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + 1e-14
    a /= a.sum()
    return a


def left_exp_distribution(n: int, lam: float) -> np.ndarray:
    """Monotonically decreasing exponential (left‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + 1e-14
    a /= a.sum()
    return a


def linear_decline_distribution(n: int) -> np.ndarray:
    """Linear decline from left to right (a_i ∝ n‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) + 1e-14
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max utilities for the stochastic phase
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max (log‑sum‑exp) weights for a vector of convolution entries.
    w_k = exp(alpha * conv_k) / Σ_j exp(alpha * conv_j)
    """
    # For numerical stability use the log‑sum‑exp trick
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exps = np.exp(scaled - max_scaled)
    w = exps / np.sum(exps)
    return w


def smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) objective w.r.t. a.
    ∂/∂a_i  ( (1/α) log Σ_k exp(α conv_k) )  =  2 * Σ_j a_j w_{i+j}
    where w are the soft‑max weights.
    """
    w = softmax_weights(conv, alpha)               # length = 2*n-1
    w_rev = w[::-1]
    # convolution a * w_rev  (full length = n + (2n-1) - 1 = 3n-2)
    conv_aw = convolution_fft(a, w_rev)
    L = len(w)                                     # = 2*n-1
    n = a.shape[0]
    # g_i = 2 * conv_aw[L-1-i]   (see derivation in analysis)
    g = 2.0 * conv_aw[(L - 1) - np.arange(n)]
    return g


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent / exponentiated‑gradient step on the simplex:
        a' ∝ a * exp(-η * grad)
    The result is re‑normalised to sum = 1.
    """
    # Clip to avoid overflow/underflow
    exp_term = np.exp(-eta * grad)
    a_new = a * exp_term
    total = a_new.sum()
    if total == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / total
    return a_new


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within `budget_s` seconds.
    Returns the sequence as a plain Python list (its sum is ≈ 1.0).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 3.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic sweep over a rich set of parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Lengths to explore – a trade‑off between quality and speed
    lengths = [4000, 6000, 8000, 12000, 16000, 20000, 26000, 32000]

    # Parameter grids
    p_vals = np.arange(0.1, 8.01, 0.2)          # for power‑law families
    p_fine = np.arange(8.2, 15.01, 0.5)        # extra fine grid for left‑heavy
    all_p = np.concatenate([p_vals, p_fine])

    shift_cs = [0.5, 1.0, 2.0, 5.0]

    lam_vals = np.logspace(-4, -0.2, num=40)  # for exponentials

    cos_exps = np.arange(0.5, 12.5, 0.5)

    print("=== Deterministic sweep start ===")
    for n in lengths:
        if time.time() > deadline:
            break

        # -------- symmetric families --------
        for p in all_p:
            a = u_min_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (sym‑U‑min)  n={n:5d} p={p:5.2f}  score={sc:.6f}")

            a = u_prod_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (sym‑U‑prod) n={n:5d} p={p:5.2f}  score={sc:.6f}")

            # shifted symmetric U‑shapes
            for c in shift_cs:
                a = u_min_shift_distribution(n, p, c)
                sc = score_from_vector(a)
                if sc < best_score:
                    best_score, best_seq = sc, a.copy()
                    print(f"new best (sym‑shift) n={n:5d} p={p:5.2f} c={c:.1f}  score={sc:.6f}")

        # exponential (symmetric)
        for lam in lam_vals:
            a = exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (sym‑exp)   n={n:5d} lam={lam:.5g}  score={sc:.6f}")

        # cosine family
        for p in cos_exps:
            a = cos_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (cos)       n={n:5d} p={p:4.1f}  score={sc:.6f}")

        # -------- asymmetric families --------
        for p in all_p:
            a = left_power_distribution(n, p)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (left‑pow)  n={n:5d} p={p:5.2f}  score={sc:.6f}")

        for lam in lam_vals:
            a = left_exp_distribution(n, lam)
            sc = score_from_vector(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"new best (left‑exp)  n={n:5d} lam={lam:.5g}  score={sc:.6f}")

        # linear decline (asymmetric)
        a = linear_decline_distribution(n)
        sc = score_from_vector(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
            print(f"new best (linear)    n={n:5d}  score={sc:.6f}")

        # occasional time check
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2. External construction (if provided)
    # ------------------------------------------------------------------
    try:
        external = np.array(height_sequence_1, dtype=float)   # may be undefined
        if external.size:
            external = np.maximum(external, 0.0)
            s = external.sum()
            if s > 0.0:
                external /= s
                sc = score_from_vector(external)
                if sc < best_score:
                    best_score, best_seq = sc, external.copy()
                    print(f"using external height_sequence_1  score={sc:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = score_from_vector(best_seq)
        print(f"fallback random seed  n={n}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Stochastic refinement with smooth‑max gradient descent
    # ------------------------------------------------------------------
    current = best_seq.copy()
    n = current.shape[0]
    current = np.maximum(current, 0.0)
    current /= current.sum()
    cur_conv = full_convolution_fft(current)
    cur_score = float(2.0 * n * cur_conv.max())

    # hyper‑parameters

    eta = 0.6
    eta_decay = 0.9998

    while time.time() < deadline:

        # -------- 4a. smooth gradient ----------
        grad = smooth_gradient(current, cur_conv, alpha)

        # -------- 4b. exponentiated‑gradient step ----------
        cand = exponential_update(current, grad, eta)

        # -------- 4c. occasional mass‑transfer move ----------
        if iteration % 256 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            if current[i] > 0.0:

            else:

                cand_score = float(2.0 * n * cand_conv.max())
        else:
            cand_conv = full_convolution_fft(cand)
            cand_score = float(2.0 * n * cand_conv.max())

        # -------- 4d. simulated‑annealing acceptance ----------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp((cur_score - cand_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current, cur_conv, cur_score = cand, cand_conv, cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                last_improve = iteration
                print(f"Iter {iteration:7d}: new best score = {best_score:.6f}")

        # -------- 4e. schedule updates ----------
        temperature *= temp_decay
        eta *= eta_decay
        alpha *= alpha_growth

        # -------- 4f. re‑heat if stuck ----------
        if iteration - last_improve > 120000:

            current = np.maximum(current + bump, 0.0)
            current /= current.sum()
            cur_conv = full_convolution_fft(current)
            cur_score = float(2.0 * n * cur_conv.max())
            temperature = 1.0
            eta = 0.6
            alpha = 1.0
            last_improve = iteration
            print(f"Iter {iteration:7d}: re‑heat after stagnation")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found
    # ------------------------------------------------------------------
    print("\n=== Search finished ===")
    print(f"Best score found: {best_score:.6f}   (length {len(best_seq)})")
    return best_seq.tolist()
