def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np
    # Setup seed, RNG
    deadline = time.time() + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # Determine base n_small
    n_small = 2000
    # Hyperparameter grid
    p_vals = np.linspace(0.2, 1.2, 51)  # 0.02 step approx
    offset_vals = np.array([0.5, 1.0, 2.0, 4.0])  # offsets >0
    best_val = float('inf')
    best_a = None
    best_params = None
    for off in offset_vals:
        idx = np.arange(n_small, dtype=np.float64) + off
        for p in p_vals:
            a = np.power(idx, -p)
            if a.sum()==0:
                continue
            a = a / a.sum()
            # compute conv via FFT
            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            score = 2 * n_small * max_conv
            if score < best_val:
                best_val = score
                best_a = a.copy()
                best_params = (off,p)
    print("[grid search] best score", best_val, "params", best_params)

    # Use best_a as start for iterative refinement
    a = best_a.copy()
    n = n_small
    # Optimize via subgradient descent
    vel = np.zeros_like(a)
    momentum = 0.9
    step = 0.5
    step_min = 1e-6
    step_decay = 0.9998
    stagn = 0
    best_val = 2 * n * np.convolve(a, a).max()
    best_seq = a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        # gradient mask for max entries
        mask = (conv >= max_val - 1e-15)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = w / w_sum
        rev_a = a[::-1]
        conv_rev = _full_convolution_fft(rev_a, w)
        g = 2.0 * conv_rev[n-1:n-1+n]
        # momentum
        vel = momentum * vel + (1-momentum) * g
        a_new = a - step * vel
        a_new = _project_simplex(a_new)
        a = a_new

        # step decay
        step = max(step_min, step * step_decay)

        # scoring
        score = 2 * n * max_val
        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it}] new best {best_val:.12f}")
        else:
            stagn += 1

        if stagn > 5000:
            # perturb
            noise = rng.normal(scale=0.005, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            step = max(step_min, step * 1.5)
            print("[perturb] iteration", it)

        if it % 2000 == 0:
            a = a / a.sum()

    # finalize polishing with subgradient steps
    a = best_seq.copy()
    for _ in range(5000):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        idxs = np.flatnonzero(conv >= mx - 1e-15)
        idx = int(idxs[0])
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi+1)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]
        a = _project_simplex(a - 1e-6 * g)

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print("[result] final_score", final_score, "length", len(final_seq))
    return final_seq

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a simple two‑phase strategy:

1. **Coarse family search** – explore one‑parameter families
   a_i ∝ (i + δ)^(-p) (δ > 0, p > 0) on a modest size
   (n≈2000).  The best shape from this grid is taken as a warm‑start.

2. **Refinement** – run a projected (Euclidean) sub‑gradient descent
   on the simplex, where the sub‑gradient is the average of the
   gradients of all currently maximal convolution entries.
   Momentum, step‑size decay and occasional random perturbations
   are employed.  Finally a short max‑entry polishing step is applied.

All arithmetic is performed with double precision NumPy arrays.
The routine respects the 1000 s time budget and returns the best
sequence discovered (clamped to the limits required by `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x | x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using real FFT (zero‑padded to next power of two).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()      # smallest power of two ≥ size
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Main candidate generator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative sequence that (approximately) minimizes the
    evaluation function.  The function runs for at most `budget_s`
    seconds (a 5‑second safety margin is kept) and prints progress.
    """
    # ------------------------------------------------------------------
    # Global time budget.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a modest problem size for the heavy optimisation stage.
    # (The theoretical optimum is size‑independent; a smaller size
    #  speeds up the inner loop dramatically.)
    # ------------------------------------------------------------------
    n_small = 2000                     # fast enough for many iterations
    # ------------------------------------------------------------------
    # Phase 1 – coarse grid search over a simple power‑law family.
    # ------------------------------------------------------------------
    # family: a_i ∝ (i + delta)^(-p)   (delta > 0, p > 0)
    delta_vals = np.linspace(0.2, 4.0, 20)          # 20 candidates
    p_vals = np.linspace(0.20, 1.30, 56)          # step ≈0.02
    best_score = float('inf')
    best_a = None
    best_params = None

    print("[phase 1] start grid search (n=%d)" % n_small)

    for delta in delta_vals:
        idx = np.arange(n_small, dtype=np.float64) + delta
        for p in p_vals:
            a = np.power(idx, -p)
            s = a.sum()
            if s == 0.0:
                continue
            a = a / s
            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            score = 2.0 * n_small * max_conv          # sum(a)==1 ⇒ score = 2·n·max_conv
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_params = (float(delta), float(p))
    print("[phase 1] best grid score = %.12f  params = %s" %
          (best_score, best_params))

    # ------------------------------------------------------------------
    # Phase 2 – refined sub‑gradient descent on the simplex.
    # ------------------------------------------------------------------
    a = best_a.copy()
    n = n_small
    vel = np.zeros_like(a)               # momentum buffer
    momentum = 0.95
    step = 0.6                           # initial Euclidean step size
    step_min = 5e-7
    step_decay = 0.99985
    stagn = 0
    it = 0
    best_val = best_score
    best_seq = a.copy()

    # small tolerance when detecting the active (maximal) set
    active_tol = 1e-15

    while time.time() < deadline:
        it += 1

        # ----- convolution and current score -----
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # ----- keep the global best -----
        if cur_score < best_val - 1e-13:
            best_val = cur_score
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it:6d}] new best = {best_val:.12f}  step = {step:.2e}")
        else:
            stagn += 1

        # ----- build the sub‑gradient of the max‑entry(s) -----
        # mask of (almost) maximal entries
        mask = (conv >= max_conv - active_tol)
        w = mask.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum                       # normalise to a probability vector

        # gradient = 2 * (rev(a) * w)  (central slice of the full convolution)
        rev_a = a[::-1]
        conv_rev = _full_convolution_fft(rev_a, w)   # length = 3n-2
        g = 2.0 * conv_rev[n - 1:n - 1 + n]

        # ----- momentum update -----
        vel = momentum * vel + (1.0 - momentum) * g

        # ----- Euclidean step + projection onto the simplex -----
        a_new = a - step * vel
        a_new = _project_simplex(a_new)
        a = a_new

        # ----- decay the step size -----

        # ----- occasional random perturbation to escape plateaus -----
        if stagn > 5000:

            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            step = max(step_min, step * 1.4)
            print(f"[perturb] iteration {it}")

        # ----- periodic re‑normalisation (numerical safety) -----
        if it % 3000 == 0:
            a = a / a.sum()

    # ------------------------------------------------------------------
    # Phase 3 – pure max‑entry sub‑gradient polishing (few thousand steps).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    for _ in range(2500):
        conv = _full_convolution_fft(a, a)
        mx = conv.max()
        # pick the first index attaining the maximum
        idx = int(np.flatnonzero(conv >= mx - 1e-15)[0])
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        j = idx - i_range
        g = np.zeros_like(a)
        g[i_range] = 2.0 * a[j]
        a = _project_simplex(a - 3e-6 * g)

    # ------------------------------------------------------------------
    # Final evaluation & return.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in a.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"[result] final_score = {final_score:.12f}, length = {len(final_seq)}")
    return final_seq
