"""
Propose a candidate step‑function using a multi‑restart Adam optimiser.

The objective that ``evaluate_sequence`` returns is
    2·n·max_k (a*a)[k]   (the sum of the coefficients is forced to 1).
Thus we minimise the peak of the self‑convolution of a non‑negative vector.
The code runs many short Adam optimisations from different random starts
(soft‑max parametrisation guarantees feasibility).  The best vector found
before the time budget expires is returned.

The optimiser uses a smooth approximation of the max (soft‑max with a
temperature that is linearly cooled) and occasional random “jitter’’ moves
to escape plateaux.  All operations are performed with FFT‑based convolutions
so that each iteration costs O(n log n) and fits comfortably into the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via (real) FFT – reused many times
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using a real FFT.
    Length of the result is len(x)+len(y)-1.
    """
    n = x.shape[0]
    m = y.shape[0]
    total = n + m - 1
    # next power of two for efficient FFT
    size = 1 << (total - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:total]
    # Small negative rounding errors can appear; clip them to zero.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max (soft‑max weighting) w.r.t. a
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a weight vector w (soft‑max over the convolution entries)
    compute the gradient of  Σ_k w_k (a*a)[k] w.r.t. a.
    The formula is
        ∇_a = 2 * (w reversed) * a   (convolution)
    """
    n = a.shape[0]
    # reverse the weight vector; convolution with a yields the desired result
    u = w[::-1]
    conv = _full_convolution_fft(u, a)          # length = 3n‑2
    # indices that correspond to the original components of a
    start = 2 * n - 2
    idx = start - np.arange(n)
    g = 2.0 * conv[idx]
    return g

# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises a step‑function of length ``n`` (default 1800) using many
    short Adam runs from random starts.  The best coefficient vector found
    before the time budget expires is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a tiny safety margin (framework may kill a little early)
    deadline = start_time + budget_s - 2.0

    # Problem size – a few thousand entries give a good trade‑off
    # --------------------------------------------------------------
    n = int(kwargs.get("length", 1800))

    # Adam hyper‑parameters (these work well for many restarts)
    beta1, beta2 = 0.90, 0.999
    eps_adam = 1e-8
    base_lr = 0.35          # relatively aggressive initial learning rate

    # bookkeeping for the overall best solution
    best_val = np.inf
    best_a = None

    # ------------------------------------------------------------------
    # Helper: evaluate the current a (sum is already 1)
    # ------------------------------------------------------------------
    def _objective(a_vec: np.ndarray) -> float:
        """Return the raw value 2·n·max(b) for a vector that already sums to 1."""
        conv = _full_convolution_fft(a_vec, a_vec)
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # Main optimisation loop – repeatedly start a short Adam run
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:

        eps = 1e-12

        beta1_pow, beta2_pow = 1.0, 1.0

            # progress of this restart (0 → 1)
            prog = it / max_iters
            tau_start = M * 0.45
            tau_end   = M * 1e-7
            tau = tau_start * (1.0 - prog) + tau_end * prog

            # ---- soft‑max weights over the convolution entries ----
            if tau > 0.0:
                exp_args = (conv - M) / tau
                # shift for numerical stability
                max_arg = exp_args.max()
                exp_vals = np.exp(exp_args - max_arg)
                sum_exp = exp_vals.sum()
                if sum_exp == 0.0 or not np.isfinite(sum_exp):
                    w = np.zeros_like(conv)
                    w[np.argmax(conv)] = 1.0
                else:
                    w = exp_vals / sum_exp
            else:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0

            # ---- gradient w.r.t. a (smoothed sub‑gradient) ----
            grad_a = _gradient_from_weights(a, w)

            # ---- chain rule back to the logits x (soft‑max Jacobian) ----
            dot = np.dot(grad_a, a)                   # <grad_a , a>
            grad_x = a * (grad_a - dot)               # ∇_x (softmax)

            # ---- Adam update for x ----
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            # learning‑rate schedule (linear decay over the restart)
            lr = base_lr * (1.0 - prog) + 1e-5 * prog
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---- retrieve a from the updated logits ----
            max_x = x.max()
            exp_x = np.exp(x - max_x)          # shift for stability
            a = exp_x / exp_x.sum()            # stays on the simplex

            # ---- occasional random jitter (helps escape flat regions) ----
            if np.random.rand() < 0.025:       # ~2.5 % chance per iteration
                i = np.random.randint(n)
                # pick a different index j
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
                # move a random fraction of the mass from i to j
                delta = np.random.rand() * a[i] * 0.5
                a[i] -= delta
                a[j] += delta
                # re‑normalise and restart the Adam moments (a sudden change)
                a = a / a.sum()
                x = np.log(np.maximum(a, eps))
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow, beta2_pow = 1.0, 1.0

        # end of one restart – start the next one if time permits
    # ------------------------------------------------------------------
    # Report the best sequence seen
    # ------------------------------------------------------------------
    if best_a is None:
        # Very unlikely – fall back to a uniform vector
        best_a = np.full(n, 1.0 / n)

    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
