"""
Propose a step‑function (non‑negative heights) that minimises
`evaluate_sequence`.  The search consists of three phases:

1. **Initialisation** – build a uniform or interpolated start vector.
2. **LP refinement** – given the current distribution `a`,
   solve the linear programme
        maximise   Σ g_j
        subject to  (a * g)_k ≤ M  for all k,
                    g_j ≥ 0,
   where `M = max(a * a)`.  The solution `g` has a larger L¹‑norm
   while respecting the same convolution bound.  After normalising `g`
   we mix it with `a` (convex combination with weight ½).  Repeating a
   handful of LP steps monotonically reduces `M`.
3. **Deterministic hill‑climbing** – starting from the best LP result,
   repeatedly move a small amount of mass from the index that most
   contributes to the current maximal convolution entry to a position
   that influences only low‑valued entries.  The convolution is updated
   in O(n) time using the bilinearity of the convolution, and only
   improving moves are accepted.  When the search stagnates a small
   random perturbation of the best solution is performed.

The whole routine respects the 1000 s budget, prints intermediate
bounds and finally returns the best sequence found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a*a – result length 2·len(a)-1."""
    return np.convolve(a, a)   # NumPy's implementation is fast enough for n≤4000


def _lp_max_sum_under_conv(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   subject to   (a * g)_k ≤ M  ∀k , g ≥ 0.
    Returns the optimal vector g (may be None if the solver fails).
    """
    n = a.shape[0]
    # Build sparse matrix A where A[k, j] = a[k-j] if 0 ≤ k-j < n else 0.
    # Use broadcasting to generate indices efficiently.
    I = np.arange(n)[:, None]          # shape (n,1)
    J = np.arange(n)[None, :]          # shape (1,n)
    rows = (I + J).ravel()             # k = i + j
    cols = J.ravel()                   # column = j
    data = a[I.ravel()]                # a[i] repeated for each j
    A = scipy.sparse.csr_matrix((data, (rows, cols)),
                                shape=(2 * n - 1, n))

    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n)                    # maximise Σ g  ↔ minimise −Σ g

    # Use the high‑performance HiGHS interior‑point solver.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=[(0, None)] * n,
                                method='highs',
                                options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective.  Returns a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 2000 gives a good trade‑off between expressiveness
    # and the size of the LP (≈4 M non‑zeros).
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Initialise a – either interpolate a known construction or use uniform.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        a0 = np.full(n, 1.0 / n)

    a0 /= a0.sum()
    best_a = a0.copy()
    best_val = float('inf')
    cur_b = _full_convolution(best_a)
    best_val = 2.0 * n * cur_b.max()
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP refinement (≈30 % of the budget)
    # ------------------------------------------------------------------
    lp_deadline = start_time + 0.30 * budget_s
    a = best_a.copy()
    lp_iter = 0
    while time.time() < lp_deadline and lp_iter < 12:
        M = np.max(_full_convolution(a))
        g = _lp_max_sum_under_conv(a, M)
        if g is None:
            print("[LP] linear programme failed – aborting LP phase")
            break
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        g_scaled = g / sum_g                     # enforce Σ g = 1
        # convex combination with weight t = 0.5 (theoretical guarantee)
        t = 0.5
        a = (1.0 - t) * a + t * g_scaled
        a = np.maximum(a, 0.0)
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LP {lp_iter}] new best = {best_val:.6f}")
        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – deterministic hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    a = best_a.copy()
    b = _full_convolution(a)                     # length 2n-1
    cur_val = 2.0 * n * b.max()
    best_val = cur_val
    best_a = a.copy()

    max_frac = 0.5                      # at most this fraction of source mass per move
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0

    while time.time() < deadline:
        # ---- locate the index that attains the current maximum ----
        max_idx = int(np.argmax(b))

        # ---- compute contributions of each a_i to that max entry ----
        i_vals = np.arange(n)
        j_vals = max_idx - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        contrib = np.zeros(n)
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]

        # source index: the coefficient that contributes most to the peak
        if np.all(contrib == 0):
            i = int(np.argmax(a))          # fallback
        else:
            i = int(np.argmax(contrib))

        if a[i] <= 1e-14:
            continue                        # nothing to move

        # ---- choose a target index that influences only low‑valued entries ----
        # quick heuristic: compare the worst value in the two edge windows
        max_win0 = b[0:n].max()
        max_win_end = b[n - 1:2 * n - 1].max()
        if max_win0 < max_win_end:
            j = 0
        else:
            j = n - 1
        if j == i:
            j = n - 1 if i == 0 else 0

        # ---- amount of mass to transfer ----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- fast incremental update of the convolution (linear in n) ----
        L = 2 * n - 1
        b_new = b.copy()

        # cross‑terms 2·δ·a added / removed by shifting a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a

        # pure‑δ² terms
        idx2j = 2 * j
        if idx2j < L:
            b_new[idx2j] += delta * delta
        idx2i = 2 * i
        if idx2i < L:
            b_new[idx2i] += delta * delta
        idx_ij = i + j
        if idx_ij < L:
            b_new[idx_ij] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ---- accept only strictly improving moves ----
        if new_val < cur_val - 1e-12:
            a[i] -= delta
            a[j] += delta
            # (sum stays exactly 1, no renormalisation required)
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {iteration}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---- occasional random perturbation after long stagnation ----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration}] stagnation – perturbing")

        iteration += 1

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return final_seq
